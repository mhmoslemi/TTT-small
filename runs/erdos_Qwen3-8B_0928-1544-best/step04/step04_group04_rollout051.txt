import numpy as np
from scipy.optimize import minimize, Bounds
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n = 70
    dx = 2.0 / n
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n)
        h[::2] = 1.0
    h = h.astype(float)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    bounds = Bounds([0.0]*n, [1.0]*n)
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros_like(h_clip)
        scale = (n / 2) / sum_h
        return h_clip * scale
    
    best_h = h
    best_c5 = objective(h)
    
    for _ in range(15):
        perturbation = np.random.normal(0, 0.1, n)
        perturbed_h = project_h(best_h + perturbation)
        result = minimize(
            fun=objective,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            tol=1e-8,
            options={'maxiter': 150, 'ftol': 1e-9}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    final_result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8,
        options={'maxiter': 150, 'ftol': 1e-9}
    )
    
    if final_result.success and final_result.fun < best_c5:
        best_c5 = final_result.fun
        best_h = final_result.x
    
    return best_h, best_c5, n
