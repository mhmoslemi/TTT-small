import numpy as np
from scipy.optimize import minimize
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(70)
        initial_h = (initial_h / np.sum(initial_h) * 35.0)
    np.random.seed(seed)
    n = 70
    dx = 2.0 / n
    pop_size = 50
    mut_rate = 0.05
    cross_rate = 0.7
    tour_size = 3

    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale

    def fitness(ind):
        corr = np.correlate(ind, 1 - ind, mode='full')
        return np.max(corr) * dx

    def tournament_selection(pop, fit):
        indices = np.random.choice(range(pop_size), tour_size, replace=False)
        return pop[np.argmin(fit[indices])]

    pop = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n)
        h = (h / np.sum(h) * 35.0)
        pop.append(h)

    best_h = initial_h.copy()
    best_c5 = fitness(best_h)
    start_time = time.time()

    for _ in range(100):
        fit = [fitness(ind) for ind in pop]
        best_idx = np.argmin(fit)
        if fit[best_idx] < best_c5:
            best_c5 = fit[best_idx]
            best_h = pop[best_idx].copy()
        parents = [tournament_selection(pop, fit) for _ in range(pop_size)]
        new_pop = []
        for i in range(pop_size):
            p1, p2 = parents[i % pop_size], parents[(i+1) % pop_size]
            if np.random.rand() < cross_rate:
                child = np.random.rand(n)
                for j in range(n):
                    child[j] = p1[j] * np.random.rand() + p2[j] * (1 - np.random.rand())
            else:
                child = p1.copy()
            if np.random.rand() < mut_rate:
                noise = np.random.normal(0, 0.05, n)
                child += noise
            child = project_h(child)
            new_pop.append(child)
        pop = new_pop
        if time.time() - start_time > budget_s:
            break

    top = sorted(pop, key=fitness)[:5]
    for cand in top:
        res = minimize(lambda x: fitness(x), cand, method='SLSQP',
                       bounds=[(0, 1)]*n, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}],
                       options={'maxiter': 150, 'ftol': 1e-9})
        if res.success:
            if res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x
    return best_h, best_c5, n
