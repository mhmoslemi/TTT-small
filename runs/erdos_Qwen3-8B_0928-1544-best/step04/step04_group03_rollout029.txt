import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = initial_h_values.copy()

        idx = np.random.choice(70, 35, replace=False)

        cand[idx] = 1.0
        population.append(cand)
    population = np.array([np.array(p) for p in population], dtype=object)
    best_h = population[0]
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / 70.0)
    start_time = time.time()
    for gen in range(100):
        c5s = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5s.append(np.max(corr) * (2.0 / 70.0))
        order = np.argsort(c5s)
        parents = population[order[:int(len(population)*0.2)]]
        new_pop = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1]
            child = (p1 + p2) / 2
            s = np.sum(child)
            child = child * 35.0 / s
            noise = np.random.normal(0, 0.05, 70)
            child = np.clip(child + noise, 0, 1)
            s = np.sum(child)
            child = child * 35.0 / s
            new_pop.append(child)
        combined = np.vstack([population, new_pop])

        combined_order = np.argsort(c5s)
        population = combined[combined_order[:50]]
        if time.time() - start_time > budget_s - 500:
            break
        current = population[0]
        curr_c5 = np.max(np.correlate(current, 1 - current, mode='full')) * (2.0 / 70.0)
        if curr_c5 < best_c5:
            best_h = current
            best_c5 = curr_c5
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * (2.0 / 70.0)
    bounds = Bounds([0.0]*70, [1.0]*70)
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - 35.0}
    result = minimize(objective, best_h, bounds=bounds, constraints=cons, method='SLSQP', options={'maxiter': 200})
