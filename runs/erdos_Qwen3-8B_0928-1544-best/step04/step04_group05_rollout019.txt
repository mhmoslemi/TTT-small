import numpy as np

def run(seed=42, budget_s=1100, **kwargs):

        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h
        population.append(h)

        corr = np.correlate(h, 1 - h, mode='full')

    def crossover(p1, p2):
        split = np.random.randint(1, n_points)
        return np.concatenate([p1[:split], p2[split:]]), np.concatenate([p2[:split], p1[split:]])
    
    def mutate(h):
        h_new = h.copy()
        for i in range(n_points):

        h_new = h_new * (n_points / 2) / sum_h

    best_c5 = float('inf')

    start_time = time.time()

        if time.time() - start_time > budget_s:
