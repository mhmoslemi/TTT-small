import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h = initial_h_values.copy()
    else:
        h = np.full(70, 0.5)
    n = 70
    dx = 2.0 / n
    
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale
    
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best = project_h(h)
    best_c5 = obj(best)
    
    for _ in range(100):
        pop = np.random.normal(best, 0.01, (n, n))
        for i in range(n):
            pop[i] = project_h(pop[i])
        vals = [obj(pop[i]) for i in range(n)]
        idx = np.argmin(vals)
        best = pop[idx]
        best_c5 = vals[idx]
    
    return project_h(best), best_c5, n
