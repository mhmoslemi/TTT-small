import numpy as np
from scipy.optimize import minimize, Bounds
import random

def run(seed=42, budget_s=1100, **kwargs):

        initial_h = np.full(70, 0.5)
    
    dx = 2.0 / 70
    n = 70
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale
    
    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = np.copy(initial_h)
    current_h = project_h(initial_h)
    current_c5 = objective(current_h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h
    
    for _ in range(150):
        perturbation = np.random.normal(0, 0.05, size=n)
        perturbed_h = current_h + perturbation
        perturbed_h = project_h(perturbed_h)
        
        result = minimize(
            fun=objective,
            x0=perturbed_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            tol=1e-8,
            options={'maxiter': 250, 'ftol': 1e-9, 'eps': 1e-8}
        )

            new_c5 = result.fun

                best_c5 = new_c5
                best_h = result.x

    result_final = minimize(

    )
