import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    required_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def project(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.zeros_like(h)
        return h_clipped * (required_sum / sum_clipped)
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    population = [initial_h.copy() for _ in range(50)]
    scores = np.array([evaluate_c5(h) for h in population])
    
    time_used = 0
    max_iter = int(budget_s // 5)
    temp = 1.0
    cool_rate = 0.995
    sigma = 0.2
    
    for _ in range(max_iter):
        sorted_indices = np.argsort(scores)
        top_candidates = sorted_indices[:int(len(population) * 0.25)]
        
        new_population = []
        for idx in top_candidates:
            h = population[idx].copy()
            noise = np.random.normal(0, sigma, n_points)
            h_perturbed = h + noise
            h_perturbed = project(h_perturbed)
            
            result = minimize(
                fun=evaluate_c5,
                x0=h_perturbed,
                method='Nelder-Mead',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                options={'maxiter': 100, 'xatol': 1e-6}
            )
            new_candidate = project(result.x)
            new_population.append(new_candidate)
        
        population = new_population + [population[i] for i in sorted_indices[int(len(population) * 0.25):]]
        scores = np.array([evaluate_c5(h) for h in population])
        
        temp *= cool_rate
        sigma *= cool_rate
        
        time_used += 5
        if time_used >= budget_s:
            break
    
    best_idx = np.argmin(scores)
    best_h = population[best_idx]
    best_c5 = scores[best_idx]
    return best_h, best_c5, n_points
