import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    resolutions = [14, 28, 56, 70]
    best_h = None
    best_c5 = float('inf')
    time_used = 0
    max_time_per_level = budget_s // len(resolutions)

    # Handle initial_h_values for downsampled starting point
    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = initial_h_values[::5].astype(np.float64)
    else:
        initial_h = np.full(resolutions[0], 0.5).astype(np.float64)

    current_h = initial_h.copy()
    n_points = len(current_h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    for i, target_n in enumerate(resolutions):
        if i == 0:
            h_prev = current_h
        else:
            upsample_factor = target_n // len(current_h)
            h_prev = np.repeat(current_h, upsample_factor).astype(np.float64)
            n_points = target_n

        x0 = h_prev.copy()
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        result = minimize(
            fun=compute_c5,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )

        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()

        current_h = result.x.copy()
        current_h = current_h.astype(np.float64)
        n_points = target_n
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        time_used += max_time_per_level
        if time_used >= budget_s:
            break

    return best_h, best_c5, len(best_h)
