import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n = 70
    dx = 2.0 / n
    pop_size = 50
    generations = 100
    tournament_size = 5
    mutation_rate = 0.1
    mutation_step = 0.05

    # Initialize population
    pop = []
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        pop.append(initial_h_values.copy())
    for _ in range(pop_size - len(pop)):
        h = np.random.rand(n)
        h_sum = np.sum(h)
        h = h * (n / 2) / h_sum
        pop.append(h)

    best_h = pop[0]
    best_c5 = evaluate_erdos_solution(pop[0])
    start_time = time.time()

    for gen in range(generations):
        # Evaluate all
        fitness = [evaluate_erdos_solution(h) for h in pop]
        # Update best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = pop[best_idx]
        # Tournament selection
        parents = []
        for _ in range(pop_size // 2):
            tour = np.random.choice(pop_size, tournament_size, replace=False)
            winner = tour[np.argmin([fitness[i] for i in tour])]
            parents.append(pop[winner])
        # Create offspring
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 >= len(parents):
                offspring.append(parents[i])
                continue
            p1, p2 = parents[i], parents[i+1]
            mid = np.random.randint(1, n)
            c1 = np.concatenate([p1[:mid], p2[mid:]])
            c2 = np.concatenate([p2[:mid], p1[mid:]])
            # Mutate c1
            if np.random.rand() < mutation_rate:
                mut_idx = np.random.choice(n, int(mutation_rate * n), replace=False)
                c1[mut_idx] += np.random.normal(0, mutation_step, size=len(mut_idx))
                c1 = np.clip(c1, 0, 1)
                c1_sum = np.sum(c1)
                if c1_sum > 0:
                    c1 *= (n / 2) / c1_sum
            # Mutate c2
            if np.random.rand() < mutation_rate:
                mut_idx = np.random.choice(n, int(mutation_rate * n), replace=False)
                c2[mut_idx] += np.random.normal(0, mutation_step, size=len(mut_idx))
                c2 = np.clip(c2, 0, 1)
                c2_sum = np.sum(c2)
                if c2_sum > 0:
                    c2 *= (n / 2) / c2_sum
            offspring.append(c1)
            offspring.append(c2)
        # Combine and select
        combined = pop + offspring
        fit = [evaluate_erdos_solution(h) for h in combined]
        sort_idx = np.argsort(fit)
        pop = [combined[i] for i in sort_idx[:pop_size]]
        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n
