        initial_h = initial_h_values
    else:
        initial_h = np.full(n_points_list[0], 0.5)
    
    current_h = initial_h.copy()
    dx_list = [2.0 / n for n in n_points_list]
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx_list[len(h)-1]
    
    def constraint_sum(h):
        return np.sum(h) - (len(h) / 2)
    
    bounds = [(0.0, 1.0)] * len(current_h)
    best_h = current_h.copy()
    best_c5 = objective(current_h)
    
    for i, n_points in enumerate(n_points_list):
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
        
        if i == 0:
            current_h = initial_h.copy()
        else:
            previous_len = n_points_list[i-1]
            if n_points == 2 * previous_len:
                current_h = np.repeat(current_h, 2)
            elif previous_len == 56 and n_points == 70:
                current_h = np.concatenate([current_h, np.full(14, 0.5)])
        
        # Optimize
        bounds = [(0.0, 1.0)] * len(current_h)
        result = minimize(
            fun=objective,
            x0=current_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
        current_h = result.x
    
    return best_h, best_c5, n_points_list[-1]
