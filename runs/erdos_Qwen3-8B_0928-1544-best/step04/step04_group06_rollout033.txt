        initial_h = project_h(initial_h)
    
    pop_size = 100
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n)
        h = project_h(h)
        population.append(h)
    
    best_h = initial_h.copy()
    best_c5 = fitness(best_h)
    
    crossover_rate = 0.25
    mutation_rate = 0.05
    mutation_std = 0.03
    
    for _ in range(300):
        new_population = []
        for _ in range(pop_size):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            
            if np.random.rand() < crossover_rate:
                point = np.random.randint(1, n)
                child = np.concatenate([parent1[:point], parent2[point:]])
            else:
                child = parent1.copy()
            
            perturbation = np.random.normal(0, mutation_std, size=n)
            child += perturbation * (np.random.rand(n) < mutation_rate)
            
            child = project_h(child)
            
            current_c5 = fitness(child)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = child.copy()
            
            new_population.append(child)
        
        population = new_population
    
    return best_h, best_c5, n
