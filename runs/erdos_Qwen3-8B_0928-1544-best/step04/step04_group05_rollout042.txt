import numpy as np

    dx = 2.0 / n_points

        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n_points)
        scale = (n_points / 2) / sum_h
        return h_clip * scale
    
    pop_size = 50
    population = []
    if 'initial_h_values' in globals():
        population.append(initial_h)
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = initial_h if 'initial_h_values' in globals() else initial_h
    
    start_time = time.time()
    
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        
        fitness = []

            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        parents = []
        for _ in range(pop_size):
