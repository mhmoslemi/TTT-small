    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.zeros(n_points)
        h_initial[::2] = 1.0
    # Ensure sum is exactly 35
    if np.sum(h_initial) != 35:
        diff = 35 - np.sum(h_initial)
        zeros = np.where(h_initial == 0)[0]
        h_initial[zeros[:diff]] = 1.0
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    current_h = h_initial.copy()
    best_h = current_h.copy()
    best_c5 = compute_c5(best_h)
    T = 1.0
    alpha = 0.999
    max_steps = 1000
    t0 = time.time()
    for step in range(max_steps):
        if time.time() - t0 > budget_s - 500:
            break
        noise = np.random.normal(0, T, n_points)
        h_perturbed = current_h + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        if sum_perturbed > 1e-8:
            h_perturbed = h_perturbed * (35.0 / sum_perturbed)
        else:
            h_perturbed = np.zeros(n_points)
        new_c5 = compute_c5(h_perturbed)
        if new_c5 < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = new_c5
        else:
            prob = np.exp((best_c5 - new_c5) / T)
            if np.random.rand() < prob:
                current_h = h_perturbed.copy()
        T *= alpha
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    def constraint_sum(h):
        return np.sum(h) - 35.0
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
    )
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    return best_h, best_c5, n_points
