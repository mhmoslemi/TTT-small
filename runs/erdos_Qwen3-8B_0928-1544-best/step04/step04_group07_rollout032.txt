import numpy as np
from scipy.optimize import minimize, Bounds
import random

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(70, 0.5)
    
    np.random.seed(seed)
    n = len(initial_h)
    dx = 2.0 / n

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    current_h = initial_h.copy()
    temp = 0.1
    cooling_rate = 0.995
    steps = 150
    
    for _ in range(steps):
        perturb = np.random.normal(0, temp, n)
        perturbed = current_h + perturb
        perturbed = project_h(perturbed)
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            tol=1e-8,
            options={'maxiter': 150, 'ftol': 1e-9}
        )
        if result.success:
            new_c5 = result.fun
            if new_c5 < best_c5:

        temp *= cooling_rate
    
    result_final = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8,
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    
    if result_final.success and result_final.fun < best_c5:
        best_c5 = result_final.fun
        best_h = result_final.x
    
    return best_h, best_c5, n
