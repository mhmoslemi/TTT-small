    n_points = 70
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    max_iterations = 1000
    alpha = 0.5
    start_time = time.time()
    learning_rate_decay = 0.999
    no_improvement = 0
    improvement_threshold = 1e-6

    for _ in range(max_iterations):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        k_max = np.argmax(corr)
        
        grad = np.zeros(n_points)
        for j in range(n_points):
            j_plus_k = j + k_max
            term1 = (1 - h[j_plus_k]) * dx if j_plus_k < n_points else 0.0
            j_minus_k = j - k_max
            term2 = h[j_minus_k] * dx if j_minus_k >= 0 else 0.0
            grad[j] = term1 - term2
        
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        sum_h_new = np.sum(h_new)
        scale_factor = (n_points / 2) / sum_h_new if sum_h_new != 0 else 1.0
        h_new = h_new * scale_factor
        
        h = h_new
        current_c5 = max_corr * dx if max_corr > 0 else 0.0
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > 50:
                break
        
        if time.time() - start_time > budget_s:
            break
        
        alpha *= learning_rate_decay
    
    return best_h, best_c5, n_points
