import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=np.float64)
    else:
        initial_h = np.zeros(n_points, dtype=np.float64)
        indices = np.random.choice(n_points, size=ones_count, replace=False)
        initial_h[indices] = 1.0
    
    mu = 100
    pop_size = mu
    population = [initial_h.copy()]
    for _ in range(mu - 1):
        idxs = np.random.choice(n_points, size=ones_count, replace=False)
        cand = np.zeros(n_points, dtype=np.float64)
        cand[idxs] = 1.0
        population.append(cand)

    best_h = initial_h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    t_remaining = budget_s

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    while t_remaining > 0.1:
        t_remaining = budget_s - (time.time() - start_time)

        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best.copy()

        sorted_idx = np.argsort(fitness)
        parents = [current_pop[i] for i in sorted_idx[:mu // 2]]

            child = np.zeros(n_points, dtype=np.float64)

                if np.random.rand() < 0.5:

                    child[i] = p2[i]
            m_rate = 0.05
            mutate = np.random.choice(n_points, size=int(m_rate * n_points), replace=False)
            child[mutate] = 1.0 - child[mutate]

                else:
                    zeros_pos = np.where(child == 0.0)[0]
                    flip = np.random.choice(zeros_pos, size=int(-diff), replace=False)
                    child[flip] = 1.0

            new_pop.append(child)

        population = parents + new_pop
        time.sleep(0.01)

    return best_h, best_c5, n_points
