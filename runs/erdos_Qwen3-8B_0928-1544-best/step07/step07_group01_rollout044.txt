        initial_h = np.random.choice([0,1], n_points, p=[0.5, 0.5])
        cnt = np.sum(initial_h)
        diff = ones_needed - cnt
        if diff > 0:
            nz = np.where(initial_h == 0)[0]
            if len(nz) >= diff:
                initial_h[np.random.choice(nz, diff, replace=False)] = 1
        elif diff < 0:
            no = np.where(initial_h == 1)[0]
            if len(no) >= -diff:
                initial_h[np.random.choice(no, -diff, replace=False)] = 0

    pop_size = 50
    population = [initial_h.copy() for _ in range(pop_size)]
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * (2.0 / n_points)

    def evaluate(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)

    def tournament_selection(pop, size=5):
        return min(random.sample(pop, size), key=evaluate)

    def crossover(a, b):
        idx = np.random.randint(1, n_points - 1)
        return np.concatenate([a[:idx], b[idx:]]), np.concatenate([b[:idx], a[idx:]])

    def mutate(h, rate=0.02):
        h = h.copy()
        for _ in range(10):
            if np.random.rand() < rate:
                i = np.random.randint(n_points)
                h[i] ^= 1
                if np.sum(h) != ones_needed:
                    if h[i] == 1:
                        ones = np.where(h == 1)[0]
                        if ones.size:
                            h[np.random.choice(ones)] ^= 1
                    else:
                        zeros = np.where(h == 0)[0]
                        if zeros.size:
                            h[np.random.choice(zeros)] ^= 1
        return h

    start = time.time()
    for gen in range(200):
        if time.time() - start > budget_s:
            break
        new_pop = []
        for _ in range(pop_size // 2):
            p1 = tournament_selection(population)
            p2 = tournament_selection(population)
            c1, c2 = crossover(p1, p2)
            new_pop.append(mutate(c1))
            new_pop.append(mutate(c2))
        population = new_pop
        for h in population:
            curr = evaluate(h)
            if curr < best_c5:
                best_c5 = curr
                best_h = h.copy()
    return best_h, best_c5, n_points
