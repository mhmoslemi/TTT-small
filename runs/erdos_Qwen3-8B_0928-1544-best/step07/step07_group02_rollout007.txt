            res = minimize_scalar(objective, bounds=(0.0, 1.0), method='bounded')
            if res.success:
                h[i] = res.x
        current_sum = np.sum(h)
        delta = target_sum - current_sum
        h[-1] += delta
        h[-1] = np.clip(h[-1], 0.0, 1.0)
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if time.time() - start_time > budget_s:
            break
    return best_h, best_c5, n_points
