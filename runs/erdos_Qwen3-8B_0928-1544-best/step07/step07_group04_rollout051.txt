import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # Initialize population
    pop_size = 20
    population = [initial_h.copy() for _ in range(pop_size)]
    for _ in range(pop_size):
        perturb = np.random.normal(0, 0.1, n_points)
        population[_] += perturb
        population[_] = np.clip(population[_], 0, 1)
        population[_] *= (ones_count / np.sum(population[_]))
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Evaluate and sort population
        fitness = [compute_c5(ind) for ind in population]
        ranked = sorted(zip(population, fitness), key=lambda x: x[1])
        
        # Select top 20% parents
        top_parents = [ind for ind, fit in ranked[:int(0.2 * pop_size)]]
        new_population = [ranked[0][0]]  # Elitism
        
        # Generate new individuals
        while len(new_population) < pop_size:

            alpha = np.random.uniform(0.3, 0.7)
            child = alpha * p1 + (1 - alpha) * p2
            mutation = np.random.normal(0, 0.05, n_points)
            child += mutation
            child = np.clip(child, 0, 1)
            child = child * (ones_count / np.sum(child))
            new_population.append(child)
        
        population = new_population
        
        # Update best
