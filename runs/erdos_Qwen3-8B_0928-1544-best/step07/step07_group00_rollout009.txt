        new_pop = elites.copy()
        while len(new_pop) < population_size:
            p1, p2 = np.random.choice(elite_size, 2, replace=False)
            parent1 = population[sorted_idx[p1]]
            parent2 = population[sorted_idx[p2]]
            
            child = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            child = normalize(child, n_ones)
            
            if np.random.rand() < 0.1:
                idx = np.random.randint(n_points)
                delta = np.random.normal(0, 0.05)
                child[idx] += delta
                child = normalize(child, n_ones)
            
            new_pop.append(child)
        population = new_pop
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    final_candidates = [population[i] for i in np.argsort([evaluate_erdos_solution(h) for h in population])[:5]]
    
    for h in final_candidates:
        def obj(x):
            return evaluate_erdos_solution(x)
        
        res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_ones}])
        if res.success:
            curr = res.fun
            if curr < best_c5:
                best_c5 = curr
                best_h = res.x
    
    return best_h, best_c5, n_points
