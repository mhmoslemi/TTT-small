        seed=seed,
        polish=True,
        disp=False
    )

    best_h = np.clip(result_de.x, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    # Local refinement with COBYLA
    result_local = minimize(
        fun=objective,
        x0=best_h,
        method='COBYLA',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': lambda x: np.sum(np.clip(x, 0, 1)) - target_sum},
        options={'maxiter': 1000, 'ftol': 1e-9, 'eps': 1e-8}
    )
    best_h = result_local.x
    best_c5 = result_local.fun

    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Additional random perturbation-based refinement
        for _ in range(10):
            perturb = np.random.normal(0, 0.1, n_points)
            perturb = np.clip(perturb, -0.5, 0.5)
            candidate = best_h + perturb
            candidate = np.clip(candidate, 0, 1)
            candidate = candidate * target_sum / np.sum(candidate)
            current_c5 = objective(candidate)
            if current_c5 < best_c5:
                best_h = candidate
                best_c5 = current_c5

    return best_h, best_c5, n_points
