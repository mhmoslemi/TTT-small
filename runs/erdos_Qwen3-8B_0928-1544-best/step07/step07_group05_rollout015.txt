from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2

    def objective(h):

        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    # Multi-start optimization with adaptive perturbation
    initial_h = np.clip(initial_h, 0.0, 1.0)
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    
    for _ in range(20):  # 20 initial trials

        h_perturbed = initial_h + perturb
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
