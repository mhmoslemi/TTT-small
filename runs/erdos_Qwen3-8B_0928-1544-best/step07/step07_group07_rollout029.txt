import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()

        initial_h = np.full(n_points, 0.5)
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    T = 0.8
    alpha = 0.995
    max_iter = 10000
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        
        perturb_indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
        noise = np.random.normal(0, T, len(perturb_indices))
        candidate = initial_h.copy()
        candidate[perturb_indices] += noise
        
        candidate = np.clip(candidate, 0, 1)
        candidate *= (ones_count / np.sum(candidate))
        
        current_c5 = evaluate_erdos_solution(initial_h)

        if candidate_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - candidate_c5) / T):

        if candidate_c5 < best_c5:
            best_h = candidate.copy()
            best_c5 = candidate_c5
        
        T *= alpha
