import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:

    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()

    start_time = time.time()
    
    # Initial optimizer run
    result = minimize(
        fun=objective,
        x0=initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    if result.success:
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5

    # Iterative refinement

        result = minimize(
            fun=objective,
            x0=h_perturbed,

            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
