import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70  # 70 samples for efficiency
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for it in range(200):  # 200 iterations for thorough exploration
        # Perturb current best solution
        sigma = 0.15 * (1 - (it / 200))
        h_perturbed = best_h + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        
        # Minimize objective while satisfying the normalization constraint
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={
                'maxiter': 100,
                'ftol': 1e-9,
                'eps': 1e-8,
                'disp': False
            }
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                # Early exit if no improvement for 50 iterations
                if it > 50 and abs(current_c5 - best_c5) < 1e-5:
                    break
    
    # Ensure time budget is respected
    if time.time() - start_time > budget_s - 10:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
