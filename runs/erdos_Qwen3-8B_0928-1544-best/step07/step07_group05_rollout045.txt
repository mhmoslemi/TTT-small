import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)

        return np.max(corr) * (2.0 / n_points)
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    current_h = initial_h.copy()
    current_c5 = best_c5
    
    T = 1.0
    alpha = 0.95
    steps = 0
    start_time = time.time()
    perturb_frac = 0.1
    patience_counter = 0
    
    while time.time() - start_time < budget_s:
        perturb_indices = np.random.choice(n_points, int(perturb_frac * n_points), replace=False)
        new_h = current_h.copy()
        for i in perturb_indices:
            new_h[i] = 1.0 - new_h[i]
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h_sum = np.sum(new_h)
        if new_h_sum == 0:
            new_h = np.zeros(n_points)
        else:
            new_h = new_h * (ones_count / new_h_sum)
        
        new_c5 = compute_c5(new_h)
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = new_h.copy()
            current_c5 = new_c5

        T *= alpha
        steps += 1
