    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()

        initial_h = np.full(n_points, 0.5)
    
    bounds = [(0.0, 1.0)] * n_points

        penalty = 1e4 * (np.sum(h) - ones_count) ** 2
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5 + penalty
    
    start_time = time.time()
    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=int(budget_s * 0.5),

        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-7,
        seed=seed,
        polish=True,

    )
    
    best_h = result.x
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * (2.0 / n_points)
    
    return best_h, best_c5, n_points
