        return corr
    
    def get_max_lag(corr):
        return np.argmax(corr)
    
    def get_overlap_sum(h, lag):
        if lag >= 0:
            return np.dot(h[:n_points - lag], h[lag:])
        else:
            return np.dot(h[:n_points + lag], h[lag:])
    
    def optimize_for_lag(h, lag):
        def obj(x):
            return -get_overlap_sum(x, lag)
        
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                {'type': 'ineq', 'fun': lambda x: x},
                {'type': 'ineq', 'fun': lambda x: np.ones_like(x) - x})
        
        res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=cons,
                       options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8})
        return res.x
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        corr = compute_corr(best_h)
        k_max = get_max_lag(corr)
        new_h = optimize_for_lag(best_h, k_max)
        new_c5 = corr[k_max] * (2.0 / n_points)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
        else:
            break
    
    return best_h, best_c5, n_points
