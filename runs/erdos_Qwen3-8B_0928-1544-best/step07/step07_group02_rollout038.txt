import numpy as np
from scipy.optimize import minimize_scalar
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    h = np.zeros(n_points)
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():

        h = h * (target_sum / np.sum(h))
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    max_passes = 5  # Number of optimization passes
    for pass_num in range(max_passes):
        # Coordinate-wise optimization
        for i in range(n_points):
            # Define objective for scalar optimization
            def obj(h_i):
                h_new = h.copy()
                h_new[i] = h_i
                corr = np.correlate(h_new, 1 - h_new, mode='full')
                return np.max(corr) * (2.0 / n_points)

            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-6})

        if prob.status == 'OPTIMAL':

            best_h = h.copy()
        # Early termination
        if time.time() - start_time > budget_s:
