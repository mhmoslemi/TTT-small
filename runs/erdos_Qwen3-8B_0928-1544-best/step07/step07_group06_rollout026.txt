from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70

        h = np.random.rand(n_points)
        h = h * ones_count / np.sum(h)
        harmony_memory.append(h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()

    start_time = time.time()
