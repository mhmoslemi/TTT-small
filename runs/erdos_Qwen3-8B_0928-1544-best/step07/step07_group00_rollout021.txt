import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        parent_h = initial_h_values.copy()
        parent_h /= np.sum(parent_h) * (n_points / 2)
    else:
        parent_h = np.random.rand(n_points)
        parent_h /= np.sum(parent_h) * (n_points / 2)

    swarm_size = 100
    particles = np.random.rand(swarm_size, n_points)
    particles /= np.sum(particles, axis=1, keepdims=True) * (n_points / 2)
    velocities = np.random.uniform(-0.1, 0.1, (swarm_size, n_points))
    best_positions = particles.copy()
    best_costs = np.full(swarm_size, np.inf)
    global_best = parent_h.copy()
    global_cost = evaluate_erdos_solution(global_best)

    start_time = time.time()

    while time.time() - start_time < budget_s:
        w = max(0.4, 0.9 - (time.time() - start_time) / budget_s * 0.5)
        costs = np.array([evaluate_erdos_solution(p) for p in particles])
        mask = costs < best_costs
        best_positions[mask] = particles[mask]
        best_costs[mask] = costs[mask]
        if np.min(costs) < global_cost:
            global_cost = np.min(costs)
            global_best = particles[np.argmin(costs)]
        for i in range(swarm_size):
            rand1 = np.random.rand(n_points)
            rand2 = np.random.rand(n_points)
            velocities[i] = w * velocities[i] + 2.0 * rand1 * (best_positions[i] - particles[i]) + 2.0 * rand2 * (global_best - particles[i])
            particles[i] = particles[i] + velocities[i]
        particles = np.clip(particles, 0, 1)
        for i in range(swarm_size):
            particles[i] = particles[i] / np.sum(particles[i]) * (n_points / 2)

    final_h = np.clip(global_best, 0, 1)
    final_h = final_h / np.sum(final_h) * (n_points / 2)
    final_c5 = evaluate_erdos_solution(final_h)
    return final_h, final_c5, n_points
