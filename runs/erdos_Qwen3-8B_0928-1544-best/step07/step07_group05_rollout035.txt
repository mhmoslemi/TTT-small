import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    population_size = 50
    population = []
    for _ in range(population_size):
        if _ == 0:
            h = initial_h.copy()
        else:
            h = initial_h.copy() + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum) if h_sum else np.full(n_points, 0.5)
        population.append(h)
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        
        fitness = np.array([compute_c5(h) for h in population])
        sorted_idx = np.argsort(fitness)
        
        new_pop = []
        for _ in range(population_size):
            a, b = random.sample(sorted_idx[:20], 2)
            parent1, parent2 = population[a], population[b]
            split = np.random.randint(1, n_points)
            child = np.concatenate([parent1[:split], parent2[split:]])
            if np.random.random() < 0.1:
                perturb = np.random.normal(0, 0.1, n_points)
                child += perturb
            child = np.clip(child, 0.0, 1.0)
            child_sum = np.sum(child)
            child = child * (ones_count / child_sum) if child_sum else np.full(n_points, 0.5)
            new_pop.append(child)
        
        population = new_pop[:]
        sorted_idx = np.argsort([compute_c5(h) for h in population])
        population[:int(0.1*population_size)] = [population[i] for i in sorted_idx[:int(0.1*population_size)]]
    
    fitness = [compute_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    return population[best_idx], fitness[best_idx], n_points
