import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    h = h * (ones_count / np.sum(h))
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    
    start_time = time.time()
    step = 0
    temp = 1.0
    decay_rate = 0.995
    
    while time.time() - start_time < budget_s:
        step += 1
        # Perturb with noise scaled by temperature
        noise = np.random.normal(0, temp, n_points)
        h_perturbed = h + noise
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        # Evaluate new c5
        corr = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
        new_c5 = np.max(corr) * (2.0 / n_points)
        
        # Accept with probability
        if new_c5 < best_c5:
            h = h_perturbed.copy()
            best_c5 = new_c5
            best_h = h.copy()
        else:
            delta = new_c5 - best_c5
            prob = np.exp(-delta / temp)
            if np.random.rand() < prob:
                h = h_perturbed.copy()
                best_c5 = new_c5
                best_h = h.copy()
        
        # Local optimization every 50 steps
        if step % 50 == 0:
            def objective(x):
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * (2.0 / n_points)
            res = minimize(
                fun=objective,
                x0=h,
                method='SLSQP',
                bounds=[(0.0, 1.0)] * n_points,
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count}
            )
            if res.success:
                h = res.x
                best_c5 = res.fun
                best_h = h.copy()
        
        # Cool down
        temp *= decay_rate
    
    return best_h, best_c5, n_points
