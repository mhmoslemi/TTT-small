    selected = int(len(population) * 0.2)
    start_time = time.time()
    best_h = initial_h_bin
    best_c5 = evaluate_c5(best_h)
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness_values = [fitness(h) for h in population]
        sorted_idx = np.argsort(fitness_values)
        parents = [population[i] for i in sorted_idx[:selected]]
        
        new_population = []
        while len(new_population) < len(population):
            p1, p2 = np.random.choice(parents, size=2)
            child = p1.copy()
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = p2[i]
            if np.random.rand() < 0.1:
                idx1, idx2 = np.random.randint(n_points, size=2)
                if child[idx1] != child[idx2]:
                    child[idx1], child[idx2] = child[idx2], child[idx1]
            new_population.append(child)
        
        population = new_population
        for h in population:
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    candidates = [best_h]
    for h in population:
        if evaluate_c5(h) < best_c5:
            candidates.append(h)
    candidates = candidates[:5]
    
    for h in candidates:
        def obj(x):
            return evaluate_c5(x)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, h, bounds=[(0, 1)]*n_points, constraints=cons, method='SLSQP', options={'maxiter': 100})
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
    
    return best_h, best_c5, n_points
