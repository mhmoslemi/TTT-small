        selected_pop = [population[i] for i in selected]
        
        new_pop = []
        while len(new_pop) < population_size:
            parent1, parent2 = np.random.choice(selected_pop, 2, replace=False)
            child = (parent1 + parent2) / 2
            
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                sum_child = np.sum(child)
                if sum_child > 0:
                    child = child * (ones_count / sum_child)
                else:
                    child = np.zeros_like(child)
                    child[0] = 1.0
            
            new_pop.append(child)
        
        population = selected_pop + new_pop
        population_fitness = [-evaluate(h) for h in population]
        best_fitness = max(population_fitness)
        best_h = population[population_fitness.index(best_fitness)]
    
    # Convex refinement
    top_candidates = np.argsort(population_fitness)[-5:]
    candidates = [population[i] for i in top_candidates]
    
    best_c5 = float('inf')
    best_h_refined = best_h
    
    for h_candidate in candidates:
        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * (2.0 / n_points)
        
        result = minimize(
            fun=objective,
            x0=h_candidate,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n_points,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h_refined = result.x
    
    return best_h_refined, best_c5, n_points
