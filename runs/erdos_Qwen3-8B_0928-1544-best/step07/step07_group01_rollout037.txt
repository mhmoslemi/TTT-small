import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    initial_h = initial_h * (ones_count / np.sum(initial_h))

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points

    def optimize(start_h, max_iter=200, tolerance=1e-10):
        result = minimize(fun=objective, x0=start_h, method='SLSQP',
                          bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum},
                          options={'maxiter': max_iter, 'ftol': tolerance, 'eps': 1e-8})
        return result.x if result.success else start_h

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for _ in range(100):
        perturb = np.random.normal(0, 0.1, n_points)
        perturbed_h = initial_h + perturb
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (ones_count / np.sum(perturbed_h))
        current_h = optimize(perturbed_h)
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        elapsed = time.time() - start_time
        if elapsed >= budget_s:

    return best_h, best_c5, n_points
