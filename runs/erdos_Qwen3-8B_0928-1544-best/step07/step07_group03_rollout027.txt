            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (particles[i] - best_h) + social * r2 * (particles[i] - best_h)
            particles[i] += velocities[i]
            
            # Project to feasible region
            sum_h = np.sum(particles[i])
            if sum_h != 0:
                scaled = particles[i] * (target_sum / sum_h)
                scaled = np.clip(scaled, 0, 1)
                particles[i] = scaled
            
            new_cost = calculate_cost(particles[i])
            if new_cost < personal_best[i]:
                personal_best[i] = new_cost
                if new_cost < best_cost:
                    best_cost = new_cost
                    best_h = particles[i].copy()
    
    # SLSQP refinement
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8})
    best_h = res.x
    best_c5 = res.fun
    
    return best_h, best_c5, n_points
