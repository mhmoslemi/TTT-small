import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    bounds = [(0.0, 1.0)] * n_points
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count}
    
    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()
    best_c5_list = np.inf
    
    def perturb(h, scale):
        return np.clip(h + np.random.normal(0, scale, n_points), 0.0, 1.0) * (ones_count / np.sum(h))
    
    def optimize(start):
        result = minimize(
            fun=objective,
            x0=start,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 100, 'eps': 1e-6, 'ftol': 1e-9}
        )
        if result.success:
            return result.x
        return start
    
    for i in range(20):
        if time.time() - start_time > budget_s:
            break
        noisy = perturb(h, 0.1)
        if np.min(noisy) < 0:
            noisy = np.clip(noisy, 0, 1)
        noisy = noisy * (ones_count / np.sum(noisy))
        h = optimize(noisy)
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if current_c5 < best_c5_list:
            best_c5_list = current_c5
    
    while time.time() - start_time < budget_s:
        noisy = perturb(best_h, 0.05)
        noisy = np.clip(noisy, 0, 1)
        noisy = noisy * (ones_count / np.sum(noisy))
        h = optimize(noisy)
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if current_c5 < best_c5_list:
            best_c5_list = current_c5
    
    return best_h, best_c5, n_points
