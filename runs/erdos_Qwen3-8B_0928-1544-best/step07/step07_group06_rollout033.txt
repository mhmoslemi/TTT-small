import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 120  # Increased for better resolution
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy().astype(float)
    else:
        initial_h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    # Optimized parameters
    learning_rate = 0.1
    decay_rate = 0.995
    max_iter = 2000
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
        
        # Identify worst shift
        k_max = np.argmax(corr)
        k_shift = k_max - n_points + 1
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j_plus = (i + k_shift) % n_points
            j_minus = (i - k_shift) % n_points
            grad[i] = (1 - best_h[j_plus]) - best_h[j_minus]
        
        # Gradient descent step with dx scaling
        step = learning_rate * dx * grad

        candidate_h = np.clip(candidate_h, 0.0, 1.0)
        candidate_h = candidate_h * (ones_count / np.sum(candidate_h))
        
        # SLSQP refinement
        result = minimize(
