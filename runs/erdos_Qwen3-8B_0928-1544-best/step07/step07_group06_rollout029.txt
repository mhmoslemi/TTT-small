            indices = np.random.choice(n_points, size=ones_count, replace=False)
            h = np.zeros(n_points, dtype=int)
            h[indices] = 1
            population.append(h)
    else:
        for _ in range(population_size):
            indices = np.random.choice(n_points, size=ones_count, replace=False)
            h = np.zeros(n_points, dtype=int)
            h[indices] = 1
            population.append(h)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h) for h in population]
        sorted_idx = np.argsort(fitness)
        parents = [population[i] for i in sorted_idx[:population_size//2]]
        offspring = []
        for _ in range(population_size - len(parents)):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = np.zeros(n_points, dtype=int)
            for i in range(n_points):
                child[i] = p1[i] if np.random.rand() < 0.5 else p2[i]
            for i in range(n_points):
                if np.random.rand() < 0.05:
                    child[i] ^= 1
            while np.sum(child) != ones_count:
                cs = np.sum(child)
                if cs > ones_count:
                    idx = np.where(child == 1)[0]
                    flip = np.random.choice(idx, size=cs - ones_count, replace=False)
                    child[flip] = 0
                else:
                    idx = np.where(child == 0)[0]
                    flip = np.random.choice(idx, size=ones_count - cs, replace=False)
                    child[flip] = 1
            offspring.append(child)
        population = parents + offspring
        for h in population:
            c = compute_c5(h)
            if c < best_c5:
                best_c5 = c
                best_h = h
    return best_h, best_c5, n_points
