import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        return np.max(corr * dx)
    
    population = [initial_h]
    for _ in range(20):
        h = np.random.rand(n_points)
        h *= ones_count / np.sum(h)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = initial_h
    start_time = time.time()
    
    while time.time() - start_time < 0.2 * budget_s:
        fitness = np.array([compute_c5(h) for h in population])
        idx = np.argmin(fitness)
        best_c5, best_h = fitness[idx], population[idx]
        
        top_idx = np.argsort(fitness)[:int(len(population)*0.2)]
        selected = population[top_idx]
        
        new_pop = selected.copy()
        while len(new_pop) < len(population):
            p1, p2 = np.random.choice(selected, 2, replace=False)
            child = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = p1[i]
                else:
                    child[i] = p2[i]
            mutate = np.random.rand(n_points) < 0.1
            child[mutate] += np.random.normal(0, 0.05, np.sum(mutate))
            child = np.clip(child, 0, 1)
            tot = np.sum(child)
            child = child * (ones_count / tot) if tot > 0 else np.random.rand(n_points) * (ones_count / np.sum(np.random.rand(n_points)))
            new_pop.append(child)
        population = new_pop
    
    def objective(h):
        return compute_c5(h)
    
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count}
    
    for h in population:
        res = minimize(objective, h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=constraint, options={'maxiter':200})
        if res.success and res.fun < best_c5:
            best_c5, best_h = res.fun, res.x
    
    return best_h, best_c5, n_points
