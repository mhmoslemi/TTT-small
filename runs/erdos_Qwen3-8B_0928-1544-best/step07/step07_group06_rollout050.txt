    for i in range(n_points):
        j_plus = (i + k_shift) % n_points
        j_minus = (i - k_shift) % n_points
        grad[i] = (1 - best_h[j_plus]) - best_h[j_minus]
    
    learning_rate = 0.1
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        step = learning_rate * grad
        candidate_h = best_h - step
        candidate_h = np.clip(candidate_h, 0.0, 1.0)
        total = np.sum(candidate_h)
        if total == 0:
            candidate_h[0] = 1.0
        else:
            scale = ones_count / total
            candidate_h = candidate_h * scale
        candidate_h = np.clip(candidate_h, 0.0, 1.0)
        current_c5 = compute_c5(candidate_h)
        if current_c5 < compute_c5(best_h):
            best_h = candidate_h.copy()
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        k_shift = k_max - n_points + 1
        grad = np.zeros(n_points)
        for i in range(n_points):
            j_plus = (i + k_shift) % n_points
            j_minus = (i - k_shift) % n_points
            grad[i] = (1 - best_h[j_plus]) - best_h[j_minus]
        learning_rate *= 0.99
    
    return best_h, best_c5, n_points
