            h = h * (n_points / 2) / np.sum(h)
            guesses.append(h)
        for phase in np.linspace(0, 2 * np.pi, 5):
            h = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points + phase)
            guesses.append(h)
        return guesses
    
    while current_stage < len(stages) and time.time() - start_time < budget_s:
        n_points, required_sum = stages[current_stage]
        initial_guesses = generate_initial_guesses(n_points)
        if best_h is not None:
            initial_guesses.append(upscale(best_h, n_points))
        stage_best_h = None
        stage_best_c5 = float('inf')
        
        for guess in initial_guesses:
            def objective(h):
                corr = np.correlate(h, 1 - h, mode='full')
                return np.max(corr) * (2.0 / n_points)
            
            def constraint_sum(h):
                return np.sum(h) - required_sum
            
            bounds = [(0.0, 1.0)] * n_points
            constraints = [{'type': 'eq', 'fun': constraint_sum}]
            
            method = 'SLSQP'
            options = {'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
            
            result = minimize(objective, guess, bounds=bounds, constraints=constraints, method=method, options=options)
            if result.success:
                current_c5 = result.fun
                if current_c5 < stage_best_c5:
                    stage_best_c5 = current_c5
                    stage_best_h = result.x
        
        if stage_best_h is not None:
            if stage_best_c5 < best_c5:
                best_c5 = stage_best_c5
                best_h = stage_best_h.copy()
            if current_stage < len(stages) - 1:
                next_n_points = stages[current_stage + 1][0]
                best_h = upscale(best_h, next_n_points)
                current_stage += 1
    
    return best_h, best_c5, len(best_h) if best_h is not None else 0
