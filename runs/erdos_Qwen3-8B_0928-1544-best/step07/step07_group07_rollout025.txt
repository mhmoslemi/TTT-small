            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            
            if n_points < max_points:
                new_n_points = 2 * n_points
                new_h = np.zeros(new_n_points)
                for i in range(n_points):
                    new_h[2*i] = current_h[i]
                    new_h[2*i + 1] = current_h[i]
                current_h = new_h
                n_points = new_n_points
        else:
            break
    
    return best_h, best_c5, n_points
