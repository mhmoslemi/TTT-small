import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    population_size = 150
    generations = 30
    mutation_rate = 0.15
    population = []
    
    for _ in range(population_size):
        vec = np.zeros(n_points)
        indices = np.random.choice(n_points, size=ones_count, replace=False)
        vec[indices] = 1
        population.append(vec)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    for _ in range(generations):
        fitness = np.array([evaluate(ind) for ind in population])
        elite_size = int(population_size * 0.2)
        elite = np.array(population[np.argsort(fitness)[:elite_size]])
        
        new_population = []
        for _ in range(population_size - elite_size):
            p1, p2 = np.random.choice(elite, size=2, replace=False)
            child = p1.copy()
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = 1 - child[i]
            ones = np.sum(child)
            if ones != ones_count:
                diff = ones - ones_count
                if diff > 0:
                    idx = np.where(child == 1)[0]
                    if idx.size > 0:
                        to_remove = np.random.choice(idx, size=diff, replace=False)
                        child[to_remove] = 0
                else:
                    idx = np.where(child == 0)[0]
                    if idx.size > 0:
                        to_add = np.random.choice(idx, size=-diff, replace=False)
                        child[to_add] = 1
            new_population.append(child)
        population = elite.tolist() + new_population
    
    best_h = initial_h.copy()
    best_c5 = float('inf')
    
    for ind in population:
        h_initial = ind.astype(float)
        bounds = [(0.0, 1.0)] * n_points
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - ones_count}
        result = minimize(lambda h: evaluate(h), h_initial, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 200})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
