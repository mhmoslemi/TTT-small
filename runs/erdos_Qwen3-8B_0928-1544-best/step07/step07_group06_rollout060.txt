        population = [initial_h_values.copy()]
    else:
        n_points = 70
        total_ones = n_points // 2
        population = [np.random.randint(0, 2, n_points)]
        for _ in range(99):
            h = np.random.randint(0, 2, n_points)
            s = np.sum(h)
            if s != total_ones:
                diff = s - total_ones
                if diff > 0:
                    idxs = np.where(h == 1)[0]
                    np.random.shuffle(idxs)
                    for i in idxs[:diff]:
                        h[i] = 0
                else:
                    idxs = np.where(h == 0)[0]
                    np.random.shuffle(idxs)
                    for i in idxs[: -diff]:
                        h[i] = 1
            population.append(h)

    def compute_c5(h):
        vals = np.zeros(n_points)
        for k in range(n_points):
            shifted = np.roll(1 - h, k)
            vals[k] = np.dot(h, shifted)
        return np.max(vals) * (2.0 / n_points)

    best_c5 = float('inf')
    best_h = population[0]
    start_time = time.time()

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        fit = [compute_c5(h) for h in population]
        best_idx = np.argmin(fit)
        if fit[best_idx] < best_c5:
            best_c5, best_h = fit[best_idx], population[best_idx]
        parents = [population[i] for i in np.argsort(fit)[:len(population)//2]]
        new_pop = []
        for _ in range(len(population) - len(parents)):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
            cnt = np.sum(child)
            diff = cnt - total_ones
            if diff != 0:
                if diff > 0:
                    idxs = np.where(child == 1)[0]
                    np.random.shuffle(idxs)
                    for i in idxs[:diff]:
                        child[i] = 0
                else:
                    idxs = np.where(child == 0)[0]
                    np.random.shuffle(idxs)
                    for i in idxs[: -diff]:
                        child[i] = 1
            new_pop.append(child)
        for i in range(len(new_pop)):
            population[ -len(new_pop)+i ] = new_pop[i]

    return best_h, best_c5, n_points
