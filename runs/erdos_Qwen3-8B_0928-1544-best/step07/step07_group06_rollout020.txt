import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    population_size = 100
    initial_population = []
    
    if 'initial_h_values' in globals():

        initial_h = np.round(initial_h).astype(int)

            candidate = np.random.choice([0, 1], size=n_points)
        initial_population.append(candidate)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points, dtype=int)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        current_pop = np.array(initial_population, dtype=int)
        fitness = []
        for h in current_pop:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            fitness.append(c5)
        
        best_idx = np.argmin(fitness)
        current_best = current_pop[best_idx]
        current_best_c5 = fitness[best_idx]

            best_c5 = current_best_c5
            best_h = current_best.copy()
        
        parents = current_pop[np.argsort(fitness)][:population_size // 2]
        new_pop = []
        
        for _ in range(population_size - len(parents)):
