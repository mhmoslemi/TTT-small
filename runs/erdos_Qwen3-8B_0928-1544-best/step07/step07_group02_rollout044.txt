            current_c5 = result.fun
            if current_c5 < best_medium_c5:
                best_medium_c5 = current_c5
                best_medium_h = result.x
        if time.time() - start_time > budget_s:
            break

    # Upscale to final grid
    final_h = upscale(best_medium_h, n_final)

    # Final optimization
    def objective_final(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_final)

    def constraint_sum_final(h):
        return np.sum(h) - ones_final

    result_final = minimize(
        fun=objective_final,
        x0=final_h,
        method='COBYLA',
        bounds=[(0.0, 1.0)]*n_final,
        constraints={'type': 'eq', 'fun': constraint_sum_final},
        options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8}
    )

    best_h = result_final.x
    best_c5 = result_final.fun
    return best_h, best_c5, n_final
