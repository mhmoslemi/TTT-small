import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    # Ensure sum is correct
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    
    # Parameters
    alpha = 0.1
    step_decay = 0.995
    max_iter = 1000
    freeze_steps = 100
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Compute correlation
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = best_h.copy()
        
        # Compute gradient
        grad = (1 - 2 * best_h) * np.sign(corr)
        
        # Update h
        best_h += alpha * grad
        
        # Clip and rescale
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (ones_count / np.sum(best_h))
        
        alpha *= step_decay
        
        # Freeze early if convergence
        if _ > freeze_steps and np.abs(best_c5 - 0.38085) < 1e-5:
            break
    
    return best_h, best_c5, n_points
