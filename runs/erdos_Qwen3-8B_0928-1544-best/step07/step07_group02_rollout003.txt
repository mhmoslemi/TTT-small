import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    bounds = [(0.0, 1.0)] * n_points
    initial_guesses = []

    if 'initial_h_values' in globals():
        initial_guesses.append(initial_h_values.copy())

    # Generate random, sinusoidal, and step function guesses
    for _ in range(5):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (ones_count / np.sum(h_rand))
        initial_guesses.append(h_rand)

    x = np.linspace(0, 2, n_points)
    sin_guess = 0.5 * (1 + np.sin(np.pi * x / 2))
    sin_guess = np.clip(sin_guess, 0, 1)
    sin_guess = sin_guess * (ones_count / np.sum(sin_guess))
    initial_guesses.append(sin_guess)

    step_guess = np.zeros(n_points)
    step_guess[:ones_count] = 1
    initial_guesses.append(step_guess)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    for guess in initial_guesses:
        result = minimize(
            fun=objective,
            x0=guess,
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 250, 'disp': False}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n_points
