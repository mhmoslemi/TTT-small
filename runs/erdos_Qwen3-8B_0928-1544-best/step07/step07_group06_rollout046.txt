import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    T = budget_s
    time_used = 0

    if 'initial_h_values' in globals():
        h0 = initial_h_values.copy()
    else:
        h0 = np.random.rand(n_points)
        h0 /= np.sum(h0)
        h0 *= ones_count
        h0 = np.clip(h0, 0, 1)

    def calc_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        return np.max(corr) * dx

    def calc_grad(h, k):
        grad = np.zeros(n_points)
        for i in range(n_points):
            j_plus = (i + k) % n_points
            j_minus = (i - k) % n_points
            grad[i] = (1 - h[j_plus]) - h[j_minus]
        return grad

    def optimize_c5(h0):
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count})
        result = minimize(lambda x: calc_c5(x), x0=h0, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, options={'maxiter': 100})
        return result

    def ga_phase(h0, pop_size=50, gens=200, time_per_gen=5):
        population = [np.random.dirichlet(np.ones(n_points), size=1)[0] * ones_count for _ in range(pop_size)]
        best_pop = population[0]
        best_c5 = calc_c5(best_pop)
        for gen in range(gens):

            best_idx = np.argmin(fitness)
            if fitness[best_idx] < best_c5:
                best_c5 = fitness[best_idx]
                best_pop = population[best_idx]
            selected = np.argsort(fitness)[:int(pop_size * 0.5)]
            selected_pop = [population[i] for i in selected]
            new_pop = []
            for i in range(0, len(selected_pop), 2):
                parent1 = selected_pop[i]
                parent2 = selected_pop[i+1] if i+1 < len(selected_pop) else parent1
                child = (parent1 + parent2) / 2
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                if np.sum(child) > 0:
                    child = child / np.sum(child) * ones_count
                new_pop.append(child)
            population = new_pop
            time_used += time_per_gen
        return best_pop, best_c5

    best_h, best_c5 = h0, calc_c5(h0)
    h0_refined = optimize_c5(h0).x
    best_c5 = min(best_c5, calc_c5(h0_refined))
    best_h = h0_refined

    if 'initial_h_values' in globals():
        h_candidate = initial_h_values
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate / np.sum(h_candidate) * ones_count
        h_candidate = np.clip(h_candidate, 0, 1)
        current_c5 = calc_c5(h_candidate)
        if current_c5 < best_c5:
            best_h, best_c5 = h_candidate, current_c5

    h_pop, c_pop = ga_phase(h0, pop_size=50, gens=200, time_per_gen=5)
    best_c5 = min(best_c5, c_pop)
    best_h = h_pop

    if time_used < T:
        result = optimize_c5(best_h)
        best_c5 = min(best_c5, result.fun)
        best_h = result.x

    return best_h, best_c5, n_points
