import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def gen_candidate():
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        return h * (ones_count / h_sum)
    
    def fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # Genetic algorithm
    pop = [initial_h] + [gen_candidate() for _ in range(199)]
    pop = np.array([np.clip(x, 0, 1) for x in pop])
    fit = np.array([fitness(x) for x in pop])
    best_idx = np.argmin(fit)
    best_h = pop[best_idx]
    best_c5 = fit[best_idx]
    start_time = time.time()
    
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        idxs = np.argsort(fit)[:int(0.2 * len(pop))]
        parents = pop[idxs]
        children = []
        for i in range(0, len(parents), 2):
            if i+1 < len(parents):
                p1, p2 = parents[i], parents[i+1]
                child = np.random.choice([p1, p2], size=n_points)
                child = np.clip(child, 0, 1)
                child_sum = np.sum(child)
                children.append(child * (ones_count / child_sum))
        pop = np.vstack([parents, children])
        fit = np.array([fitness(x) for x in pop])
        pop = pop[np.argsort(fit)[:n_points]]
        fit = fit[np.argsort(fit)[:n_points]]
        best_idx = np.argmin(fit)
        best_h, best_c5 = pop[best_idx], fit[best_idx]
    
    # Local refinement
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 100})
    best_h, best_c5 = res.x, res.fun if res.success else best_c5
    
    return best_h, best_c5, n_points
