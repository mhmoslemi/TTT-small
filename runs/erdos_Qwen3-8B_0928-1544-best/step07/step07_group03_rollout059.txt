            h_clipped = np.clip(particles[i], 0.0, 1.0)
            current_sum = np.sum(h_clipped)
            if current_sum > 0:
                particles[i] = h_clipped * (target_sum / current_sum)
    
    # Run SLSQP on the best solution
    best_h = gbest_position.copy()
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
    bounds = [(0.0, 1.0)] * n_points
    
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8})
    best_h = res.x
    best_c5 = res.fun
    
    return best_h, best_c5, n_points
