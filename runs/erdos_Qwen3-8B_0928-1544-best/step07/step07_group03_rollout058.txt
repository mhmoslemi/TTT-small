        particles[i] = np.clip(particles[i], 0, 1)
        sum_p = np.sum(particles[i])
        if sum_p != 0:
            particles[i] = particles[i] * (target_sum / sum_p)
    velocity = np.random.rand(n_particles, n_points) * 0.1
    pbest = particles.copy()
    pbest_vals = np.array([obj(p) for p in particles])
    gbest = pbest[np.argmin(pbest_vals)].copy()
    gbest_val = np.min(pbest_vals)
    best_h = gbest.copy()
    best_c5 = gbest_val
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocity[i] = inertia * velocity[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] = particles[i] + velocity[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_p = np.sum(particles[i])
            if sum_p != 0:
                particles[i] = particles[i] * (target_sum / sum_p)
        for i in range(n_particles):
            current_val = obj(particles[i])
            if current_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
        current_index = np.argmin(pbest_vals)
        if pbest_vals[current_index] < gbest_val:
            gbest = pbest[current_index].copy()
            gbest_val = pbest_vals[current_index]
        if gbest_val < best_c5:
            best_h = gbest.copy()
            best_c5 = gbest_val
    res = minimize(obj, best_h, method='SLSQP', bounds=[(0.0, 1.0)] * n_points, constraints={'type': 'eq', 'fun': cons})
    best_h = res.x
    best_c5 = res.fun
    return best_h, best_c5, n_points
