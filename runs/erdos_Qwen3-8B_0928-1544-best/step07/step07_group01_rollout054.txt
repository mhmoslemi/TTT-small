import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 50  # Reduced to speed up computation
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Simulated annealing with adaptive cooling
    temp = 1.0
    cooling_rate = 0.995
    for step in range(200):
        if time.time() - start_time > budget_s:
            break
        # Perturbation with Gaussian noise
        h_perturbed = initial_h + np.random.normal(0, temp, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        # Local optimization
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-7, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                initial_h = result.x
        
        temp *= cooling_rate
    
    return best_h, best_c5, n_points
