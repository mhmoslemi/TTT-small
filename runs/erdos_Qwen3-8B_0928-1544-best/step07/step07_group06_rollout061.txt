        parents = [population[i] for i in sorted_indices[:population_size//2]]
        
        offspring = []
        for _ in range(population_size - len(parents)):
            p1, p2 = np.random.choice(parents, size=2, replace=False)
            child = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = p1[i]
                else:
                    child[i] = p2[i]
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0, 1)
            sum_child = np.sum(child)
            if abs(sum_child - ones_count) > 1e-6:
                scale = (ones_count - sum_child) / n_points
                child += scale
                child = np.clip(child, 0, 1)
            offspring.append(child)
        
        new_population = parents + offspring
        new_fitness = np.array([evaluate_fitness(h) for h in new_population])
        combined = np.column_stack((new_fitness, new_population))
        combined.sort()
        new_population = combined[:population_size, 1:]
        
        current_best = new_population[0]
        current_c5 = evaluate_fitness(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best
        population = new_population
    
    return best_h, best_c5, n_points
