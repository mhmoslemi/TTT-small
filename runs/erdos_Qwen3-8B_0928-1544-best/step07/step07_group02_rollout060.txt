    best_h = np.zeros(n_points)
    start_time = time.time()

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / len(h))

    def cons(h):
        return np.sum(h) - len(h) / 2

    def _opt(x0, res_n, method='COBYLA', **kwargs):
        x0 = np.clip(x0, 0, 1)
        x0 = x0 * (res_n / 2) / np.sum(x0)
        res = minimize(obj, x0, method=method, bounds=bounds, constraints={'type': 'eq', 'fun': cons}, **kwargs)
        return res

    for g in startup:

        if res.success:
            if res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x

        if time.time() - start_time > budget_s:

    if time.time() - start_time < budget_s:

            x = np.linspace(0, 2, len(h))

        n_points = 35
