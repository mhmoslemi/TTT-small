        total = np.sum(h)
        scaling = (n_points / 2) / total
        return h * scaling
    
    initial_h = adjust_h(initial_h)
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    # Optimization objective
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / len(h)
        return max_corr * dx
    
    while time.time() - start_time < budget_s:
        current_n = len(best_h)
        if current_n >= max_points:
            break
        # Refine grid by inserting points at largest gradient locations
        diffs = np.abs(best_h[1:] - best_h[:-1])
        sort_indices = np.argsort(diffs)[::-1]
        num_refine = min(20, len(sort_indices))
        selected_edges = sort_indices[:num_refine]
        # Insert new points in reverse order to avoid index shifting
        insert_indices = [idx + 1 for idx in selected_edges]
        insert_indices.sort(reverse=True)
        new_h = best_h.copy()
        for idx in insert_indices:
            new_h = np.insert(new_h, idx, (new_h[idx-1] + new_h[idx])/2)
        new_n = len(new_h)
        if new_n > max_points:
            new_h = new_h[:max_points]
        # Prepare optimization
        bounds = [(0.0, 1.0)] * new_n
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - new_n / 2}
        result = minimize(
            fun=objective,
            x0=new_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to previous best
            pass
    
    return best_h, best_c5, len(best_h)
