import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    n_ones = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = np.round(initial_h_values).astype(int)
        initial_h[initial_h == 1] = 1
        initial_h[initial_h == 0] = 0
    else:
        initial_h = np.zeros(n_points, dtype=int)
        indices = np.random.choice(n_points, size=n_ones, replace=False)
        initial_h[indices] = 1
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    population_size = 100
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.zeros(n_points, dtype=int)
        indices = np.random.choice(n_points, size=n_ones, replace=False)
        h[indices] = 1
        population.append(h)
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    generations = 0
    
    while time.time() - start_time < budget_s:
        candidates = []
        for h in population:
            c5 = compute_c5(h)
            candidates.append((c5, h))
        candidates.sort()
        parents = [h for (c5, h) in candidates[:population_size // 2]]
        
        offspring = []
        for _ in range(population_size - len(parents)):
            p1, p2 = random.sample(parents, 2)
            child = np.zeros(n_points, dtype=int)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = p1[i]
                else:
                    child[i] = p2[i]
            for i in range(n_points):
                if random.random() < 0.05:
                    child[i] ^= 1
            current_sum = np.sum(child)
            if current_sum != n_ones:
                delta = current_sum - n_ones
                for _ in range(abs(delta)):
                    if delta > 0:
                        idx = np.random.choice(np.where(child == 1)[0])
                        child[idx] = 0
                    else:
                        idx = np.random.choice(np.where(child == 0)[0])
                        child[idx] = 1
            offspring.append(child)
        
        population = parents + offspring
        for (c5, h) in candidates:
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        generations += 1
    
    return best_h, best_c5, n_points
