from scipy.optimize import minimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()

    # Feasibility projection function
    def project_feasible(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clipped)
        if np.isclose(sum_h, 0.0):
            return h_clipped
        scale = (n_points / 2) / sum_h
        return h_clipped * scale
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()

        iter += 1

        if iter < 5:

        h_perturbed = initial_h + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        # Set up optimization
        def objective_func(x):
            return objective(x)
        
        def constraint_func(x):
            return constraint_sum(x)
        
        constraints = [{'type': 'eq', 'fun': constraint_func}]
        bounds_opt = Bounds(0.0, 1.0)
        
        # Use L-BFGS-B for scalability
        result = minimize(
            fun=objective_func,
            x0=h_perturbed,
