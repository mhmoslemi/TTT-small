    decay_rate = 0.999

        if current_c5 < best_c5:

        # Find the worst shift
        k_max = np.argmax(corr)

            grad[i] = (1 - best_h[j_plus]) - best_h[j_minus]
        
        # Update h in the direction that reduces the overlap

        candidate_h = best_h - step

        # Use SLSQP to refine the candidate
        result = minimize(
