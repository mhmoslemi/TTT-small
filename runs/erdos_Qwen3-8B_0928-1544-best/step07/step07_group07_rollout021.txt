import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    T = 0.5
    alpha = 0.995
    max_iter = 10000
    iter_count = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        current_h = best_h.copy()
        perturb_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        perturb_amount = np.random.normal(0, T, len(perturb_indices))
        current_h[perturb_indices] += perturb_amount
        
        current_h = np.clip(current_h, 0.0, 1.0)
        current_h = current_h * (ones_count / np.sum(current_h))
        
        candidate_c5 = evaluate_erdos_solution(current_h)
        if candidate_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - candidate_c5) / T):
