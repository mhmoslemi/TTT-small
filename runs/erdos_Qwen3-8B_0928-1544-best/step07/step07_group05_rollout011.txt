import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def scale_clip(h):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if s == 0:
            return h
        scale = target_sum / s
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        return h
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def evaluate_fitness(h):
        return -compute_c5(h)
    
    population_size = 50
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.normal(0, 0.1, n_points)
        h = initial_h + h
        h = np.clip(h, 0.0, 1.0)
        h = scale_clip(h)
        population.append(h)
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = compute_c5(initial_h)
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate_fitness(ind) for ind in population]
        parents = []
        for _ in range(population_size):
            idxs = np.random.choice(len(population), 3, replace=False)
            best_idx = idxs[np.argmax(fitness[idxs])]
            parents.append(population[best_idx])
        new_population = []
        for i in range(0, population_size, 2):
            p1, p2 = parents[i], parents[i+1] if i+1 < len(parents) else parents[i]
            if np.random.rand() < 0.8:
                split = np.random.randint(1, n_points)
                c1, c2 = np.concatenate([p1[:split], p2[split:]]), np.concatenate([p2[:split], p1[split:]])
            else:
                c1, c2 = p1.copy(), p2.copy()
            for c in [c1, c2]:
                for j in range(n_points):
                    if np.random.rand() < 0.1:
                        c[j] += np.random.normal(0, 0.1)
                        c[j] = np.clip(c[j], 0.0, 1.0)
                c = scale_clip(c)
            new_population.extend([c1, c2])
        population = new_population[:population_size]
        current_fitness = [evaluate_fitness(ind) for ind in population]
        idx = np.argmax(current_fitness)
        if -current_fitness[idx] < best_c5:
            best_c5 = -current_fitness[idx]
            best_h = population[idx].copy()
    return best_h, best_c5, n_points
