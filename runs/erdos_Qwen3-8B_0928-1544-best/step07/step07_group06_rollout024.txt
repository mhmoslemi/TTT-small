import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Increase points for better precision
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    # Parameters for gradient-based refinement
    learning_rate = 0.2
    decay_rate = 0.995
    max_iterations = 800
    iteration = 0
    
    while time.time() - start_time < budget_s and iteration < max_iterations:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        current_c5 = np.max(corr) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
        
        # Find top 3 shifts
        top_shifts = np.argsort(corr)[::-1][:3]
        grad = np.zeros(n_points)
        for shift_idx in top_shifts:
            k_shift = shift_idx - n_points + 1
            for i in range(n_points):
                j_plus = (i + k_shift) % n_points
                j_minus = (i - k_shift) % n_points
                grad[i] += (1 - best_h[j_plus]) - best_h[j_minus]
        
        # Update h in the direction that reduces the overlap
        step = learning_rate * grad
        candidate_h = best_h - step
        candidate_h = np.clip(candidate_h, 0.0, 1.0)
        candidate_h = candidate_h * (ones_count / np.sum(candidate_h))
        
        # Use SLSQP to refine the candidate
        result = minimize(
            fun=objective,
            x0=candidate_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            learning_rate *= decay_rate
        iteration += 1
    
    return best_h, best_c5, n_points
