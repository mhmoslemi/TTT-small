from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    ones_count = n_points // 2
    if 'initial_h_values' in globals():

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    def optimize(start_h):
        result = minimize(
            fun=objective,
            x0=start_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - ones_count},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            return result.x, result.fun
        return start_h, best_c5
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.1, n_points)
        h_perturbed = best_h + perturb
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        h_perturbed, c = optimize(h_perturbed)
        if c < best_c5:
            best_c5 = c
            best_h = h_perturbed
    
    while time.time() - start_time < budget_s:
