import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.zeros(n_points)
        idx = np.random.choice(n_points, ones_count, replace=False)
        initial_h[idx] = 1

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points

    # Genetic Algorithm
    pop_size = 100
    pop = []
    for _ in range(pop_size):
        idx = np.random.choice(n_points, ones_count, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1
        pop.append(h)
    if 'initial_h_values' in globals():
        pop.append(initial_h)

    for _ in range(50):
        fits = np.array([evaluate_c5(h) for h in pop])
        best_idx = np.argmin(fits)
        best_h = pop[best_idx]
        new_pop = [best_h]  # elitism
        while len(new_pop) < pop_size:
            a, b = np.random.choice(pop, 2, replace=False)
            child = a.copy()
            crossover = np.random.rand(n_points) < 0.5
            child[crossover] = b[crossover]
            if np.sum(child) != ones_count:
                child[(child == 1) & (np.random.rand(n_points) < 0.5)] = 0
                child[(child == 0) & (np.random.rand(n_points) < 0.5)] = 1
            new_pop.append(child)
        pop = new_pop

    best_h = min(pop, key=evaluate_c5)
    best_c5 = evaluate_c5(best_h)

    # Local refinement
    def objective(x):
        h = x.copy()
        h /= np.sum(h)
        h *= ones_count
        h = np.clip(h, 0, ones_count)
        return evaluate_c5(h)

    result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-8, options={'maxiter': 300})
    if result.success:
        best_h = result.x
        best_c5 = result.fun

    return best_h, best_c5, n_points
