from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    def optimize_with_perturbations(start_h):
        best_h_local = start_h.copy()
        best_c5_local = objective(start_h)
        for _ in range(300):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, 0.15, n_points)
            h_perturbed = start_h + perturb
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)

            result = minimize(
                fun=objective,
                x0=h_perturbed,

                constraints={'type': 'eq', 'fun': constraint_sum},

            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5_local:
