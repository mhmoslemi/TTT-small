import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    temp_start = 1.0
    temp_end = 0.01
    decay_rate = 0.999

    for step in range(steps):

        h_perturbed = best_h + np.random.normal(0, noise_scale, n_points)
