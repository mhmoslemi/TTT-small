import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    def optimize_with_perturbations(start_h, pert_sigma):
        best_h_local = start_h.copy()
        best_c5_local = objective(start_h)
        for _ in range(150):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, pert_sigma, n_points)
            h_perturbed = start_h + perturb
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
            result = minimize(
                fun=objective,
                x0=h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-5}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5_local:
                    best_c5_local = current_c5
                    best_h_local = result.x
                start_h = best_h_local
        return best_h_local, best_c5_local
    
    best_c5 = np.inf
    current_h = initial_h.copy()
    pert_schedules = [0.2, 0.1, 0.05]

        if c < best_c5:
