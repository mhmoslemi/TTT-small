import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70

    alpha = 0.01
    decay_rate = 0.995
    max_iter = 10000
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s:
        iter_count += 1
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_val = np.max(corr)
        sign = np.sign(max_corr_val)

        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
