import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 4 == 0:
                h[i] = 1.0
            elif i % 4 == 1:
                h[i] = 0.5
            elif i % 4 == 2:
                h[i] = 0.5
            else:
                h[i] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    max_iterations = 5000
    iteration = 0
    learning_rate = 0.1
    while iteration < max_iterations and time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        m = idx_max - (n_points - 1)
        gradient = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + m < n_points:
                gradient[j] += 1 - h[j + m]
            if 0 <= j - m < n_points:
                gradient[j] -= h[j - m]
        alpha = learning_rate
        h_candidate = h + alpha * (-gradient)
        h_candidate = np.clip(h_candidate, 0.0, 1.0)
        h_candidate_sum = np.sum(h_candidate)
        if h_candidate_sum == 0:
            h = h.copy()
            continue
        h_candidate = h_candidate * (n_points / 2) / h_candidate_sum
        corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        new_c5 = np.max(corr_candidate) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_candidate.copy()
            h = h_candidate.copy()
            learning_rate *= 0.999
        else:
            learning_rate *= 1.01
        if iteration % 50 == 0 and iteration != 0:
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2) / np.sum(h)
        iteration += 1
    return best_h, best_c5, n_points
