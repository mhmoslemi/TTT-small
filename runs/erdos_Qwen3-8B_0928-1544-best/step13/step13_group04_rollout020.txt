import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    k = 7
    n_coarse = n_points // k
    ones_count = n_points // 2

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    # Initialize coarse grid
    if 'initial_h_values' in globals():
        initial_coarse = initial_h_values[::k]

    h_coarse = initial_coarse * (n_coarse / 2) / initial_coarse_sum

    # Coarse grid optimization
    max_iterations_coarse = 500
    step_size_coarse = 0.1
    for _ in range(max_iterations_coarse):
        corr = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        max_corr_idx = np.argmax(corr)
        lag = max_corr_idx - (n_coarse - 1)
        gradient = np.zeros(n_coarse)
        for i in range(n_coarse):
            j = i + lag
            if 0 <= j < n_coarse:
                gradient[i] += (1 - h_coarse[j])
        h_coarse = np.clip(h_coarse - step_size_coarse * gradient, 0.0, 1.0)
        h_coarse_sum = np.sum(h_coarse)
        if h_coarse_sum > 0:
            h_coarse = h_coarse * (n_coarse / 2) / h_coarse_sum

    # Upscale to fine grid
    h_fine = np.repeat(h_coarse, k)
    h_fine_sum = np.sum(h_fine)
    h_fine_scaled = h_fine * (ones_count / h_fine_sum)

    # Fine grid optimization
    max_iterations_fine = 800
    step_size_fine = 0.1
    best_h = h_fine_scaled.copy()
    best_c5 = compute_C5(h_fine_scaled)
    start_time = time.time()
    for _ in range(max_iterations_fine):

        corr = np.correlate(h_fine_scaled, 1 - h_fine_scaled, mode='full')

        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = i + lag
