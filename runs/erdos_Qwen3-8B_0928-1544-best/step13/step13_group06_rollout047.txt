import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    max_iterations = 2000
    learning_rate = 0.1
    decay_rate = 0.99
    start_time = time.time()
    
    for iter in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        derivatives = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + k
            i_minus_k = i - k
            term1 = 0.0

            if 0 <= i_minus_k < n_points:
