        centroid = np.mean(sorted_simplex[:-1], axis=0)
        reflection = centroid + 1.0 * (centroid - worst_p)
        reflection = adjust_h(reflection, target_sum)
        new_fitness = compute_c5(reflection)

        if new_fitness < sorted_fitness[-1]:
            new_simplex = sorted_simplex[:-1] + [reflection]
            new_fitness_list = sorted_fitness[:-1] + [new_fitness]
            simplex, fitness = new_simplex, new_fitness_list
        else:
            contraction = centroid + 0.5 * (centroid - worst_p)
            contraction = adjust_h(contraction, target_sum)
            new_contract = compute_c5(contraction)
            if new_contract < sorted_fitness[-1]:
                new_simplex = sorted_simplex[:-1] + [contraction]
                new_fitness_list = sorted_fitness[:-1] + [new_contract]
                simplex, fitness = new_simplex, new_fitness_list
            else:
                shrink = best_p + 0.5 * (np.array(simplex) - best_p)
                shrink = [adjust_h(s, target_sum) for s in shrink]
                simplex = shrink
                fitness = [compute_c5(s) for s in simplex]

        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_c5 = fitness[current_best]
            best_h = simplex[current_best]

        iterations += 1

    return best_h, best_c5, n_points
