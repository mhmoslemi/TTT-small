import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points

    population = [initial_h.copy()]

        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = [np.max(np.correlate(ind, 1 - ind, mode='full')) * dx for ind in population]
        if np.min(fitness) < best_c5:
            best_c5 = np.min(fitness)
            best_h = population[fitness.index(np.min(fitness))]
        parents = []
        for _ in range(25):
            idx = np.random.choice(50, 5, replace=False)
            winner = population[idx[np.argmin(fitness[idx])]]
            parents.append(winner)

        for i in range(0, len(parents), 2):

            p2 = parents[i+1] if i+1 < len(parents) else parents[0]
            split = np.random.randint(1, n_points-1)
            off1 = np.concatenate([p1[:split], p2[split:]])
            off2 = np.concatenate([p2[:split], p1[split:]])
            offspring.append(off1)
            offspring.append(off2)
        mutated = []
        for child in offspring:
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.05, size=n_points)
                child += noise

            child_sum = np.sum(child)

            mutated.append(child)
        population = mutated[:50]
