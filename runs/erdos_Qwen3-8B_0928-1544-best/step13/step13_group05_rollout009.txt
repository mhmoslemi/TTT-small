import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 100
    ones_count = n_points // 2
    h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    h = h * (ones_count / np.sum(h))

    # Convert to symbolic variables
    h_vars = cp.Variable(n_points)
    C = cp.Variable()
    constraints = [cp.sum(h_vars) == ones_count]
    for k in range(n_points):
        idx = (np.arange(n_points) + k) % n_points
        expr = cp.sum(h_vars * (1 - h_vars[idx]))
        constraints.append(expr <= C * n_points / 2)
    
    problem = cp.Problem(cp.Minimize(C), constraints)
    problem.solve(solver='SCIPY', eps=1e-6, time_limit=budget_s - 60)

    if problem.status == 'optimal':
        h_opt = h_vars.value
        c5 = C.value
        return h_opt, c5, n_points
    else:
        return h, np.max(np.correlate(h, 1 - h, mode='full') * (2.0 / n_points)), n_points
