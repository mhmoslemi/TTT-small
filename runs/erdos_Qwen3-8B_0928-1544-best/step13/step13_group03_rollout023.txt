import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_coarse = 35
    n_fine = 70
    max_iter = 500
    initial_temp = 0.5
    temp_decay = 0.95
    std_coarse = 0.1
    std_fine = 0.05

    def objective(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * (2.0 / len(h))

    # Initialize coarse grid
    if 'initial_h_values' in globals():
        coarse_h = initial_h_values[::2]
    else:
        base = np.zeros(n_coarse)
        base[:17] = 1.0
        base[17] = 0.5
        base = np.convolve(base, np.ones(5)/5, mode='same')
        sum_base = np.sum(base)
        coarse_h = base * (n_coarse / 2) / sum_base

    # Simulated Annealing
    def sa(initial, n_points, temp, decay, std, max_iter, start_time):
        h = initial.copy()
        best_h = h.copy()
        best_c5 = objective(h)
        current_temp = temp
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                return best_h, best_c5
            noise = np.random.normal(0, std, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                continue

            current_c5 = objective(h)
            delta = current_c5 - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / current_temp):
                best_h = h.copy()
                best_c5 = current_c5
            current_temp *= decay
        return best_h, best_c5

    start_time = time.time()
    coarse_h, coarse_c5 = sa(coarse_h, n_coarse, initial_temp, temp_decay, std_coarse, max_iter, start_time)
    if time.time() - start_time > budget_s:
        return coarse_h, coarse_c5, n_coarse

    # Interpolate to fine grid
    x_coarse = np.linspace(0, 2, n_coarse)
