import numpy as np
import time
import scipy.optimize as opt
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    initial_h_sum = np.sum(initial_h)
    initial_h_scaled = initial_h * (ones_count / initial_h_sum) if initial_h_sum != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = initial_h_scaled.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    population_size = 50
    population = [initial_h_scaled]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        population.append(h * (ones_count / h_sum))
    
    F = 0.8
    CR = 0.9
    mutation_rate = 0.3
    max_stagnation = 100
    stagnation = 0
    
    while time.time() - start_time < budget_s:
        for _ in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[_])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            new_c5 = objective(u_scaled)
            if new_c5 < best_c5:
                best_h = u_scaled.copy()
                best_c5 = new_c5
                stagnation = 0
            else:
                stagnation += 1
            if stagnation > max_stagnation:
                F = max(0.5, F - 0.02)
                CR = min(0.99, CR + 0.02)
                stagnation = 0
        if time.time() - start_time >= budget_s:
            break
    
    return best_h, best_c5, n_points
