import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_coarse = 70
    k = 4
    n_fine = n_coarse * k
    
    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n)
    
    def calc_gradient(h, lag, n):
        grad = np.zeros(n)
        for i in range(n):
            j = i + lag
            if 0 <= j < n:
                grad[i] += (1 - h[j])
                grad[j] -= h[i]
        return grad
    
    best_h = h_coarse.copy()
