import time
import cvxopt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

    else:

        h = h * (ones_count / h_sum) if h_sum > 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
    
    def compute_overlaps(h):
        dx = 2.0 / n_points
        overlaps = []
        for k in range(n_points):
            shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            overlaps.append(overlap)
        return overlaps
    
    best_c5 = max(compute_overlaps(h))
    start_time = time.time()
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        overlaps = compute_overlaps(h)
        max_overlap = max(overlaps)
        best_k = overlaps.index(max_overlap)
        
        Q = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = (i + best_k) % n_points
            Q[i][j] = 1.0
        P = -2.0 * Q
        q = np.zeros(n_points)
        A = np.ones((1, n_points))
        b = np.array([n_points / 2.0])
        
        G = np.zeros((2 * n_points, n_points))
        h_vec = np.zeros(2 * n_points)
        for i in range(n_points):
            G[i][i] = -1.0
            h_vec[i] = 0.0
            G[i + n_points][i] = 1.0
            h_vec[i + n_points] = 1.0
        
        try:
            P_cvp = cvxopt.matrix(P)
            q_cvp = cvxopt.matrix(q)
            G_cvp = cvxopt.matrix(G)
            h_vec_cvp = cvxopt.matrix(h_vec)
            A_cvp = cvxopt.matrix(A)
            b_cvp = cvxopt.matrix(b)
            sol = cvxopt.solvers.qp(P_cvp, q_cvp, G_cvp, h_vec_cvp, A_cvp, b_cvp)
        except:
