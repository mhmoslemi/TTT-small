import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()

        h = np.random.rand(n_points)
        h = h / np.sum(h) * ones_count

    result = minimize(
        objective,
        h,
        method='SLSQP',
