            velocity_new = w * velocity + c1 * r1 * (personal_best[i] - h) + c2 * r2 * (global_best - h)
            velocity_new = np.clip(velocity_new, -1.0, 1.0)
            velocities[i] = velocity_new
            h_new = h + velocity_new
            h_new = np.clip(h_new, 0.0, 1.0)
            s_new = np.sum(h_new)
            if s_new == 0:
                h_new = np.zeros(n_points)
            else:
                h_new = h_new * (ones_count / s_new)
            particles[i] = h_new.copy()
            c5_new = evaluate_erdos_solution(particles[i])
            if c5_new < evaluate_erdos_solution(personal_best[i]):
                personal_best[i] = particles[i].copy()
                if c5_new < best_c5:
                    best_c5 = c5_new
                    global_best = particles[i].copy()
    s_global = np.sum(global_best)
    if s_global != 0:
        global_best = global_best * (ones_count / s_global)
    return global_best, best_c5, n_points
