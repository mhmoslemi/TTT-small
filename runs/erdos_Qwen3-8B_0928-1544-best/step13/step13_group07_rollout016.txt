import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    max_iterations = 1000
    velocity = np.zeros(n_points)
    momentum = 0.9
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        m = idx_max - (n_points - 1)
        gradient = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + m < n_points:
                gradient[j] += 1 - h[j + m]
            if 0 <= j - m < n_points:
                gradient[j] -= h[j - m]
        velocity = momentum * velocity + gradient
        best_alpha = 0.0
        best_new_c5 = best_c5
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.001]:
            h_candidate = h + alpha * velocity
            h_candidate = np.clip(h_candidate, 0.0, 1.0)
            h_candidate_sum = np.sum(h_candidate)
            if h_candidate_sum == 0:
                continue
            h_candidate = h_candidate * (n_points / 2) / h_candidate_sum
            corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            new_c5 = np.max(corr_candidate) * dx
            if new_c5 < best_new_c5:
                best_alpha = alpha
                best_new_c5 = new_c5
        if best_alpha > 0:
            h = h + best_alpha * velocity
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (n_points / 2) / h_sum
            corr_new = np.correlate(h, 1 - h, mode='full')
            new_c5 = np.max(corr_new) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    return best_h, best_c5, n_points
