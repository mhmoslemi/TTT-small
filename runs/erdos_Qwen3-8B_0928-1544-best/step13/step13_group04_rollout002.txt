import time

def run(seed=42, budget_s=1100, **kwargs):

    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)

    for _ in range(19):
        h = np.random.rand(n_points)
        h = (h * (ones_count / np.sum(h))).astype(np.float64)
        particles.append(h)
    velocities = [np.random.uniform(-1.0, 1.0, n_points) for _ in range(20)]
    pbest = particles[:]
    pbest_values = [compute_C5(p) * (2.0 / n_points) for p in particles]
    gbest = min(pbest, key=lambda p: compute_C5(p) * (2.0 / n_points))
    gbest_value = min(pbest_values)
    start_time = time.time()
    max_iter = 1000
    omega = 0.9
    c1, c2 = 2.0, 2.0
    vel_limits = np.array([[-1.0, 1.0]] * n_points)
    no_improve = 0
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(20):
            p = particles[i]

            v = omega * v + c1 * r1 * (pbest_i - p) + c2 * r2 * (gbest - p)

            new_p = p + v
            new_p = np.clip(new_p, 0.0, 1.0)
            new_p_sum = np.sum(new_p)
            if new_p_sum == 0:
                new_p = np.zeros(n_points)
            else:
                new_p = new_p * (ones_count / new_p_sum)

            if new_c5 < compute_C5(p) * (2.0 / n_points):
                particles[i] = new_p

def compute_C5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr)
