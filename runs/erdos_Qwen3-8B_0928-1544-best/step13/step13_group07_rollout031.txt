            if 0 <= j - m < n_points:
                gradient[j] -= h[j - m]
        best_alpha = 0.0
        best_new_c5 = best_c5
        alpha = 1.0
        while alpha > 1e-5:
            h_candidate = h + alpha * (-gradient)
            h_candidate = np.clip(h_candidate, 0.0, 1.0)
            h_candidate_sum = np.sum(h_candidate)
            if h_candidate_sum == 0:
                alpha /= 2
                continue
            h_candidate = h_candidate * (n_points / 2) / h_candidate_sum
            corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            new_c5 = np.max(corr_candidate) * dx
            if new_c5 < best_new_c5:
                best_alpha = alpha
                best_new_c5 = new_c5
                alpha *= 1.5  # Exploit better directions
            else:
                alpha *= 0.5
        if best_alpha > 0:
            h = h + best_alpha * (-gradient)
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (n_points / 2) / h_sum
            corr_new = np.correlate(h, 1 - h, mode='full')
            new_c5 = np.max(corr_new) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        if _ % 50 == 0:
            perturb = np.random.normal(0, 0.01, n_points)
            h += perturb
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (n_points / 2) / h_sum
    return best_h, best_c5, n_points
