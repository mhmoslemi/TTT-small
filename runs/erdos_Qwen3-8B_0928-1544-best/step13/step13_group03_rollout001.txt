import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    ones_count = n_points // 2
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        # Create a symmetric initial pattern: 1s in middle 0.5 of the domain, smooth with a moving average
        pattern = np.zeros(n_points)
        middle_start = n_points // 2 - n_points // 4
        middle_end = n_points // 2 + n_points // 4
        pattern[middle_start:middle_end] = 1.0
        # Smooth with a moving average
        window_size = 5
        for i in range(window_size // 2, n_points - window_size // 2):
            pattern[i] = np.mean(pattern[i - window_size // 2:i + window_size // 2])
        initial_h = pattern

    # Normalize and clip
    initial_sum = np.sum(initial_h)
    initial_h = initial_h * (ones_count / initial_sum)
    initial_h = np.clip(initial_h, 0.0, 1.0)

    def objective(h_vec):
        h = h_vec.copy()
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    # Constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - ones_count})

    result = minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
    )

    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
