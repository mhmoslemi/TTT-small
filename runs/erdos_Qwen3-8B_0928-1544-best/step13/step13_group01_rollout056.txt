import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.random.rand(n_points)
        initial_h_sum = np.sum(initial_h)
        initial_h = initial_h * (ones_count / initial_h_sum)
    population_size = 200
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points)
        h = h * (ones_count / h_sum)
        population.append(h)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(initial_h)
    start_time = time.time()
    F = 0.8
    CR = 0.9
    stagnation = 0

    def mutate(parent, a, b, c):
        mutant = population[a] + F * (population[b] - population[c])
        trial = parent.copy()
        for i in range(n_points):
            if np.random.rand() < CR:
                trial[i] = mutant[i]
        trial = np.clip(trial, 0.0, 1.0)
        trial_sum = np.sum(trial)
        if trial_sum == 0:
            return None
        trial = trial * (ones_count / trial_sum)
        return trial

    while time.time() - start_time < budget_s:
        for _ in range(population_size):
            a, b, c = random.sample(range(population_size), 3)
            trial = mutate(population[_], a, b, c)
            if trial is None:
                continue
            new_c5 = compute_c5(trial)
            if new_c5 < compute_c5(population[_]):
                population[_] = trial
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = trial
                    stagnation = 0
            else:
                stagnation += 1
        if stagnation > 500:
            F = max(0.8, F - 0.01)
            CR = min(0.9, CR + 0.01)
            stagnation = 0
    return best_h, best_c5, n_points
