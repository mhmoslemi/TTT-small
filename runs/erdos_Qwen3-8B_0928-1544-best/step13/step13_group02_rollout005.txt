import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def scale_h(h):
        return h * (ones_count / np.sum(h))

    # Coarse phase: n_points=35
    coarse_n_points = 35
    coarse_ones_count = coarse_n_points // 2
    if 'initial_h_values' in globals():
        coarse_h = initial_h_values[:coarse_n_points]
    else:
        coarse_h = np.random.rand(coarse_n_points)
        coarse_h = scale_h(coarse_h)

    # Coarse GA
    pop_size = 50
    pop = [coarse_h.copy() for _ in range(pop_size)]
    max_iter = 100
    for _ in range(max_iter):
        for _ in range(pop_size):
            a, b, c = np.random.choice(range(pop_size), 3, replace=False)
            h1, h2, h3 = pop[a], pop[b], pop[c]
            h = h1 + np.random.rand(coarse_n_points) * (h2 - h3)
            h = np.clip(h, 0, 1)
            h = scale_h(h)
            fit = objective(h)
            if fit < objective(pop[_]):
                pop[_] = h.copy()
    best_coarse = min(pop, key=objective)

    # Upsample to 70 points
    fine_h = np.repeat(best_coarse, 2)
    fine_h = scale_h(fine_h)

    # Fine phase: Gradient descent
    steps = 1000
    step_size = 0.1
    for _ in range(steps):
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = fine_h.copy()
            h_plus[i] += step_size
            h_plus = scale_h(h_plus)
            h_minus = fine_h.copy()
            h_minus[i] -= step_size
            h_minus = scale_h(h_minus)
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * step_size)
        fine_h = fine_h - grad * 0.01
        fine_h = np.clip(fine_h, 0, 1)
        fine_h = scale_h(fine_h)

    best_c5 = objective(fine_h)
    return fine_h, best_c5, n_points
