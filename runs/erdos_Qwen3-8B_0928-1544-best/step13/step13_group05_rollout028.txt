import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 100
        initial_h = np.random.rand(n_points) * 0.5
        initial_h_sum = initial_h.sum()
        initial_h = initial_h * (n_points // 2) / initial_h_sum
    ones_count = n_points // 2
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def rescale(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return h * (ones_count / s)
    
    population = [rescale(initial_h.copy()) for _ in range(50)]
    best_h = population[0]
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for _ in range(50):
            idx = np.random.randint(len(population))
            h = population[idx].copy()
            noise = np.random.normal(0, 0.1, n_points)
            h += noise
            h = rescale(h)
            new_pop.append(h)
        new_evals = [objective(h) for h in new_pop]
        best_idx = np.argmin(new_evals)
        if new_evals[best_idx] < best_c5:
            best_h = new_pop[best_idx]
            best_c5 = new_evals[best_idx]
        population = [h for h in new_pop]
    
    result = opt.minimize(lambda x: objective(x), best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, options={'maxiter': 50})
    best_h = result.x
    best_h = rescale(best_h)
    return best_h, objective(best_h), n_points
