import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * ones_count
        h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial / np.sum(h_initial) * ones_count

    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / n_points)

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - ones_count}]
    bounds = [(0, 1) for _ in range(n_points)]

    result = minimize(objective, h_initial, method='COBYLA', bounds=bounds, constraints=cons, options={'maxiter': 500})
    if result.success:
        best_h = result.x
    else:
        best_h = h_initial.copy()

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * ones_count
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
