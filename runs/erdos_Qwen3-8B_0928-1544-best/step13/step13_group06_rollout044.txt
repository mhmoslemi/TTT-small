import numpy as np
from scipy.optimize import minimize, LinearConstraint
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2  # 35
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * ones_count
        h = np.clip(h, 0, 1)
    
    def objective(h_vec):
        h = np.clip(h_vec, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = LinearConstraint(np.ones(n_points), [35.0], [35.0])
    
    start_time = time.time()
    result = minimize(
        objective,
        h,
        method='COBYLA',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'disp': False}
    )
    
    if time.time() - start_time > budget_s:
        best_h = h
    else:
        best_h = result.x
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * ones_count
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
