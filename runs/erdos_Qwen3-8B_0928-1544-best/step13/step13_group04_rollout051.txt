            new_c5 = compute_C5(particles[i])
            if new_c5 < pbest_c5[i]:
                personal_best[i] = particles[i].copy()
                pbest_c5[i] = new_c5
            if new_c5 < global_best_c5:
                global_best = particles[i].copy()
                global_best_c5 = new_c5
                best_h = global_best.copy()
                best_c5 = global_best_c5
        omega *= 0.99
    h = global_best
    h_sum = np.sum(h)
    if h_sum == 0:
        h = np.zeros(n_points)
    else:
        h = h * (ones_count / h_sum)
    return h, best_c5, n_points
