        res = opt.minimize(obj_func, x0, bounds=bounds, constraints=cons, method='SLSQP')
        if res.success:
            h[:n_left] = res.x

        # Optimize second half
        fixed_first = h[:n_left]
        required_sum_second = S_total - np.sum(fixed_first)
        x0 = h[n_left:]

        bounds = [(0.0, 1.0) for _ in range(n_points - n_left)]

        def cons_sum(x):
            return np.sum(x) - required_sum_second

        cons = [{'type': 'eq', 'fun': cons_sum}]

        def obj_func(x):
            h_opt = np.concatenate((fixed_first, x))
            return compute_overlap(h_opt)

        res = opt.minimize(obj_func, x0, bounds=bounds, constraints=cons, method='SLSQP')
        if res.success:
            h[n_left:] = res.x

        current_c5 = compute_overlap(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    return best_h, best_c5, n_points
