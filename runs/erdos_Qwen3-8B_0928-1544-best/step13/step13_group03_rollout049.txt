import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    desired_sum = n_points // 2  # 70

    # Initialize h
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        # Resample to 140 points
        x = np.linspace(0, 2, len(initial_h))
        x_new = np.linspace(0, 2, n_points)
        h_resampled = np.interp(x_new, x, initial_h)
        h_resampled = np.clip(h_resampled, 0, 1)
        sum_h = np.sum(h_resampled)
        if sum_h != 0:
            h_resampled = h_resampled * (desired_sum / sum_h)
        initial_h = h_resampled
    else:
        # Create symmetric pattern
        initial_h = np.zeros(n_points)
        initial_h[:n_points//4] = 1
        initial_h[n_points//2:3*n_points//4] = 1
        # Smooth the initial pattern
        kernel = np.ones(5) / 5
        initial_h = np.convolve(initial_h, kernel, mode='same')

    # Define the objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    # Define constraints
    constraints = [
        {'type': 'eq', 'fun': lambda h: np.sum(h) - desired_sum}
    ]

    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization
    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 300, 'ftol': 1e-8})

    # Extract the best solution
    best_h = result.x
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
