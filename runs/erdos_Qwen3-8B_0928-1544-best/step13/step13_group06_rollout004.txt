import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2  # 35
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
    h = h / np.sum(h) * ones_count
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    start_time = time.time()
    
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    method = 'COBYLA'
    options = {'maxiter': 1000, 'disp': False}
    res = optimize.minimize(objective, h, method=method, bounds=bounds, constraints=constraints, options=options)
    
    if time.time() - start_time > budget_s:
        return best_h, best_c5, n_points
    
    h_opt = res.x
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt / np.sum(h_opt) * ones_count
    current_c5 = np.max(np.correlate(h_opt, 1 - h_opt, mode='full')) * (2.0 / n_points)
