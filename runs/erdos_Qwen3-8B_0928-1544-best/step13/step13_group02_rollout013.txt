import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    h_sum = np.sum(h)
    h_scaled = h * (ones_count / h_sum)
    population = [h_scaled]
    population_size = 150
    F_start, F_end = 0.6, 0.9
    CR_start, CR_end = 0.9, 1.0
    mutation_rate = 0.25
    initial_population = [np.copy(h_scaled) for _ in range(population_size - 1)]
    
    for _ in range(population_size - 1):
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h_scaled = h * (ones_count / h_sum)
        initial_population.append(h_scaled)
    
    population.extend(initial_population)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()
    F = F_start
    CR = CR_start
    stagnation = 0
    
    while time.time() - start_time < budget_s:
        for _ in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[_])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            new_c5 = objective(u_scaled)
            if new_c5 < objective(population[_]):
                population[_] = u_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u_scaled.copy()
                    stagnation = 0
            else:
                stagnation += 1
            if stagnation > 50:
                F = max(F_start, F - 0.01)
                CR = min(CR_end, CR + 0.01)
                stagnation = 0
    
    return best_h, best_c5, n_points
