        personal_best_c5.append(compute_C5(h_scaled))
    else:
        for _ in range(num_particles):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            h_scaled = h * (ones_count / h_sum)
            particles.append(h_scaled)
            velocity = np.random.uniform(-velocity_limit, velocity_limit, size=n_points)
            velocities.append(velocity)
            personal_best.append(h_scaled)
            personal_best_c5.append(compute_C5(h_scaled))

    global_best = None
    global_best_c5 = float('inf')
    for i in range(len(particles)):
        if personal_best_c5[i] < global_best_c5:
            global_best = particles[i].copy()
            global_best_c5 = personal_best_c5[i]

    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        for i in range(len(particles)):
            r1 = np.random.rand()
            r2 = np.random.rand()
            v = velocities[i].copy()
            pbest = personal_best[i].copy()
            gbest = global_best.copy()
            v = omega * v + c1 * r1 * (pbest - particles[i]) + c2 * r2 * (gbest - particles[i])
            v = np.clip(v, -velocity_limit, velocity_limit)
            new_pos = particles[i] + v
            new_pos = np.clip(new_pos, 0.0, 1.0)
            new_pos_sum = np.sum(new_pos)
            if new_pos_sum == 0:
                continue
            new_pos_scaled = new_pos * (ones_count / new_pos_sum)
            new_c5 = compute_C5(new_pos_scaled)
            if new_c5 < personal_best_c5[i]:
                personal_best[i] = new_pos_scaled.copy()
                personal_best_c5[i] = new_c5
                if new_c5 < global_best_c5:
                    global_best = new_pos_scaled.copy()
                    global_best_c5 = new_c5
            particles[i] = new_pos_scaled.copy()
            velocities[i] = v.copy()

    h_sum = np.sum(global_best)
    if h_sum == 0:
        global_best = np.zeros(n_points)
    else:
        global_best = global_best * (ones_count / h_sum)

    return global_best, global_best_c5, n_points
