import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points

    for _ in range(max_iterations):

            if 0 <= j + m < n_points:

                gradient[j] -= h[j - m]
        gradient = momentum * prev_grad + gradient
        prev_grad = gradient.copy()
