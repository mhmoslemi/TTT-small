        if initial_h_sum != 0:
            initial_h = initial_h * (target_sum / initial_h_sum)
    
    def objective(h):
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def generate_population(size=100):
        pop = [initial_h.copy()]
        for _ in range(size-1):
            h = np.random.rand(n_points)
            h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
            pop.append(h)
        return pop
    
    population = generate_population(100)
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    F = 0.8
    CR = 0.9
    mutation_rate = 0.25
    stagnation = 0
    max_stagnation = 100
    
    def evolve_population(pop, F, CR, mutation_rate):
        new_pop = []
        for _ in range(len(pop)):
            a, b, c = np.random.choice(len(pop), 3, replace=False)
            a_vec, b_vec, c_vec = pop[a], pop[b], pop[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(pop[_])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u = u * (target_sum / u_sum)
            new_c5 = objective(u)

                new_pop.append(u)
                if new_c5 < best_c5:
