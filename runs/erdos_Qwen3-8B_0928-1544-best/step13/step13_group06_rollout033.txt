import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    if 'initial_h_values' in globals():
        particles = [initial_h_values.copy()]
    else:
        particles = []
        for _ in range(50):
            h = np.random.rand(n_points)
            h = h / np.sum(h) * target_sum
            h = np.clip(h, 0, 1)
            particles.append(h)
    particles = np.array(particles)
    velocities = np.random.uniform(-1, 1, (50, n_points))
    pbest = particles.copy()
    pbest_fitness = np.array([evaluate_erdos_solution(p) for p in particles])
    gbest = particles[np.argmin(pbest_fitness)]
    gbest_fitness = np.min(pbest_fitness)
    start_time = time.time()
    max_iter = 100000
    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        omega = 0.9 - (0.9 - 0.4) * (t / max_iter)
        r1 = np.random.rand(50, n_points)
        r2 = np.random.rand(50, n_points)
        velocities = omega * velocities + 2.0 * r1 * (pbest - particles) + 2.0 * r2 * (gbest - particles)
        velocities = np.clip(velocities, -2, 2)
        particles += velocities
        particles = np.clip(particles, 0, 1)
        for i in range(50):
            s = np.sum(particles[i])
            if s > 0:
                particles[i] = particles[i] * (target_sum / s)
            else:
                particles[i] = np.full(n_points, target_sum / n_points)
