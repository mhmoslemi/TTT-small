import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum)

    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    learning_rate = 0.05
    max_iterations = int(budget_s / 0.1)  # Adjust for time

    for _ in range(max_iterations):
        max_c5 = -np.inf
        best_k = 0
        for k in range(n_points):
            shifted = np.roll(h, k)
            current_c = np.dot(h, 1 - shifted) * (2.0 / n_points)
            if current_c > max_c5:
                max_c5 = current_c
                best_k = k

        idx_plus = (np.arange(n_points) + best_k) % n_points
        idx_minus = (np.arange(n_points) - best_k) % n_points
        gradient = (1 - h[idx_plus]) - h[idx_minus]

        h_new = h - learning_rate * gradient
        h_new = np.clip(h_new, 0.0, 1.0)
        h_sum = np.sum(h_new)
        if h_sum != 0:
            h = h_new * (ones_count / h_sum)
        else:
            h = h_new

        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()

        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n_points
