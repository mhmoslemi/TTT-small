import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    dx = 2.0 / n_points

    n_particles = 50
    particles = []
    velocities = []

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h = np.clip(initial_h, 0, 1)
        sum_initial = np.sum(initial_h)
        if sum_initial > 1e-8:
            initial_h = initial_h * ones_count / sum_initial
        particles.append(initial_h)
        velocities.append(np.zeros(n_points))
    else:
        for _ in range(n_particles):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 1e-8:
                h = h * ones_count / sum_h
            particles.append(h)
            velocities.append(np.random.uniform(-1, 1, n_points))

    pbest = [np.copy(p) for p in particles]
    pbest_fitness = [0.0] * n_particles
    for i in range(n_particles):
        corr = np.correlate(particles[i], 1 - particles[i], mode='full')
        pbest_fitness[i] = np.max(corr) * dx

    gbest_idx = np.argmin(pbest_fitness)
    gbest = np.copy(pbest[gbest_idx])
    gbest_fitness = pbest_fitness[gbest_idx]

    start_time = time.time()

    for iteration in range(100000):
        if time.time() - start_time > budget_s:
            break
        inertia = 0.9 - (iteration / 100000) * 0.5
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = inertia * velocities[i] + 2.0 * r1 * (pbest[i] - particles[i]) + 2.0 * r2 * (gbest - particles[i])
