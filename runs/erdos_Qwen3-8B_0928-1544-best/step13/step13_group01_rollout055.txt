import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h0 = initial_h_values
    else:
        h0 = np.random.rand(n_points) * 0.5
        h0_sum = np.sum(h0)
        h0 = h0 * (ones_count / h0_sum)
    population = [h0]
    population_size = 200
    F_start, F_end = 0.2, 0.9
    CR_start, CR_end = 0.7, 0.99
    mutation_rate = 0.1
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()
    F = F_start
    CR = CR_start
    stagnation = 0
    
    for _ in range(population_size - 1):
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h = h * (ones_count / h_sum)
        population.append(h)
    
    while time.time() - start_time < budget_s:
        for _ in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            v = population[a] + F * (population[b] - population[c])
            u = np.copy(population[_])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
