        h = initial_h_values
    else:
        h = np.random.rand(n_points) * 0.5
    # Normalize to ensure sum constraint
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum == 0:
        h = np.random.rand(n_points) * 0.5
    h = h * (ones_count / h_sum)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    dx = 2.0 / n_points
    learning_rate = 0.05
    max_iterations = int(budget_s / 10)
    start_time = time.time()
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Compute maximum C5
        max_c5 = 0.0
        active_k = 0
        for k in range(n_points):
            shifted = np.roll(h, k)
            dot = np.dot(h, shifted)
            c_k = dx * (h_sum - dot)
            if c_k > max_c5:
                max_c5 = c_k
                active_k = k
        # Compute gradient
        grad = 1.0 - np.roll(h, active_k) - np.roll(h, -active_k)
        # Update h
        h = h - learning_rate * grad
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            continue
        h = h * (ones_count / h_sum)
        h = np.clip(h, 0, 1)
    # Final evaluation
    max_c5 = 0.0
    h_sum = np.sum(h)
    dx = 2.0 / n_points
    for k in range(n_points):
        shifted = np.roll(h, k)
        dot = np.dot(h, shifted)
        c_k = dx * (h_sum - dot)
        if c_k > max_c5:
            max_c5 = c_k
    return h, max_c5, n_points
