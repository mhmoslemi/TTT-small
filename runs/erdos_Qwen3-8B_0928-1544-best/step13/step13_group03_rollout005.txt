import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 100
    mutation_rate = 0.2
    mutation_std = 0.1
    crossover_rate = 0.8
    max_generations = 200

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        population = [initial_h]
        for _ in range(population_size - 1):
            perturbed = initial_h + np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            sum_p = np.sum(perturbed)
            if sum_p == 0:
                perturbed = np.random.rand(n_points) * 0.5
            else:
                perturbed = perturbed * (ones_count / sum_p)
            population.append(perturbed)
    else:
        population = [np.random.rand(n_points) for _ in range(population_size)]
        for i in range(population_size):
            sum_p = np.sum(population[i])
            population[i] = population[i] * (ones_count / sum_p)

    def objective(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = float('inf')

    start_time = time.time()
    for _ in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        fitness = [objective(indiv) for indiv in population]
        sorted_indices = np.argsort(fitness)
        top_half = [population[i] for i in sorted_indices[:population_size // 2]]
        new_population = []
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(top_half, 2, replace=False)
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points)
                child = np.concatenate([parent1[:split], parent2[split:]])
            else:
                child = parent1.copy()
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_std, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                sum_c = np.sum(child)
                if sum_c == 0:
                    child = np.random.rand(n_points) * 0.5
                else:
                    child = child * (ones_count / sum_c)
            new_population.append(child)
        population = new_population
        for indiv in population:
            c5 = objective(indiv)
            if c5 < best_c5:
                best_c5 = c5
                best_h = indiv.copy()
    return best_h, best_c5, n_points
