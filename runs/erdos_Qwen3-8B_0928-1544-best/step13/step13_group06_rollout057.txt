                pbest_positions[i] = particles[i].copy()
                pbest_fitness[i] = current_fitness

            if current_fitness < gbest_fitness:
                gbest_position = particles[i].copy()
                gbest_fitness = current_fitness
                best_h = gbest_position.copy()
                best_c5 = gbest_fitness

    # Final projection
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best == 0:
        best_h = np.random.rand(n_points)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
    else:
        best_h = best_h * target_sum / sum_best

    return best_h, best_c5, n_points
