            j_minus_k = (j - k) % n_points

        h -= learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            continue
        h *= ones_count / h_sum
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        learning_rate *= decay

    return best_h, best_c5, n_points
