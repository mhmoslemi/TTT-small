import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    initial_sum = np.sum(initial_h)
    if initial_sum != 0:
        initial_h = initial_h * (ones_count / initial_sum)
    h_prev = initial_h.copy()
    dx = 2.0 / n_points
    max_iter = 100
    best_c5 = float('inf')
    best_h = h_prev.copy()
    
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        corr = np.correlate(h_prev, 1 - h_prev, mode='full')
        I_k_prev = []
        grad_k_prev = []
        shifts = np.arange(2 * n_points - 1)
        for k in shifts:
            I_k = corr[k] * dx
            I_k_prev.append(I_k)
            grad_k = np.zeros(n_points)
            for j in range(n_points):
                if j + k < n_points:
                    term1 = 1.0 - h_prev[j + k]
                else:
                    term1 = 0.0
                if j - k >= 0:
                    term2 = h_prev[j - k]
                else:
                    term2 = 0.0
                grad_k[j] = term1 - term2
            grad_k_prev.append(grad_k)
        h = cp.Variable(n_points)
        C = cp.Variable()
        constraints = []
        for k in range(len(shifts)):
            rhs = np.dot(grad_k_prev[k], h_prev) - I_k_prev[k]
            constraints.append(cp.sum(grad_k_prev[k] * h) <= C + rhs)
        for j in range(n_points):
            constraints.append(h[j] >= 0.0)
            constraints.append(h[j] <= 1.0)
        constraints.append(cp.sum(h) == ones_count)
        prob = cp.Problem(cp.Minimize(C), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_prev = h.value
                current_c5 = C.value
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_prev.copy()
        except:
            pass
    return best_h, best_c5, n_points
