import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 100
    ones_count = n_points // 2
    population_size = 200
    w = 0.9
    c1 = 2.0
    c2 = 2.5
    max_v = 0.2
    max_iterations = 800
    stagnation_threshold = 50

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init_sum = np.sum(h_init)
        h_init_scaled = h_init * (ones_count / h_init_sum)
    else:
        h_init = np.random.rand(n_points)
        h_init_sum = np.sum(h_init)
        h_init_scaled = h_init * (ones_count / h_init_sum)

    population = [h_init_scaled.copy()]
    for _ in range(population_size - 1):
        noise = np.random.uniform(0.9, 1.1, n_points)
        h = population[0] * noise
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h_scaled = h * (ones_count / h_sum)
        population.append(h_scaled)

    velocities = [np.zeros(n_points) for _ in range(population_size)]
    personal_best = [p.copy() for p in population]
    personal_best_values = [objective(p) for p in population]
    global_best = population[0].copy()
    global_best_value = objective(population[0])
    best_h = global_best.copy()
    best_c5 = global_best_value
    start_time = time.time()

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            new_velocity = velocities[i] + w * velocities[i] + c1 * r1 * (personal_best[i] - population[i]) + c2 * r2 * (global_best - population[i])
            new_velocity = np.clip(new_velocity, -max_v, max_v)
            velocities[i] = new_velocity
            new_pos = population[i] + new_velocity
            new_pos = np.clip(new_pos, 0, 1)
            new_pos_sum = np.sum(new_pos)
            if new_pos_sum == 0:
                continue
            new_pos_scaled = new_pos * (ones_count / new_pos_sum)
            population[i] = new_pos_scaled
            new_value = objective(population[i])
            if new_value < personal_best_values[i]:
                personal_best[i] = population[i].copy()
                personal_best_values[i] = new_value
                if new_value < global_best_value:
                    global_best = population[i].copy()
                    global_best_value = new_value
                    best_h = global_best.copy()
                    best_c5 = global_best_value
        if iteration % 50 == 0 and iteration > 0:
            if np.max(np.abs(np.array(personal_best_values) - best_c5)) < 1e-6:
                max_v += 0.02
                if max_v > 0.2:
                    max_v = 0.2
                num_replace = int(population_size * 0.1)
                indices = np.argsort(personal_best_values)[-num_replace:]
                for idx in indices:
                    noise = np.random.uniform(0.9, 1.1, n_points)
