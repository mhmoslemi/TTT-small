import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2

    def objective(h):
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            return float('inf')
        h_scaled = h * (ones_count / h_sum)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(objective, bounds, popsize=30, maxiter=200, tol=1e-6, mutation=(0.7, 0.9), recombination=0.9, seed=seed, callback=lambda x: None)
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
