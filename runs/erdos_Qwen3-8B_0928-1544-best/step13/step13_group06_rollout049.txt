        if time.time() - start_time > budget_s:
            break
        current_inertia = inertia_start - iter * 0.001
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * current_inertia + 2.0 * r1 * (pbest[i] - particles[i]) + 2.0 * r2 * (gbest - particles[i])
            vel = np.clip(vel, -2.0 * (2.0 / n_points), 2.0 * (2.0 / n_points))
            velocities[i] = vel
            particles[i] += vel
            particles[i] = np.clip(particles[i], 0, 1)
            sum_p = np.sum(particles[i])
            if sum_p != target_sum:
                scale = target_sum / sum_p
                particles[i] *= scale
                particles[i] = np.clip(particles[i], 0, 1)
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            fitness = np.max(corr) * (2.0 / n_points)
            if fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness
            if fitness < gbest_fitness:
                gbest = particles[i].copy()
                gbest_fitness = fitness
    best_h = gbest
    best_c5 = gbest_fitness
    return best_h, best_c5, n_points
