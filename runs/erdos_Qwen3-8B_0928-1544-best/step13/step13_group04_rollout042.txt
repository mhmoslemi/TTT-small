        if time.time() - start_time > budget_s:
            break
        for i in range(len(particles)):
            r1, r2 = np.random.rand(), np.random.rand()
            current = particles[i]
            pbest_i = pbest[i]
            gbest = global_best
            vel = omega * velocities[i] + c1 * r1 * (pbest_i - current) + c2 * r2 * (gbest - current)
            vel = np.clip(vel, -1.0, 1.0)
            velocities[i] = vel
            new_pos = current + vel
            new_pos = np.clip(new_pos, 0.0, 1.0)
            sum_h = np.sum(new_pos)
            if sum_h == 0:
                new_pos = np.full(n_points, 0.5)
            else:
                new_pos = new_pos * (ones_count / sum_h)
            particles[i] = new_pos.copy()
            new_c5 = compute_C5(particles[i])
            if new_c5 < best_c5[i]:
                best_c5[i] = new_c5
                pbest[i] = particles[i].copy()
            if new_c5 < best_c5[global_best_idx]:
                global_best_idx = i
                global_best = particles[i].copy()
                best_c5[global_best_idx] = new_c5
        omega *= 0.9995
    h_vals = global_best.copy()
    return h_vals, best_c5[global_best_idx], n_points
