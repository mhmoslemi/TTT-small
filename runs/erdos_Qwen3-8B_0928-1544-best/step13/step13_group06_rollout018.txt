        h[idx] = np.clip(h[idx], 0, 1)
        h = h / np.sum(h) * ones_count
        current_c5 = np.max(corr) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * ones_count
    return best_h, best_c5, n_points
