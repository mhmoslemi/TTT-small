def run(seed=42, budget_s=1100, **kwargs):

        if time.time() - start_time > budget_s:
            break
        t += 1
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        gradients = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + k
            i_minus_k = i - k
