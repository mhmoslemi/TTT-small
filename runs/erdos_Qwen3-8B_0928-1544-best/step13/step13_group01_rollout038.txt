import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h0 = initial_h_values
    else:
        h0 = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h0)
        h0 = h0 * (ones_count / h_sum)
    population = [h0]
    population_size = 200
    F_start, F_end = 0.9, 0.5
    CR_start, CR_end = 0.95, 0.6
    mutation_rate = 0.2
    tournament_size = 3
    stagnation = 0
    best_h = population[0].copy()
    best_c5 = np.inf
    start_time = time.time()
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    for _ in range(population_size - 1):
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h = h * (ones_count / h_sum)
        population.append(h)
    
    F = F_start
    CR = CR_start
    while time.time() - start_time < budget_s:
        for _ in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            v = population[a] + F * (population[b] - population[c])
            u = np.copy(population[_])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            new_c5 = objective(u_scaled)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = u_scaled.copy()
                stagnation = 0
            else:
                stagnation += 1
            if np.random.rand() < mutation_rate * 0.1:
                perturb = np.random.normal(0, 0.1, n_points)
                u_perturb = u_scaled + perturb
                u_perturb = np.clip(u_perturb, 0.0, 1.0)
                u_perturb_sum = np.sum(u_perturb)
                if u_perturb_sum != 0:
                    u_perturb = u_perturb * (ones_count / u_perturb_sum)
                    new_c5_perturb = objective(u_perturb)
                    if new_c5_perturb < best_c5:
                        best_c5 = new_c5_perturb
                        best_h = u_perturb.copy()
                        stagnation = 0
            if stagnation > 100:
                F = max(F_start, F - 0.02)
                CR = min(CR_end, CR + 0.02)
                stagnation = 0
        if time.time() - start_time >= budget_s:
            break
    return best_h, best_c5, n_points
