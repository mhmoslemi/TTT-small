import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 100
    ones_count = n_points // 2
    population_size = 50
    max_iterations = 500
    mutation_rate = 0.1
    tournament_size = 3
    sigma = 0.5
    max_v = 0.1

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)

        h = np.random.rand(n_points)
        h = h * (ones_count / np.sum(h))
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            return h
        return h * (ones_count / h_sum)

    population = [clip_and_scale(np.random.rand(n_points) * (ones_count / np.sum(np.random.rand(n_points)))) for _ in range(population_size)]
    best_h = population[0].copy()
    best_c5 = evaluate(best_h)

    start_time = time.time()
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate(ind) for ind in population]
        parents = []
        for _ in range(population_size // 2):
            idxs = np.random.choice(population_size, tournament_size, replace=False)
            candidates = [population[i] for i in idxs]
            best = min(candidates, key=lambda x: evaluate(x))
            parents.append(best)
        children = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1] if i+1 < len(parents) else parents[i]
            child1 = np.copy(p1)
            child2 = np.copy(p2)
            for j in range(n_points):

                    child1[j] = p2[j]
                    child2[j] = p1[j]
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
