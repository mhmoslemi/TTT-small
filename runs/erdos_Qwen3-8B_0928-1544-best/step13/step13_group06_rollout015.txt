    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    max_iterations = 600
    learning_rate = 0.05
    start_time = time.time()
    lambda_values = np.ones(2 * n_points - 1)
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        
        # Update lambda values for the current k
        current_overlap = corr[max_index]
        lambda_values[max_index] *= 1.1 if current_overlap > 0 else 1.0
        
        # Compute gradient for the current k
        derivatives = np.zeros(n_points)
        i_plus_k = (np.arange(n_points) + k) % n_points
        i_minus_k = (np.arange(n_points) - k) % n_points
        term1 = 1 - h[i_plus_k]
        term2 = h[i_minus_k]
        derivatives += term1 - term2
        
        # Apply gradient descent with lambda as a weight
        direction = np.sign(-derivatives)
        delta = learning_rate * direction
        h += delta
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * ones_count
        
        current_c5 = np.max(corr) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * ones_count
    return best_h, best_c5, n_points
