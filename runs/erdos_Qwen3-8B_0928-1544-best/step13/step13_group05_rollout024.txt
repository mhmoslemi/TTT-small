import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 120
    dx = 2.0 / n_points
    ones_count = n_points // 2
    population_size = 50
    w = 0.5
    c1 = 2.0
    c2 = 2.5
    max_v = 0.6
    max_iterations = 1000
    stagnation_threshold = 100
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    initial_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.random.rand(n_points)
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        perturb = np.random.uniform(0.9, 1.1, n_points)
        candidate = initial_h * perturb
        candidate = np.clip(candidate, 0, 1)
        candidate_sum = np.sum(candidate)
        candidate_scaled = candidate * (ones_count / candidate_sum)
        population.append(candidate_scaled)
    
    velocities = [np.zeros(n_points) for _ in range(population_size)]
    personal_best = [p.copy() for p in population]
    global_best = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            new_velocity = velocities[i] + w * velocities[i] + c1 * r1 * (personal_best[i] - population[i]) + c2 * r2 * (global_best - population[i])
            new_velocity = np.clip(new_velocity, -max_v, max_v)
            new_pos = population[i] + new_velocity
            new_pos = np.clip(new_pos, 0, 1)
            new_pos_sum = np.sum(new_pos)
            if new_pos_sum == 0:
                continue
            new_pos_scaled = new_pos * (ones_count / new_pos_sum)
            population[i] = new_pos_scaled
            new_val = objective(population[i])
            if new_val < objective(personal_best[i]):
                personal_best[i] = population[i].copy()
                if new_val < best_c5:
                    best_c5 = new_val
                    global_best = population[i].copy()
        if iteration % 50 == 0:
            if np.abs(best_c5 - objective(global_best)) < 1e-6:
                max_v += 0.05
                if max_v > 0.6:
                    max_v = 0.6
    
    best_h = np.clip(global_best, 0, 1)
    best_h_sum = np.sum(best_h)
    best_h_scaled = best_h * (ones_count / best_h_sum)
    final_c5 = objective(best_h_scaled)
    return best_h_scaled, final_c5, n_points
