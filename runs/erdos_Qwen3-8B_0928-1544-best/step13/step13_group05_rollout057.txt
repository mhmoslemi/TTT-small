import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 100
    ones_count = n_points // 2
    initial_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    initial_h = np.clip(initial_h, 0, 1)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraints(h):
        sum_h = np.sum(h)
        return [sum_h - n_points / 2]
    
    def optimize():
        res = opt.minimize(
            fun=objective,
            x0=initial_h,
            method='SLSQP',
            constraints={'type': 'eq', 'fun': constraints},
            bounds=[(0, 1) for _ in range(n_points)],
            options={
                'ftol': 1e-10,
                'maxiter': 100,
                'disp': False,
                'eps': 1e-8
            }
        )
        return res.x
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    
    while time.time() - start_time < budget_s - 60:
        h_candidate = optimize()
        h_candidate = np.clip(h_candidate, 0, 1)
        h_scaled = h_candidate * (ones_count / np.sum(h_candidate))
        c5 = objective(h_scaled)
        if c5 < best_c5:
            best_h = h_scaled
            best_c5 = c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    best_h_scaled = best_h * (ones_count / best_h_sum)
    best_c5 = objective(best_h_scaled)
    
    return best_h_scaled, best_c5, n_points
