        sum_fixed = np.sum(fixed_part)
        sum_left = (n_points / 2) - sum_fixed
        bounds = [(0.0, 1.0) for _ in range(n_points//2)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - sum_left})
        def obj_left(x):
            h_opt = np.concatenate([x, fixed_part])
            return compute_max_overlap(h_opt)
        initial_guess = h[:n_points//2].copy()
        result = optimize.minimize(obj_left, initial_guess, method='L-BFGS-B', bounds=bounds, constraints=cons)
        variables_opt = np.clip(result.x, 0.0, 1.0)
        sum_opt = np.sum(variables_opt)
        scaling_factor = sum_left / sum_opt if sum_opt != 0 else 1.0
        h[:n_points//2] = variables_opt * scaling_factor
        
        # Optimize right half
        fixed_part = h[:n_points//2]
        sum_fixed = np.sum(fixed_part)
        sum_right = (n_points / 2) - sum_fixed
        bounds = [(0.0, 1.0) for _ in range(n_points//2)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - sum_right})
        def obj_right(x):
            h_opt = np.concatenate([fixed_part, x])
            return compute_max_overlap(h_opt)
        initial_guess = h[n_points//2:].copy()
        result = optimize.minimize(obj_right, initial_guess, method='L-BFGS-B', bounds=bounds, constraints=cons)
        variables_opt = np.clip(result.x, 0.0, 1.0)
        sum_opt = np.sum(variables_opt)
        scaling_factor = sum_right / sum_opt if sum_opt != 0 else 1.0
        h[n_points//2:] = variables_opt * scaling_factor
        
        # Track best solution
        current_c5 = compute_max_overlap(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Final clamping and scaling
    h = np.clip(best_h, 0.0, 1.0)
    total_sum = np.sum(h)
    scaling_factor = (n_points / 2) / total_sum
    h = h * scaling_factor
    
    return h, best_c5, n_points
