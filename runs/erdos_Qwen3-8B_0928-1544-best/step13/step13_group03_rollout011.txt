        current_h = initial_h_values.copy()
    else:
        current_h = np.zeros(n_points)
        current_h[:n_points//2] = 1
    best_c5 = float('inf')
    best_h = current_h.copy()
    for _ in range(100):
        left = slice(0, n_points//2)
        init_left = current_h[left]
        bounds = [(0, 1)] * len(init_left)
        def obj_left(x):
            temp = current_h.copy()
            temp[left] = x
            corr = np.correlate(temp, 1 - temp, mode='full')
            return np.max(corr) * dx
        res = minimize(obj_left, init_left, bounds=bounds, method='L-BFGS-B', options={'maxiter': 20})
        if res.success:
            current_h[left] = res.x
        right = slice(n_points//2, n_points)
        init_right = current_h[right]
        bounds = [(0, 1)] * len(init_right)
        def obj_right(x):
            temp = current_h.copy()
            temp[right] = x
            corr = np.correlate(temp, 1 - temp, mode='full')
            return np.max(corr) * dx
        res = minimize(obj_right, init_right, bounds=bounds, method='L-BFGS-B', options={'maxiter': 20})
        if res.success:
            current_h[right] = res.x
        current_h = np.clip(current_h, 0, 1)
        current_sum = np.sum(current_h)
        if current_sum != 0:
            current_h *= target_sum / current_sum
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = current_h.copy()
    return best_h, best_c5, n_points
