import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    n = n_points
    target_sum = n / 2

    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n)
        h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else np.random.rand(n) * 0.5 * (target_sum / 0.5)

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    max_iterations = 1000  # Adjust based on time budget

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        # Compute overlaps for all shifts
        overlaps = []
        for k in range(n):
            shifted_h = np.roll(h, k)
            dot_product = np.dot(h, shifted_h)
            overlap = 1 - (dot_product * (2.0 / n))
            overlaps.append(overlap)

        # Find the shift with maximum overlap
        max_overlap = max(overlaps)
        k_max = overlaps.index(max_overlap)

        # Prepare permutation for this k_max
        perm = np.roll(np.arange(n), k_max)

        # Formulate QP to maximize dot product
        variables = cp.Variable(n)
        objective_expr = sum([variables[i] * variables[perm[i]] for i in range(n)])
        objective = cp.Maximize(objective_expr * (2.0 / n))
        constraints = [cp.sum(variables) == target_sum, 0 <= variables, variables <= 1]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == 'OPTIMAL':
                new_h = variables.value
                new_h = np.clip(new_h, 0, 1)
                sum_new = np.sum(new_h)

                    continue

                current_c5 = max_overlap

                    best_c5 = current_c5
                    best_h = new_h.copy()
                h = new_h

            continue
