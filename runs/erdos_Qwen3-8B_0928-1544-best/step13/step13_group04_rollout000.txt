                        simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
        
        for p in simplex:
            current_c5 = compute_C5(p)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = p.copy()
    
    return best_h, best_c5, n_points
