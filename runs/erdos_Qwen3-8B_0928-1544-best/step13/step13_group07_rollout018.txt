import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h_pop = [initial_h_values.copy()]
    else:
        h_pop = [np.random.rand(n_points) * 0.5]
        h_pop[0] = h_pop[0] * (n_points / 2) / np.sum(h_pop[0])
    
    def normalize_h(h):
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    best_h = h_pop[0].copy()
    best_c5 = np.max(np.correlate(h_pop[0], 1 - h_pop[0], mode='full')) * dx
    
    start_time = time.time()
    population_size = 50
    while len(h_pop) < population_size:
        h = np.random.rand(n_points) * 0.5
        h = normalize_h(h)
        h_pop.append(h)
    
    max_generations = 1000
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = []
        for h in h_pop:
            corr = np.correlate(h, 1 - h, mode='full')
            fitness.append(np.max(corr) * dx)
        # Select parents via tournament
        new_pop = []
        for _ in range(population_size):
            tournament = np.random.choice(h_pop, size=5, replace=False)
            winner = min(tournament, key=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx)
            new_pop.append(winner.copy())
        # Crossover and mutation
        for i in range(0, len(new_pop), 2):
            if i + 1 >= len(new_pop):
                break
            p1, p2 = new_pop[i], new_pop[i+1]
            # Crossover
            cp = np.random.randint(1, n_points)
            c1 = np.concatenate([p1[:cp], p2[cp:]])
            c2 = np.concatenate([p2[:cp], p1[cp:]])
            # Mutation
            mu = 0.05
            c1 = np.clip(c1 + np.random.normal(0, mu, size=n_points), 0, 1)
            c1 = normalize_h(c1)
            c2 = np.clip(c2 + np.random.normal(0, mu, size=n_points), 0, 1)
            c2 = normalize_h(c2)
            new_pop[i], new_pop[i+1] = c1, c2
        h_pop = new_pop
        # Update best solution
        for h in h_pop:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx

                best_c5 = c5
                best_h = h.copy()
