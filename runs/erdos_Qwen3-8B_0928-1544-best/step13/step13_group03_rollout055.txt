    best_fine = curr_fine.copy()
    best_c5_fine = compute_c5(curr_fine)
    temp_f = temp_fine

    for _ in range(iter_per_stage):
        if time.time() - time_start > max_time:
            break
        perturb = np.random.normal(0, perturb_fine, n_fine)
        new_h = curr_fine + perturb
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new == 0:
            continue
        new_h = new_h * (n_fine / 2) / sum_new
        new_c5 = compute_c5(new_h)
        delta = new_c5 - best_c5_fine
        if delta < 0 or np.random.rand() < np.exp(-delta / temp_f):
            curr_fine = new_h.copy()
            if new_c5 < best_c5_fine:
                best_c5_fine = new_c5
                best_fine = curr_fine.copy()
        temp_f *= cool_rate

    return best_fine, best_c5_fine, n_fine
