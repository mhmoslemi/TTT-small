import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        n_points = 100
        h = np.random.rand(n_points) * 0.5
        h = h * (n_points // 2) / np.sum(h)
    n_points = len(h)
    ones_count = n_points // 2
    best_h = h.copy()
    best_c5 = np.inf

    temp = 1.0
    cooling_rate = 0.9995
    iterations = 0
    max_iter = int(budget_s * 10)
    start_time = time.time()

    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * (2.0 / n_points)

    while iterations < max_iter and time.time() - start_time < budget_s:
        iterations += 1
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = np.clip(h + perturb, 0, 1)
        h_perturb_sum = np.sum(h_perturb)
        if h_perturb_sum == 0:
            continue
        h_perturb = h_perturb * (ones_count / h_perturb_sum)
        c5_perturb = compute_c5(h_perturb)
        if c5_perturb < best_c5 or (np.random.rand() < np.exp((best_c5 - c5_perturb) / temp)):
            best_h = h_perturb.copy()
            best_c5 = c5_perturb
        temp *= cooling_rate

    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    best_h_scaled = best_h * (ones_count / best_h_sum)
    final_c5 = compute_c5(best_h_scaled)
    return best_h_scaled, final_c5, n_points
