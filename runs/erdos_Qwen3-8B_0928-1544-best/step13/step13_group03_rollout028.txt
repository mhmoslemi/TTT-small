def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    population_size = 100
    gen_limit = 100
    mutation_rate = 0.2
    crossover_rate = 0.8

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        sum_initial = np.sum(initial_h)
        scale = target_sum / sum_initial
        initial_h = np.clip(initial_h * scale, 0.0, 1.0)
        population = [initial_h]
        while len(population) < population_size:
            h = np.random.rand(n_points)
            sum_h = np.sum(h)
            scale = target_sum / sum_h
            h = np.clip(h * scale, 0.0, 1.0)
            population.append(h)
    else:
        population = []
        while len(population) < population_size:
            h = np.random.rand(n_points)
            sum_h = np.sum(h)
            scale = target_sum / sum_h
            h = np.clip(h * scale, 0.0, 1.0)
            population.append(h)

    best_h = population[0]
    best_c5 = float('inf')
    start_time = time.time()

    for gen in range(gen_limit):
        if time.time() - start_time > budget_s:
            break
        fitness = []
        for h in population:
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            fitness.append(-c5)
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        parents = sorted_population[:population_size // 2]
        new_population = []
        for _ in range(population_size // 2):
            if np.random.rand() < crossover_rate:
                p1, p2 = np.random.choice(parents, size=2, replace=False)
                split = np.random.randint(1, n_points)
                child = np.concatenate((p1[:split], p2[split:]))
            else:
                child = np.random.choice(parents)
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
            child = np.clip(child, 0.0, 1.0)
            sum_child = np.sum(child)
            if sum_child == 0:
                child = np.random.rand(n_points)
                sum_child = np.sum(child)
                scale = target_sum / sum_child
                child = np.clip(child * scale, 0.0, 1.0)
            else:
                scale = target_sum / sum_child
                child = np.clip(child * scale, 0.0, 1.0)
            new_population.append(child)
        population = sorted_population[:population_size // 2] + new_population
        for h in population:
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    return best_h, best_c5, n_points
