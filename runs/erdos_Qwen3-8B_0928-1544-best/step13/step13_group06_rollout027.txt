    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    max_iterations = 10000
    learning_rate = 0.01
    momentum = 0.9
    decay_rate = 0.999
    epsilon = 1e-8
    # Adam optimizer parameters
    beta1 = 0.9
    beta2 = 0.999
    # Initialize Adam moving averages
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        t += 1
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        # Correct gradient calculation based on k's sign
        derivatives = np.zeros(n_points)
        if k > 0:
            for i in range(n_points):
                if i + k < n_points:
                    derivatives[i] = 1 - h[i + k]
                else:
                    derivatives[i] = 0
        else:
            k_abs = -k
            for i in range(n_points):
                if i - k_abs >= 0:
                    derivatives[i] = 1 - h[i - k_abs]
                else:
                    derivatives[i] = 0
        
        # Adam update rule
        m = beta1 * m + (1 - beta1) * derivatives
        v = beta2 * v + (1 - beta2) * (derivatives ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Update h
        h += learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        h = np.clip(h, 0, 1)
        # Enforce the sum constraint
        h = h / np.sum(h) * ones_count
        # Compute current C5
        current_c5 = np.max(corr) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * ones_count
    return best_h, best_c5, n_points
