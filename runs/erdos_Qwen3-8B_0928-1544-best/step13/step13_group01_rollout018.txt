            h_p = clip_and_scale(h_p)
            obj_p = compute_c5(h_p)
            h_n = h.copy()
            h_n[i] -= eps
            h_n = clip_and_scale(h_n)
            obj_n = compute_c5(h_n)
            gradient[i] = (obj_p - obj_n) / (2 * eps)

        m = beta1 * m + (1 - beta1) * gradient
        v = beta2 * v + (1 - beta2) * (gradient ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = alpha * m_hat / (np.sqrt(v_hat) + epsilon)
        h_new = h - step

        h_new = clip_and_scale(h_new)

        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
        else:
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                h = h_new
                best_c5 = new_c5

        T *= cooling_rate

    return best_h, best_c5, n_points
