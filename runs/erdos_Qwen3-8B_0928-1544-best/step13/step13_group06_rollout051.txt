    beta1 = 0.9

    epsilon = 1e-8
    m = np.zeros(n_points)

    t = 0
    max_iterations = 10000  # adjust based on time
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        t += 1
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        
        # Compute gradient for the current k
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + k
            i_minus_k = i - k

            if 0 <= i_plus_k < n_points:

        h = h / np.sum(h) * ones_count
