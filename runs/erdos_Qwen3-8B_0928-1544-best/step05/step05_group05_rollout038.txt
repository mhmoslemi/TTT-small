import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    required_sum = n_points / 2.0
    
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        h = h * (n_points / len(h))  # Rescale to target length
    else:
        h = np.full(n_points, 0.5)
    
    def clip_and_scale(x):
        x = np.clip(x, 0.0, 1.0)
        total = np.sum(x)
        if total == 0:
            return np.full_like(x, 0.5)
        factor = required_sum / total
        return x * factor
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - required_sum
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Perturb current solution
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturbed = clip_and_scale(h + perturb)
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        h = best_h
    
    return best_h, best_c5, n_points
