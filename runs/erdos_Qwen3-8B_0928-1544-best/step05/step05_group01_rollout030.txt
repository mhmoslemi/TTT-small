import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    m = 100

    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        fireflies = [initial_h_values.copy()]
    else:
        fireflies = [np.zeros(n_points)]
        indices = np.random.choice(n_points, 35, replace=False)
        fireflies[0][indices] = 1.0

    for _ in range(m-1):
        sol = np.zeros(n_points)
        indices = np.random.choice(n_points, 35, replace=False)
        sol[indices] = 1.0
        fireflies.append(sol)

    def evaluate(f):
        corr = np.correlate(f, 1 - f, mode='full')
        return np.max(corr) * dx

    fitness = [evaluate(f) for f in fireflies]
    best = np.argmin(fitness)
    best_val = fitness[best]
    c5 = best_val

    alpha = 0.1
    beta = 0.2
    gamma = 0.5
    max_iter = 300
    perturb = 0.05

    for _ in range(max_iter):
        idx = np.argsort(fitness)
        sorted_f = np.array([fireflies[i] for i in idx])
        sorted_f_ = sorted_f.copy()
        for i in range(m):
            for j in range(i+1, m):
                if sorted_f[j] < sorted_f[i]:
                    dist = np.linalg.norm(sorted_f[i] - sorted_f[j])
                    move = beta * np.exp(-gamma * dist)
                    sorted_f[i] += move * (sorted_f[j] - sorted_f[i]) + alpha * np.random.randn(n_points)
            if np.random.rand() < perturb:
                sorted_f[i] += alpha * np.random.randn(n_points)
        for i in range(m):
            sorted_f[i] = np.sort(sorted_f[i])[::-1]
            sorted_f[i] = np.where(sorted_f[i][:35] > 0, 1.0, 0.0)
        for i in range(m):
            fireflies[idx[i]] = sorted_f[i]
            fitness[idx[i]] = evaluate(sorted_f[i])
        best = np.argmin(fitness)
        if fitness[best] < best_val:
            best_val = fitness[best]
            c5 = best_val

    return fireflies[best], c5, n_points
