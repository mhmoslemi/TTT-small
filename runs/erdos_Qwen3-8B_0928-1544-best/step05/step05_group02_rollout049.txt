import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):

    target_sum = n_points / 2
    swarm_size = 50
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    vel_clip = 0.2
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None

    if 'initial_h_values' in globals():
        h_init = initial_h_values
    else:
        x = np.linspace(0, 2, n_points)
        h_init = (1 + np.sin(np.pi * x)) / 2
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (target_sum / np.sum(h_init))

    particles = [h_init + 0.1 * np.random.randn(n_points) for _ in range(swarm_size - 1)]
    particles.append(h_init)
    particles = [np.clip(p, 0, 1) for p in particles]
    particles = [p * (target_sum / np.sum(p)) for p in particles]

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    velocities = np.random.uniform(-vel_clip, vel_clip, (swarm_size, n_points))
    pbest = [np.copy(p) for p in particles]
    pbest_vals = [evaluate(p) for p in pbest]
    gbest = np.copy(pbest[np.argmin(pbest_vals)])
    gbest_val = np.min(pbest_vals)

    for _ in range(max_iter):
        current_time = time.time()
        if current_time - start_time > budget_s:
            break

        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] +
                            c1 * r1 * (pbest[i] - particles[i]) +
                            c2 * r2 * (gbest - particles[i]))
            velocities[i] = np.clip(velocities[i], -vel_clip, vel_clip)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))

        current_vals = [evaluate(p) for p in particles]
        for i in range(swarm_size):
            if current_vals[i] < pbest_vals[i]:
                pbest[i] = np.copy(particles[i])
                pbest_vals[i] = current_vals[i]

        new_gbest = np.argmin(pbest_vals)
