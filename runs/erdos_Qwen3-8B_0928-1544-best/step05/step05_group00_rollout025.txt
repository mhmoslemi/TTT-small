def halton_sequence(n, dim=1):
    # Returns n points in [0,1] using the Halton sequence for the given dimension
    # Here, for dim=1, we use the Van der Corput sequence
    # For this example, I'll generate the sequence up to n samples
    sequence = np.zeros(n)
    for i in range(n):
        # Convert i to base 2
        j = i + 1
        d = 0
        f = 0
        while j > 0:
            r = j % 2
            f += r * (0.5 ** d)
            j = j // 2
            d += 1
        sequence[i] = f
    return sequence
