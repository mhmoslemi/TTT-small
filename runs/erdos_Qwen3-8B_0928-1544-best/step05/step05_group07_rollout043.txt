import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    
    if not hasattr(initial_h, '__array__') or initial_h.size == 0:
        initial_h = np.zeros(n_points)
        for i in range(n_points // 2):
            initial_h[i*2] = 1.0

    # Normalize initial population
    def normalize(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / s, 0, 1)

    population = [normalize(initial_h.copy()) for _ in range(100)]
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = initial_h.copy()
    best_c5 = evaluate(initial_h)
    time_used = 0
    max_generations = 500
    
    for gen in range(max_generations):
        fitness = [evaluate(h) for h in population]
        selected = np.argsort(fitness)[:int(0.3 * len(population))]
        
        new_population = []
        for _ in range(len(population)):
            if np.random.rand() < 0.5:
                p1, p2 = np.random.choice(selected, 2, replace=False)
                child = (population[p1] + population[p2]) / 2
            else:
                p = np.random.choice(selected)
                child = population[p].copy()
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
                child = normalize(child)
            new_population.append(child)
        
        population = new_population
        current_best = min(population, key=evaluate)
        current_c5 = evaluate(current_best)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
        
        if gen % 10 == 0:
            def obj(x):
                return evaluate(x)
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
            res = minimize(obj, best_h, method='COBYLA', bounds=[(0,1)]*n_points, constraints=cons)
            if res.success:
                best_h = res.x
                best_c5 = evaluate(best_h)
        
        time_used += 0.05
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
