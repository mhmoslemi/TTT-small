import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2  # 35.0
    h = np.zeros(n_points)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))
    best_c5 = float('inf')
    best_h = h.copy()
    eta = 0.1
    decay_rate = 0.99
    max_iterations = 1000
    start_time = time.time()
    for _ in range(max_iterations):
        current_time = time.time()
        if current_time - start_time > budget_s:
            break
        # Compute cross-correlation
        corr_full = np.correlate(h, 1 - h, mode='full')
        max_corr_val = np.max(corr_full)
        max_index = np.argmax(corr_full)
        shift = max_index - (n_points - 1)
        k_star = shift % n_points
        # Compute gradient
        grad = np.zeros(n_points)
        for j in range(n_points):
            j_plus = (j + k_star) % n_points
            j_minus = (j - k_star) % n_points
            grad[j] = (1 - h[j_plus]) - h[j_minus]
        # Update h
        h = h - eta * grad
        # Clip and scale
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (target_sum / sum_h)
        # Evaluate new C5
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Decay learning rate
        eta *= decay_rate
    return best_h, best_c5, n_points
