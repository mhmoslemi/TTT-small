import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    T = n_points / 2  # Target sum for h
    
    # Initialize with provided initial guess
    if 'initial_h_values' in globals() and hasattr(initial_h_values, '__array__'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points // 2):
            h[i*2] = 1.0
    
    # Project initial guess to satisfy the sum constraint
    def project_to_sum(h, target):
        h = np.clip(h, 0.0, 1.0)
        S = np.sum(h)
        if abs(S - target) < 1e-10:
            return h
        n = len(h)
        D = target - S
        if D > 0:
            up = 1.0 - h
            total_up = np.sum(up)
            if total_up > 0:
                delta = D / total_up
                h += delta * up
        else:
            down = h
            total_down = np.sum(down)
            if total_down > 0:
                delta = -D / total_down
                h -= delta * down
        h = np.clip(h, 0.0, 1.0)
        return h
    
    h = project_to_sum(h, T)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = h.copy()
    best_c5 = objective(h)
    time_used = 0
    max_iter = int(budget_s / 4)
    
    for iter in range(max_iter):
        # Perturb current best
        sigma = 0.05 * (1 - iter / max_iter)
        h_perturbed = best_h + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = project_to_sum(h_perturbed, T)
        
        # Local optimization with COBYLA
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - T},
