import numpy as np
from scipy.optimize import minimize
from scipy import optimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = int(n_points / 2)
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    population = [initial_h.copy()]
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def feasibility_proj(h):
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        if total == 0:
            return h
        scale = (ones_count / total) if total > 0 else 1.0
        return h * scale
    
    def perturb(h, sigma, temp):
        return feasibility_proj(h + np.random.normal(0, sigma * temp, n_points))
    
    T0 = 0.8
    alpha = 0.995
    sigma0 = 0.2
    time_used = 0
    max_iter = int(budget_s // 25)
    
    for iter in range(max_iter):
        if len(population) < 30:
            for _ in range(30 - len(population)):
                perturbed = perturb(initial_h, sigma0, T0)
                population.append(perturbed)
        else:
            population.sort(key=lambda x: objective(x))
            keep_ratio = 0.3
            top = population[:int(len(population) * keep_ratio)]
            population = top.copy()
        
        best = population[0]
        temp = T0 * alpha ** iter
        sigma = sigma0 * alpha ** iter
        
        for _ in range(1):
            h_perturbed = perturb(best, sigma, temp)
            res = minimize(objective, h_perturbed, method='L-BFGS-B', bounds=[(0, 1)]*n_points, options={'ftol': 1e-6, 'maxiter': 50})
            if res.success and objective(res.x) < objective(best):
                best = res.x
        
        population = [best] + [perturb(best, sigma * 0.5, temp * 0.5) for _ in range(29)]
        time_used += 25
        if time_used >= budget_s:
            break
    
    best_h = population[0]
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
