import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        penalty = np.sum(h**2 * (1 - h)**2)
        return c5 + 1e-4 * penalty

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points

    def generate_initial_population(base, size=20):
        pop = [base.copy()]
        for _ in range(size - 1):
            perturb = np.random.normal(0, 0.1, n_points)
            h = base + perturb
            h = np.clip(h, 0.0, 1.0)
            pop.append(h)
        return pop

    initial_population = generate_initial_population(initial_h)

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    max_iter = int(budget_s / 100)

    for _ in range(max_iter):
        idx = np.random.randint(len(initial_population))
        current_h = initial_population[idx]
        result = minimize(
            fun=objective,
            x0=current_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
            initial_population.append(result.x)
            initial_population.pop(np.random.randint(len(initial_population)))

    return best_h, best_c5, n_points
