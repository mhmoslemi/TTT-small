        for i in range(n_points // 2):
            initial_h[i * 2] = 1.0
    h = initial_h.copy()
    scale = ones_count / np.sum(h)
    h = h * scale
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    start_time = time.time()
    max_iter = int(budget_s / 4)
    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        sorted_indices = np.argsort(corr)
        top_indices = sorted_indices[-30:]
        def obj_func(h_current):
            corr_current = np.correlate(h_current, 1 - h_current, mode='full')
            return np.sum(corr_current[top_indices] ** 2)
        result = minimize(
            fun=obj_func,
            x0=h,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - ones_count}],
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            current_h = result.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * (2.0 / n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        h = result.x
        if time.time() - start_time > budget_s:
            break
    return best_h, best_c5, n_points
