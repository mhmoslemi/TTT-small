    # Normalize initial guess
    initial_sum = np.sum(initial_h)
    if initial_sum > 0:
        initial_h = initial_h * target_sum / initial_sum
    
    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def normalize(h):
        s = np.sum(h)
        return h * target_sum / s if s > 0 else np.zeros_like(h)
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()
    
    generations = 100
    mutation_rate = 0.3
    crossover_rate = 0.5
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        fitness = [evaluate(h) for h in population]
        sorted_idx = np.argsort(fitness)
        parents_idx = sorted_idx[:int(len(population)*0.3)]
        parents = [population[i] for i in parents_idx]
        
        new_population = []
        for _ in range(population_size):
            if np.random.rand() < crossover_rate:
                p1, p2 = np.random.choice(parents, size=2, replace=False)
                child = (p1 + p2) / 2
            else:
                h = np.random.normal(0, 0.1, n_points)
                child = population[np.random.choice(len(population))] + h
                child = np.clip(child, 0, 1)
            child = normalize(child)
            new_population.append(child)
        
        combined = population + new_population
        combined_fitness = [evaluate(h) for h in combined]
        sorted_combined = sorted(zip(combined_fitness, combined), key=lambda x: x[0])
        population = [h for (f, h) in sorted_combined[:population_size]]
        
        current_best = population[0]
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best
    
    def objective(h):
        return evaluate(h)
    
    result = minimize(
        fun=objective,
        x0=best_h,
        method='COBYLA',
        bounds=[(0, 1)]*n_points,
        constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum},
        options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    
    return best_h, best_c5, n_points
