import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = ones_count / sum_h
        h = h * scale
        return evaluate(h)

    # Local optimization with bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds)
    best_h = result.x
    best_c5 = evaluate(best_h)
    return best_h, best_c5, n_points
