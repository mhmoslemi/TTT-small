import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.3 * np.sin(np.pi * x) + 0.2 * np.sin(2 * np.pi * x)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def objective(params):
        A, B, phi, psi = params
        x = np.linspace(0, 2, n_points)
        h = 0.5 + A * np.sin(np.pi * x + phi) + B * np.sin(2 * np.pi * x + psi)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        return evaluate_c5(h)
    
    # Initial guess
    best_params = [0.3, 0.2, 0.0, 0.0]
    best_c5 = evaluate_c5(h)
    
    # Optimization
    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_params, method='L-BFGS-B', bounds=[(0, 0.5), (0, 0.5), (0, 2*np.pi), (0, 2*np.pi)])
        if res.success:
            best_params = res.x
            best_c5 = res.fun
        # Perturb slightly for exploration
        best_params += np.random.normal(0, 0.05, 4)
        best_params = np.clip(best_params, [0, 0, 0, 0], [0.5, 0.5, 2*np.pi, 2*np.pi])
    
    # Generate final h from best_params
    x = np.linspace(0, 2, n_points)
    h_best = 0.5 + best_params[0] * np.sin(np.pi * x + best_params[2]) + best_params[1] * np.sin(2 * np.pi * x + best_params[3])
    h_best = np.clip(h_best, 0, 1)
