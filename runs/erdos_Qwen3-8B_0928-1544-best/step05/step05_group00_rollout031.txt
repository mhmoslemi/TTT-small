        return c5 + penalty
    
    def constraint_sum(h):

    bounds = [(0.0, 1.0)] * n_points
    
    # Enhanced initialization with more diverse points
    initial_population = [initial_h.copy()]
    for _ in range(24):
        perturbed = initial_h + np.random.normal(0, 0.1, n_points)
        initial_population.append(np.clip(perturbed, 0, 1))
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    max_iter = int(budget_s / 100)
    
    for _ in range(max_iter):
        current_h = initial_population[np.random.randint(len(initial_population))]
        result = minimize(
