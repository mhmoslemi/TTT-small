from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)

    if not hasattr(initial_h, '__array__') or initial_h.size == 0:

            initial_h[i] = 1.0
    initial_h = np.clip(initial_h, 0.0, 1.0)
    sum_h = np.sum(initial_h)
    if sum_h != 0:
        initial_h = initial_h * (ones_count / sum_h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    time_used = 0
    max_iter = int(budget_s / 4)
    num_shifts = 30
    
    for _ in range(max_iter):
        # Compute overlaps and select top shifts
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        corr_indices = np.argsort(corr)[::-1]  # Highest overlaps first
        top_shifts = corr_indices[:num_shifts]
        
        # Create cost function using selected shifts
        def cost_func(h):
            return np.max(corr[top_shifts] @ h) * (2.0 / n_points)
        
        # Define optimization problem
        result = minimize(
            fun=objective,
            x0=best_h,
            method='L-BFGS-B',
            bounds=bounds,
