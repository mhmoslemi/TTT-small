import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_needed = 35
    
    num_particles = 50
    particles = np.zeros((num_particles, n_points))
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        particles[0, :] = initial_h_values.copy()
        for i in range(1, num_particles):
            idx = np.random.choice(n_points, ones_needed, replace=False)
            particles[i, idx] = 1.0
    else:
        for i in range(num_particles):
            idx = np.random.choice(n_points, ones_needed, replace=False)
            particles[i, idx] = 1.0
    
    velocities = np.random.uniform(-0.1, 0.1, (num_particles, n_points))
    
    personal_best = particles.copy()
    global_best = particles[0, :].copy()
    best_c5 = np.inf
    dx = 2.0 / n_points
    
    window_shift_prob = 0.1
    w = 0.7
    c1 = c2 = 2.0
    max_iterations = 1000
    time_budget = budget_s
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > time_budget:
            break
        
        r1 = np.random.rand(num_particles, n_points)
        r2 = np.random.rand(num_particles, n_points)
        velocities = w * velocities + c1 * r1 * (personal_best - particles) + c2 * r2 * (global_best - particles)
        velocities = np.clip(velocities, -0.2, 0.2)
        
        particles += velocities
        
        for i in range(num_particles):
            if np.random.rand() < window_shift_prob:
                start_idx = np.random.randint(0, n_points)
                window = particles[i, start_idx:start_idx+10].copy()
                shift = np.random.choice([-1, 1])
                window += shift
                window = np.clip(window, 0.0, 1.0)
                particles[i, start_idx:start_idx+10] = window
        
        for i in range(num_particles):
            particles[i, :] = np.clip(particles[i, :], 0.0, 1.0)
            current_sum = np.sum(particles[i, :])
            if current_sum > ones_needed:
                diff = current_sum - ones_needed
                indices = np.argsort(particles[i, :])[:diff]
                particles[i, indices] = 0.0
            elif current_sum < ones_needed:
                diff = ones_needed - current_sum
                indices = np.argsort(particles[i, :])[-diff:]
                particles[i, indices] = 1.0
        
        objectives = []
        for h in particles:
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            obj = max_corr * dx
            objectives.append(obj)
        
        for i in range(num_particles):
            current_obj = objectives[i]
            if current_obj < np.mean(objectives):
                if current_obj < np.linalg.norm(personal_best[i] - particles[i]):
                    personal_best[i] = particles[i].copy()
                    if current_obj < best_c5:
                        best_c5 = current_obj
                        global_best = particles[i].copy()
                if current_obj < best_c5:
                    best_c5 = current_obj
                    global_best = particles[i].copy()
        
        top_indices = np.argsort(objectives)[:10]
        for i in top_indices:
            h_initial = particles[i, :].copy()
            def obj(x):
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * dx
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - ones_needed}
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            result = minimize(obj, h_initial, method='SLSQP', bounds=bounds, constraints=cons)
            if result.success:
                if result.fun < objectives[i]:
                    particles[i, :] = result.x
                    if result.fun < best_c5:
                        best_c5 = result.fun
                        global_best = result.x
    
    return (global_best, best_c5, n_points)
