        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = ones_count / sum_h
        return h * scale
    
    def generate_triangular():
        x = np.linspace(0, 2, n_points)
        h = np.clip(np.abs(x - 1.0), 0, 1)
        sum_h = np.sum(h)
        scale = ones_count / sum_h
        return h * scale
    
    population = []
