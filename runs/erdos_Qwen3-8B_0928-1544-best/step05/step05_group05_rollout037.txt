import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    required_sum = n_points / 2.0

    def objective(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)

    if 'initial_h_values' in globals():
        current_h = initial_h_values.copy()
    else:
        current_h = np.full(n_points, 0.5)

    # Ensure sum constraint
    current_h = current_h * (required_sum / np.sum(current_h))
    current_c5 = objective(current_h)
    best_c5 = current_c5
    best_h = current_h.copy()

    T = 1.0
    alpha = 0.996
    max_iter = int(budget_s / 0.001)  # 1000 iterations per second

    start_time = time.time()
    iter_count = 0

    while iter_count < max_iter and time.time() - start_time < budget_s:
        perturb_indices = np.random.choice(n_points, size=5, replace=False)
        noise = np.random.normal(0, 0.05, 5)
        perturbed_h = current_h.copy()
        perturbed_h[perturb_indices] += noise
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (required_sum / np.sum(perturbed_h))

        new_c5 = objective(perturbed_h)

        if new_c5 < current_c5:
            current_h = perturbed_h.copy()
            current_c5 = new_c5
        else:
            delta = new_c5 - current_c5
            prob = np.exp(-delta / T)
            if np.random.rand() < prob:
                current_h = perturbed_h.copy()
                current_c5 = new_c5

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()

        T *= alpha
        iter_count += 1

    return best_h, best_c5, n_points
