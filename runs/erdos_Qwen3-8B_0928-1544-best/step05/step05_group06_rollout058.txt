def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    s = n_points / 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    h = initial_h.copy()

    max_iter = int(budget_s / 2)  # 500 iterations with 2 seconds each

    for _ in range(max_iter):
        corr_cross = np.correlate(h, h, mode='full')
        cross_corr_values = corr_cross * (2.0 / n_points)
        max_index = np.argmax(cross_corr_values)
        k_val = max_index - (n_points - 1)

        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = i + k_val
            if 0 <= j < n_points:
                A[i, j] = 2.0 / n_points

        h_var = cp.Variable(n_points)
        objective = cp.Maximize(cp.quad_form(h_var, A))
        constraints = [

            problem.solve(solver=cp.ECOS, verbose=False)
            if problem.status == 'OPTIMAL':
                h = h_var.value
        except:
