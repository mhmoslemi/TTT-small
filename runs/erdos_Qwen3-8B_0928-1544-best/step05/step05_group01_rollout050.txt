import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(70)
        h[:35] = 1.0
    target_sum = 35

    def calculate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / len(h))

    best_h = h.copy()
    best_c5 = calculate_c5(h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        n = len(h)
        num_perturb = int(n * 0.1)
        indices = np.random.choice(n, size=num_perturb, replace=False)
        for i in indices:
            current_val = h[i]
            if current_val == 0.0:
                new_val = np.random.choice([0.5, 1.0])
            elif current_val == 1.0:
                new_val = np.random.choice([0.0, 0.5])
            else:
                new_val = np.random.choice([0.0, 1.0])
            h[i] = new_val

        current_sum = np.sum(h)
        diff = current_sum - target_sum
        if abs(diff) > 1e-6:
            while abs(diff) > 1e-6:
                candidates = []
                for i in range(n):
                    val = h[i]
                    if val == 0.0:
                        if diff < 0:
                            candidates.append((i, 0.5, 1.0))
                    elif val == 1.0:
                        if diff > 0:
                            candidates.append((i, -0.5, -1.0))
                    elif val == 0.5:
                        if diff > 0:
                            candidates.append((i, 0.5, 1.0))
                        elif diff < 0:
                            candidates.append((i, -0.5, -0.5))
                if not candidates:
                    break
                _, delta1, delta2 = candidates[0]
                if diff > 0:
                    h[candidates[0][0]] += delta1
                    diff -= delta1
                else:
                    h[candidates[0][0]] += delta2
                    diff -= delta2

        current_c5 = calculate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    return best_h, best_c5, 70
