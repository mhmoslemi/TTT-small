import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def objective(h):
        return evaluate(h)

    def constraint_func(h):
        return np.sum(h) - ones_count

    constraints = [{'type': 'eq', 'fun': constraint_func}]
    
    def optimize_from_initial():
        res = minimize(objective, initial_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=constraints, tol=1e-8)
        return res.x, evaluate(res.x)

    def perturb_and_scale(h, scale_factor=0.5):
        perturb = np.random.normal(0, scale_factor, n_points)
        h_perturb = h + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        sum_h = np.sum(h_perturb)
        scale = ones_count / sum_h
        return h_perturb * scale

    best_h, best_c5 = optimize_from_initial()
    time_used = 0
    max_iter = int(budget_s // 5)
    
    for _ in range(max_iter):
        h = perturb_and_scale(best_h)
        res = minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=constraints, tol=1e-8)
        if res.success:
            current_h = res.x
            current_c5 = evaluate(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        time_used += 5
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
