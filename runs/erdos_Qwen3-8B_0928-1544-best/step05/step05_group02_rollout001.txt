import time

                new_h = new_h * (target_sum / new_h_sum)
            particles.append(new_h)
    else:
        x = np.linspace(0, 2, n_points)
        sine_h = (1 + np.sin(np.pi * x)) / 2

    best_c5 = evaluate_c5(global_best)
