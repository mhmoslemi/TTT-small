    best_h = h_initial
    best_c5 = evaluate_erdos_solution(h_initial)
    for h_start in start_points:
        result = minimize(objective, h_start, method='L-BFGS-B', bounds=bnds, constraints=cons, options={'maxiter': 100})
        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
    
    return best_h, best_c5, n_points
