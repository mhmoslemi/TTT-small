import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    h = cp.Variable(n_points)
    t = cp.Variable((n_points, n_points))
    C = cp.Variable(nonneg=True)

    constraints = []
    for i in range(n_points):
        for j in range(n_points):
            constraints.append(t[i, j] <= h[i])
            constraints.append(t[i, j] <= h[j])
            constraints.append(t[i, j] >= h[i] + h[j] - 1)

    constraints.append(cp.sum(h) == target_sum)
    for i in range(n_points):
        constraints.append(h[i] >= 0)
        constraints.append(h[i] <= 1)

    for k in range(2 * n_points - 1):
        shift_sum = 0.0
        for i in range(n_points):
            idx = (i + k) % n_points
            shift_sum += t[i, idx]
        constraints.append(1.0 - shift_sum * dx <= C)

    objective = cp.Minimize(C)

    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.MOSEK, verbose=False)

    h_values = h.value
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    c5_bound = np.max(corr) * dx

    return h_values, c5_bound, n_points
