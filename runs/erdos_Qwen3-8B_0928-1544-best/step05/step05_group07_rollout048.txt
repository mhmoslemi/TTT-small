import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def project_to_sum(h, target_sum):
    h_clipped = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h_clipped)
    if np.isclose(current_sum, target_sum):
        return h_clipped.copy()
    deficit = target_sum - current_sum
    sorted_idx = np.argsort(h_clipped)
    for i in sorted_idx:
        if deficit <= 0:
            break
        val = h_clipped[i]
        if val == 1.0:
            continue
        max_inc = 1.0 - val
        if max_inc > deficit:
            h_clipped[i] += deficit
            deficit = 0
        else:
            h_clipped[i] = 1.0
            deficit -= max_inc
    surplus = current_sum - target_sum
    if surplus > 0:
        sorted_idx = np.argsort(h_clipped)[::-1]
        for i in sorted_idx:
            if surplus <= 0:
                break
            val = h_clipped[i]
            if val == 0.0:
                continue
            max_dec = val
            if max_dec > surplus:
                h_clipped[i] -= surplus
                surplus = 0
            else:
                h_clipped[i] = 0.0
                surplus -= max_dec
    return h_clipped
