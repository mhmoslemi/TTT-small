import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = int(n_points / 2)
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    h = initial_h.copy()
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * (2.0 / n_points)
        # Select the worst shift
        k = np.argmax(corr) - len(h) + 1
        # Create a constraint for the worst shift
        def constraint(x):
            if k < 0:
                return np.sum(x) - ones_count
            else:
                return np.sum([x[i] * (1 - x[i + k]) for i in range(n_points) if i + k < n_points]) - max_overlap
        cons = [{'type': 'eq', 'fun': constraint}]
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 20})
        if res.success:
            h = res.x
