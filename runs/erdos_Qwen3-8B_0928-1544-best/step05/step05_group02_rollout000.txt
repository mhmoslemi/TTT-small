import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    swarm_size = 50
    max_iterations = 10000
    omega = 0.7
    c1 = 2.0
    c2 = 2.0
    velocity_decay = 0.999
    start_time = time.time()
    
    # Initialize swarm
    if 'initial_h_values' in globals():
        h_particles = [initial_h_values.copy()]
        for _ in range(swarm_size - 1):
            noise = np.random.normal(0, 0.1, n_points)
            h = initial_h_values + noise
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum != 0:
                h = h * (target_sum / h_sum)
            h_particles.append(h)
    else:
        h_particles = [np.random.rand(n_points) * (target_sum / np.sum(np.random.rand(n_points))) for _ in range(swarm_size)]
    
    # Initial velocities
    velocities = np.random.uniform(-0.1, 0.1, (swarm_size, n_points))
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    # Best personal and global
    personal_best = [h.copy() for h in h_particles]
    global_best = h_particles[0]
    best_c5 = evaluate_c5(global_best)
    
    # Main loop
    while time.time() - start_time < budget_s:
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + c1 * r1 * (personal_best[i] - h_particles[i]) + c2 * r2 * (global_best - h_particles[i])
            h_particles[i] += velocities[i]
            h_particles[i] = np.clip(h_particles[i], 0, 1)
            h_sum = np.sum(h_particles[i])
            if h_sum != 0:
                h_particles[i] = h_particles[i] * (target_sum / h_sum)
        
        # Update personal and global bests
        for i in range(swarm_size):
            current_c5 = evaluate_c5(h_particles[i])
            if current_c5 < evaluate_c5(personal_best[i]):
                personal_best[i] = h_particles[i].copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_particles[i].copy()
                    global_best = personal_best[i]
    
    return best_h, best_c5, n_points
