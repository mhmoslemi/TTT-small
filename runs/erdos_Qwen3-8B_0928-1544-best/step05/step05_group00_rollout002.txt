def run(seed=42, budget_s=1100, **kwargs):

            elif i % 3 == 1:
                initial_h[i] = 0.5
            else:
                initial_h[i] = 0.75
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    constraints = [LinearConstraint(bounds, lb=np.zeros(n_points), ub=np.ones(n_points))]
