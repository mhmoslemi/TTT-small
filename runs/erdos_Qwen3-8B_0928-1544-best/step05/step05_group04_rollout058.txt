    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 * (1 + np.sin(np.pi * (i - n_points / 2) / n_points))
        h = np.clip(h, 0.0, 1.0)
        h = h * (ones_count / np.sum(h))

    best_h = h.copy()
    best_c5 = compute_c5(h)

    time_used = 0
    max_iter = int(budget_s / 2)
    
    for _ in range(max_iter):
        time_used += 1
        if time_used > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        
        grad = np.zeros(n_points)
        for j in range(n_points):
            if (j + k) >= 0 and (j + k) < n_points:
                term1 = 1 - h[j + k]
            else:
                term1 = 0.0
            if (j - k) >= 0 and (j - k) < n_points:
                term2 = -h[j - k]
            else:
                term2 = 0.0
            grad[j] = term1 + term2
        
        h += 0.05 * grad
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (ones_count / sum_h)
        else:
            h = np.zeros(n_points)
            h[0] = 1.0
        
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    return best_h, best_c5, n_points

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
