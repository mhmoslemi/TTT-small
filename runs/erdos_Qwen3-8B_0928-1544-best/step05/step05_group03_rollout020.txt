                h = 0.5 + 0.5 * np.sin(np.pi * x)
                h = np.clip(h, 0, 1)
                sum_h = np.sum(h)
                scale = ones_count / sum_h
                new_h = h * scale
                new_candidates.append(new_h)

        new_population = sorted_pop[:population_size - len(new_candidates)] + new_candidates
        population = new_population
        current_best = min(zip(scores, population), key=lambda x: x[0])[1]
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best
        time_used += 5
        if time_used >= budget_s:
            break

    return best_h, best_c5, n_points
