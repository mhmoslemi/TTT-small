import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

            population.append(perturbed)
    else:
        population = [np.full(n_points, 0.5) * (required_sum / np.sum(np.full(n_points, 0.5))) for _ in range(30)]

        if current_best[0] < best_c5:
