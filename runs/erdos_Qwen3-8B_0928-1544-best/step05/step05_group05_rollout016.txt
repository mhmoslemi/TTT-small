                child = np.clip(child, 0, 1)
                child = child * (required_sum / np.sum(child))
            new_pop.append(child)

        indices = np.argsort(fitness)
        for i in range(pop_size // 2):
            pop[indices[-i - 1]] = new_pop[i]

    return best_h, best_c5, n_points
