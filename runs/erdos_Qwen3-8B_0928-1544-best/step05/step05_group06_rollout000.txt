import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    def feasibility_proj(h):
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        if total == 0:
            return h
        return h * (ones_count / total)
    
    def perturb(h, sigma, temp):
        return feasibility_proj(h + np.random.normal(0, sigma * temp, n_points))
    
    T0 = 1.0
    alpha = 0.999
    sigma0 = 0.2
    time_used = 0
    max_iter = int(budget_s // 25)
    population = [initial_h.copy()]
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    for iter in range(max_iter):
        if len(population) < 50:
            for _ in range(50 - len(population)):
                perturbed = perturb(initial_h, sigma0, T0)
                population.append(perturbed)
        else:
            population.sort(key=lambda x: objective(x))
            keep_ratio = 0.25
            population = population[:int(len(population) * keep_ratio)]
        
        for h in population:
            result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'maxiter': 20})
            if result.success:
                current_h = result.x
                current_c5 = objective(current_h)
                if current_c5 < best_c5:
                    best_h = current_h
                    best_c5 = current_c5
        
        time_used += 25
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
