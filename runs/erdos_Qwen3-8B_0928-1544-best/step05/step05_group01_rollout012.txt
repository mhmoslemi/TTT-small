import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 70
    target_sum = n // 2  # 35
    dx = 2.0 / n

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        if np.isclose(current_sum, target_sum):
            return h
        diff = target_sum - current_sum
        if diff > 0:
            indices = np.argsort(h)
            for i in indices:
                if h[i] < 1.0:
                    delta = min(1.0 - h[i], diff)
                    h[i] += delta
                    diff -= delta
                    if diff == 0:
                        break
        else:
            indices = np.argsort(-h)
            for i in indices:
                if h[i] > 0.0:
                    delta = min(h[i], -diff)
                    h[i] -= delta
                    diff += delta
                    if diff == 0:
                        break
        return h

    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.zeros(n)
        idx = np.random.choice(n, target_sum, replace=False)
        h_initial[idx] = 1.0

    num_particles = 50
    swarm = np.zeros((num_particles, n))
    velocities = np.zeros((num_particles, n))
    personal_best = np.zeros((num_particles, n))
    global_best = h_initial.copy()
    global_best_value = compute_c5(global_best)

    for i in range(num_particles):
        if i == 0:
            swarm[i] = h_initial
            velocities[i] = np.random.uniform(-0.1, 0.1, n)
        else:
            idx = np.random.choice(n, target_sum, replace=False)
            swarm[i][idx] = 1.0
            velocities[i] = np.random.uniform(-0.1, 0.1, n)
        personal_best[i] = swarm[i].copy()
        personal_best_values = np.array([compute_c5(swarm[i]) for i in range(num_particles)])
        global_best_value = np.min(personal_best_values)

    w, c1, c2 = 0.7, 2.0, 2.0
    max_iter = 1000
    iteration = 0

    while iteration < max_iter and time.time() - start_time < budget_s:
        for i in range(num_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + c1 * r1 * (personal_best[i] - swarm[i]) + c2 * r2 * (global_best - swarm[i])
            velocities[i] = np.clip(velocities[i], -0.2, 0.2)
            swarm[i] += velocities[i]
            if np.random.rand() < 0.1:
                shift = np.random.randint(-10, 10)
                swarm[i] = np.roll(swarm[i], shift)
            swarm[i] = repair(swarm[i])
            current_val = compute_c5(swarm[i])
            if current_val < compute_c5(personal_best[i]):
                personal_best[i] = swarm[i].copy()
                personal_best_values[i] = current_val
            if current_val < global_best_value:
                global_best = swarm[i].copy()
                global_best_value = current_val

        top_indices = np.argsort(personal_best_values)[:10]
        for i in top_indices:
            h = personal_best[i].copy()
            def obj(x):

            bounds = [(0.0, 1.0)] * n

                if refined_val < global_best_value:
                    global_best = refined_h.copy()
                    global_best_value = refined_val
        iteration += 1
