import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    
    num_initial = 50
    initial_population = [initial_h.copy()]
    for _ in range(num_initial - 1):
        perturbed = initial_h + np.random.normal(0, 0.25, n_points)
        initial_population.append(np.clip(perturbed, 0, 1))
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    max_iter = int(budget_s / 200)
    
    for _ in range(max_iter):
        current_h = initial_population[np.random.randint(len(initial_population))]
        result = minimize(
            fun=objective,
            x0=current_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
            initial_population.append(result.x)
            initial_population.pop(np.random.randint(len(initial_population)))
    
    return best_h, best_c5, n_points
