from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70

    if 'initial_h_values' in globals():

        initial_candidates = [initial_h]

            h_perturbed = initial_h + np.random.normal(0, 0.05, n_points)
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            h_perturbed = h_perturbed * (required_sum / np.sum(h_perturbed))
            initial_candidates.append(h_perturbed)
    else:
        initial_h = np.full(n_points, 0.5)
        initial_candidates = [initial_h]

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - required_sum

    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_candidates[0].copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        current_h = initial_candidates[np.random.randint(len(initial_candidates))]
        result = minimize(
