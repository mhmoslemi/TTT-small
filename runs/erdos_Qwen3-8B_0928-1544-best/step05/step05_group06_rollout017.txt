import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def feasibility_proj(h):
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        if current_sum == 0:
            return h
        scale = target_sum / current_sum
        return h * scale

    bounds = Bounds(0.0, 1.0)

    best_h = h.copy()
    best_c5 = objective(best_h)
    time_used = 0
    max_iter = int(budget_s // 50)  # 22 iterations with 50 seconds each

    for _ in range(max_iter):
        # Perturb
        perturbation = norm.rvs(0, 0.1, size=n_points)
        h_perturbed = h + perturbation
        h_perturbed = feasibility_proj(h_perturbed)

        # Local optimization
        result = minimize(
            objective,
            h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-7, 'eps': 1e-8}
        )

        # Ensure feasible
        if result.success:
            h_candidate = feasibility_proj(result.x)
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
                h = best_h

        time_used += 50
        if time_used >= budget_s:
            break

    return best_h, best_c5, n_points
