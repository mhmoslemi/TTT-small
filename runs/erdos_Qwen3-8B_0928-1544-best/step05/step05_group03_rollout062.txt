from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def objective(h):
        total = np.sum(h)
        if total == 0:
            return float('inf')
        scale = (n_points / 2) / total
        scaled_h = h * scale
        corr = np.correlate(scaled_h, 1 - scaled_h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=200,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-8,
        x0=initial_h
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        best_c5 = float('inf')
    return best_h, best_c5, n_points
