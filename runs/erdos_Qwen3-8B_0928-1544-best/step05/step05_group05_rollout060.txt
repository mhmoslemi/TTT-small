import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 120
    required_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.full(n_points, 0.5)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def obj_penalty(h):
        penalty = 1e2 * np.abs(np.sum(h) - required_sum)
        return objective(h) + penalty

    bounds = [(0.0, 1.0)] * n_points
    result = differential_evolution(
        obj_penalty,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        maxiter=400,
        tol=1e-9,
        seed=seed
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
