def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    if 'initial_h_values' in globals():

    res = minimize(objective, h, bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, method='L-BFGS-B', options={'maxiter': max_iter})
