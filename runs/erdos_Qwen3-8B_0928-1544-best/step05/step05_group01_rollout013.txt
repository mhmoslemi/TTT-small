import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    required_sum = 35
    swarm_size = 50
    max_iter = 1000
    inertia = 0.7
    c1 = c2 = 2.0
    velocity_clamp = 0.2
    window_shift_prob = 0.1
    window_size = 10

    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        delta = required_sum - current_sum
        if delta != 0:
            if delta > 0:
                idx = np.argsort(h)
                for i in idx:
                    if h[i] < 1:
                        add = min(delta, 1 - h[i])
                        h[i] += add
                        delta -= add
                        if delta == 0:
                            break
            else:
                idx = np.argsort(-h)
                for i in idx:
                    if h[i] > 0:
                        sub = min(-delta, h[i])
                        h[i] -= sub
                        delta += sub
                        if delta == 0:
                            break
        return h

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.zeros(n_points)
        np.random.shuffle(np.random.permutation([1]*35 + [0]*35))
        initial_h = np.random.choice([0, 1], size=n_points, p=[0.6, 0.4])
        while np.sum(initial_h) != required_sum:
            idx = np.where(initial_h == 0)[0]
            if np.sum(initial_h) < required_sum:
                initial_h[idx[:required_sum - np.sum(initial_h)]] = 1
            else:
                initial_h[idx[:np.sum(initial_h) - required_sum]] = 0

    swarm = np.array([initial_h.copy() for _ in range(swarm_size)])

    best_h = swarm[0].copy()

                swarm[i] = repair(swarm[i])
