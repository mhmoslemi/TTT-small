import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    swarm_size = 50
    max_iterations = 1000
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0

    if 'initial_h_values' in globals():
        swarm = [initial_h_values]
    else:
        swarm = []
    for _ in range(swarm_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * target_sum / np.sum(h)
        swarm.append(h)

    personal_best = [np.copy(h) for h in swarm]
    global_best = np.copy(personal_best[0])
    best_c5 = np.max(np.correlate(global_best, 1 - global_best, mode='full')) * (2.0 / n_points)
    best_h = np.copy(global_best)

    velocities = np.zeros((swarm_size, n_points))
    start_time = time.time()

    for _ in range(max_iterations):
        current_time = time.time()
        if current_time - start_time > budget_s:
            break

        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = inertia * velocities[i] + c1 * r1 * (personal_best[i] - swarm[i]) + c2 * r2 * (global_best - swarm[i])
            swarm[i] += velocities[i]
            swarm[i] = np.clip(swarm[i], 0, 1)
            h_sum = np.sum(swarm[i])
            if h_sum != 0:
                swarm[i] = swarm[i] * (target_sum / h_sum)

        for i in range(swarm_size):
            current_c5 = np.max(np.correlate(swarm[i], 1 - swarm[i], mode='full')) * (2.0 / n_points)
            if current_c5 < np.max(np.correlate(personal_best[i], 1 - personal_best[i], mode='full')) * (2.0 / n_points):
                personal_best[i] = np.copy(swarm[i])
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = np.copy(swarm[i])

        for i in range(swarm_size):
            if np.max(np.correlate(personal_best[i], 1 - personal_best[i], mode='full')) * (2.0 / n_points) < best_c5:
                global_best = np.copy(personal_best[i])
                best_c5 = np.max(np.correlate(global_best, 1 - global_best, mode='full')) * (2.0 / n_points)

    return best_h, best_c5, n_points
