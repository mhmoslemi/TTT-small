    def compute_gradient(h, k):
        n = len(h)
        dx = 2.0 / n_points
        grad = np.zeros(n)
        for i in range(n):
            term1 = 1.0 - h[(i + k) % n] if 0 <= i + k < n else 0.0
            term2 = h[(i - k) % n] if 0 <= i - k < n else 0.0
            grad[i] = (term1 - term2) * dx
        return grad
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    for iteration in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        m = np.argmax(corr)
        k = m - n_points + 1
        grad = compute_gradient(h, k)
        
        alpha = 0.5
        while True:
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (required_sum / np.sum(h_new))
            new_c5 = compute_c5(h_new)
            if new_c5 < compute_c5(h):
                break
            alpha *= 0.5
        
        h = h_new.copy()
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if iteration % 50 == 0:
            perturbation = np.random.normal(0, 0.05, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (required_sum / np.sum(h))
    
    return best_h, best_c5, n_points
