        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = overlap(best_h)
    time_used = 0
    max_iter = int(budget_s * 0.8 / 5)
    
    for _ in range(max_iter):
        # Perturb
        h_pert = best_h + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (ones_count / np.sum(h_pert))
        
        # Optimize
        result = minimize(
            fun=overlap,
            x0=h_pert,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        time_used += 5
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
