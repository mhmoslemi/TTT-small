from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    
    initial_population = [initial_h.copy()]
    for _ in range(49):
        perturbed = initial_h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        initial_population.append(perturbed)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    lambda_penalty = 1e-3
    
    for _ in range(150):
        current_h = initial_population[np.random.randint(len(initial_population))]
        result = minimize(
            fun=lambda x: objective(x) + lambda_penalty * np.sum((x - 0.5) ** 2),  # Improved penalty
            x0=current_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
