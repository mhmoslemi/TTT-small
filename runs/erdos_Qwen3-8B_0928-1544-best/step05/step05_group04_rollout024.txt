def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():

    if 'initial_h_values' in globals():

    velocities = np.random.uniform(-0.5, 0.5, (n_particles, n_points))
