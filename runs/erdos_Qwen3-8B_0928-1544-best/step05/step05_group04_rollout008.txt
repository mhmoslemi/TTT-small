            velocities[i] = vel

            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            particles[i] = particles[i] * (ones_count / np.sum(particles[i]))

            current_value = np.max(np.correlate(particles[i], 1 - particles[i], 'full')) * dx
            if current_value < personal_best_values[i]:
                personal_best[i] = np.copy(particles[i])
                personal_best_values[i] = current_value
                if current_value < global_best_value:
                    global_best = np.copy(particles[i])
                    global_best_value = current_value

        if time.time() - start_time >= budget_s:
            break

    return global_best, global_best_value, n_points
