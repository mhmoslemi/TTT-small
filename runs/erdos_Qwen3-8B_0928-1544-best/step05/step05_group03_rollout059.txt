        perturb = np.random.normal(0, 0.15, n_points)
        h_perturb = h + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        sum_h = np.sum(h_perturb)
        scale = ones_count / sum_h if sum_h != 0 else 1.0
        return h_perturb * scale
    
    best_h = initial_h.copy()
    best_c5 = evaluate(initial_h)
    
    population = [initial_h.copy()]
    for _ in range(30):
        h = population[np.random.randint(len(population))]
        h = perturb(h)
        h = optimize_candidate(h)
        population.append(h)
    
    time_used = 0
    max_iter = int(budget_s // 4)
    for _ in range(max_iter):
        population.sort(key=evaluate)
        parents = population[:10]
        new_population = []
        while len(new_population) < 30:
            h1, h2 = np.random.choice(parents, 2, replace=False)
            h = (h1 + h2) / 2
            h = perturb(h)
            h = optimize_candidate(h)
            new_population.append(h)
        population = new_population
        current_best = min(population, key=evaluate)
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best
        time_used += 100
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
