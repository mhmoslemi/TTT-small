import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n = 70
    m = 100
    max_iter = 300
    gamma = 0.5
    beta = 0.2

    # Initialize population
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        initial = initial_h_values.copy()
    else:
        initial = np.zeros(n)
        indices = np.random.choice(n, 35, replace=False)
        initial[indices] = 1.0

    fireflies = [initial]
    while len(fireflies) < m:
        h = np.zeros(n)
        indices = np.random.choice(n, 35, replace=False)
        h[indices] = 1.0
        fireflies.append(h)
    fireflies = np.array(fireflies)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n)

    best_h = initial.copy()
    best_c5 = objective(initial)

    for _ in range(max_iter):
        c5_values = np.array([objective(f) for f in fireflies])
        sorted_indices = np.argsort(c5_values)
        sorted_fireflies = fireflies[sorted_indices]
        sorted_c5 = c5_values[sorted_indices]

        for i in range(m):
            current = sorted_fireflies[i]
            best = sorted_fireflies[0]
            distance = np.linalg.norm(current - best)
            attraction = beta * np.exp(-gamma * distance) * (best - current)
            new_pos = current + attraction
            if np.random.rand() < perturb_prob:
                noise = np.random.normal(0, alpha, n)
                new_pos += noise
            indices = np.argsort(new_pos)[::-1]
            new_pos_repaired = np.zeros(n)
