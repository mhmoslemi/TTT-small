            if 0 <= j_plus_k < n_points:
                term1 = 1 - h[j_plus_k]
            else:
                term1 = 0
            j_minus_k = j - k
            if 0 <= j_minus_k < n_points:
                term2 = h[j_minus_k]
            else:
                term2 = 0
            gradient[j] = term1 - term2

        # Update h
        h = h - learning_rate * gradient

        # Clip to [0, 1]
        h = np.clip(h, 0, 1)

        # Scale to target_sum
        h_sum = np.sum(h)
        if h_sum != 0:
            scale_factor = target_sum / h_sum
            h = h * scale_factor

        # Evaluate C5
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Adjust learning rate
        learning_rate *= 0.995

        # Random perturbation to avoid local minima
        if np.random.rand() < 0.01:
            noise = np.random.normal(0, 0.05, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum != 0:
                h = h * (target_sum / h_sum)

    return best_h, best_c5, n_points
