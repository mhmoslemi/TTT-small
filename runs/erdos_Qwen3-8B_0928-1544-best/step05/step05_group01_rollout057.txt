import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n = len(initial_h)
    dx = 2.0 / n
    target_sum = n // 2  # 35 for n=70

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Perturb
        h_perturbed = best_h.copy()
        p = 0.1
        for i in range(n):
            if np.random.rand() < p:
                h_perturbed[i] = 1 - h_perturbed[i]
        # Repair
        sum_p = np.sum(h_perturbed)
        if sum_p > target_sum:
            excess = sum_p - target_sum
            idxs = np.where(h_perturbed == 1)[0]
            for j in range(excess):
                h_perturbed[idxs[j]] = 0
        elif sum_p < target_sum:
            needed = target_sum - sum_p
            idxs = np.where(h_perturbed == 0)[0]
            for j in range(needed):
                h_perturbed[idxs[j]] = 1
        # Evaluate
        c5_p = compute_c5(h_perturbed)
        if c5_p < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5_p

        # Shift +1
        non_zero = np.where(best_h == 1)[0]
        if len(non_zero) > 0:
            shifted_pos = (non_zero + 1) % n
            h_shifted = np.zeros(n)
            h_shifted[shifted_pos] = 1
            c5_s = compute_c5(h_shifted)
            if c5_s < best_c5:
                best_h = h_shifted.copy()
                best_c5 = c5_s

        # Shift -1
        non_zero = np.where(best_h == 1)[0]
        if len(non_zero) > 0:
            shifted_pos = (non_zero - 1) % n
            h_shifted = np.zeros(n)
            h_shifted[shifted_pos] = 1
            c5_s = compute_c5(h_shifted)
            if c5_s < best_c5:
                best_h = h_shifted.copy()
                best_c5 = c5_s

    return best_h, best_c5, n
