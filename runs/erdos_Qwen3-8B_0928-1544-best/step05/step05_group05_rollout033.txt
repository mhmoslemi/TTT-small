import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    required_sum = n_points / 2.0
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_candidates = [initial_h]
    else:
        initial_h = np.full(n_points, 0.5)
        initial_candidates = [initial_h]

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - required_sum

    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_candidates[0].copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        current_h = initial_candidates[np.random.randint(len(initial_candidates))]
        current_h = np.clip(current_h, 0.0, 1.0)
        current_h *= (required_sum / np.sum(current_h))
        result = minimize(
            fun=objective,
            x0=current_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        initial_candidates.append(result.x)
        if len(initial_candidates) > 5:
            initial_candidates.pop(0)

    return best_h, best_c5, n_points
