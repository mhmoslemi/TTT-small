    initial_h = initial_h.copy()
    sum_h = np.sum(initial_h)
    initial_h *= (ones_count / sum_h)
    initial_h = np.clip(initial_h, 0, 1)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def local_optimize(h, k):
        x = np.linspace(0, 2, n_points)
        h = h.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = ones_count / sum_h
        h *= scale
        h = np.clip(h, 0, 1)
        obj = lambda x: x[0] * (1 - x[1]) + x[1] * (1 - x[0])
        lags = np.correlate(h, 1 - h, mode='full')
        worst_k = np.argmax(lags)
        dx = 2.0 / n_points
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            for j in range(n_points):
                if (i + worst_k) % n_points == j:
                    A[i, j] = 1
        problem = cp.Problem(cp.Minimize(np.dot(h, 1 - h) * dx), [sum(h) == ones_count])
        problem.solve() if problem.is_dcp() else None
        return h.value

    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    best_h = initial_h.copy()
    best_c5 = evaluate(initial_h)
    time_used = 0
    max_iter = int(budget_s // 10)

    for _ in range(max_iter):
        scores = [evaluate(h) for h in population]
        sorted_indices = np.argsort(scores)
        parents = [population[i] for i in sorted_indices[:10]]
        new_candidates = []
        for h in parents:
            h = local_optimize(h, np.argmax(np.correlate(h, 1 - h, mode='full')))
            new_candidates.append(h)
        population = sorted(parents)[:population_size//2] + new_candidates[:population_size//2]
        current_best = min(zip(scores, population), key=lambda x: x[0])[1]
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best
        time_used += 10
        if time_used >= budget_s:
            break
    return best_h, best_c5, n_points
