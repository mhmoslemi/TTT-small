    start_time = time.time()

    for iteration in range(max_iterations):
        current_time = time.time()
        if current_time - start_time > budget_s:
            break

        dx = 2.0 / n_points
        max_val = 0.0
        k_star = 0
        for k in range(n_points):
            shifted = np.roll(h, k)
            current_val = np.dot(h, 1 - shifted)
            c5_candidate = current_val * dx
            if c5_candidate > max_val:
                max_val = c5_candidate
                k_star = k

        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()

        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_minus_k = (i - k_star) % n_points
            i_plus_k = (i + k_star) % n_points
            gradient[i] = (1.0 - h[i_minus_k]) - h[i_plus_k]

        h_new = h - learning_rate * gradient
        h_clipped = np.clip(h_new, 0, 1)
        sum_clipped = np.sum(h_clipped)

        if sum_clipped != 0:
            scale_factor = target_sum / sum_clipped
            h = h_clipped * scale_factor
        else:
            h = h_clipped

        if adapt_learning_rate:
            learning_rate *= 0.995

    return best_h, best_c5, n_points
