def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    sum_h = 35
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    
    def clip_rescale(h):
        h = np.clip(h, 0.0, 1.0)
        if np.sum(h) == 0:
            return np.zeros(n_points)
        scale = sum_h / np.sum(h)
        return h * scale
    
    h = clip_rescale(h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - sum_h
    
    def optimize_h(h_initial):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        res = minimize(objective, h_initial, method='SLSQP', 
                       constraints=[{'type': 'eq', 'fun': constraint_sum}],
                       bounds=bounds,
                       options={'ftol': 1e-8, 'maxiter': 100})
        return res.x
    
    best_h = h
    best_c5 = objective(best_h)
    time_used = 0
    max_iter = int(budget_s // 100)
    
    for _ in range(max_iter):
        perturbed = h + np.random.normal(0, 0.05, n_points)
        perturbed = clip_rescale(perturbed)
        new_h = optimize_h(perturbed)
        new_c5 = objective(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
        time_used += 100
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
