def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)

    def project_sum(h):
        h = h.copy()
        current_sum = np.sum(h)
        if np.isclose(current_sum, target_sum):
            return h
        delta = target_sum - current_sum
        if delta > 0:
            # Need to add delta to values < 1.0
            indices = np.where(h < 1.0)[0]
            if len(indices) == 0:
                return h
            max_possible = np.sum(1.0 - h[indices])
            if delta > max_possible:
                h[indices] = 1.0
                return h
            else:
                h[indices] += delta / len(indices)
        else:
            # Need to subtract (-delta) from values > 0.0
            indices = np.where(h > 0.0)[0]
            if len(indices) == 0:

            min_possible = np.sum(h[indices])
            if -delta > min_possible:
                h[indices] = 0.0
                return h
            else:
                h[indices] -= (-delta) / len(indices)
        return h
    
    initial_h = project_sum(initial_h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    max_iter = int(budget_s / 4.5)

        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = project_sum(h_perturbed)
        
        # Optimize
        result = minimize(
