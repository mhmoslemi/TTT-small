import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    required_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure sum constraint
    h = h * (required_sum / np.sum(h))
    
    # Initialize velocity and parameters
    velocity = np.zeros(n_points)
    beta = 0.9  # Momentum coefficient
    alpha = 0.1  # Initial learning rate
    step_decay = 0.999  # Learning rate decay
    eps = 1e-5  # Perturbation for finite differences
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Compute objective
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        
        # Compute gradient via finite differences
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Perturb and clip
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus[i] = np.clip(h_plus[i], 0.0, 1.0)
            # Renormalize
            h_plus = h_plus * (required_sum / np.sum(h_plus))
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * (2.0 / n_points)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus[i] = np.clip(h_minus[i], 0.0, 1.0)
            h_minus = h_minus * (required_sum / np.sum(h_minus))
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * (2.0 / n_points)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update velocity and h
        velocity = beta * velocity + (1 - beta) * grad
        h += alpha * velocity
        
        # Clip and renormalize
        h = np.clip(h, 0.0, 1.0)
        h = h * (required_sum / np.sum(h))
        
        # Decay learning rate
        alpha *= step_decay
        
        # Periodic perturbation to escape local minima
        if np.random.rand() < 0.01:  # 1% chance
            perturb = np.random.normal(0, 0.05, n_points)
            h += perturb
            h = np.clip(h, 0.0, 1.0)
            h = h * (required_sum / np.sum(h))
    
    # Final check
    corr = np.correlate(h, 1 - h, mode='full')
    c5_final = np.max(corr) * (2.0 / n_points)
    h = h * (required_sum / np.sum(h))
    
    return h, c5_final, n_points
