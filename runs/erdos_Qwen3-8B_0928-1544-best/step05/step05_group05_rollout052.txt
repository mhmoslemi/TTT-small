from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    bounds = [(0.0, 1.0)] * n_points

    def constraint_sum(h):
        return np.sum(h) - required_sum

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    max_iter = int(budget_s / 5.0)

    while iter_count < max_iter and time.time() - start_time < budget_s:

        result = minimize(

            x0=current_h,
            method='L-BFGS-B',

            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
