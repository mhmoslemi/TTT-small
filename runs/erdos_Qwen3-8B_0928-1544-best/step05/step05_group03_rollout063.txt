import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = initial_h.copy()
    best_c5 = evaluate(initial_h)
    time_used = 0
    max_iter = int(budget_s // 5)

    for _ in range(max_iter):
        # Evaluate and sort population
        scores = [evaluate(h) for h in population]
        idx = np.argsort(scores)
        sorted_pop = [population[i] for i in idx]
        parents = sorted_pop[:int(0.3 * population_size)]
        
        # Generate new candidates
        new_candidates = []
        for h in parents:
            # Random perturbation
            perturb = np.random.normal(0, 0.15, n_points)
            h_perturb = h + perturb
            h_perturb = np.clip(h_perturb, 0, 1)
            sum_h = np.sum(h_perturb)
            scale = ones_count / sum_h
            new_h = h_perturb * scale
            new_candidates.append(new_h)
        
        # Replace worst half
        replace_count = int(0.5 * population_size)
        new_population = sorted_pop[:population_size - replace_count] + new_candidates[:replace_count]
        population = new_population
        
        # Update best
        current_best = min(zip(scores, population), key=lambda x: x[0])[1]
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best
        
        time_used += 5
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
