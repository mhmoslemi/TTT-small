from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.full(n_points, 0.5)
    dx = 2.0 / n_points

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points

    num_initial = 15
    initial_population = [initial_h.copy()]
    for _ in range(num_initial - 1):
        perturb = np.random.choice([-0.25, 0.25], size=n_points)
        initial_population.append(initial_h + perturb)

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    max_iter = int(budget_s / 100)

    for _ in range(max_iter):
        current_h = initial_population[np.random.randint(len(initial_population))]
        result = minimize(
            fun=objective,
            x0=current_h,
