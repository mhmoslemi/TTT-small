import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.full(n_points, 0.5)

    result = differential_evolution(

        popsize=20,
        maxiter=int(budget_s / 100),
        tol=1e-5,
        mutation=(0.5, 1),
        recombination=0.9,
        disp=False
    )
    
    best_h = result.x
    best_c5 = objective(best_h) - 1e4 * np.abs(np.sum(best_h) - ones_count)
    
    return best_h, best_c5, n_points
