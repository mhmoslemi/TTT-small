import time

def run(seed=42, budget_s=1100, **kwargs):

    swarm += [np.random.uniform(0, 1, n_points) * (target_sum / np.random.uniform(0.9, 1.1, n_points).sum()) for _ in range(swarm_size - 1)]

    velocities = np.random.uniform(-vel_scale, vel_scale, (swarm_size, n_points))
    personal_best = [np.copy(h) for h in swarm]
    global_best = np.copy(personal_best[0])
    best_c5 = float('inf')

    def evaluate(h):

        return np.max(corr) * (2.0 / n_points)

        if time.time() - start_time > budget_s:
            break

                   c1 * r1 * (personal_best[i] - swarm[i]) +
                   c2 * r2 * (global_best - swarm[i]))
