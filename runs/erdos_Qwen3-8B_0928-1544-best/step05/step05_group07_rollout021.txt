import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 70
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points // 2):
            h[i*2] = 1.0
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.zeros(n_points)
        for i in range(n_points // 2):
            h[i*2] = 1.0
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * (2.0 / n_points)
    best_h = h.copy()
    max_iter = int(budget_s / 4)
    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        top_indices = np.argsort(corr)[::-1][:30]
        def objective(x):
            return np.sum(np.correlate(x, 1 - x, mode='full')[top_indices] ** 2)
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-9)
        if result.success:
            h = result.x
            new_corr = np.correlate(h, 1 - h, mode='full')
            new_c5 = np.max(new_corr) * (2.0 / n_points)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * (2.0 / n_points)
    return best_h, final_c5, n_points
