import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    h_initial = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def perturb(h, scale=0.2):
        h_new = h + np.random.normal(0, scale, n_points)
        h_new = np.clip(h_new, 0, 1)
        norm_factor = ones_count / np.sum(h_new)
        return h_new * norm_factor
    
    def harmonic_perturb(h):
        x = np.linspace(0, 2, n_points)
        h_new = 0.5 + 0.5 * np.sin(np.pi * x)
        h_new = np.clip(h_new, 0, 1)
        norm_factor = ones_count / np.sum(h_new)
        return h_new * norm_factor
    
    def refined_perturb(h, h_target):
        diff = h - h_target
        scale = np.std(diff)
        h_new = h + np.random.normal(0, scale * 0.6, n_points)
        h_new = np.clip(h_new, 0, 1)
        norm_factor = ones_count / np.sum(h_new)
        return h_new * norm_factor
    
    def smooth_harmonic(h):
        x = np.linspace(0, 2, n_points)
        h_new = 0.5 + 0.5 * np.sin(np.pi * x)
        h_new = np.clip(h_new, 0, 1)
        norm_factor = ones_count / np.sum(h_new)
        return h_new * norm_factor
    
    h = h_initial.copy()
    best_h = h
    best_c5 = evaluate(h)
    time_used = 0
    max_iter = int(budget_s // 5)
    
    for _ in range(max_iter):
        h_perturb = perturb(h)
        h_harmonic = harmonic_perturb(h)
        h_smooth = smooth_harmonic(h)
        h_candidates = [h_perturb, h_harmonic, h_smooth]
        
        for candidate in h_candidates:
            c = evaluate(candidate)
            if c < best_c5:
                best_h = candidate.copy()
                best_c5 = c
        
        h = refined_perturb(h, best_h)
        time_used += 5
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
