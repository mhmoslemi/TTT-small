import time

def run(seed=42, budget_s=1100, **kwargs):

        initial_pop.append(h)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = initial_pop[0].copy()
    best_c5 = evaluate(best_h)

        min_score = min(scores)

        selected = []
        for _ in range(2):

            selected.append(population[idx])
