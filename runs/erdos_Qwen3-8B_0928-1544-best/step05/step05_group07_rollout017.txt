            available = np.sum(1.0 - h_clipped)
            delta = target_sum - current_sum
            if available >= delta:
                non_1_mask = h_clipped < 1.0
                non_1 = h_clipped[non_1_mask]
                available_non_1 = 1.0 - non_1
                factor = delta / np.sum(available_non_1)
                h_clipped[non_1_mask] += factor * available_non_1
            return h_clipped
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    time_used = 0
    max_iter = int(budget_s / 4)
    
    for iter in range(max_iter):
        sigma = 0.1 * (1 - iter / max_iter)
        h_perturbed = best_h + np.random.normal(0, sigma, n_points)
        h_perturbed = project_h(h_perturbed, ones_count)
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        time_used += 4
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
