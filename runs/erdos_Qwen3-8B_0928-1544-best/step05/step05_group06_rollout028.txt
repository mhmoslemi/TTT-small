def compute_C5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_overlap = np.max(corr)
    dx = 2.0 / len(h)
    return max_overlap * dx

def compute_gradient(h, k_shift):
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        if 0 <= j + k_shift < n:
            grad[j] += 1.0 - h[j + k_shift]
        if 0 <= j - k_shift < n:
            grad[j] -= h[j - k_shift]
    return grad
