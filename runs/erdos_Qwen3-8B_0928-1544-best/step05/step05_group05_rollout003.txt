    while time.time() - start_time < budget_s:
        current_h = initial_candidates[np.random.randint(len(initial_candidates))]
        result = minimize(
            fun=objective,
            x0=current_h,
            method='L-BFGS-B',  # More efficient optimization
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        initial_candidates.append(result.x)
        initial_candidates = initial_candidates[-5:]

    return best_h, best_c5, n_points
