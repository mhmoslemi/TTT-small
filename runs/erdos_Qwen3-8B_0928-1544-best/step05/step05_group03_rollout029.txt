    best_h = initial_h.copy()
    best_c5 = evaluate(initial_h)
    time_used = 0
    max_iter = int(budget_s // 5)
    
    for _ in range(max_iter):
        scores = [evaluate(h) for h in population]
        sorted_indices = np.argsort(scores)
        sorted_pop = [population[i] for i in sorted_indices]
        parents = sorted_pop[:int(0.3 * population_size)]
        new_candidates = []
        
        for h in parents:
            perturb = np.random.normal(0, 0.1, n_points)
            h_perturb = h + perturb
            h_perturb = np.clip(h_perturb, 0, 1)
            sum_h = np.sum(h_perturb)
            scale = ones_count / sum_h
            new_h = h_perturb * scale
            new_candidates.append(new_h)
        
        replace_count = int(0.6 * population_size)
        population = sorted_pop[:population_size - replace_count] + new_candidates[:replace_count]
        current_best = min(zip(scores, population), key=lambda x: x[0])[1]
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best
        
        time_used += 5
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
