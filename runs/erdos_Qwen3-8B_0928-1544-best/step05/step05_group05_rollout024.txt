import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    required_sum = n_points / 2.0
    pop_size = 30
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def mutate(parent):
        noise = np.random.normal(0, 0.05, n_points)
        mutated = np.clip(parent + noise, 0.0, 1.0)
        sum_mutated = np.sum(mutated)
        if sum_mutated != 0:
            mutated = mutated * (required_sum / sum_mutated)
        return np.clip(mutated, 0.0, 1.0)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # Initialize population with initial_h and perturbed versions
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        perturbed = initial_h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed = perturbed * (required_sum / np.sum(perturbed))
        population.append(perturbed)
    
    best_h = population[0]
    best_c5 = evaluate(best_h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        fitness = np.array([evaluate(h) for h in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx]
        
        # Tournament selection
        parents = []
        for _ in range(pop_size // 2):
            indices = np.random.choice(np.arange(pop_size), 5, replace=False)
            winner = indices[np.argmin(fitness[indices])]
            parents.append(population[winner])
        
        # Generate offspring
        offspring = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1]
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            mutated = mutate(child)
            offspring.append(mutated)
        
        # Replace worst individuals
        combined = population + offspring
        fitness_combined = np.array([evaluate(h) for h in combined])
        sorted_indices = np.argsort(fitness_combined)
        population = [combined[i] for i in sorted_indices[:pop_size]]
    
    return best_h, best_c5, n_points
