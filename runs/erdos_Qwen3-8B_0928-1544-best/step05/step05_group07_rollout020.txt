        bounds = [(0.0, 1.0)] * n_points
        
        result = minimize(obj_func, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': cons_func})
        
        if result.success:
            best_h = result.x
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            overlaps = corr * dx
            current_c5 = np.max(overlaps)
            if current_c5 < best_c5:
                best_c5 = current_c5
        
        time_used += 4
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
