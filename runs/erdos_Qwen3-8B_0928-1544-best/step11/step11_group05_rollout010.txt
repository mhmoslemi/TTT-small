import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    population_size = 50

    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (target_sum / initial_sum)
        population = [initial_h_scaled]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (target_sum / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (target_sum / h_sum)
            population.append(h_scaled)

    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        h_scaled = h_clipped * (target_sum / np.sum(h_clipped)) if np.sum(h_clipped) != 0 else h_clipped
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()

    F = 0.8
    CR = 0.95
    evals = 0

    while time.time() - start_time < budget_s:
        for _ in range(population_size):
            indices = np.random.choice(population_size, 3, replace=False)
            a, b, c = indices
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[_])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0, 1)

            if u_sum == 0:

            u_scaled = u * (target_sum / u_sum)

            new_c5 = objective(u_scaled)
            if new_c5 < objective(population[_]):
                population[_] = u_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u_scaled.copy()
        evals += population_size

    return best_h, best_c5, n_points
