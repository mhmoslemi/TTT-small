import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    swarm_size = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    local_opt_interval = 100
    local_tol = 1e-6

    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        positions = [initial_h_scaled]
        velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(swarm_size - 1)]
        for _ in range(swarm_size - 1):
            h = np.random.uniform(0.4, 0.6, n_points)
            h_sum = np.sum(h)
            h_scaled = h * (ones_count / h_sum)
            positions.append(h_scaled)
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        positions = []
        velocities = []
        for _ in range(swarm_size):
            h = np.random.uniform(0.4, 0.6, n_points)
            h_sum = np.sum(h)
            h_scaled = h * (ones_count / h_sum)
            positions.append(h_scaled)
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))

    personal_best = [pos.copy() for pos in positions]
    personal_best_values = [objective(pos) for pos in positions]
    global_best = positions[0].copy()
    global_best_value = personal_best_values[0]

    start_time = time.time()

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def clip_and_rescale(h):
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            return np.zeros(n_points)
        return h * (ones_count / h_sum)

    def local_optimize(h):
        def local_obj(x):
            return objective(x)
        constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count})
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        res = minimize(local_obj, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=local_tol)
        return res.x if res.success else h

    while time.time() - start_time < budget_s:
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + c1 * r1 * (personal_best[i] - positions[i]) + c2 * r2 * (global_best - positions[i])
            positions[i] = positions[i] + velocities[i]
            positions[i] = clip_and_rescale(positions[i])
            current_value = objective(positions[i])
            if current_value < personal_best_values[i]:
                personal_best[i] = positions[i].copy()
                personal_best_values[i] = current_value
                if current_value < global_best_value:
                    global_best = positions[i].copy()
                    global_best_value = current_value

        if (time.time() - start_time) > budget_s - 60:
            break

        if (time.time() - start_time) % local_opt_interval < 1e-6:
            global_best = local_optimize(global_best)

    return global_best, global_best_value, n_points

def objective(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n_points)
