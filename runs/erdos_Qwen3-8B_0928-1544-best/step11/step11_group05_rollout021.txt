                h = np.random.rand(n_points) * 0.6
            h_scaled = h * (target_sum / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.6
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.6
            h_scaled = h * (target_sum / h_sum)
            population.append(h_scaled)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = objective(population[0])
    
    start_time = time.time()
    
    F = 0.7  # Mutation factor

    switch_threshold = 1e-5  # Switch to BFGS if progress stalls
    mutation_rate = 0.2
    
    def bfgs_optimize(h):
        bounds = np.array([[0.0, 1.0]] * n_points)
        res = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, options={'maxiter': 200})
        return res.x
    
    while time.time() - start_time < budget_s:
        for i in range(population_size):
            if np.random.rand() < mutation_rate:
                indices = np.random.choice(population_size, 3, replace=False)
                a, b, c = indices

                v = a_vec + F * (b_vec - c_vec)
                
                u = np.copy(population[i])

                    if np.random.rand() < CR:
                        u[j] = v[j]
                
                u = np.clip(u, 0.0, 1.0)
                u_sum = np.sum(u)
                u_scaled = u * (target_sum / u_sum)
                u_scaled = np.clip(u_scaled, 0.0, 1.0)
                
                new_c5 = objective(u_scaled)

                    if new_c5 < best_c5:
