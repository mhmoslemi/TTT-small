            a, b, c = indices
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = project(u)
            new_c5 = objective_de(u)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = u.copy()
            if new_c5 < objective_de(population[i]):
                population[i] = u.copy()
        if iterations % 50 == 0:
            F = 0.8 if F < 0.8 else 0.5
            CR = 0.95 if CR < 0.95 else 0.8

    def objective_LBFGS(h):
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        penalty = 1e6 * (np.sum(h) - ones_count)**2
        return c5 + penalty

    if best_h is None:
        best_h = np.random.rand(n_points) * 0.5
        best_h_sum = np.sum(best_h)
        best_h = best_h * (ones_count / best_h_sum)

    res = minimize(
        objective_LBFGS,
        best_h,
        bounds=[(0, 1) for _ in range(n_points)],
        method='L-BFGS-B',
        options={'maxiter': 100}
    )
    best_h = res.x
    best_c5 = objective_de(best_h)
    return best_h, best_c5, n_points
