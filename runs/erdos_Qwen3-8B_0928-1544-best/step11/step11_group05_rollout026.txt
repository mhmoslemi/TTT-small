import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    initial_guess = None

    if 'initial_h_values' in globals():
        initial_guess = initial_h_values.copy()
        initial_guess_sum = np.sum(initial_guess)
        initial_guess = initial_guess * (target_sum / initial_guess_sum)
    else:
        initial_guess = np.random.rand(n_points) * 0.5
        initial_guess_sum = np.sum(initial_guess)
        initial_guess = initial_guess * (target_sum / initial_guess_sum)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        penalty = 1000 * np.abs(np.sum(h) - target_sum)
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds, options={'maxiter': 1000, 'ftol': 1e-8})

    elapsed = time.time() - start_time if 'start_time' in locals() else 0
    if elapsed >= budget_s:

    return (result.x, result.fun, n_points)
