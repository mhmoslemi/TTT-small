import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    population_size = 100
    best_c5 = float('inf')
    best_h = None

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        population = [initial_h_scaled]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()

    # PSO parameters
    inertia = 0.9
    c1, c2 = 2.0, 2.0
    max_velocity = 0.5
    velocities = np.random.uniform(-max_velocity, max_velocity, (population_size, n_points))

    while time.time() - start_time < budget_s:
        for i in range(population_size):
            # Update velocity and position
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] += (c1 * r1 * (population[i] - best_h) + c2 * r2 * (np.array([best_h] * population_size)[i] - population[i]))
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            population[i] += velocities[i]
            population[i] = np.clip(population[i], 0, 1)
            population[i] *= (ones_count / np.sum(population[i]))
            c5 = objective(population[i])
            if c5 < best_c5:
                best_c5 = c5
                best_h = population[i].copy()
        # Reduce inertia weight
        inertia -= (0.9 - 0.1) / (budget_s - (time.time() - start_time))

    return best_h, best_c5, n_points
