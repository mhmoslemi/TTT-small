    w = 0.7
    c1, c2 = 2.0, 2.0

    global_best = np.array(population[0])

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

        if h_sum == 0:
            return h
