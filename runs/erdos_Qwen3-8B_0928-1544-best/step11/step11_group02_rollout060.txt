import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n = 100
    dx = 2.0 / n

    best_c5 = float('inf')
    best_h = np.zeros(n)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')[n-1:2*n-1]) * dx
