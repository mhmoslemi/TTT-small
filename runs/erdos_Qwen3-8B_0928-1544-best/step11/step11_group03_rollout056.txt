import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    dx = 2.0 / n_points

    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h = h * (ones_count / h_sum)
        h = np.clip(h, 0.0, 1.0)

    # Transform h to unconstrained z
    z = np.log(h / (1 - h))
    z = np.clip(z, -6, 6)

    # Objective function
    def obj(z):
        z_clipped = np.clip(z, -6, 6)
        h = 1 / (1 + np.exp(-z_clipped))
        h_sum = np.sum(h)
        scaling = ones_count / h_sum
        h_scaled = h * scaling
        h_scaled = np.clip(h_scaled, 0.0, 1.0)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    # Phase 1: L-BFGS optimization
    result = minimize(obj, z, method='L-BFGS-B', jac=None, options={'maxiter': 300})
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        z_best = result.x
    else:
        # Phase 2: Nelder-Mead refinement
        try:
            nelder_result = minimize(obj, result.x, method='Nelder-Mead', options={'maxiter': 100})
            z_best = nelder_result.x
        except:
            z_best = result.x
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            z_best = result.x

    # Final conversion to h and evaluation
    z_clipped = np.clip(z_best, -6, 6)
    h = 1 / (1 + np.exp(-z_clipped))
    h_sum = np.sum(h)
    scaling = ones_count / h_sum
    h_scaled = h * scaling
    h_scaled = np.clip(h_scaled, 0.0, 1.0)
    corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
    final_c5 = np.max(corr) * dx
    return h_scaled, final_c5, n_points
