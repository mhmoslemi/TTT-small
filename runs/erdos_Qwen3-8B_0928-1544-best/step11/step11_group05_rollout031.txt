import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = 35
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def perturb(h, idx, delta):
        new_h = h.copy()
        new_h[idx] += delta
        new_sum = np.sum(new_h)
        if new_sum > target_sum:
            diff = new_sum - target_sum
            new_h[np.argmax(new_h)] -= diff
        elif new_sum < target_sum:
            diff = target_sum - new_sum
            new_h[np.argmin(new_h)] += diff
        new_h = np.clip(new_h, 0, 1)
        return new_h
    
    best_h = h.copy()
    best_c5 = compute_overlap(h)
    for _ in range(500):
        for i in range(n_points):
            for delta in [0.01, -0.01]:
                trial_h = perturb(h, i, delta)
                trial_c5 = compute_overlap(trial_h)
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial_h.copy()
                    break
    return best_h, best_c5, n_points
