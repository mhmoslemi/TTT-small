import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        h_sum = np.sum(h)
        h = h * (target_sum / h_sum)
    else:
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        h = h * (target_sum / h_sum)
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    
    start_time = time.time()
    
    def obj(x):
        return _compute_c5(x)
    
    def _compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def _scale(h):
        s = np.sum(h)
        return np.clip(h * (target_sum / s), 0, 1)
    
    while time.time() - start_time < budget_s:
        h = best_h.copy()
        res = minimize(obj, h, method='BFGS', bounds=[(0,1)]*n_points)
        h = _scale(res.x)
        c5 = obj(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        h_mut = best_h.copy()
        for i in range(n_points):
            h_mut[i] += np.random.normal(0, 0.1)
            h_mut[i] = np.clip(h_mut[i], 0, 1)
        h_mut = _scale(h_mut)
        c5_mut = obj(h_mut)
        if c5_mut < best_c5:
            best_c5 = c5_mut
            best_h = h_mut.copy()
    
    return best_h, best_c5, n_points
