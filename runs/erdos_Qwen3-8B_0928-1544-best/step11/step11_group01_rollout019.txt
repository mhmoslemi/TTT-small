import numpy as np
import time
import cvxpy as cp
import matplotlib.pyplot as plt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.linspace(0, 1, n_points)
    h = initial_h.copy()
    h = (h * (ones_count / np.sum(h))).tolist()
    h = np.array(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        H = np.fft.fft(h)
        tau = cp.Variable()
        constraints = [cp.abs(H[k]) <= tau for k in range(n_points)]
        prob = cp.Problem(cp.Minimize(tau), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-7, max_iter=1000)
            if prob.status != 'OPTIMAL':
                break
            tau_opt = float(tau.value)
            new_H = np.zeros_like(H)
            for k in range(n_points):
                new_H[k] = H[k] / np.abs(H[k]) * tau_opt if np.abs(H[k]) > 1e-10 else H[k]
            h_new = np.fft.ifft(new_H).real
            h_new = np.clip(h_new, 0.0, 1.0)
            h_sum = np.sum(h_new)
            if h_sum > 1e-6:
                h_new = h_new * (ones_count / h_sum)
            h = h_new
            c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        except:
            break

    return best_h, best_c5, n_points
