import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    dx = 2.0 / n_points

    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.uniform(0.4, 0.6, n_points)
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Compute current correlation
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        worst_shift_index = np.argmax(corr)
        delta = worst_shift_index - (n_points - 1)
        
        # Define objective function for current delta
        def objective_sub(h):
            total = 0.0
            for i in range(n_points):
                j = i + delta
                if 0 <= j < n_points:
                    total += h[i] * (1 - h[j])
            return total
        
        # Optimization settings
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = ({'type': 'eq', 'fun': lambda h_val: np.sum(h_val) - ones_count})
        
        # Run SLSQP optimization
        result = minimize(
            objective_sub,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            tol=1e-8
        )
        
        if result.success:
            h = result.x
            new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    
    return best_h, best_c5, n_points
