from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.random.uniform(0.4, 0.6, n_points)
    
    # Scale to meet the constraint
    h = h_initial * (ones_count / np.sum(h_initial))

    def objective(x):
        corr = correlate(x, 1 - x, mode='full')

    # Constraints and bounds
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count})
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def evaluate_and_refine(h_candidate):
        res = minimize(objective, h_candidate, method='SLSQP',
                       bounds=bounds, constraints=constraints, tol=1e-6)
        return res.x, res.fun

    # Initial optimization
    best_h, best_c5 = evaluate_and_refine(h)
    elapsed = 0

    # Timestamp to track time
    start_time = time.time()

    # Additional refinement cycles
    for _ in range(5):
        # Perturb current best solution
        perturbation = np.random.uniform(-0.05, 0.05, n_points)

        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)

        h_candidate, c5 = evaluate_and_refine(perturbed_h)
        if c5 < best_c5:

        elapsed = time.time() - start_time
        if elapsed >= budget_s:
