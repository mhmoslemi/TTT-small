import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():

        h0_sum = np.sum(h0)
        h0 = h0 * (target_sum / h0_sum)
        h0 = np.clip(h0, 0.0, 1.0)

    def objective(h):

        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)

    elapsed_time = time.time() - start_time

    h_best = result.x
    c5_best = objective(h_best)
    return h_best, c5_best, n_points
