import numpy as np
import time
import scipy.fft as fft
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def perturb(h):
        h = h.copy()
        idxs = np.random.choice(n_points, size=10, replace=False)
        for i in idxs:
            h[i] += np.random.uniform(-0.1, 0.1)
            h[i] = np.clip(h[i], 0, 1)
        return h
    
    population = [initial_h]
    for _ in range(99):
        if np.random.rand() < 0.3:
            h = perturb(population[np.random.choice(len(population))])
        else:
            h = np.random.rand(n_points) * 0.5
            h = h * (ones_count / np.sum(h))
        population.append(h)
    
    best_h = initial_h
    best_c5 = evaluate(best_h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        new_pop = []
        for _ in range(len(population)):
            parent1 = population[np.random.choice(len(population))]
            parent2 = population[np.random.choice(len(population))]
            child = (parent1 + parent2) / 2
            if np.random.rand() < 0.1:
                child = perturb(child)
            child = np.clip(child, 0, 1)
            child = child * (ones_count / np.sum(child))
            new_pop.append(child)
        
        population = new_pop
        for h in population:
            curr_c5 = evaluate(h)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h
    
    return best_h, best_c5, n_points
