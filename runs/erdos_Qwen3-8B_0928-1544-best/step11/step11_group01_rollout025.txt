import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.linspace(0, 1, n_points)
