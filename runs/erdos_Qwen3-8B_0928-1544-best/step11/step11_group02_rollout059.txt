def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = 50  # n_points // 2
    population_size = 100

    if 'initial_h_values' in globals():

            h = np.random.rand(n_points) * 0.5
