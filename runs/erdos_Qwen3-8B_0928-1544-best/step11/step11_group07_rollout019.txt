import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 50
    mutation_rate = 0.15
    mutation_scale = 0.1
    cooling_factor = 0.995
    tabu_size = 20
    tabu_list = []
    elite_ratio = 0.05

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        if np.sum(initial_h) == 0:
            initial_h = np.random.rand(n_points) * 0.5
        initial_h_scaled = initial_h * (ones_count / np.sum(initial_h))
        population = [initial_h_scaled]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            if np.sum(h) == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / np.sum(h))
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            if np.sum(h) == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / np.sum(h))
            population.append(h_scaled)

    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h) for h in population]
        total_fitness = np.sum(fitness)
        probabilities = fitness / total_fitness
        indices = np.random.choice(range(population_size), size=population_size, p=probabilities)
        parents = [population[i] for i in indices]

        offspring = []
        for i in range(0, population_size, 2):
            h1 = parents[i]
            if i + 1 < population_size:
                h2 = parents[i + 1]
            else:
                h2 = h1
            alpha = np.random.uniform(0.4, 0.6)
            child = alpha * h1 + (1 - alpha) * h2
            offspring.append(child)

        for j in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_scale, n_points)
                offspring[j] += noise
                offspring[j] = np.clip(offspring[j], 0, 1)
                offspring[j] = offspring[j] * (ones_count / np.sum(offspring[j]))

        new_population = []
        for h in offspring:
            is_tabu = False
            for t in tabu_list:
                if np.linalg.norm(h - t) < 0.02:
                    is_tabu = True
                    break
            if not is_tabu:
                new_population.append(h)
                if len(tabu_list) >= tabu_size:
                    tabu_list.pop(0)
                tabu_list.append(h)

        while len(new_population) < population_size:
            h = np.random.rand(n_points) * 0.5
            if np.sum(h) == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / np.sum(h))
            new_population.append(h_scaled)

        population = new_population
        mutation_scale *= cooling_factor

        new_fitness = [compute_c5(h) for h in population]
        sorted_indices = np.argsort(new_fitness)
        elite_count = int(population_size * elite_ratio)
        population = [population[i] for i in sorted_indices[:elite_count]]

    return best_h, best_c5, n_points
