import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 50
    temp_initial = 1.0
    alpha = 0.995
    p_perturb = 0.05
    mutation_scale = 0.05
    cooldown = 200
    max_iter = int(budget_s / 100)

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        population = [initial_h_scaled]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            h_scaled = h * (ones_count / h_sum) if h_sum > 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
            population.append(h_scaled)
    else:
        population = [np.random.rand(n_points) * 0.5 * (ones_count / 0.5) for _ in range(population_size)]
    
    personal_best = [np.copy(p) for p in population]
    best_h = np.copy(population[0])
    best_c5 = np.inf

    start_time = time.time()
    temp = temp_initial
    cooldown_count = 0

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(population_size):
            h = population[i]
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
            h_scaled = h * (ones_count / h_sum)
            new_h = h_scaled.copy()
            if np.random.rand() < p_perturb:
                pert = np.random.uniform(-mutation_scale, mutation_scale, n_points)
                new_h += pert
                new_h = np.clip(new_h, 0, 1)
                new_h_sum = np.sum(new_h)
                if new_h_sum == 0:
                    new_h = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)

            population[i] = new_h
        for i in range(population_size):
