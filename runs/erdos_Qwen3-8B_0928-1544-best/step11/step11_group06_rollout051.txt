        h_scaled = h * (ones_count / h_sum)

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def constraint(h_vec):
        return np.sum(h_vec) - ones_count

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = minimize(objective, x0=h_scaled, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, tol=1e-8)
    best_h = result.x
    best_c5 = result.fun

    # Perturb and optimize
    for _ in range(20):
        delta = np.random.uniform(-0.02, 0.02, n_points)
        perturbed = best_h + delta
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum > 0:
            perturbed = perturbed * (ones_count / perturbed_sum)
        result = minimize(objective, x0=perturbed, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, tol=1e-8)
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    return best_h, best_c5, n_points
