import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2  # 35

    # Initialize solution with initial_h_values or random
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        sum_h = np.sum(h)
        if np.isclose(sum_h, 0):
            h = np.random.rand(n_points) * 0.5
        h = h * (target_sum / sum_h)
    else:
        h = np.random.rand(n_points) * 0.5
        sum_h = np.sum(h)
        if np.isclose(sum_h, 0):
            h = np.random.rand(n_points) * 0.5
        h = h * (target_sum / sum_h)

    # Objective function
    def objective(h_vec):
        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        return np.max(corr) * (2.0 / n_points)

    # Define bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

    # Optimization with hybrid strategy
    result = opt.minimize(
        fun=objective,
        x0=h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        tol=1e-6,
        options={'maxiter': 500, 'disp': False}
    )

    best_h = result.x
    best_c5 = objective(best_h)
    # Clip and scale after optimization
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    if not np.isclose(sum_best, target_sum):
        best_h = best_h * (target_sum / sum_best)

    return best_h, best_c5, n_points
