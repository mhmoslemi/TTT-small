            if h_sum == 0:
                h = np.random.uniform(0.4, 0.6, n_points)
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.uniform(0.4, 0.6, n_points)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.uniform(0.4, 0.6, n_points)
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)

    # Objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])

    start_time = time.time()

    F = 1.0  # Higher mutation factor
    CR = 0.98  # Higher crossover rate

    def evaluate_population():
        nonlocal best_c5, best_h
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec = population[a]
            b_vec = population[b]
            c_vec = population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            new_c5 = objective(u_scaled)
            if new_c5 < objective(population[i]):
                population[i] = u_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u_scaled.copy()

    while time.time() - start_time < budget_s:
        evaluate_population()

            def local_objective(x):
                h = x
                corr = np.correlate(h, 1 - h, mode='full')

            constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count})
