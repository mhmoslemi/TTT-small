from scipy.optimize import minimize

        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (ones_count / np.sum(h)) if np.sum(h) != 0 else np.full(n_points, 1.0 / n_points) * ones_count
        return objective(h)
    
    def bounds():
        return [(0.0, 1.0) for _ in range(n_points)]
    
    def minimize_c5():
        result = minimize(
            lambda x: evaluate(x),
            x0=initial_h,
            method='SLSQP',
            bounds=bounds(),
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'ftol': 1e-8, 'maxiter': 1000}
        )
        return result
    
    start_time = time.time()
