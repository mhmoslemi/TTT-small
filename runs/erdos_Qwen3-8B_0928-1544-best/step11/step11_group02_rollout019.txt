        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (target_sum / initial_h_sum)
        population = [initial_h_scaled]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (target_sum / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (target_sum / h_sum)
            population.append(h_scaled)

    velocities = [np.random.uniform(-max_velocity, max_velocity, size=n_points) for _ in range(population_size)]
    personal_best = [p.copy() for p in population]
    personal_best_c5 = [objective(p) for p in population]
    best_c5 = min(personal_best_c5)
    best_h = population[personal_best_c5.index(best_c5)].copy()

    start_time = time.time()

    while time.time() - start_time < budget_s:
        inertia_weight = max(inertia_weight - (inertia_decay / (budget_s / 2)), 0.1)
        for i in range(population_size):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocity = velocities[i] + c1 * r1 * (personal_best[i] - population[i]) + c2 * r2 * (best_h - population[i])
            velocity = np.clip(velocity, -max_velocity, max_velocity)
            velocities[i] = velocity
            population[i] += velocity
            population[i] = np.clip(population[i], 0, 1)
            current_sum = np.sum(population[i])
            if current_sum == 0:
                h = np.random.rand(n_points) * 0.5
                h_sum = np.sum(h)
                if h_sum == 0:
                    h = np.random.rand(n_points) * 0.5
                population[i] = h * (target_sum / h_sum)
            else:
                population[i] = population[i] * (target_sum / current_sum)
            current_c5 = objective(population[i])
            if current_c5 < personal_best_c5[i]:
                personal_best[i] = population[i].copy()
                personal_best_c5[i] = current_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = population[i].copy()
    return best_h, best_c5, n_points
