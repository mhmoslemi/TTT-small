            valid_i = np.arange(n_points)[valid_mask]
            return np.sum(x[valid_i] * (1.0 - x[valid_i + k])) * dx
        def grad_func(x):
            return grad
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = minimize(objective, h, method='SLSQP', jac=grad_func, bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
                h = h * (ones_count / np.sum(h))
            else:
                h = h * (ones_count / h_sum)
            new_c5 = np.max(compute_overlap(h))
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    return best_h, best_c5, n_points
