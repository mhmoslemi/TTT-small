        corr = np.correlate(h, 1 - h, mode='full')

        return result.x if result.success else h
    
    # Coordinate descent with line search
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - len(best_h) + 1
        
        # Compute gradient for max lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + lag) % n_points

        # Update in the direction of -gradient
