        return np.max(corr) * dx
    
    def scale(x):
        t = np.sum(x)
        if t == 0:
            return x
        return x * (target_sum / t)
    
    def bounds_func(x):
        return np.clip(scale(x), 0.0, 1.0)
    
    def evaluate(x):
        return obj(bounds_func(x))
    
    def mutate(parent, scale=0.4):
        perturb = np.random.randn(n_points) * scale
        child = parent + perturb
        child = scale(child)
        return np.clip(child, 0.0, 1.0)
    
    def crossovers(parent, child, p=0.9):
        return np.where(np.random.rand(n_points) < p, child, parent)
    
    population = [h]
    for _ in range(49):
        new_h = mutate(population[0])
        population.append(new_h)
    
    best_h = population[0]
    best_c5 = evaluate(population[0])
    
    while time.time() - start_time < budget_s:
        new_population = []
        for i in range(len(population)):
            a, b, c = np.random.choice(len(population), 3, replace=False)
            p = population[i]
            p1, p2, p3 = population[a], population[b], population[c]
            child = mutate(p1)
            child = crossovers(p, child)
            child = bounds_func(child)
            new_c5 = evaluate(child)
            if new_c5 < evaluate(p):
                population[i] = child
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = child
        population = [np.clip(scale(p), 0.0, 1.0) for p in population]
    
    return best_h, best_c5, n_points
