import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    HM_size = 50
    HM = []
    for _ in range(HM_size):

        h += noise
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.zeros(n_points)
        else:
            h = h * (ones_count / h_sum)
        HM.append(h)
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = initial_h.copy()
    best_c5 = compute_C5(initial_h)
    BW = 0.2
    time_start = time.time()
    
    while time.time() - time_start < budget_s:
        new_h = np.random.choice(HM)
        adjusted = new_h + np.random.uniform(-BW, BW, n_points)
        adjusted = np.clip(adjusted, 0, 1)
        adjusted_sum = np.sum(adjusted)
        if adjusted_sum == 0:
            adjusted = np.zeros(n_points)
        else:
            adjusted = adjusted * (ones_count / adjusted_sum)
        current_c5 = compute_C5(adjusted)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = adjusted.copy()
        if current_c5 < np.min([compute_C5(h) for h in HM]):
            worst_idx = np.argmin([compute_C5(h) for h in HM])
            HM[worst_idx] = adjusted.copy()
        if (time.time() - time_start) % 100 < 1:
            BW *= 0.95
    
    return best_h, best_c5, n_points
