import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()

    HM = []
    for _ in range(HM_size):
        h = initial_h.copy()

        h = np.clip(h, 0, 1)

        if sum_h > 0:
            h = h * (ones_count / sum_h)
        HM.append(h)

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    
    iter_count = 0
    while time.time() - start_time < budget_s:
        iter_count += 1
        new_h = np.zeros(n_points)
        for i in range(n_points):
            idx = np.random.choice(HM_size)
            new_h[i] = HM[idx][i] + np.random.uniform(-BW, BW)
            new_h[i] = np.clip(new_h[i], 0, 1)
        sum_h = np.sum(new_h)
        if sum_h == 0:
            new_h = np.zeros(n_points)
        else:
            new_h = new_h * (ones_count / sum_h)
        
        current_c5 = evaluate_erdos_solution(new_h)
        
        worst_idx = np.argmin([evaluate_erdos_solution(h) for h in HM])
        if current_c5 < evaluate_erdos_solution(HM[worst_idx]):
            HM[worst_idx] = new_h.copy()
        
        if current_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = current_c5
        
        if iter_count % 100 == 0:
            BW *= 0.95
    
    return best_h, best_c5, n_points
