import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    population_size = 120  # Larger population for more diverse exploration
    
    # Initialize population
    if 'initial_h_values' in globals():

            h = initial_h_scaled + np.random.uniform(-0.05, 0.05, n_points)
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.uniform(0.4, 0.6, n_points)
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.uniform(0.4, 0.6, n_points)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.uniform(0.4, 0.6, n_points)
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    
    # Objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = objective(population[0])
    
    start_time = time.time()
    
    F = 1.0  # Increased mutation factor for stronger exploration
    CR = 0.95  # High crossover probability to diversify
    
    def evaluate_population():
        nonlocal best_c5, best_h
        for i in range(population_size):
            # Randomly select three distinct individuals
            indices = np.random.choice(population_size, 3, replace=False)
            a, b, c = indices
            a_vec = population[a]
            b_vec = population[b]
            c_vec = population[c]
            
            # Mutate: v = a + F*(b - c)
            v = a_vec + F * (b_vec - c_vec)
            
            # Crossover: u is a mix of v and current individual i
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            
            # Clip to [0,1]
            u = np.clip(u, 0.0, 1.0)
            
            # Scale to meet sum constraint
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            
            # Evaluate
            new_c5 = objective(u_scaled)
            
            # Check if improvement
            if new_c5 < objective(population[i]):
                population[i] = u_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u_scaled.copy()
    
    # Run the evolution process
    while time.time() - start_time < budget_s:
        evaluate_population()
        # Apply local optimization every 50 iterations
        if (time.time() - start_time) % 100 < 10:
