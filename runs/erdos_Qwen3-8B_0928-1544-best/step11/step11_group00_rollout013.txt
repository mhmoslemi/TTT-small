import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    swarm_size = 50
    inertia_start, inertia_end = 0.8, 0.2
    phi_p, phi_g = 2.0, 2.0
    vel_max = 0.5

    if 'initial_h_values' in globals():

        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        population = [initial_h_scaled]
        for _ in range(swarm_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(swarm_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    positions = np.array([p.copy() for p in population])
    velocities = np.random.rand(swarm_size, n_points) * vel_max
    personal_best = positions.copy()
    personal_best_val = np.array([objective(p) for p in positions])
    global_best = positions[0].copy()
    global_best_val = objective(positions[0])

    start_time = time.time()
    inertia = inertia_start
    stagnation = 0

    while time.time() - start_time < budget_s:
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = inertia * velocities[i] + phi_p * r1 * (personal_best[i] - positions[i]) + phi_g * r2 * (global_best - positions[i])
            velocities[i] = np.clip(velocities[i], -vel_max, vel_max)
            positions[i] = np.clip(positions[i] + velocities[i], 0.0, 1.0)
            pos_sum = np.sum(positions[i])
            if pos_sum == 0:
                continue
            positions[i] = positions[i] * (ones_count / pos_sum)
            current_val = objective(positions[i])
            if current_val < personal_best_val[i]:
                personal_best[i] = positions[i].copy()
                personal_best_val[i] = current_val
                if current_val < global_best_val:
                    global_best = positions[i].copy()
                    global_best_val = current_val
                    best_h = positions[i].copy()
                    best_c5 = current_val
            stagnation += 1
            if stagnation > 500:
                inertia = max(inertia_start, inertia - 0.001)
                stagnation = 0

    return best_h, best_c5, n_points
