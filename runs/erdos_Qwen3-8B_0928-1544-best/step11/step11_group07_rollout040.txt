import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 0.5
    h = h * (ones_count / np.sum(h))
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    learning_rate = 0.05
    max_iterations = int(budget_s / (n_points * 10))
    epsilon = 1e-4
    for _ in range(max_iterations):
        correlate = np.correlate(h, 1 - h, mode='full')
        overlap = correlate * (2.0 / n_points)
        k_max = np.argmax(overlap)
        grads = np.zeros_like(h)
        for i in range(n_points):
            pert = h.copy()
            pert[i] += epsilon
            pert_sum = np.sum(pert)
            if pert_sum == 0:
                pert = h.copy()
            else:
                pert = pert * (ones_count / pert_sum)
            corr_pert = np.correlate(pert, 1 - pert, mode='full')
            delta = corr_pert[k_max] - overlap[k_max]
            grads[i] = delta / epsilon
        h -= learning_rate * grads
        h = np.clip(h, 0, 1)
        h = h * (ones_count / np.sum(h))
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * (2.0 / n_points))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5, n_points
