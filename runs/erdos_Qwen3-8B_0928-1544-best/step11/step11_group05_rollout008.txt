import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    population_size = 100  # Increased population for better exploration
    
    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        # Scale to meet sum constraint
        initial_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (target_sum / initial_sum)
        population = [initial_h_scaled]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (target_sum / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (target_sum / h_sum)
            population.append(h_scaled)
    
    # Objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = objective(population[0])
    
    start_time = time.time()
    
    F = 0.6  # Reduced mutation factor for finer adjustments
    CR = 0.95  # High crossover rate to ensure diversity
    
    while time.time() - start_time < budget_s:
        for i in range(population_size):
            # Select three distinct individuals
            indices = np.random.choice(population_size, 3, replace=False)
            a, b, c = indices
            a_vec = population[a]
            b_vec = population[b]
            c_vec = population[c]
            
            # Mutate: v = a + F*(b - c)
            v = a_vec + F * (b_vec - c_vec)
            
            # Crossover: u is a mix of v and current individual i
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            
            # Clip to [0,1]
            u = np.clip(u, 0.0, 1.0)
            
            # Scale to meet sum constraint
            u_sum = np.sum(u)
            if u_sum == 0:
                continue  # Skip if sum is zero (unlikely)
            u_scaled = u * (target_sum / u_sum)
            u_scaled = np.clip(u_scaled, 0.0, 1.0)
            
            # Evaluate
            new_c5 = objective(u_scaled)
            
            # Check if improvement
            if new_c5 < objective(population[i]):
                population[i] = u_scaled
                if new_c5 < best_c5:
