import time

def run(seed=42, budget_s=1100, **kwargs):

    CR_start, CR_end = 0.5, 0.9
    stagnation = 0
    max_stagnation = 50

    if 'initial_h_values' in globals():

        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        population = [initial_h_scaled]
        for _ in range(population_size - 1):
            h = initial_h_scaled + np.random.randn(n_points) * 0.3
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                continue
            population.append(h * (ones_count / h_sum))
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.6
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.6
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()
    F = F_start
    CR = CR_start

    while time.time() - start_time < budget_s:
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:

            u_scaled = u * (ones_count / u_sum)
            new_c5 = objective(u_scaled)
            if new_c5 < best_c5:
