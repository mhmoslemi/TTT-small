                        child = np.full(n_points, 1.0 / n_points) * target_sum
                    else:
                        scaling_factor = target_sum / current_sum
                        child = child * scaling_factor
            
            new_population.append(child)
        
        population = new_population
    
    return best_h, best_c5, n_points
