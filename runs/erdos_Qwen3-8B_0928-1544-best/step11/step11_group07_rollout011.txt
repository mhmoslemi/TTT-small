        current_c5 = [compute_c5(h) for h in population]
        best_idx = np.argmin(current_c5)
        if current_c5[best_idx] < best_c5:
            best_c5 = current_c5[best_idx]
            best_h = population[best_idx]
        mutation_scale *= cooling_factor

    return best_h, best_c5, n_points
