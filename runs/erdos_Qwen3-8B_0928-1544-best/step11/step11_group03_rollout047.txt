                    scaling_factor = required_sum / child_sum_scaled
                    child *= scaling_factor
                child = np.clip(child, 0.0, 1.0)
            new_population.append(child)
        
        population = new_population
        for i in range(population_size):
            current_c5 = objective(population[i])
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = population[i].copy()
    
    return best_h, best_c5, n_points
