    learning_rate = 0.1
    min_lr = 1e-5
    perturb_interval = 500

    while iteration < max_iterations and time.time() - start_time < budget_s:
        current_c5 = compute_c5(h, n_points, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        overlap = np.correlate(h, 1 - h, mode='full')
        max_overlap_val = np.max(overlap)
        max_lags_indices = np.where(overlap == max_overlap_val)[0]
        gradient = np.zeros(n_points)

        for lag_index in max_lags_indices:
            lag_val = lag_index - (n_points - 1)
            for j in range(n_points):
                contrib1 = 0.0
                if 0 <= j + lag_val < n_points:
                    contrib1 = (1 - h[j + lag_val]) * dx
                contrib2 = 0.0
                if 0 <= j - lag_val < n_points:
                    contrib2 = h[j - lag_val] * dx
                gradient[j] += (contrib1 - contrib2)

        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scaling = (n_points / 2.0) / total if total != 0 else 1.0
        h = h * scaling

        if compute_c5(h, n_points, dx) < best_c5:
            learning_rate = min(0.1, learning_rate * 1.2)
        else:
            learning_rate = max(min_lr, learning_rate * 0.5)

        if iteration % perturb_interval == 0:
            perturb = np.random.randn(n_points) * learning_rate * 0.1
            h += perturb
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            scaling = (n_points / 2.0) / total if total != 0 else 1.0
            h = h * scaling
            if compute_c5(h, n_points, dx) < best_c5:
                best_c5 = compute_c5(h, n_points, dx)
                best_h = h.copy()

        iteration += 1

    return best_h, best_c5, n_points

def compute_c5(h, n_points, dx):
    overlap = np.correlate(h, 1 - h, mode='full')
    return np.max(overlap) * dx
