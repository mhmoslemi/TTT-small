    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        curr_max = corr[k_max]

        def objective(h_vec):
            k = k_max
            total = 0.0
            for i in range(n_points - k):
                total += h_vec[i] * (1 - h_vec[i + k])
            return total * (2.0 / n_points)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - ones_count

        cons = [{'type': 'eq', 'fun': constraint_sum}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-7)
        if res.success:
            h = res.x
            new_c5 = compute_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()

    return best_h, best_c5, n_points

def compute_c5(h_values):
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    return np.max(corr) * (2.0 / len(h_values))
