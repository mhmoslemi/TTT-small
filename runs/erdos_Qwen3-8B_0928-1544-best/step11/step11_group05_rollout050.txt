import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.random.rand(70)
        h_initial = h_initial * (35.0 / np.sum(h_initial))
    n_points = 70
    dx = 2.0 / n_points

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def constraint(h_vec):
        return np.sum(h_vec) - 35.0

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 1000})
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
