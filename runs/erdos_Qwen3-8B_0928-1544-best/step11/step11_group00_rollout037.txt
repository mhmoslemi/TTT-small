import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 200
    w_start, w_end = 0.8, 0.2
    phi_p, phi_g = 2.0, 2.0
    max_velocity = 0.1
    max_iterations = 1000
    stagnation_threshold = 50

    if 'initial_h_values' in globals():

        initial_sum = np.sum(initial_h)
        initial_scaled = initial_h * (ones_count / initial_sum)
        population = [initial_scaled]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)

    velocities = [np.zeros(n_points) for _ in range(population_size)]
    pbest = population.copy()
    gbest = min(pbest, key=lambda x: _objective(x, n_points))
    best_c5 = _objective(gbest, n_points)
    best_h = gbest

    start_time = time.time()
    w = w_start
    stagnation = 0

    while time.time() - start_time < budget_s:
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i]
            curr = population[i]
            pbest_i = pbest[i]
            new_vel = w * vel + phi_p * r1 * (pbest_i - curr) + phi_g * r2 * (gbest - curr)
            new_vel = np.clip(new_vel, -max_velocity, max_velocity)
            velocities[i] = new_vel
            new_pos = curr + new_vel
            new_pos = np.clip(new_pos, 0.0, 1.0)
            new_sum = np.sum(new_pos)
            if new_sum == 0:
                continue
            new_pos = new_pos * (ones_count / new_sum)
            population[i] = new_pos
            curr_c5 = _objective(population[i], n_points)
            if curr_c5 < _objective(pbest[i], n_points):
                pbest[i] = population[i].copy()
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    best_h = population[i].copy()
                    stagnation = 0
            else:
                stagnation += 1
        if stagnation > stagnation_threshold:
            w = max(w_start, w - 0.005)
            stagnation = 0

    return best_h, best_c5, n_points

def _objective(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n_points)
