    T = 0.5
    cooling_rate = 0.99
    tabu = set()
    tabu_size = 20
    start_time = time.time()
    while time.time() - start_time < budget_s:
        perturbed = current_h.copy()
        num_perturb = np.random.randint(5, 15)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for i in indices:
            perturbed[i] += np.random.uniform(-0.1, 0.1)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        scale = target_sum / perturbed_sum
        perturbed = perturbed * scale
        perturbed = np.clip(perturbed, 0, 1)
        key = tuple(perturbed.round(5))
        if key in tabu:
            continue
        new_c5 = compute_c5(perturbed)
        current_c5 = compute_c5(current_h)
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            current_h = perturbed.copy()
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = current_h.copy()
        tabu.add(key)
        if len(tabu) > tabu_size:
            tabu.remove(min(tabu))
        T *= cooling_rate
    return best_h, best_c5, n_points
