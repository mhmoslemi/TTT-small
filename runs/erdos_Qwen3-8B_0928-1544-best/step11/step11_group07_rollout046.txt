import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    temp = 1.0
    cooling_rate = 0.995
    start_time = time.time()
    while time.time() - start_time < budget_s:
        delta = 0.08
        h_perturbed = h + np.random.uniform(-delta, delta, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed_sum = np.sum(h_perturbed)
        if h_perturbed_sum == 0:
            h_perturbed = np.random.rand(n_points)
            h_perturbed_sum = np.sum(h_perturbed)
            h_perturbed = h_perturbed * (ones_count / h_perturbed_sum)
        else:
            h_perturbed = h_perturbed * (ones_count / h_perturbed_sum)
        new_c5 = compute_c5(h_perturbed, n_points)
        if new_c5 < best_c5:
            h = h_perturbed.copy()
            best_c5 = new_c5
        else:
            prob = np.exp((new_c5 - best_c5) / temp)
            if np.random.rand() < prob:
                h = h_perturbed.copy()
                best_c5 = new_c5
        temp *= cooling_rate
    return best_h, best_c5, n_points

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    dx = 2.0 / n_points
    return max_corr * dx
