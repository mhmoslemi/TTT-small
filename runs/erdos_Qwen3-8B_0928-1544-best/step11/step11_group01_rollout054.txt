        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):

                grad[i] += (1 - h[i + lag])
            if i - lag >= 0:
                grad[i] -= h[i - lag]

        direction = -learning_rate * grad
        h_new = h + direction
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        if h_new_sum == 0:
            # Avoid division by zero
            h_new = h
        else:
            scale = ones_count / h_new_sum
            h_new = h_new * scale
        new_c5 = max_corr * (2.0 / n_points)
        if new_c5 < best_c5:

            best_h = h_new.copy()
