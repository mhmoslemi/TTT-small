import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 100
    F = 0.5
    CR = 0.8
    max_generations = int(budget_s / (population_size * 2))
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        scaling_factor = ones_count / initial_h_sum
        initial_h_scaled = initial_h * scaling_factor
        initial_h_scaled = np.clip(initial_h_scaled, 0.0, 1.0)
        population.append(initial_h_scaled)
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            scaling_factor = ones_count / h_sum
            h_scaled = h * scaling_factor
            h_scaled = np.clip(h_scaled, 0.0, 1.0)
            population.append(h_scaled)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            scaling_factor = ones_count / h_sum
            h_scaled = h * scaling_factor
            h_scaled = np.clip(h_scaled, 0.0, 1.0)
            population.append(h_scaled)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate all
        fitness = [objective(h) for h in population]
        # Generate new population
        new_population = []
        for _ in range(population_size):
            # Select three distinct parents for mutation
            indices = np.random.choice(population_size, 3, replace=False)
            a, b, c = indices
            parent = population[a]
            # Generate mutant
            mutant = parent + F * (population[b] - population[c])
            # Crossover between parent and mutant
            offspring = np.copy(parent)
            for i in range(n_points):
                if np.random.rand() < CR:
                    offspring[i] = mutant[i]
            # Scale and clip
            sum_offspring = np.sum(offspring)
            if sum_offspring == 0:
                continue
            scaling_factor = ones_count / sum_offspring
            offspring_scaled = offspring * scaling_factor
            offspring_scaled = np.clip(offspring_scaled, 0.0, 1.0)
            sum_scaled = np.sum(offspring_scaled)
            if sum_scaled == 0:
                continue
            scaling_factor_new = ones_count / sum_scaled
            offspring_scaled = offspring_scaled * scaling_factor_new
            new_population.append(offspring_scaled)
        # Replace population
        new_fitness = [objective(h) for h in new_population]
        combined = list(zip(population, fitness, new_population, new_fitness))
        combined.sort(key=lambda x: x[1])
        population = [x[2] for x in combined[:population_size]]
        # Update best solution
        for h, f in zip(new_population, new_fitness):
            if f < best_c5:
                best_c5 = f
                best_h = h.copy()
    return best_h, best_c5, n_points
