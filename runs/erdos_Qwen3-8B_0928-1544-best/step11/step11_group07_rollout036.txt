import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h = h * (ones_count / h_sum)
    
    best_h = h.copy()
    best_c5 = 0.0
    dx = 2.0 / n_points
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k_max = max_index - (n_points - 1)
        
        # Compute gradient for current h and k_max
        gradient = np.zeros(n_points)
        for i in range(n_points):
            term1 = 0.0
            if i + k_max < n_points:
                term1 = 1.0 - h[i + k_max]
            term2 = 0.0
            if i - k_max >= 0:
                term2 = -h[i - k_max]
            gradient[i] = term1 + term2
        
        # Define objective function for S(k_max)
        def objective(x):
            h_new = x
            overlap = 0.0
            for i in range(n_points - k_max):
                overlap += h_new[i] * (1.0 - h_new[i + k_max])
            return overlap * dx
        
        # Define constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        
        # Optimize
        result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, jac=gradient)
        h = result.x
        
        # Update best_c5
        curr_corr = np.correlate(h, 1 - h, mode='full')
