import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    pop_size = 100
    mutation_rate = 0.15
    crossover_rate = 0.85

    if 'initial_h_values' in globals():
        h0 = initial_h_values.copy()
        h0_sum = np.sum(h0)
        h0 = np.clip(h0 * (ones_count / h0_sum), 0.0, 1.0)
        population = [h0]
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h = np.clip(h * (ones_count / h_sum), 0.0, 1.0)
            population.append(h)
    else:
        population = [np.clip(np.random.rand(n_points) * (ones_count / np.sum(np.random.rand(n_points))), 0.0, 1.0) for _ in range(pop_size)]

    def obj(h):
        return np.max(np.correlate(h, 1 - h, 'full')) * (2.0 / n_points)

    best = population[0]
    best_c5 = obj(best)

    start = time.time()

    while time.time() - start < budget_s:
        fitness = [obj(ind) for ind in population]
        new_pop = []
        for _ in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            p1, p2, p3 = population[a], population[b], population[c]
            if np.random.rand() < crossover_rate:
                child = np.random.choice([p1, p2, p3], size=n_points)
                child = (child * np.random.rand(n_points)).clip(0, 1)
                sum_c = np.sum(child)
                if sum_c != 0:
                    child = (child * (ones_count / sum_c)).clip(0, 1)
            else:
                child = population[np.random.choice(pop_size)].copy()
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < 0.2:
                        child[i] += np.random.uniform(-0.1, 0.1)
                        child[i] = np.clip(child[i], 0.0, 1.0)
                sum_m = np.sum(child)
                if sum_m != 0:

            new_pop.append(child)
        fitness_new = [obj(ind) for ind in new_pop]

        if min(fitness_new) < best_c5:

            best = new_pop[np.argmin(fitness_new)]
