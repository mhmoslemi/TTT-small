import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 50
    tournament_size = 3
    mutation_rate = 0.2
    elite_ratio = 0.1
    max_generations = 1000
    best_c5 = float('inf')
    best_h = None

    def normalize(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            return np.random.rand(n_points) * 0.5 * (ones_count / (np.random.rand(n_points).sum()))
        return h * (ones_count / h_sum)

    if 'initial_h_values' in globals():

        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            initial_pop.append(normalize(h))
        population = initial_pop
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            population.append(normalize(h))

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    for h in population:
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    start_time = time.time()
    for gen in range(max_generations):
        if time.time() - start_time >= budget_s:

        for _ in range(population_size):

            selected.append(min(candidates, key=evaluate))
        new_pop = []
        for i in range(0, population_size, 2):
            p1, p2 = selected[i], selected[i+1]
            alpha = np.random.uniform(0.4, 0.6)
            c1 = alpha * p1 + (1 - alpha) * p2
            c2 = (1 - alpha) * p1 + alpha * p2
            for c in [c1, c2]:

                best_h = h.copy()
