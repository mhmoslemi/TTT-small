import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.uniform(0.4, 0.6, n_points)
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    max_iter = 5000
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmin(corr)
        k_max = np.clip(k_max, 0, len(corr) - 1)
        
        delta = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = (i + k_max) % n_points
            i_minus_k = (i - k_max) % n_points
            deriv = (1 - h[i_plus_k] - h[i_minus_k]) * dx
            if deriv > 0:
                delta[i] = -0.01
            elif deriv < 0:
                delta[i] = 0.01
        
        h = np.clip(h + delta, 0.0, 1.0)
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum)
        
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    def local_obj(x):
        return compute_c5(x)
    
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count})
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    res = minimize(local_obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    
    return best_h, best_c5, n_points
