            constraints.append(cp.square(H[i].real) + cp.square(H[i].imag) <= tau)
        prob = cp.Problem(cp.Minimize(tau), constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-6)
            if prob.status != 'OPTIMAL':
                continue
            H_scaled = H * np.sqrt(tau.value)
            h_new = scipy.fft.ifft(H_scaled).real
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new_sum = np.sum(h_new)
            if h_new_sum == 0:
                h_new = np.ones(n_points) * (ones_count / n_points)
            else:
                h_new = h_new * (ones_count / h_new_sum)
            # Evaluate C5
            corr = np.correlate(h_new, 1 - h_new, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h = h_new.copy()
        except:
            continue
    
    return best_h, best_c5, n_points
