import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2.0

    # Initialize h based on available initial_h_values
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_sum = np.sum(initial_h)
        best_h = initial_h * (target_sum / initial_sum)
        best_h = np.clip(best_h, 0.0, 1.0)
    else:
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        best_h = h * (target_sum / h_sum)
        best_h = np.clip(best_h, 0.0, 1.0)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_c5 = compute_c5(best_h)
    sigma = 0.1  # Initial mutation strength
    best_h_orig = best_h.copy()

    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Generate a new candidate by perturbing best_h
        perturbation = np.random.normal(0, sigma, n_points)
        candidate = best_h + perturbation
        candidate = np.clip(candidate, 0.0, 1.0)
        candidate_sum = np.sum(candidate)
        if candidate_sum == 0:
            continue
        candidate_scaled = candidate * (target_sum / candidate_sum)
        new_c5 = compute_c5(candidate_scaled)

        # Update if improvement is found
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = candidate_scaled.copy()

        else:

    return best_h, best_c5, n_points
