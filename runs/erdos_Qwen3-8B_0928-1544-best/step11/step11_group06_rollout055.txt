            current_sum = np.sum(new_h)
            if np.isclose(current_sum, target_sum):
                new_c5, _ = compute_max_overlap(new_h, n_points)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
    
    return best_h, best_c5, n_points
