import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points

    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if np.isclose(total, 0):
            h = np.ones(n_points) * (n_points / (2.0 * n_points))
        else:
            scaling = (n_points / 2.0) / total
            h *= scaling
    else:
        # Fallback to triangular wave
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        if np.sum(h) == 0:
            h = np.ones(n_points) * 0.5
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scaling = (n_points / 2.0) / total
        h *= scaling

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    max_iterations = int(budget_s * 20)
    learning_rate = 0.2
    min_lr = 1e-5

    for _ in range(max_iterations):
        if time.time() - start_time >= budget_s:
            break
        # Compute overlap and find critical lag
        overlap = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(overlap)
        k = max_index - (n_points - 1)
        # Gradient for lag k
        gradient = np.zeros(n_points)
        for j in range(n_points):
            shifted_j = j + k
            if 0 <= shifted_j < n_points:
                gradient[j] = (1 - h[shifted_j]) * dx
        # Update h
        h -= learning_rate * gradient
        # Project to feasible region
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if np.isclose(total, 0):
            h = np.ones(n_points) * (n_points / (2.0 * n_points))
        else:
            scaling = (n_points / 2.0) / total
            h *= scaling
