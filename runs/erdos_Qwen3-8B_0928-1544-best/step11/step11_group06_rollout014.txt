import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.uniform(0.4, 0.6, n_points)
    h = h * (ones_count / np.sum(h))
    h = np.clip(h, 0.0, 1.0)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap_idx = np.argmax(corr)
        k_max = max_overlap_idx - (n_points - 1)
        
        def obj(x):
            k = k_max
            if k > 0:
                valid_i = np.arange(0, n_points - k)
                s = np.sum(x[valid_i] * (1 - x[valid_i + k]))
            elif k < 0:
                valid_i = np.arange(-k, n_points)
                s = np.sum(x[valid_i] * (1 - x[valid_i + k]))
            else:
                s = np.sum(x * (1 - x))
            return s * (2.0 / n_points)
        
        constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count})
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
        if res.success:
            h = res.x
            h = np.clip(h, 0.0, 1.0)
            h = h * (ones_count / np.sum(h))
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    return best_h, best_c5, n_points
