import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 200
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        population = [initial_h_scaled]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    velocity = np.zeros((population_size, n_points))
    pbest = np.array(population)
    pbest_fitness = np.array([objective(h) for h in population])
    gbest = population[np.argmin(pbest_fitness)]
    gbest_fitness = np.min(pbest_fitness)
    
    start_time = time.time()
    max_velocity = 0.1
    inertia_weight = 0.8
    phi_p = 2.0
    phi_g = 2.0
    stagnation = 0
    
    while time.time() - start_time < budget_s:
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocity[i] = inertia_weight * velocity[i] + phi_p * r1 * (pbest[i] - population[i]) + phi_g * r2 * (gbest - population[i])
            velocity[i] = np.clip(velocity[i], -max_velocity, max_velocity)
            population[i] += velocity[i]
            population[i] = np.clip(population[i], 0.0, 1.0)
            current_sum = np.sum(population[i])
            if current_sum == 0:
                continue
            population[i] *= (ones_count / current_sum)
            current_fitness = objective(population[i])
            if current_fitness < pbest_fitness[i]:
                pbest[i] = population[i].copy()
                pbest_fitness[i] = current_fitness
                if current_fitness < gbest_fitness:
                    gbest = population[i].copy()
                    gbest_fitness = current_fitness
                stagnation = 0
            else:
                stagnation += 1
            if stagnation > 100:
                inertia_weight = max(0.4, inertia_weight - 0.01)
                phi_p = min(3.0, phi_p + 0.05)
                phi_g = min(3.0, phi_g + 0.05)
                stagnation = 0
    
    return gbest, gbest_fitness, n_points
