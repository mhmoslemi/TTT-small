    n_points = 70
    ones_count = n_points // 2
    population_size = 150
    F = 0.9
    CR = 0.8
    mutation_ratio = 0.3
    selection_pressure = 0.95

    def initial_h():
        if 'initial_h_values' in globals():
            h = initial_h_values.copy()
            h_sum = np.sum(h)
            h_scaled = h * (ones_count / h_sum)
            return h_scaled
        else:
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            h_scaled = h * (ones_count / h_sum)
            return h_scaled

    population = [initial_h()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h_scaled = h * (ones_count / h_sum)
        population.append(h_scaled)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()

    while time.time() - start_time < budget_s:
        new_population = []
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]

            u_sum = np.sum(u)

            current_c5 = objective(population[i])
            new_c5 = objective(u_scaled)
            if new_c5 < current_c5:
