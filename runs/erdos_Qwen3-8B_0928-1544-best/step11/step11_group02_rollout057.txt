            u_scaled = u * (ones_count / u_sum)
            new_c5 = obj_h(u_scaled)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = u_scaled.copy()
            if new_c5 < obj_h(population[i]):
                population[i] = u_scaled.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining < 500:
            F = 0.8
            CR = 0.95
        else:
            F = 0.5
            CR = 0.8

    def obj_local(x):
        h = x
        sum_h = np.sum(h)
        penalty = 1e6 * (sum_h - n_points/2)**2
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points) + penalty

    bounds = [(0, 1)] * n_points
    res = minimize(obj_local, best_h, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
    final_h = res.x
    final_h = np.clip(final_h, 0, 1)
    final_sum = np.sum(final_h)
    if np.abs(final_sum - n_points/2) > 1e-8:
        final_h = final_h * (n_points/2) / final_sum
    final_c5 = obj_local(final_h) - 1e6 * (np.sum(final_h) - n_points/2)**2

    return final_h, final_c5, n_points
