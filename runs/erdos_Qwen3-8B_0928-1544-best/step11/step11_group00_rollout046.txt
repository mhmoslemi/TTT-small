                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u_scaled.copy()
                    stagnation = 0
            else:
                stagnation += 1
            if stagnation > 20:
                F = max(F_start, F - 0.02)
                CR = min(CR_end, CR + 0.02)
                stagnation = 0
            if time.time() - start_time > budget_s - 10:
                break
        if time.time() - start_time > budget_s - 10:
            break

    return best_h, best_c5, n_points
