import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

    def normalize_h(h):
        h_sum = np.sum(h)
        if h_sum == 0:
            return np.zeros_like(h)
        scaling_factor = ones_count / h_sum
        return np.clip(h * scaling_factor, 0.0, 1.0)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def compute_gradient(h):
        grad = np.zeros_like(h)
        epsilon = 1e-5
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = normalize_h(h_plus)
            c5_plus = compute_c5(h_plus)
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = normalize_h(h_minus)
            c5_minus = compute_c5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
        return grad

    def line_search(h, grad, tau, current_c5, n_points, ones_count):
        alpha = 1.0
        max_iter = 20
        while max_iter > 0:
            h_candidate = h - alpha * grad
            h_candidate = normalize_h(h_candidate)
            if np.any(h_candidate < 0) or np.any(h_candidate > 1):
                alpha *= 0.5
                max_iter -= 1
                continue
            new_c5 = compute_c5(h_candidate)
            new_barrier = np.sum(np.log(1 - h_candidate) - np.log(h_candidate)) / tau
            total_new = new_c5 + new_barrier
            total_old = current_c5 + (np.sum(np.log(1 - h) - np.log(h)) / tau)
            if total_new <= total_old - 0.1 * alpha * np.dot(grad, grad):
                return alpha
            alpha *= 0.5
            max_iter -= 1
        return 0.0

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        h = normalize_h(h)
    else:
        h = np.random.rand(n_points)
        h = normalize_h(h)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    tau = 1e6
    min_tau = 1e-3
    tau_decay = 0.95
    max_iter = 200

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        grad_c5 = compute_gradient(h)
        gradient_barrier = np.log(1 - h) - np.log(h)
        gradient_total = grad_c5 + gradient_barrier / tau
        alpha = line_search(h, gradient_total, tau, best_c5, n_points, ones_count)
        if alpha > 0:
            h = h - alpha * gradient_total
            h = normalize_h(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        tau *= tau_decay

    return best_h, best_c5, n_points
