        max_corr_index = np.argmax(corr)
        shift = max_corr_index - (n_points - 1)
        k_max_val = int(shift)
        def objective(h_vars):
            if k_max_val < 0:
                lower = k_max_val
                valid_j = np.arange(lower, n_points)
                valid_hj = h_vars[valid_j]
                valid_hj_minusk = h_vars[valid_j - k_max_val]
                return np.dot(valid_hj, 1 - valid_hj_minusk) * dx
            else:
                valid_i = np.arange(n_points - k_max_val)
                valid_hi = h_vars[valid_i]
                valid_hi_plusk = h_vars[valid_i + k_max_val]
                return np.dot(valid_hi, 1 - valid_hi_plusk) * dx
        def gradient(h_vars):
            grad = np.zeros(n_points)
            if k_max_val < 0:
                lower = k_max_val
                valid_j = np.arange(lower, n_points)
                grad[valid_j] += (1 - h_vars[valid_j - k_max_val]) * dx
                j_minus_k = valid_j - k_max_val
                grad[j_minus_k] -= h_vars[j_minus_k] * dx
            else:
                valid_i = np.arange(n_points - k_max_val)
                grad[valid_i] += (1 - h_vars[valid_i + k_max_val]) * dx
                i_plus_k = valid_i + k_max_val
                grad[i_plus_k] -= h_vars[valid_i] * dx
            return grad
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = minimize(objective, h, method='SLSQP', jac=gradient, bounds=bounds, constraints=cons, options={'maxiter': 10})
        h = result.x
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    return best_h, best_c5, n_points
