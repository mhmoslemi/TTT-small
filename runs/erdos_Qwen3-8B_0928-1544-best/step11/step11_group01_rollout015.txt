def run(seed=42, budget_s=1100, **kwargs):

    h = h_scaled.copy()
    
    learning_rate = 0.1
    lr_decay = 0.999
    max_iterations = 100000
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    
    for it in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        j_max = np.argmax(corr)
        k_max = j_max - (n_points - 1)
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus = i + k_max
            term1 = 1 - (h[i_plus] if i_plus < n_points else 0.0)
            i_minus = i - k_max
            term2 = h[i_minus] if i_minus >= 0 else 0.0
            gradient[i] = term1 - term2
        
        h -= learning_rate * gradient
        h = np.clip(h, 0.0, 1.0)
