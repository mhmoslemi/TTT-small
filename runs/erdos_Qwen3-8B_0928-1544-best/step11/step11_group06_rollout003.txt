import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    dx = 2.0 / n_points

    # Check if initial_h_values is available
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.uniform(0.4, 0.6, n_points)

    h_sum = np.sum(initial_h)
    if h_sum == 0:
        initial_h = np.random.uniform(0.4, 0.6, n_points)
        h_sum = np.sum(initial_h)
    initial_h_scaled = initial_h * (ones_count / h_sum)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - ones_count})
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = minimize(objective, initial_h_scaled, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)

    best_h = result.x if result.success else initial_h_scaled
    best_c5 = result.fun if result.success else objective(best_h)

    return best_h, best_c5, n_points
