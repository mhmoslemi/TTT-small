        h_initial = np.interp(new_x, original_x, initial_h_values)
    else:
        h_initial = np.random.rand(n_points)
    h_current = h_initial * (sum_required / np.sum(h_initial))
    best_c5 = np.max(np.correlate(h_current, 1 - h_current, mode='full')) * (2.0 / n_points)
    best_h = h_current.copy()
    T = 0.5
    alpha = 0.99
    tabu = []
    start_time = time.time()
    while time.time() - start_time < budget_s:
        perturb_idx = np.random.choice(n_points, size=10, replace=False)
        h_temp = h_current.copy()
        for idx in perturb_idx:
            h_temp[idx] += np.random.uniform(-0.1, 0.1)
            h_temp[idx] = np.clip(h_temp[idx], 0.0, 1.0)
        current_sum = np.sum(h_temp)
        if current_sum == 0:
            continue
        h_candidate = h_temp * (sum_required / current_sum)
        in_tabu = False
        for t in tabu:
            if np.allclose(t, h_candidate):
                in_tabu = True
                break
        if in_tabu:
            continue
        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        new_c5 = np.max(corr) * (2.0 / n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_candidate.copy()
            tabu.append(h_candidate)
            if len(tabu) > 20:
                tabu.pop(0)
        else:
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                best_c5 = new_c5
                best_h = h_candidate.copy()
                tabu.append(h_candidate)
                if len(tabu) > 20:
                    tabu.pop(0)
        T *= alpha
        if T < 1e-5:
            break
    return best_h, best_c5, n_points
