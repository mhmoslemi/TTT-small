import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.random.rand(n_points) * 0.5
        h_initial_sum = np.sum(h_initial)
        h_initial = h_initial * target_sum / h_initial_sum

    def objective(h):

        h_sum = np.sum(h_clipped)
        if h_sum == 0:
            return np.inf
        h_normalized = h_clipped * target_sum / h_sum
        corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    result = differential_evolution(
        objective, bounds,
        strategy='best1bin',

        popsize=50,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        result = differential_evolution(
            objective, bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=50,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
    best_h = result.x
    best_c5 = result.fun
    return best_h, best_c5, n_points
