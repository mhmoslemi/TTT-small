import time
from scipy.fft import fft, ifft

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 100
    ones_count = n_points // 2
    if 'initial_h_values' in globals():

    h_sum = np.sum(h)

    def apply_lowpass_filter(h, cutoff_ratio=0.2):
        fft_h = fft(h)
        n = len(h)
        cutoff = int(cutoff_ratio * n)
        fft_h[cutoff:] = 0
        fft_h[n - cutoff:] = 0
        return np.real(ifft(fft_h))

    h = apply_lowpass_filter(h)
    h = np.clip(h, 0, 1)
    h_sum_current = np.sum(h)
    h = h * (ones_count / h_sum_current)

    learning_rate = 0.1
    max_iter = 1000
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr)
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()

        max_k_index = np.argmax(corr)
        k = max_k_index - (n_points - 1)
        k = max(0, min(k, n_points - 1))

        grad = np.zeros(n_points)
        for p in range(n_points):
            p_plus_k = (p + k) % n_points
            p_minus_k = (p - k) % n_points
            grad[p] = (1 - h[p_plus_k]) - h[p_minus_k]
