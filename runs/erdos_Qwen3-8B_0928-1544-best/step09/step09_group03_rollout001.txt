            term1 = 1.0 - h[i_plus_k]
            term2 = h[i_minus_k]
            gradient[j] = dx * (term1 - term2)
        h = h - learning_rate * gradient
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.full(n_points, 0.5)
        else:
            h = h * (ones_count / h_sum)
        current_c5 = current_max
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        learning_rate *= decay_rate
    return best_h, best_c5, n_points
