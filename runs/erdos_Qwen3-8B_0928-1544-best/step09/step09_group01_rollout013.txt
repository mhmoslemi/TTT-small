        initial_h_scaled = np.clip(initial_h_scaled, 0.0, 1.0)
        best_h = initial_h_scaled.copy()
        best_c5 = _evaluate_erdos_solution(best_h)
    else:
        # Random initialization
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h_scaled = h * (ones_count / h_sum)
        h_scaled = np.clip(h_scaled, 0.0, 1.0)
        best_h = h_scaled.copy()
        best_c5 = _evaluate_erdos_solution(best_h)
    
    # Define objective function with constraint enforcement
    def objective(h_flat):
        h = h_flat.copy()
        h_sum = np.sum(h)
        # Scale to meet the constraint
        if h_sum != 0:
            h = h * (ones_count / h_sum)
        h = np.clip(h, 0.0, 1.0)
        # Compute overlap
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # Bounds for each parameter
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use differential evolution with improved settings
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,  # Adjust for budget
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed,
        disp=False
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # If initial_h is better, return it
    if 'initial_h_values' in globals():
        initial_scaled = _evaluate_erdos_solution(initial_h_scaled)
        if initial_scaled < best_c5:
            best_c5 = initial_scaled
            best_h = initial_h_scaled
    
    # Final evaluation
    final_c5 = _evaluate_erdos_solution(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Ensure sum constraint is satisfied
    h_sum = np.sum(best_h)
    if h_sum != 0:
        best_h = best_h * (ones_count / h_sum)
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, final_c5, n_points
