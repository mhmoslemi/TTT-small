    best_h = initial_h.copy()
    best_c5 = obj(initial_h)
    start_time = time.time()
    
    # Firefly loop
    for gen in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Evaluate brightness
        fitness = [obj(h) for h in pop]
        # Sort by brightness
        sorted_idx = np.argsort(fitness)
        sorted_pop = [pop[i] for i in sorted_idx]
        sorted_fit = np.array(fitness)[sorted_idx]
        
        # Update positions
        for i in range(pop_size):
            if i < pop_size * elite_ratio:
                continue  # Elite stays
