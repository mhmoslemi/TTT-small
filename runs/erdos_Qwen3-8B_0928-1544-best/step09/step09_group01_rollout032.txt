            u = np.clip(u, 0, 1)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            current_c5 = objective(population[i])
            new_c5 = objective(u_scaled)
            if new_c5 < current_c5:
                population[i] = u_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u_scaled.copy()
        if gen % 10 == 0:
            top = sorted(population, key=objective)[:5]
            for h in top:
                def local_obj(x):
                    x = np.clip(x, 0, 1)
                    x_sum = np.sum(x)
                    if x_sum == 0:
                        return 1e9
                    x_scaled = x * (ones_count / x_sum)
                    return objective(x_scaled)
                res = scipy.optimize.minimize(local_obj, h, method='Nelder-Mead', tol=1e-5)
                if res.success:
                    candidate = res.x
                    candidate = np.clip(candidate, 0, 1)
                    candidate_sum = np.sum(candidate)
                    if candidate_sum == 0:
                        continue
                    candidate_scaled = candidate * (ones_count / candidate_sum)
                    if objective(candidate_scaled) < objective(h):
                        h[:] = candidate_scaled.copy()
        if gen % 50 == 0:
            if best_c5 < 0.38:
                F = min(F + 0.05, 1.0)
                CR = max(CR - 0.05, 0.5)
            else:
                F = max(F - 0.05, 0.5)
                CR = min(CR + 0.05, 0.95)
    return best_h, best_c5, n_points
