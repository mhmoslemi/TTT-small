import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_h = np.random.rand(n_points)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    initial_h = np.clip(initial_h, 0.0, 1.0)
    initial_h = initial_h * target_sum / np.sum(initial_h)

    pop_size = 100
    positions = [initial_h.copy() for _ in range(pop_size)]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(pop_size)]

    pbest_positions = [pos.copy() for pos in positions]
    pbest_values = [objective(pos) for pos in positions]
    gbest_position = positions[0].copy()
    gbest_value = objective(positions[0])

    max_iterations = 600
    start_time = time.time()

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            current_h = positions[i]
            current_fitness = objective(current_h)
            if current_fitness < pbest_values[i]:
                pbest_positions[i] = current_h.copy()
                pbest_values[i] = current_fitness
            if current_fitness < gbest_value:
                gbest_position = current_h.copy()
                gbest_value = current_fitness
        for i in range(pop_size):
            inertia_weight = 0.9 - (iteration / max_iterations) * 0.5
            r1 = np.random.rand()
            r2 = np.random.rand()
            cognitive_term = 2.0 * r1 * (pbest_positions[i] - positions[i])
            social_term = 2.0 * r2 * (gbest_position - positions[i])
            velocities[i] = inertia_weight * velocities[i] + cognitive_term + social_term
            positions[i] = positions[i] + velocities[i]
            positions[i] = np.clip(positions[i], 0.0, 1.0)
            positions[i] = positions[i] * target_sum / np.sum(positions[i])

    best_h = gbest_position.copy()

    def objective_for_SLSQP(x):
        return objective(x)

    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0)] * n_points
    result = minimize(fun=objective_for_SLSQP, x0=best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9})
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else gbest_value
