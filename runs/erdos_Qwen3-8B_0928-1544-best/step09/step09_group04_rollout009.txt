from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    initial_h = initial_h_values.copy()
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h * target_sum / np.sum(initial_h)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0)] * n_points

    def optimize_candidate(h_initial):
        result = minimize(
            fun=objective,
            x0=h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={
                'maxiter': 300,
                'ftol': 1e-8,
                'eps': 1e-8,
                'disp': False
            }
        )
        final_h = result.x
        final_sum = np.sum(final_h)
        final_h = final_h * target_sum / final_sum
        return final_h, result.fun

    best_h, best_c5 = optimize_candidate(initial_h)
    start_time = time.time()
