import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():

        h = np.random.rand(n_points)

    best_h = h.copy()
    best_C5 = current_C5
    T = 0.1 * current_C5
    alpha = 0.995
    min_T = 1e-5
    start_time = time.time()

    while time.time() - start_time < budget_s:
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.min([h[i], 1 - h[j]])
        epsilon = np.random.uniform(0.01, 0.1)
        delta_transfer = epsilon * delta
        h_new = h.copy()
        h_new[i] -= delta_transfer

        new_C5 = compute_C5(h_new)
