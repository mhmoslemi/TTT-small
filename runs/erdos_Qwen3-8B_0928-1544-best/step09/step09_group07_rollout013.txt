from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        n_low = 35
        h_low = np.zeros(n_low)
        np.random.shuffle(np.arange(n_low))
        h_low[:17] = 1.0
        h_low = h_low * (target_sum / 2) / np.sum(h_low)  # scale to 17.5
        x_low = np.linspace(0, 2, n_low)
        x_high = np.linspace(0, 2, n_points)
        h = np.interp(x_high, x_low, h_low)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0)] * n_points

    # Differential Evolution
    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        tol=1e-7,
        mutation=(0.5, 1.0),
        recombination=0.8,
        polish=True,
        maxiter=200,
        disp=False,
        vectorized=True,
        callback=lambda x: None
    )

    final_h = result.x
    final_c5 = result.fun

    # Gradient-based polishing
    result_polish = minimize(
        fun=objective,
        x0=final_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )

    final_h = result_polish.x if result_polish.success else final_h
    final_c5 = result_polish.fun if result_polish.success else final_c5

    return final_h, final_c5, n_points
