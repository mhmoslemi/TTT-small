    if initial_h_sum != 0:
        initial_h = initial_h * (target_sum / initial_h_sum)
    best_h = initial_h.copy()
    best_c5 = calc_overlap(best_h, dx, n_points)
    
    max_iterations = 1000
    learning_rate = 0.01
    epsilon = 1e-4
    tolerance = 1e-6
    convergence_count = 0
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        overlaps = corr * dx
        k_max = np.argmax(overlaps)
        current_overlap = overlaps[k_max]
        
        if current_overlap < tolerance:
            break
        
        grad = compute_gradient(best_h, k_max, dx, target_sum, n_points)
        
        step = learning_rate
        for _ in range(10):
            perturbed_h = best_h.copy()
            perturbed_h = perturbed_h - step * grad
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_sum = np.sum(perturbed_h)
            if perturbed_sum == 0:
                break
            perturbed_h = perturbed_h * (target_sum / perturbed_sum)
            new_corr = np.correlate(perturbed_h, 1 - perturbed_h, mode='full')
            new_overlap = new_corr[k_max] * dx
            if new_overlap < current_overlap:
                best_h = perturbed_h.copy()
                current_overlap = new_overlap
                break
            else:
                step *= 0.5
        
        if current_overlap < best_c5:
            best_c5 = current_overlap
            best_h = best_h.copy()
        
        if abs(best_c5 - current_overlap) < tolerance:
            convergence_count += 1
            if convergence_count > 50:
                break
        else:
            convergence_count = 0
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        fun=lambda h: calc_overlap(h, dx, n_points),
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5
    
    return final_h, final_c5, n_points

def compute_gradient(h, k, dx, target_sum, n_points):
    n = len(h)
    original_overlap = np.correlate(h, 1 - h, mode='full')[k] * dx
    grad = np.zeros(n)
    epsilon = 1e-4
    for j in range(n):
        perturbed_h = h.copy()
        perturbed_h[j] += epsilon
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_sum = np.sum(perturbed_h)
        if perturbed_sum == 0:
            continue
        perturbed_h = perturbed_h * (target_sum / perturbed_sum)
        perturbed_overlap = np.correlate(perturbed_h, 1 - perturbed_h, mode='full')[k] * dx
        grad[j] = (perturbed_overlap - original_overlap) / epsilon
    return grad

def calc_overlap(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)
