from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_low = 35
    n_high = 70
    dx_low = 2.0 / n_low
    dx_high = 2.0 / n_high
    target_sum_low = n_low / 2
    target_sum_high = n_high / 2

    # Generate initial coarse_h
    if 'initial_h_values' in globals():

        coarse_h[:17] = 1.0
        coarse_h[-1] = 0.5
