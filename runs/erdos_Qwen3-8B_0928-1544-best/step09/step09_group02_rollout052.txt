def run(seed=42, budget_s=1100, **kwargs):

        h = h * (ones_count / total)

        h = np.full(n_points, 1.0 / n_points) * ones_count
    
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()

    alpha = 0.1
    
    for _ in range(max_iter):
        # Compute overlaps for all shifts
        overlaps = []

            shifted = np.roll(h, k)
            overlap = np.dot(h, 1 - shifted) * dx
            overlaps.append(overlap)
        max_overlap = max(overlaps)
        k_max = overlaps.index(max_overlap)
        
        # Compute gradient

        shifted_neg_k = np.roll(h, -k_max)
        grad = (1 - shifted_k) - shifted_neg_k
        grad *= dx
        
        # Update h
