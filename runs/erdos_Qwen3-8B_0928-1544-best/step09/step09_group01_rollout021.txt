        bounds = Bounds(0, 1)
        result = minimize(
            lambda h: evaluate(h),
            h0,
            bounds=bounds,
            method='L-BFGS-B',
            jac=lambda h: gradient(h, k_max)
        )
        return result.x
    
    t0 = time.time()
    while time.time() - t0 < T:
        k, c5 = get_max_k(best_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = optimize(best_h)
    return best_h, best_c5, n_points
