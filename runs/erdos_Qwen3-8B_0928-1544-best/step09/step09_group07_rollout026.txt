import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h with provided initial values or a structured pattern
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        # Construct a structured initial guess (alternating 1s and 0s)
        initial_h = np.zeros(n_points)
        initial_h[::2] = 1.0
        initial_h = np.clip(initial_h, 0, 1)

    # Define objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Constraint function for sum
    def constraint_sum(h):
        return np.sum(h) - target_sum

    # Bounds for h values
    bounds = [(0.0, 1.0)] * n_points

    # DE optimization with constraints enforced via penalty
    def objective_with_penalty(h):
        penalty = 1000 * abs(np.sum(h) - target_sum)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx + penalty

    # Run DE with penalty method
    result_de = differential_evolution(
        fun=objective_with_penalty,
        bounds=bounds,
        strategy='best1bin',
        popsize=100,
        tol=1e-8,
        mutation=(0.5, 1.0),
        recombination=0.9,
        maxiter=500,
        disp=False,
        polish=True,
        x0=initial_h
    )

    # Refine with SLSQP for better convergence
    result_slsqp = minimize(
        fun=objective,
        x0=result_de.x,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )

    # Final adjustments
    final_h = np.clip(result_slsqp.x, 0.0, 1.0)
    sum_h = np.sum(final_h)
    if abs(sum_h - target_sum) > 1e-8:
        scale = target_sum / sum_h
        final_h *= scale

    return final_h, result_slsqp.fun, n_points
