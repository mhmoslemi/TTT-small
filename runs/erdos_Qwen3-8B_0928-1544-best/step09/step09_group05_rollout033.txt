import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Diversity in initial population
    def generate_population(pop_size):
        pop = []
        # Include initial_h with random perturbation
        perturbed = initial_h + np.random.normal(0, 0.1, n_points)
        pop.append(np.clip(perturbed, 0, 1) * (target_sum / np.sum(np.clip(perturbed, 0, 1))))
        # Add random samples
        for _ in range(pop_size - 1):
            h_rand = np.random.uniform(0, 1, n_points)
            h_rand = np.clip(h_rand, 0, 1) * (target_sum / np.sum(h_rand))
            pop.append(h_rand)
        return pop
    
    pop_size = 150
    pop = generate_population(pop_size)
    best_h = initial_h.copy()
    best_c5 = obj(initial_h)
    start_time = time.time()
    
    # Global search with DE-like strategy
    for gen in range(800):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + 0.5 * (pop[a] - pop[b])  # Adjust mutation factor
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < 0.95:  # Crossover rate
                    trial[j] = mutant[j]
            trial_sum = np.sum(trial)
            trial = np.clip(trial, 0, 1) * (target_sum / trial_sum) if trial_sum != 0 else trial
            trial_obj = obj(trial)
            if trial_obj < obj(pop[i]):
                pop[i] = trial
                if trial_obj < best_c5:
                    best_c5 = trial_obj
                    best_h = trial.copy()
    
    # Local refinement with SLSQP
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        fun=obj,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 400, 'ftol': 1e-10, 'eps': 1e-9}
    )
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5
    
    return final_h, final_c5, n_points
