                child2 = child2 * target_sum / child2_sum
            new_population.append(child1)
            new_population.append(child2)

        combined = initial_population + new_population
        combined_fitness = [evaluate_fitness(h) for h in combined]
        indices = np.argsort(combined_fitness)
        initial_population = [combined[i] for i in indices[:pop_size]]

        current_best = initial_population[0]
        current_c5 = evaluate_fitness(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5

    return best_h, best_c5, n_points
