import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    dx = 2.0 / n_points

    bounds = [(0.0, 1.0)] * n_points
    start_time = time.time()
