import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 50
    max_no_improvement = 100
    F = 0.8
    CR = 0.9
    dx = 2.0 / n_points

    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        population.append(initial_h_scaled)
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5

                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)

    # Objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()
    no_improvement_count = 0

    while time.time() - start_time < budget_s:
