            offspring.append(child1)
            offspring.append(child2)
        
        # Mutate offspring
        mutated_offspring = []
        for child in offspring:
            noise = np.random.normal(0, 0.1, n_points)
            mutated = child + noise
            mutated = np.clip(mutated, 0, 1)
            mutated_sum = np.sum(mutated)
            mutated = mutated * target_sum / mutated_sum
            mutated_offspring.append(mutated)
        
        # Evaluate offspring
        offspring_fitness = np.array([evaluate_fitness(h) for h in mutated_offspring])
        
        # Combine parents and offspring
        combined = np.vstack((initial_population, mutated_offspring))
        combined_fitness = np.concatenate((population_fitness, offspring_fitness))
        
        # Select top half
        indices = np.argsort(combined_fitness)[:population_size]
        initial_population = combined[indices]
        population_fitness = combined_fitness[indices]
        
        # Update best
        current_best_idx = np.argmin(population_fitness)
        if population_fitness[current_best_idx] < best_c5:
            best_h = initial_population[current_best_idx]
            best_c5 = population_fitness[current_best_idx]
    
    return best_h, best_c5, n_points
