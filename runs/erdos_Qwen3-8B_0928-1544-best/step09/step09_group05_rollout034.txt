                movement = beta * delta * np.exp(-gamma * distance**2)
                pop[i] += alpha * np.random.normal(0, 0.1, n_points)
                pop[i] += movement
                pop[i] = np.clip(pop[i], 0, 1)
                pop[i] = pop[i] * (target_sum / np.sum(pop[i]))
        for h in pop:
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
    
    # Transition to 140 points if time remains
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0:
        n_points, target_sum = 140, 70
        best_h_replicated = np.repeat(best_h, 2)
        pop = [np.clip(best_h_replicated + 0.4 * np.random.normal(0, 0.1, n_points), 0, 1) * (target_sum / np.sum(np.clip(best_h_replicated + 0.4 * np.random.normal(0, 0.1, n_points), 0, 1))) for _ in range(pop_size)]
        alpha, beta, gamma = 0.5, 0.02, 0.5
        for _ in range(300):
            if time.time() - start_time > budget_s:
                break
            brightness = np.array([evaluate_erdos_solution(h) for h in pop])
            sorted_idx = np.argsort(brightness)
            for i in range(pop_size):
                if i == sorted_idx[0]:
                    continue
                j = np.random.choice(sorted_idx)
                if brightness[j] < brightness[i]:
                    delta = pop[j] - pop[i]
                    distance = np.linalg.norm(delta)
                    movement = beta * delta * np.exp(-gamma * distance**2)
                    pop[i] += alpha * np.random.normal(0, 0.1, n_points)
                    pop[i] += movement
                    pop[i] = np.clip(pop[i], 0, 1)
                    pop[i] = pop[i] * (target_sum / np.sum(pop[i]))
            for h in pop:
                c5 = evaluate_erdos_solution(h)
                if c5 < best_c5:
                    best_c5, best_h = c5, h.copy()
    
    # Local refinement
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        fun=lambda x: evaluate_erdos_solution(x),
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5
    
    return final_h, final_c5, n_points
