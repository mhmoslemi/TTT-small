    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    h_sum = np.sum(h)
    if h_sum != 0:
        h *= target_sum / h_sum
    else:
        h = np.full(n_points, 0.5)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    learning_rate = 0.05
    max_iterations = 1000
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        shifted_indices = (np.arange(n_points) + k) % n_points
        gradient = (1 - h[shifted_indices]) * dx
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h *= target_sum / h_sum
        else:
            h = np.full(n_points, 0.5)
        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        elif abs(new_c5 - best_c5) < 1e-5:
            learning_rate *= 0.5

    def obj(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        fun=obj,
        x0=best_h.flatten(),
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-9}
    )

    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5
    return final_h, final_c5, n_points
