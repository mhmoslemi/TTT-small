import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * ones_count
        h = np.clip(h, 0, 1)

    def project(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return np.zeros_like(h)
        scale = ones_count / sum_h
        return h_clipped * scale

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for _ in range(10000):
        if time.time() - start_time > budget_s:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)

        grad = np.zeros(n_points)
        for i in range(n_points):
            j_plus = (i + k) % n_points
            j_minus = (i - k) % n_points
            grad[i] = (1 - h[j_plus] - h[j_minus]) * dx

        current_c5 = compute_c5(h)
        alpha = 0.1
        beta = 0.5
        for _ in range(20):
            h_candidate = h - alpha * grad
            h_proj = project(h_candidate)
            corr_candidate = np.correlate(h_proj, 1 - h_proj, mode='full')
            new_c5 = np.max(corr_candidate) * dx
            if new_c5 <= current_c5 - alpha * np.dot(grad, grad) * 0.1:
                break
            alpha *= beta
            if alpha < 1e-8:
                break
        h = h_proj
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5

    return best_h, best_c5, n_points
