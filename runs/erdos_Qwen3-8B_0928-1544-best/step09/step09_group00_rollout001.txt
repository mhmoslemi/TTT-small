import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    pop_size = 30
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = initial_h.copy()
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (ones_count / h_sum)
        population.append(h.copy())
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    F = 0.8
    CR = 0.9
    no_improve = 0
    
    while True:
        if time.time() - start_time >= budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = np.copy(mutant)
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial = np.zeros(n_points)
            else:
                trial = trial * (ones_count / trial_sum)
            current_c5 = objective(trial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = trial.copy()
            if current_c5 < objective(population[i]):
                population[i] = trial.copy()
        
        if objective(best_h) < best_c5:
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 10:
            F = max(0.5, F - 0.05)
            CR = min(1.0, CR + 0.05)
            no_improve = 0
    
    # Refinement with convex optimization
    h_var = cp.Variable(n_points, name='h')
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == ones_count]
    objective_func = cp.Minimize(cp.Max(cp.correlate(h_var, 1 - h_var, mode='full') * (2.0 / n_points)))
    prob = cp.Problem(objective_func, constraints)
    prob.solve()
    
    if prob.status == 'optimal':
        best_h = h_var.value
    else:
        best_h = best_h
    
    return best_h, best_c5, n_points
