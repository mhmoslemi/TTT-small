import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    pop_size = 50
    population = [initial_h.copy() for _ in range(pop_size)]
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    F = 0.8
    CR = 0.9
    F_decay = 0.95
    CR_inc = 0.05
    no_improvement = 0
    
    def refine_solution(h):
        constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count})
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full') * (2.0 / n_points)), h, 
                       method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-5)
        if res.success:
            return res.x
        return h
    
    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = np.copy(mutant)
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial = np.zeros(n_points)
            else:
                trial = trial * (ones_count / trial_sum)
            
            current_c5 = objective(trial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = trial.copy()
            
            if current_c5 < objective(population[i]):
                population[i] = trial.copy()
        
        if objective(best_h) < best_c5:
            no_improvement = 0
            best_h = refine_solution(best_h)
        else:
            no_improvement += 1
        
        if no_improvement > 10:
            F *= F_decay
            CR = np.clip(CR + CR_inc, 0.7, 1.0)
            no_improvement = 0
    
    return best_h, best_c5, n_points
