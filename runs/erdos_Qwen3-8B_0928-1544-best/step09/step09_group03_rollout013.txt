import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()

        h = np.full(n_points, 0.5)
    total = np.sum(h)
    scaling = (n_points / 2) / total
    h = h * scaling

    def scale(arr):
        total = np.sum(arr)
        if total == 0:
            return np.zeros_like(arr)

        scaled = arr * scaling
        return np.clip(scaled, 0.0, 1.0)

    best_h = h.copy()

        eps = 1e-5

            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = scale(h_plus)
            c_plus = evaluate_erdos_solution(h_plus)
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = scale(h_minus)
