import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = initial_h_values.copy()
    
    pop_size = 200
    mutation_rate = 0.1
    crossover_rate = 0.7
    tournament_size = 5
    max_iter = int(budget_s / 1.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def rescale(h):
        return h * target_sum / np.sum(h)
    
    population = np.zeros((pop_size, n_points))
    for i in range(pop_size):
        h = initial_h.copy()
        noise = np.random.normal(0, 0.2, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        population[i] = rescale(h)
    
    best_h = population[0]
    best_c5 = objective(best_h)
    
    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        fitness = np.array([objective(ind) for ind in population])
        cur_best_idx = np.argmin(fitness)
        if fitness[cur_best_idx] < best_c5:
            best_c5 = fitness[cur_best_idx]
            best_h = population[cur_best_idx]
        
        parents = []
        for _ in range(pop_size // 2):
            idxs = np.random.choice(pop_size, tournament_size, replace=False)
            winner = idxs[np.argmin(fitness[idxs])]
            parents.append(population[winner])
        
        offspring = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1] if i+1 < len(parents) else parents[i]
            if np.random.rand() < crossover_rate:
                pt = np.random.randint(1, n_points - 1)
                child1 = np.concatenate((p1[:pt], p2[pt:]))
                child2 = np.concatenate((p2[:pt], p1[pt:]))
            else:
                child1, child2 = p1.copy(), p2.copy()
            offspring.append(child1)
            offspring.append(child2)
        
        for i in range(len(offspring)):
            child = offspring[i]
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = rescale(child)
            offspring[i] = child
        
        population = np.vstack((population, offspring))
        population = population[np.argsort(fitness[np.arange(pop_size)] + np.random.rand(pop_size) * 0.01)[:pop_size]]
    
    return best_h, best_c5, n_points
