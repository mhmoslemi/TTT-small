                mutant = initial_h.copy()
            
            mutant_c5 = evaluate(mutant)
            current_c5 = evaluate(population[i])
            
            if mutant_c5 < current_c5:
                population[i] = mutant.copy()
        
        current_best = population[0]
        current_best_c5 = evaluate(current_best)
        for h in population[1:]:
            h_c5 = evaluate(h)
            if h_c5 < current_best_c5:
                current_best = h
                current_best_c5 = h_c5
        
        if current_best_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_best_c5
        
        mutation_strength *= mutation_decay
    
    return best_h, best_c5, n_points
