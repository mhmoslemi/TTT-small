def run(seed=42, budget_s=1100, **kwargs):

        initial_h = np.full(n_points, 0.5)
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    
    n_particles = 50
    particles = np.random.rand(n_particles, n_points)
    particles = np.clip(particles, 0.0, 1.0)
    particles = particles * (ones_count / np.sum(particles, axis=1, keepdims=True))
    particles[0] = initial_h.copy()
    particles[0] = particles[0] * (ones_count / np.sum(particles[0]))
    
    velocities = np.random.rand(n_particles, n_points) * 0.1
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    pbest = particles.copy()
    pbest_values = np.array([evaluate(p) for p in particles])
    gbest = particles[np.argmin(pbest_values)]
    gbest_value = np.min(pbest_values)
    best_h = gbest
    best_c5 = gbest_value
    
    start_time = time.time()
    time_remaining = budget_s
    
    for _ in range(100):  # Limit iterations to avoid infinite loops
        if time.time() - start_time >= budget_s:
            break

            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = 0.8 * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -1.0, 1.0)
            particles[i] += velocities[i]
        
        for i in range(n_particles):
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            particles[i] = particles[i] * (ones_count / np.sum(particles[i]))
        
        for i in range(n_particles):

            if current_c5 < pbest_values[i]:
