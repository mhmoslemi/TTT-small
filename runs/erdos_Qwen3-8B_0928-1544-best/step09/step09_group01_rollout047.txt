import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    population_size = 50
    F = 0.8
    CR = 0.7
    elite_ratio = 0.2

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        constrained_h = initial_h * (ones_count / np.sum(initial_h))
        population = [constrained_h]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            constrained_h = h * (ones_count / h_sum)
            population.append(constrained_h)
    else:
        # Thue-Morse-inspired initial guess
        thue_morse = np.array([bin(i).count('1') % 2 for i in range(n_points)])
        thue_sum = np.sum(thue_morse)
        constrained_h = thue_morse * (ones_count / thue_sum)
        population = [constrained_h]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            constrained_h = h * (ones_count / h_sum)
            population.append(constrained_h)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()

    while time.time() - start_time < budget_s:
        fitness = [(objective(h), i) for i, h in enumerate(population)]
        fitness.sort()
        elites = [population[i] for _, i in fitness[:int(elite_ratio * population_size)]]
        new_pop = elites.copy()
        while len(new_pop) < population_size:
            a, b, c = np.random.choice(len(population), 3, replace=False)
            v = population[a] + F * (population[b] - population[c])
            u = population[a].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0, 1)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            new_c5 = objective(u_scaled)
            if new_c5 < fitness[-1][0]:
                new_pop.append(u_scaled)
        population = new_pop

        current_best = min(population, key=objective)
        if objective(current_best) < best_c5:
            best_c5 = objective(current_best)
            best_h = current_best.copy()

    return best_h, best_c5, n_points
