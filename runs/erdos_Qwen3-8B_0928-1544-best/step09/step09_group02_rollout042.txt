import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    pop_size = 50
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals() and np.random.rand() < 0.3:
            h = initial_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (ones_count / np.sum(h))
        else:
            h = np.random.rand(n_points)
            h = h * (ones_count / np.sum(h))
        population.append(h)
    
    def compute_C5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    
    def eval_fitness(h):
        return compute_C5(h)
    
    fitness = [eval_fitness(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_score = fitness[best_idx]
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        new_pop = []
        for _ in range(pop_size):
            parent1_idx = np.random.choice(pop_size)
            parent2_idx = np.random.choice(pop_size)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            if np.random.rand() < 0.8:
                split_idx = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:split_idx], parent2[split_idx:]])
            else:
                child = parent1
            
            if np.random.rand() < 0.2:
                mutation = np.random.normal(0, 0.05, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
            
            child_sum = np.sum(child)
            if child_sum != 0:
                child = child * (ones_count / child_sum)
            else:
                child = np.full(n_points, 0.5)
            
            new_pop.append(child)
        
        new_fitness = [eval_fitness(h) for h in new_pop]
        combined = sorted(zip(population, fitness, new_pop, new_fitness),
                          key=lambda x: x[3])
        population = [x[2] for x in combined[:pop_size]]
        fitness = [x[3] for x in combined[:pop_size]]
        
        if np.min(fitness) < best_score:
            best_score = np.min(fitness)
            best_h = combined[np.argmin(fitness)][2]
    
    return best_h, best_score, n_points
