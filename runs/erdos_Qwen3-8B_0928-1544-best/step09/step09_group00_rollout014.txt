import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    a_initial = np.full(n_points, 0.0)
    if 'initial_h_values' in globals():
        a_initial = initial_h_values.copy() - 0.5
        a_initial -= a_initial.mean()
    else:
        a_initial = np.random.rand(n_points) * 1.0 - 0.5
        a_initial -= a_initial.mean()
    particles = np.random.rand(30, n_points) * 1.0 - 0.5
    particles[:, :] = a_initial[np.newaxis, :] + (particles - a_initial[np.newaxis, :]) * 0.1
    particles -= particles.mean(axis=1, keepdims=True)
    velocities = np.random.rand(30, n_points) * 0.1
    personal_best = particles.copy()
    personal_best_vals = np.inf * np.ones(30)
    global_best = particles[0].copy()
    global_best_val = float('inf')
    
    def evaluate(a):
        h = 0.5 + a
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        for i in range(30):
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.7 * velocities[i] + 2.0 * r1 * (personal_best[i] - particles[i]) + 2.0 * r2 * (global_best - particles[i])
            particles[i] = particles[i] + velocities[i]
            a_clipped = np.clip(particles[i], -0.5, 0.5)
            sum_clipped = np.sum(a_clipped)
            a_clipped -= sum_clipped / n_points
            particles[i] = a_clipped
            curr_val = evaluate(particles[i])
            if curr_val < personal_best_vals[i]:
                personal_best_vals[i] = curr_val
                personal_best[i] = particles[i].copy()
            if curr_val < global_best_val:
                global_best_val = curr_val
                global_best = particles[i].copy()
    
    best_a = global_best
    best_h = 0.5 + best_a
    return best_h, global_best_val, n_points
