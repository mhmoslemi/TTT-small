import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    # Precompute shifts for full convolution
    shift_indices = np.zeros((2 * n_points - 1, n_points), dtype=int)
    for pos in range(2 * n_points - 1):
        k = pos - (n_points - 1)
        for i in range(n_points):
            shift_indices[pos, i] = (i + k) % n_points
    
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    constraints = []
    for pos in range(2 * n_points - 1):
        expr = 0
        for i in range(n_points):
            expr += h[i] * (1 - h[shift_indices[pos, i]])
        constraints.append(expr <= t)
    
    constraints.append(cp.sum(h) == ones_count)
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Objective: minimize t
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    while time.time() - start_time < budget_s:
        try:
            prob.solve(max_time=10, eps=1e-4)
            if prob.status == 'optimal':
                current_h = h.value
                # Normalize to ensure sum is correct
                current_h = current_h * (ones_count / np.sum(current_h))
                current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full') * (2.0 / n_points))
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
        except:
            pass
    
    return best_h, best_c5, n_points
