        trial /= np.sum(trial) * dx
        trial *= target_sum
        trial = optimize(trial)
        c5 = objective(trial)
        if c5 < best_c5:
            best_h, best_c5 = trial, c5

    # Finale refinement
    if time.time() - start_time < budget_s:
        result = minimize(
            objective, best_h, method='L-BFGS-B',
            bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'ftol': 1e-10, 'maxiter': 200, 'eps': 1e-8}
        )
        best_h = result.x if result.success else best_h
        best_c5 = objective(best_h)

    return best_h, best_c5, n_points
