from scipy.optimize import minimize
import time

def make_symmetric(h, n_points):
    if n_points % 2 == 0:
        half = n_points // 2
        return np.concatenate([h[:half], h[:half][::-1]])
    else:
        half = n_points // 2
        return np.concatenate([h[:half], [h[half]], h[:half][::-1]])

    def objective(h):

        corr = np.correlate(h_sym, 1 - h_sym, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
