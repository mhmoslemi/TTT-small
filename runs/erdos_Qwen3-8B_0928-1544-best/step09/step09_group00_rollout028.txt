            curr_c5 = compute_c5(new_h, n_points)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = new_h.copy()
        
        # Optimize right
        def obj_right(x):
            new_h = np.concatenate([left, x])
            return compute_c5(new_h, n_points)
        cons_right = ({'type': 'eq', 'fun': lambda x: np.sum(x) - (total_sum - np.sum(left))})
        bnds_right = [(0., 1.) for _ in range(len(right))]
        res_right = minimize(obj_right, right, method='SLSQP', bounds=bnds_right, constraints=cons_right)
        if res_right.success:
            right = res_right.x
            new_h = np.concatenate([left, right])
            curr_c5 = compute_c5(new_h, n_points)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = new_h.copy()
    
    return best_h, best_c5, n_points
