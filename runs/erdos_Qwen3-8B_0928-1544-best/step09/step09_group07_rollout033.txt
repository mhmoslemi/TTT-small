import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_low = 35
    n_high = 70
    dx_low = 2.0 / n_low
    dx_high = 2.0 / n_high
    target_sum_low = n_low / 2
    target_sum_high = n_high / 2

    if 'initial_h_values' in globals():
        h_initial_low = initial_h_values[::2]
    else:
        h_initial_low = np.zeros(n_low)
        h_initial_low[:int(n_low/2)] = 1
        if n_low % 2 == 1:
            h_initial_low[int(n_low/2)] += 0.5
        h_initial_low = np.clip(h_initial_low, 0.0, 1.0)

    def coarse_objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx_low

    def coarse_constraint(h):
        return np.sum(h) - target_sum_low

    bounds_coarse = [(0.0, 1.0)] * n_low
    result_coarse = minimize(
        fun=coarse_objective,
        x0=h_initial_low,
        method='SLSQP',
        bounds=bounds_coarse,
        constraints={'type': 'eq', 'fun': coarse_constraint},
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9}
    )

    h_high = np.repeat(result_coarse.x, 2)
    assert np.isclose(np.sum(h_high), target_sum_high)

    def high_objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx_high

    def high_constraint(h):
        return np.sum(h) - target_sum_high

    bounds_high = [(0.0, 1.0)] * n_high
    result_high = minimize(
        fun=high_objective,
        x0=h_high,
        method='SLSQP',
        bounds=bounds_high,
        constraints={'type': 'eq', 'fun': high_constraint},
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )

    c5_coarse = result_coarse.fun
    c5_high = result_high.fun
    best_h = result_high.x if c5_high < c5_coarse else result_coarse.x
    best_c5 = min(c5_coarse, c5_high)
    return best_h, best_c5, n_high
