        best_index = np.argmin(current_values)
        best_candidate = new_population[best_index]
        if evaluate(best_candidate) < best_c5:
            best_c5 = evaluate(best_candidate)
            best_h = best_candidate.copy()
        population = new_population
        if np.abs(current_values[best_index] - best_c5) < 1e-6:
            mutation_strength *= mutation_decay
    return best_h, best_c5, n_points
