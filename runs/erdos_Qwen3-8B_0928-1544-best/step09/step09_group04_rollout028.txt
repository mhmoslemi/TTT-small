    best_h = initial_h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx]
        tournament_size = int(pop_size * 0.2)
        parents = []
        for _ in range(pop_size):
            idx = np.random.choice(pop_size, tournament_size)
            winner = idx[np.argmin(fitness[idx])]
            parents.append(population[winner])
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 >= len(parents):
                break
            p1, p2 = parents[i], parents[i+1]
            split = np.random.randint(1, n_points - 1)
            c1 = np.concatenate([p1[:split], p2[split:]])
            c2 = np.concatenate([p2[:split], p1[split:]])
            noise1 = np.random.normal(0, 0.1, n_points)
            noise2 = np.random.normal(0, 0.1, n_points)
            c1 += noise1
            c2 += noise2
            c1 = np.clip(c1, 0, 1)
            c2 = np.clip(c2, 0, 1)
            c1 = c1 * target_sum / np.sum(c1)
            c2 = c2 * target_sum / np.sum(c2)
            offspring.append(c1)
            offspring.append(c2)
        population = offspring[:pop_size]
    
    return best_h, best_c5, n_points
