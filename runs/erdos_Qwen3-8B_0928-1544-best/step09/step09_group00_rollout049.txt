    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def clamp_and_scale(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * (ones_count / np.sum(h))
        return h
    
    h_initial = initial_h.copy()
    h_initial = clamp_and_scale(h_initial)
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    
    pop_size = 30
    population = [h_initial.copy() for _ in range(pop_size)]
    for i in range(1, pop_size):
        h = np.random.rand(n_points)
        h = clamp_and_scale(h)
        population[i] = h.copy()
    
    F = 0.7
    CR = 0.9
    n_gen = 0
    start_time = time.time()
    prev_best_c5 = float('inf')
    no_improvement_count = 0
    max_no_improvement = 50
    
    while True:
        if time.time() - start_time >= budget_s:
            break
        n_gen += 1
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = mutant.copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = clamp_and_scale(trial)
            current_c5 = objective(trial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = trial.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
            if current_c5 < objective(population[i]):
                population[i] = trial.copy()
            if no_improvement_count >= max_no_improvement:
                break
        if best_c5 < prev_best_c5:
            prev_best_c5 = best_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        if no_improvement_count >= max_no_improvement * 2:
            break
    
    return best_h, best_c5, n_points
