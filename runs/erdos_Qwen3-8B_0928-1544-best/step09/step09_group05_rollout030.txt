import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with provided or uniform distribution
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    h = initial_h.copy()
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))

    def obj(h_vec):
        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        return np.max(corr) * dx

    # Iterative gradient descent
    learning_rate = 0.1
    max_iter = 2000
    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_k = np.argmax(corr)
        shift = max_k

        # Compute gradient for the worst shift
        gradient = np.zeros_like(h)
        for i in range(n_points):
            j = i + shift
            if 0 <= j < n_points:
                gradient[i] = 1.0 - h[j]
        grad_norm = np.linalg.norm(gradient)

        # Avoid division by zero
        if grad_norm == 0:
            continue

        # Gradient descent step
        h -= learning_rate * gradient / grad_norm
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))

        current_c5 = obj(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        elif current_c5 >= best_c5 and learning_rate < 1e-8:
            break
        learning_rate *= 0.9995

    # Local refinement with SLSQP
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        fun=obj,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
    )
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5

    return final_h, final_c5, n_points
