from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

        return np.max(corr) * dx

    pop_size = 100
    max_gen = 600
    F, CR = 0.8, 0.9
    pop = [np.clip(initial_h + np.random.normal(0, 0.1, n_points), 0, 1) * (target_sum / np.sum(np.clip(initial_h + np.random.normal(0, 0.1, n_points), 0, 1))) for _ in range(pop_size)]
    
    best_h = initial_h.copy()
    best_c5 = obj(initial_h)
    start_time = time.time()
    
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):

            mutant = pop[a] + F * (pop[b] - pop[c])
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial_sum = np.sum(trial)

            trial_obj = obj(trial)

                if trial_obj < best_c5:
                    best_c5 = trial_obj
