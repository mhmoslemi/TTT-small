import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * (ones_count / np.sum(h))
    start_time = time.time()

    best_h = h.copy()
    best_C5 = current_C5
    T = current_C5 * 1.5
    alpha = 0.999
    T_min = 1e-5
    
    while time.time() - start_time < budget_s:
        for _ in range(100):
            i = np.random.randint(n_points)
            j = np.random.randint(n_points)
            delta = np.random.rand() * min(h[i], 1 - h[j])
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            new_C5 = compute_C5(h_new)
            delta_C5 = new_C5 - current_C5

                h = h_new.copy()

                best_C5 = current_C5
                best_h = h.copy()
        T *= alpha
        if T < T_min:
            break
    
    return best_h, best_C5, n_points
