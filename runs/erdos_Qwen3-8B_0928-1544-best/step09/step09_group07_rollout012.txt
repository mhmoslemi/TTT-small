    else:
        # Structured initial pattern: alternate 1.0 and 0.0, with last 1.0 replaced by 0.5
        low_res_n = 35
        pattern = np.zeros(low_res_n)
        pattern[::2] = 1.0
        pattern[-1] = 0.5  # Adjust to sum 17.5
        h_high = np.repeat(pattern, 2)
        initial_h = h_high
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    
    # Differential Evolution setup
    pop_size = 100
    max_gen = 500
    F = 0.7
    CR = 0.9
    
    # Initialize population
    pop = []
    for _ in range(pop_size):
        h = initial_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (target_sum / h_sum)
        pop.append(h)
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + F * (pop[a] - pop[b])
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0.0, 1.0)
            trial_sum = np.sum(trial)
            if trial_sum != 0:
                trial = trial * (target_sum / trial_sum)
            trial_obj = objective(trial)
            if trial_obj < objective(pop[i]):
                pop[i] = trial
                if trial_obj < best_c5:
                    best_c5 = trial_obj
                    best_h = trial.copy()
    
    # SLSQP refinement
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5
    
    return final_h, final_c5, n_points
