import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        a = initial_h - 0.5
    else:
        a = np.random.uniform(-0.5, 0.5, n_points)
        a[-1] = a[-1] - np.sum(a)
    
    def clip_and_normalize(a):
        a = np.clip(a, -0.5, 0.5)
        total = np.sum(a)
        a[-1] = a[-1] - total
        a[-1] = np.clip(a[-1], -0.5, 0.5)
        return a
    
    a = clip_and_normalize(a)
    
    def objective(a):
        h = 0.5 + a
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / n_points)
    
    def eval_func(x):
        x = clip_and_normalize(x)
        return objective(x)
    
    # PSO parameters
    omega = 0.7
    c1 = 2.0
    c2 = 2.0
    vel = np.zeros_like(a)
    particles = [a.copy()]
    pbest = [objective(a)]
    gbest = objective(a)
    gbest_pos = a.copy()
    
    start_time = time.time()
    
    for _ in range(1000):
        if time.time() - start_time >= budget_s:
            break
        for i in range(len(particles)):
            r1, r2 = np.random.rand(), np.random.rand()
            vel[i] = omega * vel[i] + c1 * r1 * (particles[i] - particles[i])
            vel[i] = omega * vel[i] + c2 * r2 * (gbest_pos - particles[i])
            particles[i] = particles[i] + vel[i]
            particles[i] = clip_and_normalize(particles[i])
            cost = objective(particles[i])
            if cost < pbest[i]:
                pbest[i] = cost
                particles[i] = clip_and_normalize(particles[i])
            if cost < gbest:
                gbest = cost
                gbest_pos = particles[i].copy()
    
    best_h = 0.5 + gbest_pos
    return best_h, gbest, n_points
