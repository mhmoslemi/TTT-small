    current_C5 = compute_C5(h)
    best_C5 = current_C5
    best_h = h.copy()

    T = 0.1 * current_C5
    if T < 1e-5:
        T = 1e-5

    alpha = 0.995
    min_T = 1e-5
    start_time = time.time()

    while time.time() - start_time < budget_s:
        i, j = np.random.choice(n_points, 2, replace=False)
        max_transfer = min(h[i], 1.0 - h[j])
        delta = np.random.uniform(0.0, 0.1) * max_transfer
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        h_new = np.clip(h_new, 0.0, 1.0)

        new_C5 = compute_C5(h_new)
        delta_C5 = new_C5 - current_C5
        if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / T):
            current_C5 = new_C5
            h = h_new

        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()

        T *= alpha

    return best_h, best_C5, n_points
