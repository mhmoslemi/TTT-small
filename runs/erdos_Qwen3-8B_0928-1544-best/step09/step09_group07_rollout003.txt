    n_low = 35
    n_high = 70
    dx_low = 2.0 / n_low
    target_sum_low = n_low / 2
    dx_high = 2.0 / n_high
    target_sum_high = n_high / 2
    
    # Generate initial low-res pattern
    h_low = np.zeros(n_low)
    for i in range(n_low):
        if i % 2 == 1:
            h_low[i] = 1.0
    h_low[-1] = 0.5
    
    def compute_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # DE for low-res
    pop_size_low = 100
    F_low = 0.8
    CR_low = 0.9
    max_gen_low = 300
    start_time = time.time()
    
    pop_low = [h_low + np.random.normal(0, 0.05, n_low) for _ in range(pop_size_low)]
    for h in pop_low:
        h = np.clip(h, 0, 1)
        if np.sum(h) != 0:
            h *= target_sum_low / np.sum(h)
        h_low = h
    
    best_h_low = h_low.copy()
    best_c5_low = compute_c5(best_h_low, dx_low)
    
    for gen in range(max_gen_low):
        if time.time() - start_time > budget_s - 200:
            break
        for i in range(pop_size_low):
            a, b, c = np.random.choice(pop_size_low, 3, replace=False)
            mutant = pop_low[a] + F_low * (pop_low[a] - pop_low[b])
            trial = pop_low[i].copy()
            for j in range(n_low):
                if np.random.rand() < CR_low:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            if np.sum(trial) != 0:
                trial *= target_sum_low / np.sum(trial)
            trial_c5 = compute_c5(trial, dx_low)
            if trial_c5 < compute_c5(pop_low[i], dx_low):
                pop_low[i] = trial.copy()
                if trial_c5 < best_c5_low:
                    best_h_low = trial.copy()
                    best_c5_low = trial_c5
    
    # Upsample to high-res
    h_high = np.repeat(best_h_low, 2)
    h_high = np.clip(h_high, 0, 1)
    if np.sum(h_high) != target_sum_high:
        h_high *= target_sum_high / np.sum(h_high)
    
    # DE for high-res
    pop_size_high = 200
    F_high = 0.7
    CR_high = 0.8
    max_gen_high = 400
    start_time = time.time()
    
    pop_high = [h_high + np.random.normal(0, 0.05, n_high) for _ in range(pop_size_high)]
    for h in pop_high:
        h = np.clip(h, 0, 1)
        if np.sum(h) != 0:
            h *= target_sum_high / np.sum(h)
    
    best_h_high = h_high.copy()
    best_c5_high = compute_c5(best_h_high, dx_high)
    
    for gen in range(max_gen_high):
        if time.time() - start_time > budget_s - 100:
            break
        for i in range(pop_size_high):
            a, b, c = np.random.choice(pop_size_high, 3, replace=False)
            mutant = pop_high[a] + F_high * (pop_high[a] - pop_high[b])
            trial = pop_high[i].copy()
            for j in range(n_high):
                if np.random.rand() < CR_high:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            if np.sum(trial) != 0:
                trial *= target_sum_high / np.sum(trial)
            trial_c5 = compute_c5(trial, dx_high)
            if trial_c5 < compute_c5(pop_high[i], dx_high):
                pop_high[i] = trial.copy()
                if trial_c5 < best_c5_high:
                    best_h_high = trial.copy()
                    best_c5_high = trial_c5
    
    # SLSQP refinement
    bounds = [(0.0, 1.0)] * n_high
    constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum_high}
    result = minimize(
        fun=lambda h: compute_c5(h, dx_high),
        x0=best_h_high,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
    )
    
    final_h = result.x if result.success else best_h_high
    final_c5 = result.fun if result.success else best_c5_high
    return final_h, final_c5, n_high
