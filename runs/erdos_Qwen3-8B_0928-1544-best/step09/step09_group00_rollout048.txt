import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    pop_size = 20
    population = [initial_h.copy() for _ in range(pop_size)]
    for i in range(pop_size):
        if i == 0:
            continue
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (ones_count / h_sum)
        population[i] = h.copy()
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    F = 0.8
    CR = 0.9
    
    while True:
        if time.time() - start_time >= budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = np.copy(mutant)
            for j in range(n_points):

                    trial[j] = mutant[j]
