    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0)] * n_points

    # Improved Genetic Algorithm Parameters
    max_pop_size = 200
    F = 0.9
    CR = 0.85
    max_generations = 1000
    pop = []
    for _ in range(max_pop_size):
        h = initial_h + np.random.normal(0, 0.15, n_points)
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        h = h * (target_sum / h_sum) if h_sum != 0 else h
        pop.append(h)

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()

    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        for i in range(max_pop_size):
            a, b, c = np.random.choice(max_pop_size, 3, replace=False)
            mutant = pop[a] + F * (pop[a] - pop[b])
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial_sum = np.sum(trial)
            if trial_sum != 0:
                trial = trial * (target_sum / trial_sum)
            trial_obj = objective(trial)
            if trial_obj < objective(pop[i]):
                pop[i] = trial
                if trial_obj < best_c5:
                    best_c5 = trial_obj
                    best_h = trial.copy()

    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 500, 'ftol': 1e-12, 'eps': 1e-9}
    )
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5

    return final_h, final_c5, n_points
