import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    # Initialize the h
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = np.inf
    
    # Learning rate parameters
    learning_rate = 0.5
    learning_rate_decay = 0.95
    # Max iterations
    iters = 0
    max_iters = int(budget_s / (n_points**2))  # Adjust based on time per iteration
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Compute correlations for all shifts
        corr = np.correlate(h, 1 - h, mode='full')
        # The maximum value is at the first n_points positions
        corr_max = corr[:n_points]
        current_max = np.max(corr_max) * (2.0 / n_points)
        
        # If found a better solution
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        
        # Find the k_max (shift) that gives the maximum
        k_max = np.argmax(corr_max)
        
        # Compute the gradient for this k_max
        grad = np.zeros(n_points)
        for j in range(n_points):
            # (j + k_max) mod n_points
            j_plus_k = (j + k_max) % n_points
            # (j - k_max) mod n_points
            j_minus_k = (j - k_max) % n_points
            grad[j] = (1 - h[j_plus_k]) - h[j_minus_k]
        grad *= (2.0 / n_points)  # dx is 2.0 / n_points
        
        # Update h with gradient descent
        h -= learning_rate * grad
        
        # Clip and normalize
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        if total > 0:
            h = h * (ones_count / total)
        else:
            h = np.full(n_points, 0.5)
        
        # Decay learning rate
        learning_rate *= learning_rate_decay
        
        iters += 1
    
    return best_h, best_c5, n_points
