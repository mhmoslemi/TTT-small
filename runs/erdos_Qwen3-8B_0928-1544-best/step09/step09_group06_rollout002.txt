import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    h = np.clip(initial_h, 0.0, 1.0)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = float('inf')
    
    max_iter = 1000
    learning_rate = 0.1
    decay = 0.995
    noise_freq = 200
    start_time = time.time()
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k_max = max_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for j in range(n_points):
            pos1 = j + k_max
            term1 = (1 - h[pos1]) * dx if 0 <= pos1 < n_points else 0.0
            pos2 = j - k_max
            term2 = -h[pos2] * dx if 0 <= pos2 < n_points else 0.0
            grad[j] = term1 + term2
        
        h = h - learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        h = h * target_sum / np.sum(h)
        
        curr_c5 = corr[np.argmax(corr)] * dx
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()

            h = h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0.0, 1.0)

        learning_rate *= decay
