    generations = 200
    best_low = initial_low.copy()
    best_c5_low = float('inf')

    def eval_c5_low(h_low):
        h_high = np.repeat(h_low, 2)
        corr = np.correlate(h_high, 1 - h_high, mode='full')
        return np.max(corr) * dx

    # Initialize population
    pop = [initial_low.copy() for _ in range(pop_size)]
    for _ in range(pop_size):
        h = pop[_].copy()
        for i in range(n_low):
            if np.random.rand() < 0.1:
                h[i] = 1 - h[i]
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (17.5 / h_sum)
        pop[_] = h

    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for _ in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a].copy()
            for j in range(n_low):
                if np.random.rand() < 0.8:
                    mutant[j] = pop[b][j]
            mutant_sum = np.sum(mutant)
            if mutant_sum > 0:
                mutant = mutant * (17.5 / mutant_sum)
            c5 = eval_c5_low(mutant)
            if c5 < best_c5_low:
                best_c5_low = c5
                best_low = mutant.copy()
            new_pop.append(mutant)
        pop = new_pop

    # Upscale to high-resolution
    best_h_high = np.repeat(best_low, 2)

    # Differential Evolution
    def objective_high(h_high):
        h = h_high.copy()
        h_sum = np.sum(h)
        if abs(h_sum - target_sum) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
    result_de = differential_evolution(
        objective_high, bounds, maxiter=500, popsize=50, tol=1e-10, mutation=(0.5, 1), recombination=0.8, strategy='best1bin', polish=True, disp=False, seed=seed
    )

    best_h_de = result_de.x
    if result_de.success:
        # SLSQP refinement
        def objective_slsqp(h):
            h_sum = np.sum(h)
            if abs(h_sum - target_sum) > 1e-6:
                return np.inf
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        def constraint_sum(x):
            return np.sum(x) - target_sum

        result_slsqp = minimize(
            objective_slsqp, best_h_de, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
        )
        best_h = result_slsqp.x
        final_c5 = result_slsqp.fun
    else:
        best_h = best_h_de
        final_c5 = result_de.fun

    # Final sum correction
    best_h = best_h * (target_sum / np.sum(best_h)) if np.sum(best_h) > 0 else best_h
    return best_h, final_c5, n_points
