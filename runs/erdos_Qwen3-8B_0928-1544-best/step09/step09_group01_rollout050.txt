    h = np.zeros(n_points)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        while h_sum == 0:
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
        h = h * (ones_count / h_sum)
    
    def compute_c5(h):
        overlaps = []
        for k in range(n_points):
            shifted = np.roll(h, k)
            overlap = np.dot(h, 1 - shifted)
            overlaps.append(overlap)
        return max(overlaps) * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    alpha = 0.1
    step_decay = 0.5
    max_steps = 10000
    start_time = time.time()
    
    for _ in range(max_steps):
        if time.time() - start_time > budget_s:
            break
        
        # Compute overlaps
        overlaps = []
        for k in range(n_points):
            shifted = np.roll(h, k)
            overlap = np.dot(h, 1 - shifted)
            overlaps.append(overlap)
        current_max = max(overlaps)
        k_max = overlaps.index(current_max)
        
        # Compute gradient
        shifted_forward = np.roll(h, k_max)
        shifted_backward = np.roll(h, -k_max)
        grad = 1 - (shifted_forward + shifted_backward)
        
        # Step direction
        direction = -grad
        
        # Step
        h_new = h + alpha * direction
        
        # Clip to [0, 1]
        h_new = np.clip(h_new, 0.0, 1.0)
        
        # Rescale to maintain sum constraint
        h_sum_new = np.sum(h_new)
        if h_sum_new == 0:
            continue
        h_new = h_new * (ones_count / h_sum_new)
        
        # Evaluate new candidate
        new_c5 = compute_c5(h_new)
        
        # Update best
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            alpha = max(alpha * 1.1, 0.1)
        else:
            alpha *= step_decay
        
        # Early stopping if alpha becomes too small
        if alpha < 1e-6:
            break
    
    return best_h, best_c5, n_points
