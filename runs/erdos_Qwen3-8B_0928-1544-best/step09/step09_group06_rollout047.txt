import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    h = np.clip(h, 0.0, 1.0)
    h_sum = np.sum(h)
    h = h * target_sum / h_sum
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    learning_rate = 1e-3
    learning_rate_decay = 0.99
    no_improvement = 0
    max_no_improvement = 100
    noise_magnitude = 1e-3
    
    start_time = time.time()
    
    for _ in range(10000):
        if time.time() - start_time > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k_max = max_index - (n_points - 1)
        gradient = np.zeros(n_points)
        for j in range(n_points):
            i_plus = (j + k_max) % n_points
            i_minus = (j - k_max) % n_points
            gradient[j] = (1 - h[i_plus]) - h[i_minus]
        gradient *= dx
        
        h -= learning_rate * gradient
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        h = h * target_sum / h_sum
        
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        if no_improvement >= max_no_improvement:
            learning_rate *= learning_rate_decay
            no_improvement = 0
        
        if _ % 100 == 0:
            h += noise_magnitude * np.random.randn(n_points)
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            h = h * target_sum / h_sum
        
        if _ > 500 and no_improvement > 100:
            break
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        h = h * target_sum / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0)] * n_points
