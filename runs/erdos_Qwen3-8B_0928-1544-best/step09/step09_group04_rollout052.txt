import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values.copy()
    
    # Initialize population
    population_size = 200
    population = [initial_h.copy() * target_sum / np.sum(initial_h) + np.random.normal(0, 0.15, n_points) for _ in range(population_size)]
    population = [np.clip(h, 0, 1) * target_sum / np.sum(np.clip(h, 0, 1)) for h in population]
    
    best_h = population[0]
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = np.zeros(population_size)
        for i, h in enumerate(population):
            corr = np.correlate(h, 1 - h, mode='full')
            fitness[i] = np.max(corr) * dx
        # Select parents
        indices = np.argsort(fitness)
        selected = indices[:int(population_size * 0.5)]
        # Generate new population
        new_population = []
        for i in range(0, len(selected), 2):
            p1, p2 = population[selected[i]], population[selected[i+1]]
            # Crossover
            cp = np.random.randint(1, n_points-1)
            c1 = np.concatenate([p1[:cp], p2[cp:]])
            c2 = np.concatenate([p2[:cp], p1[cp:]])
            # Mutation
            c1 += np.random.normal(0, 0.1, n_points)
            c2 += np.random.normal(0, 0.1, n_points)
            # Normalize
            c1 = np.clip(c1, 0, 1) * target_sum / np.sum(np.clip(c1, 0, 1))
            c2 = np.clip(c2, 0, 1) * target_sum / np.sum(np.clip(c2, 0, 1))
            new_population.extend([c1, c2])
        # Update population
        population = new_population[:population_size]
        # Update best
        for h in population:
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_h, best_c5 = h, c5
    return best_h, best_c5, n_points
