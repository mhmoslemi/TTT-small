import time

def run(seed=42, budget_s=1100, **kwargs):

        if _ < 2:
            h = initial_h.copy()
        else:
            h = np.random.rand(n_points)
        h = h * (ones_count / np.sum(h))
        population.append(h)
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    fitness = [compute_C5(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_c5 = fitness[best_idx]
    
    tournament_size = 5
    mutation_rate = 0.2
    crossover_rate = 0.8
    max_generations = 200
    stagnation_threshold = 20
    stagnation_count = 0
    
    start_time = time.time()
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for _ in range(population_size):
            # Tournament selection
            tournament = np.random.choice(population_size, tournament_size)
            selected = tournament[np.argmin([fitness[i] for i in tournament])]
            parent1 = population[selected]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                parent2 = population[np.random.choice(population_size)]
                crossover_point = np.random.randint(n_points)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:

            # Clip and normalize

        # Evaluate new population
        new_fitness = [compute_C5(h) for h in new_population]
