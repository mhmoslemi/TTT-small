    a_initial = to_a(initial_h)
    
    num_particles = 30
    particles = [np.random.rand(n_points) * 0.5 - 0.5 for _ in range(num_particles)]
    particles = [to_a(p) for p in particles]
    velocities = [np.random.rand(n_points) * 0.1 for _ in range(num_particles)]
    
    def compute_c5(a):
        h = 0.5 + a
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    personal_best = [p.copy() for p in particles]
    global_best = particles[0].copy()
    best_c5 = compute_c5(global_best)
    
    omega = 0.7
    c1 = 2.0
    c2 = 2.0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = omega * velocities[i] + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            particles[i] += velocities[i]
            
            a = np.clip(particles[i], -0.5, 0.5)
            s = np.sum(a)
            if abs(s) > 1e-8:
                a[-1] -= s
            particles[i] = a
            
            current_c5 = compute_c5(particles[i])
            if current_c5 < compute_c5(personal_best[i]):
                personal_best[i] = particles[i].copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    global_best = particles[i].copy()
    
    h = 0.5 + global_best
    return h, best_c5, n_points
