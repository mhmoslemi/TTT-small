def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    # Function to compute C₅
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Initialize population
    pop_size = 100
    pop = []
    # Generate initial population
    for _ in range(pop_size):
        h = initial_h.copy()
        # Add some random perturbation
        perturb = np.random.normal(0, 0.2, n_points)
        h = h + perturb
        h = np.clip(h, 0, 1)
        # Scale to meet sum constraint
        h_sum = np.sum(h)
        h = h * (target_sum / h_sum) if h_sum != 0 else h
        pop.append(h)
    
    # Function to evaluate brightness (obj value)
    brightness = [obj(h) for h in pop]
    
    best_h = initial_h.copy()
    best_c5 = obj(initial_h)
    start_time = time.time()
    
    # Parameters for FA
    alpha = 0.8
    beta = 1.0
    gamma = 0.5
    max_iter = 400
    decay_alpha = 0.99
    decay_beta = 0.99
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Sort the population by brightness (lowest first)
        sorted_indices = np.argsort(brightness)
        sorted_pop = [pop[i] for i in sorted_indices]
        sorted_brightness = [brightness[i] for i in sorted_indices]
        
        for i in range(pop_size):
            # Select a random brighter firefly
            # Pick a firefly j that is in the top half (brighter)
            j = np.random.choice(range(pop_size), size=1, replace=False)[0]
            if sorted_brightness[i] > sorted_brightness[j]:
                # Move towards j
                h_i = sorted_pop[i].copy()
                h_j = sorted_pop[j].copy()
                distance = np.linalg.norm(h_i - h_j)
                # Compute the movement
                delta = beta * (h_j - h_i) * np.exp(-gamma * distance) + alpha * np.random.normal(0, 0.1, n_points)
                new_h = h_i + delta
                # Clip and scale
                new_h = np.clip(new_h, 0, 1)
                new_h_sum = np.sum(new_h)
                new_h = new_h * (target_sum / new_h_sum) if new_h_sum != 0 else new_h
                # Update the population
                pop[i] = new_h
                brightness[i] = obj(new_h)
                # Update best solution
                if brightness[i] < best_c5:
                    best_c5 = brightness[i]
                    best_h = pop[i].copy()
        
        # Decay alpha and beta
        alpha *= decay_alpha
        beta *= decay_beta
    
    # After FA, perform local refinement with SLSQP
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        fun=obj,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5
    
    return final_h, final_c5, n_points
