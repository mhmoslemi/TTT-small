import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:

        h = h * (ones_count / np.sum(h))
        particles.append(h)
        velocities.append(np.zeros(n_points))
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    pbest = [np.copy(p) for p in particles]
    pbest_values = [compute_C5(p) for p in pbest]
    gbest = np.copy(particles[np.argmin(pbest_values)])
    gbest_value = np.min(pbest_values)
    
    omega = 0.8
    c1, c2 = 1.5, 1.5
    max_stagnation = 50
