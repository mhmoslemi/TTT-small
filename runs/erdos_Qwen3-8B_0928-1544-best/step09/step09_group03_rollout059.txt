            velocities[i] = np.clip(velocities[i], -0.5, 0.5)
            particles[i] += velocities[i]

        for i in range(50):
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            particles[i] = particles[i] * (ones_count / np.sum(particles[i]))

        for i in range(50):
            current_c5 = evaluate(particles[i])
            if current_c5 < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_c5
            if current_c5 < gbest_value:
                gbest = particles[i].copy()
                gbest_value = current_c5
                best_h = gbest
                best_c5 = gbest_value

    return best_h, best_c5, n_points
