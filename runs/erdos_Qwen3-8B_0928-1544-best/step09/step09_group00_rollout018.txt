import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        h_initial = np.full(n_points, 0.5)
    a_initial = h_initial - 0.5

    def evaluate_c5(a):
        h = 0.5 + a
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    pop_size = 20
    omega = 0.7
    c1 = 2.0
    c2 = 2.0
    velocities = np.random.rand(pop_size, n_points) * 0.1
    population = [a_initial.copy() for _ in range(pop_size)]
    pbest = [a.copy() for a in population]
    pbest_values = [evaluate_c5(a) for a in population]
    gbest = population[0].copy()
    gbest_value = pbest_values[0]
    best_a = gbest.copy()
    best_c5 = gbest_value

    start_time = time.time()

    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + c1 * r1 * (pbest[i] - population[i]) + c2 * r2 * (gbest - population[i])
            population[i] = population[i] + velocities[i]
            population[i] = np.clip(population[i], -0.5, 0.5)
            sum_a = np.sum(population[i])
            if sum_a != 0:
                population[i] = population[i] - sum_a / n_points
            current_c5 = evaluate_c5(population[i])
            if current_c5 < pbest_values[i]:
                pbest[i] = population[i].copy()
                pbest_values[i] = current_c5
        gbest_idx = np.argmin(pbest_values)
        if pbest_values[gbest_idx] < gbest_value:
            gbest = pbest[gbest_idx].copy()
            gbest_value = pbest_values[gbest_idx]
        if (time.time() - start_time) < budget_s * 0.8:
            def objective(a):
                return evaluate_c5(a)
            bounds = [(-0.5, 0.5) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': lambda a: np.sum(a)})
            result = minimize(objective, gbest, method='SLSQP', bounds=bounds, constraints=cons)
            if result.success:
                gbest = result.x
                gbest_value = result.fun
        if gbest_value < best_c5:
            best_c5 = gbest_value
            best_a = gbest.copy()
    best_h = 0.5 + best_a
    return best_h, best_c5, n_points
