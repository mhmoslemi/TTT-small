import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    n_particles = 30
    particles = np.zeros((n_particles, n_points))
    particles[0] = initial_h.copy()
    particles[0] = particles[0] * (ones_count / np.sum(particles[0]))
    for i in range(1, n_particles):
        h = np.random.rand(n_points)
        h = h * (ones_count / np.sum(h))
        particles[i] = h
    
    velocities = np.zeros((n_particles, n_points))
    velocities = velocities * 0.1
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    pbest = particles.copy()
    pbest_values = np.array([evaluate(p) for p in particles])
    gbest = particles[np.argmin(pbest_values)]
    gbest_value = np.min(pbest_values)
    best_h = gbest
    best_c5 = gbest_value
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = 0.7 * velocities[i] + 2.0 * r1 * (pbest[i] - particles[i]) + 2.0 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -1.0, 1.0)
            particles[i] += velocities[i]
        
        for i in range(n_particles):
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            particles[i] = particles[i] * (ones_count / np.sum(particles[i]))

            current_c5 = evaluate(particles[i])

                pbest_values[i] = current_c5
