import time
from cvxopt import matrix, solvers

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    a = initial_h.copy() - 0.5
    a = a - np.mean(a)  # Ensure sum(a) == 0
    left_block = a[:35]
    right_block = a[35:]
    best_a = a.copy()
    best_c5 = compute_c5(a)
    start_time = time.time()
    
    def compute_c5(a):
        h = 0.5 + a
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def build_lp(current_block, fixed_block, m, n_points):
        s = -np.sum(fixed_block)
        num_constraints = n_points
        G = np.zeros((num_constraints, m + 1))
        h = np.zeros(num_constraints)
        for k in range(n_points):
            c_k = np.zeros(m)
            d_k = 0.0
            for j in range(m):
                idx = (j + k) % n_points
                val = 0.5 - fixed_block[idx]
                c_k[j] = val
                d_k += 0.5 * val
            G[k, :m] = c_k
            G[k, m] = -1.0
            h[k] = -d_k
        A = np.zeros((1, m + 1))
        A[0, :m] = 1.0
        b = np.array([s])
        lb = np.full(m + 1, -np.inf)
        ub = np.full(m + 1, np.inf)
        for i in range(m):
            lb[i] = -0.5
            ub[i] = 0.5
        return G, h, A, b, lb, ub
    
    def solve_lp(G, h, A, b, lb, ub):
        try:
            c = np.zeros(G.shape[1])
            c[-1] = 1.0
            sol = solvers.lp(matrix(c), matrix(G), matrix(h), matrix(A), matrix(b), matrix(lb), matrix(ub))
            x = np.array(sol['x'])
            return x
        except:
            return np.zeros(G.shape[1])
    
    iteration = 0
    while time.time() - start_time < budget_s:
        # Optimize left block
        G, h, A, b, lb, ub = build_lp(left_block, right_block, 35, n_points)
        x = solve_lp(G, h, A, b, lb, ub)
        if x.shape[0] > 0:
            optimized = x[:35]
            optimized = np.clip(optimized, -0.5, 0.5)
            left_block[:] = optimized
        # Optimize right block
        G, h, A, b, lb, ub = build_lp(right_block, left_block, 35, n_points)
        x = solve_lp(G, h, A, b, lb, ub)

            right_block[:] = optimized
        # Update best
        current_c5 = compute_c5(a)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_a = a.copy()
        iteration += 1
        if iteration > 1000:
            break
    return best_a, best_c5, n_points
