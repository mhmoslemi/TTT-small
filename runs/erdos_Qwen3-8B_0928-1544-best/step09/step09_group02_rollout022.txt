            if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / T):
                h = h_new.copy()
                current_C5 = new_C5
            
            if current_C5 < best_C5:
                best_h = h.copy()
                best_C5 = current_C5
        
        T *= alpha
    
    return best_h, best_C5, n_points
