from scipy.optimize import minimize
import time

        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points

        h_sum = np.sum(h)
        h = h * target_sum / h_sum
        pop.append(h)
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
