    if 'initial_h_values' in globals():
        particles[0] = initial_h.copy()
        particles[0] = particles[0] * (ones_count / np.sum(particles[0]))
    
    velocities = np.random.rand(n_particles, n_points) * 0.05

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    pbest = particles.copy()
    pbest_values = np.array([evaluate(p) for p in particles])
    gbest = particles[np.argmin(pbest_values)]
    gbest_value = np.min(pbest_values)
    best_h = gbest
    best_c5 = gbest_value
    
    start_time = time.time()
    
    iterations = 0
    while time.time() - start_time < budget_s:

            velocities[i] = 0.7 * velocities[i] + 2.0 * r1 * (pbest[i] - particles[i]) + 2.0 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -1.0, 1.0)
            particles[i] += velocities[i]
