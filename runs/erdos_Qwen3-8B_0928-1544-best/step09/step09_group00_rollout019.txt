import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum_a = 1 - 0.5 * n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        a = initial_h - 0.5
    else:
        a = np.random.uniform(-0.5, 0.5, n_points)
        current_sum = np.sum(a)
        delta = target_sum_a - current_sum
        a[-1] += delta
        a[-1] = np.clip(a[-1], -0.5, 0.5)

    def clip_and_adjust(a):
        clipped = np.clip(a, -0.5, 0.5)
        current_sum = np.sum(clipped)
        delta = target_sum_a - current_sum
        if np.abs(delta) > 1e-6:
            clipped[0] += delta
            clipped[0] = np.clip(clipped[0], -0.5, 0.5)
        return clipped

    a = clip_and_adjust(a)

    def objective(a_flat):
        a = a_flat.reshape(n_points)
        a = clip_and_adjust(a)
        h = 0.5 + a
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    # PSO parameters
    pop_size = 20
    population = np.random.uniform(-0.5, 0.5, (pop_size, n_points))
    for i in range(pop_size):
        population[i] = clip_and_adjust(population[i])
    
    best_a = population[0].copy()
    best_c5 = objective(best_a)
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time >= budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + np.random.rand(n_points) * (population[b] - population[c])
            trial = np.copy(mutant)
            for j in range(n_points):
                if np.random.rand() < 0.9:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = clip_and_adjust(trial)
            current_c5 = objective(trial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_a = trial.copy()
            if current_c5 < objective(population[i]):
                population[i] = trial.copy()

    # Local refinement with convex optimization
    def conv_expt(a_flat):
        a = a_flat.reshape(n_points)
        a = clip_and_adjust(a)
        h = 0.5 + a
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    res = minimize(conv_expt, best_a, method='L-BFGS-B', bounds=[(-0.5, 0.5)]*n_points)
    best_a = res.x
    best_a = clip_and_adjust(best_a)
    best_c5 = conv_expt(best_a)

    h_values = 0.5 + best_a
    return h_values, best_c5, n_points
