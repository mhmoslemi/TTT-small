    best_c5 = evaluate(h)
    start_time = time.time()
    
    max_iter = 200
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_perturbed = h.copy()
            h_perturbed[i] += eps
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            grad[i] = (evaluate(h_perturbed) - evaluate(h)) / eps
        
        bounds = [(-h[i], 1 - h[i]) for i in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x)}]
        
        def obj(x):
            return 0.5 * np.dot(x, x) + np.dot(grad, x)
        
        try:
            res = minimize(obj, np.zeros(n_points), method='SLSQP', bounds=bounds, constraints=cons)
            if res.success:
                delta = res.x
            else:
                delta = -grad * 0.01
        except:
            delta = -grad * 0.01
        
        h_new = h + delta
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        current_c5 = evaluate(h_new)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5, n_points
