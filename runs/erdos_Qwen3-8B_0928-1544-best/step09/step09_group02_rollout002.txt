import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    population_size = 50
    population = []
    for _ in range(population_size):
        if np.random.rand() < 0.2:
            h = initial_h.copy()
            perturb = np.random.normal(0, 0.1, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            h = h * (ones_count / total)
            population.append(h)
        else:
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            h = h * (ones_count / total)
            population.append(h)
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0]
    best_c5 = compute_C5(best_h)
    
    tournament_size = 5
    crossover_rate = 0.8
    mutation_rate = 0.2
    generations = 200
    max_stagnation = 50
    stagnation_count = 0
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        fitness = np.array([compute_C5(h) for h in population])
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx]
        
        new_population = []
        for _ in range(population_size):
            tournament = np.random.choice(range(population_size), tournament_size, replace=False)
            tournament_fitness = fitness[tournament]
            winner_idx = tournament[np.argmin(tournament_fitness)]
            parent1 = population[winner_idx]
            
            tournament = np.random.choice(range(population_size), tournament_size, replace=False)
            tournament_fitness = fitness[tournament]
            winner_idx = tournament[np.argmin(tournament_fitness)]
            parent2 = population[winner_idx]
            
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:split], parent2[split:]])
                total = np.sum(child)
                child = child * (ones_count / total)
                child = np.clip(child, 0, 1)
            else:
                child = parent1.copy()
                total = np.sum(child)
                child = child * (ones_count / total)
                child = np.clip(child, 0, 1)
            
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                total = np.sum(child)
                child = child * (ones_count / total)
            
            new_population.append(child)
        
        population = new_population
        
        if best_c5 == np.min(fitness):
            stagnation_count += 1

        else:
            stagnation_count = 0
    
    return best_h, best_c5, n_points
