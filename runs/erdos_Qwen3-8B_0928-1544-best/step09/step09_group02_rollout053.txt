        if time.time() - start_time > budget_s:
            break

        current_C5 = max_corr * dx
        
        idx = np.argmax(corr)
        k_star = idx - (n_points - 1)
        
        j_plus_k = (np.arange(n_points) + k_star) % n_points
        j_minus_k = (np.arange(n_points) - k_star) % n_points
        grad = (1 - h[j_plus_k] - h[j_minus_k]) * dx

        backtracks = 0

            clipped = np.clip(h_candidate, 0.0, 1.0)
            sum_clipped = np.sum(clipped)

            corr_new = np.correlate(scaled, 1 - scaled, mode='full')
