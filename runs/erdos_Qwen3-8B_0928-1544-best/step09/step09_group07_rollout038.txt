    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    max_pop_size = 200
    F = 0.8
    CR = 0.85
    max_generations = 1000
    pop = []
    for _ in range(max_pop_size):
        h = initial_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        h = h * target_sum / h_sum
        pop.append(h)
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 100:
            break
        for i in range(max_pop_size):
            a, b, c = np.random.choice(max_pop_size, 3, replace=False)
            mutant = pop[a] + F * (pop[a] - pop[b])
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial_sum = np.sum(trial)
            if trial_sum != 0:
                trial = trial * (target_sum / trial_sum)
            trial_obj = objective(trial)
            if trial_obj < objective(pop[i]):
                pop[i] = trial
                if trial_obj < best_c5:
                    best_c5 = trial_obj
                    best_h = trial.copy()
    
    res = minimize(
        fun=objective,
        x0=best_h,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        method='SLSQP',
        options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
    )
    final_h = res.x if res.success else best_h
    final_c5 = res.fun if res.success else best_c5
    
    return final_h, final_c5, n_points
