    max_gen = 600
    F, CR = 0.85, 0.95
    pop = [np.clip(initial_h + np.random.normal(0, 0.1, n_points), 0, 1) * (target_sum / np.sum(np.clip(initial_h + np.random.normal(0, 0.1, n_points), 0, 1))) for _ in range(pop_size)]
    
    start_time = time.time()
    
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + F * (pop[a] - pop[b])
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            trial = np.clip(trial * (target_sum / trial_sum) if trial_sum != 0 else trial, 0, 1)
            trial_c5 = np.max(np.correlate(trial, 1 - trial, mode='full')) * dx
            if trial_c5 < np.max(np.correlate(pop[i], 1 - pop[i], mode='full')) * dx:
                pop[i] = trial.copy()
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        fun=lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * dx,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5
    
    return final_h, final_c5, n_points
