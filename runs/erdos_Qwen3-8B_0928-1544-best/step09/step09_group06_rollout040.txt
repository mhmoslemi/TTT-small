import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    # Initial population with genetic algorithm
    pop_size = 20
    max_gen = 400
    F = 0.8
    CR = 0.9
    pop = [initial_h + np.random.normal(0, 0.05, n_points) for _ in range(pop_size)]
    
    for i in range(pop_size):
        h = pop[i]
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        pop[i] = h * target_sum / h_sum
    
    best_h = initial_h.copy()
    best_c5 = get_overlap(best_h, dx)
    start_time = time.time()
    
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + F * (pop[a] - pop[b])
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0.0, 1.0)
            trial_sum = np.sum(trial)
            trial = trial * target_sum / trial_sum
            trial_c5 = get_overlap(trial, dx)
            if trial_c5 < best_c5:
                best_c5 = trial_c5
                best_h = trial.copy()
            if trial_c5 < get_overlap(pop[i], dx):
                pop[i] = trial.copy()
    
    # refiner
    def objective(h):
        return get_overlap(h, dx)
    
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-9})
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5
    
    return final_h, final_c5, n_points

def get_overlap(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr
