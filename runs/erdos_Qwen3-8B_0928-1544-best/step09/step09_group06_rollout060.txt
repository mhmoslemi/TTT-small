import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_particles = 100
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    particles = np.random.rand(n_particles, n_points)
    velocities = np.random.rand(n_particles, n_points)
    
    # Clip and scale initial population
    for i in range(n_particles):
        particles[i] = np.clip(particles[i], 0.0, 1.0)
        particles[i] = particles[i] * target_sum / np.sum(particles[i])
    
    personal_best = particles.copy()
    global_best = particles[0].copy()
    best_c5 = np.inf
    
    max_iterations = 600
    inertia_start, inertia_end = 0.9, 0.4
    cognitive, social = 2.0, 2.0
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Fitness evaluation
        for i in range(n_particles):
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            current_c5 = np.max(corr) * dx
            
            # Update personal best
            pb_corr = np.correlate(personal_best[i], 1 - personal_best[i], mode='full')
            pb_c5 = np.max(pb_corr) * dx
            if current_c5 < pb_c5:
                personal_best[i] = particles[i].copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    global_best = particles[i].copy()
        
        # Velocity and position update
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            inertia = inertia_start - (iteration / max_iterations) * (inertia_start - inertia_end)
            velocities[i] = velocities[i] * inertia + cognitive * r1 * (personal_best[i] - particles[i]) + social * r2 * (global_best - particles[i])
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
    
    # SLSQP refinement
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        fun=objective, 
        x0=global_best, 
        method='SLSQP', 
        bounds=bounds, 
        constraints={'type': 'eq', 'fun': constraint}, 
        options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-9}
    )
    final_h = result.x if result.success else global_best
    final_c5 = result.fun if result.success else best_c5
    
    return final_h, final_c5, n_points
