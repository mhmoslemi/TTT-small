    current_h = np.clip(current_h, 0, 1)

    best_h = current_h.copy()
    best_c5 = evaluate_erdos_solution(current_h)
    tabu_set = set()
    tabu_queue = deque(maxlen=50)
    np.random.seed(seed)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        perturbed_h = current_h.copy()
        for _ in range(3):
            idx = np.random.randint(n_points)
            perturbation = np.random.uniform(-0.05, 0.05)
            perturbed_h[idx] += perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        total = np.sum(perturbed_h)
        scaling_factor = (n_points / 2) / total
        perturbed_h *= scaling_factor

        rounded = tuple(np.round(perturbed_h, 5))
        if rounded in tabu_set:
            continue

        current_c5 = evaluate_erdos_solution(perturbed_h)
        if current_c5 < best_c5:
            best_h = perturbed_h.copy()
            best_c5 = current_c5
        tabu_set.add(rounded)
        tabu_queue.append(rounded)
        current_h = perturbed_h.copy()

    return (best_h, best_c5, n_points)
