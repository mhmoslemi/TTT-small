import numpy as np
from scipy.optimize import minimize
import scipy

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 128
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

    else:
        h_initial = np.full(n_points, 0.5)

    if np.sum(h_initial) != n_points / 2:

    best_c5 = obj_func(best_h)
