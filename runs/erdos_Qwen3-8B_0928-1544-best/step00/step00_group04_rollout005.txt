        start_time = time_used
        # Compute cross-correlation
        shifted = np.roll(h, 1)
        corr = correlate(h, 1 - shifted, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
        # Check for early stopping
        if c5 < 0.380876:
            break
        # Find maximum lag
        max_loc = np.argmax(corr)
        k = max_loc - n_points + 1  # Adjust for full mode
        if k < 0:
            k += n_points
        # Ensure k is within [0, n_points)
        k = k % n_points
        # Formulate optimization problem
        h_vars = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(h_vars[i] * (1 - h_vars[(i + k) % n_points]) for i in range(n_points)) * dx)
        constraints = [0 <= h_vars, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=100)
            if prob.status == 'optimal':
                h = h_vars.value
        except:
            pass
        time_used = time_used + (time_used - start_time) * 1.5
        if time_used >= budget_s:
            break
    return h, best_c5, n_points
