        return np.max(cor)
    
    cons = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - n_points / 2,
        'jac': lambda h: np.ones_like(h)
    }]
    
    bnds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bnds,
        constraints=cons,
        options={'maxiter': 500, 'disp': False}
    )
    
    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)
    else:
        best_h = h_initial
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
