import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in kwargs:
        h = np.array(kwargs['initial_h_values'], dtype=float)
        n_points = len(h)

        n_points = 95
        h = np.random.rand(n_points)

    dx = 2.0 / n_points
    max_iter = int(budget_s / 50)  # ~20 iterations per second
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)

    start = time.time()
    for _ in range(max_iter):
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(cross_corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        max_idx = np.argmax(cross_corr)
        lag = max_idx - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if 0 <= i + lag < n_points:
                grad[i] += 1 - h[i + lag]
            if 0 <= i - lag < n_points:
