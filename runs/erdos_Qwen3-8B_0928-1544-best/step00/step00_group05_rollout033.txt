            j_minus_s = j[:, np.newaxis] - s[np.newaxis, :]
            valid_part2 = (j_minus_s >= 0) & (j_minus_s < n_points)
            part2 = np.where(valid_part2, -h[j_minus_s.astype(int)], 0)
            
            gradient_part = part1 + part2
            grad = np.sum(gradient_part * exp_corr[np.newaxis, :], axis=1) * beta / S
        
        # Update h
        h = h - eta * grad
        
        # Project h to [0, 1] and ensure integral constraint
        clipped = np.clip(h, 0, 1)
        total_clipped = np.sum(clipped)
        if total_clipped > 0:
            scale = (n_points / 2) / total_clipped
            h = clipped * scale
        
        # Evaluate and update best
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return best_h, best_c5, n_points
