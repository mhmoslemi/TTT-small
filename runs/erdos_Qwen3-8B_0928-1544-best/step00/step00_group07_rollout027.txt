        if total == 0:
            h.fill(0.5)
        else:
            h = h * target_sum / total
        particles.append(h)
        velocities.append(np.random.uniform(-0.2, 0.2, n_points))
        pbest.append(h.copy())
        pbest_cost.append(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    gbest = initial_h.copy()
    gbest_cost = pbest_cost[0]
    start_time = time.time()

    for iter in range(1000):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            w = 0.7 - 0.3 * (iter / 1000)
            c1, c2 = 2.0, 2.0
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            v = velocities[i]
            h = particles[i]
            pbest_h = pbest[i]
            gbest_h = gbest
            new_v = w * v + c1 * r1 * (pbest_h - h) + c2 * r2 * (gbest_h - h)
            new_v = np.clip(new_v, -0.2, 0.2)
            new_h = h + new_v
            new_h = np.clip(new_h, 0.0, 1.0)
            total_new = np.sum(new_h)
            if total_new == 0:
                new_h.fill(0.5)
            else:
                new_h = new_h * target_sum / total_new
            particles[i] = new_h
            velocities[i] = new_v
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            current_cost = np.max(corr) * dx
            if current_cost < pbest_cost[i]:
                pbest[i] = particles[i].copy()
                pbest_cost[i] = current_cost
        best_idx = np.argmin(pbest_cost)
        if pbest_cost[best_idx] < gbest_cost:
            gbest = pbest[best_idx].copy()
            gbest_cost = pbest_cost[best_idx]

    assert np.all(gbest >= 0) and np.all(gbest <= 1)
    assert np.isclose(np.sum(gbest), target_sum)
    return (gbest, gbest_cost, n_points)
