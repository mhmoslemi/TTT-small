import numpy as np

import time

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    target_sum = n_points / 2
    initial_sum = np.sum(initial_h)
    if abs(initial_sum - target_sum) > 1e-6:
        initial_h = np.full(n_points, 0.5)
    h = np.array(initial_h, dtype=float)
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    max_iter = 1000
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        c5_val = evaluate(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        shifted = (np.arange(n_points) + lag) % n_points
        
        x = cp.Variable(n_points)
        objective = cp.Maximize(cp.sum(x[i] * x[shifted[i]] for i in range(n_points)))
        constraints = [0 <= x, x <= 1, cp.sum(x) == target_sum]
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve()
        except:
            break
        
        if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INFEASIBLE]:
            break
        
        h_new = np.array(x.value)
        h = (1 - 0.1) * h + 0.1 * h_new
        h = np.clip(h, 0, 1)
        h = h - (np.sum(h) - target_sum) / n_points
    
    final_c5 = evaluate(h)
    return (h, final_c5, n_points)
