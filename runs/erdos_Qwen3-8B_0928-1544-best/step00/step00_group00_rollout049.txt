    target_sum = n_points / 2
    h = h / np.sum(h) * target_sum

    def compute_c5(h):
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        return cross_corr.max()

    def objective(h):
        return compute_c5(h)

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        tol=1e-6,
        options={'maxiter': 200, 'ftol': 1e-6, 'gtol': 1e-6}
    )

    best_h = result.x
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
