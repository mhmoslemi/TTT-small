    for temp in np.linspace(T, 0.1, 100):
        for _ in range(100):
            idx = np.random.choice(n_points, 3, replace=False)
            h_new = h.copy()
            for i in idx:
                h_new[i] = 1 - h_new[i]
            # Enforce sum constraint
            sum_h = np.sum(h_new)
            if np.abs(sum_h - n_points / 2) > 1e-6:
                idx = np.random.choice(n_points, 1)
                h_new[idx] = 1 - h_new[idx]
            c5_new = evaluate_erdos_solution(h_new)
            if c5_new < best_c5 or np.random.rand() < np.exp((best_c5 - c5_new) / temp):
                best_c5 = c5_new
                h = h_new
    return h, best_c5, n_points
