        max_val = np.max(cross_corr)
        max_idx = np.argmax(cross_corr)
        k = max_idx - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            i_plus_k = i + k
            if 0 <= i_plus_k < n:

            i_minus_k = i - k
            if 0 <= i_minus_k < n:
                grad[i] -= h[i_minus_k]
        h_new = h - alpha * grad
