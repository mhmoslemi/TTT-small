    best_h = h.copy()
    
    for _ in range(max_iterations):
        c = np.correlate(h, 1 - h, mode='full') * dx
        max_val = np.max(c)
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()
        
        m = np.argmax(c)
        k = m - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx
        
        direction = -grad
        h_new = h + direction
        h_new = np.clip(h_new, 0, 1)
        sum_new = np.sum(h_new)
        scale = (n_points / 2) / sum_new if sum_new != 0 else 1.0
        h_new *= scale
        c_new = np.correlate(h_new, 1 - h_new, mode='full') * dx
        new_c5 = np.max(c_new)
        
        if new_c5 < max_val:
            h = h_new
        else:
            h = h + 0.5 * direction
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
            h *= scale
    
    return (best_h, best_c5, n_points)
