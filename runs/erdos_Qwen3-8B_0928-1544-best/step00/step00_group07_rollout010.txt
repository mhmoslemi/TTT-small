    velocities = [np.random.uniform(-0.2, 0.2, n_points) for _ in range(n_particles)]
    personal_best = particles.copy()
    global_best = initial_h.copy()
    best_c5 = evaluate_erdos_solution(initial_h)
    best_h = initial_h.copy()
    
    w = 0.7
    c1, c2 = 2.0, 2.0
    max_iter = 1000
    start_time = time.time()
    
    while time.time() - start_time < budget_s and max_iter > 0:
        for i in range(n_particles):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            vel = velocities[i] * w + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            vel = np.clip(vel, -0.2, 0.2)
            new_h = particles[i] + vel
            new_h = np.clip(new_h, 0, 1)
            sum_h = np.sum(new_h)
            
            if sum_h == 0:
                new_h = np.random.rand(n_points)
                new_h = np.clip(new_h, 0, 1)
                sum_h = np.sum(new_h)
                scaling = target_sum / sum_h
                new_h = np.clip(new_h * scaling, 0, 1)
            else:
                scaling = target_sum / sum_h
                new_h = np.clip(new_h * scaling, 0, 1)
            
            new_h = np.clip(new_h, 0, 1)
            sum_h = np.sum(new_h)
            if abs(sum_h - target_sum) > 1e-6:
                diff = target_sum - sum_h
                delta = diff / (n_points)
                new_h = np.clip(new_h + delta, 0, 1)
            
            particles[i] = new_h
            current_c5 = evaluate_erdos_solution(particles[i])
            
            if current_c5 < evaluate_erdos_solution(personal_best[i]):
                personal_best[i] = particles[i].copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = particles[i].copy()
                    global_best = best_h.copy()
        
        max_iter -= 1
    
    return (best_h, best_c5, n_points)
