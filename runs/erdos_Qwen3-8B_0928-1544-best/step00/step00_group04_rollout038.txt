    while time.time() - start_time < budget_s:
        fitness = [compute_c5(ind) for ind in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()

        tournament = np.random.choice(range(pop_size), size=tournament_size, replace=False)
        tournament_fitness = [fitness[i] for i in tournament]
        parent1_idx = tournament[np.argmin(tournament_fitness)]
        parent2_idx = np.random.choice(range(pop_size), size=1, replace=False)[0]
        while parent2_idx == parent1_idx:
            parent2_idx = np.random.choice(range(pop_size), size=1, replace=False)[0]

        parent1 = population[parent1_idx].copy()
        parent2 = population[parent2_idx].copy()

        mask = np.random.rand(n_points) < 0.5
        child1 = parent1 * mask + parent2 * (1 - mask)
        child2 = parent2 * mask + parent1 * (1 - mask)

        mut_mask1 = np.random.rand(n_points) < mutation_rate
        mut_mask2 = np.random.rand(n_points) < mutation_rate
        child1[mut_mask1] += np.random.normal(0, mutation_scale, np.sum(mut_mask1))
        child2[mut_mask2] += np.random.normal(0, mutation_scale, np.sum(mut_mask2))

        child1 = clip_and_norm(child1)
        child2 = clip_and_norm(child2)

        worst_idx = np.argmax(fitness)
        population[worst_idx] = child1
        if worst_idx + 1 < pop_size:
            population[worst_idx + 1] = child2
        else:
            population[worst_idx] = child2

    return (best_h, best_c5, n_points)
