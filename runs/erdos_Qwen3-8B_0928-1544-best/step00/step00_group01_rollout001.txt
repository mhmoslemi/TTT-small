import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h_arr = np.array(h)
        corr = np.correlate(h_arr, 1 - h_arr, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def constraint_func(h):
        return np.sum(h) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_func}]

    h_initial = initial_h.copy()
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)

    if result.success:
        h_values = np.clip(result.x, 0.0, 1.0)
    else:
        h_values = np.clip(initial_h, 0.0, 1.0)

    if result.success:
        c5_bound = result.fun
    else:
        h_arr = np.array(h_values)
        corr = np.correlate(h_arr, 1 - h_arr, mode='full')
        c5_bound = np.max(corr) * dx

    return (h_values, c5_bound, n_points)
