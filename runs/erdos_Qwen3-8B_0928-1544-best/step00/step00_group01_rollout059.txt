from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    best_h = initial_h_values.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Precompute some perturbations
    perturbations = [np.random.normal(0, 0.05, size=n_points) for _ in range(20)]

        current_sum = np.sum(h_perturb)
        if current_sum == 0:

        scale = (n_points / 2) / current_sum
        h_perturb = h_perturb * scale
        h_perturb = np.clip(h_perturb, 0, 1)
        
        res = minimize(obj, h_perturb, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}],
