import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    n_points = len(initial_h_values)
    h_initial = initial_h_values.copy()
    dx = 2.0 / n_points
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=constraint, tol=1e-6)
    h_opt = result.x
    
    corr_opt = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5_bound = np.max(corr_opt) * dx
    
    # Ensure h_opt is within [0,1] and sum is correct
    h_opt = np.clip(h_opt, 0, 1)
    sum_h = np.sum(h_opt)
    if abs(sum_h - n_points / 2) > 1e-6:
        h_opt = h_opt * (n_points / 2) / sum_h
    
    return (h_opt, c5_bound, n_points)
