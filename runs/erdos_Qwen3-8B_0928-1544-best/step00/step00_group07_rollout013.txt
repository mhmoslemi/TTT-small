import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 128
    # Use initial_h_values if available
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:

    def objective(h):

        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    # Constraints: sum(h) == n_points / 2
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2, 'jac': lambda h: np.ones(n_points)}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = optimize.minimize(
        fun=objective,
        x0=h_start,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'disp': False}
    )
    
    elapsed = time.time() - start_time
    # Use initial guess if optimization exceeds budget or fails
    if elapsed >= budget_s or not result.success:
        h_values = h_start
        c5_bound = objective(h_start)
    else:

        c5_bound = result.fun
    
    # Validate solution to ensure constraints

    if abs(sum_h - n_points / 2) > 1e-6:
        h_values = h_start
        c5_bound = objective(h_start)
    
    # Recompute C5 to ensure accuracy
    c5_bound = objective(h_values)
    
    return h_values, c5_bound, n_points
