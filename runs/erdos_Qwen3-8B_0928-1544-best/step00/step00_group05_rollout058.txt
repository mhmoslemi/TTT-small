    time_start = time.time()
    
    while time.time() - time_start < budget_s:
        h_perturbed = perturb_h(h, n_points)
        new_c5 = evaluate_erdos_solution(h_perturbed)
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0:
            h = h_perturbed
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            if random.random() < np.exp(-delta_c5 / T):
                h = h_perturbed
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        T *= alpha
    return (best_h, best_c5, n_points)
