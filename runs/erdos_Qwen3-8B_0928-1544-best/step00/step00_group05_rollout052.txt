        return np.sum(h) - n_points / 2

    # Bounds and constraints
    bounds = Bounds(0.0, 1.0)

    # Start with initial guess
    x0 = initial_h_values

    # Optimization settings
    result = minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=LinearConstraint(A=[[1.0]*n_points], b=[n_points / 2]),
        method='L-BFGS-B',
        options={'maxiter': 200, 'ftol': 1e-8}
    )

    # Check time to avoid exceeding budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        return (initial_h_values, 0.38085, n_points)

    h_values = result.x
    c5_bound = objective(h_values)

    return (h_values, c5_bound, n_points)
