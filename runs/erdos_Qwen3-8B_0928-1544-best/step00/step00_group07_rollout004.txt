import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    # Initialize with provided starting point or uniform distribution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = initial_h_values
        n_points = len(h)
    else:
        n_points = 128
        h = np.full(n_points, 0.5)
    
    # Objective function: maximize overlap integral (minimize C5)
    def objective(h_vals):
        dx = 2.0 / n_points
        corr = np.correlate(h_vals, 1 - h_vals, mode='full') * dx
        return np.max(corr)
    
    # Constraint: total integral equals 1
    def constraint(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint, 'jac': lambda x: np.ones_like(x)}]
    
    # Optimization settings
    result = minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 500, 'ftol': 1e-8, 'disp': False}
    )
    
    # Fallback to initial guess if optimization fails
    if not result.success:
        h_opt = h
        c5 = objective(h_opt)
    else:
        h_opt = result.x
        c5 = result.fun
    
    # Final check and return
    c5 = objective(h_opt)
    return (h_opt, c5, n_points)
