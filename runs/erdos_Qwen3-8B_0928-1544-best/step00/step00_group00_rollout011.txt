import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        n_points = len(h_initial)

        corr = np.correlate(h, 1 - h, mode='full')
