def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_coarse = 64
    n_fine = 128
    dx_fine = 2 / n_fine

    if 'initial_h_values' in globals():

            h /= np.sum(h) * n_coarse / 2
            h *= n_coarse / 2

        h_shifted = 1 - h
        corr = np.correlate(h, h_shifted, mode='full')
        return np.max(corr) * (2.0 / n)

    num_generations = 150
    pop_size = 50

    for _ in range(num_generations):
        fitness = [evaluate_c5(h, n_coarse) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]

        new_pop = []
        for _ in range(pop_size):
            idx1, idx2 = np.random.randint(0, pop_size, 2)
            p1, p2 = population[idx1], population[idx2]
            if fitness[idx1] < fitness[idx2]:
                parent = p1
            else:
                parent = p2

            cross = np.random.randint(1, n_coarse - 1)
            child = np.concatenate([parent[:cross], parent[cross:]])

            noise = np.random.normal(0, 0.1, n_coarse)

        if np.sum(h_fine) != 0:

    final_h = result.x
    c5_bound = evaluate_c5(final_h, n_fine)
    return final_h, c5_bound, n_fine
