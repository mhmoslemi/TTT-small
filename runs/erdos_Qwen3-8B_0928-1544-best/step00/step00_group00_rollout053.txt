                indices = np.argsort(particles[i])
                for j in indices:
                    if diff <= 0:
                        break
                    particles[i][j] += diff
                    diff = 0
            current_val = evaluate(particles[i])
            if current_val < personal_best_val[i]:
                personal_best[i] = particles[i].copy()
                personal_best_val[i] = current_val
            if current_val < global_best_val:
                global_best = particles[i].copy()
                global_best_val = current_val

    return (global_best.tolist(), global_best_val, n_points)
