import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    n_points = 45
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def project(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum < target_sum:
            deficit = target_sum - current_sum
            indices = np.where(h < 1)[0]
            if indices.size > 0:
                idx_sorted = np.argsort(h[indices])
                for i in indices[idx_sorted]:
                    if deficit <= 0:
                        break
                    add = np.clip(deficit, 0, 1 - h[i])
                    h[i] += add
                    deficit -= add
        elif current_sum > target_sum:
            excess = current_sum - target_sum
            indices = np.where(h > 0)[0]
            if indices.size > 0:
                idx_sorted = np.argsort(h[indices])[::-1]
                for i in indices[idx_sorted]:
                    if excess <= 0:
                        break
                    sub = np.clip(excess, 0, h[i])
                    h[i] -= sub
                    excess -= sub
        return h

    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_initial = initial_h_values.copy()
        h_initial = project(h_initial)
    else:
        h_initial = np.random.rand(n_points)
        h_initial /= np.sum(h_initial) * target_sum

    N_particles = 100
    particles = [h_initial.copy() for _ in range(N_particles // 2)]
    for _ in range(N_particles // 2):
        h = np.random.rand(n_points)
        h /= np.sum(h) * target_sum
        particles.append(h)
    velocities = [np.random.rand(n_points) - 0.5 for _ in range(N_particles)]

    personal_best = [p.copy() for p in particles]
    personal_best_values = [evaluate_erdos_solution(p) for p in particles]
    global_best = min(personal_best, key=evaluate_erdos_solution)
    global_best_value = evaluate_erdos_solution(global_best)

    inertia = 0.8
    cognitive = 1.2
    social = 1.2
    max_iterations = 1000
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:

        for i in range(N_particles):
            v = velocities[i] * inertia + cognitive * np.random.rand(n_points) * (personal_best[i] - particles[i]) + social * np.random.rand(n_points) * (global_best - particles[i])
            velocities[i] = v
            particles[i] += v
            particles[i] = project(particles[i])
        new_personal = []
        for i in range(N_particles):
            val = evaluate_erdos_solution(particles[i])

                global_best = particles[i].copy()

    return (best_h.tolist(), global_best_value, n_points)
