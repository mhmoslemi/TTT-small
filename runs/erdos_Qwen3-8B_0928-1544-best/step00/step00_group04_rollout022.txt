import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

        initial = None

    def compute_c5(h):
        dx = 2.0 / n_points
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr) * dx

    def normalize(h):
        s = np.sum(h)
        if s == 0:
            h[0] = 1.0
        else:
            h = h * (50.0 / s)
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            h[0] = 1.0
        else:
            h = h * (50.0 / s)
        return h

    pop_size = 20
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = normalize(h)
        population.append(h)
    if initial is not None and len(initial) == n_points:
        population.append(initial)

    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h) for h in population]
        sorted_idx = np.argsort(fitness)
        top_half = sorted_idx[:pop_size // 2]
        new_pop = []
        for i in range(pop_size // 2):
            p1, p2 = population[top_half[i % len(top_half)]], population[top_half[(i+1) % len(top_half)]]
            mask = np.random.rand(n_points) < 0.5
            c1 = p1 * mask + p2 * (1 - mask)

            for c in [c1, c2]:
                mutate = np.random.rand(n_points) < 0.1
                c[mutate] += np.random.normal(0, 0.01, np.sum(mutate))
