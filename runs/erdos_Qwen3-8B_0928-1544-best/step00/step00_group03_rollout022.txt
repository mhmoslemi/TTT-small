        if c5 < p['pbest_fitness']:
            p['pbest'] = p['h'].copy()
    gbest = swarm[0]['pbest'].copy()
    gbest_fitness = swarm[0]['pbest_fitness']
    for p in swarm:
        if p['pbest_fitness'] < gbest_fitness:
            gbest_fitness = p['pbest_fitness']
            gbest = p['pbest'].copy()
    
    start_time = time.time()
    max_iter = 100
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for p in swarm:
            r1, r2 = np.random.rand(), np.random.rand()
            w = 0.7
            p['vel'] = w * p['vel'] + 2 * r1 * (p['pbest'] - p['h']) + 2 * r2 * (gbest - p['h'])
            p['vel'] = np.clip(p['vel'], -0.5, 0.5)
            p['h'] += p['vel']
            p['h'] = np.clip(p['h'], 0, 1)
            sum_h = np.sum(p['h'])
            p['h'] = p['h'] * (n_points / 2) / sum_h
            c5 = evaluate_erdos_solution(p['h'])
            if c5 < p['pbest_fitness']:
                p['pbest_fitness'] = c5
                p['pbest'] = p['h'].copy()
            if c5 < gbest_fitness:
                gbest_fitness = c5
                gbest = p['h'].copy()
    
    best_h = gbest
    best_c5 = gbest_fitness
    return (best_h.tolist(), best_c5, n_points)
