def run(seed=42, budget_s=1100, **kwargs):

        n_points = len(h)
    else:
        n_points = 95
        h = np.full(n_points, 0.5)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_c5 = float('inf')
    best_h = h.copy()
    learning_rate = 0.01
    max_iter = 11000
    time_start = time.time()

    for _ in range(max_iter):
        # Compute cross-correlation
        c = np.correlate(h, 1 - h, mode='full')
        j_max = np.argmax(c)
        s = j_max - (n_points - 1)
        # Compute gradient
        g = np.zeros_like(h)
        for i in range(n_points):
            j = i + s
            if 0 <= j < n_points:
                g[i] = 1.0 - h[j]
        # Line search
        best_alpha = 0.0
        best_new_c5 = float('inf')
        for alpha in [0.01, 0.005, 0.001, 0.0005]:
            h_candidate = h - alpha * g
            h_candidate = np.clip(h_candidate, 0.0, 1.0)
            sum_candidate = np.sum(h_candidate)
            if sum_candidate == 0:
                continue
            scaling = target_sum / sum_candidate
            h_candidate *= scaling
            c_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            new_c5 = np.max(c_candidate) * dx
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_alpha = alpha
        # Update h
        h = h - best_alpha * g
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != 0:
            scaling = target_sum / sum_h
            h *= scaling
        # Evaluate current C5
        c = np.correlate(h, 1 - h, mode='full')

            best_h = h.copy()
        # Check time

            break
    return best_h, best_c5, n_points
