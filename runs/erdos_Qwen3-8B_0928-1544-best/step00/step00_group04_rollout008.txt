import numpy as np

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()

    for _ in range(max_iter):
        # Compute h from z
        h_current = 1 / (1 + np.exp(-z))
        # Check sum
        sum_h = np.sum(h_current)
        if abs(sum_h - sum_h_target) > 1e-6:
            # Adjust z to make sum correct by scaling
            scale = sum_h_target / sum_h
            z = z * scale
            h_current = 1 / (1 + np.exp(-z))

        # Compute overlap integrals
        correlate = np.correlate(h_current, 1 - h_current, mode='full')
        max_corr = np.max(correlate)

        k = k_max_idx - (n_points - 1)

        if k < -n_points + 1 or k >= n_points:
            break
