        population = [new_population[i] for i in sorted_indices[:pop_size]]

        current_fitnesses = evaluate(population)
        current_best_idx = np.argmin(current_fitnesses)
        current_best_c5 = current_fitnesses[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()

    return (best_h, best_c5, n_points)
