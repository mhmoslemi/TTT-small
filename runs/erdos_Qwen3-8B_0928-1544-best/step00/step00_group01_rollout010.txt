import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - sum_required
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    total_time = 0
    
    n_starts = 10

        h_perturbed = np.clip(initial_h_values + perturbation, 0, 1)
        
        start_time = time.time()
        result = opt.minimize(compute_c5, h_perturbed, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-5)
        elapsed_time = time.time() - start_time
        total_time += elapsed_time
        
        if result.success:
