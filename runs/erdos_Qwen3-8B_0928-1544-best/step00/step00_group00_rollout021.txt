            particles[i] = projected_h
            fitness = compute_c5(projected_h)
            if fitness < pbest_fitness[i]:
                pbest[i] = projected_h.copy()
                pbest_fitness[i] = fitness
        for i in range(swarm_size):
            if pbest_fitness[i] < global_best_fitness:
                global_best = pbest[i].copy()
                global_best_fitness = pbest_fitness[i]
    
    return (global_best, global_best_fitness, n_points)
