import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    initial_h_values = eval('initial_h_values') if 'initial_h_values' in dir() else None
    n_points = 47 if initial_h_values is None else len(initial_h_values)
