        pop = np.array([h] * pop_size)
    else:
        pop = np.random.rand(pop_size, n_points)
        for i in range(pop_size):
            pop[i] = pop[i] * (n_points / 2) / np.sum(pop[i])
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / n_points)
    
    def evaluate(solution):
        return compute_c5(solution)
    
    best_h = pop[0].copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    # Genetic Algorithm
    for gen in range(max_generations):
        # Tournament selection
        parents = []
        for _ in range(2):
            indices = np.random.choice(pop_size, tournament_size)
            winner = indices[np.argmin([evaluate(pop[i]) for i in indices])]
            parents.append(pop[winner].copy())
        
        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1]
            cross_idx = np.random.randint(1, n_points - 1)
            child = np.concatenate((p1[:cross_idx], p2[cross_idx:]))
            offspring.append(child)
        
        # Mutation
        for j in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child = offspring[j] + noise
                child = np.clip(child, 0, 1)
                total = np.sum(child)
                if total > 0:
                    child *= (n_points / 2) / total
                else:
                    child = np.random.rand(n_points) * (n_points / 2)
                offspring[j] = child
        
        # Evaluate and replace
        for o in offspring:
            c5 = evaluate(o)
            if c5 < best_c5:
                best_c5 = c5
                best_h = o.copy()
        
        # Replace population
        for i in range(len(offspring)):
            if np.random.rand() < 0.5:
                pop[i] = offspring[i]
        
        # Check time
        if time.time() - start_time > time_limit:
            break

    # Upscale to fine grid
    x_coarse = np.linspace(0, 2, n_points)
    x_fine = np.linspace(0, 2, fine_n_points)
    h_fine = np.interp(x_fine, x_coarse, best_h)
    h_fine = h_fine * (fine_n_points / 2) / np.sum(h_fine)
    h_fine = np.clip(h_fine, 0, 1)

    # Gradient refinement
    def objective(x):
        h = x.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / fine_n_points)
    
    result = minimize(objective, h_fine, method='L-BFGS-B',
                      bounds=[(0, 1)]*fine_n_points, tol=1e-6,
                      options={'maxiter': 100})
    best_h_fine = result.x
    final_c5 = result.fun
    
    # Final adjustment
    best_h_fine = np.clip(best_h_fine, 0, 1)
    total = np.sum(best_h_fine)
    if total > 0:
        best_h_fine *= (fine_n_points / 2) / total
    else:
        best_h_fine = np.random.rand(fine_n_points) * (fine_n_points / 2)
    
    return best_h_fine, final_c5, fine_n_points
