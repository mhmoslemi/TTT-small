    result = opt.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.8,
        tol=1e-5,
        maxiter=500,
        polish=True,
        seed=seed,
        disp=False
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        # Early termination if time is tight
        result = opt.minimize(
            objective,
            result.x,
            bounds=bounds,
            method='L-BFGS-B',
            tol=1e-5
        )

    h_opt = result.x
    c5_bound = objective(h_opt)
    return (h_opt, c5_bound, n_points)
