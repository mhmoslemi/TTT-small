import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        sum_h = np.sum(initial_h)
        if not np.isclose(sum_h * dx, 1.0):
            initial_h = np.full(n_points, 0.5)
        current_h = np.array(initial_h, dtype=float)
    else:
        current_h = np.full(n_points, 0.5)
    
    best_h = current_h.copy()
    best_c5 = evaluate_erdos_solution(current_h)
    
    start_time = time.time()
    tabu_list = []
    max_iter = int(budget_s / 10)
    
    def perturb(h):
        noise = np.random.normal(0, 0.1, n_points)
        new_h = h + noise
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new == 0:
            return new_h
        scale = (n_points / 2) / sum_new
        return new_h * scale
    
    for _ in range(max_iter):
        candidate = perturb(current_h)
        candidate_c5 = evaluate_erdos_solution(candidate)
        
        if (candidate_c5 < best_c5 and 
            all(not np.allclose(candidate, h, atol=1e-3) for h, _ in tabu_list)):
            current_h = candidate.copy()
            best_c5 = candidate_c5
            best_h = current_h.copy()
            if len(tabu_list) > 20:
                tabu_list.pop(0)
            tabu_list.append((candidate.copy(), candidate_c5))
        
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)
