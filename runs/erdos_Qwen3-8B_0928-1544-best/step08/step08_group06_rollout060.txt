            social = c2 * r2 * (best_solution - particles[i])
            velocities[i] = omega * velocities[i] + cognitive + social
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            particles[i] = particles[i] * (sum_target / np.sum(particles[i]))
            current_value = compute_c5(particles[i])
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
            if current_value < best_value:
                best_value = current_value
                best_solution = particles[i].copy()
    return best_solution, best_value, n_points

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
