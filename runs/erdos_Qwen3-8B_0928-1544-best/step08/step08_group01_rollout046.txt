    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * (ones_count / np.sum(h))
    h = np.clip(h, 0, 1)
    scale = (n_points / 2) / np.sum(h)
    h = h * scale
    population = [h.copy()]
    for _ in range(99):
        h = np.random.rand(n_points)
        h = h * (ones_count / np.sum(h))
        h = np.clip(h, 0, 1)
        scale = (n_points / 2) / np.sum(h)
        population.append(h * scale)
    
    generations = 50
    for _ in range(generations):
        evaluations = []
        for p in population:
            corr = np.correlate(p, 1 - p, mode='full')
            evaluations.append(np.max(corr) * 2 / n_points)
        argsort = np.argsort(evaluations)
        sorted_pop = [population[i] for i in argsort]
        top_half = sorted_pop[:len(sorted_pop)//2]
        offspring = []
        for i in range(0, len(top_half), 2):
            if i + 1 < len(top_half):
                p1, p2 = top_half[i], top_half[i+1]
                c1, c2 = p1.copy(), p2.copy()
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        c1[j], c2[j] = p2[j], p1[j]
                for j in range(n_points):
                    if np.random.rand() < 0.1:
                        if np.random.rand() < 0.33:
                            c1[j] = 0.0
                        elif np.random.rand() < 0.66:
                            c1[j] = 1.0
                        else:
                            c1[j] = np.random.rand()
                        if np.random.rand() < 0.33:
                            c2[j] = 0.0
                        elif np.random.rand() < 0.66:
                            c2[j] = 1.0
                        else:
                            c2[j] = np.random.rand()
                c1 = np.clip(c1, 0, 1)
                c1 = c1 * (n_points / 2) / np.sum(c1)
                c2 = np.clip(c2, 0, 1)
                c2 = c2 * (n_points / 2) / np.sum(c2)
                offspring.append(c1)
                offspring.append(c2)
        new_pop = sorted_pop[:len(sorted_pop)//2] + offspring
        population = new_pop
    
    evaluations = []
    for p in population:
        corr = np.correlate(p, 1 - p, mode='full')
        evaluations.append(np.max(corr) * 2 / n_points)
    argsort = np.argsort(evaluations)
    top_solutions = [population[i] for i in argsort[:10]]
    
    best_h = np.zeros(n_points)
    best_c5 = float('inf')
    for h_candidate in top_solutions:
        h = h_candidate.copy()
        def objective(x):
            h = x.copy()
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * 2 / n_points
        def constraint(x):
            return np.sum(x) - ones_count
        bounds = [(0., 1.)]*n_points
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints={'type':'eq', 'fun':constraint}, options={'maxiter':300})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    return best_h, best_c5, n_points
