    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    h = np.clip(h, 0, 1)
    if np.sum(h) != ones_count:
        h = h / np.sum(h) * ones_count
    best_h = h.copy()
    best_c5 = float('inf')
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    def compute_gradient(h):
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if np.abs(h_plus_sum - ones_count) > 1e-6:
                h_plus = h_plus / h_plus_sum * ones_count
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * (2.0 / n_points)
            grad[i] = (c5_plus - objective(h)) / eps
        return grad
    
    def frank_wolfe_step(h, grad):
        clamped = np.clip(grad, 0, 1)
        clamped_sum = np.sum(clamped)
        if np.abs(clamped_sum - ones_count) < 1e-6:
            p = clamped
        else:
            scale = ones_count / clamped_sum
            p = clamped * scale
            p = np.clip(p, 0, 1)
        p_sum = np.sum(p)
        if np.abs(p_sum - ones_count) < 1e-6:
            pass
        else:
            scale = ones_count / p_sum
            p = p * scale
            p = np.clip(p, 0, 1)
        return p
    
    start_time = time.time()
    iterations = 0
    while time.time() - start_time < budget_s:
        if iterations > 500:
            break
        iterations += 1
        grad = compute_gradient(h)
        p = frank_wolfe_step(h, grad)
        h_candidate = h + 0.3 * p
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate_sum = np.sum(h_candidate)
        if np.abs(h_candidate_sum - ones_count) > 1e-6:
            h_candidate = h_candidate / h_candidate_sum * ones_count
        c5_candidate = objective(h_candidate)
        if c5_candidate < objective(h):
            h = h_candidate
            best_c5 = c5_candidate
            best_h = h.copy()
        # Add perturbation to escape local minima
        if iterations % 10 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h_perturb = h + noise
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb_sum = np.sum(h_perturb)
            if np.abs(h_perturb_sum - ones_count) > 1e-6:
                h_perturb = h_perturb / h_perturb_sum * ones_count
            c5_perturb = objective(h_perturb)
            if c5_perturb < best_c5:
                best_h = h_perturb.copy()
                best_c5 = c5_perturb
    
    return best_h, best_c5, n_points
