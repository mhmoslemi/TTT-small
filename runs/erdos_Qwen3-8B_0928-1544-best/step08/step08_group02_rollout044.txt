from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 100  # Increased to 100 for better resolution
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    # Adaptive refinement with dynamic perturbation
    for _ in range(20):  # More iterations for better convergence
        if time.time() - start_time >= budget_s:
            break
        # Get current best
        current_c5 = objective(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        
        # Identify most detrimental shift
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        
        # Focus on regions contributing to max overlap
        target_mask = np.zeros(n_points, dtype=bool)
        for x in range(n_points - k):
            target_mask[x] = True

        perturbation = np.zeros(n_points)
        perturbation[target_mask] = np.random.normal(0, 0.03, sum(target_mask))
        perturbation = np.clip(perturbation, -0.2, 0.2)
        
        # Adjust to maintain sum constraint
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        norm = np.sum(h_perturbed)
        if norm > 0:
            h_perturbed = h_perturbed * (ones_count / norm)
        
        result = minimize(
            fun=objective,
