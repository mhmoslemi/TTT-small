import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Use initial_h as a seed for the random population
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    population = [initial_h + np.random.normal(0, 0.05, n_points) for _ in range(20)]
    population = [np.clip(p, 0, 1) for p in population]
    for i in range(len(population)):
        population[i] = population[i] * target_sum / np.sum(population[i])
    
    # Differential Evolution
    def de_objective(h):
        return objective(h)
    
    result_de = differential_evolution(
        de_objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        maxiter=200,
        tol=1e-6,
        mutation=(0.4, 0.7),
        recombination=0.8,
        seed=seed
    )
    best_h_de = result_de.x
    best_c5_de = result_de.fun
    
    # SLSQP Refinement
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    result_slsqp = minimize(
        fun=objective,
        x0=best_h_de,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    final_h = result_slsqp.x
    final_c5 = result_slsqp.fun
