import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    n_particles = 20
    particles = np.random.rand(n_particles, n_points)
    particles = np.clip(particles, 0.0, 1.0)
    particles = particles * (ones_count / np.sum(particles, axis=1, keepdims=True))
    if 'initial_h_values' in globals():
        particles[0] = initial_h.copy()
        particles[0] = particles[0] * (ones_count / np.sum(particles[0]))
    
    velocities = np.random.rand(n_particles, n_points) * 0.1

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    pbest = particles.copy()
    pbest_values = np.array([evaluate(p) for p in particles])
    gbest = particles[np.argmin(pbest_values)]
    gbest_value = np.min(pbest_values)
    best_h = gbest
    best_c5 = gbest_value
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = 0.7 * velocities[i] + 2.0 * r1 * (pbest[i] - particles[i]) + 2.0 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -1.0, 1.0)
            particles[i] += velocities[i]
        
        for i in range(n_particles):
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            particles[i] = particles[i] * (ones_count / np.sum(particles[i]))
        
        for i in range(n_particles):
            current_c5 = evaluate(particles[i])
            if current_c5 < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_c5
            if current_c5 < gbest_value:
                gbest = particles[i].copy()
                gbest_value = current_c5
                best_h = gbest
                best_c5 = gbest_value
    
    return best_h, best_c5, n_points
