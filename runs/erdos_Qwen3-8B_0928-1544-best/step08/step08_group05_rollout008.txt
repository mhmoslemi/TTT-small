import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    # DE configuration
    de_result = differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=200,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.9,
        maxiter=50,
        seed=seed,
        disp=False
    )
    
    best_de_h = de_result.x
    sum_de = np.sum(best_de_h)
    
    # Repair to meet sum constraint
    if sum_de == 0:
        best_de_h = np.full(n_points, 0.5)
    else:
        scale = ones_count / sum_de
        best_de_h = np.clip(best_de_h * scale, 0, 1)
    # Adjust final element if necessary
    if np.sum(best_de_h) != ones_count:
        excess = ones_count - np.sum(best_de_h)
        best_de_h[-1] = np.clip(best_de_h[-1] + excess, 0, 1)
    
    # Local optimization with SLSQP
    slsqp_result = minimize(
        fun=objective,
        x0=best_de_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    best_h = slsqp_result.x
    best_c5 = slsqp_result.fun
    
    return best_h, best_c5, n_points
