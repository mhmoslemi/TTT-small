        return np.sum(h_vec) - target_sum

    # Initial optimization with trust-constr for better convergence
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        fun=objective,
        x0=h,
        method='trust-constr',
        bounds=Bounds(bounds),
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={
            'maxiter': 300,
            'ftol': 1e-9,
            'gtol': 1e-9,
            'xscale': 'jac',
            'debug': False,
        }
    )

    best_h = result.x
    best_c5 = result.fun

    # Refine further with local descent to escape local minima
    def refine_step(h_vec):
        h_tmp = h_vec.copy()
        h_tmp = np.clip(h_tmp, 0.0, 1.0)
        h_tmp_sum = np.sum(h_tmp)
        h_tmp = h_tmp * (target_sum / h_tmp_sum) if h_tmp_sum != 0 else h_tmp
        return h_tmp

    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        # Perturb and optimize
        perturb = np.random.normal(0, 0.2, n_points)
        h_perturb = h_best + perturb
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        h_perturb = h_perturb * (target_sum / np.sum(h_perturb))
        res = minimize(
            fun=objective,
            x0=h_perturb,
            method='trust-constr',
            bounds=Bounds(bounds),
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={
                'maxiter': 100,
                'ftol': 1e-9,
                'gtol': 1e-9,
                'xscale': 'jac',
                'debug': False,
            }
        )
        if res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    return best_h, best_c5, n_points
