        initial_sparse[positions] = 1
        for _ in range(50):
            mutant = initial_sparse + np.random.normal(0, 0.2, n_points)
            mutant = np.clip(mutant, 0.0, 1.0)
            sum_mut = np.sum(mutant)
            if sum_mut > 0:
                mutant = mutant * (ones_count / sum_mut)
            population.append(mutant)

    best_h = population[0]
    best_c5 = compute_c5(best_h)

    sigma = 0.25
    last_decay_time = time.time()
    start_time = time.time()

    while time.time() - start_time < budget_s:
        current_c5s = [compute_c5(h) for h in population]
        sorted_indices = np.argsort(current_c5s)
        sorted_population = [population[i] for i in sorted_indices]
        top_parents = sorted_population[:25]

        children = []
        for parent in top_parents:
            for _ in range(2):
                noise = np.random.normal(0, sigma, n_points)
                child = parent + noise
                child = np.clip(child, 0.0, 1.0)
                sum_child = np.sum(child)
                if sum_child > 0:
                    child = child * (ones_count / sum_child)
                children.append(child)

        current_c5s_children = [compute_c5(h) for h in children]
        combined = sorted_population + children
        combined_sorted = sorted(combined, key=lambda x: compute_c5(x))
        new_population = combined_sorted[:50]

        population = new_population

        current_best_c5 = compute_c5(population[0])
        if current_best_c5 < best_c5:
            best_h = population[0]
            best_c5 = current_best_c5

        current_time = time.time()
        if current_time - last_decay_time >= 10:
            sigma *= 0.95
            last_decay_time = current_time

    return best_h, best_c5, n_points
