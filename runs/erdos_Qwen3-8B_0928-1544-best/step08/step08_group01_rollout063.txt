import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    start_time = time.time()
    gen = 0
    
    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        scale = ones_count / np.sum(h)
        return h * scale
    
    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                if np.random.rand() < 0.33:
                    h[i] = 0.0
                elif np.random.rand() < 0.34:
                    h[i] = 1.0
                else:
                    h[i] = np.random.rand()
        return clip_rescale(h)
    
    pop = [clip_rescale(np.random.rand(n_points)) for _ in range(100)]
    for _ in range(50):
        if time.time() - start_time > budget_s - 10:
            break
        gen += 1
        scores = np.array([objective(ind) for ind in pop])
        idxs = np.argsort(scores)
        pop = np.array(pop)[idxs[:50]]
        new_pop = []
        for i in range(0, 50, 2):
            if i+1 < 50:
                p1, p2 = pop[i], pop[i+1]
                child1 = np.where(np.random.rand(n_points) < 0.5, p1, p2)
                child2 = np.where(np.random.rand(n_points) < 0.5, p2, p1)
                new_pop.append(mutate(child1))
                new_pop.append(mutate(child2))
        pop = np.vstack([pop, new_pop])
        pop = np.array([clip_rescale(ind) for ind in pop])
        scores = np.array([objective(ind) for ind in pop])
        idxs = np.argsort(scores)
        pop = np.array(pop)[idxs[:100]]
        if np.min(scores) < best_c5:
            best_c5 = np.min(scores)
            best_h = pop[np.argmin(scores)]
    
    top_solutions = pop[np.argsort(objective(pop))[:10]]
    for sol in top_solutions:
        result = minimize(
            fun=objective,
            x0=sol,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-8}
        )

            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
