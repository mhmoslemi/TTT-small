    CR = 0.9

            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            h = h * target_sum / h_sum if h_sum != 0 else h
            pop.append(h)
        return pop
    
    pop = generate_initial_population()
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    for gen in range(max_generations):

                    trial[j] = mutant[j]
            trial_sum = np.sum(trial)
            if trial_sum != 0:
