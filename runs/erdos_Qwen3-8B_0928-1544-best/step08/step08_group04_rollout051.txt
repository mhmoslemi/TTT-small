        perturbation = np.random.normal(0, 0.1, n_points)
        perturbed_h = initial_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (target_sum / np.sum(perturbed_h))
        result = minimize(
            fun=lambda h: objective_with_penalty(h, lambda_penalty),
            x0=perturbed_h,
            method='Nelder-Mead',
            bounds=bounds,
            options={'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 500}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            lambda_penalty *= 1.2
        else:
            lambda_penalty /= 2.0
        initial_h = result.x
    return best_h, best_c5, n_points
