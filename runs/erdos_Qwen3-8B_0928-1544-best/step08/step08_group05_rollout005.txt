        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        # Use a more precise optimization method with tighter tolerances
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            sigma *= 0.95  # Reduce perturbation scale over time
    
    return best_h, best_c5, n_points
