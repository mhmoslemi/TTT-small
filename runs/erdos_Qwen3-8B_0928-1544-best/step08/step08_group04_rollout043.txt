import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.full(n_points, 0.5)
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed /= (np.sum(h_perturbed) / target_sum)
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            h_initial = result.x
    
    return best_h, best_c5, n_points
