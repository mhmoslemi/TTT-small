import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = 35
    pop_size = 50
    mutation_strength = 0.25
    sigma_decay = 0.95
    time_start = time.time()
    time_remaining = budget_s

    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        for _ in range(10):
            h = initial_h + np.random.normal(0, mutation_strength, n_points)
            h = np.clip(h, 0, 1)
            s = np.sum(h)
            if s > 0:
                h = h * (ones_count / s)
            population.append(h.copy())
        while len(population) < pop_size:
            h = np.zeros(n_points)
            idx = np.random.choice(n_points, ones_count, replace=False)
            h[idx] = 1
            population.append(h)
    else:
        for _ in range(pop_size):
            h = np.zeros(n_points)
            idx = np.random.choice(n_points, ones_count, replace=False)
            h[idx] = 1
            population.append(h)

    # Evaluate candidates
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    scores = np.array([evaluate(h) for h in population])
    best_idx = np.argmin(scores)
    best_h = population[best_idx].copy()
    best_c5 = scores[best_idx]

    while time_remaining > 0:
        elapsed = time.time() - time_start
        if elapsed % 10 < 1:  # Decay mutation strength every 10 seconds
            mutation_strength *= sigma_decay
        time_remaining = budget_s - (time.time() - time_start)

        # Select top parents
        parent_indices = np.argsort(scores)[:25]
        parents = [population[i] for i in parent_indices]

        # Generate children
        new_population = []
        for parent in parents:
            for _ in range(2):
                mut = parent + np.random.normal(0, mutation_strength, n_points)
                mut = np.clip(mut, 0, 1)
                s = np.sum(mut)
                if s > 0:
                    mut = mut * (ones_count / s)
                else:
                    mut = np.random.rand(n_points)
                    s = np.sum(mut)
                    mut = mut * (ones_count / s)
                new_population.append(mut)

        # Evaluate and merge
        new_scores = np.array([evaluate(h) for h in new_population])
        combined = population + new_population
        combined_scores = np.concatenate([scores, new_scores])
        sorted_indices = np.argsort(combined_scores)
        population = [combined[i] for i in sorted_indices[:pop_size]]
        scores = combined_scores[sorted_indices[:pop_size]]

        # Update best solution
        current_best = np.argmin(scores)
        if scores[current_best] < best_c5:
            best_c5 = scores[current_best]
            best_h = population[current_best].copy()

    return best_h, best_c5, n_points
