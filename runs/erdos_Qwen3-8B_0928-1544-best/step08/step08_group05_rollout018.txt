from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2

    def objective(h):

        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    # Multi-perturbation optimization with adaptive sigma and enhanced convergence
    for _ in range(10):  # Increased iterations for exploration
        if time.time() - start_time > budget_s:

        h_perturbed = initial_h + np.random.normal(0, sigma, n_points)

        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
