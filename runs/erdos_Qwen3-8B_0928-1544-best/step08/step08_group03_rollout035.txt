from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    population_size = 50
    elite_ratio = 0.2
    mutation_rate = 0.15

    for _ in range(100):
        if time.time() - start_time >= budget_s:
            break
        # Evaluate current population
        pop = np.random.choice(range(n_points), size=population_size, replace=True)
        candidates = [initial_h.copy() for _ in range(population_size)]
        for i in range(population_size):
            candidates[i][pop[i]] = np.random.uniform(0, 1)
            candidates[i] = candidates[i] * (ones_count / np.sum(candidates[i]))
        candidate_scores = [objective(c) for c in candidates]
        # Select elites
        elite_indices = np.argsort(candidate_scores)[:int(population_size * elite_ratio)]
        elites = [candidates[i] for i in elite_indices]
        # Crossover
        new_population = []
        for _ in range(population_size // 2):
            p1, p2 = np.random.choice(elite_indices, 2, replace=False)
            child1 = elites[p1].copy()
            child2 = elites[p2].copy()
            split = np.random.randint(1, n_points - 1)
            child1[:split] = elites[p1][:split]
            child1[split:] = elites[p2][split:]
            child2[:split] = elites[p2][:split]
            child2[split:] = elites[p1][split:]
            new_population.extend([child1, child2])
        # Mutate
        for i in range(len(new_population)):
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(n_points)
                new_population[i][idx] = np.clip(new_population[i][idx] + np.random.normal(0, 0.05), 0, 1)
                new_population[i] = new_population[i] * (ones_count / np.sum(new_population[i]))
        # Evaluate new population
        for idx, h in enumerate(new_population):
            h = h * (ones_count / np.sum(h))
            score = objective(h)
            if score < best_c5:
                best_c5 = score
                best_h = h.copy()
        # Local optimization on top candidates
        top_indices = np.argsort([objective(h) for h in new_population])[:5]
        for i in top_indices:
            h = new_population[i].copy()
            h = h * (ones_count / np.sum(h))
