        for _ in range(population_size):
            h = np.random.uniform(0.0, 1.0, n_points)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.zeros_like(h)
            else:
                scale = ones_count / sum_h
                h = h * scale
                h = np.clip(h, 0.0, 1.0)
            population.append(h)
    population = np.array(population)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()

    F = 0.8
    CR = 0.7
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        current_c5s = [np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in population]
        best_index = np.argmin(current_c5s)
        best_individual = population[best_index]

        mutants = []
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = best_individual + F * (population[a] - population[b])
            mutants.append(mutant)

        trials = []
        for i in range(population_size):
            mutant = mutants[i]
            target = population[i]
            trial = np.copy(target)
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial_repaired = np.zeros_like(trial)
            else:
                scale = ones_count / trial_sum
                trial_repaired = trial * scale
                trial_repaired = np.clip(trial_repaired, 0.0, 1.0)
            trials.append(trial_repaired)

        trial_c5s = [np.max(np.correlate(t, 1 - t, mode='full')) * dx for t in trials]
        for i in range(population_size):
            if trial_c5s[i] < current_c5s[i]:
                population[i] = trials[i]

        current_best = np.min(current_c5s)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin(current_c5s)]

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8}
    )
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5

    return final_h, final_c5, n_points
