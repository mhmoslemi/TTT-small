import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - (n_points / 2)

    bounds = [(0.0, 1.0)] * n_points

    def project_h(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / sum_h
        return h_clipped * scale

    current_h = initial_h.copy()
    best_h = current_h.copy()
    best_c5 = objective(current_h)
    start_time = time.time()
    sa_steps = 0

    while sa_steps < 1000 and time.time() - start_time < budget_s:
        noise = np.random.normal(0, 0.1, n_points)
        h_perturbed = current_h + noise
        h_perturbed = project_h(h_perturbed)
        new_c5 = objective(h_perturbed)
        current_c5 = objective(current_h)
        if new_c5 < current_c5 or np.random.rand() < np.exp((new_c5 - current_c5) / (1.0 * (1 - sa_steps / 1000))):
            current_h = h_perturbed.copy()
        if objective(current_h) < best_c5:
            best_h = current_h.copy()
            best_c5 = objective(current_h)
        sa_steps += 1

    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
    )
