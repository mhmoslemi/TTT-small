import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    best_c5 = float('inf')
    best_h = None
    
    if 'initial_h_values' in globals():
        initial_h = np.clip(initial_h_values, 0, 1)
        initial_h = initial_h * (ones_count / np.sum(initial_h))
        population = [initial_h]
        for _ in range(99):
            h = np.random.rand(n_points)
            h = h * (ones_count / np.sum(h))
            population.append(h)
    else:
        population = [np.random.rand(n_points) * (ones_count / np.sum(np.random.rand(n_points))) for _ in range(100)]
    
    start_time = time.time()
    generations = 0
    max_gen = int(budget_s * 0.3)  # 30% of budget for GA
    
    while generations < max_gen:
        c5s = [np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points) for h in population]
        indices = np.argsort(c5s)
        sorted_pop = [population[i] for i in indices]
        top_half = sorted_pop[:len(sorted_pop)//2]
        offspring = []
        
        for i in range(len(top_half)):
            for j in range(i+1, len(top_half)):
                p1, p2 = top_half[i], top_half[j]
                c1 = np.where(np.random.rand(n_points) < 0.5, p1, p2)
                c2 = np.where(np.random.rand(n_points) < 0.5, p2, p1)
                
                for k in range(n_points):
                    if np.random.rand() < 0.1:
                        m = np.random.rand()
                        c1[k] = 0 if m < 0.333 else 1 if m < 0.666 else np.random.rand()
                        c2[k] = 0 if m < 0.333 else 1 if m < 0.666 else np.random.rand()
                
                c1 = np.clip(c1, 0, 1) * (ones_count / np.sum(c1))
                c2 = np.clip(c2, 0, 1) * (ones_count / np.sum(c2))
                offspring.extend([c1, c2])
        
        new_pop = sorted_pop[:len(sorted_pop)//2]
        new_pop += offspring[:len(sorted_pop)//2]
        population = new_pop
        
        for i in range(len(sorted_pop)):
            if c5s[indices[i]] < best_c5:
                best_c5 = c5s[indices[i]]
                best_h = sorted_pop[i]
        
        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break
        generations += 1
    
    for h_candidate in sorted_pop[:10]:
        def obj(x):
            return np.max(np.correlate(x, 1 - x, mode='full')) * (2.0 / n_points)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count}
        res = minimize(obj, h_candidate, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, options={'maxiter': 150})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
    
    return best_h, best_c5, n_points
