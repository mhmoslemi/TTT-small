from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()

        h = np.full(n_points, 0.5)
    h = h * (ones_count / np.sum(h))

    def objective_with_alpha(x, alpha, dx, ones_count):
        corr = np.correlate(x, 1 - x, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx + alpha * (np.sum(x) - ones_count) ** 2

    best_h = h.copy()

        result = minimize(
            fun=lambda x: objective_with_alpha(x, alpha, dx, ones_count),
