                noise = np.random.normal(0, sigma, n_points)
                mutated = h + noise
                mutated = np.clip(mutated, 0, 1)
                sum_mut = np.sum(mutated)
                if sum_mut != 0:
                    mutated = mutated * (target_sum / sum_mut)
                new_population.append(mutated)
                if len(new_population) >= 50:
                    break
            if len(new_population) >= 50:
                break

        all_individuals = population + new_population
        all_evals = [compute_c5(h) for h in all_individuals]
        sorted_indices = np.argsort(all_evals)
        population = [all_individuals[i] for i in sorted_indices[:50]]

        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        sigma *= decay_rate

    return best_h, best_c5, n_points
