from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    decay_factor = 0.995
    
    while time.time() - start_time < budget_s:
        # Adaptive perturbation with larger initial exploration
        time_frac = (time.time() - start_time) / budget_s
        sigma = 0.15 * (1 - time_frac)  # Larger initial noise, decay with time
        h_perturbed = initial_h + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        result = minimize(
            fun=objective,
