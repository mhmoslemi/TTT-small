import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    initial_h = np.clip(initial_h, 0, 1)
    if np.sum(initial_h) > 0:
        initial_h *= ones_count / np.sum(initial_h)
    else:
        initial_h = np.full(n_points, 0.5)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def rescale(h):
        h_clipped = np.clip(h, 0, 1)
        if np.sum(h_clipped) == 0:
            return np.full(n_points, 0.5)
        return h_clipped * (ones_count / np.sum(h_clipped))
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    population = [initial_h.copy()]
    for _ in range(99):
        h = np.random.rand(n_points)
        h = h * (ones_count / np.sum(h))
        population.append(h)
    
    start_time = time.time()
    generation = 0
    while time.time() - start_time < budget_s:
        scores = [evaluate(h) for h in population]
        sorted_indices = np.argsort(scores)
        top_half = np.array(population)[sorted_indices[:len(population)//2]]
        offspring = []
        
        for _ in range(len(population) - len(top_half)):
            p1, p2 = np.random.choice(top_half, 2, replace=False)
            child = np.random.choice([p1, p2], size=n_points)

                if np.random.rand() < 0.1:

                        child[i] = 0.0

                        child[i] = 1.0
                    else:
                        child[i] = np.random.rand()
            child = rescale(child)
            offspring.append(child)
        
        population = np.vstack((top_half, offspring)).tolist()
        for h in population:
            current_c5 = evaluate(h)
            if current_c5 < best_c5:
