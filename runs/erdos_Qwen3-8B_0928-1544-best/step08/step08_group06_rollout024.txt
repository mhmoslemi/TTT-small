import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 120  # Increased resolution for better precision
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        # Structured initial guess with uniform distribution of 1s
        initial_h = np.zeros(n_points)
        uniform_step = n_points / ones_count
        for i in range(ones_count):
            idx = int(i * uniform_step)
            initial_h[idx] = 1.0
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        sigma = 0.25 * (1 - (time.time() - start_time) / budget_s)
        perturbed = best_h + np.random.normal(0, sigma, n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        # Rescale to maintain the integral constraint
        perturbed = perturbed * (ones_count / np.sum(perturbed))
        
        # Use L-BFGS-B for faster convergence with bounds
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-10, 'disp': False}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Final normalization to ensure sum constraint
    best_h = best_h * (ones_count / np.sum(best_h))
    return best_h, best_c5, n_points
