```python
import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    ones_count = n_points // 2

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        h_perturbed = initial_h + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        sum_h = np.sum(h_perturbed)
        if sum_h != 0:
            h_perturbed = h_perturbed * (ones_count / sum_h)

        def obj_with_penalty(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr) * (2.0 / n_points)
            sum_h = np.sum(h)
            penalty = 1000 * (sum_h - ones_count)**2
            return max_corr + penalty

        result = minimize(
            fun=obj_with_penalty,
            x0=h_perturbed,
            method='Nelder-Mead',
            bounds=bounds,
            options={'xatol': 1e-8, 'fatol': 1e-8, 'maxiter': 150}
        )

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            initial_h = result.x

    return best_h, best_c5, n_points
```
