import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    def project(h):
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * (ones_count / total)

    pop_size = 30
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        vec = np.random.rand(n_points)
        vec = vec * (ones_count / np.sum(vec))
        population.append(vec)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for gen in range(200):
        if time.time() - start_time >= budget_s:
            break
        fitness = [objective(ind) for ind in population]
        elites = np.argsort(fitness)[:int(len(population)*0.2)]
        parents = np.random.choice(elites, size=int(len(population)*0.4), replace=False)
        
        offspring = []
        for i in range(0, len(parents), 2):
            p1, p2 = population[parents[i]], population[parents[i+1]]
            split = np.random.randint(1, n_points-1)
            child = np.concatenate([p1[:split], p2[split:]])
            child = project(child)
            offspring.append(child)
        
        mut_rate = max(0.05, 0.15 - 0.1 * (gen / 200))
        for i in range(len(offspring)):
            if np.random.rand() < mut_rate:
                perturb = np.random.normal(0, 0.1, n_points)
                offspring[i] += perturb
                offspring[i] = project(offspring[i])
        
        new_population = []
        for ind in offspring:
            res = minimize(
                fun=objective,
                x0=ind,
                method='SLSQP',
                bounds=[(0.0, 1.0)] * n_points,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}
            )
            new_population.append(res.x)
        
        population = np.array(population) + np.array(new_population)
        population = population / np.sum(population) * ones_count
        population = np.clip(population, 0.0, 1.0)
        
        for ind in population:
            if objective(ind) < best_c5:
                best_c5 = objective(ind)
                best_h = ind.copy()
    
    return best_h, best_c5, n_points
