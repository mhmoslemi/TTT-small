def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(n_points)
        initial_h = initial_h * (target_sum / np.sum(initial_h))

        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h)
    
    pop_size = 100
    pop = [np.random.rand(n_points) * (target_sum / np.sum(np.random.rand(n_points))) for _ in range(pop_size)]
    pop = [np.clip(h, 0, 1) for h in pop]
    
    for _ in range(100):
        parents = sorted(pop, key=evaluate_c5)[:pop_size//2]
        offspring = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1]
            for _ in range(2):
                child = np.random.choice([p1, p2], size=n_points)
                for j in range(n_points):
                    if np.random.rand() < 0.1:

        combined = pop + offspring
        combined.sort(key=evaluate_c5)
        pop = combined[:pop_size]
        current_c5 = evaluate_c5(pop[0])
        if current_c5 < best_c5:

            best_h = pop[0]
