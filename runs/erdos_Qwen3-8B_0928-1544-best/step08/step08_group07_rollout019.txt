import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for _ in range(300):
        sigma = 0.07 * (1 - (_ / 300))
        h_perturbed = initial_h + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun

                best_h = result.x
                initial_h = result.x
    
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        for _ in range(200):
