import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    n_particles = 20
    omega = 0.7
    c1, c2 = 2.0, 2.0
    max_vel = 1.0
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(n_points)

    for _ in range(n_particles - 1):
        p = np.random.rand(n_points)
        p = np.clip(p, 0, 1)
        p = p * (ones_count / np.sum(p))
        particles.append(p)

    pbest_values = [np.max(np.correlate(p, 1 - p, mode='full')) * (2.0 / n_points) for p in particles]
    gbest = particles[np.argmin(pbest_values)]
    gbest_value = np.min(pbest_values)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
