import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    population_size = 20
    population = [initial_h.copy() for _ in range(population_size)]
    for i in range(1, population_size):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (ones_count / np.sum(h))
        population[i] = h.copy()
    best_h = initial_h.copy()
    best_c5 = np.inf
    dx = 2.0 / n_points
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    start_time = time.time()
    F = 0.7
    CR = 0.9
    for gen in range(1000):
        if time.time() - start_time >= budget_s:
            break
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec = population[a]
            b_vec = population[b]
            c_vec = population[c]
            mutant = a_vec + F * (b_vec - c_vec)
            trial = population[i].copy()
            mask = np.random.rand(n_points) < CR
            trial[mask] = mutant[mask]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial = np.full(n_points, 0.5)
            else:
                trial = trial * (ones_count / trial_sum)
            current_c5 = compute_c5(trial)
            parent_c5 = compute_c5(population[i])
            if current_c5 < parent_c5:
                population[i] = trial.copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = trial.copy()
    return best_h, best_c5, n_points
