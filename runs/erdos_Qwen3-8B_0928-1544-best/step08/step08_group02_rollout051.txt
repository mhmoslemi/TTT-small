def run(seed=42, budget_s=1100, **kwargs):

    pbest_c5 = [objective(p) for p in particles]
    
    gbest_idx = np.argmin(pbest_c5)
    gbest_h = pbest_h[gbest_idx].copy()

    max_stagnation = 100
    stagnation = 0
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:

            velocities[i] = omega * velocities[i] + c1 * r1 * (pbest_h[i] - particles[i]) + c2 * r2 * (gbest_h - particles[i])

            new_h = new_h * (ones_count / np.sum(new_h))
            
            current_c5 = objective(new_h)
            
            if current_c5 < pbest_c5[i]:
                pbest_h[i] = new_h.copy()
                pbest_c5[i] = current_c5
