import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    best_h = None
    best_c5 = 1.0
    n_points = 70
    target_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    initial_h = np.clip(initial_h, 0.0, 1.0)
    sum_h = np.sum(initial_h)
    if sum_h > 0:
        initial_h = initial_h * (target_sum / sum_h)

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def repair(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.full_like(h, 0.5)
        scale = target_sum / sum_clipped
        return h_clipped * scale

    bounds = [(0.0, 1.0)] * n_points

    start_time = time.time()

    try:
        result = differential_evolution(
            fun=obj,
            bounds=bounds,
            popsize=50,
            tol=1e-6,
            maxiter=1000,
            mutation=(0.5, 1.0),
            recombination=0.85,
            seed=seed,
            strategy='best1bin'
        )
        best_h = repair(result.x)
        best_c5 = result.fun
    except:
        best_h = initial_h
        best_c5 = obj(best_h)

    elapsed = time.time() - start_time
    if elapsed < budget_s:
        try:
            result = differential_evolution(
                fun=obj,
                bounds=bounds,
                popsize=75,
                tol=1e-6,
                maxiter=1000,
                mutation=(0.6, 0.9),
                recombination=0.8,
                seed=seed + 1,
                strategy='best1bin'
            )
            best_h = repair(result.x)
            best_c5 = result.fun
        except:
            pass

    final_h = np.clip(best_h, 0.0, 1.0)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    return final_h, best_c5, n_points
