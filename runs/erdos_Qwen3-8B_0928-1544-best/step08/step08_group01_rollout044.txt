import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    T = n_points / 2
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        sum_initial = np.sum(initial_h)
        if sum_initial > 0:
            initial_h = initial_h * (T / sum_initial)
        else:
            initial_h = np.full(n_points, T / n_points)
    else:
        initial_h = np.random.rand(n_points)
        initial_h = initial_h * (T / np.sum(initial_h))
    
    num_particles = 50
    particles_positions = np.random.rand(num_particles, n_points)
    particles_velocities = np.random.normal(0, 0.1, (num_particles, n_points))
    
    if 'initial_h_values' in globals():
        particles_positions[0, :] = initial_h
        for i in range(1, num_particles):
            guess = np.random.rand(n_points)
            guess = guess * (T / np.sum(guess))
            particles_positions[i, :] = guess
    else:
        for i in range(num_particles):
            guess = np.random.rand(n_points)
            guess = guess * (T / np.sum(guess))
            particles_positions[i, :] = guess
    
    pbest_positions = particles_positions.copy()
    pbest_c5 = np.full(num_particles, float('inf'))
    gbest_positions = particles_positions[0].copy()
    gbest_c5 = float('inf')
    
    w, c1, c2 = 0.7, 2.0, 2.0
    dx = 2.0 / n_points
    
    start_time = time.time()
    max_iter = 1000
    
    for _ in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        for i in range(num_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            particles_velocities[i] = w * particles_velocities[i] + c1 * r1 * (pbest_positions[i] - particles_positions[i]) + c2 * r2 * (gbest_positions - particles_positions[i])
            
            particles_positions[i] += particles_velocities[i]
            particles_positions[i] = np.clip(particles_positions[i], 0, 1)
            sum_pos = np.sum(particles_positions[i])
            if sum_pos > 0:
                particles_positions[i] *= (T / sum_pos)
            
            h = particles_positions[i]
            ones_minus_h = 1 - h
            corr = np.correlate(h, ones_minus_h, mode='full')
            current_c5 = np.max(corr) * dx
            
            if current_c5 < pbest_c5[i]:
                pbest_positions[i] = particles_positions[i].copy()
                pbest_c5[i] = current_c5
            
            if current_c5 < gbest_c5:
                gbest_positions = particles_positions[i].copy()
                gbest_c5 = current_c5
    
    best_h = gbest_positions
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best > 0:
        best_h *= (T / sum_best)
    
    return best_h, gbest_c5, n_points
