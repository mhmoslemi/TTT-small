import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    best_h = initial_h.copy()
    best_c5 = 0.0
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * (2.0 / n_points)
    
    start_time = time.time()
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        i_max = np.argmax(corr)
        k_max = i_max - (n_points - 1)
        indices = (k_max - np.arange(n_points)) % n_points
        grad = 1 - best_h[indices]
        step_dir = -grad
        
        alpha = 0.1
        improved = False
        for _ in range(10):
            h_candidate = best_h + alpha * step_dir
            h_candidate = np.clip(h_candidate, 0, 1)
            sum_candidate = np.sum(h_candidate)
            if sum_candidate == 0:
                sum_candidate = 1.0
            scale = (n_points / 2) / sum_candidate
            h_candidate = h_candidate * scale
            new_c5 = compute_c5(h_candidate)
            if new_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = new_c5
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-6:
                break
        
        if not improved:
            break
    
    return best_h, best_c5, n_points
