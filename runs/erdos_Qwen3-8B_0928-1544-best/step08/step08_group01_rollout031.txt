            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            new_evaluations.append(c5)
        sorted_new = np.argsort(new_evaluations)
        population = [new_population[i] for i in sorted_new[:100]]
        best_idx = np.argmin(evaluations)
        current_c5 = evaluations[best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[best_idx]
    top_solutions = population[:10]
    for h in top_solutions:
        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * (2.0 / n_points)
        bounds = [(0.0, 1.0)] * n_points
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-8})
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
    best_h = np.clip(best_h, 0, 1)
    if np.sum(best_h) != target_sum:
        best_h = best_h * (target_sum / np.sum(best_h))
    return best_h, best_c5, n_points
