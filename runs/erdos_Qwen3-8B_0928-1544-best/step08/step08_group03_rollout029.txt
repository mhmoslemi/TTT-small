            z = improvement / std
            ei = improvement * norm.cdf(z) + std * norm.pdf(z)
            return -ei
        
        result = minimize(
            fun=acquisition,
            x0=best_h,
            bounds=[(0.0, 1.0)] * n_points,
            method='SLSQP',
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count}]
        )
        if not result.success:
            continue
        
        h_candidate = result.x
        h_candidate = project_h(h_candidate)
        current_c5 = compute_c5(h_candidate)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_candidate
        
        X_new = np.array([h_candidate]).reshape(1, -1)
        y_new = np.array([current_c5])
        gp.fit(X_new, y_new)
        initial_h = h_candidate
    
    return best_h, best_c5, n_points
