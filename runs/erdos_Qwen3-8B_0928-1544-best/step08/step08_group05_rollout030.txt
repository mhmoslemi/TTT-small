    best_c5 = compute_c5(initial_h)
    start_time = time.time()
    
    learning_rate = 0.05
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(initial_h, 1 - initial_h, mode='full')
        max_overlap = np.max(corr)
        k_max = np.argmax(corr)
        k = k_max
        m = np.arange(n_points)
        indices_plus = m + k
        indices_plus_valid = indices_plus < n_points
        grad = np.zeros(n_points)
        grad[indices_plus_valid] += 1 - initial_h[indices_plus[indices_plus_valid]]
        indices_minus = m - k
        indices_minus_valid = indices_minus >= 0
        grad[indices_minus_valid] -= initial_h[indices_minus[indices_minus_valid]]
        
        initial_h = initial_h - learning_rate * grad
        initial_h = np.clip(initial_h, 0, 1)
        sum_h = np.sum(initial_h)
        delta = (ones_count - sum_h) / n_points
        initial_h += delta
        initial_h = np.clip(initial_h, 0, 1)
        sum_h = np.sum(initial_h)
        delta = (ones_count - sum_h) / n_points
        initial_h += delta
        initial_h = np.clip(initial_h, 0, 1)
        
        current_c5 = compute_c5(initial_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = initial_h.copy()
        learning_rate *= 0.995
    
    return best_h, best_c5, n_points
