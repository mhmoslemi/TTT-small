import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    levels = [35, 70]
    best_c5 = float('inf')
    best_h = None
    total_time = 0

    for n_points in levels:
        dx = 2.0 / n_points
        target = n_points / 2.0

        if n_points == 35:
            if 'initial_h_values' in globals():
                current_h = initial_h_values.copy()
            else:
                current_h = np.full(n_points, 0.5)
        else:
            if best_h is None:
                continue
            prev_h = best_h
            current_h = np.zeros(n_points)
            idx = 0
            for i in range(n_points):
                current_h[i] = prev_h[idx]
                idx += 1
                if idx < len(prev_h):
                    current_h[i+1] = prev_h[idx]
                else:
                    break
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (target / np.sum(current_h))

        h_perturbed = current_h + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (target / np.sum(h_perturbed))

        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            penalty = 10 * (np.sum(h) - target)**2
            return c5 + penalty

        bounds = [(0.0, 1.0)] * n_points
        start_time = time.time()
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='Nelder-Mead',
            bounds=bounds,
            options={'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 300}
        )
        elapsed = time.time() - start_time
        total_time += elapsed

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (target / np.sum(best_h))
        else:
            best_h = current_h.copy()
            best_c5 = objective(best_h)

        if total_time >= budget_s:
            break

    return best_h, best_c5, levels[-1] if best_h is not None else 70
