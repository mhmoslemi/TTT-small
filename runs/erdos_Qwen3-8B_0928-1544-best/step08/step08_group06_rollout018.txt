        if sum_mutated == 0:
            return np.zeros(n_points)
        scale = ones_count / sum_mutated
        mutated = mutated * scale
        return mutated

    population_size = 50
    population = []
    if 'initial_h_values' in globals():
        for _ in range(10):
            population.append(mutate(initial_guess, 0.25))
        while len(population) < population_size:
            population.append(mutate(initial_guess, 0.25))
    else:
        for _ in range(10):
            population.append(mutate(initial_guess, 0.25))
        while len(population) < population_size:
            population.append(mutate(initial_guess, 0.25))
    population = np.array(population)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = initial_guess.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    sigma = 0.25
    decay_factor = 0.95

    while time.time() - start_time < budget_s:
        c5_values = np.array([compute_c5(h) for h in population])
        top_indices = np.argsort(c5_values)[:25]
        parents = population[top_indices]

        children = []
        for parent in parents:
            noise = np.random.normal(0, sigma, n_points)
            mutated = parent + noise
            mutated = np.clip(mutated, 0.0, 1.0)
            sum_mutated = np.sum(mutated)
            if sum_mutated == 0:
                mutated = np.zeros(n_points)
            else:
                scale = ones_count / sum_mutated
                mutated = mutated * scale
            children.append(mutated)
        children = np.array(children)

        all_candidates = np.vstack([parents, children])
        all_c5 = np.array([compute_c5(h) for h in all_candidates])
        new_population = all_candidates[np.argsort(all_c5)[:population_size]]
        population = new_population

        current_best_index = np.argmin(all_c5)
        if all_c5[current_best_index] < best_c5:
            best_c5 = all_c5[current_best_index]
            best_h = all_candidates[current_best_index]

        if (time.time() - start_time) > 10:
            sigma *= decay_factor

    return best_h, best_c5, n_points
