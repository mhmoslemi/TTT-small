import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    n_particles = 20
    max_stagnation = 50
    omega = 0.7
    c1 = 2.0
    c2 = 2.0
    velocity_scale = 0.1

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        particles = [np.random.rand(n_points) for _ in range(n_particles - 1)]
        particles.append(initial_h)
    else:
        particles = [np.random.rand(n_points) for _ in range(n_particles)]

    def clip_scale(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return np.full_like(h_clipped, ones_count / n_points)
        scale = ones_count / sum_h
        return h_clipped * scale

    particles = [clip_scale(p) for p in particles]
    pbest = particles.copy()
    gbest = min(pbest, key=lambda x: objective(x))
    best_c5 = objective(gbest)
    velocities = [np.random.rand(n_points) * velocity_scale for _ in range(n_particles)]

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    start_time = time.time()
    stagnation_count = 0
    iteration = 0

    while True:
        if time.time() - start_time >= budget_s:
            break
        iteration += 1
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = omega * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -0.5, 0.5)
            particles[i] += velocities[i]
            particles[i] = clip_scale(particles[i])
        new_pbest = []
        for i in range(n_particles):
            current_c5 = objective(particles[i])
            if current_c5 < objective(pbest[i]):
                pbest[i] = particles[i].copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    gbest = pbest[i].copy()
            new_pbest.append(pbest[i])
        if np.allclose(new_pbest, pbest, atol=1e-8):
            stagnation_count += 1
            if stagnation_count > max_stagnation:
                particles = [np.random.rand(n_points) for _ in range(n_particles)]
                particles = [clip_scale(p) for p in particles]
                pbest = particles.copy()
                gbest = min(pbest, key=lambda x: objective(x))
                best_c5 = objective(gbest)
                stagnation_count = 0
        else:
            stagnation_count = 0

    return gbest, best_c5, n_points
