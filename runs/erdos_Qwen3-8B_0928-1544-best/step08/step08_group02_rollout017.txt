import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    bounds = [(0.0, 1.0)] * n_points
    pop_size = 20
    F = 0.8
    CR = 0.9
    max_gen = int(budget_s / 100)
    
    population = np.random.uniform(0, 1, (pop_size, n_points))
    for i in range(pop_size):
        population[i] = population[i] * (ones_count / np.sum(population[i]))
    population[0] = initial_h
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    for gen in range(max_gen):
        if time.time() - start_time >= budget_s:
            break
        new_pop = []
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            v = population[a] + F * (population[b] - population[c])
            trial = np.where(np.random.rand(n_points) < CR, v, population[i])
            trial = np.clip(trial, 0.0, 1.0)
            trial = trial * (ones_count / np.sum(trial))
            new_pop.append(trial)
        population = np.array(new_pop)
        for h in population:
            curr = objective(h)
            if curr < best_c5:
                best_c5 = curr
                best_h = h.copy()
    
    return best_h, best_c5, n_points
