import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 20
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    # Scale initial_h to ensure sum constraint
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    population = [initial_h]
    # Generate remaining population members
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (ones_count / np.sum(h))
        population.append(h)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(initial_h)
    start_time = time.time()
    F, CR = 0.7, 0.9
    
    while time.time() - start_time < budget_s:
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_ind, b_ind, c_ind = population[a], population[b], population[c]
            # Mutation
            v = a_ind + F * (b_ind - c_ind)
            # Crossover
            u = a_ind.copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            # Clip
            u = np.clip(u, 0.0, 1.0)
            # Repair sum constraint
            sum_u = np.sum(u)
            if sum_u != 0:
                u = u * (ones_count / sum_u)
            # Evaluate
            current_c5 = compute_c5(u)
            parent_c5 = compute_c5(a_ind)
            if current_c5 < parent_c5:
                population[i] = u
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = u.copy()
    
    return best_h, best_c5, n_points
