from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    F = 0.8
    CR = 0.9
    pop_size = 100
    max_generations = 600
    pop = []
    
    # Initialize population based on initial_h
    for _ in range(pop_size):
        perturb = np.random.normal(0, 0.1, n_points)
        h = initial_h + perturb
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        h = h * (target_sum / h_sum)
        pop.append(h)
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    # Differential Evolution phase
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + F * (pop[a] - pop[b])
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            # Enforce constraints
            trial_sum = np.sum(trial)
            trial = trial * (target_sum / trial_sum)
            trial = np.clip(trial, 0.0, 1.0)
            trial_obj = objective(trial)
            if trial_obj < objective(pop[i]):
                pop[i] = trial.copy()
                if trial_obj < best_c5:
                    best_c5 = trial_obj
                    best_h = trial.copy()
    
    # SLSQP refinement
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    final_h = result.x if result.success else best_h
    final_c5 = result.fun if result.success else best_c5
