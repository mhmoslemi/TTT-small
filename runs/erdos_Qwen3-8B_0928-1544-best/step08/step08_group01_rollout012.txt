def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    
    # Initialize h from initial_h_values if available

        else:
            initial_h = np.full(n_points, target_sum / n_points)

        personal_best_c5.append(c5)
