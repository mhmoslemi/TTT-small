import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    start_time = time.time()
    
    initial_guesses = []
    if 'initial_h_values' in globals():
        base_guess = initial_h_values.copy()
        for _ in range(5):
            perturbed = base_guess + np.random.normal(0, 0.1, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (ones_count / np.sum(perturbed))
            initial_guesses.append(perturbed)
    for _ in range(15):
        guess = np.random.rand(n_points)
        guess = guess * (ones_count / np.sum(guess))
        initial_guesses.append(guess)
    
    for guess in initial_guesses:
        result = minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-9}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
