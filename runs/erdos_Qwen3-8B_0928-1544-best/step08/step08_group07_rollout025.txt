            child = alpha * parent1 + (1 - alpha) * parent2
            child = np.clip(child, 0, 1)
            sum_child = np.sum(child)
            if sum_child == 0:
                child = np.random.rand(n_points) * 0.5
                child = child * (ones_count_fine / np.sum(child))
            else:
                child = child * (ones_count_fine / sum_child)
            new_pop.append(child)
        population = new_pop
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    best_h_fine_final = population[np.argmin([evaluate_fine(h) for h in population])]
    best_c5 = evaluate_fine(best_h_fine_final)
    return best_h_fine_final, best_c5, n_points
