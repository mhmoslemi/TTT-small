import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(n_points)
        initial_h = initial_h * ones_count / np.sum(initial_h)
    population_size = 40
    population = [initial_h.copy() for _ in range(population_size)]
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full') * (2.0 / n_points))
    start_time = time.time()
    sigma = 0.05
    max_generations = 100

    for gen in range(max_generations):
        if time.time() - start_time >= budget_s:
            break
        new_population = []
        for _ in range(population_size):
            parent = population[np.random.randint(len(population))]
            noise = np.random.normal(0, sigma, n_points)
            h = parent + noise
            h = np.clip(h, 0, 1)
            h = h * ones_count / np.sum(h)
            new_population.append(h)
        scores = []
        for h in new_population:
            corr = np.correlate(h, 1 - h, mode='full')
            dx = 2.0 / n_points
            overlap_vals = corr * dx
            f1 = np.max(overlap_vals)
            scores.append(f1)
        new_population.sort(key=lambda x: np.max(np.correlate(x, 1 - x, mode='full') * (2.0 / n_points)))
        top_10_percent = new_population[:int(len(new_population)*0.1)]
        population = top_10_percent
        for h in new_population:
            corr = np.correlate(h, 1 - h, mode='full')
            dx = 2.0 / n_points
            overlap_vals = corr * dx
            f1 = np.max(overlap_vals)
            if f1 < best_c5:
                best_c5 = f1
                best_h = h.copy()
    return best_h, best_c5, n_points
