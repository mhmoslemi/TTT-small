import numpy as np
from scipy.optimize import minimize

    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    # Initialize h
    if 'initial_h_values' in globals():

    def objective(h):

    # Compute C5 bound
    h_opt = result.x
