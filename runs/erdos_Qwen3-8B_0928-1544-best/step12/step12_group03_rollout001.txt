import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    target_sum = n_points // 2
    best_h = None
    best_c5 = float('inf')
    epsilon = 0.1
    stagnation = 0
    max_stagnation = 50
    backtrack_steps = 0

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        original_x = np.linspace(0, 2, len(initial_h))
        new_x = np.linspace(0, 2, n_points)
        best_h = np.interp(new_x, original_x, initial_h)
    else:
        h = np.random.rand(n_points) * 0.5
        sum_h = np.sum(h)
        if sum_h == 0:
            best_h = np.random.rand(n_points) * 0.5 * (target_sum / 0.5)
        else:
            best_h = h * (target_sum / sum_h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_c5 = compute_c5(best_h)

    start_time = time.time()

    while time.time() - start_time < budget_s:
        s = np.random.choice([-1, 0, 1], size=n_points)
        delta = s * epsilon
        candidate = best_h + delta
        candidate = np.clip(candidate, 0, 1)
        sum_candidate = np.sum(candidate)
        if sum_candidate == 0:
            candidate = np.random.rand(n_points) * 0.5
            sum_candidate = np.sum(candidate)
            candidate = candidate * (target_sum / sum_candidate)
        else:
            candidate = candidate * (target_sum / sum_candidate)
        new_c5 = compute_c5(candidate)

        if new_c5 < best_c5:
            best_h = candidate.copy()
            best_c5 = new_c5
            stagnation = 0
            epsilon *= 0.8
        else:
            epsilon *= 1.1
            stagnation += 1

        if stagnation > max_stagnation:
            if backtrack_steps < 3:
                best_h = best_h.copy()
                best_c5 = compute_c5(best_h)
                s = np.random.choice([-1, 0, 1], size=n_points)
                perturb = s * 0.5
                best_h += perturb
                best_h = np.clip(best_h, 0, 1)
                sum_h = np.sum(best_h)
                if sum_h != 0:
                    best_h = best_h * (target_sum / sum_h)
                best_c5 = compute_c5(best_h)
                backtrack_steps += 1
                stagnation = 0
            else:
                epsilon = 0.1
                backtrack_steps = 0

    return best_h, best_c5, n_points
