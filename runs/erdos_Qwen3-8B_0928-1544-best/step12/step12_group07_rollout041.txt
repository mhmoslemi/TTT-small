    np.random.seed(seed)
    n_points = 140
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points)
        h = h * (target_sum / h_sum)

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_val = np.max(corr)
        return max_val * (2.0 / n_points)

    def grad_obj(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for x in range(n_points):
            y = x + k
            if 0 <= y < n_points:
                grad[x] = 1.0 - h_vec[y]
        return grad

    constraints = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - target_sum
    }]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        jac=grad_obj,
        bounds=bounds,
        constraints=constraints,
        tol=1e-8,
        options={'maxiter': 500}
    )
    best_h = result.x
    c5 = objective(best_h)
    return best_h, c5, n_points
