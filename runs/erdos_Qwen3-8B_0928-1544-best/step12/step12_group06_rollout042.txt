import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    n_half = 35
    dx = 2.0 / n_points
    population_size = 100
    mutation_rate = 0.1
    time_start = time.time()
    
    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        sum_initial = np.sum(initial_h)
        initial_h = initial_h * n_half / sum_initial
        population = [initial_h]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points)
            sum_h = np.sum(h)
            h = h * n_half / sum_h
            population.append(h)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points)
            sum_h = np.sum(h)
            h = h * n_half / sum_h
            population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    while time.time() - time_start < budget_s:
        # Evaluate and sort population
        c5_values = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            c5 = max_corr * dx
            c5_values.append(c5)
        sorted_indices = np.argsort(c5_values)
        best_idx = sorted_indices[0]
        if c5_values[best_idx] < best_c5:
            best_c5 = c5_values[best_idx]
            best_h = population[best_idx].copy()
        
        # Elitism
        num_elite = int(population_size * 0.2)
        elites = [population[i] for i in sorted_indices[:num_elite]]
        new_population = elites.copy()
        remaining = population_size - num_elite
        
        # Selection and crossover
        for _ in range(remaining):
            # Tournament selection
            tournament = np.random.choice(len(population), size=5, replace=False)
            parent1 = population[tournament[np.argmin(c5_values[tournament])]]
            tournament2 = np.random.choice(len(population), size=5, replace=False)
            parent2 = population[tournament2[np.argmin(c5_values[tournament2])]]
            
            # Crossover
            child = np.zeros_like(parent1)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.uniform(-0.05, 0.05)
            child = np.clip(child, 0, 1)
            sum_child = np.sum(child)
            if sum_child == 0:
                h = np.random.rand(n_points)
                sum_h = np.sum(h)
                h = h * n_half / sum_h
                child = h
            else:
                child = child * n_half / sum_child
            
            new_population.append(child)
        
        population = new_population
    
    return best_h, best_c5, n_points
