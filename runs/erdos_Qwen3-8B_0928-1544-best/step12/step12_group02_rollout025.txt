import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:ones_count] = 1.0
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (ones_count / h_sum)
    else:
        h = np.full(n_points, 1.0 / n_points)
    
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        overlaps = corr * dx
        return np.sum(overlaps**2)
    
    def constraint_sum(x):
        return np.sum(x) - ones_count
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-9, 'maxiter': 200})
    best_h = result.x
    
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    max_overlap = np.max(corr * dx)
    
    return best_h, max_overlap, n_points
