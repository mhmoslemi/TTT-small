        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

        if time.time() - start_time >= budget_s:
            break

    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best > 0:
        best_h = best_h * (ones_count / sum_best)

    return best_h, best_c5, n_points
