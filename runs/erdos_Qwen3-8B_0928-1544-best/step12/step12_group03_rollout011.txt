import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    initial_guesses = []

    if 'initial_h_values' in globals():
        initial_guesses.append(initial_h_values.copy())

    # Add uniform and random guesses
    initial_guesses.append(np.full(n_points, 0.5))
    for _ in range(4):
        h = np.random.rand(n_points)
        h = h * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        else:
            h = h * (target_sum / h_sum)
        initial_guesses.append(h)

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    def objective(x):
        dx = 2.0 / n_points
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    for guess in initial_guesses:
        h = np.clip(guess, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        else:
            h = h * (target_sum / h_sum)
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()
        if time.time() - start_time >= budget_s:
            break

    return best_h, best_c5, n_points
