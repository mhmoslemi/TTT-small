                corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
                c5 = np.max(corr) * (2.0 / n_points)
                if c5 < min_c5:
                    min_c5 = c5
                    best_v = v
            if min_c5 < best_c5:
                best_c5 = min_c5
                h = h.copy()
                h[i] = best_v
                new_sum = current_sum - current_h_i + best_v
                scale_factor = target_sum / new_sum
                h *= scale_factor
            else:
                perturb = np.random.rand(n_points) * 0.01 - 0.005
                h += perturb
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                if h_sum == 0:
                    h = np.random.rand(n_points) * 0.5 * (target_sum / 0.5)
                else:
                    h = h * (target_sum / h_sum)
    return best_h, best_c5, n_points
