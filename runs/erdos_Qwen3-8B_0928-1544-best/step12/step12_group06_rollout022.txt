import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()

        h = h * (target_sum / np.sum(h))

    best_c5 = float('inf')
    max_iterations = int(budget_s / 5)
    start_time = time.time()

    for _ in range(max_iterations):
        corr = np.correlate(h, 1 - h, mode='full')
        corr_n = corr[n_points - 1: 2 * n_points - 1]
        k = np.argmax(corr_n)
        k = (k + 0) % n_points  # Ensure k is in [0, n_points)

        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j1 = (i + k) % n_points
            j2 = (i - k) % n_points
            A[i, j1] = 1.0
            A[i, j2] = 1.0

        h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
        objective = cp.Maximize(dx * (cp.quad_form(h_var, A) / 2))
        constraints = [cp.sum(h_var) == target_sum]
        prob = cp.Problem(objective, constraints)

        try:

            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INFEASIBLE]:
                new_h = h_var.value
            else:
                new_h = h.copy()
        except:
            new_h = h.copy()

        new_h = np.clip(new_h, 0.0, 1.0)

        if current_c5 < best_c5:
