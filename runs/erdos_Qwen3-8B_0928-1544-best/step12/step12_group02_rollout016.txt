        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.sum(corr**2) * dx**2

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    options = {'maxiter': 100, 'ftol': 1e-9, 'disp': False}
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options=options)

    if result.success:
        best_h = result.x
    else:
        best_h = h

    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5 = np.max(corr) * dx

    return best_h, c5, n_points
