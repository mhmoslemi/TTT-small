import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    required_sum = n_points / 2

    else:
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:

        h = h * (required_sum / h_sum)
    h = np.clip(h, 0.0, 1.0)
    best_h = h.copy()
    best_val = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    T_initial = 0.5
    T_final = 1e-4
    cooling_rate = 0.99
    T = T_initial
    start = time.time()
    
    while time.time() - start < budget_s:
        if T < T_final:
            break
        for _ in range(1000):
            noise = np.random.randn(n_points) * T
            h_try = h + noise
            h_try = np.clip(h_try, 0.0, 1.0)
            h_sum = np.sum(h_try)
            if h_sum == 0:

            h = h_try.copy()

    return best_h, best_val, n_points
