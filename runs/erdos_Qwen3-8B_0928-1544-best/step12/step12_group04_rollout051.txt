        h *= required_sum / np.sum(h)

    best_h = h.copy()
    best_val = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    T = 0.5
    cooling_rate = 0.99
    start = time.time()
    
    while T > 1e-4 and time.time() - start < budget_s:

            if val < best_val or np.random.rand() < np.exp((best_val - val) / T):
