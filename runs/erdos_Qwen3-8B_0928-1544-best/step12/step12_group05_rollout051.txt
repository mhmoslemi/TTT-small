        h = h * (target_sum / h_sum)
        s0 = np.zeros(n_points + 1)
        s0[0] = 0.0
        for i in range(n_points):
            s0[i + 1] = s0[i] + h[i] * dx

    s = cp.Variable(n_points + 1)
    C5 = cp.Variable()

    constraints = [
        s[0] == 0.0,
        s[n_points] == 1.0,
    ]

    for i in range(n_points):
        constraints.append(s[i + 1] - s[i] >= 0.0)
        constraints.append(s[i + 1] - s[i] <= dx)

    for k in range(1, n_points + 1):
        for i in range(n_points - k + 1):
            constraints.append(s[i + k] - s[i] <= C5)

    objective = cp.Minimize(C5)
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
    except:
        try:
            prob.solve(solver=cp.GUROBI, verbose=False)
        except:
            prob.solve(solver=cp.SCS, verbose=False)

    if prob.status != 'OPTIMAL':
        s_opt = s0
    else:
        s_opt = s.value

    h_values = np.zeros(n_points)
    for i in range(n_points):
        h_values[i] = (s_opt[i + 1] - s_opt[i]) / dx

    c5_bound = C5.value if prob.status == 'OPTIMAL' else 1.0

    assert np.all(h_values >= 0.0) and np.all(h_values <= 1.0)
    assert np.isclose(np.sum(h_values) * dx, 1.0)

    return h_values, c5_bound, n_points
