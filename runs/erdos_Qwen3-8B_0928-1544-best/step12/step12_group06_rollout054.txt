import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    max_iter = int(budget_s / 10)  # Adjusted iteration limit for time efficiency

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        h = h * (35.0 / h_sum)

    def compute_overlaps(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return corr

    best_h = h.copy()
    best_c5 = np.max(compute_overlaps(h)) * dx
    T = time.time()

    for _ in range(max_iter):
        corr = compute_overlaps(h)
        # Find the worst shift
        worst_k = np.argmax(corr)
        # For this worst_k, define the objective function
        def objective(x):
            total = 0.0
            for i in range(n_points):
                j = (i + worst_k) % n_points
                total += x[i] * (1.0 - x[j])
            return total

        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 35.0}]
        try:
            result = minimize(objective, h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
            if result.success:
                h = result.x
        except:
            pass  # Skip if optimization fails

        # Recompute best C5
        current_c5 = np.max(compute_overlaps(h)) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        else:
            break  # Stop if no improvement

    return best_h, best_c5, n_points
