        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum)

    # Rescale to ensure integral constraint
    h = h * (ones_count / np.sum(h))

    # Define continuous variable
    h_var = cp.Variable(n_points)
    constraints = [0 <= h_var, h_var <= 1]
    constraints.append(cp.sum(h_var) == ones_count)

    # Define objective: minimize max cross-correlation
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial c5 value
    initial_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = initial_c5

    # Optimization steps
    for _ in range(100):
        # Define objective function
        objective = cp.Minimize(cp.Max(cp.multiply(h_var, 1 - h_var)))

        # Solve problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, max_iter=100, eps=1e-6)
            if prob.status == 'OPTIMAL':
                h_new = h_var.value
                c5 = compute_c5(h_new)
                if c5 < best_c5:

                    best_h = h_new.copy()
