import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    start_time = time.time()
    learning_rate = 0.1
    max_iterations = 1000

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    for _ in range(max_iterations):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        lag_indices = np.where(corr == max_corr)[0]
        lag_index = lag_indices[0]
        k = lag_index - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= (i + k) < n_points:
                gradient[i] = 1 - h[i + k]
        alpha = learning_rate
        for _ in range(5):
            new_h = h - alpha * gradient
            new_h = np.clip(new_h, 0.0, 1.0)
            sum_new = np.sum(new_h)

            new_corr = compute_c5(new_h)
            if new_corr < max_corr:

        h = new_h
