import time
import scipy.optimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140  # Higher resolution for better accuracy
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum)
    else:
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h = h * (ones_count / h_sum)

    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)

    start_time = time.time()

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        
        gradient = np.zeros(n_points)
        for x in range(n_points):
            x_shifted = x + k
            if 0 <= x_shifted < n_points:
                gradient[x] = 1 - h[x_shifted]
        
        direction = -gradient
        alpha = 0.15
        new_h = h + alpha * direction
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h_sum = np.sum(new_h)
        if new_h_sum == 0:
            continue
        new_h = new_h * (ones_count / new_h_sum)
        
        new_c5 = compute_c5(new_h, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        h = new_h

    return best_h, best_c5, n_points

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_correlation = np.max(corr) * (2.0 / n_points)
    return max_correlation
