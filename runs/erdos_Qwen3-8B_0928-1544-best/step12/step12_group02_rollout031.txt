        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            trial = pop[i] * (1 - CR) + mutant * CR
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                continue
            trial_scaled = trial * (ones_count / trial_sum)
            curr_c5 = objective(pop[i])
            new_c5 = objective(trial_scaled)
            if new_c5 < curr_c5:
                pop[i] = trial_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = trial_scaled.copy()
                stagnation = 0
            else:
                stagnation += 1
            if stagnation > 200:
                F = max(F_start, F - 0.01)
                CR = min(CR_end, CR + 0.01)
                stagnation = 0

    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0:
        def grad_objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            j_max = np.argmax(corr)
            s = j_max - n_points + 1
            gradient = np.zeros(n_points)
            for i in range(n_points):
                if i + s < n_points:
                    gradient[i] += (1 - h[i + s])
                if i - s >= 0:
                    gradient[i] -= h[i - s]
            return gradient

        def constraint_sum(h):
            return np.sum(h) - n_points / 2

        cons = [
            {'type': 'eq', 'fun': constraint_sum},
            {'type': 'ineq', 'fun': lambda h: h},
            {'type': 'ineq', 'fun': lambda h: np.ones_like(h) - h}
        ]

        res = minimize(objective, best_h, method='SLSQP', jac=grad_objective, constraints=cons, options={'maxiter': 100})
        if res.success:
            best_h = res.x
            best_c5 = objective(best_h)

    return best_h, best_c5, n_points
