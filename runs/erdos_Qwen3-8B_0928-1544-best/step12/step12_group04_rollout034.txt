import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    required_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        if len(h) != n_points:

    else:

    h = h * (required_sum / np.sum(h))
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h.copy()
    best_val = compute_c5(h)
    
    T_initial = 0.5
    T_final = 1e-4
    cooling_rate = 0.99
    T = T_initial
    iterations_per_temp = 1000
    time_start = time.time()
    
    while T > T_final and time.time() - time_start < budget_s:
        for _ in range(iterations_per_temp):
            perturbation = np.random.randn(n_points) * T
            h_try = h + perturbation
            h_clipped = np.clip(h_try, 0.0, 1.0)
            total = np.sum(h_clipped)
            if total == 0:
                h_clipped = np.random.rand(n_points) * 0.5
                total = np.sum(h_clipped)
            h_clipped = h_clipped * (required_sum / total)
            val_try = compute_c5(h_clipped)
            if val_try < best_val:
                h = h_clipped.copy()
                best_val = val_try

                if np.random.rand() < np.exp(-delta / T):
