            worst_indices = np.argsort([compute_c5(p) for p in population])[:int(population_size * 0.1)]
            for idx in worst_indices:
                h = np.random.rand(n_points) * 0.5
                h_sum = np.sum(h)
                if h_sum == 0:
                    h = np.random.rand(n_points) * 0.5
                h = h * (ones_count / h_sum)
                population[idx] = h.copy()
                velocities[idx] = np.zeros(n_points)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
