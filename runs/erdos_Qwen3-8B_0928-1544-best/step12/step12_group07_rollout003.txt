    best_h = population[0].copy()
    best_c5 = compute_c5(population[0], n_points)

    start_time = time.time()
    gen = 0

    while time.time() - start_time < budget_s:
        gen += 1

        fitness = [compute_c5(h, n_points) for h in population]
        sorted_indices = np.argsort(fitness)
        selected = [population[i] for i in sorted_indices[:30]]

        new_population = []
        for i in range(0, len(selected), 2):
            if i + 1 >= len(selected):
                break
            p1, p2 = selected[i], selected[i + 1]
            split = np.random.randint(1, n_points)
            c1, c2 = np.concatenate([p1[:split], p2[split:]]), np.concatenate([p2[:split], p1[split:]])
            new_population.append(c1)
            new_population.append(c2)

        for h in new_population:
            for _ in range(2):
                idx = np.random.randint(n_points)
                h[idx] += np.random.uniform(-0.1, 0.1)
                h[idx] = np.clip(h[idx], 0, 1)
            h_sum = np.sum(h)
            h = h * (target_sum / h_sum)
        new_population = [np.clip(h, 0, 1) for h in new_population]

        combined = population + new_population
        fitness_combined = [compute_c5(h, n_points) for h in combined]
        sorted_indices = np.argsort(fitness_combined)
        population = [combined[i] for i in sorted_indices[:60]]

        for h in population:
            c5 = compute_c5(h, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        if gen > 500:
            break

    return best_h, best_c5, n_points

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * (2.0 / n_points)
    return max_corr
