            c_pert_minus = compute_c5(h_pert_minus)

            grad[i] = (c_pert_plus - c_pert_minus) / (2 * eps)

        if np.abs(np.max(grad)) < 1e-5:
            break

        grad_norm = np.linalg.norm(grad)
        step_size = 0.3 / (1 + grad_norm)

        new_h = h.copy()
        new_h += -step_size * grad
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum == 0:
            continue
        new_h = new_h * (n_half / new_h_sum)
        c_new = compute_c5(new_h)
        if c_new < best_c5:
            h = new_h
            best_c5 = c_new

        if _ % 100 == 0:
            pert = 0.01 * (2 * np.random.rand(n_points) - 1)
            h = np.clip(h + pert, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
                h = h * (n_half / np.sum(h))
            else:
                h = h * (n_half / h_sum)

        if time.time() - T > budget_s:
            break

    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum == 0:
        h = np.random.rand(n_points) * 0.5
        h = h * (n_half / np.sum(h))
    else:
        h = h * (n_half / h_sum)
    best_c5 = compute_c5(h)
    return h, best_c5, n_points
