                off = np.clip(off, 0, 1)
                actual_sum = np.sum(off)
                if actual_sum != 0:
                    off = off * (target_sum / actual_sum)
                offspring[i] = off
        # Replace
        combined = population + offspring
        fitness_combined = [compute_c5(ind, n_points) for ind in combined]
        sorted_indices = np.argsort(fitness_combined)
        new_pop = [combined[i] for i in sorted_indices[:population_size]]
        population = new_pop
        # Update best
        current_best = population[0]
        current_c5 = compute_c5(current_best, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
    
    return best_h, best_c5, n_points

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    return max_val * (2.0 / n_points)
