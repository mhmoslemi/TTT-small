import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    population_size = 100
    elite_ratio = 0.2
    mutation_rate = 0.1
    tournament_size = 5
    max_generations = 200
    time_budget = budget_s

    if 'initial_h_values' in globals():

    else:
        h_pop = np.random.rand(population_size, n_points)
        h_pop = np.clip(h_pop, 0, 1)
        sums = np.sum(h_pop, axis=1)
        h_pop = h_pop * (n_points / 2) / sums[:, np.newaxis]

    best_h = h_pop[0]
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()

    for _ in range(max_generations):
        fitness = np.array([evaluate_erdos_solution(h) for h in h_pop])
        sorted_indices = np.argsort(fitness)
        elites = h_pop[sorted_indices[:int(elite_ratio * population_size)]]
        parents = []
        for _ in range(population_size - len(elites)):
            tournament = np.random.choice(sorted_indices, tournament_size, replace=False)
            winner = tournament[np.argmin(fitness[tournament])]
            parents.append(h_pop[winner])

        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1, p2 = parents[i], parents[i+1]
                child1 = p1 * 0.5 + p2 * 0.5
                child2 = p2 * 0.5 + p1 * 0.5
                offspring.extend([child1, child2])

        for j in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                offspring[j] += np.random.uniform(-0.1, 0.1, n_points)
                offspring[j] = np.clip(offspring[j], 0, 1)
                sums_off = np.sum(offspring[j])
                if sums_off == 0:
                    offspring[j] = np.random.rand(n_points)
                else:
                    offspring[j] = offspring[j] * (n_points / 2) / sums_off

        new_pop = np.vstack([elites, offspring])
        new_fitness = np.array([evaluate_erdos_solution(h) for h in new_pop])
        sorted_new = np.argsort(new_fitness)
        h_pop = new_pop[sorted_new[:population_size]]

        current_best = np.argmin(fitness)
        if evaluate_erdos_solution(h_pop[current_best]) < best_c5:
