            child1 = child1 * (target_sum / child1_sum)
            
            mut_indices = np.random.choice(n_points, int(n_points * 0.02), replace=False)
            child2[mut_indices] += np.random.uniform(-0.1, 0.1, size=len(mut_indices))
            child2 = np.clip(child2, 0.0, 1.0)
            child2_sum = np.sum(child2)
            child2 = child2 * (target_sum / child2_sum)
            
            offspring.append(child1)
            offspring.append(child2)
        
        combined = population + offspring
        combined_fitness = [compute_c5(indiv, n_points) for indiv in combined]
        sorted_combined = sorted(zip(combined_fitness, combined), key=lambda x: x[0])
        new_population = [indiv for _, indiv in sorted_combined[:60]]
        population = new_population
    
    return best_h, best_c5, n_points

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n_points)
