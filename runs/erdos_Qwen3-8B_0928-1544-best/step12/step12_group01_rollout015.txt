    population = [initial_scaled]
    for _ in range(max_tries - 1):
        h = np.random.rand(n_points) * (ones_count / n_points)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * (ones_count / n_points)

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = evaluate(population[0])
    start_time = time.time()

        new_population = []

            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
