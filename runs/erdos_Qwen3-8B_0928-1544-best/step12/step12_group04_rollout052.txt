            child1_sum = np.sum(child1)
            if child1_sum == 0:
                child1 = parent1.copy()
            else:
                child1 = child1 * (required_sum / child1_sum)
            # Mutate child2
            noise2 = np.random.normal(0, 0.1, n_points)
            child2 += noise2
            child2 = np.clip(child2, 0, 1)
            child2_sum = np.sum(child2)
            if child2_sum == 0:
                child2 = parent2.copy()
            else:
                child2 = child2 * (required_sum / child2_sum)
            new_offspring.append(child1)
            new_offspring.append(child2)
        new_offspring = new_offspring[:pop_size]
        combined = pop + new_offspring
        combined_sorted = sorted(combined, key=lambda x: evaluate(x))
        pop = combined_sorted[:pop_size]
        for h in pop:
            current_val = evaluate(h)
            if current_val < best_val:
                best_val = current_val
                best_h = h.copy()
    
    return best_h, best_val, n_points
