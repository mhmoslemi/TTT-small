import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 300  # Increased population size for better exploration
    F_start, F_end = 0.6, 1.2  # Adjusted F range
    CR_start, CR_end = 0.5, 1.0  # Adjusted CR range
    mutation_rate = 0.3  # Increased mutation rate for exploration

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        population = [initial_h_scaled]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()
    F = F_start
    CR = CR_start
    stagnation = 0

    while time.time() - start_time < budget_s:
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR or np.random.rand() < mutation_rate:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            current_c5 = objective(population[i])
            new_c5 = objective(u_scaled)
            if new_c5 < current_c5:
                population[i] = u_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u_scaled.copy()
                    stagnation = 0
            else:
                stagnation += 1
            if stagnation > 50:
                F = max(F_start, F - 0.01)
                CR = min(CR_end, CR + 0.01)
                stagnation = 0

    return best_h, best_c5, n_points
