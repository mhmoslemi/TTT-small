import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    pop_size = 100
    F_start, F_end = 0.8, 1.0
    CR_start, CR_end = 0.7, 1.0
    mutation_rate = 0.3
    stagnation_threshold = 15
    max_iter = 1000
    best_h = None
    best_c5 = 1e10

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        population = [initial_h_scaled]
        for _ in range(pop_size - 1):
            h = initial_h_scaled + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            if h_sum > 0:
                h_scaled = h * (ones_count / h_sum)
                population.append(h_scaled)
    else:
        population = []
        for _ in range(pop_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    start_time = time.time()
    F = F_start
    CR = CR_start
    stagnation = 0
    best_h = population[0].copy()
    best_c5 = objective(population[0])

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = mutant[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            new_c5 = objective(u_scaled)
            if new_c5 < objective(population[i]):
                population[i] = u_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u_scaled.copy()
                    stagnation = 0
            else:
                stagnation += 1
        if stagnation > stagnation_threshold:
            F = max(F_start, F - 0.01)
            CR = min(CR_end, CR + 0.01)
            stagnation = 0

    return best_h, best_c5, n_points
