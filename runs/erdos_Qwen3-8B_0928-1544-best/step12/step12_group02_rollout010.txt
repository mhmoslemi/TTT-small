import numpy as np

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2
    pop_size = 200
    max_iter = 1000
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    vel_scale = 0.1

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    if 'initial_h_values' in globals():

            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (ones_count / h_sum) if h_sum > 0 else np.random.rand(n_points)
            pop.append(h)
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = h * (ones_count / np.sum(h)) if np.sum(h) > 0 else np.random.rand(n_points)
