import time

def run(seed=42, budget_s=1100, **kwargs):

    dx = 2.0 / n_points

    elite_ratio = 0.2

        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx

    best_h = h_pop[0]

        if time.time() - start_time > budget_s:

        # Evaluate

                child[mask] += np.random.normal(0, 0.1, np.sum(mask))
