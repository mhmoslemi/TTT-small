            population.append(h_scaled)
    else:
        population = []
        for _ in range(50):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()

    population_size = 50
    F_start, F_end = 0.6, 0.9
    CR_start, CR_end = 0.8, 1.0
    mutation_rate = 0.2
    stagnation_threshold = 30

    F = F_start
    CR = CR_start
    stagnation = 0

    while time.time() - start_time < budget_s * 0.8:
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            current_c5 = objective(population[i])
            new_c5 = objective(u_scaled)
            if new_c5 < current_c5:
                population[i] = u_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u_scaled.copy()
                    stagnation = 0
            else:
                stagnation += 1
            if stagnation > stagnation_threshold:
                F = max(F_start, F - 0.01)
                CR = min(CR_end, CR + 0.01)
                stagnation = 0

    if time.time() - start_time < budget_s:
        print("Applying Nelder-Mead optimization...")
        def objective_nelder_mead(h_vec):
            h = np.clip(h_vec, 0.0, 1.0)
            h_sum = np.sum(h)
            if h_sum == 0:
                return np.inf
            h_scaled = h * (ones_count / h_sum)
            corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
            return np.max(corr) * dx

        initial_guess = best_h.copy()
        initial_guess += np.random.normal(0, 0.01, n_points)
        initial_guess = np.clip(initial_guess, 0.0, 1.0)
        initial_guess_sum = np.sum(initial_guess)
        if initial_guess_sum == 0:
            initial_guess = best_h.copy()
        else:
            initial_guess = initial_guess * (ones_count / initial_guess_sum)

        res = minimize(objective_nelder_mead, initial_guess, method='Nelder-Mead',
                       options={'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 100})
        best_h = res.x.copy()
        best_c5 = objective(best_h)

    return best_h, best_c5, n_points
