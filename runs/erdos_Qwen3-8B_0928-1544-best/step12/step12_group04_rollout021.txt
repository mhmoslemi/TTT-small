def run(seed=42, budget_s=1100, **kwargs):

            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h = h * (required_sum / h_sum)
            pop.append(h)
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h = h * (required_sum / h_sum)
            pop.append(h)
