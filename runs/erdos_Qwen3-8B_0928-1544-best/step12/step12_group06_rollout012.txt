import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = int(budget_s * 0.85)

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        h = h * (target_sum / h_sum)

    def compute_c5(h_val):
        corr = np.correlate(h_val, 1.0 - h_val, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = time.time()
    mutation_rate = 0.15
    mutation_scale = 0.07

    for _ in range(max_iter):
        current_c5 = compute_c5(h)
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_pert = h.copy()
            h_pert[i] += 1e-4
            h_pert = np.clip(h_pert, 0, 1)
            h_pert_sum = np.sum(h_pert)
            if h_pert_sum == 0:
                continue
            h_pert = h_pert * (target_sum / h_pert_sum)
            grad[i] = (compute_c5(h_pert) - current_c5) / 1e-4

        if np.abs(np.max(grad)) < 1e-5:
            break

        neg_grad = -grad
        idx = np.argmax(neg_grad)
        step_size = 0.2
        new_h = h.copy()
        new_h[idx] += step_size
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum == 0:
            continue
        new_h = new_h * (target_sum / new_h_sum)
        c_new = compute_c5(new_h)
        if c_new < best_c5:
            h = new_h
            best_c5 = c_new

        if np.random.rand() < mutation_rate:
            mutate_h = h.copy()
            for i in range(n_points):
                mutate_h[i] += np.random.normal(0, mutation_scale)
            mutate_h = np.clip(mutate_h, 0, 1)
            mutate_h_sum = np.sum(mutate_h)
            if mutate_h_sum == 0:
                continue
            mutate_h = mutate_h * (target_sum / mutate_h_sum)
            c_mut = compute_c5(mutate_h)
            if c_mut < best_c5:
                h = mutate_h
                best_c5 = c_mut

        if time.time() - T > budget_s * 0.9:
            break

    return h, best_c5, n_points
