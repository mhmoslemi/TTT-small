import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    c = n_points / 2

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        h = h * (c / h_sum)

    def project(h, c):
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        if np.isclose(current_sum, c):
            return h
        deficit = c - current_sum
        if deficit > 0:
            non_full = np.where(h < 1.0)[0]
            if not non_full.size:
                return h
            total_available = np.sum(1.0 - h[non_full])
            h[non_full] += deficit * (1.0 - h[non_full]) / total_available
        else:
            surplus = current_sum - c
            non_zero = np.where(h > 0.0)[0]
            if not non_zero.size:
                return h
            total_available = np.sum(h[non_zero])
            h[non_zero] -= surplus * h[non_zero] / total_available
        return h

    h = project(h, c)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)

    learning_rate = 0.1
    start_time = time.time()
    iteration = 0

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + k < n_points:
                gradient[i] += 1.0 - h[i + k]
            if 0 <= i - k < n_points:
                gradient[i] -= h[i - k]
        h_new = h - learning_rate * gradient
        h = project(h_new, c)
        current_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(current_corr) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        iteration += 1
        if iteration % 100 == 0:
            learning_rate *= 0.99

    return best_h, best_c5, n_points
