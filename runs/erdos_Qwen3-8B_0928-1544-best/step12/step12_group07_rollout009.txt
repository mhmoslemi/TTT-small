import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    dx = 2.0 / n_points
    ones_count = n_points // 2
    if 'initial_h_values' in globals():

        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum)
    best_h = h.copy()
    best_c5 = 1.0
    
    def obj(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - ones_count})
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    points = [h]
    for _ in range(4):
        pt = h + np.random.uniform(-0.05, 0.05, n_points)
        pt = np.clip(pt, 0, 1)
        s = np.sum(pt)
        pt = pt * (ones_count / s) if s > 1e-6 else pt
        points.append(pt)
    
    for p in points:
        res = minimize(obj, p, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-6, options={'maxiter': 500})
        if res.success and res.fun < best_c5:
