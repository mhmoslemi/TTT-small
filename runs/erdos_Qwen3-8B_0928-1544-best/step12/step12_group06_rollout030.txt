import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    m = n_points / 2
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:

        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h = h * (m / h_sum)
    
    start_time = time.time()
    max_iter = int(budget_s / 2)  # Limit to avoid excessive computation
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr[n_points-1:2*n_points-1]
        worst_k = np.argmax(max_corr)
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            A[i, (i + worst_k) % n_points] = 1
        
        h_qp = cp.Variable(n_points)
        obj = cp.Maximize(cp.quad_form(h_qp, A))
        constraints = [cp.sum(h_qp) == m, 0 <= h_qp, h_qp <= 1]
        prob = cp.Problem(obj, constraints)
        try:

    c5_bound = np.max(corr_final[n_points-1:2*n_points-1]) * dx
