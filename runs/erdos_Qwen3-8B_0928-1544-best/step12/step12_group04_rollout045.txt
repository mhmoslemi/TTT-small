import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    required_sum = n_points / 2.0
    pop_size = 200
    if 'initial_h_values' in globals():
        initial = initial_h_values
        h0 = initial * (required_sum / np.sum(initial))
        pop = [h0]
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points) * 0.5
            h = h * (required_sum / np.sum(h))
            pop.append(h)
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points) * 0.5
            h = h * (required_sum / np.sum(h))
            pop.append(h)
    
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = pop[0].copy()
    best_val = obj(pop[0])
    start = time.time()
    
    while time.time() - start < budget_s:
        fitnesses = [obj(h) for h in pop]
        sorted_indices = np.argsort(fitnesses)
        num_parents = int(pop_size * 0.5)
        parent_indices = sorted_indices[:num_parents]
        parent_pop = [pop[i] for i in parent_indices]
        offspring = []
        
        for i in range(0, num_parents, 2):
            p1 = parent_pop[i]
            p2 = parent_pop[i+1] if i+1 < num_parents else parent_pop[i]
            crossover_point = np.random.randint(1, n_points)
            child1 = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
            child2 = np.concatenate([p2[:crossover_point], p1[crossover_point:]])
            
            if np.random.rand() < 0.5:
                noise = np.random.normal(0, 0.1, n_points)
                child1 += noise
                child1 = np.clip(child1, 0.0, 1.0)
                child1_sum = np.sum(child1)
                if child1_sum > 0:
                    child1 = child1 * (required_sum / child1_sum)
                else:
                    child1 = p1.copy()
            
            if np.random.rand() < 0.5:
                noise = np.random.normal(0, 0.1, n_points)
                child2 += noise
                child2 = np.clip(child2, 0.0, 1.0)
                child2_sum = np.sum(child2)
                if child2_sum > 0:
                    child2 = child2 * (required_sum / child2_sum)
                else:
                    child2 = p2.copy()
            
            offspring.append(child1)
            offspring.append(child2)
        
        new_pop = []
        for _ in range(pop_size):
            if offspring:
                new_pop.append(offspring.pop(0))
            else:
                new_pop.append(pop[0].copy())
        
        pop = new_pop
        current_best = min(pop, key=obj)
        current_val = obj(current_best)
        if current_val < best_val:
            best_val = current_val
            best_h = current_best.copy()
    
    return best_h, best_val, n_points
