        for _ in range(500):  # Reduced iterations per temperature for time efficiency
            noise = T * np.random.randn(n_points)
            h_try = h_current + noise
            h_try = np.clip(h_try, 0, 1)
            h_sum = np.sum(h_try)
            if h_sum == 0:
                h_try = h_current.copy()
            else:
                h_try = h_try * required_sum / h_sum
            new_obj = compute_c5(h_try)
            if new_obj < best_obj:
                best_h = h_try.copy()
                best_obj = new_obj
            else:
                prob = np.exp((best_obj - new_obj) / T)
                if np.random.rand() < prob:
                    h_current = h_try.copy()
                    if new_obj < best_obj:
                        best_h = h_try.copy()
                        best_obj = new_obj
        T *= cooling_rate
        if T < T_final:
            break
    return best_h, best_obj, n_points
