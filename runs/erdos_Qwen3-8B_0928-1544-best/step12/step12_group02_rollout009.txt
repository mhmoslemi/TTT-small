from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h_initial = np.clip(initial_h_values, 0, 1)
        h_initial_sum = np.sum(h_initial)
        if h_initial_sum == 0:
            h_initial = np.random.rand(n_points)
            h_initial_sum = np.sum(h_initial)
        h_initial_scaled = h_initial * (ones_count / h_initial_sum)
    else:
        h_initial = np.zeros(n_points)
        h_initial[:ones_count] = 1.0
        h_initial_sum = ones_count
        h_initial_scaled = h_initial.copy()

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.sum((corr * dx) ** 2)

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result = minimize(
        objective,
        h_initial_scaled,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-5,
        options={'maxiter': 500}
    )
    elapsed_time = time.time() - start_time

    best_h = result.x
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    max_overlap = np.max(corr * dx)

    return best_h, max_overlap, n_points
