import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    resolution_schedule = [10, 20, 35, 70]
    initial_temp = 0.5
    final_temp = 0.1
    init_learning_rate = 0.1
    decay_rate = 0.999
    noise_scale = 0.1

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        h = h * (ones_count / np.sum(h))
    else:
        h = np.random.rand(n_points)
        h = h * (ones_count / np.sum(h))

    def obj_func(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def latent_to_h(a):
        return 1 / (1 + np.exp(-a)) * (ones_count / np.sum(1 / (1 + np.exp(-a))))

    def h_to_latent(h):
        return np.log(h / (1 - h + 1e-12))

    best_h = h.copy()
    best_c5 = obj_func(h)
    current_h = h.copy()
    current_a = h_to_latent(current_h)
    current_temp = initial_temp
    current_lr = init_learning_rate
    time_start = time.time()
    iter_count = 0

    for res in resolution_schedule:

            current_h = current_h[:res]
            current_a = h_to_latent(current_h)
        n = res

            a_trial = current_a + perturb
