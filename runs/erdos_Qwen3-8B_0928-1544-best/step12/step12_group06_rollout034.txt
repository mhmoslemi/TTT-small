def run(seed=42, budget_s=1100, **kwargs):

    mutation_rate = 0.1

    max_gen = int(budget_s // 2)

            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h = h * (n_points / 2) / h_sum
            pop.append(h)
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h = h * (n_points / 2) / h_sum
            pop.append(h)

    def compute_c5(h_val):

        return np.max(corr) * dx

    best_h = pop[0]
    best_c5 = compute_c5(pop[0])
    start_time = time.time()

    while time.time() - start_time < budget_s:
        scores = [compute_c5(h) for h in pop]

        elite_count = int(elitism_ratio * pop_size)
