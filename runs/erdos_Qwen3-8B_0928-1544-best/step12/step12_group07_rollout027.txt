            initial_h = initial_h * (target_sum / initial_h_sum)
        population = [initial_h.copy() for _ in range(n_population)]
    else:
        population = []
        for _ in range(n_population):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            h = h * (target_sum / h_sum)
            population.append(h)
    
    start_time = time.time()
    
    while generations < max_generations and time.time() - start_time < budget_s:
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            c5 = max_corr * (2.0 / n_points)
            fitness.append(c5)
        
        sorted_indices = np.argsort(fitness)
        selected_indices = sorted_indices[:int(n_population / 2)]
        selected_parents = [population[i] for i in selected_indices]
        
        offspring = []
        for i in range(len(selected_parents)):
            parent1 = selected_parents[i]
            parent2 = selected_parents[np.random.randint(len(selected_parents))]
            if np.random.rand() < crossover_rate:
                split_point = np.random.randint(1, n_points)
                child = np.concatenate([parent1[:split_point], parent2[split_point:]])
            else:
                child = parent1.copy()
            
            if np.random.rand() < mutation_rate:
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        noise = np.random.uniform(-mutation_strength, mutation_strength)
                        child[j] += noise
                        child[j] = np.clip(child[j], 0.0, 1.0)
                current_sum = np.sum(child)
                if current_sum < target_sum:
                    max_possible = np.sum(1.0 - child)
                    if max_possible > 0:
                        delta = target_sum - current_sum
                        increase = delta / max_possible
                        child += increase
                        child = np.clip(child, 0.0, 1.0)
                elif current_sum > target_sum:
                    max_possible = np.sum(child)
                    if max_possible > 0:
                        delta = current_sum - target_sum
                        decrease = delta / max_possible
                        child -= decrease
                        child = np.clip(child, 0.0, 1.0)
            offspring.append(child)
        
        new_population = selected_parents + offspring
        new_population = new_population[:n_population]
        population = new_population.copy()
        
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            c5 = max_corr * (2.0 / n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        generations += 1
    
    return best_h, best_c5, n_points
