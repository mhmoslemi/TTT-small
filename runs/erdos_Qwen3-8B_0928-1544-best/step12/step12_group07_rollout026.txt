            m = c.copy()
            m[mut] += delta[mut]
            m = np.clip(m, 0, 1)
            s = np.sum(m)
            m *= target_sum / s if s != 0 else 1
            m = np.clip(m, 0, 1)
            mutated.append(m)
        combined = population + mutated
        corr_list = []
        for h in combined:
            corr = np.correlate(h, 1 - h, mode='full')
            corr_list.append(np.max(corr) * (2.0 / n_points))
        sorted_idx = np.argsort(corr_list)
        population = [combined[i] for i in sorted_idx[:60]]
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            current_max = np.max(corr) * (2.0 / n_points)
            if current_max < best_c5:
                best_c5 = current_max
                best_h = h.copy()
    return best_h, best_c5, n_points
