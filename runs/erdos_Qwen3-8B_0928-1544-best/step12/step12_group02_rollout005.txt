import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    pop_size = 200
    inertia_start, inertia_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    vel_clip = 0.5
    max_iter = int(budget_s / 0.01)  # 100 iterations per 1s

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        pop = [initial_h_scaled]
        for _ in range(pop_size - 1):
            h = initial_h_scaled + np.random.rand(n_points) * 0.1
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / np.sum(h))
            pop.append(h_scaled)
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / h_sum)
            pop.append(h_scaled)

    # Initialize velocities
    velocities = np.random.rand(pop_size, n_points) * 0.1

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = pop[0]
    best_c5 = objective(pop[0])
    start_time = time.time()
    inertia = inertia_start
    stagnation = 0
    best_global = pop[0]

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] + c1 * r1 * (pop[i] - best_global) + c2 * r2 * (pop[i] - best_global)
            vel = np.clip(vel, -vel_clip, vel_clip)
            velocities[i] = vel
            # Update position
            new_pos = pop[i] + vel
            new_pos = np.clip(new_pos, 0, 1)
            new_pos_sum = np.sum(new_pos)
            if new_pos_sum == 0:
                continue
            new_pos_scaled = new_pos * (ones_count / new_pos_sum)
            # Evaluate
            curr_c5 = objective(pop[i])
            new_c5 = objective(new_pos_scaled)
            if new_c5 < curr_c5:
                pop[i] = new_pos_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_pos_scaled.copy()
                if new_c5 < objective(best_global):
                    best_global = new_pos_scaled.copy()
                stagnation = 0
            else:
                stagnation += 1
            if stagnation > 200:
                inertia = max(inertia_start, inertia - 0.001)
                stagnation = 0
        if time.time() - start_time > budget_s:
            break
    return best_h, best_c5, n_points
