    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_scaled = initial_h * (ones_count / np.sum(initial_h))
        population = [initial_h_scaled]
        for _ in range(population_size - 1):
            h = initial_h_scaled.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            if h_sum == 0:
                continue
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)

    def compute_c5(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h_clipped)
        if h_sum == 0:
            return float('inf')
        scaling = (n_points / 2) / h_sum
        h_scaled = h_clipped * scaling
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    stagnation = 0

    while time.time() - start_time < budget_s:
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            current_c5 = compute_c5(population[i])
            new_c5 = compute_c5(u_scaled)
            if new_c5 < current_c5:
                population[i] = u_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u_scaled.copy()
                    stagnation = 0
            else:
                stagnation += 1
        if stagnation > 30:
            result = minimize(compute_c5, best_h, method='Nelder-Mead', options={'maxiter': max_nelder_iter})
            best_h = result.x
            best_c5 = compute_c5(best_h)
            stagnation = 0
    return best_h, best_c5, n_points
