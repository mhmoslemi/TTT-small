            h_scaled = h * (ones_count / h_sum) if h_sum != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(swarm_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            h_scaled = h * (ones_count / h_sum) if h_sum != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
            population.append(h_scaled)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
