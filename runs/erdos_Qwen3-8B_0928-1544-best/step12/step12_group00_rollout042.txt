import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    target_sum = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.random.rand(n_points) * 0.5
        initial_h = initial_h * (target_sum / np.sum(initial_h))
    particles = [initial_h.copy() for _ in range(50)]
    p_bests = [p.copy() for p in particles]
    best_h = p_bests[0].copy()
    best_c5 = float('inf')
    w, c1, c2, max_v = 0.7, 2.0, 2.0, 0.1
    max_iter = 500
    stagnation = 0
    start_time = time.time()

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(len(particles)):
            curr_c5 = objective(particles[i])
            if curr_c5 < objective(p_bests[i]):
                p_bests[i] = particles[i].copy()
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    best_h = particles[i].copy()
            r1, r2 = np.random.rand(), np.random.rand()
            velocities = w * (p_bests[i] - particles[i]) * c1 * r1 + (best_h - particles[i]) * c2 * r2
            particles[i] += velocities
            particles[i] = np.clip(particles[i], 0, 1)
            sum_h = np.sum(particles[i])
            if sum_h:
                particles[i] *= (target_sum / sum_h)
            else:
                particles[i] = np.random.rand(n_points) * 0.5
                particles[i] *= (target_sum / np.sum(particles[i]))
        stagnation += 1
        if stagnation > 50:
            w = max(0.4, w - 0.01)
            stagnation = 0
    best_h = np.clip(best_h, 0, 1)
    best_h *= (target_sum / np.sum(best_h))
    return best_h, best_c5, n_points
