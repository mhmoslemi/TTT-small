def obj(h, n_points):
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    required_sum = n_points / 2
    pop_size = 300
    
    if 'initial_h_values' in globals():
        initial = initial_h_values
        h0 = initial * (required_sum / np.sum(initial))
        pop = [h0]
        perturb_size = 0.2
        for _ in range(pop_size - 1):
            perturb = np.random.rand(n_points) * perturb_size
            h = initial + perturb
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h = h * (required_sum / np.sum(h))
            pop.append(h)
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h = h * (required_sum / h_sum)
            pop.append(h)
    
    pop = np.array(pop, dtype=np.float64)
    velocities = np.zeros((pop_size, n_points), dtype=np.float64)
    pbest = pop.copy()
    pbest_vals = np.array([obj(h, n_points) for h in pop])
    gbest = pop[0].copy()
    gbest_val = pbest_vals[0]
    best_h = gbest.copy()
    
    w, phi_p, phi_g = 0.7, 2.0, 2.0
    max_vel = 0.2
    start = time.time()
    
    while time.time() - start < budget_s:
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            vel_update = phi_p * r1 * (pbest[i] - pop[i]) + phi_g * r2 * (gbest - pop[i])
            velocities[i] = w * velocities[i] + vel_update
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            new_pos = pop[i] + velocities[i]
            
            # Reflect boundaries
            new_pos[new_pos < 0] = -new_pos[new_pos < 0]
            new_pos[new_pos > 1] = 2 - new_pos[new_pos > 1]
            velocities[i][new_pos < 0] = -velocities[i][new_pos < 0]
            velocities[i][new_pos > 1] = -velocities[i][new_pos > 1]
            new_pos = np.clip(new_pos, 0.0, 1.0)
            
            new_sum = np.sum(new_pos)
            if new_sum == 0:
                new_pos = pop[i].copy()
            else:
                new_pos = new_pos * (required_sum / new_sum)
            
            new_val = obj(new_pos, n_points)
            if new_val < pbest_vals[i]:
                pbest[i] = new_pos.copy()
                pbest_vals[i] = new_val
                if new_val < gbest_val:
                    gbest = new_pos.copy()
                    gbest_val = new_val
                    best_h = new_pos.copy()
            pop[i] = new_pos
        
        if time.time() - start > budget_s - 1:
            break
    
    return best_h, gbest_val, n_points
