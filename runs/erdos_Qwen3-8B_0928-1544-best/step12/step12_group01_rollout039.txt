import time

        h_sum = np.sum(h)

            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
        h_scaled = h * (ones_count / h_sum)

    population = [h_scaled]

        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)

            h = np.random.rand(n_points) * 0.5
        h_scaled = h * (ones_count / h_sum)
        population.append(h_scaled)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()

    start_time = time.time()
    F = 0.8
    CR = 0.9

            vec_a, vec_b, vec_c = population[a], population[b], population[c]
            mutant = vec_a + F * (vec_b - vec_c)
            child = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    child[j] = mutant[j]
            child = np.clip(child, 0.0, 1.0)
            child_sum = np.sum(child)
            if child_sum == 0:
                continue
            child_scaled = child * (ones_count / child_sum)
            current_c5 = objective(population[i])
            new_c5 = objective(child_scaled)
