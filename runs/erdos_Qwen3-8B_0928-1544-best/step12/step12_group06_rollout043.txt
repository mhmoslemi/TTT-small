def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    n_half = n_points // 2

    P = 100  # Population size
    max_gen = 200  # Max generations
    mutation_rate = 0.1

    def compute_c5(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = None

    # Initialize population
    population = []
    if initial_h is not None:
        h = initial_h.copy()
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        else:
            h = h * (n_half / h_sum)
        population.append(h)
        for _ in range(P - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h = h * (n_half / h_sum)
            population.append(h)
    else:
        for _ in range(P):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h = h * (n_half / h_sum)
            population.append(h)

    best_h = population[0]
    best_c5 = compute_c5(population[0])
    T = time.time()

    for gen in range(max_gen):
        if time.time() - T > budget_s:
            break
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        if compute_c5(population[best_idx]) < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = compute_c5(population[best_idx])

        # Elitism: keep top 20%
        sorted_indices = np.argsort(fitness)
        elites = population[:int(P * 0.2)]
        new_population = elites.copy()

        # Select parents
        parents = []
        for _ in range(P - len(elites)):
            tournament = np.random.choice(P, size=5, replace=False)
            winner_idx = tournament[np.argmin([fitness[i] for i in tournament])]
            parents.append(population[winner_idx])
