        pop = []
        for _ in range(200):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / h_sum)
            pop.append(h_scaled)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = pop[0]
    best_c5 = objective(pop[0])
    start_time = time.time()
    F, CR = 0.6, 0.6
    stagnation = 0
    max_iterations = 1000

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s * 0.7:
            break
        for i in range(len(pop)):
            a, b, c = np.random.choice(len(pop), 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            trial = pop[i] * (1 - CR) + mutant * CR
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                continue
            trial_scaled = trial * (ones_count / trial_sum)
            curr_c5 = objective(pop[i])
            new_c5 = objective(trial_scaled)
            if new_c5 < curr_c5:
                pop[i] = trial_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = trial_scaled.copy()
                stagnation = 0
            else:
                stagnation += 1
        if stagnation > 200:
            F = min(0.9, F + 0.01)
            CR = min(0.95, CR + 0.01)
            stagnation = 0

    # SQP refinement
    def objective_grad(h):
        grad = np.zeros_like(h)
        eps = 1e-6
        for i in range(len(h)):
            h_p = h.copy()
            h_p[i] += eps
            h_m = h.copy()
            h_m[i] -= eps
            grad[i] = (objective(h_p) - objective(h_m)) / (2 * eps)
        return grad

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(objective, best_h, method='SLSQP', jac=objective_grad, bounds=bounds, options={'maxiter': 100})
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
