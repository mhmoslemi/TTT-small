        if time.time() - start_time > budget_s:
            return initial_h, np.correlate(initial_h, 1 - initial_h, mode='full')[0] * dx, n_points
        h_opt = np.zeros(n_points)
        for i in range(n_points):
            h_opt[i] = (s.value[i+1] - s.value[i]) / dx
    except:
        h_opt = initial_h.copy()
    
    def objective(h):
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            return np.inf
        h_norm = h * (ones_count / h_sum)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(objective, h_opt, method='L-BFGS-B', bounds=bounds, options={'maxiter': 150})
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
