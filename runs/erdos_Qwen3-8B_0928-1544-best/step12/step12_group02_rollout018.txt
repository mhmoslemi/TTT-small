            personal_c5 = objective(personal_bests[i])
            if current_c5 < personal_c5:
                personal_bests[i] = pop[i].copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    global_best = pop[i].copy()

    return global_best, best_c5, n_points
