import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    initial_h_sum = np.sum(initial_h)
    initial_h = initial_h * (ones_count / initial_h_sum)
    
    def cost_func(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraints(h):
        return np.array([np.sum(h) - ones_count])
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = opt.minimize(cost_func, initial_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraints}, tol=1e-7, options={'maxiter': 1000})
    
    best_h = result.x
    best_c5 = cost_func(best_h)
    
    return best_h, best_c5, n_points
