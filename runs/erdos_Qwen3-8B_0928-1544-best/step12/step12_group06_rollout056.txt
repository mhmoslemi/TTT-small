import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    n_half = n_points // 2
    dx = 2.0 / n_points
    max_iter = int(budget_s / 3)

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h = h * (n_half / h_sum)

    def compute_c5(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = time.time()

    for _ in range(max_iter):
        grad = np.zeros(n_points)
        best_k = np.argmax(np.correlate(h, 1 - h, mode='full'))
        for i in range(n_points):
            h_pert = h.copy()
            h_pert[i] += 1e-4
            h_pert = np.clip(h_pert, 0, 1)
            h_pert_sum = np.sum(h_pert)
            if h_pert_sum == 0:
                c_pert = np.inf
            else:
                h_pert = h_pert * (n_half / h_pert_sum)
                c_pert = compute_c5(h_pert)
            grad[i] = (c_pert - best_c5) / 1e-4

        if np.abs(np.max(grad)) < 1e-5:
            break

        neg_grad = -grad
        idx = np.argmax(neg_grad)
        step_size = 0.15 / (1.0 + np.sqrt(np.sum(grad**2)))
        new_h = h.copy()
        new_h[idx] += step_size
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum == 0:
            continue
        new_h = new_h * (n_half / new_h_sum)
        c_new = compute_c5(new_h)
        if c_new < best_c5:
