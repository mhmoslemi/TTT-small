import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    initial_guesses = []
    
    if 'initial_h_values' in globals():
        initial_guesses.append(initial_h_values.copy())
    # Add uniform guess
    initial_guesses.append(np.full(n_points, 0.5))
    # Add 8 random guesses
    for _ in range(8):
        h = np.random.rand(n_points) * 0.5
        scale = target_sum / np.sum(h)
        h = h * scale
        h = np.clip(h, 0, 1)
        initial_guesses.append(h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    for guess in initial_guesses:
        h = guess.copy()
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
        scale = target_sum / h_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        if time.time() - start_time >= budget_s:
            break
        try:
            res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint})
            if res.success and res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x.copy()
        except:
            pass
    
    if best_h is None:
        best_h = np.random.rand(n_points) * 0.5
        best_h_sum = np.sum(best_h)
        best_h = best_h * (target_sum / best_h_sum)
        best_h = np.clip(best_h, 0, 1)
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
