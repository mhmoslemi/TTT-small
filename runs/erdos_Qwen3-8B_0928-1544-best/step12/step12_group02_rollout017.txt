import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:ones_count] = 1.0
        h = h * (ones_count / np.sum(h))
    h = np.clip(h, 0, 1)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.sum(corr ** 2)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def compute_c5(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    method = 'SLSQP'
    options = {'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}

    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    for _ in range(50):
        result = minimize(objective, h, method=method, bounds=bounds, constraints=cons, options=options)
        if not result.success:
            break

        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        h = h_clipped * (ones_count / sum_clipped) if sum_clipped != 0 else h_clipped
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n_points
