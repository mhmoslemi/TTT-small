import time

from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2
    budget = budget_s

    initial_h = initial_h_values.copy()

    def evaluate(h):
        h = np.clip(h, 0, 1)
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def rescale(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h) + 0.5
        return h * (ones_count / s)
    
    def optimize():
        population = [rescale(scaled_initial)]
        for _ in range(199):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            scaled = h * (ones_count / h_sum)
            population.append(scaled)
        best_h = scaled_initial.copy()
        best_c5 = evaluate(best_h)
        elapsed = 0
        F, CR = 0.9, 0.95
        for _ in range(500):
            if elapsed > budget:
                break
            for i in range(200):
                a, b, c = np.random.choice(200, 3, replace=False)
                v = population[a] + F * (population[b] - population[c])
                u = population[i].copy()

                    if np.random.rand() < CR:
                        u[j] = v[j]
                u = np.clip(u, 0, 1)

                        best_h = scaled.copy()

            elapsed = time.time() - start_time
