def sqp_phase(h_initial, n_points):
    dx = 2.0 / n_points
    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        j = max_index
        k0 = j - (n_points -1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            # term1: i +k0
            if 0 <= i + k0 < n_points:
                grad[i] += 1 - h[i +k0]
            # term2: i -k0
            if 0 <= i - k0 < n_points:
                grad[i] -= h[i -k0]
        return grad

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
    bounds = [(0, 1) for _ in range(n_points)]

    result = minimize(objective, h_initial, method='SLSQP', jac=gradient, bounds=bounds, constraints=cons, tol=1e-6)
    return result.x
