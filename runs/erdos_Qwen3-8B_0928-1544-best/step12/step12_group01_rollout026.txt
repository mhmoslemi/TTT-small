def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * (2.0 / n_points)

def upscale(h, old_n, new_n):
    return np.kron(h, np.ones((new_n // old_n,)))[:new_n]
