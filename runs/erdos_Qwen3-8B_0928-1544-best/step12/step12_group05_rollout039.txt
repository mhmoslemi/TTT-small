            return float('inf')
        h_scaled = h * (ones_count / h_sum)
        return objective(h_scaled)
    
    bounds = Bounds(0.0, 1.0)
    result = minimize(local_objective, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-6)
    best_h = result.x
    best_c5 = objective(best_h * (ones_count / np.sum(best_h)))
    
    return best_h, best_c5, n_points
