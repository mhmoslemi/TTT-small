import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    required_sum = 70.0
    h = initial_h_values.copy()
    h = h * (required_sum / np.sum(h))
    h += 0.1 * np.random.randn(n_points)
    h = np.clip(h, 0, 1)
    h = h * (required_sum / np.sum(h))
    best_h = h.copy()
    best_val = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    T = 0.5
    T_final = 1e-4
    cooling_rate = 0.99
    start_time = time.time()
    while T > T_final and time.time() - start_time < budget_s:
        for _ in range(1000):
            h_try = h + T * np.random.randn(n_points)
            h_try = np.clip(h_try, 0, 1)
            h_try_sum = np.sum(h_try)
            if h_try_sum == 0:
                h_try = h.copy()
            else:
                h_try = h_try * (required_sum / h_try_sum)
            new_val = np.max(np.correlate(h_try, 1 - h_try, mode='full')) * (2.0 / n_points)
            if new_val < best_val or np.random.rand() < np.exp((best_val - new_val)/T):
                h = h_try.copy()
                best_val = new_val
        T *= cooling_rate
    return best_h, best_val, n_points
