                h = np.random.rand(n_points)
            h_scaled = h * (n_points / 2) / h_sum
            pop.append(h_scaled)
        velocities = [np.random.rand(n_points) * 0.1 for _ in pop]

    pbest = [h.copy() for h in pop]
    pbest_values = [evaluate_erdos_solution(h) for h in pop]
    gbest = pop[np.argmin(pbest_values)]
    gbest_value = min(pbest_values)

    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    start_time = time.time()
    iteration = 0
    while time.time() - start_time < budget_s:
        w = w_start - (w_start - w_end) * (iteration / 2000)
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest[i] - pop[i]) + c2 * r2 * (gbest - pop[i])
            pop[i] += velocities[i]
            pop[i] = np.clip(pop[i], 0, 1)
            h_sum = np.sum(pop[i])
            if h_sum == 0:
                continue
            pop[i] = pop[i] * (n_points / 2) / h_sum
            curr = evaluate_erdos_solution(pop[i])
            if curr < pbest_values[i]:
                pbest[i] = pop[i].copy()
                pbest_values[i] = curr
                if curr < gbest_value:
                    gbest = pop[i].copy()
                    gbest_value = curr
        iteration += 1
    return gbest, gbest_value, n_points
