                temp_h_sum = temp_h.sum()
                if temp_h_sum == 0:
                    continue
                scale = required_sum / temp_h_sum
                temp_h *= scale
                temp_h = np.clip(temp_h, 0, 1)
                c = compute_c5(temp_h)
                if c < best_c:
                    best_v = v
                    best_c = c
            if best_c < compute_c5(h):
                h[i] = best_v
                improved = True
        if not improved:
            h += np.random.uniform(-0.03, 0.03, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            scale = required_sum / h_sum
            h *= scale
            h = np.clip(h, 0, 1)
        if compute_c5(h) < best_c5:
            best_h = h.copy()
            best_c5 = compute_c5(h)
        iteration += 1
        
        # Early termination if no improvement
        if iteration > 1000 and not improved:
            break
    
    return best_h, best_c5, n_points
