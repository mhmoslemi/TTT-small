import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    required_sum = n_points / 2
    pop_size = 300
    if 'initial_h_values' in globals():
        initial = initial_h_values
        h0 = initial * (required_sum / np.sum(initial))
        pop = [h0]
        for _ in range(pop_size - 1):
            h = initial + np.random.rand(n_points) * 0.1
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h = h * (required_sum / h_sum)
            pop.append(h)
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h = h * (required_sum / h_sum)
            pop.append(h)
    velocities = np.zeros((pop_size, n_points))
    w, phi_p, phi_g = 0.7, 2.0, 2.0
    max_velocity = 0.2
    best_h = pop[0].copy()
    best_val = np.max(np.correlate(pop[0], 1 - pop[0], mode='full')) * (2.0 / n_points)
    pbest = [h.copy() for h in pop]
    pbest_vals = [np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points) for h in pop]
    gbest = best_h.copy()
    gbest_val = best_val
    start = time.time()
    while time.time() - start < budget_s:
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            vel_update = phi_p * r1 * (pbest[i] - pop[i]) + phi_g * r2 * (gbest - pop[i])
            velocities[i] = w * velocities[i] + vel_update

            if np.any(mask):

            new_sum = np.sum(new_pos)
            if new_sum == 0:
                new_pos = pop[i].copy()
            else:
                new_pos *= required_sum / new_sum
