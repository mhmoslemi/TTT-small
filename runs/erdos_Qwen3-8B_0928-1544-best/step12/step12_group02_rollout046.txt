            grad[j] = (obj_plus - obj_minus) / (2 * eps)
        return grad
    result = minimize(objective, best_h, method='SLSQP', jac=grad_func, bounds=bounds, constraints=cons, options={'maxiter': 100})
    if result.success:
        best_h_result = result.x
        best_c5_result = result.fun
    else:
        best_h_result = best_h
        best_c5_result = best_c5
    return best_h_result, best_c5_result, n_points
