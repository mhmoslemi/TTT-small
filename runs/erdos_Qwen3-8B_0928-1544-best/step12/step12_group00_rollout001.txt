import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    initial_h_sum = np.sum(initial_h)
    initial_h_scaled = initial_h * (ones_count / initial_h_sum)
    
    # Convert h to a matrix for SDP relaxation
    def problem_setup():
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1]
        constraints.append(cp.sum(h) == ones_count)
        obj = cp.Minimize(cp.Max(cp.sum(h * (1 - cp.Variable(n_points)), axis=0)))
        return h, constraints, obj
    
    h, constraints, obj = problem_setup()
    prob = cp.Problem(obj, constraints)
    start_time = time.time()
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, max_iters=1000)
    except:
        pass
    
    # Fallback to initial guess if SDP fails
    h_values = initial_h_scaled.copy() if prob.status != 'OPTIMAL' else h.value
    h_values = np.clip(h_values, 0.0, 1.0)
    h_values = h_values * (ones_count / np.sum(h_values))
    
    # Compute C5
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    c5_bound = np.max(corr) * (2.0 / n_points)
    return h_values, c5_bound, n_points
