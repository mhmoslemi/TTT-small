from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points // 2
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.full(n_points, 0.5)
        noise = np.random.normal(0, 0.05, n_points)
        initial_h += noise
        initial_h = np.clip(initial_h, 0, 1)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def repair(p):
        p = np.clip(p, 0, 1)
        s = np.sum(p)
        delta = (target_sum - s) / n_points
        p += delta
        p = np.clip(p, 0, 1)
        return p
    
    particles = []
    for _ in range(10):
        p = initial_h.copy()
        noise = np.random.normal(0, 0.05, n_points)
        p += noise
        p = repair(p)
        particles.append(p)
    for _ in range(20):
        p = np.sin(np.linspace(0, np.pi, n_points)) * 0.5 + 0.5

        p = repair(p)
        particles.append(p)
    for _ in range(20):
        p = (np.random.rand(n_points) * 2 - 1) * 0.5 + 0.5
        p = repair(p)
        particles.append(p)
    particles = np.array(particles)
    
    PS = particles.shape[0]
    velocities = np.random.rand(PS, n_points) * 0.1
