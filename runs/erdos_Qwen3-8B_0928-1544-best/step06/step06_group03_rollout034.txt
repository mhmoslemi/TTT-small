            if trial_c5 < compute_c5(target):
                pop[i] = trial.copy()
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    final_h = np.clip(best_h, 1e-8, 1.0 - 1e-8)
    final_h = final_h * (ones_count / np.sum(final_h))
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, n_points
