import numpy as np
from scipy.optimize import differential_evolution
from scipy.interpolate import interp1d

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.full(n_points, 0.5)
        perturb = np.random.choice([0.25, 0.75], size=n_points, p=[0.2, 0.2])
        initial_h += perturb
        initial_h = np.clip(initial_h, 0, 1)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        sum_h = np.sum(h)
        # Penalize deviation from required sum
        penalty = 1e3 * (sum_h - target_sum)**2
        return max_corr + penalty

    bounds = [(0.0, 1.0)] * n_points
    # Use differential evolution to find the optimal h
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=200,
        popsize=15,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )

    h_values = result.x
    # Enforce sum constraint and clip values
    sum_h = np.sum(h_values)
    if sum_h != target_sum:
        scale = target_sum / sum_h
        h_values = h_values * scale

    # Final C5 computation
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    c5_bound = np.max(corr) * dx
    return h_values, c5_bound, n_points
