    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):

        return total - ones_count

    bounds = [(0.0, 1.0)] * n_points

        new_h = initial_h + perturb
        new_h = np.clip(new_h, 0, 1)
        initial_population.append(new_h)

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    max_iter = int(budget_s / 25)
