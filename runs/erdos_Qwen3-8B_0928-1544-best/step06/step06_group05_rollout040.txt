    def objective(h):

        return np.max(corr) * dx

        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        k_idx = np.argmax(corr)
        n = len(h)
        shift_k = k_idx - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            term1 = 1.0 - h[(i + shift_k) % n] if (i + shift_k) < n else 0.0
            term2 = h[(i - shift_k) % n] if (i - shift_k) >= 0 else 0.0
            grad[i] = dx * (term1 - term2)
        return max_corr * dx, grad
