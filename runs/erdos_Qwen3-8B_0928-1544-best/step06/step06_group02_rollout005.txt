import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()

        for i in range(n_points // 2):

    # Scale to meet sum constraint
    if np.sum(initial_h) == 0:
        initial_h = np.zeros(n_points)
    else:
        initial_h = initial_h * (target_sum / np.sum(initial_h))

    best_h = initial_h.copy()
    best_c5 = float('inf')

    # Constraint function
    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Function to evaluate C5
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial best_c5
    best_c5 = compute_c5(best_h)

    # Main loop with time tracking
    start_time = time.time()

    max_iter = 100  # Carrion supporter
    for _ in range(max_iter):
        # Compute current worst shift
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        worst_k = np.argmax(corr)
