import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    
    # Initialize population
    pop_size = 30
    pop = [initial_h.copy() + np.random.normal(0, 0.1, n_points) for _ in range(pop_size - 1)]
    pop[0] = initial_h.copy()
    pop = [np.clip(p, 0, 1) for p in pop]
    
    # Rescale to meet sum constraint
    for i in range(pop_size):
        s = np.sum(pop[i])
        if s == 0:
            pop[i] = np.full(n_points, 1.0 / n_points) * ones_count
        else:
            pop[i] = pop[i] * (ones_count / s)
    
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        return np.max(corr) * (2.0 / len(h_vec))
    
    best_h = pop[0].copy()
    best_c5 = compute_c5(best_h)
    time_used = 0
    max_iter = int(budget_s / 100)
    F = 0.8
    CR = 0.9
    
    for _ in range(max_iter):
        for i in range(pop_size):
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
            mutant = a + F * (b - c)
            
            trial = np.random.rand(n_points) < CR
            trial = np.where(trial, mutant, pop[i])
            trial = np.clip(trial, 0, 1)
            
            s = np.sum(trial)

                trial = np.full(n_points, 1.0 / n_points) * ones_count

                trial = trial * (ones_count / s)
