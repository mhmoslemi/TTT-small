import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    time_used = 0
    max_iter = 500
    iter_step = 3

    for iter in range(max_iter):
        start_time = time.time()
        # Greedy adjustment
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr_idx = np.argmax(corr)
        k = max_corr_idx - n_points + 1
        k = k % n_points
        shift = k
        overlaps = best_h * (1 - np.roll(best_h, shift))
        contrib = overlaps
        big_contrib = np.argpartition(contrib, -5)[-5:]
        
        delta = 0.01
        for i in big_contrib:
            if best_h[i] > delta:
                best_h[i] -= delta
                toi = (i + shift) % n_points
                if best_h[toi] < 1.0 - delta:
                    best_h[toi] += delta
        
        # Normalize
        best_h = best_h * (ones_count / np.sum(best_h))
        
        # Perturb
        if iter % iter_step == 0:
            sigma = 0.05 * (1 - iter / max_iter)
            perturb = np.random.normal(0, sigma, n_points)
            best_h += perturb
            best_h = np.clip(best_h, 0.0, 1.0)
            best_h = best_h * (ones_count / np.sum(best_h))
        
        # Run local optimizer
        result = minimize(
            fun=objective,
            x0=best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        time_used += time.time() - start_time
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
