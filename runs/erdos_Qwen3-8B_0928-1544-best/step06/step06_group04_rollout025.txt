import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h_sum = np.sum(h)
        if h_sum !=0:
            h = h * (target_sum / h_sum)
    else:
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        if h_sum !=0:
            h = h * (target_sum / h_sum)
    
    def evaluate_c5(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    temp = 1.0
    cooling_rate = 0.995
    iteration = 0
    max_iterations = int(budget_s / 0.1)
    
    while iteration < max_iterations and time.time() - start_time < budget_s:
        # Perturb h
        idx = np.random.randint(n_points)
        delta = np.random.uniform(-0.05, 0.05)
        new_h = h.copy()
        new_h[idx] += delta
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum != 0:
            new_h = new_h * (target_sum / new_h_sum)
        new_c5 = evaluate_c5(new_h)
        
        # Acceptance
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
        else:
            prob = np.exp( -(new_c5 - best_c5) / temp )
            if np.random.rand() < prob:
                best_h = new_h.copy()
                best_c5 = new_c5
        
        # Update temperature
        temp *= cooling_rate
        iteration += 1
    
    return best_h, best_c5, n_points

        return np.max(corr) * (2.0 / n_points)
    
    best_h = h.copy()
