import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.zeros(n_points)
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = np.inf
    time_used = 0.0
    max_iter = int(budget_s / 2)
    
    for _ in range(max_iter):
        # Perturb current best
        perturb = np.random.normal(0, 0.08, n_points)
        h_perturbed = best_h + perturb
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        time_used += 3.0
        if time_used >= budget_s:
            break
    
    h_final = np.clip(best_h, 1e-8, 1.0 - 1e-8)
    h_final = h_final * (ones_count / np.sum(h_final))
    corr = np.correlate(h_final, 1.0 - h_final, mode='full')
    best_c5 = np.max(corr) * (2.0 / n_points)
    
    return h_final, best_c5, n_points
