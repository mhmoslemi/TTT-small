import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    
    # Initialize with provided or default pattern
    try:
        h = initial_h_values
    except:
        h = np.zeros(n_points)
        for i in range(n_points // 2):
            h[i*2] = 1.0
    h = h * (target_sum / np.sum(h)) if np.sum(h) > 0 else h
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def grad(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        h_shifted = np.roll(h, k)
        return 1.0 - h_shifted
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = h.copy()
    best_c5 = objective(h)
    time_used = 0
    max_iter = int(budget_s / 2)
    
    for _ in range(max_iter):
        # Estimate gradient
        grad_est = grad(best_h)
        # Use a simple descent with momentum
        h_new = best_h - 0.05 * grad_est
        h_new = np.clip(h_new, 0.0, 1.0)
        # Adjust sum
        def adjust_sum(h):
            while np.sum(h) > target_sum:
                h[np.argmax(h)] -= 1e-5
            while np.sum(h) < target_sum:
                h[np.argmin(h)] += 1e-5
            return h
        h_new = adjust_sum(h_new)
        
        # Local optimization
        result = minimize(
            fun=objective,
            x0=h_new,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        time_used += 2
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
