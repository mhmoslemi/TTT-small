def run(seed=42, budget_s=1100, **kwargs):

        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (target_sum / h_sum)
        return h
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h_sum = np.sum(initial_h)
        if initial_h_sum != 0:
            initial_h = initial_h * (target_sum / initial_h_sum)
        population = [initial_h]
    else:
        population = [create_candidate()]
    
    while len(population) < population_size:
        population.append(create_candidate())
    
    fitness = [evaluate_c5(h) for h in population]
    best_h = population[np.argmin(fitness)]
    best_c5 = np.min(fitness)
    
    while time.time() - start_time < budget_s:

            selected = tournament[np.argmin([evaluate_c5(h) for h in tournament])]
            new_population.append(selected)

                c1, c2 = np.zeros(n_points), np.zeros(n_points)

                        c1[j], c2[j] = p1[j], p2[j]
