    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    h = initial_h.copy()
    h = h * (n_points / 2) / np.sum(h)
    best_c5 = max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    best_h = h.copy()
    perturbation_scale = 0.25
    cooling_rate = 0.999
    max_iter = int(budget_s / 200)
    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for j in range(n_points):
            if j + k < n_points:
                grad[j] += (1 - h[j + k])
            if j - k >= 0:
                grad[j] -= h[j - k]
        alpha = 0.1
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        c5_new = max(np.correlate(h_new, 1 - h_new, mode='full')) * (2.0 / n_points)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h - 0.05 * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h += perturbation_scale * np.sin(h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    h = best_h
    h = h * (n_points / 2) / np.sum(h)
    return h, best_c5, n_points
