def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2.0
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
        initial_h = np.clip(initial_h, 0.0, 1.0)
    
    initial_points = [initial_h.copy()]
    for _ in range(5):
        h = initial_h.copy()
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        if current_sum > 0:
            h *= target_sum / current_sum
        initial_points.append(h.copy())
    
    best_h = initial_h.copy()
    best_c5 = compute_max_overlap(best_h, n_points)
    time_start = time.time()
    
    for h_start in initial_points:
        h = h_start.copy()
        for _ in range(100):
            if time.time() - time_start > budget_s:
                break
            overlaps = compute_overlaps(h, 2.0 / n_points)
            max_overlap = np.max(overlaps)
            if max_overlap == 0:
                break
            best_k = np.argmax(overlaps) + 1
            
            gradient = np.zeros_like(h)
            for j in range(best_k, n_points):
                gradient[j] = 1 - h[j]
            
            alpha = 0.01
            h = h - alpha * gradient
            h = np.clip(h, 0.0, 1.0)
            current_sum = np.sum(h)
            if current_sum > 0:
                h *= target_sum / current_sum
            
            new_c5 = compute_max_overlap(h, n_points)
            if new_c5 < max_overlap:
                best_h = h.copy()
                best_c5 = new_c5
            else:
                break
    
    return best_h, best_c5, n_points
