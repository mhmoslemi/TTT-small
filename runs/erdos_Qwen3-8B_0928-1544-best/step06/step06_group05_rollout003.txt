import numpy as np
import cma
import time

def repair(h, target_sum):
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    delta = (target_sum - s) / len(h)
    h += delta
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_population = []
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        for _ in range(5):
            perturb = np.random.normal(0, 0.05, n_points)
            candidate = initial_h + perturb
            candidate = np.clip(candidate, 0, 1)
            initial_population.append(candidate)
    else:
        for _ in range(5):
            random_h = np.random.rand(n_points)
            s = np.sum(random_h)
            delta = (target_sum - s) / n_points
            random_h += delta
            random_h = np.clip(random_h, 0, 1)
            initial_population.append(random_h)
    
    for _ in range(5):
        periodic = np.tile([0.5, 0.5], n_points // 2)
        perturb = np.random.normal(0, 0.05, n_points)
        candidate = periodic + perturb
        candidate = np.clip(candidate, 0, 1)
        initial_population.append(candidate)
    
    initial_population = initial_population[:50]
    initial_population = [repair(h, target_sum) for h in initial_population]
    
    x0 = initial_population[0]
    sigma = 0.2
    start_time = time.time()
    best_h = x0.copy()
    best_c5 = np.inf
    
    es = cma.CMAEvolutionStrategy(x0, sigma, {'popsize': 50})
    while time.time() - start_time < budget_s:
        population = es.ask()
        costs = []
        for x in population:
            x_repair = repair(x, target_sum)
            corr = np.correlate(x_repair, 1 - x_repair, mode='full')
            c5 = np.max(corr) * dx
            costs.append(c5)
        es.tell(population, costs)
        current_c5 = np.max(np.correlate(es.x, 1 - es.x, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = es.x.copy()
    
    return best_h, best_c5, n_points
