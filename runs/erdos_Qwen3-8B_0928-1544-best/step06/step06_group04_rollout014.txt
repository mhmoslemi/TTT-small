import numpy as np
import time

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial_sum = np.sum(h_initial)
        if h_initial_sum != 0:
            h_initial = h_initial * (target_sum / h_initial_sum)
        population.append(h_initial)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (target_sum / h_sum)
        population.append(h)
    
    population_size = 50
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (target_sum / h_sum)
        population.append(h)
    
    # Evaluate initial fitness
    fitness = [evaluate_c5(h, dx) for h in population]
    best_index = np.argmin(fitness)
    best_h = population[best_index]
    best_c5 = fitness[best_index]
    time_start = time.time()
    
    while time.time() - time_start < budget_s:
        # Tournament selection
        tournament_size = 5
        parents = []
        for _ in range(population_size // 2):
            indices = np.random.choice(len(population), tournament_size, replace=False)
            selected = [population[i] for i in indices]
            winner = min(selected, key=lambda x: evaluate_c5(x, dx))
            parents.append(winner)
        
        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            p1 = parents[i]
            p2 = parents[i+1] if i+1 < len(parents) else p1
