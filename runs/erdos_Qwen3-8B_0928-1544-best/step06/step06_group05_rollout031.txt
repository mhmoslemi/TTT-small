import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        x = np.linspace(0, 2, n_points)
        freq = 2.0 / (n_points / 2)
        initial_h = 0.5 + 0.4 * np.sin(2 * np.pi * x * freq)
        initial_h = np.clip(initial_h, 0, 1)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points

    res = minimize(
        fun=objective,
        x0=initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8}
    )

    best_h = res.x
    best_c5 = res.fun
    return best_h, best_c5, n_points
