import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        # Ensure initial_h is for the current n_points
        if len(initial_h) != n_points:
            from scipy.interpolate import interp1d
            x = np.linspace(0, 2, len(initial_h))
            f = interp1d(x, initial_h, kind='linear', fill_value='extrap')
            initial_h = f(np.linspace(0, 2, n_points))
        h = np.clip(initial_h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.full(n_points, 0.5)
    best_h = h.copy()
    best_c5 = float('inf')
    time_used = 0
    max_iter = 10000
    learning_rate = 0.1
    decay_rate = 0.999
    for iter in range(max_iter):
        start_time = time.time()
        max_overlap = -np.inf
        best_k = 0
        for k in range(n_points):
            shifted = np.roll(h, k)
            overlap = np.dot(h, 1 - shifted) * dx
            if overlap > max_overlap:
                max_overlap = overlap
                best_k = k
        grad = np.zeros(n_points)
        for j in range(n_points):
            j_plus_k = (j + best_k) % n_points
            j_minus_k = (j - best_k) % n_points
            term1 = 1 - h[j_plus_k]
            term2 = h[j_minus_k]
            grad[j] = dx * (term1 - term2)
        h -= learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        if current_sum < target_sum - 1e-8:
            scale = target_sum / current_sum
            h *= scale
        elif current_sum > target_sum + 1e-8:
            scale = target_sum / current_sum
            h *= scale
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        elapsed = time.time() - start_time
        time_used += elapsed
        if time_used >= budget_s:
            break
        learning_rate *= decay_rate
    return best_h, best_c5, n_points
