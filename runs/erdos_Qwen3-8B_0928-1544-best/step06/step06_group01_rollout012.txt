def run(seed=42, budget_s=1100, **kwargs):

        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * (2.0 / n_points))
    
    if not np.isclose(np.sum(initial_h), ones_count):
        initial_h = initial_h * ones_count / np.sum(initial_h)
    
    population_size = 50

    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * ones_count / np.sum(h)
        population.append(h)
    
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(initial_h)
    time_used = 0
    max_generations = 1000
    
    for gen in range(max_generations):
        if time_used >= budget_s:
            break
        
        fitness = np.array([evaluate_c5(h) for h in population])
        sorted_indices = np.argsort(fitness)
        selected_parents = [population[i] for i in sorted_indices[:population_size//2]]
        
        offspring = []
        for _ in range(population_size):
            parent1, parent2 = np.random.choice(selected_parents, 2, replace=False)
            child = (parent1 + parent2) / 2
            child = np.clip(child, 0, 1)
            child = child * ones_count / np.sum(child)
            offspring.append(child)
        
        offspring = np.array(offspring)
        fitness_offspring = evaluate_c5(offspring.mean(axis=0))
        offspring_fitness = np.array([evaluate_c5(h) for h in offspring])
        combined = np.vstack((population, offspring))
        combined_fitness = np.concatenate((fitness, offspring_fitness))
        
        sorted_indices = np.argsort(combined_fitness)
