        c5 = objective(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # SLSQP refinement with perturbations
    while time.time() - start_time < budget_s:
        h_perturbed = best_h + np.random.normal(0, 0.1 * (1 - (time.time() - start_time) / budget_s), n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (sum_h / np.sum(h_perturbed))
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
