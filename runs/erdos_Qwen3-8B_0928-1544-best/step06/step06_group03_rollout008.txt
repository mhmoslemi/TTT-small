def run(seed=42, budget_s=1100, **kwargs):

    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)

    particles = np.tile(initial_h, (num_particles, 1))
    velocities = np.random.uniform(-0.2, 0.2, (num_particles, n_points))
    
    # Personal best positions and fitness
    personal_best = particles.copy()
    personal_best_fitness = np.full(num_particles, np.inf)
    
    # Global best position and fitness
    global_best = particles[0].copy()
    global_best_fitness = np.inf
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # PSO parameters
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    max_iter = int(budget_s / 2)

        # Update velocities and positions
        r1, r2 = np.random.rand(num_particles, n_points), np.random.rand(num_particles, n_points)
        velocities = (w_end + (w_start - w_end) * iter / max_iter) * velocities + \
                     c1 * r1 * (personal_best - particles) + \
                     c2 * r2 * (global_best - particles)
        
        particles += velocities
        
        # Clip positions to [0, 1]
        particles = np.clip(particles, 0.0, 1.0)

        particles = particles * (ones_count / particles_sum[:, np.newaxis])
        
        # Evaluate fitness
        fitness = np.array([objective(p) for p in particles])
        
        # Update personal best
