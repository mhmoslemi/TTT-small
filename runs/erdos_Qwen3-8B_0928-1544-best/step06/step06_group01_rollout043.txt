import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    time_used = 0
    max_iter = 100
    
    for _ in range(max_iter):
        start = time.time()
        h = best_h.copy()
        # Compute worst shift and adjust
        corr = np.correlate(h, 1 - h, mode='full')
        worst_idx = np.argmax(corr)
        k = worst_idx - n_points + 1
        shift = k % n_points
        # Adjust h to reduce overlap for shift
        alpha = 0.1
        h[:n_points - shift] *= 1 - alpha
        h[shift:] *= 1 + alpha
        h = np.clip(h, 0.0, 1.0)
        h = h * (ones_count / np.sum(h))
        
        # Local optimization
        res = minimize(
            fun=objective,
            x0=h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        
        time_used += time.time() - start
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
