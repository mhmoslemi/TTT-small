        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    if abs(total - ones_count_fine) > 1e-6:
        best_h = best_h * (ones_count_fine / total)
    
    return best_h, best_c5, N_fine
