def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    
    if not hasattr(initial_h, '__array__') or initial_h.size == 0:
        initial_h = np.zeros(n_points)
        for i in range(n_points // 2):
            initial_h[i*2] = 1.0
    initial_h = initial_h * (target_sum / np.sum(initial_h)) if np.sum(initial_h) > 0 else initial_h
    
    def cross_corr_max(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = cross_corr_max(best_h)
    low = 0.0
    high = 0.38085
    tolerance = 1e-5
    time_used = 0
    max_iter = int(budget_s / 2)
    
    for _ in range(max_iter):
        mid = (low + high) / 2
        current_h = best_h.copy()
        current_c = best_c5
        for iter in range(10):
            sigma = 0.1 * (1 - iter / 10)
            h_perturbed = current_h + np.random.normal(0, sigma, n_points)
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            
            def adjust_sum(h):
                while np.sum(h) > target_sum:
                    h[np.argmax(h)] -= 1e-5
                while np.sum(h) < target_sum:
                    h[np.argmin(h)] += 1e-5
                return h
            
            h_perturbed = adjust_sum(h_perturbed)
            
            result = minimize(
                fun=lambda h: cross_corr_max(h),
                x0=h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8}
            )
            
            if result.success and result.fun < current_c:
                current_c = result.fun
                current_h = result.x
        
        if current_c <= mid:
            high = mid
        else:
            low = mid
        
        time_used += 2
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
