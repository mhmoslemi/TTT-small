import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    
    def clip_and_adjust(h):
        h = np.clip(h, 0.0, 1.0)
        while np.sum(h) > target_sum:
            idx = np.argmax(h)
            h[idx] -= 1e-5
        while np.sum(h) < target_sum:
            idx = np.argmin(h)
            h[idx] += 1e-5
        return h
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    time_used = 0
    max_iter = int(budget_s / 2)
    
    for iter in range(max_iter):
        # Perturb with adaptive noise
        sigma = 0.2 * (1 - iter / max_iter)
        h_perturbed = best_h + np.random.normal(0, sigma, n_points)
        h_perturbed = clip_and_adjust(h_perturbed)
        
        # Local optimization with COBYLA
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 300, 'ftol': 1e-8, 'disp': False}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Early stopping
        time_used += 2
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
