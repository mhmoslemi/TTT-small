        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        
        # Adjust sum constraint with clipping and element-wise adjustment
        def adjust_sum(h):
            while np.sum(h) > target_sum:
                h[np.argmax(h)] -= 1e-4
            while np.sum(h) < target_sum:
                h[np.argmin(h)] += 1e-4
            return h
        
        h_perturbed = adjust_sum(h_perturbed)
        
        # Local optimization with SLSQP
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Early stopping
        time_used += 2
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
