import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    required_sum = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.full(n_points, 0.5)
        perturb = np.random.choice([0.25, 0.75], size=n_points, p=[0.2, 0.2])
        initial_h += perturb
        initial_h = np.clip(initial_h, 0, 1)
        # Add periodic patterns for diversity
        initial_h[::2] += 0.25
        initial_h[1::2] -= 0.25
        initial_h = np.clip(initial_h, 0, 1)

    def repair(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if np.isclose(total, required_sum):
            return h
        delta = (required_sum - total) / len(h)
        h += delta
        h = np.clip(h, 0, 1)
        return h

    def objective(h):
        h_repaired = repair(h)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    
    result = differential_evolution(
        objective,
        bounds,
        strategy='rand',
        popsize=50,
        tol=1e-10,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        maxiter=int((budget_s - 500) / 10),
        disp=False
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 100:
        best_h = initial_h
        best_c5 = objective(best_h)
    else:
        best_h = result.x
        best_c5 = objective(best_h)

    return best_h, best_c5, n_points
