    fitness = np.array([evaluate_c5(h) for h in population])
    best_h = population[0]
    best_c5 = fitness[0]
    
    for gen in range(max_gen):
        current_time = time.time()
        if current_time - start_time > budget_s:
            break
        
        # Elitism
        elite_count = int(pop_size * elite_ratio)
        elites = np.argsort(fitness)[:elite_count]
        new_population = [population[i] for i in elites]
        
        # Generate remaining individuals
        for _ in range(pop_size - elite_count):
            # Tournament selection
            tournament_indices = np.random.choice(range(pop_size), size=tournament_size)
            best_idx = tournament_indices[np.argmin(fitness[tournament_indices])]
            parent1 = population[best_idx]
            tournament_indices2 = np.random.choice(range(pop_size), size=tournament_size)
            best_idx2 = tournament_indices2[np.argmin(fitness[tournament_indices2])]
            parent2 = population[best_idx2]
            
            # Crossover
            child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mut_rate:
                    delta = np.random.uniform(-0.1, 0.1)
                    child[i] += delta
                    child[i] = np.clip(child[i], 0, 1)
            
            # Rescale to target_sum
            child_sum = np.sum(child)
            if child_sum != 0:
                child = child * (target_sum / child_sum)
            
            new_population.append(child)
        
        # Update population
        population = new_population
        
        # Recompute fitness
        for i in range(pop_size):
            fitness[i] = evaluate_c5(population[i])
        
        # Update best solution
        if np.min(fitness) < best_c5:
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            best_c5 = fitness[best_idx]
    
    return best_h, best_c5, n_points
