    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        # Construct a smooth initial guess using windowed sinusoidal pattern
        t = np.linspace(0, 2, n_points)
        initial_h = 0.5 + 0.5 * np.cos(2 * np.pi * (t - 1))
        initial_h = np.clip(initial_h, 0, 1)
    
    # Define critical shifts that influence maximum overlap
    critical_shifts = [0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 1.75]
    
    def obj_func(h):
        # Compute cross-correlation for critical shifts
        corr = np.array([np.correlate(h, 1 - h, mode='full')[k] for k in critical_shifts])
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    
    # Differential Evolution with smart initial population
    def de_obj(h):
        return obj_func(h)
    
    if 'initial_h_values' in globals():
        # Spread initial guesses around the base solution
        def generate_perturbations(h_base):

            for _ in range(30):
                perturbed = h_base + 0.2 * np.random.normal(0, 0.2, n_points)
                perturbed = np.clip(perturbed, 0, 1)
                perturbations.append(perturbed)
            return perturbations
        initial_guesses = generate_perturbations(initial_h)
    else:
        # Fallback to random initialization
        initial_guesses = [np.random.uniform(0, 1, n_points) for _ in range(30)]

    max_evals = int(budget_s / 250)  # 4 seconds per eval for n=70
    
    best_h = initial_h.copy()
    best_c5 = obj_func(best_h)
    
    # Use differential evolution for global search
    result = differential_evolution(
        de_obj,
        bounds,
        strategy='best1bin',
        popsize=40,
        maxiter=100,
        tol=1e-7,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        polish=True,
        disp=False
    )
    
    best_h = result.x
    best_c5 = obj_func(best_h)
    
    # Add constraint enforcement via penalty
    def obj_with_penalty(h):
        obj = obj_func(h)
        constraint_violation = abs(np.sum(h) - ones_count) * 1000
        return obj + constraint_violation
    
    # Local refinement with SLSQP
    result_local = minimize(
        fun=obj_with_penalty,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    best_h = result_local.x
    best_c5 = obj_func(best_h)
    
    return best_h, best_c5, n_points
