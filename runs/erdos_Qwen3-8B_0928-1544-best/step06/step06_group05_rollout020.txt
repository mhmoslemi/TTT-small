    particles = [initial_h.copy()]
    for _ in range(49):
        if np.random.rand() < 0.5:
            h = np.random.rand(n_points)
            h = (h - 0.5) * 0.4 + 0.5
        else:
            h = np.tile([0.5, 0.5], (n_points // 2, 1)).flatten()
        h = np.clip(h, 0, 1)
        h = h * (ones_count / np.sum(h)) if np.sum(h) != 0 else h
        particles.append(h)

    n_particles = len(particles)
    pbest = [p.copy() for p in particles]
    pbest_vals = [np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles]

    gbest_val = min(pbest_vals)
    best_h = gbest
    best_c5 = gbest_val

    inertia = 0.7

    max_velocity = 0.5

    velocities = [np.random.rand(n_points) * max_velocity - 0.5 * max_velocity for _ in particles]

    start_time = time.time()

    while time.time() - start_time < budget_s:

            r1, r2 = np.random.rand(2)

            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)

            # Update position
            particles[i] += velocities[i]

                    delta = (ones_count - s) / n_points
                    particles[i] += delta

            # Compute objective
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
