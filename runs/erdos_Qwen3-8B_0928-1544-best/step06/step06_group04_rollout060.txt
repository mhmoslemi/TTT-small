import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2

    best_h = h.copy()

        corr = np.correlate(h, 1 - h, mode='full')
        c = corr * (2.0 / n_points)
        max_c = np.max(c)

            best_h = h.copy()

        grad = (np.exp(beta * c) / Z) * (1 - 2 * h)
        h += eta * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:

        beta *= 0.99
