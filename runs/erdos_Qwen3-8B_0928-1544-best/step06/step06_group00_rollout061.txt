                return h_sa
            h_new = h_sa.copy()
            idx = np.random.randint(n_points)

            delta = objective(h_new) - objective(h_sa)

                h_sa = h_new
            temp *= cooling
        return h_sa
    
    def pso_search():

        pbest = particles.copy()
        gbest = particles[0].copy()
        for _ in range(1000):
            if time.time() - start_time > budget_s:
                return gbest
            for i in range(50):
                r1, r2 = np.random.rand(2)

                particles[i] = particles[i] * (ones_count / np.sum(particles[i]))
                if objective(particles[i]) < objective(pbest[i]):
                    pbest[i] = particles[i].copy()
            gbest = pbest[np.argmin([objective(p) for p in pbest])]

    def ga_search():
