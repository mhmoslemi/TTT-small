from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    n_ones = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        # Generate a random initial guess
        initial_h = np.zeros(n_points)
        indices = np.random.choice(n_points, n_ones, replace=False)
        initial_h[indices] = 1.0

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - n_ones

    bounds = [(0.0, 1.0)] * n_points

    # Build diverse initial population
    initial_population = [initial_h.copy()]
    # Add random configuration
    random_h = np.zeros(n_points)
    indices = np.random.choice(n_points, n_ones, replace=False)
    random_h[indices] = 1.0
    initial_population.append(random_h)
    # Add structured configuration (every other position)
    structured_h = np.zeros(n_points)
    structured_h[::2] = 1.0
    initial_population.append(structured_h)
    # Add perturbed versions
    for _ in range(5):
        perturbed = initial_h + np.random.normal(0, 0.15, n_points)
        initial_population.append(np.clip(perturbed, 0, 1))

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    max_iter = int(budget_s / 200)

    for _ in range(max_iter):
        current_h = initial_population[np.random.randint(len(initial_population))]
        result = minimize(
            fun=objective,
            x0=current_h,
            method='SLSQP',
            bounds=Bounds(0, 1),
            constraints={'type': 'eq', 'fun': constraint_sum},

        if result.success and result.fun < best_c5:

            best_h = result.x
            # Replace a random individual in the population

                initial_population[idx] = result.x
            else:
                initial_population.append(result.x)
