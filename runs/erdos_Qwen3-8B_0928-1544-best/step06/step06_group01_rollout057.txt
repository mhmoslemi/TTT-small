import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    time_used = 0.0
    max_iter = 1500
    learning_rate = 0.1
    decay_rate = 0.995
    eps = 1e-6
    
    for _ in range(max_iter):
        start_time = time.time()
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - n_points + 1
        
        grad = np.zeros_like(best_h)
        for i in range(n_points):
            i_plus_lag = (i + lag) % n_points
            i_minus_lag = (i - lag) % n_points
            grad[i] = (1 - best_h[i_plus_lag]) - best_h[i_minus_lag]
        grad *= (2.0 / n_points)
        
        best_h -= learning_rate * grad
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = best_h * (ones_count / np.sum(best_h))
        
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        
        learning_rate *= decay_rate
        time_used += time.time() - start_time
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
