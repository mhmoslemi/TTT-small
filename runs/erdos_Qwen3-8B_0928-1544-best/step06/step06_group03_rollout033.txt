import numpy as np
from scipy.optimize import Bounds

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    
    # Initialize swarm: 20 particles, starting with initial_h
    swarm_size = 20
    positions = np.zeros((swarm_size, n_points))
    velocities = np.zeros((swarm_size, n_points))
    
    # First particle is the initial solution
    positions[0] = initial_h
    velocities[0] = np.zeros(n_points)
    
    # Perturb others around initial solution
    for i in range(1, swarm_size):
        positions[i] = initial_h + np.random.normal(0, 0.1, n_points)
        positions[i] = np.clip(positions[i], 0, 1)
        positions[i] = (positions[i] * target_sum) / np.sum(positions[i])
    
    # PSO parameters
    w = 0.9
    c1 = c2 = 2.0
    max_iter = 500
    velocity_limit = 0.2
    
    # Track best solutions
    personal_best = positions.copy()
    global_best = np.min(positions, axis=0)
    best_score = np.inf
    
    for iter in range(max_iter):
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] +
                            c1 * r1 * (personal_best[i] - positions[i]) +
                            c2 * r2 * (global_best - positions[i]))
            velocities[i] = np.clip(velocities[i], -velocity_limit, velocity_limit)
            positions[i] = np.clip(positions[i] + velocities[i], 0, 1)
            # Rescale to meet sum constraint
            positions[i] = (positions[i] * target_sum) / np.sum(positions[i])
        
        # Evaluate new positions
        for i in range(swarm_size):
            h = positions[i]
            corr = np.correlate(h, 1.0 - h, mode='full')
            score = np.max(corr) * (2.0 / n_points)
            if score < best_score:
                best_score = score
                global_best = h.copy()
            # Update personal best
            if score < np.min(np.correlate(personal_best[i], 1.0 - personal_best[i], mode='full')) * (2.0 / n_points):
                personal_best[i] = h.copy()
        
        # Early termination if budget is exceeded
        if iter * 2 >= budget_s:
            break
    
    # Final validation
    h_final = np.clip(global_best, 1e-8, 1.0 - 1e-8)
    h_final = (h_final * target_sum) / np.sum(h_final)
    corr = np.correlate(h_final, 1.0 - h_final, mode='full')
    best_c5 = np.max(corr) * (2.0 / n_points)
    
    return h_final, best_c5, n_points
