import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70

            while np.sum(initial_h) > ones_count:
                idx = np.random.randint(n_points)
                if initial_h[idx] == 1:
                    initial_h[idx] = 0
        elif np.sum(initial_h) < ones_count:

                idx = np.random.randint(n_points)
                if initial_h[idx] == 0:
                    initial_h[idx] = 1
    initial_h = np.clip(initial_h, 0.0, 1.0)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        return np.max(corr) * dx

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    time_start = time.time()

    while time.time() - time_start < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        dx = 2.0 / n_points
        all_overlaps = corr[n_points-1:2*n_points-1]
        max_overlap_index = np.argmax(all_overlaps)
        worst_k = max_overlap_index + 1
        current_c5 = all_overlaps[max_overlap_index] * dx

        def constraint(h):
            start = 0
            end = n_points - worst_k
            sum_overlap = np.sum(h[start:end] * (1 - h[worst_k:end + worst_k]))
            return current_c5 * n_points / 2 - sum_overlap

        constraints = [
            {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2},
            {'type': 'ineq', 'fun': constraint}
        ]

        result = minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n_points,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-8}
        )

        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

            idx = np.argmax(best_h)
            best_h[idx] -= excess
