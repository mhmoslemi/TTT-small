        h_new = np.clip(h_new, 0, 1)
        sum_h = np.sum(h_new)
        h_new = h_new * (ones_count / sum_h) if sum_h > 0 else h_new
        
        h = h_new
        
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        elapsed_time = time.time() - time_start
        if elapsed_time >= budget_s:
            break
    
    return best_h, best_c5, n_points
