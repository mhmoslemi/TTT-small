from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2

    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    
    num_initial = 100
    initial_population = [initial_h.copy()]
    for _ in range(num_initial - 1):

        initial_population.append(np.clip(perturbed, 0, 1))
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    max_iter = int(budget_s / 250)
    
    for _ in range(max_iter):

        result = minimize(

            x0=current_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8, 'disp': False}
