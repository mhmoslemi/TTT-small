import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_low = 35
    n_high = 70
    budget_low = 0.5 * budget_s
    budget_high = 0.5 * budget_s

    if 'initial_h_values' in globals():
        h_low = initial_h_values.copy()
    else:
        h_low = np.full(n_low, 0.5)

    def optimize_h(h_initial, n_points, budget):
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * (2.0 / n_points)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2

        bounds = [(0.0, 1.0)] * n_points
        x0 = h_initial.copy()
        options = {'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
        result = minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options=options
        )
        return result.x, result.fun

    start_time = time.time()
    h_low_opt, c5_low = optimize_h(h_low, n_low, budget_low)
    time_used = time.time() - start_time
    
    if time_used >= budget_s:
        return h_low_opt, c5_low, n_low

    h_high_initial = np.repeat(h_low_opt, 2)
    h_high_opt, c5_high = optimize_h(h_high_initial, n_high, budget_high)
    time_used_high = time.time() - start_time

    if time_used_high >= budget_s:
        return h_low_opt, c5_low, n_low

    if c5_low < c5_high:
        return h_low_opt, c5_low, n_low
    else:
        return h_high_opt, c5_high, n_high
