import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    pop_size = 50
    tournament_size = 5
    mutation_rate = 0.1
    max_generations = 200
    start_time = time.time()
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h_sum = np.sum(initial_h)
        if initial_h_sum != 0:
            initial_h = initial_h * (target_sum / initial_h_sum)
        population.append(initial_h)
    
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (target_sum / h_sum)
        population.append(h)
    
    fitness = [evaluate_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_c5 = fitness[best_idx]
    
    # Main loop
    for gen in range(max_generations):
        current_time = time.time()
        if current_time - start_time > budget_s:
            break
        
        # Tournament selection
        new_population = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size, replace=False)
            best_tournament = tournament[np.argmin(fitness[tournament])]
            new_population.append(population[best_tournament])
        
        # Crossover
        for i in range(0, pop_size, 2):
            if i+1 < pop_size:
                p1, p2 = new_population[i], new_population[i+1]
                child1 = np.where(np.random.rand(n_points) < 0.5, p1, p2)
                child2 = np.where(np.random.rand(n_points) < 0.5, p2, p1)
                new_population[i], new_population[i+1] = child1, child2
        
        # Mutation
        for i in range(pop_size):
            h = new_population[i]
            if np.random.rand() < mutation_rate:
                mutation = np.random.rand(n_points) * 0.2
                h = np.clip(h + mutation, 0, 1)
                h_sum = np.sum(h)
                if h_sum != 0:
                    h = h * (target_sum / h_sum)
                new_population[i] = h
        
        # Evaluate and replace
        new_fitness = [evaluate_c5(h) for h in new_population]
        population = new_population
        fitness = new_fitness
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx]
    
    return best_h, best_c5, n_points
