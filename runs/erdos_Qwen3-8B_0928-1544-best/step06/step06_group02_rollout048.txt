import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    
    # Ensure initial_h meets sum constraint
    if np.sum(initial_h) > 0:
        initial_h = initial_h * (target_sum / np.sum(initial_h))
    else:
        for i in range(n_points // 2):
            initial_h[i*2] = 1.0
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    time_used = 0
    
    # Optimize with COBYLA and perturbative steps
    for _ in range(200):
        # Perturb initial guess
        perturb = np.random.normal(0, 0.15, n_points)
        h_perturbed = best_h + perturb
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        if np.sum(h_perturbed) > 0:
            h_perturbed = h_perturbed * (target_sum / np.sum(h_perturbed))
        
        # Local optimization
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'disp': False}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        time_used += 15  # Approximated time per iteration
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
