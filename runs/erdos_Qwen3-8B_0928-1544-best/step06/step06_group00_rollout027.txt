def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    # Genetic search parameters
    pop_size = 50
    generations = 10
    elite_ratio = 0.2
    mutation_scale = 0.05
    
    # Initialize population
    population = [initial_h.copy() for _ in range(pop_size)]
    for _ in range(pop_size):
        perturb = np.random.normal(0, 0.1, n_points)
        new_h = initial_h + perturb
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = new_h * (ones_count / np.sum(new_h))
        population.append(new_h)
    population = np.array(population)
    
    for gen in range(generations):
        fitness = np.array([objective(ind) for ind in population])
        elite_idx = np.argsort(fitness)[:int(pop_size * elite_ratio)]
        elites = population[elite_idx]
        new_population = elites.copy()
        while len(new_population) < pop_size:
            p1, p2 = np.random.choice(elite_idx, 2, replace=False)
            child = (population[p1] + population[p2]) / 2
            noise = np.random.normal(0, mutation_scale, n_points)
            child += noise
            child = np.clip(child, 0.0, 1.0)
            child = child * (ones_count / np.sum(child))
            new_population = np.vstack((new_population, child))
        population = new_population
    
    best_candidate = population[np.argmin(objective(population))]
    best_c5 = objective(best_candidate)
    best_h = best_candidate
    
    # Final optimization
    while time.time() - start_time < budget_s:
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = best_h + perturb
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed = perturbed * (ones_count / np.sum(perturbed))
        
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            break
    
    return best_h, best_c5, n_points
