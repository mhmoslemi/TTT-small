        initial_population.append(perturbed)
    
    best_h = initial_h.copy()

    max_iter = int(budget_s / 200)

        fitness = [compute_c5(h, n_points) for h in initial_population]
        # Sort by fitness and select top 50%
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:int(len(initial_population) * 0.5)]
        top_candidates = [initial_population[i] for i in top_indices]

        new_population = []

                # Blend crossover
                child1 = parent1 * 0.5 + parent2 * 0.5
                child2 = parent2 * 0.5 + parent1 * 0.5
                new_population.extend([child1, child2])
            else:
                new_population.append(top_candidates[i])
        
        # Apply mutation
        for child in new_population:
            perturbed = child + np.random.normal(0, 0.05, n_points)

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * (2.0 / n_points)
