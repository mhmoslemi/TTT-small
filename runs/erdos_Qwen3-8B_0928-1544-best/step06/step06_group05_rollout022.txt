from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):

    def objective(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points

    # Initialize best solution
    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    # Set optimization parameters
    max_iter = int(budget_s / 60)
    method = 'L-BFGS-B'
    tol = 1e-6

    for _ in range(max_iter):
        # Perturb current best solution

        perturbed = best_h + perturb
        perturbed = np.clip(perturbed, 0, 1)

        # Ensure sum constraint
        sum_perturbed = np.sum(perturbed)

            perturbed += delta
            perturbed = np.clip(perturbed, 0, 1)

        # Optimize from perturbed solution
