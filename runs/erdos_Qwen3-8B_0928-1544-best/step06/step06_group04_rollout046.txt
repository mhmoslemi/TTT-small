import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    pop_size = 50
    mutation_rate = 0.1
    tournament_size = 3

    def adjust_sum(h, target_sum):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        scaled = h * scale
        clipped = np.clip(scaled, 0, 1)
        return clipped

    def evaluate_h(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = adjust_sum(initial_h, target_sum)
        population.append(initial_h)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = adjust_sum(h, target_sum)
        population.append(h)
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = adjust_sum(h, target_sum)
        population.append(h)

    fitness = [evaluate_h(h) for h in population]
    best_index = np.argmin(fitness)
    best_h = population[best_index].copy()
    best_fitness = fitness[best_index]
    start_time = time.time()

    while True:
        current_time = time.time()
        if current_time - start_time > budget_s:
            break

        # Selection
        new_population = []
        for _ in range(pop_size):
            tournament = np.random.choice(population, size=tournament_size)
            winner_idx = np.argmin([evaluate_h(h) for h in tournament])
            new_population.append(tournament[winner_idx])

        # Crossover and mutation
        offspring = []
        for i in range(0, pop_size, 2):
            if i + 1 >= pop_size:
                break
            parent1, parent2 = new_population[i], new_population[i + 1]
            child1 = np.zeros(n_points)
            child2 = np.zeros(n_points)
            for j in range(n_points):
                if np.random.rand() < 0.5:
                    child1[j] = parent1[j]
                    child2[j] = parent2[j]
                else:
                    child1[j] = parent2[j]
                    child2[j] = parent1[j]
            # Mutate
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    delta = np.random.uniform(-0.1, 0.1)
                    child1[j] = np.clip(child1[j] + delta, 0, 1)
                    child2[j] = np.clip(child2[j] + delta, 0, 1)
            child1 = adjust_sum(child1, target_sum)
            child2 = adjust_sum(child2, target_sum)
            offspring.append(child1)
            offspring.append(child2)

        # Handle odd population
        if pop_size % 2 != 0:
            parent = new_population[-1]
            child = np.copy(parent)
            for j in range(n_points):

                    child[j] = np.clip(child[j] + delta, 0, 1)
            child = adjust_sum(child, target_sum)
