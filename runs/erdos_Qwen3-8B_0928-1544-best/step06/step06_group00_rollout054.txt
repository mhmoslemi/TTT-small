import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        dx = 2.0 / n_points
        current_c5 = corr[k_max] * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
        
        # Prepare objective for current k_max
        def objective(h_vec):
            corr_current = np.correlate(h_vec, 1 - h_vec, mode='full')
            return corr_current[k_max] * dx
        
        # Perform constrained optimization
        result = minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Early termination if no improvement
        if np.isclose(current_c5, best_c5, atol=1e-9):
            break
    
    return best_h, best_c5, n_points
