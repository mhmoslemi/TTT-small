import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    pop_size = 15
    population = []
    h = initial_h.copy()
    h = np.clip(h, 0, 1)
    h = h * (ones_count / np.sum(h))
    population.append(h)
    for _ in range(pop_size - 1):
        perturb = np.random.normal(0, 0.05, n_points)
        h = initial_h + perturb
        h = np.clip(h, 0, 1)
        h = h * (ones_count / np.sum(h))
        population.append(h)
    F = 0.8
    CR = 0.9
    max_iter = int(budget_s / 200)
    best_h = population[0]
    def compute_c5(h_vec):
        dx = 2.0 / n_points
        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        return np.max(corr) * dx
    best_c5 = compute_c5(best_h)
    for _ in range(max_iter):
        for i in range(len(population)):
            j, k, l = np.random.choice(len(population), 3, replace=False)
            a, b, c = population[j], population[k], population[l]
            mutant = a + F * (b - c)
            trial = population[i].copy()
            for idx in range(n_points):
                if np.random.rand() < CR:
                    trial[idx] = mutant[idx]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            trial = trial if trial_sum > 0 else np.zeros(n_points)
            trial = trial * (ones_count / trial_sum)
            c5_trial = compute_c5(trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = trial.copy()
            if c5_trial < compute_c5(population[i]):
                population[i] = trial.copy()
    final_h = np.clip(best_h, 1e-8, 1.0 - 1e-8)
    final_h = final_h * (ones_count / np.sum(final_h))
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, n_points
