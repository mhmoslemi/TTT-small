            j = (i + shift) % n_points
            C[i, j] = 1
        h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
        obj = cp.Maximize(cp.quad_form(h_var, C))
        constraints = [cp.sum(h_var) == ones_count]
        try:
            prob = cp.Problem(obj, constraints)
            prob.solve(solver='ECOS', eps=1e-10, verbose=False)
            if prob.status == 'OPTIMAL':
                h = np.clip(h_var.value, 0.0, 1.0)
                h = h * (ones_count / np.sum(h))
        except:
            pass
    
    return best_h, best_c5, n_points
