        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    max_iter = int(budget_s / 100)
    
    for _ in range(max_iter):
        current_h = initial_population[np.random.randint(len(initial_population))]

        current_h += np.random.normal(0, 0.1, n_points)
        current_h = np.clip(current_h, 0, 1)

        adjustment = (target_sum - current_sum) / n_points
