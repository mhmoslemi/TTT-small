import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    
    # Scale initial guess if provided
    if hasattr(initial_h, '__array__') and initial_h.size > 0:
        initial_h = initial_h * (target_sum / np.sum(initial_h)) if np.sum(initial_h) > 0 else initial_h
    else:
        # Binary initial guess
        initial_h = np.zeros(n_points)
        for i in range(n_points // 2):
            initial_h[i*2] = 1.0
        initial_h = initial_h * (target_sum / np.sum(initial_h))
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Early stopping parameters
    no_improvement = 0
    max_iterations = int(budget_s / 2)
    
    for _ in range(max_iterations):
        # Perturb current best with controlled noise
        sigma = 0.15 * (1 - _ / max_iterations)
        h_perturbed = best_h + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        
        # Directly optimize with SLSQP, letting the solver handle the sum constraint
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Early stopping if no improvement for 100 iterations
        if no_improvement >= 100:
            break
        
        # Check budget time
        if time.time() - start_time >= budget_s:
            break
    
    return best_h, best_c5, n_points
