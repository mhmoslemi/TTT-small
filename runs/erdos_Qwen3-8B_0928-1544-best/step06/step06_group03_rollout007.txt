import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    pop_size = 50
    max_gen = 350
    best_h = None
    best_c5 = float('inf')
    
    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.random.rand(n_points)
    initial_h = np.clip(initial_h / np.sum(initial_h) * ones_count, 0.0, 1.0)
    
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        perturbed = initial_h + np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed = perturbed * ones_count / np.sum(perturbed)
        population.append(perturbed)
    
    time_used = 0
    mutation_rate = 0.15
    
    for gen in range(max_gen):
        if time_used >= budget_s:
            break
        
        f_vals = [compute_c5(ind) for ind in population]
        curr_best = population[np.argmin(f_vals)]
        if compute_c5(curr_best) < best_c5:
            best_c5 = compute_c5(curr_best)
            best_h = curr_best
        
        parents = []
        for _ in range(pop_size // 2):
            t = np.random.choice(population, 3, replace=False)
            winner = min(t, key=compute_c5)
            parents.append(winner)
        
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 >= len(parents):
                break
            p1, p2 = parents[i], parents[i+1]
            cp = np.random.randint(1, n_points)
            c1 = np.concatenate([p1[:cp], p2[cp:]])
            c2 = np.concatenate([p2[:cp], p1[cp:]])
            offspring.append(c1)
            offspring.append(c2)
        
        for child in offspring:
            if np.random.rand() < mutation_rate:
                m_idx = np.random.choice(n_points, int(0.25 * n_points), replace=False)
                child[m_idx] += np.random.normal(0, 0.05, len(m_idx))
                child = np.clip(child, 0.0, 1.0)
                child = child * ones_count / np.sum(child)
