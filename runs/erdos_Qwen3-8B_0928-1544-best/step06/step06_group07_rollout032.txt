        grad[:n_points - k_max] += 1 - best_h[k_max:]
        grad[k_max:] -= best_h[:n_points - k_max]
        
        alpha = 0.2
        for _ in range(10):
            h_trial = best_h - alpha * grad
            h_trial = np.clip(h_trial, 0, 1)
            sum_trial = np.sum(h_trial)
            delta = sum_h - sum_trial
            if delta != 0:
                h_trial += delta * (h_trial / max(sum_trial, 1e-10))
                h_trial = np.clip(h_trial, 0, 1)
            
            new_overlap = np.max(compute_overlaps(h_trial))
            if new_overlap < max_overlap:
                best_h = h_trial.copy()
                break
            alpha *= 0.5
    
    return best_h, best_c5, n_points
