import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.linspace(0, 1, n_points)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        delta = (ones_count - current_sum) / n_points
        h += delta
        h = np.clip(h, 0, 1)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-8}
    )

    best_h = result.x if result.success else h
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
