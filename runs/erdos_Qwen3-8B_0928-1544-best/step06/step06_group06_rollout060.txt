import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    required_sum = n_points / 2
    if 'initial_h_values' in globals():

        return max_corr * dx

    def fitness(h):

    best_h = initial_h.copy()
    best_c5 = fitness(best_h)
    population = initial_population.copy()
    remaining_time = budget_s

    for _ in range(int(remaining_time / 50)):

        parent2 = np.random.choice(population)
