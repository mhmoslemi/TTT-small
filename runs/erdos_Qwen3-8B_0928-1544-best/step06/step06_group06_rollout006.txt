        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h = h * scale
        
        if compute_c5(h) < best_c5_val:
            best_c5_val = compute_c5(h)
            best_h = h.copy()
    
    return best_h, best_c5_val, n_points
