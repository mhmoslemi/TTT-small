import numpy as np
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points).copy()
    
    # Initialize population
    population_size = 30
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        perturb = np.random.normal(0, 0.1, n_points)
        new_h = initial_h + perturb
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h_sum = np.sum(new_h)
        new_h = new_h * (ones_count / new_h_sum if new_h_sum > 1e-8 else np.zeros_like(new_h))
        population.append(new_h)
    
    # DE parameters
    F = 0.8
    CR = 0.9
    max_generations = 2500
    
    def calc_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * (2.0 / len(h))
    
    best_h = initial_h.copy()
    best_c5 = calc_c5(best_h)
    
    # Evaluate initial population
    for h in population:
        c5 = calc_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Main DE loop
    for gen in range(max_generations):
        np.random.shuffle(population)
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            target = population[i]
            vec1, vec2, vec3 = population[a], population[b], population[c]
            
            # Mutation
            mutant = vec1 + F * (vec2 - vec3)
            
            # Crossover
            trial = np.zeros_like(target)
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
                else:
                    trial[j] = target[j]
            
            # Clip and scale
            trial = np.clip(trial, 0.0, 1.0)
            trial_sum = np.sum(trial)
            if trial_sum > 1e-8:
                trial = trial * (ones_count / trial_sum)
            else:
                trial = np.zeros_like(trial)
            
            # Evaluate and replace
            trial_c5 = calc_c5(trial)
            current_c5 = calc_c5(target)
            if trial_c5 < current_c5:
                population[i] = trial.copy()
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
    
    # Final validation
    final_h = np.clip(best_h, 1e-8, 1.0 - 1e-8)
    final_h_sum = np.sum(final_h)
    final_h = final_h * (ones_count / final_h_sum) if final_h_sum > 1e-8 else np.zeros_like(final_h)
    final_c5 = calc_c5(final_h)
    
    return final_h, final_c5, n_points
