import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 400
    target_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    
    # Adjust initial_h to meet the sum constraint
    initial_sum = np.sum(initial_h)
    if initial_sum == 0:
        initial_h = np.zeros(n_points)
        for i in range(n_points // 2):
            initial_h[i*2] = 1.0
    if initial_sum > 0:
        scaling = target_sum / initial_sum
        initial_h = initial_h * scaling
    else:
        initial_h = np.zeros(n_points)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    time_used = 0
    max_iter = int(budget_s / 2)
    
    for _ in range(max_iter):
        result = minimize(
            fun=objective,
            x0=best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        time_used += 20
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
