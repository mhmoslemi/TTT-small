from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    dx = 2.0 / n_points

    def objective(h):

        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    initial_population = [initial_h.copy()]
    num_initial = 120
    
    for _ in range(num_initial - 1):
        perturb = np.random.rand(n_points)
        initial_population.append(initial_h + (perturb - 0.5) * 0.3)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    max_iter = int(budget_s / 15)
    
    for _ in range(max_iter):
        current_h = initial_population[np.random.randint(len(initial_population))]
        res = minimize(
            fun=objective,
            x0=current_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 30, 'ftol': 1e-10, 'eps': 1e-8}

            best_h = res.x
