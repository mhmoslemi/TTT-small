import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        n_points = len(initial_h)
    else:
        n_points = 100
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0)] * n_points
    
    initial_population = [initial_h.copy()]
    for _ in range(49):
        perturbed = initial_h + np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed != 0:
            perturbed = perturbed * (n_points / 2) / sum_perturbed
        initial_population.append(perturbed)
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    elapsed = 0
    
    max_iter = int(budget_s / 100)
    for _ in range(max_iter):
        current_h = initial_population[np.random.randint(len(initial_population))]
        result = minimize(
            fun=objective,
            x0=current_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 20, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
            initial_population.append(result.x)
            initial_population.pop(np.random.randint(len(initial_population)))
        elapsed += 100
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
