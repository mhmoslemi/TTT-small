import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    max_iterations = 500
    swarm_size = 30
    inertia = 0.5
    c1 = 2.0
    c2 = 2.5
    start_time = time.time()
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    particles = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h_sum = np.sum(initial_h)
        if initial_h_sum != 0:
            initial_h = initial_h * (target_sum / initial_h_sum)
        particles.append(initial_h)
        velocities.append(np.zeros(n_points))
        personal_best.append(initial_h.copy())
        personal_best_vals.append(evaluate_c5(initial_h))
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (target_sum / h_sum)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        personal_best.append(h.copy())
        personal_best_vals.append(evaluate_c5(h))
    
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (target_sum / h_sum)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        personal_best.append(h.copy())
        personal_best_vals.append(evaluate_c5(h))
    
    global_best = particles[0].copy()
    global_best_val = personal_best_vals[0]
    for i in range(1, swarm_size):
        if personal_best_vals[i] < global_best_val:
            global_best = particles[i].copy()
            global_best_val = personal_best_vals[i]
    
    for _ in range(max_iterations):
        current_time = time.time()
        if current_time - start_time > budget_s:
            break
        
        for i in range(swarm_size):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = (inertia * velocities[i] +
                             c1 * r1 * (personal_best[i] - particles[i]) +
                             c2 * r2 * (global_best - particles[i]))
            
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            h_sum = np.sum(particles[i])
            if h_sum != 0:
                particles[i] = particles[i] * (target_sum / h_sum)
            
            current_val = evaluate_c5(particles[i])
            if current_val < personal_best_vals[i]:
                personal_best[i] = particles[i].copy()
                personal_best_vals[i] = current_val
                if current_val < global_best_val:
                    global_best = particles[i].copy()
                    global_best_val = current_val
    
    return global_best, global_best_val, n_points
