import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    # Initialize h with provided values or structured pattern
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        # Alternate 1s and 0s to create a structured initial guess
        initial_h = np.zeros(n_points)
        initial_h[:ones_count] = 1.0
        initial_h[ones_count:] = 0.0

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points
    initial_population = [initial_h.copy()]
    num_initial = 10

    for _ in range(num_initial - 1):
        perturb = np.random.rand(n_points) * 0.1 - 0.05
        initial_population.append(initial_h + perturb)
        initial_population[-1] = np.clip(initial_population[-1], 0, 1)

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    max_iter = int(budget_s / 100)

    for _ in range(max_iter):
        current_h = initial_population[np.random.randint(len(initial_population))]
        res = minimize(
            fun=objective,
            x0=current_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-10, 'eps': 1e-8}
        )
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
            initial_population.append(res.x)
            initial_population.pop(np.random.randint(len(initial_population)))

    return best_h, best_c5, n_points
