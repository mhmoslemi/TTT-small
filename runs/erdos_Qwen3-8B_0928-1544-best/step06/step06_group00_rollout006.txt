    dx = 2.0 / n_points
    start_time = time.time()
    iter_limit = 200

    def compute_overlap(h, shift):
        shifted = np.roll(h, shift)
        overlap = np.sum(h * (1 - shifted)) * dx
        return overlap

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points

    for _ in range(iter_limit):
        if time.time() - start_time >= budget_s:
            break
        # Compute all overlaps
        overlaps = []
        for k in range(n_points):
            orl = compute_overlap(best_h, k)
            overlaps.append(orl)
        current_max = max(overlaps)
        if current_max < best_c5:
            best_c5 = current_max
            best_h = best_h.copy()
        k_max = overlaps.index(current_max)
        # Define objective for this shift
        def objective(h):
            return compute_overlap(h, k_max)
        # Optimization
        result = minimize(
            fun=objective,
            x0=best_h,
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'disp': False}
        )
        if result.success:
            if result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
        # Early termination if no improvement
        if _ > 50 and abs(best_c5 - 1.0) < 1e-8:
            break

    return best_h, best_c5, n_points
