import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h /= np.sum(h)
        h *= n_points / 2
        h = np.clip(h, 0, 1)
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    def repair(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.full_like(h, 0.5)
        return h * (n_points / 2) / s
    best_h = h
    best_c5 = compute_c5(h)
    initial_pop = [h.copy() for _ in range(50)]
    for i in range(1, 50):
        noise = np.random.normal(0, 0.03, n_points)
        initial_pop[i] = (h + noise) * (n_points / 2) / np.sum(np.clip(h + noise, 0, 1))
    pop = initial_pop
    sigma = 0.05
    start_time = time.time()
    while time.time() - start_time < budget_s:
        offspring = []
        for h in pop:
            noise = np.random.normal(0, sigma, n_points)
            mut = repair(h + noise)
            offspring.append(mut)
        c5s = [compute_c5(h) for h in offspring]
        best_idx = np.argmin(c5s)
        if c5s[best_idx] < best_c5:
            best_c5 = c5s[best_idx]
            best_h = offspring[best_idx].copy()
        new_pop = [offspring[best_idx]] + [np.random.choice(offspring) for _ in range(49)]
        pop = new_pop
        sigma *= 0.995 if best_c5 < 0.38088 else 1.01
    return best_h, best_c5, n_points
