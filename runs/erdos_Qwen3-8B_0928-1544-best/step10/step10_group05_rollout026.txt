    else:
        initial_h = np.full(n_points, 0.5)
    
    # Create initial population
    population_size = 50
    population = [initial_h.copy()]
    
    for _ in range(population_size - 1):
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = initial_h + perturb
        current_sum = new_h.sum()
        delta = current_sum - target_sum
        adjustment = delta / n_points
        new_h -= adjustment
        new_h = np.clip(new_h, 0, 1)
        population.append(new_h)
    
    def compute_c5(h_vec):
        h = h_vec
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * 2.0 / n_points
    
    fitness = [compute_c5(ind) for ind in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_c5 = fitness[best_idx]
    
    tournament_size = 5
    mutation_rate = 0.1
    num_generations = 200
    
    start_time = time.time()
    
    for gen in range(num_generations):
        if time.time() - start_time > budget_s:
            break
        
        parents = []
        for _ in range(population_size):
            tournament = np.random.choice(population, tournament_size, replace=False)
            tour_fitness = [compute_c5(ind) for ind in tournament]
            best_tour = tournament[np.argmin(tour_fitness)]
            parents.append(best_tour)
        
        offspring = []
        for i in range(0, population_size, 2):
            parent1 = parents[i]
            parent2 = parents[i+1] if i+1 < len(parents) else parents[0]
            split = np.random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:split], parent2[split:]])
            child2 = np.concatenate([parent2[:split], parent1[split:]])
            offspring.append(child1)
            offspring.append(child2)
        
        for j in range(len(offspring)):
            child = offspring[j].copy()
            for k in range(n_points):
                if np.random.rand() < mutation_rate:
                    delta = np.random.uniform(-0.05, 0.05)
                    child[k] += delta
            current_sum = child.sum()
            delta = current_sum - target_sum
            adjustment = delta / n_points
            child -= adjustment
            child = np.clip(child, 0, 1)
            offspring[j] = child
        
        combined = population + offspring
        combined_fitness = fitness + [compute_c5(ind) for ind in offspring]
        sorted_indices = np.argsort(combined_fitness)
        new_population = [combined[i] for i in sorted_indices[:population_size]]
        fitness = [compute_c5(ind) for ind in new_population]
        best_idx = np.argmin(fitness)
        best_h = new_population[best_idx]
        best_c5 = fitness[best_idx]
    
    return best_h, best_c5, n_points
