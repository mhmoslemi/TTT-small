import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()

        h = np.random.rand(n_points) * 0.5 + 0.5
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum)

    def compute_c5(h_val):
        if len(h_val) != n_points:
            h_val = np.resize(h_val, n_points)
        overlaps = []
        for k in range(n_points):
            shifted = h_val[(k + np.arange(n_points)) % n_points]
            overlaps.append(np.dot(h_val, shifted) * dx)
        return np.max(overlaps)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    for _ in range(200):  # Increased iterations for refinement
        if time.time() - start_time >= budget_s:
