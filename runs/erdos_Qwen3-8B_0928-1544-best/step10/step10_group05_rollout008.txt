import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # Check if initial_h_values is available

        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    # Initialize a as initial_h - 0.5
    a = initial_h - 0.5
    # Now, a is in [-0.5, 0.5] with sum zero
    # Compute initial best
    def compute_c5(a_vec):
        h = a_vec + 0.5
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_a = a.copy()
    best_c5 = compute_c5(best_a)
    
    T = 1.0
    cooling_rate = 0.99
    max_iterations = 10000
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time >= budget_s:
            break
        # Generate candidate
        a_candidate = best_a.copy()
        # Add noise
        noise = np.random.normal(0, 0.1, n_points)
        a_candidate += noise
        # Clip to bounds
        a_candidate = np.clip(a_candidate, -0.5, 0.5)
        # Adjust sum to zero
        sum_candidate = a_candidate.sum()
        adjustment = -sum_candidate / n_points
        a_candidate += adjustment
        a_candidate = np.clip(a_candidate, -0.5, 0.5)
        # Compute C5
        new_c5 = compute_c5(a_candidate)
        # Acceptance
        if new_c5 < best_c5:

        else:

            if np.random.rand() < np.exp(-delta / T):
                best_a = a_candidate.copy()
                best_c5 = new_c5
        # Cool temperature
        T *= cooling_rate
    
    # Convert best_a back to h_values
    h_values = best_a + 0.5
    return h_values, best_c5, n_points

import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    a = initial_h - 0.5
    def compute_c5(a_vec):
        h = a_vec + 0.5
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    current_a = a.copy()
    best_a = current_a.copy()
    best_c5 = compute_c5(best_a)
    
    T = 1.0
    cooling_rate = 0.99
    max_iterations = 10000
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time >= budget_s:
            break
        a_candidate = current_a.copy()
        noise = np.random.normal(0, 0.05, n_points)
        a_candidate += noise
        a_candidate = np.clip(a_candidate, -0.5, 0.5)
        sum_candidate = a_candidate.sum()
        adjustment = -sum_candidate / n_points
        a_candidate += adjustment
        a_candidate = np.clip(a_candidate, -0.5, 0.5)
        new_c5 = compute_c5(a_candidate)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_a = a_candidate.copy()
            current_a = a_candidate.copy()
        else:
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                current_a = a_candidate.copy()
        T *= cooling_rate
    
    h_values = best_a + 0.5
    return h_values, best_c5, n_points
