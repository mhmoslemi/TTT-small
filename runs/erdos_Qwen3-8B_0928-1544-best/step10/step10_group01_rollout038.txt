        particles.append(h_scaled)
        personal_bests.append(h_scaled.copy())
        velocities.append(np.zeros(n_points))
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    global_best = particles[0].copy()
    global_best_c5 = compute_c5(global_best)
    
    w, c1, c2 = 0.9, 2.0, 2.0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * w + c1 * r1 * (personal_bests[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            velocities[i] = vel
            new_h = particles[i] + vel
            new_h = np.clip(new_h, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new == 0:
                continue
            new_h = new_h * (ones_count / sum_new)
            particles[i] = new_h
            current_c5 = compute_c5(particles[i])
            if current_c5 < compute_c5(personal_bests[i]):
                personal_bests[i] = particles[i].copy()
                if current_c5 < global_best_c5:
                    global_best = particles[i].copy()
                    global_best_c5 = current_c5
            if current_c5 < best_c5:
                best_h = particles[i].copy()
                best_c5 = current_c5
    
    # Gradient descent
    current_c5 = compute_c5(best_h)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    max_k = np.argmax(corr)
    eps = 1e-5
    grad = np.zeros(n_points)
    for j in range(n_points):
        h_perturb = best_h.copy()
        h_perturb[j] += eps
        sum_perturb = np.sum(h_perturb)
        if sum_perturb == 0:
            continue
        h_perturb = h_perturb * (ones_count / sum_perturb)
        c5_perturb = compute_c5(h_perturb)
        grad[j] = (c5_perturb - current_c5) / eps
        h_perturb[j] -= 2 * eps
        c5_perturb = compute_c5(h_perturb)
        grad[j] = (current_c5 - c5_perturb) / eps
    
    learning_rate = 0.01
    max_steps = 1000
    for _ in range(max_steps):
        new_h = best_h - learning_rate * grad
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new == 0:
            break
        new_h = new_h * (ones_count / sum_new)
        new_c5 = compute_c5(new_h)
        if new_c5 < current_c5:
            best_h = new_h
            current_c5 = new_c5
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_perturb = best_h.copy()
                h_perturb[j] += eps
                sum_perturb = np.sum(h_perturb)
                if sum_perturb == 0:
                    continue
                h_perturb = h_perturb * (ones_count / sum_perturb)
                c5_perturb = compute_c5(h_perturb)
                grad[j] = (c5_perturb - current_c5) / eps
                h_perturb[j] -= 2 * eps
                c5_perturb = compute_c5(h_perturb)
                grad[j] = (current_c5 - c5_perturb) / eps
        else:
            break
    
    return best_h, current_c5, n_points
