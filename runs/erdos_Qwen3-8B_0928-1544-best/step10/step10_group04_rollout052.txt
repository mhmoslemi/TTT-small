    dx = 2.0 / n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (ones_count / h_sum)
    
    a = h - 0.5
    a_sum = np.sum(a)
    if not np.isclose(a_sum, 0):
        adjustment = -a_sum / len(a)
        a += adjustment
    
    n_left = n_points // 2
    a_left = a[:n_left]
    a_right = a[n_left:]
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    def compute_c5(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time >= budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap_idx = np.argmax(corr)
        k_worst = max_overlap_idx - (n_points - 1)
        
        def obj_left(x):
            new_a = x.copy()
            h_vals = np.zeros(n_points)
            h_vals[:n_left] = 0.5 + new_a
            h_vals[n_left:] = 0.5 + a_right
            overlap = 0
            for j in range(n_points):
                j_new = (j + k_worst) % n_points
                val_left = h_vals[j]
                val_right = 1 - h_vals[j_new]
                overlap += val_left * val_right
            return overlap * dx
        
        bounds_left = [(-0.5, 0.5) for _ in range(n_left)]
        cons_left = {'type': 'eq', 'fun': lambda x: np.sum(x) + np.sum(a_right)}
        res_left = minimize(obj_left, a_left, method='SLSQP', bounds=bounds_left, constraints=cons_left)
        if res_left.success:
            a_left = res_left.x
            a[:n_left] = a_left
        
        def obj_right(x):
            new_a = x.copy()
            h_vals = np.zeros(n_points)
            h_vals[:n_left] = 0.5 + a_left
            h_vals[n_left:] = 0.5 + new_a
            overlap = 0
            for j in range(n_points):
                j_new = (j + k_worst) % n_points
                val_left = h_vals[j]
                val_right = 1 - h_vals[j_new]
                overlap += val_left * val_right
            return overlap * dx
        
        bounds_right = [(-0.5, 0.5) for _ in range(n_points - n_left)]
        cons_right = {'type': 'eq', 'fun': lambda x: np.sum(x) + np.sum(a_left)}
        res_right = minimize(obj_right, a_right, method='SLSQP', bounds=bounds_right, constraints=cons_right)
        if res_right.success:
            a_right = res_right.x
            a[n_left:] = a_right
        
        h = a + 0.5
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
    
    return best_h, best_c5, n_points
