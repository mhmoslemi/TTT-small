            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            current_c5 = np.max(np.correlate(population[i], 1 - population[i], mode='full')) * (2.0 / n_points)
            new_c5 = np.max(np.correlate(u_scaled, 1 - u_scaled, mode='full')) * (2.0 / n_points)
            if new_c5 < current_c5:
                population[i] = u_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u_scaled.copy()
    
    return best_h, best_c5, n_points
