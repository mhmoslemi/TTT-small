    for gen in range(max_generations):
        if time.time() - start_time >= budget_s:
            break
        new_population = []
        for _ in range(population_size):
            candidates = np.random.choice(population, tournament_size, replace=False)
            parent1 = min(candidates, key=lambda x: compute_c5(x))
            candidates = np.random.choice(population, tournament_size, replace=False)
            parent2 = min(candidates, key=lambda x: compute_c5(x))
            if np.random.rand() < crossover_rate:
                point = np.random.randint(1, n_points)
                child = np.concatenate([parent1[:point], parent2[point:]])
            else:
                child = parent1.copy()
            if np.random.rand() < mutation_rate:
                perturbation = np.random.uniform(-0.1, 0.1, n_points)
                child += perturbation
                a_sum = child.sum()
                child = child - a_sum / n_points
                child = np.clip(child, -0.5, 0.5)
            new_population.append(child.copy())
        population = new_population
        for i in range(population_size):
            current_c5 = compute_c5(population[i])
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_a = population[i].copy()
    
    best_h = best_a + 0.5
    return best_h, best_c5, n_points
