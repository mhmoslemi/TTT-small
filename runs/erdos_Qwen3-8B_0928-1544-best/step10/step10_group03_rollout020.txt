import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (ones_count / h_sum)
        else:
            h = np.zeros(n_points)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()
    alpha = 0.1
    max_iter = 2000
    improve_threshold = 1e-5
    improvement = True

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k_shift = k_max - (n_points - 1)
        gradient = np.zeros_like(h)
        for j in range(n_points):
            if 0 <= j + k_shift < n_points:
                gradient[j] += (1 - h[j + k_shift]) * (2.0 / n_points)
            if 0 <= j - k_shift < n_points:
                gradient[j] -= h[j - k_shift] * (2.0 / n_points)
        h -= alpha * gradient
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (ones_count / h_sum)
        else:
            h = np.zeros(n_points)
        current_c5 = objective(h)
        if current_c5 < best_c5 - improve_threshold:
            best_c5 = current_c5
            best_h = h.copy()
            improvement = True
        else:
            improvement = False
        if not improvement:
            alpha = max(alpha * 0.5, 1e-5)
    
    return best_h, best_c5, n_points
