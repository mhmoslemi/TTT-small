def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

    pop_size = 50

    for _ in range(pop_size - 1):
        h = initial_h.copy()

        h += noise
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (ones_count / h_sum)
        population.append(h.copy())
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = objective(best_h)
    start_time = time.time()

            mutant = population[a] + F * (population[b] - population[c])
            trial = np.copy(mutant)
