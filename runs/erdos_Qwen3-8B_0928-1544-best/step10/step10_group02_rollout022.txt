import numpy as np
from scipy.stats.qmc import SobolSequence
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        n_segments = 10
        sobol = SobolSequence(d=1, scramble=True)
        points = sobol.random(n=n_segments + 1) * 2.0
        points.sort()
        intervals = []
        for i in range(n_segments):
            start = points[i]
            end = points[i + 1]
            intervals.append((start, end, 1.0))
        total_length = sum(end - start for start, end, _ in intervals)
        for i in range(n_segments):
            start, end, height = intervals[i]
            intervals[i] = (start, end, height / total_length)
        h = np.zeros(n_points)
        for start, end, height in intervals:
            start_idx = int(start / dx)
            end_idx = int(end / dx)
            if start_idx >= end_idx:
                start_idx = end_idx - 1
            h[start_idx:end_idx] = height
        initial_h = h

    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    F = 0.7
    CR = 0.8
    pop_size = 30

    population = [h.copy()]
    for _ in range(pop_size - 1):
        h = initial_h.copy()
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (ones_count / h_sum)
        population.append(h.copy())

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for _ in range(200):
        if time.time() - start_time >= budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = np.copy(mutant)
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum > 0:
                trial = trial * (ones_count / trial_sum)
            current_c5 = objective(trial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = trial.copy()
            if current_c5 < objective(population[i]):
                population[i] = trial.copy()

    return best_h, best_c5, n_points
