from scipy.fft import fft, ifft

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    # Initialize h
    if 'initial_h_values' in globals():

    # Ensure sum constraint
    h = h * (ones_count / np.sum(h))
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    max_iter = 10000
    learning_rate = 0.1
    # Ensure Hermitian symmetry
    def ensure_hermitian(H):
        for k in range(1, n_points // 2 + 1):
            H[k] = np.conj(H[n_points - k])
            H[n_points - k] = np.conj(H[k])
        return H

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:

        H = fft(h)
        # Enforce DC constraint
        H[0] = ones_count
        # Ensure Hermitian symmetry
        H = ensure_hermitian(H)
