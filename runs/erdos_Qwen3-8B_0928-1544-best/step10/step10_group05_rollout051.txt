import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals():

        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * 2.0 / n_points
    
    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for _ in range(600):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([compute_c5(ind) for ind in population])
        parents = []

            tournament_size = 5

        split = np.random.randint(1, n_points)
