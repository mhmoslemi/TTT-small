            if 0 <= j < n_points:
                gradient[i] += (1 - h[j]) * dx
                gradient[j] -= h[i] * dx
        
        # Update h
        h = h - alpha * gradient
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.zeros(n_points)
        else:
            h = h * (ones_count / h_sum)
        
        # Random perturbation
        if np.random.rand() < perturb_prob:
            noise = np.random.randn(n_points) * 0.01
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (ones_count / np.sum(h))
        
        # Adjust alpha
        if current_max < best_c5:
            alpha *= decay_rate
            if alpha < 1e-6:
                alpha = 1e-6
        else:
            alpha = max(alpha * 0.95, 1e-6)
        
        # Early exit if improvement is minimal
        if np.abs(best_c5 - 0.38085) < 1e-6:
            break
    
    return best_h, best_c5, n_points
