import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    dx = 2.0 / n_points

    def compute_c5(h_vals):
        one_minus_h = 1.0 - h_vals
        corr = np.correlate(h_vals, one_minus_h, mode='full')
        return np.max(corr) * dx

    def clip_and_adjust(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        total = np.sum(h_vals)
        h_vals -= total / n_points
        return h_vals

    # Initialize with initial_h
    h = clip_and_adjust(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # PSO parameters
    population_size = 30
    particles = [h.copy() for _ in range(population_size)]
    velocities = [np.zeros(n_points) for _ in range(population_size)]
    pbest = [p.copy() for p in particles]
    pbest_fitness = [compute_c5(p) for p in particles]
    gbest = particles[np.argmin(pbest_fitness)]
    gbest_fitness = min(pbest_fitness)

    max_iter = 1000
