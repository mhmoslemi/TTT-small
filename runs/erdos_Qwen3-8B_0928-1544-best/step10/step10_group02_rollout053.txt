import numpy as np
import time
import scipy.optimize as optimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2
    
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    
    def c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def update_h():
        nonlocal h
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (ones_count / h_sum)
    
    update_h()
    best_c5 = c5(h)
    best_h = h.copy()
    start_time = time.time()
    
    n_segments = 10
    start = np.linspace(0, 2 - 2.0/n_segments, n_segments)
    end = start + 2.0 / n_segments
    h_seg = np.full(n_segments, 0.5) * (ones_count / n_segments)
    
    def seg_to_h(seg, n_points):
        h_vals = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            for j in range(n_segments):
                if start[j] <= x < end[j]:
                    h_vals[i] = seg[j]
                    break
        return h_vals
    
    def objective(seg):
        h_vals = seg_to_h(seg, n_points)
        return c5(h_vals)
    
    def constraint(seg):
        return np.sum((end - start) * seg) - 1
    
    cons = {'type': 'eq', 'fun': constraint}
    
    for _ in range(200):
