import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        a = initial_h_values - 0.5
    else:
        a = np.random.uniform(-0.5, 0.5, n_points)
    sum_a = np.sum(a)
    if not np.isclose(sum_a, 0):
        a += -sum_a / n_points
    best_a = a.copy()
    best_c5 = np.max(np.correlate(0.5 + a, 1 - (0.5 + a), mode='full')) * dx
    start_time = time.time()
    population = [a.copy() for _ in range(20)]
    F = 0.8
    CR = 0.9
    for gen in range(300):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for i in range(len(population)):
            idxs = np.random.choice(len(population), 3, replace=False)
            a1, a2, a3 = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            mutant = a1 + F * (a2 - a3)
            mutant = np.where(np.random.rand(n_points) < CR, mutant, population[i])
