from scipy.optimize import minimize

        total = np.sum(h)
        return h * (n_points / 2) / total if total != 0 else np.zeros_like(h)
    
    population = [repair(h) for h in population]
    best_h, best_c5 = population[0], evaluate_erdos_solution(population[0])

        if fitness[best_idx] < best_c5:
