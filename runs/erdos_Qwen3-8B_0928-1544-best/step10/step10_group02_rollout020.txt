import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total > 0:
            h = h * (ones_count / total)
        return h
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    h = clip_and_normalize(initial_h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    time_start = time.time()
    
    # Hyperparameters
    pop_size = 30
    F = 1.0
    CR = 0.3
    generations = 150
    n_gen = 0
    best_log = []
    
    while time.time() - time_start < budget_s and n_gen < generations:
        n_gen += 1
        population = [clip_and_normalize(h.copy()) for _ in range(pop_size)]
        
        # Initial population
        for _ in range(pop_size):
            mutant = np.copy(population[0])
            for i in range(n_points):
                if np.random.rand() < CR:
                    idx = np.random.randint(pop_size)
                    mutant[i] = population[idx][i] + F * (population[np.random.randint(pop_size)][i] - population[np.random.randint(pop_size)][i])
            population.append(clip_and_normalize(mutant))
        
        # Evaluate and select
        for idx in range(len(population)):
            current_c5 = compute_c5(population[idx])
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = population[idx].copy()
            if len(best_log) < 3:
                best_log.append((n_gen, current_c5))
        
        # Tournament selection
        new_pop = []
        for _ in range(pop_size):
            a, b, c = np.random.choice(len(population), 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = np.copy(mutant)
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[np.random.randint(pop_size)][j]
            trial = clip_and_normalize(trial)
            current_c5 = compute_c5(trial)

                new_pop.append(trial)
            else:
                new_pop.append(population[np.random.randint(len(population))])
        
        population = new_pop
