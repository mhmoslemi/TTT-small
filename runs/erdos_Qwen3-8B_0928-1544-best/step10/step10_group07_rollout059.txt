    result = minimize(objective, h, method='L-BFGS-B', jac=jacobian, options={'maxiter': 100, 'ftol': 1e-8})

    # Refine with perturbations
    for _ in range(200):
        if time.time() - start_time >= budget_s:
            break
        perturbation = np.random.normal(0, 0.1, size=n_points)
        trial_h = result.x + perturbation
        trial_h = clip_and_rescale(trial_h)
        trial_c5 = compute_c5(trial_h)
        if trial_c5 < best_c5:
            best_c5 = trial_c5
            best_h = trial_h

    return best_h, best_c5, n_points
