def run(seed=42, budget_s=1100, **kwargs):

        return np.max(corr) * 2.0 / n_points

        h_clip = np.clip(h_vec, 0.0, 1.0)

        return np.clip((h_clip - total / n_points), 0.0, 1.0)

    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5

    beta = 0.5
    mu = 0.1

        grad = np.zeros(n_points)
        for k in range(-n_points + 1, n_points):
