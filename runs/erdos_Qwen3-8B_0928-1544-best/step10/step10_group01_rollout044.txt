                population[i] = u_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u_scaled.copy()
        
        # Adjust parameters over time
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 300:
            mutation_factor = 0.8  # Final phase: higher exploration
            crossover_rate = 0.9
        elif remaining < 600:
            mutation_factor = 0.7
            crossover_rate = 0.85
        else:
            mutation_factor = 0.6
            crossover_rate = 0.8
    
    # Gradient descent refinement
    max_iter = 100
    learning_rate = 0.01
    for _ in range(max_iter):
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_index = np.argmax(corr)
        k = k_index - (n_points - 1)
        if k < 0:
            k = 0
        # Compute gradient for this k
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k < n_points:
                grad[i] -= best_h[i + k]
            if i - k >= 0:
                grad[i] += (1 - best_h[i - k])
        # Project gradient onto constraint space
        grad_proj = grad - (np.sum(grad) / n_points) * np.ones(n_points)
        # Update with learning rate
        best_h += learning_rate * grad_proj
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = best_h * (ones_count / np.sum(best_h))
    
    return best_h, best_c5, n_points
