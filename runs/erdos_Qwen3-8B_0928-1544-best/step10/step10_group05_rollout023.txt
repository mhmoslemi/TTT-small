    initial_h = np.clip(initial_h * scale, 0, 1)
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    beta = 0.1
    decay_rate = 0.99
    max_iter = 10000
    start_time = time.time()
    
    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        cross_corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(cross_corr)
        lag_val = max_idx - (n_points - 1)
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus = i + lag_val
            i_minus = i - lag_val
            term1 = 0.0
            if 0 <= i_plus < n_points:
                term1 = 1.0 - h[i_plus]
            term2 = 0.0
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            gradient[i] = term1 - term2
        
        h -= beta * gradient
        h = np.clip(h, 0, 1)
        sum_h = h.sum()
        scale = (n_points / 2) / (sum_h + 1e-12)
        h = np.clip(h * scale, 0, 1)
        
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        beta *= decay_rate
    
    return best_h, best_c5, n_points
