import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2
    population_size = 30  # Smaller population size for efficiency
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        population = [initial_h_scaled]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        valid_indices = np.arange(len(corr))[:n_points]
        return np.max(corr[valid_indices]) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = objective(population[0])
    
    start_time = time.time()
    
    F_start, F_end = 0.5, 0.8
    CR_start, CR_end = 0.8, 0.5
    
    while time.time() - start_time < budget_s:
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            F = F_start + (F_end - F_start) * (time.time() - start_time) / budget_s
            CR = CR_start + (CR_end - CR_start) * (time.time() - start_time) / budget_s
            
            # Mutate: v = a + F*(b - c)
            v = a_vec + F * (b_vec - c_vec)
            
            # Crossover: u is a mix of v and current individual i
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            
            # Clip to [0,1]
            u = np.clip(u, 0.0, 1.0)
            
            # Scale to meet sum constraint
            u_sum = np.sum(u)
            if u_sum == 0:
                continue  # skip if sum is zero (unlikely)
            u_scaled = u * (ones_count / u_sum)
            
            # Evaluate
            new_c5 = objective(u_scaled)
            
            # Check if improvement
            if new_c5 < objective(population[i]):
                population[i] = u_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u_scaled.copy()
    
    # Local refinement
    def refine_h(h):
        for _ in range(100):
            k_vals = np.correlate(h, 1 - h, mode='full')[::n_points]
            worst_k = np.argmax(k_vals)
            h_worst = h.copy()
            for j in range(n_points):
                if np.random.rand() < 0.5:
