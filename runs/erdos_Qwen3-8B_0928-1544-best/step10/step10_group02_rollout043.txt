import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points / 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    initial_h = np.clip(initial_h, 0, 1)
    initial_sum = np.sum(initial_h)

    else:
        initial_h = np.zeros(n_points)
    best_h = initial_h.copy()
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    best_c5 = compute_c5(best_h)
    temperature = 1.0
    cooling_rate = 0.995
    max_iterations = int(budget_s * 0.9)
    start_time = time.time()
    for _ in range(max_iterations):
        if time.time() - start_time >= budget_s:
            break
        noise = np.random.normal(0, 0.1, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum == 0:
            perturbed = np.zeros(n_points)
        else:
            perturbed = perturbed * (ones_count / perturbed_sum)
        new_c5 = compute_c5(perturbed)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = perturbed.copy()
        else:
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / temperature):
                best_h = perturbed.copy()
                best_c5 = new_c5
        temperature *= cooling_rate
    return best_h, best_c5, n_points
