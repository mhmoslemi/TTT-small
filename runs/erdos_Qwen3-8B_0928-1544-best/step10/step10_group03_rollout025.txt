import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    
    best_h = h.copy()
    best_c5 = 0.0
    max_iter = 1000
    iter_count = 0
    alpha = 0.1
    alpha_decay = 0.95
    alpha_increase = 1.05
    improve_threshold = 1e-6
    start_time = time.time()
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            idx1 = (i + k) % n_points
            idx2 = (i - k) % n_points
            term1 = 1.0 - h[idx1]
            term2 = h[idx2]
            grad[i] = dx * (term1 - term2)
        return grad
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        
        grad = compute_gradient(h)
        h_new = h - alpha * grad
        
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        if h_sum == 0:
            h_new = np.zeros(n_points)
        else:
            h_new = h_new * (ones_count / h_sum)
        
        new_c5 = objective(h_new)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            alpha *= alpha_increase
        else:
            alpha *= alpha_decay
        
        h = h_new.copy()
