    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        population = [initial_h_scaled]
    else:
        population = []
        for _ in range(5):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points)
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = population[0].copy()
    best_c5 = compute_c5(population[0])
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        if len(population) < 50:
            for _ in range(50 - len(population)):
                h = np.random.rand(n_points) * 0.5
                h = np.clip(h, 0.0, 1.0)
                h_sum = np.sum(h)
                if h_sum == 0:
                    continue
                h_scaled = h * (ones_count / h_sum)
                population.append(h_scaled)
        
        indices = np.random.choice(len(population), 3, replace=False)
        a, b, c = indices
        a_vec, b_vec, c_vec = population[a], population[b], population[c]
        mutant = a_vec + np.random.rand() * (b_vec - c_vec)
        trial = population[np.random.randint(len(population))].copy()
        
        for i in range(n_points):
            if np.random.rand() < 0.9:
                trial[i] = mutant[i]
        
        trial = np.clip(trial, 0.0, 1.0)
        trial_sum = np.sum(trial)
        if trial_sum == 0:
            continue
        trial = trial * (ones_count / trial_sum)
        
        new_c5 = compute_c5(trial)
        if new_c5 < compute_c5(population[np.argmin([compute_c5(h) for h in population])]):
            population[np.argmin([compute_c5(h) for h in population])] = trial
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = trial
    
    learning_rate = 0.01
    for _ in range(100):
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            if i + max_k < n_points:
                grad[i] += (1 - best_h[i + max_k]) * dx
            if i - max_k >= 0:
                grad[i] -= best_h[i - max_k] * dx
        
        best_h = best_h - learning_rate * grad
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h_sum = np.sum(best_h)
        if best_h_sum == 0:
            break
        best_h = best_h * (ones_count / best_h_sum)
        
        new_c5 = compute_c5(best_h)
        if new_c5 >= best_c5:
            break
    
    return best_h, best_c5, n_points
