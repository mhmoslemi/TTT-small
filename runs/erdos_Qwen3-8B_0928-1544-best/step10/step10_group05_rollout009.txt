import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()

        h = np.full(n_points, 0.5)

        one_minus_h = 1.0 - h_val
        corr = np.correlate(h_val, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * 2.0 / n_points

    best_a = a.copy()
    best_c5 = compute_c5(best_a)

    T = 1.0
    cooling_rate = 0.999
    max_iter = 10000
    delta = 0.1

    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        perturb = np.random.normal(0, delta, size=n_points)
        candidate_a = a + perturb
        candidate_a = np.clip(candidate_a, -0.5, 0.5)
        candidate_a_sum = candidate_a.sum()
        candidate_a = candidate_a - candidate_a_sum / n_points
        candidate_a = np.clip(candidate_a, -0.5, 0.5)
        
        current_c5 = compute_c5(candidate_a)
        
        if current_c5 < best_c5:
            a, best_c5 = candidate_a.copy(), current_c5
        else:
            prob = np.exp(-(current_c5 - best_c5) / T)

                a, best_c5 = candidate_a.copy(), current_c5
        
        T *= cooling_rate
        if T < 0.01:
            delta = 0.01

    best_h = a + 0.5
