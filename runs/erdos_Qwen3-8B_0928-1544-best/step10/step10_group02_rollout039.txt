import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def feasi(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros(n_points)
        return h * (ones_count / s)
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()
    
    F = 0.6
    CR = 0.9
    pop_size = 40
    population = [feasi(initial_h.copy()) for _ in range(pop_size)]

        a, b, c = np.random.choice(pop_size, 3, replace=False)
        mutant = population[a] + F * (population[b] - population[c])
        trial = np.copy(mutant)
        for j in range(n_points):
            if np.random.rand() < CR:
                trial[j] = mutant[j]
            else:
                trial[j] = population[_][j]
        trial = feasi(trial)

        if current_c5 < best_c5:

            population[_] = trial.copy()
