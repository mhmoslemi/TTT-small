import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * (ones_count / np.sum(h))
    h = h * (ones_count / np.sum(h))
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    learning_rate = 0.6
    temperature = 0.5
    decay_lr = 0.998
    decay_temp = 0.995
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + k < n_points:
                grad[i] += 1 - h[i + k]
            if 0 <= i - k < n_points:
                grad[i] -= h[i - k]
        h_new = h - learning_rate * grad
        h_new += temperature * np.random.randn(n_points)
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        if h_sum == 0:
            continue
        h_new = h_new * (ones_count / h_sum)
        new_corr = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(new_corr) * (2.0 / n_points)
        if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5) / temperature):
            best_c5 = new_c5
            best_h = h_new.copy()
        learning_rate *= decay_lr
        temperature *= decay_temp
    
    return best_h, best_c5, n_points
