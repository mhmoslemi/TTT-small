    corr = np.correlate(h, 1 - h, mode='full')

    best_h = h.copy()
    best_c5 = current_c5
    # Parameters
    learning_rate = 0.1
    temp = 0.5
    decay_lr = 0.999
    decay_temp = 0.99
    time_start = time.time()
    while time.time() - time_start < budget_s:
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        idx = np.argmax(corr)
        lag = idx - (n_points -1)
        # Compute gradient
        grad = np.zeros(n_points)
        for j in range(n_points):
            if (j + lag) >= 0 and (j + lag) < n_points:
                grad[j] += 1 - h[j + lag]
            i = j - lag
            if i >=0 and i < n_points:
                grad[j] -= h[i]
        # Update h
        new_h = h - learning_rate * grad
        # Add simulated annealing noise
        noise = np.random.normal(0, temp, n_points)
        new_h += noise
        # Clip to [0, 1]
        new_h = np.clip(new_h, 0, 1)
        # Scale to ensure sum is ones_count
        new_h_sum = np.sum(new_h)
        if new_h_sum == 0:
            continue
        scale_factor = ones_count / new_h_sum
        new_h = new_h * scale_factor
        # Compute new_c5
        corr_new = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(corr_new) * (2.0 / n_points)
        # Acceptance criteria
        if new_c5 < best_c5:
            h = new_h.copy()
            best_c5 = new_c5
        else:
            # Accept with probability
            prob = np.exp((best_c5 - new_c5) / temp)
            if np.random.rand() < prob:
                h = new_h.copy()
                best_c5 = new_c5

        temp *= decay_temp
