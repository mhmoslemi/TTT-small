import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    
    h_sum = np.sum(h)
    scale = (n_points / 2) / h_sum if h_sum > 0 else 0
    h = h * scale
    
    def _compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)
    
    T = 1.0
    cooling_rate = 0.995
    min_temp = 1e-5
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        perturbation = np.random.normal(0, 0.1, n_points)
        trial = h + perturbation
        trial = np.clip(trial, 0, 1)
        trial_sum = np.sum(trial)
        if trial_sum == 0:
            trial = np.zeros(n_points)
        else:
            trial = trial * (n_points / 2) / trial_sum
        
        new_c5 = _compute_c5(trial)
        if new_c5 < best_c5:
            h = trial.copy()
            best_c5 = new_c5
        else:
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                h = trial.copy()
        
        T *= cooling_rate
        if T < min_temp:
            break
    
    return best_h, best_c5, n_points
