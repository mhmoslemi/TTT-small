        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx

    for _ in range(300):
        if time.time() - start_time >= budget_s:
            break
        pop_fitness = [compute_c5(0.5 + vec) for vec in population]
        worst_idx = np.argmax(pop_fitness)
        a1, a2, a3 = np.random.choice(population, 3, replace=False)
        mutant = a1 + 0.8 * (a2 - a3)
        trial = population[worst_idx].copy()
        for i in range(n_points):
            if np.random.rand() < 0.9:
                trial[i] = mutant[i]
        clipped = np.clip(trial, -0.5, 0.5)
        s = np.sum(clipped)
        adjusted = clipped - s / n_points
        trial_c5 = compute_c5(0.5 + adjusted)
        if trial_c5 < pop_fitness[worst_idx]:
            population[worst_idx] = adjusted
            current_h = 0.5 + adjusted
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

    return best_h, best_c5, n_points
