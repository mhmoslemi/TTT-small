        h_new = np.clip(h_new, 0.0, 1.0)
        h_sum_new = np.sum(h_new)
        if h_sum_new == 0:
            continue
        h_new = h_new * (ones_count / h_sum_new)
        
        current_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full') * (2.0 / n_points))
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
    
    return best_h, best_c5, n_points
