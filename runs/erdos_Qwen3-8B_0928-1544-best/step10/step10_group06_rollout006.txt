        if sum_h != 0:
            h = h * (n_points / 2.0) / sum_h
        h = np.clip(h, 0, 1)
    def compute_max_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        corr_pos = corr[n_points-1:2*n_points-1]
        return np.max(corr_pos) * (2.0 / n_points)
    c5_best = compute_max_c5(h)
    start_time = time.time()
    alpha = 0.01
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        corr_pos = corr[n_points-1:2*n_points-1]
        k_max_index = np.argmax(corr_pos)
        shift_k = k_max_index
        gradient = np.zeros(n_points)
        for i in range(n_points - shift_k):
            gradient[i] += (1 - h[i + shift_k])
            gradient[i + shift_k] -= h[i]
        h = h - alpha * gradient
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n_points / 2.0) / sum_h
        h = np.clip(h, 0, 1)
        new_c5 = compute_max_c5(h)
        if new_c5 < c5_best:
            c5_best = new_c5
    return h, c5_best, n_points
