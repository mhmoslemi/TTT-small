            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = np.clip(h + perturbation, 0, 1)
        perturbed_h = project_h(perturbed_h)
        perturbed_c5 = compute_c5(perturbed_h)
        if perturbed_c5 < best_c5:
            best_c5 = perturbed_c5
            best_h = perturbed_h.copy()
        if time.time() - start_time >= budget_s:
            break
    
    return best_h, best_c5, n_points
