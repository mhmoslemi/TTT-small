import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(n_points, 0.5)
    
    a = initial_h - 0.5
    a_sum = a.sum()
    a = a - a_sum / n_points
    a = np.clip(a, -0.5, 0.5)
    
    population_size = 50  # Increased to search more thoroughly
    particles = [a.copy() for _ in range(population_size)]
    velocities = [np.zeros(n_points) for _ in range(population_size)]
    
    pbest = [a.copy() for _ in range(population_size)]
    pbest_fitness = [float('inf') for _ in range(population_size)]
    gbest = a.copy()
    gbest_fitness = float('inf')
    
    def compute_c5(a_vec):
        h = a_vec + 0.5
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * 2.0 / n_points
    
    # Evaluate initial particles
    for i in range(population_size):
        current_fitness = compute_c5(particles[i])
        pbest_fitness[i] = current_fitness
        if current_fitness < gbest_fitness:
            gbest = particles[i].copy()
            gbest_fitness = current_fitness
    
    w = 0.6  # Slightly reduced inertia for more exploration
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000  # Increased to allow more exploration
    
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        for i in range(population_size):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            # Clip to [-0.5, 0.5]
            particles[i] = np.clip(particles[i], -0.5, 0.5)
            # Adjust sum to zero
            sum_p = particles[i].sum()
            particles[i] -= sum_p / n_points
            particles[i] = np.clip(particles[i], -0.5, 0.5)
            current_fitness = compute_c5(particles[i])
            if current_fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = current_fitness

                    gbest_fitness = current_fitness
