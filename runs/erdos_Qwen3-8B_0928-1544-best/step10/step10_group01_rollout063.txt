    velocities = [np.random.rand(n_points) * 0.1 for _ in range(population_size)]

    start_time = time.time()
    while time.time() - start_time < budget_s * 0.8:
        for i in range(population_size):
            v = velocities[i]
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            cognitive = r1 * (pbest[i] - population[i])
            social = r2 * (gbest - population[i])
            v = 0.9 * v + 1.5 * cognitive + 1.5 * social
            velocities[i] = v
            population[i] += v
            population[i] = np.clip(population[i], 0.0, 1.0)
            population[i] = population[i] * (ones_count / np.sum(population[i]))

            curr_c5 = objective(population[i])
            if curr_c5 < objective(pbest[i]):
                pbest[i] = np.copy(population[i])
                if curr_c5 < gbest_c5:
                    gbest = np.copy(population[i])
                    gbest_c5 = curr_c5

    h = np.copy(gbest)
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0:
        for _ in range(1000):
            if remaining_time < 0.1:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            k_max_idx = np.argmax(corr)
            k_max = k_max_idx - n_points + 1
            grad = np.zeros(n_points)
            for i in range(n_points):
                j1 = (i + k_max) % n_points
                j2 = (i - k_max) % n_points
                grad[i] = (1 - h[j1]) - h[j2]
            h -= 0.01 * grad
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            h = h * (ones_count / h_sum)
            new_c5 = objective(h)
            if new_c5 < gbest_c5:
                gbest = np.copy(h)
                gbest_c5 = new_c5
            remaining_time -= 0.1

    return gbest, gbest_c5, n_points
