import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    
    def golden_ratio_initial(n_points):
        phi = (1 + np.sqrt(5)) / 2
        positions = []
        for i in range(ones_count):
            pos = 2.0 * ((i + 1) * phi) % 1.0
            positions.append(pos)
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        for pos in positions:
            idx = int(np.round(pos / dx))
            if 0 <= idx < n_points:
                h[idx] = 1
        return h
    
    if 'initial_h_values' in globals():

        population = [initial_h_scaled]
        for _ in range(49):
            h = np.random.rand(n_points) * 0.8
            h_sum = np.sum(h)
            if h_sum == 0:

            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    else:
        population = []
        initial_h = golden_ratio_initial(n_points)
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        population.append(initial_h_scaled)
        for _ in range(49):
            h = np.random.rand(n_points) * 0.8
            h_sum = np.sum(h)

                h = np.random.rand(n_points) * 0.8
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()

    start_time = time.time()

            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
