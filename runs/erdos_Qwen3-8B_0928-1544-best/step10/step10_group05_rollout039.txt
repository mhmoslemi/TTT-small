import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = 0.5 * np.ones(n_points)
    
    # Perturb and project to feasible region
    h += 0.01 * np.random.randn(n_points)
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    sum_h = h.sum()
    scale = target_sum / sum_h
    h = h * scale
    h = np.clip(h, 0, 1)
    
    best_c5 = float('inf')
    best_h = h.copy()
    beta = 0.1
    max_iter = 1000
    decay_rate = 0.999
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        
        # Compute cross-correlation and find maximum lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        lag = max_corr_idx - (n_points - 1)
        
        # Compute gradient for the maximum lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                grad[i] = 1 - h[j]
        
        # Update h using gradient descent
        h -= beta * grad
        
        # Clip to [0,1] and scale back to target sum
        h = np.clip(h, 0, 1)
        sum_h = h.sum()
        scale = target_sum / sum_h
        h = h * scale
        h = np.clip(h, 0, 1)
        
        # Compute C5 and update best solution
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * 2.0 / n_points
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Decay learning rate every 100 iterations
        if _ % 100 == 0 and _ > 0:
            beta *= decay_rate
    
    return best_h, best_c5, n_points
