        maxiter=200,
        polish=True
    )
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return h, compute_c5(h), n_points
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
