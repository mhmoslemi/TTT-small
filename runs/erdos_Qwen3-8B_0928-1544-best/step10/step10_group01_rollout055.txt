        eps = 1e-5
        for i in range(n_points):
            h_p = h.copy()
            h_p[i] += eps
            h_p = np.clip(h_p, 0, 1)
            h_p_sum = np.sum(h_p)
            if h_p_sum == 0:
                continue
            h_p = h_p * (ones_count / h_p_sum)
            c5_p = objective(h_p)
            h_m = h.copy()
            h_m[i] -= eps
            h_m = np.clip(h_m, 0, 1)
            h_m_sum = np.sum(h_m)
            if h_m_sum == 0:
                continue
            h_m = h_m * (ones_count / h_m_sum)
            c5_m = objective(h_m)
            grad[i] = (c5_p - c5_m) / (2 * eps)
        return grad
    
    h_current = global_best.copy()
    learning_rate = 0.01
    for _ in range(100):
        grad = compute_grad(h_current)
        h_new = h_current - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        if h_new_sum == 0:
            break
        h_new = h_new * (ones_count / h_new_sum)
        new_c5 = objective(h_new)
        if new_c5 < objective(h_current):
            h_current = h_new.copy()
        else:
            break
    
    c5_final = objective(h_current)
    return h_current, c5_final, n_points
