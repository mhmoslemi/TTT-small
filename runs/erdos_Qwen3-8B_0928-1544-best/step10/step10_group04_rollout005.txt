    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    for _ in range(150):  # Increased iteration count
        if time.time() - start_time >= budget_s:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)

        # Optimize left and right blocks
        def obj(x, k_val, h, dx):
            new_h = h.copy()
            new_h[:n_points//2] = x
            corr = np.correlate(new_h, 1 - new_h, mode='full')
            return corr[k_val] * dx

        bounds = [(0., 1.) for _ in range(n_points//2)]
        res = minimize(obj, x0=h[:n_points//2], args=(k, h, dx), bounds=bounds, method='L-BFGS-B')
        if res.success:
            h[:n_points//2] = res.x

        def obj2(x, k_val, h, dx):
            new_h = h.copy()
            new_h[n_points//2:] = x
            corr = np.correlate(new_h, 1 - new_h, mode='full')
            return corr[k_val] * dx

        bounds2 = [(0., 1.) for _ in range(n_points - n_points//2)]
        res2 = minimize(obj2, x0=h[n_points//2:], args=(k, h, dx), bounds=bounds2, method='L-BFGS-B')
        if res2.success:
            h[n_points//2:] = res2.x

        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    return best_h, best_c5, n_points
