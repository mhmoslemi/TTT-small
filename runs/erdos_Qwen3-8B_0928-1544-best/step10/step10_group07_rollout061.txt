    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / n_points)
    start_time = time.time()
    
    # First optimization: SLSQP
    result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints)
    best_h = result.x
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / n_points)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    # Perturb and re-optimize
    perturbation = np.random.normal(0, 0.05, size=n_points)
    perturbed_h = best_h + perturbation
    perturbed_h = np.clip(perturbed_h, 0, 1)
    perturbed_h_sum = np.sum(perturbed_h)
    if perturbed_h_sum > 0:
        perturbed_h = perturbed_h * (sum_h / perturbed_h_sum)
    
    # Second optimization
    result_perturbed = opt.minimize(objective, perturbed_h, method='SLSQP', bounds=bounds, constraints=constraints)
    best_h = result_perturbed.x
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / n_points)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
