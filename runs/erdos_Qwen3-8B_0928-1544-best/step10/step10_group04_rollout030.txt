        if current_c5 == 0:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        relevant_corr = corr[n_points - 1: n_points - 1 + n_points]
        worst_k = np.argmax(relevant_corr)
        worst_overlap = relevant_corr[worst_k]

        def objective(x):
            k = worst_k
            shifted = np.roll(x, k)
            return np.dot(x, 1 - shifted) * dx

        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count}]
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons)
        if res.success:
            h = np.clip(res.x, 0, 1)
        else:
            break

    return best_h, best_c5, n_points
