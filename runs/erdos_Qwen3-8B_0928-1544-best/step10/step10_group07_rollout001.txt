import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    sum_h = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (sum_h / h_sum)
    dx = 2.0 / n_points

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        overlap_integrals = corr * dx
        return np.max(overlap_integrals)

    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (sum_h / h_sum)
        return h

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - sum_h})

    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 500, 'ftol': 1e-6})
    best_h = clip_and_scale(result.x)
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
