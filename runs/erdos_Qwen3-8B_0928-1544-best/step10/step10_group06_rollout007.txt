def run(seed=42, budget_s=1100, **kwargs):

        max_corr = np.max(corr)
        return max_corr * (2.0 / n_points)

    def refine_candidate(h, n_points, steps=15, step_size=0.01):
        h = h.copy()
        for _ in range(steps):

            max_index = np.argmax(corr)
            lag = max_index - (n_points - 1)
            gradient = np.zeros(n_points)
            k = lag
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    gradient[i] += (1 - h[j])
                    gradient[j] -= h[i]
            h -= step_size * gradient
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            if current_sum != 0:
                scaling = (n_points / 2) / current_sum
                h = h * scaling
            h = np.clip(h, 0, 1)
        return h
