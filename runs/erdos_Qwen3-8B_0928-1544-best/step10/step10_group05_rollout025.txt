import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    initial_h_values = kwargs.get('initial_h_values', None)
    
    if initial_h_values is not None:
        a = initial_h_values - 0.5
    else:
        a = np.full(n_points, 0.0)
    
    def compute_c5(a_vec):
        h = a_vec + 0.5
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * 2.0 / n_points
    
    best_a = a.copy()
    best_c5 = compute_c5(best_a)
    current_a = a.copy()
    
    temp = 0.5
    cooling_rate = 0.995
    max_iter = 10000
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        delta = np.random.normal(0, 0.05, n_points)
        candidate_a = current_a + delta
        candidate_a = np.clip(candidate_a, -0.5, 0.5)
        sum_candidate = candidate_a.sum()
        candidate_a -= sum_candidate / n_points
        
        new_c5 = compute_c5(candidate_a)
        delta_c5 = new_c5 - best_c5

                    if new_c5 < best_c5:

        temp *= cooling_rate
    
    best_h = best_a + 0.5
