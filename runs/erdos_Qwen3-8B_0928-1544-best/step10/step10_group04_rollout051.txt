            trial = np.clip(trial, -0.5, 0.5)
            trial_sum = np.sum(trial)
            adjustment = -trial_sum / n_points
            trial += adjustment
            new_population.append(trial)
        population = new_population
    best_a = population[0]
    h_best = 0.5 + best_a
    h_best = np.clip(h_best, 0, 1)
    corr = np.correlate(h_best, 1 - h_best, mode='full')
    final_c5 = np.max(corr) * dx
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_best
    return best_h, best_c5, n_points
