        x0 = segments[:,0].copy()
        x0 = np.concatenate((x0, segments[:,1].copy()))
        result = minimize(pos_obj, x0, method='L-BFGS-B', bounds=[(0, 2) for _ in range(2 * n_segments)])
        new_segments = np.copy(segments)
        new_segments[:,0] = result.x[:n_segments]
        new_segments[:,1] = result.x[n_segments:]
        h_values = np.zeros(n_points)
        for i in range(n_segments):
            start = new_segments[i,0]
            end = new_segments[i,1]
            val = heights[i]
            idx_start = int(np.round(start / dx))
            idx_end = int(np.round(end / dx))
            h_values[idx_start:idx_end] = val
        h_values = np.clip(h_values, 0, 1)
        h_values = project_heights(h_values, n_segments, n_points, dx)
        return new_segments, h_values
    
    def optimize_heights(segments, heights, k_max):
        def height_obj(heights):
            h_values = np.zeros(n_points)
            for i in range(n_segments):
                start = segments[i,0]
                end = segments[i,1]
                val = heights[i]
                idx_start = int(np.round(start / dx))
                idx_end = int(np.round(end / dx))
                h_values[idx_start:idx_end] = val
            h_values = np.clip(h_values, 0, 1)
            h_values = project_heights(h_values, n_segments, n_points, dx)
            return get_c5(h_values)
        
        result = minimize(height_obj, heights, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_segments)])
        return result.x
    
    best_h = h.copy()
    best_c5 = get_c5(h)
    start_time = time.time()
    
    for _ in range(50):
        if time.time() - start_time >= budget_s:
            break
        k_values = np.arange(-2, 2, 0.1)
        k_max = k_values[np.argmax(np.correlate(h, 1 - h, mode='full') * dx)]
        
        # Optimize positions
        segments, h_values = optimize_position(segments, h, k_max)
        h = h_values.copy()
        
        # Optimize heights
        h = optimize_heights(segments, h, k_max)
        
        current_c5 = get_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Increase segments if stuck
        if np.all(np.abs(np.correlate(h, 1 - h, mode='full') * dx - best_c5) < 1e-8):
            n_segments += 2
            segments = []
            start = 0
            for i in range(n_points):
                if i == 0 or abs(h[i] - h[i-1]) > 1e-8:
                    segments.append((start, i * dx, h[i]))
                    start = i * dx
            segments.append((start, n_points * dx, h[-1]))
            segments = np.array(segments)
            n_segments = len(segments)
    
    return best_h, best_c5, n_points
