            h = np.random.rand(n_points) * 0.3 + 0.35  # Smaller jitter for diversity
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.3 + 0.35
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.3 + 0.35
            h_sum = np.sum(h)

                h = np.random.rand(n_points) * 0.3 + 0.35
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)

    # Objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])

    start_time = time.time()

    while time.time() - start_time < budget_s:

            a, b, c = indices

            c_vec = population[c]
            
            # Mutate: v = a + F*(b - c)
            v = a_vec + F * (b_vec - c_vec)
            
            # Crossover: u is a mix of v and current individual i
            u = np.copy(population[i])
            for j in range(n_points):
