import time

def run(seed=42, budget_s=1100, **kwargs):

                h = np.random.rand(n_points) * 0.5

    start_time = time.time()

    CR = 0.8
    mutation_factor_growth = 0.02
    crossover_rate = 0.8
    
    def gradient_step(h):
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_perturb = h.copy()

    return best_h, best_c5, n_points
