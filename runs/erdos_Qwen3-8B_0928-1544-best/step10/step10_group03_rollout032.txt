                best_h = trial.copy()
            if current_c5 < objective(population[i]):
                population[i] = trial.copy()
        
        # Adaptive parameters based on progress
        progress = (best_c5 - 0.38085) / (best_c5 - 0.380919) if best_c5 > 0.38085 else 1.0
        F *= decay_rate if progress < 0.1 else 1.0
        CR = max(0.7, CR - 0.05 if progress < 0.2 else CR)
        
        # Early exit if improvement stagnates
        if np.abs(best_c5 - 0.38085) < improve_threshold:
            break
    
    return best_h, best_c5, n_points
