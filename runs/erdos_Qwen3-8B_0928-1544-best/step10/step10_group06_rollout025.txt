def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()

            population[i] = np.clip(population[i], 0, 1)

        population = [np.random.rand(n_points) for _ in range(50)]
        for i in range(50):
            population[i] = np.clip(population[i], 0, 1)
            s = np.sum(population[i])
            population[i] *= (n_points / 2) / s

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = eval_c5(best_h)

    pop_size = 50
    generations = 200
    mutation_prob = 0.2

    tournament_size = 5

    start_time = time.time()

        if np.argmin(fitness) < len(population) and fitness[np.argmin(fitness)] < best_c5:
            best_h = population[np.argmin(fitness)].copy()

            t1 = random.sample(range(pop_size), tournament_size)
