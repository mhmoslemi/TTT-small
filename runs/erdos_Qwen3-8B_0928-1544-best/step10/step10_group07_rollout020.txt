        h = h_flat.reshape((n_points,))
        corr = np.correlate(h, 1 - h, mode='full')
        sum_exp = np.sum(np.exp(epsilon * corr))
        approx_max = (1 / epsilon) * np.log(sum_exp)
        return approx_max
    
    # Constraints
    constraints = (
        {'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat) - sum_h},
    )
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run optimization
    result = minimize(
        fun=objective,
        x0=h.flatten(),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Evaluate the exact C5
    best_h = result.x.reshape((n_points,))
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * (2.0 / n_points)
    
    return best_h, best_c5, n_points
