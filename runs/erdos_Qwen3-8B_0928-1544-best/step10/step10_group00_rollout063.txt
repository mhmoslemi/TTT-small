        h_clipped = np.clip(h_vec, 0.0, 1.0)
        h_sum = np.sum(h_clipped)
        if h_sum == 0:
            return np.random.rand(n_points) * 0.5
        return h_clipped * (ones_count / h_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        # Compute current c5
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient
        H_corr = fft.fft(h)
        K = fft.fft(1 - h)
        corr = fft.ifft(H_corr * np.conj(K)).real
        max_idx = np.argmax(corr)
        grad = np.zeros(n_points, dtype=np.complex128)
        if max_idx == 0:
            grad[0] = (n_points - h[0]) * (h[0] - 1)
        else:
            grad[max_idx] = -2 * np.conj(H_corr[max_idx]) * np.conj(K[max_idx])
        
        # Update frequency domain
        H_new = H - lr * grad
        # Enforce Hermitian symmetry
        H_new = np.conj(H_new[::-1])[1:] + H_new
        # Enforce DC component
        H_new[0] = n_points / 2
        h_new = fft.ifft(H_new).real
        h_new = project_h(h_new)
        
        # Update h
        h = h_new
        H = fft.fft(h)
    
    return best_h, best_c5, n_points
