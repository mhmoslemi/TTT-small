import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 100
    max_iter = 1000

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        population = [initial_h_scaled]
    else:
        population = []
    for _ in range(population_size - len(population)):
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h_scaled = h * (ones_count / h_sum)
        population.append(h_scaled)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()
    F = 0.8
    CR = 0.9
    stagnation = 0

    while time.time() - start_time < budget_s:
        iterations = int((time.time() - start_time) / (budget_s / max_iter))
        F = max(0.5, F - 0.0008 * iterations)
        CR = min(0.95, CR + 0.0004 * iterations)
        new_best = False

        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            curr = objective(population[i])
            new_val = objective(u_scaled)
            if new_val < curr:
                population[i] = u_scaled
                if new_val < best_c5:
                    best_c5 = new_val
                    best_h = u_scaled.copy()
                    new_best = True
        if not new_best:
            stagnation += 1
            if stagnation > 100:
                F = 0.8
                CR = 0.9
                stagnation = 0
    return best_h, best_c5, n_points
