import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2
    pop_size = 50
    phi1, phi2 = 1.4, 1.4
    vel_limit = 0.5
    ini_w = 0.9
    fin_w = 0.1
    temp_initial = 1.0
    alpha = 0.99

    max_stagn = 500

    if 'initial_h_values' in globals():

            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            pop.append(h_scaled)
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * 0.5
            h_scaled = h * (ones_count / h_sum)
            pop.append(h_scaled)

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = pop[0].copy()
    best_c5 = obj(pop[0])
    start_time = time.time()
    temp = temp_initial
    stagn = 0

    while time.time() - start_time < budget_s:
        best_c5_val = best_c5
        for i in range(pop_size):
            pbest = pop[i].copy()
            pbest_c5 = obj(pbest)
            if pbest_c5 < best_c5_val:
                best_c5_val = pbest_c5
                best_h = pbest.copy()
            if pbest_c5 < best_c5:
                best_c5 = pbest_c5
                best_h = pbest.copy()

        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = np.random.rand(n_points) * 0.5
            if i == 0:
                gbest = best_h.copy()
            else:
                gbest = best_h.copy()
            v = (ini_w * vel + 
                 phi1 * r1 * (pbest - pop[i]) + 
                 phi2 * r2 * (gbest - pop[i]))
            v = np.clip(v, -vel_limit, vel_limit)
            h_new = pop[i] + v
            h_new = np.clip(h_new, 0.0, 1.0)
            h_sum = np.sum(h_new)
            if h_sum == 0:
                continue
            h_scaled = h_new * (ones_count / h_sum)
            pop[i] = h_scaled

        if np.random.rand() < p_perturb:

            step = np.random.rand(n_points) * 0.1 - 0.05
