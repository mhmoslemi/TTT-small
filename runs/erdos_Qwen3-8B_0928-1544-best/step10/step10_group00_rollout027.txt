def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2
    population_size = 50
    omega_initial = 0.9

    phi1 = phi2 = 1.4
    temp_initial = 1.0
    cooling_rate = 0.97
    perturb_prob = 0.1

    # Initialize population
    population = []
    velocities = []
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h = initial_h * (ones_count / np.sum(initial_h))
        population.append(initial_h)
        velocities.append(np.zeros(n_points))
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            h = h * (ones_count / h_sum)
            population.append(h)
            velocities.append(np.zeros(n_points))
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            h = h * (ones_count / h_sum)
            population.append(h)
            velocities.append(np.zeros(n_points))

    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)

    personal_best_h = [np.copy(h) for h in population]
    best_h = np.copy(population[0])
    best_c5 = compute_c5(population[0])
    start_time = time.time()
    current_temp = temp_initial
    iteration = 0

    while time.time() - start_time < budget_s:
        iteration += 1
        # Update velocities and positions
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i]
            omega = omega_initial - (omega_initial - omega_final) * (iteration / 1000)
            vel += omega * vel
            vel += phi1 * r1 * (personal_best_h[i] - population[i])
            vel += phi2 * r2 * (best_h - population[i])
            velocities[i] = np.clip(vel, -0.5, 0.5)

            new_h = population[i] + velocities[i]

            new_h_sum = np.sum(new_h)

            if current_c5 < best_c5:
                best_c5 = current_c5
