import numpy as np

def run(seed=42, budget_s=1100, **kwargs):

    if 'initial_h_values' in globals():

        new_h = h + noise
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum == 0:
            new_h = np.zeros(n_points)
        else:

    return h, best_c5, n_points
