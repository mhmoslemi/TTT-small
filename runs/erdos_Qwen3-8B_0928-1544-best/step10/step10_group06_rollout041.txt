            new_h = new_h * scaling
            population.append(new_h)
    else:
        n_points = 80
        population = []
        for _ in range(50):
            h = np.random.rand(n_points)
            total = np.sum(h)
            scaling = (n_points / 2) / total
            h = h * scaling
            population.append(h)
    
    def compute_c5(h):
        h_np = np.array(h)
        corr = np.correlate(h_np, 1 - h_np, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def repair(h):
        clipped = np.clip(h, 0, 1)
        total = np.sum(clipped)
        scale = (n_points / 2) / total
        return clipped * scale
    
    fitness = [compute_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_c5 = fitness[best_idx]
    
    tournament_size = 5
    mutation_rate = 0.2
    crossover_rate = 0.7
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(len(population)):
            def select_parent():
                indices = np.random.randint(len(population), size=tournament_size)
                min_idx = indices[np.argmin([fitness[i] for i in indices])]
                return population[min_idx]
            parent1 = select_parent()
            parent2 = select_parent()
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:split], parent2[split:]])
            else:
                child = (parent1 + parent2) / 2
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, 0.05, n_points)
                child += mutation
            child = repair(child)
            new_population.append(child)
        new_fitness = [compute_c5(h) for h in new_population]
        for i in range(len(new_population)):
            if new_fitness[i] < best_c5:
                best_c5 = new_fitness[i]
                best_h = new_population[i]
        population = new_population
        fitness = new_fitness
    
    top_indices = np.argsort(fitness)[:10]
    for i in top_indices:
        candidate = population[i]
        def objective(x):
            h = x
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * (2.0 / n_points)
        bounds = [(0, 1) for _ in range(n_points)]
        result = scipy.optimize.minimize(objective, candidate, bounds=bounds, method='L-BFGS-B')
        if result.success:
            best_h = result.x
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
