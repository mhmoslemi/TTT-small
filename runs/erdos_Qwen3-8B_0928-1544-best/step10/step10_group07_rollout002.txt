def run(seed=42, budget_s=1100, **kwargs):

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        h = h * (sum_required / h_sum)
    
    # Define temperature for smooth max approximation
    temperature = 0.1

    def approx_objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        exp_corr = np.exp(corr / temperature)
        return np.log(np.sum(exp_corr)) * temperature * (2.0 / n_points)  # dx = 2.0 / n_points

    # Define constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required})

    # Initial optimization
    start_time = time.time()
    result = minimize(
        approx_objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100}
    )
    best_h = result.x
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / n_points)
    elapsed_time = time.time() - start_time

    # Further optimization steps
