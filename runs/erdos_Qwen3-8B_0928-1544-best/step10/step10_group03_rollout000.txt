        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        dx = 2.0 / n_points
        grad = np.zeros(n_points)
        for j in range(n_points):
            j_plus = (j + lag) % n_points
            j_minus = (j - lag) % n_points
            grad[j] = dx * ((1 - h[j_plus]) - h[j_minus])
        trial_h = h - alpha * grad
        trial_h = np.clip(trial_h, 0, 1)
        total = np.sum(trial_h)
        if total == 0:
            trial_h = np.zeros(n_points)
        else:
            trial_h = trial_h * (ones_count / total)
        new_c5 = compute_c5(trial_h)
        if new_c5 < best_c5:
            h = trial_h.copy()
            best_c5 = new_c5
            best_h = h.copy()
        alpha *= 0.99
    return best_h, best_c5, n_points

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
