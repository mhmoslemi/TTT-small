import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        h = initial_h_values
        a = h - 0.5
    else:
        a = np.random.uniform(-0.5, 0.5, n_points)

    def adjust_constraints(a_vals):
        a_vals = np.clip(a_vals, -0.5, 0.5)
        sum_a = np.sum(a_vals)
        delta = (-sum_a) / len(a_vals)
        a_vals += delta
        a_vals = np.clip(a_vals, -0.5, 0.5)
        return a_vals

    a = adjust_constraints(a.copy())

    def compute_c5(a_vals):
        h_vals = 0.5 + a_vals
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def evaluate_candidate(a_vals):
        return compute_c5(adjust_constraints(a_vals))

    population_size = 20
    pop = [a.copy() for _ in range(population_size)]
    best_a = a.copy()
    best_c5 = evaluate_candidate(best_a)

    start_time = time.time()
    F = 0.8
    CR = 0.9
    gen = 0
    max_gen = 300

    while gen < max_gen and time.time() - start_time < budget_s:
        new_pop = []
        for i in range(population_size):
            a1, a2, a3 = pop[np.random.choice(population_size, 3, replace=False)]
            mutant = a1 + F * (a2 - a3)
            trial = pop[i].copy()
            rand_idx = np.random.rand(len(trial)) < CR
            trial[rand_idx] = mutant[rand_idx]
            trial = adjust_constraints(trial)
            trial_c5 = evaluate_candidate(trial)
            if trial_c5 < evaluate_candidate(pop[i]):
                new_pop.append(trial)
            else:
                new_pop.append(pop[i])
        pop = new_pop
        current_best = min(pop, key=evaluate_candidate)
        if evaluate_candidate(current_best) < best_c5:
            best_a = current_best
            best_c5 = evaluate_candidate(best_a)
        gen += 1

    h_vals = 0.5 + best_a
    h_vals = np.clip(h_vals, 0, 1)
    return h_vals, best_c5, n_points
