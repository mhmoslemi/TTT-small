import time

def run(seed=42, budget_s=1100, **kwargs):

    else:
        initial_h_scaled = np.random.rand(n_points) * 0.5
        initial_h_sum = np.sum(initial_h_scaled)
        initial_h_scaled *= (ones_count / initial_h_sum)
    population = [initial_h_scaled]
    for _ in range(99):
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h_scaled = h * (ones_count / h_sum)
        population.append(h_scaled)
    
    def objective(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()
    F = 0.8
    CR = 0.9
    F_decay = 0.001
    CR_inc = 0.005
    improvement_count = 0
    iter_count = 0
    
    while time.time() - start_time < budget_s:
        for i in range(100):
            a, b, c = np.random.choice(100, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                u += 1e-6
                u_sum = np.sum(u)
            u_scaled = u * (ones_count / u_sum)
            if np.random.rand() < 0.1:
                u_scaled = initial_h_scaled.copy()
            current_c5 = objective(population[i])
            new_c5 = objective(u_scaled)
            if new_c5 < current_c5:
                population[i] = u_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5

        if improvement_count > 50:
