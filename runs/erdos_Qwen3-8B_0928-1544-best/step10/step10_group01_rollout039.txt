import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    swarm_size = 50
    inertia = 0.9
    cognitive = 2.0
    social = 2.0
    time_start = time.time()

    # Initialize particles
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        scale = ones_count / initial_h_sum
        particles = [initial_h * scale + np.random.rand(n_points) * 0.01 for _ in range(swarm_size)]
    else:
        particles = [np.random.rand(n_points) * 0.5 for _ in range(swarm_size)]
        for i in range(swarm_size):
            particles[i] = particles[i] * (ones_count / np.sum(particles[i]))
    
    velocities = np.zeros((swarm_size, n_points))
    pbest = [p.copy() for p in particles]
    gbest = min(pbest, key=lambda x: objective(x))
    best_h = gbest.copy()
    best_c5 = objective(gbest)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    # PSO phase
    while time.time() - time_start < budget_s * 0.8:
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (pbest[i] - particles[i]) + social * r2 * (gbest - particles[i])
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_p = np.sum(particles[i])
            if sum_p != 0:
                particles[i] = particles[i] * (ones_count / sum_p)
            curr_c5 = objective(particles[i])
            if curr_c5 < objective(pbest[i]):
                pbest[i] = particles[i].copy()
            if curr_c5 < objective(gbest):
                gbest = particles[i].copy()
                best_c5 = curr_c5
                best_h = gbest.copy()

    # Gradient descent phase
    corr = np.correlate(gbest, 1 - gbest, mode='full')
    k_index = np.argmax(corr)
    max_k = k_index

    def compute_gradient(h, k):
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_p = h.copy()
            h_p[i] += eps
            h_m = h.copy()
            h_m[i] -= eps
            corr_p = np.correlate(h_p, 1 - h_p, mode='full')
            corr_m = np.correlate(h_m, 1 - h_m, mode='full')
            grad[i] = (corr_p[k] - corr_m[k]) / (2 * eps)
        return grad

    while time.time() - time_start < budget_s:
        grad = compute_gradient(gbest, max_k)
        learning_rate = 0.01
        gbest = gbest - learning_rate * grad
        gbest = np.clip(gbest, 0, 1)
        sum_g = np.sum(gbest)
        if sum_g != 0:
            gbest = gbest * (ones_count / sum_g)
        new_c5 = objective(gbest)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = gbest.copy()

    return best_h, best_c5, n_points
