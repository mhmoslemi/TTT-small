import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    ones_count = n_points // 2
    population_size = 40

    def generate_h():

        if h_sum == 0:
            return np.random.rand(n_points) * 0.5
        return h * (ones_count / h_sum)
    
    if initial_h is not None:
        population = [initial_h]
        for _ in range(population_size - 1):
            h = generate_h()
            population.append(h)
    else:
        population = [generate_h() for _ in range(population_size)]
    
    # Objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = objective(population[0])
    
    start_time = time.time()
    
    F = 0.7  # Reduced mutation factor for finer tuning
    CR = 0.8  # Moderate crossover probability
    
    while time.time() - start_time < budget_s:
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            mutant = a_vec + F * (b_vec - c_vec)
            
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            
            # Clip and scale
            trial = np.clip(trial, 0.0, 1.0)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                continue
            trial = trial * (ones_count / trial_sum)
            
            new_c5 = objective(trial)
            if new_c5 < objective(population[i]):
                population[i] = trial
                if new_c5 < best_c5:
                    best_c5 = new_c5
