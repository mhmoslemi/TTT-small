            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)
    
    # Initialize velocities
    velocities = np.zeros((len(population), n_points))
    
    # Parameters
    phi1, phi2 = 1.4, 1.4
    w_initial, w_final = 0.9, 0.1
    cooling_rate, perturb_prob = 0.99, 0.1
    temp_initial = 1.0
    max_iter = 100000

    # Objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # Evaluate initial population
    pbest = [h.copy() for h in population]
    pbest_values = [objective(h) for h in population]
    gbest_idx = np.argmin(pbest_values)
    gbest = population[gbest_idx].copy()
    best_c5 = pbest_values[gbest_idx]
    best_h = gbest.copy()

    # Main loop
    temp = temp_initial
    iteration = 0
    while time.time() - start_time < max_time:
        iteration += 1
        w = w_initial - (w_initial - w_final) * iteration / max_iter
        
        for i in range(len(population)):
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel = w * velocities[i] + phi1 * r1 * (pbest[i] - population[i]) + phi2 * r2 * (gbest - population[i])
            velocities[i] = vel
            
            # Update position
            h_new = population[i] + vel
            h_clipped = np.clip(h_new, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum == 0:
                continue
            h_scaled = h_clipped * (ones_count / h_sum)
            
            # SA perturbation
            if np.random.rand() < perturb_prob:
                perturbation = np.random.normal(0, 0.1, size=n_points)
                h_perturbed = h_scaled + perturbation
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed_sum = np.sum(h_perturbed)
                if h_perturbed_sum == 0:
                    continue
                h_perturbed_scaled = h_perturbed * (ones_count / h_perturbed_sum)
                new_c5 = objective(h_perturbed_scaled)
                current_c5 = objective(h_scaled)
                if new_c5 < current_c5 or (np.random.rand() < np.exp((current_c5 - new_c5) / temp)):
                    h_scaled = h_perturbed_scaled
            
            population[i] = h_scaled
            current_c5 = objective(h_scaled)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest[i] = h_scaled.copy()
                pbest_values[i] = current_c5
            
            # Update global best
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_scaled.copy()
                gbest = best_h.copy()
        
        # Update temperature
        temp *= cooling_rate
    
    return best_h, best_c5, n_points
