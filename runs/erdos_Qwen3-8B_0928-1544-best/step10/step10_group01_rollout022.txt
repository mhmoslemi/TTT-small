import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    budget_s = budget_s if 'budget_s' in kwargs else 1100
    start_time = time.time()

    # Initialize particles
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h_scaled = initial_h * (ones_count / initial_h_sum)
        population = [initial_h_scaled]
    else:
        population = []
        for _ in range(50):
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            h_scaled = h * (ones_count / h_sum)
            population.append(h_scaled)

    # PSO parameters
    omega = 0.9
    rho1 = 2.0
    rho2 = 2.0
    inertia_decay = 0.99

    # Tracks best solution
    best_h = population[0].copy()
    best_c5 = _objective(population[0], n_points)

    while time.time() - start_time < budget_s:
        # Determine remaining time
        remaining = budget_s - (time.time() - start_time)
        if remaining < 50:
            break  # Not enough time for further refinement

        # PSO main loop
        for i in range(len(population)):
            # Select three distinct individuals
            indices = np.random.choice(len(population), 3, replace=False)
            a, b, c = indices
            a_vec, b_vec, c_vec = population[a], population[b], population[c]

            # Mutate: v = a + F*(b - c)
            F = 0.8
            v = a_vec + F * (b_vec - c_vec)

            # Crossover: u is a mix of v and current individual i
            u = np.copy(population[i])
            crossover_mask = np.random.rand(n_points) < 0.95
            u[crossover_mask] = v[crossover_mask]

            # Clip and scale
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)

            # Evaluate
            new_c5 = _objective(u_scaled, n_points)
            if new_c5 < _objective(population[i], n_points):
                population[i] = u_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u_scaled.copy()

        # Decay inertia weight
        omega *= inertia_decay

    # Gradient descent refinement
    remaining = budget_s - (time.time() - start_time)
    if remaining > 50:
        for _ in range(1000):
            if time.time() - start_time >= budget_s:
                break

            # Evaluate current best
            current_c5 = best_c5
            h = best_h.copy()

            # Find the shift with maximum integral
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            k = max_idx - (n_points - 1)

            # Compute gradient
            grad = np.zeros(n_points)
            for m in range(n_points):
                m_plus_k = m + k
                m_minus_k = m - k
                term1 = 1.0 if m_plus_k >= 0 and m_plus_k < n_points else 0.0
                term2 = h[m_minus_k] if m_minus_k >= 0 and m_minus_k < n_points else 0.0
                grad[m] = (1.0 - h[m_plus_k]) - term2

            # Update h with gradient descent
            eta = 0.01
            h = h - eta * grad
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            if h_sum == 0:
                continue
            h = h * (ones_count / h_sum)

            # Evaluate new solution
            new_c5 = _objective(h, n_points)
            if new_c5 < current_c5 and np.linalg.norm(h - best_h) > 1e-6:
                best_h = h.copy()
                best_c5 = new_c5

    return best_h, best_c5, n_points

def _objective(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n_points)
