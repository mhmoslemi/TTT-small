import numpy as np
from scipy.optimize import minimize, Bounds
import random

def run(seed=42, budget_s=1100, **kwargs):

        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(70, 0.5)
    
    dx = 2.0 / 70
    n = 70
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale
    
    best_c5 = float('inf')
    best_h = np.copy(initial_h)
    current_h = project_h(initial_h)
    current_c5 = objective(current_h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h
    
    num_candidates = 50
    for _ in range(num_candidates):
        if random.random() < 0.5:
            perturbation = np.random.normal(0, 0.1, size=n)
            perturbed_h = best_h + perturbation
        else:
            p = 0.1

        perturbed_h = project_h(perturbed_h)
        
        result = minimize(

            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,

        if result.success:
