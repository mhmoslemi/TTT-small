import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    start_time = time.time()
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def feasible(h):
        return np.sum(h) - 35
    
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_pop = [initial_h_values.copy()]
        for _ in range(19):
            perturb = np.random.normal(0, 0.1, n_points)
            new_h = initial_h_values + perturb
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (35.0 / np.sum(new_h))
            initial_pop.append(new_h)
    else:
        initial_pop = [np.random.rand(n_points) * 35.0 / np.sum(np.random.rand(n_points)) for _ in range(20)]
    
    best_h = initial_pop[0]
    best_c5 = evaluate_c5(best_h)
    
    max_gen = 100
    for gen in range(max_gen):
        current_time = time.time()
        if current_time - start_time > budget_s:
            break
        
        candidates = [h for h in initial_pop]
        for h in candidates:
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
        
        top5 = sorted(candidates, key=lambda h: evaluate_c5(h))[:5]
        new_pop = []
        
        for _ in range(15):
            p = np.random.choice(top5)
            perturb = np.random.normal(0, 0.1, n_points)
            new_h = p + perturb
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (35.0 / np.sum(new_h))
            new_pop.append(new_h)
        
        for _ in range(5):
            p1, p2 = np.random.choice(top5, 2, replace=False)
            child = (p1 + p2) / 2
            child = np.clip(child, 0, 1)
            child = child * (35.0 / np.sum(child))
            new_pop.append(child)
        
        initial_pop = new_pop
        
        for h in top5:
            def obj(x):
                return evaluate_c5(x)
            cons = {'type': 'eq', 'fun': feasible}
            bounds = Bounds([0]*n_points, [1]*n_points)
            res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'max_iter': 100, 'ftol': 1e-6})
            if res.success:
                if res.fun < best_c5:
                    best_c5 = res.fun
                    best_h = res.x
                idx = np.argmin([np.sum(np.abs(h - x)) for x in initial_pop])
                initial_pop[idx] = res.x
    
    return best_h, best_c5, n_points
