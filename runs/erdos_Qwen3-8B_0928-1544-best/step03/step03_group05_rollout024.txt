from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    initial_guesses = []
    
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        initial_guesses.append(initial_h_values.copy())

        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        initial_guesses.append(h_rand)

    best_c5 = np.inf

            corr = correlate(h_val, 1 - h_val, mode='full')

        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},

            options={'maxiter': 150, 'ftol': 1e-12, 'eps': 1e-8}

        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:

    best_h[best_h > 1] = 1
