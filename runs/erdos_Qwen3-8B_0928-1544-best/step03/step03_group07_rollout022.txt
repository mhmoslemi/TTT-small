import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()
    else:
        n = 100
        h = np.random.rand(n)
        h = h * (n / 2) / np.sum(h)
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)
    
    for _ in range(300):
        corr = correlate(best_h, 1 - best_h, mode='full')
        current_max = np.max(corr * dx)
        if current_max < best_c5:
            best_c5 = current_max
            best_h = best_h.copy()
        
        if best_c5 < 0.38085:
            break
        
        worst_idx = np.argmax(corr * dx)
        def target_obj(h_flat):
            h = h_flat.reshape(n_points)
            corr = correlate(h, 1 - h, mode='full')
            return corr[worst_idx] * dx
        
        res = minimize(
            target_obj, 
            best_h, 
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 50, 'ftol': 1e-12}
        )
        
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
    
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if np.abs(final_sum - n_points / 2) > 1e-6:
        final_h = final_h / final_sum * (n_points / 2)
    
    return final_h, best_c5, n_points
