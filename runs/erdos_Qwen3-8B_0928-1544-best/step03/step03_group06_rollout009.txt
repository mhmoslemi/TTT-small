        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 1
        h *= scale
    else:
        h = np.full(n_points, 0.5, dtype=np.float64)
        total = np.sum(h)
        scale = (n_points / 2) / total
        h *= scale

    best_c5 = np.inf
    best_h = h.copy()
    max_iterations = 500
    learning_rate = 0.05
    decay_rate = 0.99
    start_time = np.datetime64('now')

    for _ in range(max_iterations):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        max_corr_idx = np.argmax(corr)
        m = max_corr_idx - (n_points - 1)
        g = np.zeros_like(h)
        for j in range(n_points):
            shifted_j = j + m
            if shifted_j < n_points:
                g[j] = 1 - h[shifted_j]
        h -= learning_rate * g
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 1
        h *= scale
        current_c5 = max_corr * (2 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.38085:
            break
        learning_rate *= decay_rate
        elapsed = np.datetime64('now') - start_time
        if elapsed.total_seconds() > budget_s:
            break

    return best_h, best_c5, n_points
