import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def adjust_sum(h):
        s = np.sum(h)
        if abs(s - n_points / 2) > 1e-8:
            delta = (n_points / 2 - s) / n_points
            h += delta
            h = np.clip(h, 0.0, 1.0)
            if np.sum(h) < n_points / 2:
                h[-1] = 1.0
            else:
                h[-1] = 0.0
        return h

    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    elapsed = 0.0
    start = time.time()

    num_restarts = 50
    perturbation_scale = 0.15
    np.random.seed(seed)

    for _ in range(num_restarts):
        perturbed = best_h + np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed = adjust_sum(perturbed)

        result = minimize(
            compute_C5,
            perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'rhobeg': 0.5, 'catol': 1e-8}
        )

        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun

        elapsed = time.time() - start
        if elapsed > budget_s:

    return best_h, best_c5, n_points
