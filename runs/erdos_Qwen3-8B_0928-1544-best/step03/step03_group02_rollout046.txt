import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = np.array(initial_h_values)
    sum_h = n_points / 2
    h_initial = np.clip(h_initial, 0.0, 1.0)
    if not np.isclose(np.sum(h_initial), sum_h):
        h_initial[-1] += sum_h - np.sum(h_initial)
        h_initial = np.clip(h_initial, 0.0, 1.0)

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - sum_h

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def renew_guess(h, scale=0.05):
        noise = np.random.uniform(-scale, scale, size=n_points)
        perturbed = h + noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        diff = sum_h - np.sum(perturbed)
        perturbed[-1] += diff
        perturbed = np.clip(perturbed, 0.0, 1.0)
        return perturbed

    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    elapsed_time = 0.0
    start_time = time.time()

    num_restarts = 50
    max_iter_per_restart = 150
    perturbation_scale = 0.03

    for _ in range(num_restarts):
        perturbed = renew_guess(best_h, perturbation_scale)
        result = minimize(
            compute_C5,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': max_iter_per_restart, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        if result.success and result.fun < best_c5:
            current_h = result.x
            current_c5 = compute_C5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break

    return best_h, best_c5, n_points
