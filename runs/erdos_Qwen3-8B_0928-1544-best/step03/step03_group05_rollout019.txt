import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()
    else:
        n = int(70)
        h = np.zeros(n)
        h[:n//2] = 1.0
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h
    best_c5 = np.inf
    n_iter = 600
    tol = 1e-9
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    for _ in range(n_iter):
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=tol,
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-8}
        )
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        if _ % 10 == 0 and best_c5 < 0.38085:
            break
        h = res.x
    
    final_h = best_h
    final_h[final_h < 0] = 0
    final_h[final_h > 1] = 1
    final_sum = np.sum(final_h)
    if np.abs(final_sum - n_points / 2) > 1e-6:
        final_h = final_h / final_sum * (n_points / 2)
    
    return final_h, best_c5, n_points
