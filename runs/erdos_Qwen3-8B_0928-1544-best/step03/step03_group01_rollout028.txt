        initial_h = h

    dx = 2.0 / n
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h) - n / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = Bounds([0.0]*n, [1.0]*n)

        sum_h = np.sum(h_clip)
        scale = (n / 2) / sum_h if sum_h != 0 else 0
        return h_clip * scale
    
    best_c5 = float('inf')
    best_h = initial_h.copy()

    current_c5 = objective(current_h)
