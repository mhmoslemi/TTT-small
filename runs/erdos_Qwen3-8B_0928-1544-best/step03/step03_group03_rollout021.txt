    start_time = time.time()
    
    n = 70
    dx = 2.0 / n

    # Use initial_h_values if available
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        initial_guess = initial_h_values.copy()
    else:
        initial_guess = np.full(n, 0.5)
    
    # Ensure sum is exactly n/2
    sum_initial = np.sum(initial_guess)
    scale = (n / 2) / sum_initial
    initial_guess *= scale

    best_h = initial_guess.copy()
    best_c5 = np.correlate(best_h, 1 - best_h, mode='full').max() * dx

    # Simulated Annealing parameters
    temp_initial = 0.1
    cooling_rate = 0.99
    sigma = 0.05
    num_iterations = 5000

    current_h = best_h.copy()
    temperature = temp_initial

    for _ in range(num_iterations):
        # Generate perturbation
        delta = np.random.normal(0, sigma, n)
        new_h = current_h + delta
        new_h = np.clip(new_h, 0.0, 1.0)
        
        # Normalize sum to n/2
        sum_new = np.sum(new_h)
        scale = (n / 2) / sum_new
        new_h = new_h * scale
        
        # Compute new C5
        corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = corr.max() * dx
        
        # Acceptance probability
        if new_c5 < best_c5:
            best_c5 = new_c5
            current_h = new_h.copy()
            best_h = new_h.copy()
        else:
            delta_c = best_c5 - new_c5
            prob = np.exp(-delta_c / temperature)
            if np.random.rand() < prob:
                current_h = new_h.copy()
        
        # Cool down
        temperature *= cooling_rate
        
        # Check time budget
        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n
