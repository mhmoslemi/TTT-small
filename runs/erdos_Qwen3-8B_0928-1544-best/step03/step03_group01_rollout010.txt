    def objective(h):

        return np.max(corr) * dx

    bounds = Bounds([0.0]*n, [1.0]*n)

        sum_h = np.sum(h_clip)

        scale = (n / 2) / sum_h
        return h_clip * scale

    best_c5 = float('inf')

    current_h = project_h(initial_h)
    current_c5 = objective(current_h)
