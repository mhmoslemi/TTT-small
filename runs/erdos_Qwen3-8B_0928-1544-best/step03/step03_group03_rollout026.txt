import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    
    n = 70
    dx = 2.0 / n

    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h = initial_h_values.copy()
    else:
        h = np.full(n, 0.5)
    h = h * (n / 2) / np.sum(h)
    
    best_c5 = np.inf

    max_iterations = 10000
    alpha = 0.1
    alpha_decay = 0.999
    
    for _ in range(max_iterations):
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n - 1)
        
        gradient = np.zeros(n)
        for i in range(n):
            if 0 <= i + k < n:
                term1 = (1 - h[i + k]) * dx
            else:
                term1 = 0.0
            if 0 <= i - k < n:
                term2 = -h[i - k] * dx
            else:
                term2 = 0.0
            gradient[i] = term1 + term2
            
        h = np.clip(h - alpha * gradient, 0.0, 1.0)
        h = h * (n / 2) / np.sum(h)
        
        current_c5 = corr.max() * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
        
        if time.time() - start_time > budget_s:
            break
        
        alpha *= alpha_decay
    
    return h, best_c5, n
