        h = initial_h_values.copy()
    else:
        h = np.full(n, 0.5)
    
    # Normalize initial guess to ensure ∫h = 1
    h = h * (35.0 / np.sum(h))
    
    best_h = h.copy()
    best_c5 = np.correlate(h, 1 - h, mode='full').max() * dx
    
    max_iter = 10000
    alpha = 0.1
    decay = 0.99
    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = corr.max()
        idx = np.argmax(corr)
        lag_k = idx - (n - 1)
        
        # Compute gradient contributions for current lag k
        gradient = np.zeros(n)
        for i in range(n):
            i_plus_k = (i + lag_k) % n
            i_minus_k = (i - lag_k) % n
            gradient[i] = (1.0 - h[i_plus_k]) - h[i_minus_k]
        
        h = h - alpha * gradient
        h = np.clip(h, 0.0, 1.0)
        h = h * (35.0 / np.sum(h))
        
        current_c5 = corr.max() * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if time.time() - start_time > budget_s:
            break
        
        alpha *= decay
    
    return best_h, best_c5, n
