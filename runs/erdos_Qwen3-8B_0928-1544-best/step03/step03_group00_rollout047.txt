    best_h = h.copy()

        if elapsed > budget_s:
            break

        # Compute max correlation and k_max
        max_corr = -1.0
        k_max = 0
        for k in range(n_points):
            # Compute correlation for shift k
            shifted = np.roll(h, k)
            corr = np.sum(h * (1 - shifted))
            if corr > max_corr:
                max_corr = corr
                k_max = k

        # Compute gradient
        k = k_max
        indices1 = (np.arange(n_points) + k) % n_points

        shifted_term1 = 1.0 - h[indices1]
