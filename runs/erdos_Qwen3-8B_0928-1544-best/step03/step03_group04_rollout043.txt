import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    best_c5 = np.inf
    best_h = h
    
    for _ in range(50):
        perturbation = np.random.normal(0, 0.1, n_points)
        perturbed_h = h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        res = minimize(
            objective,
            perturbed_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-8}
        )
        
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
    
    final_h = best_h
    final_sum = np.sum(final_h)
    if np.abs(final_sum - n_points / 2) > 1e-6:
        final_h = final_h / final_sum * (n_points / 2)
    
    return final_h, best_c5, n_points
