import numpy as np
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()

    max_generations = 200
    sigma_start = 0.1
    sigma_end = 0.01
    optimization_steps = 100
    
    bounds = Bounds([0.0]*70, [1.0]*70)

        perturbation = np.random.normal(0, sigma, 70)

        def objective(h):

            return np.max(corr) * dx
        
        def constraint_sum(h):

        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
