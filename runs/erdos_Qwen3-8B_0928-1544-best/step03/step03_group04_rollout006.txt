        return max_overlap + (log_terms + sum_penalty) / mu
    
    def gradient(h_flat):
        h = h_flat.copy()
        grad = np.zeros_like(h)
        eps = 1e-6
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            obj_plus = objective(h_plus)
            h_minus = h.copy()
            h_minus[i] -= eps
            obj_minus = objective(h_minus)
            grad[i] = (obj_plus - obj_minus) / (2 * eps)
        return grad
    
    for _ in range(max_iter):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        res = minimize(objective, h, method='BFGS', jac=gradient, bounds=bounds, tol=1e-8, options={'maxiter': 100})
        h = res.x
        h = np.clip(h, eps, 1 - eps)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            scale = (n_points / 2) / sum_h
            h = h * scale
        current_c5 = objective(h) - (np.sum(-np.log(h) - np.log(1 - h)) + (np.sum(h) - n_points / 2)**2) / mu
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
        if best_c5 < 0.38085:
            break
        mu *= 1.1
    
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        scale = (n_points / 2) / sum_h
        best_h = best_h * scale
    final_c5 = np.max(correlate(best_h, 1 - best_h, mode='full')) * dx
    return best_h, final_c5, n_points
