import time

        mutant = mutant * (35.0 / np.sum(mutant))
        population.append(mutant)
    
    best_h = h_initial
    best_c5 = float('inf')
    start_time = time.time()
    max_generations = int(budget_s / 5)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    for gen in range(max_generations):

            break

        current_best = np.min(scores)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin(scores)]
        
        # Select survivors
        indices = np.argsort(scores)
        survivors = [population[i] for i in indices[:len(population)//4]]
        
        # Mutate and generate new population
        new_population = []
        for h in survivors:
            for _ in range(3):
                mutant = h + np.random.normal(0, 0.1 * (1 - gen/max_generations), n_points)
                mutant = np.clip(mutant, 0.0, 1.0)
