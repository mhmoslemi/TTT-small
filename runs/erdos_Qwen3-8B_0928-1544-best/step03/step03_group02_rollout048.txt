import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def adjust_sum(h):
        s = np.sum(h)
        delta = (n_points / 2) - s
        h[-1] += delta
        return np.clip(h, 0.0, 1.0)

    population = [adjust_sum(h_initial + np.random.normal(0, 0.05, n_points)) for _ in range(50)]
    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        fitness = [(compute_C5(h), h) for h in population]
        fitness.sort()
        new_pop = [h for _, h in fitness[:int(len(population)*0.2)]]
        while len(new_pop) < len(population):

            child = 0.5 * (a + b)
            child = adjust_sum(child)
            noise = np.random.normal(0, 0.05, n_points)
