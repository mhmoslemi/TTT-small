import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    # Access initial_h_values as global
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    best_c5 = float('inf')
    best_h = initial_h.copy()

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def adjust_sum(h):
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) < 1e-6:
            return h
        h[-1] += diff
        h = np.clip(h, 0.0, 1.0)
        return h

    # Generate initial population
    def generate_initial_population(size):
        pop = []
        for _ in range(size):
            h = initial_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0.0, 1.0)
            h = adjust_sum(h)
            pop.append(h)
        return pop

    pop_size = 50
    population = generate_initial_population(pop_size)
    start_time = time.time()

    # Genetic algorithm parameters
    mutation_rate = 0.1
    crossover_rate = 0.7

    while time.time() - start_time < budget_s:
        evaluated = []
        for h in population:
            c5 = compute_C5(h)
            evaluated.append((c5, h))
        evaluated.sort()
        num_parents = int(len(population) * 0.5)
        parents = [h for (c5, h) in evaluated[:num_parents]]
        children = []
        for _ in range(len(population) - num_parents):
            if np.random.rand() < crossover_rate:
                p1, p2 = np.random.choice(parents, 2, replace=False)
                child = (p1 + p2) / 2.0
            else:
                p = np.random.choice(parents)
                child = p.copy()
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0.0, 1.0)
                child = adjust_sum(child)
            children.append(child)
        population = parents + children

        for c5, h in evaluated:
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Local optimization on the best solution
    def objective(h):
        return compute_C5(h)
    def constraint_sum(h):
        return np.sum(h) - target_sum
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum})
    if result.success:
        best_h = result.x
        best_c5 = compute_C5(best_h)

    return best_h, best_c5, n_points
