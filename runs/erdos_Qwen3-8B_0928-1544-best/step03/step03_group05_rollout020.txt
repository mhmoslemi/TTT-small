    best_coarse = population[np.argmin([evaluate_c5(ind) for ind in population])]
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, best_coarse)
    h_fine = h_fine / (np.sum(h_fine) / (n_fine / 2))
    h_fine = np.clip(h_fine, 0, 1)

    population = [h_fine]
    for _ in range(n_fine_gens):
        scores = [evaluate_c5(ind) for ind in population]
        elite_idx = np.argsort(scores)[:int(n_pop * 0.2)]
        elites = [population[i] for i in elite_idx]
        offspring = []
        for _ in range(n_pop - len(elites)):
            parent1, parent2 = random.sample(elites, 2)
            child = (parent1 + parent2) / 2
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_std, len(child))
                child += noise
                child = np.clip(child, 0, 1)
            child = np.clip(child, 0, 1)
            offspring.append(child)
        new_population = elites + offspring
        for i in range(len(new_population)):
            new_population[i] = optimize_individual(new_population[i])
        population = new_population
        if np.min(scores) < 0.38085:
            break

    best_h = population[np.argmin([evaluate_c5(ind) for ind in population])]
    c5_bound = evaluate_c5(best_h)
    return best_h, c5_bound, n_fine
