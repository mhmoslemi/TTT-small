    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

        h = np.clip(h, 0.0, 1.0)
    
    best_h = h.copy()
    best_c5 = compute_C5(best_h)
    elapsed_time = 0
    start_time = time.time()

    max_iter = 400
    perturbation_scale = 0.07
    method = 'L-BFGS-B'

        perturbed = np.clip(perturbed, 0.0, 1.0)
        # Rebalance sum

        if not np.isclose(sum_p, n_points / 2):
            diff = (n_points / 2) - sum_p
            perturbed[-1] += diff
            perturbed = np.clip(perturbed, 0.0, 1.0)
        
        # Solve optimization
        result = minimize(
            compute_C5,
