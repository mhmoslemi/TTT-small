import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()
    else:
        n_points = 70
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    n_points = len(h)
    dx = 2.0 / n_points
    best_c5 = np.inf
    best_h = h
    
    def compute_c5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def adjust_h(h_vec):
        s = np.sum(h_vec)
        if s == 0:
            return h_vec
        return h_vec * (n_points / 2) / s

    population = [h.copy() for _ in range(50)]
    for gen in range(200):
        fitness = np.array([compute_c5(ind) for ind in population])
        best_idx = np.argmin(fitness)
        current_c5 = fitness[best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[best_idx].copy()
        top_indices = np.argsort(fitness)[:int(50 * 0.2)]
        offspring = []
        for i in range(len(top_indices) // 2):
            p1, p2 = population[top_indices[2*i]], population[top_indices[2*i + 1]]
            child = (p1 + p2) / 2
            mutate_mask = np.random.rand(n_points) < 0.3
            child[mutate_mask] += np.random.normal(0, 0.05, np.sum(mutate_mask))
            child = np.clip(child, 0, 1)
            child = adjust_h(child)
            offspring.append(child)
        for ind in offspring:
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
            res = minimize(lambda x: compute_c5(x), ind, method='SLSQP',
                           bounds=[(0, 1)]*n_points, constraints=cons,
                           tol=1e-9)
            if res.success:
                ind[:] = res.x
        scores = np.array([compute_c5(ind) for ind in population])
        replace_idx = np.argsort(scores)[-len(offspring):]
        for i, idx in enumerate(replace_idx):
            population[idx] = offspring[i]
    best_h = adjust_h(best_h)
    final_sum = np.sum(best_h)
