import numpy as np
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = 35
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.zeros(n_points)
        interval = n_points / ones_count
        for i in range(ones_count):
            pos = round(i * interval)
            h_initial[pos] = 1.0
        diff = ones_count - np.sum(h_initial)
        if diff > 0:
            zeros = np.where(h_initial == 0)[0]
            h_initial[zeros[:diff]] = 1.0
    dx = 2.0 / n_points
    best_h = h_initial.copy()
    best_c5 = float('inf')

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)

    for _ in range(150):
        perturbation = np.random.normal(0, 0.1, n_points)
        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        sum_perturbed = np.sum(h_perturbed)
        if sum_perturbed != ones_count:
            h_perturbed *= ones_count / sum_perturbed
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-9,
            options={'maxiter': 400, 'ftol': 1e-9}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            h_initial = result.x
    return best_h, best_c5, n_points
