import numpy as np
import random

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    
    n = 70
    dx = 2.0 / n
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        initial = initial_h_values.copy()
        population.append(initial)
    else:
        initial = np.full(n, 0.5)
        population.append(initial)
    
    for _ in range(49):
        h = np.random.normal(0, 0.05, n)
        h = np.clip(h, 0, 1)
        h = h * (35.0 / np.sum(h))
        population.append(h)
    
    best_h = population[0]
    best_c5 = np.correlate(best_h, 1 - best_h, mode='full').max() * dx
    
    num_generations = 100
    num_parents = 10  # top 20%
    tournament_size = 5
    mutation_rate = 0.03
    
    for gen in range(num_generations):
        # Evaluate all individuals
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = corr.max() * dx
            fitness.append(-c5)
        
        # Select parents
        parents = []
        for _ in range(num_parents):
            tournament = random.sample(range(len(population)), tournament_size)
            best_idx = tournament[np.argmin(fitness[i] for i in tournament)]
            parents.append(population[best_idx])
        
        # Generate offspring
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                parent1 = parents[i]
                parent2 = parents[i+1]
                split_idx = np.random.randint(1, n-1)
                child1 = np.concatenate([parent1[:split_idx], parent2[split_idx:]])
                child2 = np.concatenate([parent2[:split_idx], parent1[split_idx:]])
                # Normalize sum
                child1 = child1 * (35.0 / np.sum(child1))
                child2 = child2 * (35.0 / np.sum(child2))
                offspring.append(child1)
                offspring.append(child2)
        
        # Apply mutation to offspring
        for j in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                h = offspring[j]
                noise = np.random.normal(0, 0.03, n)
                mutated = np.clip(h + noise, 0, 1)
                sum_mutated = np.sum(mutated)
                if sum_mutated > 0:
                    mutated = mutated * (35.0 / sum_mutated)
                else:
                    mutated = np.zeros(n) + 0.5  # fallback
                offspring[j] = mutated
        
        # Replace population
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:int(len(population)*0.2)]
        top_individuals = [population[i] for i in top_indices]
        new_population = top_individuals + offspring[:len(population) - len(top_individuals)]
        population = new_population
        
        # Update best solution
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = corr.max() * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
        
        # Check time
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, n
