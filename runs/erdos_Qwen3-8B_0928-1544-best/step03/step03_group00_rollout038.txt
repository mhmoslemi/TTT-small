import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h = initial_h_values.copy()
    else:
        h = np.zeros(70)
        h[::2] = 1.0
    n = 70
    dx = 2.0 / n
    best_c5 = float('inf')
    best_h = h.copy()
    learning_rate = 0.05
    learning_rate_decay = 0.995
    perturb_interval = 100
    iteration = 0
    start_time = time.time()
    h = h * (35.0 / np.sum(h))
    
    while time.time() - start_time < budget_s:
        max_corr = 0.0
        k_max = 0
        for k in range(n):
            j = (np.arange(n) + k) % n
            current_corr = np.dot(h, 1 - h[j]) * dx
            if current_corr > max_corr:
                max_corr = current_corr
                k_max = k
        current_c5 = max_corr
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        j_plus = (np.arange(n) + k_max) % n
        j_minus = (np.arange(n) - k_max) % n
        gradient = dx * (1.0 - h[j_plus] - h[j_minus])
        
        h = h - learning_rate * gradient
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (35.0 / sum_h)
        else:
            h = np.zeros_like(h)
        
        if iteration % perturb_interval == 0:
            noise = np.random.normal(0, 0.05, n)
            h = h + noise
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * (35.0 / sum_h)
            else:
                h = np.zeros_like(h)
        
        learning_rate *= learning_rate_decay
        iteration += 1
    
    return best_h, best_c5, n
