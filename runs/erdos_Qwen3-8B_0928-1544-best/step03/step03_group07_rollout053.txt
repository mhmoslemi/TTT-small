    def compute_obj(h_flat, k):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full') * dx
        return corr[k]
    
    while time.time() - start_time < budget_s:
        corr = correlate(best_h, 1 - best_h, mode='full') * dx
        current_max = np.max(corr)
        if current_max < best_c5:
            best_c5 = current_max
            best_h = best_h.copy()
        if best_c5 < 0.38085:
            break
        
        worst_idx = np.argmax(corr)
        k = worst_idx - (n_points - 1)
        
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        def objective(h_flat):
            return compute_obj(h_flat, k)
        
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8, options={'maxiter': 100})
        if res.success:
            new_h = res.x
            new_h = np.clip(new_h, 0.0, 1.0)
            sum_new = np.sum(new_h)
            if abs(sum_new - n_points / 2) > 1e-6:
                new_h = new_h / sum_new * (n_points / 2)
            corr_new = correlate(new_h, 1 - new_h, mode='full') * dx
            new_max = np.max(corr_new)
            if new_max < best_c5:
                best_h = new_h.copy()
                best_c5 = new_max
        
        if best_c5 < 0.38085:
            break
    
    final_h = np.clip(best_h, 0.0, 1.0)
    sum_final = np.sum(final_h)
    if abs(sum_final - n_points / 2) > 1e-6:
        final_h = final_h / sum_final * (n_points / 2)
    
    return final_h, best_c5, n_points
