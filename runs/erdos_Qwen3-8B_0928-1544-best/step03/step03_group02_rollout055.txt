    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)

    np.random.seed(seed)
    num_restarts = 40
    max_iter_per_restart = 150
    perturbation_scale = 0.1
    methods = ['L-BFGS-B', 'COBYLA']

    for _ in range(num_restarts):
        perturbed = best_h + np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        sum_perturbed = np.sum(perturbed)
        if abs(sum_perturbed - n_points / 2) > 1e-6:

            perturbed[-1] += diff
            perturbed = np.clip(perturbed, 0.0, 1.0)

        method = methods[np.random.randint(len(methods))]
        result = minimize(
            compute_C5,
            perturbed,
            method=method,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
