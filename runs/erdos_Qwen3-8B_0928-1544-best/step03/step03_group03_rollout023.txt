import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):

    # Default initial guess
    n = 70
    dx = 2.0 / n
    
    # Use initial_h_values if available

    best_c5 = np.correlate(best_h, 1 - best_h, mode='full').max() * dx
    
    # Define objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max() * dx
    
    # Constraint: sum(h) == n/2
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    # Parameters
    num_restarts = 50
    max_iter = 1000
    tol = 1e-9
    
    for _ in range(num_restarts):
        # Random perturbation
        perturb = np.random.normal(0, 0.025, n)
        perturbed_guess = np.clip(initial_guess + perturb, 0.0, 1.0)
        
        # Use SLSQP with more iterations and tighter tolerance
        result = minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=tol,
            options={'maxiter': max_iter, 'ftol': 1e-10, 'eps': 1e-8}
        )

            if current_c5 < best_c5:
