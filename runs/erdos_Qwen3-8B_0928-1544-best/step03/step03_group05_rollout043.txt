            h_opt = h_opt * (n_points / 2) / sum_h_opt
        return h_opt

    start_time = time.time()
    while time.time() - start_time < budget_s:
        perturb = np.random.uniform(-0.05, 0.05, size=n_points)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new != 0:
            new_h = new_h * (n_points / 2) / sum_new
        optimized_h = optimize(new_h)
        current_c5 = objective(optimized_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
        if best_c5 < 0.38085:
            break

    final_h = np.clip(best_h, 0, 1)
    sum_final = np.sum(final_h)
    if sum_final != 0:
        final_h = final_h * (n_points / 2) / sum_final
    return final_h, best_c5, n_points
