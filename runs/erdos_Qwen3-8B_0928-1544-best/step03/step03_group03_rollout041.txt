import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 70
    dx = 2.0 / n
    best_c5 = float('inf')
    best_h = np.full(n, 0.5)

    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max() * dx

    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        initial_guess = initial_h_values.copy()
    else:
        initial_guess = np.full(n, 0.5)

    population = []
    population.append(initial_guess.copy())
    for _ in range(49):
        perturb = np.random.normal(0, 0.05, n)
        new_guess = initial_guess + perturb
        new_guess = np.clip(new_guess, 0, 1)
        new_guess = new_guess * 35.0 / np.sum(new_guess)
        population.append(new_guess)

    population = np.array(population)

    for gen in range(100):
        fitness = np.array([compute_overlap(h) for h in population])
        if np.min(fitness) < best_c5:
            best_c5 = np.min(fitness)
            best_h = population[np.argmin(fitness)]

        tournament_size = 5
        parents = []
        for _ in range(10):
            candidates = np.random.choice(len(population), tournament_size, replace=False)
            winner = candidates[np.argmin(fitness[candidates])]
            parents.append(population[winner])

        new_population = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                parent1 = parents[i]
                parent2 = parents[i + 1]
                split = np.random.randint(1, n - 1)
                child1 = np.concatenate([parent1[:split], parent2[split:]])
                child2 = np.concatenate([parent2[:split], parent1[split:]])
                child1 = child1 * 35.0 / np.sum(child1)
                child2 = child2 * 35.0 / np.sum(child2)
                new_population.extend([child1, child2])
            else:
                new_population.append(parents[i])

        for i in range(len(new_population)):
            child = new_population[i]
            perturb = np.random.normal(0, 0.03, n)
            mutated = child + perturb
            mutated = np.clip(mutated, 0, 1)
            mutated = mutated * 35.0 / np.sum(mutated)
            new_population[i] = mutated

        sorted_indices = np.argsort(fitness)
        keep = sorted_indices[:int(0.4 * len(population))]
        new_population.extend(population[keep])
        np.random.shuffle(new_population)
        population = np.array(new_population[:len(population)])

        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n
