            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    dx = 2.0 / n_points
    best_fitness = np.inf
    best_h = None

    def evaluate_fitness(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def normalize_h(h):
        total = h.sum()
        if total == 0:
            return h
        scale = (n_points / 2) / total
        return h * scale

    def tournament_selection(population, tournament_size):
        candidates = random.sample(population, tournament_size)
        fitness = [evaluate_fitness(h) for h in candidates]
        best_idx = np.argmin(fitness)
        return candidates[best_idx]

    def crossover(parent1, parent2):
        split = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:split], parent2[split:]])
        child = normalize_h(child)
        return child

    def mutate(h, mutation_rate):
        h_copy = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h_copy[i] = np.random.uniform(0, 1)
        h_copy = normalize_h(h_copy)
        return h_copy

    generations = 500
    tournament_size = 5
    mutation_rate = 0.1
    crossover_rate = 0.7
    elite_count = 2

    for gen in range(generations):
        fitness = [evaluate_fitness(h) for h in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_fitness:
            best_fitness = fitness[best_idx]
            best_h = population[best_idx]
        if best_fitness < 0.38085:
            break
        new_population = []
        for _ in range(population_size):
            parent1 = tournament_selection(population, tournament_size)
            parent2 = tournament_selection(population, tournament_size)
            if np.random.rand() < crossover_rate:
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
            if np.random.rand() < mutation_rate:
                child = mutate(child, mutation_rate)
            new_population.append(child)
        population = new_population
        if gen % 100 == 0:
            top_indices = np.argsort(fitness)[:5]
            top_h = [population[i] for i in top_indices]
            for h in top_h:
                def objective(x):
                    corr = correlate(x, 1 - x, mode='full')
                    return np.max(corr) * dx
                def constraint(x):
                    return np.sum(x) - n_points / 2
                bounds = [(0.0, 1.0)] * n_points
                res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, tol=1e-8, options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-8})
                if res.success and res.fun < best_fitness:
                    best_fitness = res.fun
                    best_h = res.x
    final_h = best_h.copy()
    final_h[final_h < 0] = 0
    final_h[final_h > 1] = 1
    final_sum = np.sum(final_h)
    if abs(final_sum - n_points / 2) > 1e-6:
        final_h = final_h / final_sum * (n_points / 2)
    return final_h, best_fitness, n_points
