            return max_overlap
        
        def constraint(x):
            return np.sum(x) - n_points / 2
        
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
    
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    if total != 0:
        best_h = best_h * (n_points / 2) / total
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
