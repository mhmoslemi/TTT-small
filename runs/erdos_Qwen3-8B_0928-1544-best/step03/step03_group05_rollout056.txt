from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
import time

    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    for _ in range(300):  # Reduced to allow more iterations
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-9,
            options={'maxiter': 200, 'ftol': 1e-12, 'eps': 1e-8}
        )
        
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.7:
            break

            perturb_h = best_h + noise
            perturb_h = np.clip(perturb_h, 0, 1)
            res = minimize(
                objective,
                perturb_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                tol=1e-9,
                options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-8}
