import numpy as np
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values
    else:
        h_initial = np.full(70, 0.5)
        h_initial = h_initial * 35.0 / np.sum(h_initial)
    
    dx = 2.0 / 70
    best_h = h_initial
    best_c5 = float('inf')

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - 35

    bounds = Bounds([0.0]*70, [1.0]*70)

    # Initial optimization with COBYLA
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='COBYLA',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 1000, 'disp': False}
    )
    best_h = result.x
    best_c5 = result.fun

    # Refinement with SLSQP
    result_slsqp = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 300, 'ftol': 1e-8}
    )
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun

    return best_h, best_c5, 70
