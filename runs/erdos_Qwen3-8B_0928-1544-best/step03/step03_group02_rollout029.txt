import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    from scipy.optimize import minimize
    import time

    # Assume initial_h_values is already defined and accessible globally
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    # Convert initial guess to a binary vector
    h_binary = (h_initial > 0.5).astype(float)
    sum_binary = np.sum(h_binary)
    diff = (n_points / 2) - sum_binary
    h_binary[-1] += diff
    h_binary = np.clip(h_binary, 0.0, 1.0)

    # Objective function to minimize
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Constraint: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Bounds: each h[i] must be between 0 and 1
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial best solution
    best_h = h_binary.copy()
    best_c5 = compute_C5(best_h)
    elapsed_time = 0.0
    start_time = time.time()

    # Optimization parameters
    num_restarts = 40
    max_iter_per_restart = 150
    np.random.seed(seed)

    perturbation_scales = [0.15, 0.1, 0.05, 0.02]

    for scale_idx in range(len(perturbation_scales)):
        perturbation_scale = perturbation_scales[scale_idx]
        for _ in range(num_restarts // len(perturbation_scales)):
            # Perturb the best guess slightly
            perturbed = best_h + np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
            perturbed = np.clip(perturbed, 0.0, 1.0)
            # Adjust to meet sum constraint
            sum_perturbed = np.sum(perturbed)
            if abs(sum_perturbed - n_points / 2) > 1e-6:
                diff = (n_points / 2) - sum_perturbed
                perturbed[-1] += diff
                perturbed = np.clip(perturbed, 0.0, 1.0)

            # Optimize from perturbed guess
            result = minimize(
                compute_C5,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': max_iter_per_restart, 'ftol': 1e-8, 'gtol': 1e-8}
            )

            if result.success and result.fun < best_c5:
                current_h = result.x

                    best_h = current_h

            elapsed_time = time.time() - start_time
            if elapsed_time > budget_s:
                break

    return best_h, best_c5, n_points
