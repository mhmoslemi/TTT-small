import time

def run(seed=42, budget_s=1100, **kwargs):

    from scipy.optimize import minimize
    import time

    h_initial = np.array(initial_h_values)

        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    start_time = time.time()

    # Optimization parameters
    num_restarts = 50
    max_iter = 50
    perturb_scale = 0.15

        sum_perturbed = np.sum(perturbed)

        )
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
