import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    np.random.seed(seed)
    
    # Initial setup
    initial_h = None
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        initial_h = initial_h_values.copy()
    else:
        n = 70
        initial_h = np.zeros(n)
        initial_h[:n//2] = 0.8
        initial_h[n//2:] = 0.2
        initial_h = initial_h * (n / 2) / np.sum(initial_h)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    best_h = initial_h.copy()
    best_c5 = float('inf')
    
    # Optimization parameters
    MU_START = 1e-4
    MU_END = 1e-1
    MU_MUL = 1.1
    MAX_ITER = 1000
    TOL = 1e-8
    
    def calc_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def calc_gradient(h):
        corr = correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j_plus = (i + k) % n_points
            j_minus = (i - k) % n_points
            grad[i] = (1 - h[j_plus]) - h[j_minus]
        # Add barrier terms
        h_clamped = np.clip(h, 1e-12, 1 - 1e-12)
        log_term = np.sum(1 / h_clamped - 1 / (1 - h_clamped))
        sum_diff = np.sum(h) - n_points / 2
        grad += -log_term / MU + 2 * sum_diff / MU / n_points
        return grad
    
    def eval_func(h_flat):
        h = h_flat.copy()
        h_clamped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clamped)
        if np.abs(h_sum - n_points / 2) > 1e-5:
            h_clamped = h_clamped * (n_points / 2) / h_sum
        corr = correlate(h_clamped, 1 - h_clamped, mode='full')

    # Main optimization loop
    current_mu = MU_START
