from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    best_h = np.full(n_points, 0.5)

    def constraint_sum(h):
        return np.sum(h) - 35.0
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    num_iterations = int(budget_s // 3)
