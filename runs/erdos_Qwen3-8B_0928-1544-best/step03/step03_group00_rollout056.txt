import numpy as np
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(70, 0.5)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    best_h = h_initial
    best_c5 = float('inf')
    population_size = 20
    population = [h_initial.copy()]
    for _ in range(population_size - 1):
        mutant = h_initial + np.random.normal(0, 0.05, n_points)
        mutant = np.clip(mutant, 0.0, 1.0)
        mutant = mutant * (n_points / 2.0 / np.sum(mutant))
        population.append(mutant)
    
    generation = 0
    max_generations = int(budget_s / 5)
    start_time = time.time()
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    while generation < max_generations and time.time() - start_time < budget_s:
        scores = np.array([evaluate(h) for h in population])
        idx = np.argmin(scores)
        if scores[idx] < best_c5:
            best_c5 = scores[idx]
            best_h = population[idx]
        survivors = np.argsort(scores)[:len(population)//4]
        survivors = [population[i] for i in survivors]
        new_population = []
        for h in survivors:
            for _ in range(4):
                sigma = 0.05 * (1 - generation / max_generations)
                mutant = h + np.random.normal(0, sigma, n_points)
                mutant = np.clip(mutant, 0.0, 1.0)
                mutant = mutant * (n_points / 2.0 / np.sum(mutant))
                new_population.append(mutant)
        for _ in range(2):
            p1, p2 = np.random.choice(survivors, 2, replace=False)
            offspring = (p1 + p2) / 2
            offspring = np.clip(offspring, 0.0, 1.0)
            offspring = offspring * (n_points / 2.0 / np.sum(offspring))
            new_population.append(offspring)
        population = new_population[:population_size]
        generation += 1
    
    return best_h, best_c5, n_points
