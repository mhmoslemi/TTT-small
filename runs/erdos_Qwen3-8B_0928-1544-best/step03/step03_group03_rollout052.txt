import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    n = 70
    dx = 2.0 / n

    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h = initial_h_values.copy()
    else:
        h = np.full(n, 0.5)
    
    # Ensure sum(h) == n/2
    h_sum = np.sum(h)
    if h_sum != n / 2:
        h = h * (n / 2) / h_sum
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    temp = 0.1
    cooling_rate = 0.99
    max_steps = 5000
    step = 0
    
    def compute_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    while step < max_steps and time.time() - start_time < budget_s:
        # Perturb h
        perturb = np.random.normal(0, temp, n)
        perturbed = h + perturb
        perturbed = np.clip(perturbed, 0.0, 1.0)
        # Normalize sum to n/2
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum == 0:
            perturbed = np.full(n, 0.5)
        else:
            perturbed = perturbed * (n / 2) / perturbed_sum
        
        # Calculate current C5
        current_c5 = compute_c5(perturbed, dx)
        
        # Metropolis criterion
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = perturbed.copy()
        else:
            delta = current_c5 - best_c5
            prob = np.exp(-delta / temp)
            if np.random.rand() < prob:
                best_c5 = current_c5
                best_h = perturbed.copy()
        
        temp *= cooling_rate
        step += 1
    
    return best_h, best_c5, n
