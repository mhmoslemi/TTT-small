import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    
    best_h = h
    best_c5 = np.inf
    iterations = 0
    max_iter = 120
    perturb_freq = 10
    perturb_mag = 0.1

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)

    while iterations < max_iter and budget_s > 0:
        if np.random.rand() < 0.1:
            perturb = np.random.uniform(-perturb_mag, perturb_mag, n_points)
            h = best_h + perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        try:
            res = minimize(
                objective,
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                tol=1e-10,
                options={'maxiter': 20, 'ftol': 1e-12, 'eps': 1e-8}
            )
        except:
            res = minimize(
                objective,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                tol=1e-10,
                options={'maxiter': 20, 'ftol': 1e-12, 'eps': 1e-8}
            )
        
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
                iterations = 0
            else:
                iterations += 1
        else:
            iterations += 1
        
        if best_c5 < 0.38085:
            break
        
        budget_s -= 0.1

    final_h = best_h
    final_h[final_h < 0] = 0
    final_h[final_h > 1] = 1
    final_sum = np.sum(final_h)
    if np.abs(final_sum - n_points / 2) > 1e-6:
        final_h = final_h / final_sum * (n_points / 2)
    
    return final_h, best_c5, n_points
