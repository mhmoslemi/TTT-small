import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[::2] = 1.0
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    res = minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 500, 'ftol': 1e-12}
    )
    
    if res.success and res.fun < best_c5:
        best_h = res.x
        best_c5 = res.fun
    
    final_h = best_h
    final_h[final_h < 0] = 0
    final_h[final_h > 1] = 1
    final_sum = np.sum(final_h)
    if np.abs(final_sum - n_points / 2) > 1e-6:
        final_h = final_h / final_sum * (n_points / 2)
    
    return final_h, best_c5, n_points
