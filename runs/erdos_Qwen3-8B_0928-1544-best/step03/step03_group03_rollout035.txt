        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_index = corr.argmax()
        n_corr = len(corr)
        offset = max_index - (n - 1)
        k = offset

        # Compute shifted array for lag k
        h_shifted = np.zeros_like(best_h)
        for i in range(n):
            shifted_index = i + k
            if 0 <= shifted_index < n:
                h_shifted[i] = best_h[shifted_index]
            else:
                h_shifted[i] = 0.0

        # Compute gradient
        gradient = 1.0 - h_shifted

        # Update h with gradient descent
        best_h = best_h - alpha * gradient
        best_h = np.clip(best_h, 0.0, 1.0)
        sum_h = np.sum(best_h)
        best_h = best_h / sum_h * 35  # Ensure sum is exactly 35

        # Update best_c5
        current_c5 = np.correlate(best_h, 1 - best_h, mode='full').max() * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()

        # Decay learning rate
        if iteration % iteration_interval == 0:
            alpha *= alpha_decay

        # Respect time budget
        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n
