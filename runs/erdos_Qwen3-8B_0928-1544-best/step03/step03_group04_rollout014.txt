            if 0 <= i - lag < n_points:
                grad_max[i] -= h[i - lag]
        log_grad = 1/h - 1/(1 - h)
        sum_grad = 2 * (np.sum(h) - n_points / 2)
        return grad_max - (1/mu)*(log_grad + sum_grad)
    
    mu_sequence = np.logspace(-4, -1, 5)
    
    for mu in mu_sequence:
        def wrapper_objective(h_flat):
            return objective(h_flat, mu)
        
        def wrapper_gradient(h_flat):
            return gradient(h_flat, mu)
        
        res = minimize(wrapper_objective, h, method='BFGS', jac=wrapper_gradient,
                        bounds=[(0.0, 1.0) for _ in range(n_points)],
                        options={'gtol': 1e-8, 'maxiter': 200})
        
        if res.success:
            h = res.x
            h = np.clip(h, 0.0, 1.0)
            total = np.sum(h)
            if abs(total - n_points / 2) > 1e-6:
                delta = (n_points / 2 - total) / (n_points * 1.0)
                h += delta * h
                h = np.clip(h, 0.0, 1.0)
            
            corr = correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
    
    h = np.clip(best_h, 0.0, 1.0)
    total = np.sum(h)
    if abs(total - n_points / 2) > 1e-6:
        delta = (n_points / 2 - total) / (n_points * 1.0)
        h += delta * h
        h = np.clip(h, 0.0, 1.0)
    
    return h, best_c5, n_points
