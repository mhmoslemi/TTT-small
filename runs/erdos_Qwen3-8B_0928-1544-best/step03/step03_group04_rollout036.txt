from scipy.signal import correlate

    best_c5 = np.inf
    start_time = time.time()
    
    def fitness(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def normalize(h):
        h_sum = np.sum(h)
        return h * (n_points / 2) / h_sum if h_sum != 0 else h
    
    def tournament_selection(pop, k=5):

        return pop[idx[np.argmin([fitness(pop[i]) for i in idx])]]

        child = np.copy(parent1)
        idx = np.random.choice(n_points, size=1, replace=False)
        child[idx] = parent2[idx]
        child = normalize(child)
        return child
    
    def mutate(h, rate=0.1, std=0.05):
        h = np.copy(h)
