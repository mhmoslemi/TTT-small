import numpy as np
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    
    n = 70
    dx = 2.0 / n

    # Use initial_h_values if available
    if 'initial_h_values' in globals() and len(initial_h_values) == n:

    else:
        initial_guess = np.full(n, 0.5)

    best_h = initial_guess
    best_c5 = np.correlate(best_h, 1 - best_h, mode='full').max() * dx

    # Define objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max() * dx

    # Constraint: sum(h) == n/2
    def constraint_sum(h):
        return np.sum(h) - n / 2

    bounds = Bounds([0.0]*n, [1.0]*n)
    
    # Multiple restarts with varied perturbations
    num_restarts = 50
    for _ in range(num_restarts):
        # Perturb with different strategies
        if _ % 3 == 0:
            # Random walk perturbation
            perturb = np.random.normal(0, 0.05, n)
        elif _ % 3 == 1:
            # Cyclical pattern shift
            perturb = np.random.rand(n) * 0.1 - 0.05
        else:
            # Random binary flip
            perturb = np.random.choice([-0.1, 0.1], size=n)
        
        perturbed_guess = np.clip(initial_guess + perturb, 0.0, 1.0)
        
        result = minimize(
            fun=objective,
            x0=perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
