        h_scaled = np.clip(h_clipped * scale_factor, 0.0, 1.0)
        current_c5 = np.correlate(h_scaled, 1 - h_scaled, mode='full').max() * dx

        if current_c5 < best_c5 or np.random.random() < np.exp((best_c5 - current_c5) / initial_temp):
            best_h = h_scaled.copy()
            best_c5 = current_c5

        initial_temp *= cooling_rate
        if time.time() - start_time > budget_s - 10:
            break

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max() * dx

    def constraint_sum(h):
        return np.sum(h) - n / 2

    bounds = Bounds([0.0]*n, [1.0]*n)
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-9,
        options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
    )

    if result.success:
        best_h = result.x
        best_c5 = result.fun
    elif time.time() - start_time < budget_s:
        best_h = np.clip(best_h, 0, 1)
        best_c5 = np.correlate(best_h, 1 - best_h, mode='full').max() * dx

    return best_h, best_c5, n
