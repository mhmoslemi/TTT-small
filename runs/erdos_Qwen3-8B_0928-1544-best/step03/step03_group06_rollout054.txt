import numpy as np
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    np.random.seed(seed)
    n_points = 70
    h_initial = np.random.rand(n_points)
    h_initial /= np.sum(h_initial)
    h_initial *= n_points / 2
    h = h_initial.copy()
    dx = 2.0 / n_points
    best_c5 = np.inf
    best_h = h.copy()

    learning_rate = 0.1
    beta = 10.0
    max_iterations = 1000
    iteration = 0

    while iteration < max_iterations and time.time() - start_time < budget_s:
        corr = correlate(h, 1 - h, mode='full')
        c_k = corr * dx
        sum_exp = np.sum(np.exp(beta * c_k))

        grad = np.zeros_like(h)
        for m in range(n_points):
            contrib = 0.0
            for idx in range(len(corr)):
                lag = idx - (n_points - 1)
                shifted_idx = m + lag
                if 0 <= shifted_idx < n_points:
                    contrib += np.exp(beta * c_k[idx]) / sum_exp * (1 - h[shifted_idx])
            grad[m] = dx * beta * contrib

        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 1
        h *= scale

        corr_eval = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr_eval) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.38085:
            break

        iteration += 1
        if iteration % 50 == 0:
            learning_rate *= 0.9
        if iteration % 20 == 0:
            beta *= 1.1
