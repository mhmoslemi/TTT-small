import numpy as np
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    POP_SIZE = 50
    MUT_RATE = 0.1
    COOLING = 0.99
    MAX_GEN = 500
    T = 1.0
    BEST_H = h_initial.copy()
    BEST_C5 = float('inf')

    def calc_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    def repair(h):
        h = np.clip(h, 0, 1)
        diff = n_points / 2 - np.sum(h)
        h[-1] += diff
        return np.clip(h, 0, 1)

    pop = [h_initial + np.random.normal(0, 0.05, n_points) for _ in range(POP_SIZE)]
    pop = [repair(h) for h in pop]

    start_time = time.time()

    for gen in range(MAX_GEN):
        fitness = np.array([calc_c5(h) for h in pop])
        idx = np.argsort(fitness)
        elite = pop[idx[0]]
        if fitness[idx[0]] < BEST_C5:
            BEST_C5 = fitness[idx[0]]
            BEST_H = elite.copy()

        if time.time() - start_time > budget_s:
            break

        new_pop = []
        for _ in range(POP_SIZE // 2):
            a, b = random.sample(range(POP_SIZE), 2)
            p1, p2 = pop[a], pop[b]
            child = (p1 + p2) / 2
            child = repair(child)
            new_pop.append(child)

        for h in new_pop:
            if random.random() < MUT_RATE:
                h[random.randint(0, n_points - 1)] += np.random.normal(0, 0.01)
                h = repair(h)
        pop = new_pop

    return BEST_H, BEST_C5, n_points
