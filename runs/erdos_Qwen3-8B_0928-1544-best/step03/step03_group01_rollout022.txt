import numpy as np
from scipy.optimize import minimize, Bounds
import random

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(70, 0.5)
    
    dx = 2.0 / 70
    n = 70
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale
    
    best_c5 = float('inf')
    best_h = np.copy(initial_h)
    current_h = project_h(initial_h)
    current_c5 = objective(current_h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h
    
    for _ in range(40):
        cluster_size = 7
        num_clusters = n // cluster_size
        cluster_idx = np.repeat(np.arange(num_clusters), cluster_size)
        np.random.shuffle(cluster_idx)
        perturbation = np.zeros(n)
        for i in range(num_clusters):
            cluster_start = i * cluster_size
            cluster_end = cluster_start + cluster_size
            cluster = cluster_idx[cluster_start:cluster_end]
            sigma = 0.15 if _ < 10 else 0.05
            noise = np.random.normal(0, sigma, cluster_size)
            perturbation[cluster] = noise
            
        perturbed_h = current_h + perturbation
        perturbed_h = project_h(perturbed_h)
        
        result = minimize(
            fun=objective,
            x0=perturbed_h,
            method='COBYLA',
            bounds=bounds,
            constraints=constraints,
            tol=1e-6,
            options={'maxiter': 150, 'rhobeg': 0.1, 'rhoend': 1e-4}
        )
        
        if result.success:
            new_c5 = result.fun
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = result.x

    result_final = minimize(

    )
