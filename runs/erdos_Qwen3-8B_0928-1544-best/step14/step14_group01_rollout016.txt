import numpy as np
import time
import scipy.optimize as opt
import cma

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 40
    target_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.6
    # Project initial guess to meet constraints
    initial_h_sum = np.sum(initial_h)
    if initial_h_sum != 0:
        initial_h = initial_h * (target_sum / initial_h_sum)
    else:
        initial_h = np.random.rand(n_points) * 0.6 * (target_sum / 0.6)
    initial_h = np.clip(initial_h, 0.0, 1.0)
    
    def objective(h):
        h = np.clip(h, 0.0, 1.0)
        if np.sum(h) == 0:
            return np.inf
        h = h * (target_sum / np.sum(h))
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # Initialize CMA-ES
    es = cma.CMAEvolutionStrategy(initial_h, 0.5, {'popsize': 50, 'bounds': (0.0, 1.0), 'verb_disp': 0})
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        solutions = es.ask()
        for sol in solutions:
            proj = np.clip(sol, 0.0, 1.0)
            if np.sum(proj) == 0:
                continue
            proj = proj * (target_sum / np.sum(proj))
            obj = objective(proj)
            es.tell(sol, obj)
        if es.get_nr_evaluations() % 10 == 0:
            current_best = es.best.x
            current_value = objective(current_best)
            if current_value < best_c5:
                best_h = current_best.copy()
                best_c5 = current_value
            if time.time() - start_time >= budget_s:
                break
    
    # Local refinement with SLSQP
    def slsqp_objective(x):
        h = np.clip(x, 0.0, 1.0)
        if np.sum(h) == 0:
            return np.inf
        h = h * (target_sum / np.sum(h))
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': lambda x: np.ones_like(x)}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    res = opt.minimize(slsqp_objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8, options={'maxiter': 100})
    best_h = res.x
    best_c5 = slsqp_objective(best_h)
    
    return best_h, best_c5, n_points
