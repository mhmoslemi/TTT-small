        n_points = 50
        target_sum = n_points / 2.0
        initial_h = None
    
    swarm_size = 200
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_iterations = 1000
    no_improve_limit = 50
    
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for _ in range(swarm_size):
        if initial_h is not None and len(particles) < 5:
            h = initial_h.copy()
            h = np.clip(h, 0, 1)
            actual_sum = np.sum(h)
            if actual_sum != 0:
                h = h * (target_sum / actual_sum)
            particles.append(h)
        else:
            h = np.random.rand(n_points)
            actual_sum = np.sum(h)
            if actual_sum == 0:
                h = np.random.rand(n_points) * (target_sum / 0.5)
            else:
                h = h * (target_sum / actual_sum)
            particles.append(h)
        velocities.append(np.random.rand(n_points) * 0.1)
        pbest.append(h.copy())
        pbest_values.append(float('inf'))
    
    for i in range(swarm_size):
        h = particles[i]
        corr = np.correlate(h, 1 - h, mode='full')
        pbest_values[i] = np.max(corr) * (2.0 / n_points)
    
    gbest = particles[0].copy()
    gbest_value = pbest_values[0]
    for i in range(1, swarm_size):
        if pbest_values[i] < gbest_value:
            gbest_value = pbest_values[i]
            gbest = particles[i].copy()
    
    best_h = gbest.copy()
    best_c5 = gbest_value
    start_time = time.time()
    no_improve_count = 0
    
    while time.time() - start_time < budget_s:
        for i in range(swarm_size):
            h = particles[i]
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * (2.0 / n_points)
            
            if current_c5 < pbest_values[i]:
                pbest[i] = h.copy()
                pbest_values[i] = current_c5
            
            v = velocities[i]
            r1 = np.random.rand()
            r2 = np.random.rand()
            v = w * v + c1 * r1 * (pbest[i] - h) + c2 * r2 * (gbest - h)
            velocities[i] = v
            
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            actual_sum = np.sum(h_new)
            if actual_sum != 0:
                h_new = h_new * (target_sum / actual_sum)
            else:
                h_new = np.random.rand(n_points) * (target_sum / 0.5)
            particles[i] = h_new.copy()
        
        for i in range(swarm_size):
            if pbest_values[i] < gbest_value:
                gbest_value = pbest_values[i]
                gbest = pbest[i].copy()
        
        if gbest_value < best_c5:
            best_c5 = gbest_value
            best_h = gbest.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        if no_improve_count >= no_improve_limit:
            break
    
    return best_h, best_c5, n_points
