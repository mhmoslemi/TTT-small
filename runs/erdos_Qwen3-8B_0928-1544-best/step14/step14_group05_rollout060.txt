        # Clip to [0, 1]

        # Rescale to ensure sum is n_points / 2

        if current_c5 < best_c5:

            best_h = h.copy()
            no_improve = 0
        else:

        # Adaptive learning rate
        if no_improve >= 50:

def run(seed=42, budget_s=1100, **kwargs):

    ones_count = n_points // 2
    dx = 2.0 / n_points

        h = initial_h_values.copy()
    else:

        h = np.random.rand(n_points) * 0.5
    h = h * (ones_count / h_sum)

    best_h = h.copy()
    best_c5 = 0.0
    start_time = time.time()

    learning_rate = 0.1
    decay_rate = 0.95
    iterations = 0
    max_iter = 10000
    no_improve = 0
    perturb_freq = 200
    perturb_strength = 0.005

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_c5(h)

    while time.time() - start_time < budget_s and iterations < max_iter:
        iterations += 1
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)

        x_array = np.arange(n_points)
        valid = np.logical_and(x_array + k >= 0, x_array + k < n_points)
        indices = np.where(valid)[0]
        gradient = np.zeros(n_points)

        h = h - learning_rate * gradient
        h = np.clip(h, 0.0, 1.0)
