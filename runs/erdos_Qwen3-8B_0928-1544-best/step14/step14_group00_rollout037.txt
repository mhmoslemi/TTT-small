import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = 35
    population_size = 50
    F_start, F_end = 0.8, 1.0
    CR_start, CR_end = 0.8, 1.0
    mutation_rate = 0.2
    shrink_rate = 0.98
    local_steps = 20
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    initial_h_sum = np.sum(initial_h)
    initial_h_scaled = initial_h * (ones_count / initial_h_sum) if initial_h_sum != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
    population = [initial_h_scaled]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        population.append(h * (ones_count / h_sum))
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()
    F = F_start
    CR = CR_start
    stagnation = 0
    
    def gradient_descent(h):
        h = h.copy()
        for _ in range(local_steps):
            corr = np.correlate(h, 1 - h, mode='full')
            k_idx = np.argmax(corr)
            k = k_idx - (n_points - 1)
            grad = np.zeros_like(h)
            for i in range(n_points):
                if k + i < 0:
                    term1 = 0
                else:
                    term1 = 1 - h[(k + i) % n_points]
                if i - k < 0:
                    term2 = 0
                else:
                    term2 = h[(i - k) % n_points]
                grad[i] = term1 - term2
            learning_rate = 0.1
            h = h - learning_rate * grad
            norm = np.sum(h)
            h = h * (ones_count / norm)
            h = np.clip(h, 0, 1)
        return h
    
    while time.time() - start_time < budget_s:
        for _ in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[_])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            new_h = gradient_descent(u_scaled)
            new_c5 = objective(new_h)
            if new_c5 < objective(population[_]):
                population[_] = new_h
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
                    stagnation = 0
            else:
                stagnation += 1
            if stagnation > 100:
                F = max(F_start, F - 0.01)
                CR = min(CR_end, CR + 0.01)
                stagnation = 0
            if time.time() - start_time >= budget_s:
                break
    
    return best_h, best_c5, n_points
