import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 200
    w = 0.7
    c1 = c2 = 2.0

    # Initialize population
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    initial_h = initial_h + np.random.normal(0, 0.1, n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h_sum = np.sum(initial_h)
    if initial_h_sum == 0:
        initial_h = np.random.rand(n_points) * 0.5
    initial_h = initial_h * (ones_count / initial_h_sum)

    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = initial_h.copy() + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h = h * (ones_count / h_sum)
        population.append(h)

    pbest = [h.copy() for h in population]
    pbest_c5 = [0.0] * population_size
    for i in range(population_size):
        corr = np.correlate(population[i], 1 - population[i], mode='full')
        pbest_c5[i] = np.max(corr) * (2.0 / n_points)
    gbest = pbest[np.argmin(pbest_c5)]
    gbest_c5 = min(pbest_c5)
    best_h = gbest
    best_c5 = gbest_c5
    velocities = [np.zeros(n_points) for _ in range(population_size)]

    time_start = time.time()

    while time.time() - time_start < budget_s:
        for i in range(population_size):
            h = population[i]
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * (2.0 / n_points)

            if current_c5 < pbest_c5[i]:
                pbest[i] = h.copy()
                pbest_c5[i] = current_c5
                if current_c5 < gbest_c5:
                    gbest = pbest[i].copy()
                    gbest_c5 = current_c5
                    best_h = gbest
                    best_c5 = gbest_c5

                h_new = np.random.rand(n_points) * 0.5
            h_new = h_new * (ones_count / h_sum_new)
            population[i] = h_new

    return best_h, best_c5, n_points
