        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def gradient(x):
        h = x.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        grad = np.zeros_like(h)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            grad_plus = corr_plus[max_index] * (2.0 / n_points)
            h_minus = h.copy()
            h_minus[i] -= eps
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            grad_minus = corr_minus[max_index] * (2.0 / n_points)
            grad[i] = (grad_plus - grad_minus) / (2 * eps)
        return grad
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count}]
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(objective, h, method='L-BFGS-B', jac=gradient, bounds=bounds, constraints=cons, options={'maxiter': 200})
    return result.x, objective(result.x), n_points
