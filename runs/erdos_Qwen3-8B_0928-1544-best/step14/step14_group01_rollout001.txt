import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    ones_count = n_points // 2
    bounds = [(0, 1) for _ in range(n_points)]
    
    def clip_and_normalize(h):
        s = np.sum(h)
        if s == 0:
            return np.clip(np.zeros_like(h), 0, 1)
        scaled = h * (n_points / 2) / s
        return np.clip(scaled, 0, 1)
    
    def objective(h):
        h_norm = clip_and_normalize(h)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / n_points)
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.random.rand(n_points) * 0.6
    initial_h = clip_and_normalize(initial_h)
    
    result_de = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=15,
        mutation=(0.5, 1), recombination=0.8, tol=1e-5, maxiter=200, seed=seed
    )
    best_h = result_de.x
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    result_slsqp = minimize(
        objective, best_h, method='SLSQP', bounds=bounds, constraints=cons,
        tol=1e-6, options={'maxiter': 100}
    )
    best_h = result_slsqp.x
    
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
