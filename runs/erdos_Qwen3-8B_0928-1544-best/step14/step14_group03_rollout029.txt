import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * (ones_count / np.sum(h)) if np.sum(h) != 0 else np.random.rand(n_points) * (ones_count / 0.5)
    
    # Objective function for minimization
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Compute numerical gradient
    def gradient(h):
        eps = 1e-6
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus_sum = np.sum(h_plus)
            if h_plus_sum != 0:
                h_plus = h_plus * (ones_count / h_plus_sum)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            obj_plus = np.max(corr_plus) * dx
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus_sum = np.sum(h_minus)
            if h_minus_sum != 0:
                h_minus = h_minus * (ones_count / h_minus_sum)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            obj_minus = np.max(corr_minus) * dx
            
            grad[i] = (obj_plus - obj_minus) / (2 * eps)
        return grad
    
    # Optimization constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = optimize.minimize(objective, h, method='L-BFGS-B', jac=gradient, bounds=bounds, tol=1e-6)
    
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
