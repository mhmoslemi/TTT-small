import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 0.5 + 0.25
    h = h * (n_points / 2) / h.sum() if h.sum() != 0 else np.random.rand(n_points) * 0.5 * (n_points / 2) / 0.5
    dx = 2.0 / n_points

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - (n_points / 2)

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def optimize_h(h_initial):
        result = opt.minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8})
        if result.success:
            return result.x, result.fun
        result = opt.minimize(objective, h_initial, method='Cobyla', bounds=bounds, constraints=cons,
                              options={'maxiter': 500})
        return result.x, result.fun

    h_opt, c5 = optimize_h(h)
    return h_opt, c5, n_points
