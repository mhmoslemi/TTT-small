import numpy as np
import time
import cma
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    initial_h = initial_h * (ones_count / np.sum(initial_h)) if np.sum(initial_h) != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)

    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return np.zeros(n_points) + (ones_count / n_points)
        scale = (ones_count) / sum_h
        return h_clipped * scale

    def objective(h):
        h_repaired = repair(h)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * (2.0 / n_points)

    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    for _ in range(3):
        # CMA-ES optimization
        x0 = best_h.copy()
        try:
            result = cma.fmin_cma(objective, x0, sigma=0.5, options={'maxiter': 100, 'popsize': 50, 'bounds': [(0.0, 1.0) for _ in range(n_points)]})
            best_h = result['x']
            best_c5 = objective(best_h)
        except:
            pass

        if time.time() - start_time > budget_s:
            break

        # Gradient refinement
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result_refine = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-6, options={'maxiter': 50})
        best_h = result_refine.x
        best_c5 = objective(best_h)

        if time.time() - start_time > budget_s:
            break

    best_h = repair(best_h)
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * (2.0 / n_points)
    return best_h, final_c5, n_points
