def run(seed=42, budget_s=1100, **kwargs):

    alpha = 0.1
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = -np.inf
        best_k = 0

                best_k = k
        k = best_k
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = (i + k) % n_points

            gradient[i] = (1 - h[i_plus_k]) - h[i_minus_k]
