import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points / 2  # 35
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (ones_count / np.sum(h))
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - ones_count}]
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * (2.0 / n_points)
        return max_overlap
    
    start_time = time.time()
    result = opt.minimize(
        objective, 
        h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={
            'ftol': 1e-8, 
            'maxiter': 10000,
            'eps': 1e-4
        }
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        # Use initial guess if optimization exceeded time
        c5_bound = objective(h)
        h_best = h.copy()
    else:
        c5_bound = result.fun
        h_best = result.x
    
    if np.isnan(c5_bound) or np.isinf(c5_bound):
        return h.tolist(), float('inf'), n_points
    
    return h_best.tolist(), c5_bound, n_points
