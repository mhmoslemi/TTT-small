import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

    # Initialize with provided or random values
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h_sum = np.sum(initial_h)
    if initial_h_sum == 0:
        initial_h = np.random.rand(n_points) * 0.5
        initial_h_sum = np.sum(initial_h)
    initial_h_scaled = initial_h * (ones_count / initial_h_sum)

    def objective(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h_clipped)
        if h_sum == 0:
            return float('inf')
        h_scaled = h_clipped * (ones_count / h_sum)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = initial_h_scaled.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Differential evolution with multiple restarts
    for _ in range(4):
        result = opt.differential_evolution(
            objective,
            [(0.0, 1.0)] * n_points,
            strategy='best1bin',
            maxiter=100,
            popsize=50,
            tol=1e-5,
            mutation=(0.5, 1.0),
            recombination=0.8,
            polish=True,
            disp=False,
            seed=seed + _
        )
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
        if time.time() - start_time > budget_s:
            break

    # Sequential Quadratic Programming refinement
    def slsqp_objective(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h_clipped)
        if h_sum == 0:
            return float('inf')
        h_scaled = h_clipped * (ones_count / h_sum)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - ones_count}]
    bounds_slsqp = [(0.0, 1.0)] * n_points

    result_slsqp = opt.minimize(
        slsqp_objective,
        best_h,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons,
        tol=1e-7,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    best_h = result_slsqp.x
    best_c5 = slsqp_objective(best_h)

    # Final clipping and scaling
    best_h_clipped = np.clip(best_h, 0.0, 1.0)
    best_h_scaled = best_h_clipped * (ones_count / np.sum(best_h_clipped))
    final_c5 = slsqp_objective(best_h_scaled)

    return best_h_scaled, final_c5, n_points
