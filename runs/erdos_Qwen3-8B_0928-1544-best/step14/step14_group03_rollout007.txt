import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        sorted_indices = np.argsort(h)[::-1]
        h = np.zeros_like(h)
        h[sorted_indices[:ones_count]] = 1
    else:
        h = np.zeros(n_points, dtype=int)
        indices = np.random.permutation(n_points)
        h[indices[:ones_count]] = 1
    
    corr = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_c5 = current_c5
    best_h = h.copy()
    
    temp = 1.0
    alpha = 0.99
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        h_candidate = h.copy()
        h_candidate[i], h_candidate[j] = 0, 1
        
        corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(corr_candidate) * dx
        delta = c5_candidate - current_c5
        
        if delta < 0:
            h = h_candidate
            current_c5 = c5_candidate
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            prob = np.exp(-delta / temp)
            if np.random.rand() < prob:
                h = h_candidate
                current_c5 = c5_candidate
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        
        temp *= alpha
    
    return best_h.tolist(), best_c5, n_points
