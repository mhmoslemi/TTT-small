import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    population_size = 75
    F_start, F_end = 0.8, 1.0
    CR_start, CR_end = 0.8, 1.0
    mutation_rate = 0.2
    shrink_rate = 0.99
    local_steps = 50
    local_lr = 0.05

    def norm_h(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        return h * (ones_count / total) if total != 0 else h * (ones_count / 0.5)

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    initial_h = norm_h(initial_h)
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points) * 0.5
        h = norm_h(h)
        population.append(h)

    def objective(h):
        return compute_C5(h)

    best_h = population[0].copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    F = F_start
    CR = CR_start
    stagnation = 0

    while time.time() - start_time < budget_s:
        for _ in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[_])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = norm_h(u)
            new_c5 = objective(u)
            if new_c5 < objective(population[_]):
                population[_] = u
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u.copy()
                    stagnation = 0
            else:
                stagnation += 1
            if stagnation > 50:
                F = max(F_start, F - 0.01)
                CR = min(CR_end, CR + 0.01)
                stagnation = 0

        if time.time() - start_time >= budget_s:

        top_candidates = sorted(population, key=objective)[:int(population_size * 0.2)]
        for h in top_candidates:
            h_local = h.copy()

                grad = np.zeros(n_points)

                    h_plus = h_local.copy()
