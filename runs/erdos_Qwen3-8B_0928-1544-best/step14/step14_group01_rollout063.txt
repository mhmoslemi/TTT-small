    population = [initial_h_scaled]

        if h_sum == 0:

        population.append(h * (ones_count / h_sum))

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()
    F = 0.8
