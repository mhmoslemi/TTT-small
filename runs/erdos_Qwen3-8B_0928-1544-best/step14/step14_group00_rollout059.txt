import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = 35
    dx = 2.0 / n_points
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    positions = []
    velocities = []
    for _ in range(200):
        noise = np.random.normal(0, 0.1, n_points)
        h = initial_h + noise
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
        h *= ones_count / h_sum
        positions.append(h.copy())
        velocities.append(np.zeros(n_points))

    pbest = [np.copy(p) for p in positions]
    pbest_values = np.array([objective(p) for p in positions])
    gbest = np.copy(positions[0])
    best_c5 = pbest_values[0]
    best_h = gbest.copy()

    w, c1, c2 = 0.7, 2.0, 2.0
    start_time = time.time()
    stagnation = 0

    while time.time() - start_time < budget_s:
        for i in range(200):
            current_h = np.array(positions[i])
            current_c5 = objective(current_h)
            if current_c5 < pbest_values[i]:
                pbest[i] = current_h.copy()
                pbest_values[i] = current_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
            r1, r2 = np.random.rand(), np.random.rand()
            v = velocities[i] * w + c1 * r1 * (pbest[i] - current_h) + c2 * r2 * (gbest - current_h)
            velocities[i] = v
            new_pos = current_h + v
            new_pos = np.clip(new_pos, 0, 1)
            new_sum = np.sum(new_pos)
            if new_sum == 0:
                new_pos = np.random.rand(n_points) * 0.5
                new_sum = np.sum(new_pos)
            new_pos *= ones_count / new_sum
            positions[i] = new_pos
        stagnation += 1
        if stagnation > 500:
            w = max(0.4, w - 0.01)
            stagnation = 0
    return best_h, best_c5, n_points
