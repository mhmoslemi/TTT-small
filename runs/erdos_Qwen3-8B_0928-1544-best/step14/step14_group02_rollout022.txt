        top_indices = indices[:int(len(indices) * 0.1)]
        
        delta = 0.01
        for idx in top_indices:
            i = idx
            j = (i + k) % n_points
            if h[i] > 0.5:
                h[i] -= delta
            elif h[j] < 0.5:
                h[j] += delta
        
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (ones_count / h_sum)
        else:
            h = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
        
        if iteration % 50 == 0:
            perturb_idx = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
            h[perturb_idx] += np.random.uniform(-0.15, 0.15, size=len(perturb_idx))
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            if h_sum != 0:
                h = h * (ones_count / h_sum)
            else:
                h = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
            
            corr_perturb = np.correlate(h, 1 - h, mode='full')
            max_overlap_perturb = np.max(corr_perturb) * (2.0 / n_points)
            if max_overlap_perturb < best_c5:
                best_h = h.copy()
                best_c5 = max_overlap_perturb
        
        iteration += 1
    
    return best_h, best_c5, n_points
