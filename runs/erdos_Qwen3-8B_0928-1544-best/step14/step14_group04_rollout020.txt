    h = np.clip(h, 0.0, 1.0)
    h = h * (ones_count / np.sum(h)) if np.sum(h) != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    alpha = 0.1
    dx = 2.0 / n_points
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        idx = np.argmax(corr)
        k = idx - (n_points - 1)
        # Compute gradient for the max k
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            gradient[i] = (1 - h[j]) * dx
        # Update h
        h -= alpha * gradient
        # Clip values
        h = np.clip(h, 0.0, 1.0)
        # Enforce sum constraint
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
        else:
            h = h * (ones_count / h_sum)
        # Evaluate new C5
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_max = np.max(new_corr) * dx
        if new_max < best_c5:
            best_c5 = new_max
            best_h = h.copy()
            alpha = 0.1 * 0.95
        else:
            alpha *= 0.95
        if alpha < 1e-6:
            break
    return best_h, best_c5, n_points
