import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    desired_sum = n_points / 2.0
    dx = 2.0 / n_points
    population_size = 50
    mutation_std = 0.05
    selection_ratio = 0.2
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h_sum = np.sum(initial_h)
        initial_h = initial_h * (desired_sum / initial_h_sum) if initial_h_sum != 0 else np.random.rand(n_points) * 0.5
        population.append(initial_h)
        for _ in range(population_size - 1):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            h = h * (desired_sum / h_sum) if h_sum != 0 else h
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h_sum = np.sum(h)
            h = h * (desired_sum / h_sum) if h_sum != 0 else h
            population.append(h)
    
    pop_c5 = [compute_c5(h) for h in population]
    best_idx = np.argmin(pop_c5)
    best_h, best_c5 = population[best_idx], pop_c5[best_idx]
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        top_idx = np.argsort(pop_c5)[:int(population_size * selection_ratio)]
        top_candidates = [population[i] for i in top_idx]
        
        new_candidates = []
        for h in top_candidates:
            noise = np.random.normal(0, mutation_std, n_points)
            mutated = h + noise
            mutated = np.clip(mutated, 0, 1)
            mutated_sum = np.sum(mutated)
            if mutated_sum == 0:
                mutated = np.random.rand(n_points)
                mutated = mutated * (desired_sum / np.sum(mutated))
            else:
                mutated = mutated * (desired_sum / mutated_sum)
            new_candidates.append(mutated)
        
        combined = population + new_candidates
        combined_c5 = [compute_c5(h) for h in combined]
        sorted_indices = np.argsort(combined_c5)
        new_population = [combined[i] for i in sorted_indices[:population_size]]
        
        population = new_population
        pop_c5 = [compute_c5(h) for h in population]
        
        current_min = np.min(pop_c5)
        if current_min < best_c5:
            best_c5 = current_min
            best_idx = np.argmin(pop_c5)
            best_h = population[best_idx]
    
    return best_h, best_c5, n_points
