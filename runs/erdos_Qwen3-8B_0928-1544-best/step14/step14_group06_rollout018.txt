import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5

    # Scale initial solution to satisfy the integral constraint
    initial_h_sum = np.sum(initial_h)
    initial_h_scaled = initial_h * (ones_count / initial_h_sum) if initial_h_sum != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)

    def objective(h):
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            return float('inf')
        h_scaled = h * (ones_count / h_sum)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = initial_h_scaled.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Use differentials search with restarts and adaptive parameters
    for _ in range(2):
        # Evaluate initial point
        current_h = best_h.copy()
        current_c5 = objective(current_h)

        # Differential evolution parameters
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=150,
            popsize=30,
            tol=1e-5,
            mutation=(0.5, 1.0),
            recombination=0.8,
            polish=True,
            disp=False,
            seed=seed + _
        )

        if result.fun < best_c5:
            best_h = result.x

        if time.time() - start_time > budget_s:
            break

    # Final evaluation and pruning
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h_sum = np.sum(best_h)
    best_h_scaled = best_h * (ones_count / best_h_sum) if best_h_sum != 0 else initial_h_scaled
    final_c5 = objective(best_h_scaled)
