import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 50
    required_sum = n_points / 2
    swarm_size = 200
    w = 0.7
    c1, c2 = 1.5, 1.5
    max_iter = 1000
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values, dtype=np.float64)
        h_initial = h_initial * (required_sum / np.sum(h_initial))
        h_swarm = np.tile(h_initial, (swarm_size, 1))
    else:
        h_swarm = np.random.rand(swarm_size, n_points)
        h_swarm = h_swarm * (required_sum / np.sum(h_swarm, axis=1, keepdims=True))
    v_swarm = np.random.rand(swarm_size, n_points) * 0.1
    pbest_swarm = h_swarm.copy()
    pbest_values = np.full(swarm_size, np.inf)
    gbest = h_swarm[0].copy()
    gbest_value = np.inf
    best_c5 = np.inf
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(swarm_size):
            h = h_swarm[i]
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * (2.0 / n_points)
            if current_c5 < pbest_values[i]:
                pbest_values[i] = current_c5
                pbest_swarm[i] = h.copy()
        best_index = np.argmin(pbest_values)
        if pbest_values[best_index] < gbest_value:
            gbest_value = pbest_values[best_index]
            gbest = pbest_swarm[best_index].copy()

        for i in range(swarm_size):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            v_swarm[i] = w * v_swarm[i] + c1 * r1 * (pbest_swarm[i] - h_swarm[i]) + c2 * r2 * (gbest - h_swarm[i])

            else:
                h_swarm[i] = np.random.rand(n_points) * (required_sum / 0.5)
        if best_c5 < 0.38085:

    return gbest, best_c5, n_points
