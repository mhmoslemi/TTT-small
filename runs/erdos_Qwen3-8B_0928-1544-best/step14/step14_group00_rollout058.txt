import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    pop_size = 200
    w, c1, c2 = 0.7, 2.0, 2.0
    max_iter = 100000

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 0.5
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (ones_count / h_sum)
    else:
        h = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
    particles = [h + np.random.normal(0, 0.1, n_points) for _ in range(pop_size)]
    for p in particles:
        p = np.clip(p, 0, 1)
        p_sum = np.sum(p)
        if p_sum != 0:
            p = p * (ones_count / p_sum)
        else:
            p = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
    velocities = np.zeros((pop_size, n_points))

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    pbest, pbest_c5 = particles, [obj(p) for p in particles]
    gbest, gbest_c5 = pbest[0], pbest_c5[0]
    best_h, best_c5 = gbest.copy(), gbest_c5

    start_time = time.time()
    stagnation = 0

    for _ in range(max_iter):

        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()

            p_sum = np.sum(particles[i])

                particles[i] = particles[i] * (ones_count / p_sum)
            else:
                particles[i] = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)

            curr = obj(particles[i])
            if curr < pbest_c5[i]:
