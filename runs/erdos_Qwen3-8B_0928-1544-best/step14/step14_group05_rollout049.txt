import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        h = np.random.rand(n_points) * 0.5

    # Normalize h to sum to ones_count
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (ones_count / h_sum)
    else:
        h = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    start_time = time.time()
    learning_rate = 0.1
    decay_rate = 0.95
    max_steps = 1000

    for steps in range(max_steps):
        if time.time() - start_time >= budget_s:
            break

        # Compute cross-correlation and find max lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        k_star_idx = np.argmax(corr)
        k_star = k_star_idx - (n_points - 1)

        # Compute gradient for k_star
        grad = np.zeros_like(h)
        for j in range(n_points):
            j_plus_k = j + k_star
            if 0 <= j_plus_k < n_points:
                term1 = 1.0 - h[j_plus_k]
            else:
                term1 = 0.0
            j_minus_k = j - k_star
            if 0 <= j_minus_k < n_points:
                term2 = -h[j_minus_k]
            else:
                term2 = 0.0
            grad[j] = term1 + term2

        # Update h
        h = h - learning_rate * grad
        # Clip to [0,1]
        h = np.clip(h, 0.0, 1.0)
        # Rescale to correct sum
        current_sum = np.sum(h)
        if current_sum != 0:
            h = h * (ones_count / current_sum)
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
        # Decay learning rate
        learning_rate *= decay_rate

    return best_h, best_c5, n_points

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return (np.max(corr) * 2.0) / h.size
