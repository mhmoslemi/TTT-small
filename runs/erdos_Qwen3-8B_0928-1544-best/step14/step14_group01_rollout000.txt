import numpy as np
import time
import scipy.optimize as opt
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 40
    dx = 2.0 / n_points
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    initial_h = initial_h * (ones_count / np.sum(initial_h)) if np.sum(initial_h) != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)

    def project_h(raw_h):
        h_clamped = np.clip(raw_h, 0.0, 1.0)
        sum_h = np.sum(h_clamped)
        if sum_h == 0:
            return np.zeros_like(h_clamped)
        h_scaled = h_clamped * (ones_count / sum_h)
        return h_scaled

    def objective(raw_h):
        h = project_h(raw_h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(objective, bounds, strategy='best1bin', maxiter=100, popsize=50, tol=1e-6, mutation=(0.5, 1), recombination=0.9, seed=seed, polish=True, polish_options={"maxiter": 50})
    best_h = project_h(result.x)
    best_c5 = result.fun

    # SLSQP local refinement
    x = cp.Variable(n_points)
    constraints = [cp.sum(x) == ones_count, 0 <= x, x <= 1]
    def slsqp_objective(x_val):
        h = project_h(x_val)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    # Use finite difference gradient
    def grad_objective(x_val):
        eps = 1e-6
        grad = np.zeros_like(x_val)
        for i in range(n_points):
            x_forward = x_val.copy()
            x_forward[i] += eps
            x_backward = x_val.copy()
            x_backward[i] -= eps
            f_forward = slsqp_objective(x_forward)
            f_backward = slsqp_objective(x_backward)
            grad[i] = (f_forward - f_backward) / (2 * eps)
        return grad

    results = opt.minimize(slsqp_objective, best_h, method='SLSQP', jac=grad_objective, constraints=constraints, tol=1e-6, options={'maxiter': 50})
    best_c5 = results.fun
    best_h = results.x

    return best_h, best_c5, n_points
