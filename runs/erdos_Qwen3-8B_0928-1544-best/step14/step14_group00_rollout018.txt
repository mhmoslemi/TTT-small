    gbest = population[0].copy()
    gbest_value = objective(gbest)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = int(budget_s / 0.5)
    stagnation_threshold = 100
    stagnation_count = 0
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        for i in range(population_size):
            h = population[i]
            current_val = objective(h)
            if current_val < pbest_values[i]:
                pbest[i] = h.copy()
                pbest_values[i] = current_val
                if current_val < gbest_value:
                    gbest = h.copy()
                    gbest_value = current_val
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest[i] - population[i]) + c2 * r2 * (gbest - population[i])
            population[i] += velocities[i]
            population[i] = np.clip(population[i], 0.0, 1.0)
            current_sum = np.sum(population[i])
            if current_sum == 0:
                population[i] = initial_h.copy()
            else:
                population[i] = population[i] * required_sum / current_sum
        if gbest_value == best_value:
            stagnation_count += 1
        else:
            best_value = gbest_value
            stagnation_count = 0
        if stagnation_count > stagnation_threshold:
            w = max(w - 0.01, 0.4)
            stagnation_count = 0
        if gbest_value < 0.38085:
            break

    return gbest, gbest_value, n_points
