import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5

    # Scale initial solution to satisfy the integral constraint
    initial_h_sum = np.sum(initial_h)

        sum_h = np.sum(h_clipped)
        if sum_h == 0:

        scale = (ones_count) / sum_h
        return h_clipped * scale

    def objective(h):
        h_repair = repair(h)
        corr = np.correlate(h_repair, 1 - h_repair, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = initial_h_scaled.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Use differentials search with restarts
    for _ in range(3):
        # Evaluate initial point
        current_h = best_h.copy()
        current_c5 = objective(current_h)

        # Differential evolution parameters
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=200,  # increased max iterations
            popsize=100,  # increased population size
            tol=1e-5,
            mutation=(0.5, 1.0),
            recombination=0.8,
            polish=True,
            disp=False,
            seed=seed + _
        )

        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun

        if time.time() - start_time > budget_s:
            break

    # Final evaluation and pruning
    best_h = repair(best_h)
    final_c5 = objective(best_h)

    # Gradient refinement
    def objective_refine(h):
        return objective(h)

    res = opt.minimize(objective_refine, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, options={'maxiter': 100})

    final_h_repair = repair(final_h)
    final_c5_refined = objective(final_h_repair)

    return best_h, final_c5, n_points
