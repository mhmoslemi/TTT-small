import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    h_sum = np.sum(initial_h)
    initial_h_scaled = initial_h * (ones_count / h_sum) if h_sum != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return initial_h_scaled.copy()
        return h_clipped * (ones_count / sum_h)

    def compute_c5(h):
        h_proj = project(h)
        corr = np.correlate(h_proj, 1 - h_proj, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def grad_func(x):
        eps = 1e-4
        grad = np.zeros_like(x)
        for i in range(len(x)):
            x_p = x.copy()
            x_m = x.copy()
            x_p[i] += eps
            x_m[i] -= eps
            c_p = compute_c5(project(x_p))
            c_m = compute_c5(project(x_m))
            grad[i] = (c_p - c_m) / (2 * eps)
        return grad

    def obj_func(x):
        return compute_c5(x)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(project(x)) - ones_count}]
    bounds = [(0., 1.) for _ in range(n_points)]

    start_time = time.time()
    result = minimize(obj_func, initial_h_scaled, method='SLSQP', jac=grad_func, bounds=bounds, constraints=cons, tol=1e-5, options={'maxiter': 500})
    if time.time() - start_time > budget_s - 10:
        result = minimize(obj_func, initial_h_scaled, method='L-BFGS-B', bounds=bounds, tol=1e-5, options={'maxiter': 500})
    best_h = project(result.x)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
