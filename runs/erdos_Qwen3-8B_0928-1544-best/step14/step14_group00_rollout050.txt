import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points // 2
    population_size = 200
    inertia = 0.7
    c1 = c2 = 2.0
    max_iter = int(budget_s / 0.1)

    if 'initial_h_values' in globals():
        base_h = initial_h_values.copy()
    else:
        base_h = np.random.rand(n_points) * 0.5
    positions = []
    velocities = []
    pbests = []
    pbest_c5s = []

    for _ in range(population_size):
        h = base_h + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        h = h * (target_sum / h_sum)
        positions.append(h)
        velocities.append(np.zeros(n_points))
        pbests.append(h.copy())
        corr = np.correlate(h, 1 - h, mode='full')
        pbest_c5s.append(np.max(corr) * (2.0 / n_points))

    gbest = positions[0].copy()
    gbest_c5 = pbest_c5s[0]

    start_time = time.time()
    stagnation = 0

    while time.time() - start_time < budget_s:
        for i in range(population_size):
            h = positions[i]
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * (2.0 / n_points)
            if current_c5 < pbest_c5s[i]:
                pbests[i] = h.copy()
                pbest_c5s[i] = current_c5
            if current_c5 < gbest_c5:
                gbest = h.copy()
                gbest_c5 = current_c5
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            vel = velocities[i] * inertia + c1 * r1 * (pbests[i] - h) + c2 * r2 * (gbest - h)
            velocities[i] = vel
            new_h = h + vel
            new_h = np.clip(new_h, 0, 1)
            new_sum = np.sum(new_h)
            if new_sum == 0:
                new_h = np.random.rand(n_points) * 0.5
                new_sum = np.sum(new_h)
            new_h = new_h * (target_sum / new_sum)
            positions[i] = new_h

        if np.allclose(pbest_c5s, gbest_c5, atol=1e-6):
            stagnation += 1
            if stagnation > 50:
                inertia = max(0.4, inertia - 0.01)
                stagnation = 0
        else:
            stagnation = 0

    return gbest, gbest_c5, n_points
