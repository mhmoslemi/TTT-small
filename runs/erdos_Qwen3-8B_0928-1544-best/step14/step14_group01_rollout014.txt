    while time.time() - start_time < budget_s:
        for _ in range(150):
            a, b, c = np.random.choice(len(population), 3, replace=False)
            new_h = mutate(population[a], 0.25, F, CR)
            new_c5 = evaluate(new_h)
            if new_c5 < evaluate(population[_]):
                population[_] = new_h
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
                    stagnation = 0
            else:
                stagnation += 1
        if stagnation > 50:
            F = max(0.5, F - 0.01)
            CR = min(1.0, CR + 0.01)
            stagnation = 0
        if time.time() - start_time >= budget_s:
            break

    return best_h, best_c5, n_points
