            adjusted = adjusted * target_sum / adjusted_sum
        new_c5 = _compute_c5(adjusted)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = adjusted.copy()
            stagnation = 0
        else:
            if np.abs(new_c5 - best_c5) < improvement_threshold:
                stagnation += 1
            else:
                stagnation = 0
        if stagnation > 50:
            perturb_indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
            for idx in perturb_indices:
                adjusted[idx] += np.random.uniform(-0.1, 0.1)
                adjusted[idx] = np.clip(adjusted[idx], 0.0, 1.0)
            adjusted_sum = np.sum(adjusted)
            if adjusted_sum == 0:
                adjusted = np.random.rand(n_points) * 0.5
                adjusted = adjusted * target_sum / np.sum(adjusted)
            else:
                adjusted = adjusted * target_sum / adjusted_sum
            new_c5 = _compute_c5(adjusted)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = adjusted.copy()
                stagnation = 0
            else:
                stagnation += 1
        h = adjusted.copy()
        iteration += 1
    return best_h, best_c5, n_points
