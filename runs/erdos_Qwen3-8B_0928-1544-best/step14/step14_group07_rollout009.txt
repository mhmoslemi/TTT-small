import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 50
    n_particles = 200
    ones_count = n_points / 2
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=np.float64)
    else:
        initial_h = np.random.rand(n_points)
        initial_h = initial_h * (ones_count / np.sum(initial_h))
    h_swarm = np.tile(initial_h[np.newaxis, :], (n_particles, 1))
    velocities = np.random.rand(n_particles, n_points) * 0.1
    pbest_h = h_swarm.copy()
    pbest_c5 = np.full(n_particles, np.inf)
    gbest_h = h_swarm[0].copy()
    gbest_c5 = np.inf
    w, c1, c2 = 0.7, 1.5, 1.5
    start_time = time.time()
    iterations = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s:
        for i in range(n_particles):
            h = h_swarm[i]
            corr = np.correlate(h, 1 - h, mode='full')
            max_overlap = np.max(corr) * (2.0 / n_points)
            if max_overlap < pbest_c5[i]:
                pbest_c5[i] = max_overlap
                pbest_h[i] = h.copy()
            if max_overlap < gbest_c5:
                gbest_c5 = max_overlap
                gbest_h = h.copy()
        
        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest_h[i] - h_swarm[i]) + c2 * r2 * (gbest_h - h_swarm[i])
            h_swarm[i] += velocities[i]
            h_swarm[i] = np.clip(h_swarm[i], 0.0, 1.0)
            sum_h = np.sum(h_swarm[i])
            if sum_h != 0:
                h_swarm[i] = h_swarm[i] * (ones_count / sum_h)
            else:
                h_swarm[i] = np.random.rand(n_points) * (ones_count / 0.5)
        
        iterations += 1
        if iterations > 100 and np.abs(gbest_c5 - np.min(pbest_c5)) < 1e-5:
            no_improvement += 1
            if no_improvement >= 50:
                break
        
        if gbest_c5 < 0.38085:
            break
    
    return gbest_h, gbest_c5, n_points
