import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points / 2.0
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        h = h * (ones_count / h_sum)
    best_h = h.copy()
    best_c5 = float('inf')
    alpha = 0.01
    start_time = time.time()
    improvement_threshold = 1e-6
    max_iterations = 1000
    iter_count = 0

    while time.time() - start_time < budget_s and iter_count < max_iterations:
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        max_overlap = np.max(corr) * dx
        if abs(max_overlap - best_c5) < improvement_threshold:
            break
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()

        k_max_lag = np.argmax(corr)
        lag = k_max_lag - (n_points - 1)
        shifted_forward = np.roll(h, lag)
        shifted_backward = np.roll(h, -lag)
        gradient = shifted_forward + shifted_backward
        h = h - alpha * gradient
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (ones_count / h_sum)
        else:
            h = np.random.rand(n_points) * (ones_count / 0.5)

        new_corr = np.correlate(h, 1 - h, mode='full')
        new_max_overlap = np.max(new_corr) * dx
        if new_max_overlap < best_c5:
            best_c5 = new_max_overlap
            best_h = h.copy()
            alpha *= 1.05
        else:
            alpha *= 0.5
        iter_count += 1
        if best_c5 < 0.38085:
            break

    return best_h, best_c5, n_points
