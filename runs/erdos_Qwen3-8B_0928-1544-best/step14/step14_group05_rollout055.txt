        res = opt.minimize(obj_func, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h_candidate = res.x
            current_c5 = np.max(np.correlate(h_candidate, 1 - h_candidate, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = h_candidate
                best_c5 = current_c5
    except:
        pass

    return best_h, best_c5, n_points
