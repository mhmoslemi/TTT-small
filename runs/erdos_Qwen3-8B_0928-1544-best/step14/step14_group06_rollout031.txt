import numpy as np
import time
import scipy.optimize as opt
import cma

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)

    def repair(h, n_points):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return np.full(n_points, (n_points / 2) / n_points)
        scale = (n_points / 2) / sum_h
        return h_clipped * scale

    def objective(x):
        h = repair(x, n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5

    # Initial repair
    h_initial = repair(initial_h, n_points)

    # CMA-ES setup
    es = cma.CMAEvolutionStrategy(h_initial, 0.5, {'popsize': 100})
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    max_iter = 200
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        es.optimize()
        current_h = es.result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5

    # Gradient refinement
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-6)
    final_h = repair(result.x, n_points)
    final_c5 = objective(final_h)

    return final_h, final_c5, n_points
