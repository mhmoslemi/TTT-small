import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points / 2
    if 'initial_h_values' in locals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * (ones_count / 0.5)
        else:
            h = h * (ones_count / h_sum)
    best_h = h.copy()
    best_c5 = float('inf')
    alpha = 0.1
    decay = 0.998
    momentum = 0.9
    velocity = np.zeros(n_points)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * (2.0 / n_points)
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        k_max_lag = np.argmax(corr)
        lag = k_max_lag - (n_points - 1)
        shifted = np.roll(h, lag)
        grad = shifted - np.roll(h, -lag)
        
        velocity = momentum * velocity + alpha * grad
        h += velocity
        h = np.clip(h, 0., 1.)
        h = h * (ones_count / np.sum(h))
        
        alpha *= decay
        if best_c5 < 0.38085:
            break
    
    return best_h, best_c5, n_points
