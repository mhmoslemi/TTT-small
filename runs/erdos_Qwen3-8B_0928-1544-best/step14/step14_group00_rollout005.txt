def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2
    pop_size = 50
    mut_rate = 0.3
    learn_rate = 0.1
    max_iter = int(budget_s / 2)

        pop.append(h)

        corr = np.correlate(h, 1 - h, mode='full')

    best_h = pop[0].copy()
    best_c5 = compute_c5(pop[0])

    start_time = time.time()

        if time.time() - start_time > budget_s:
            break

            k = np.argmax(corr)
