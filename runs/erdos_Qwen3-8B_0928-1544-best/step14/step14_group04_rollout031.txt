import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)

    h = np.clip(h, 0.0, 1.0)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    alpha = 0.1
    start_time = time.time()
    perturbation = 0.02  # Small noise scale for exploration
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        idx = np.argmax(corr)
        k = idx - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = (i + k) % n_points
            i_minus_k = (i - k) % n_points
            gradient[i] = h[i_plus_k] + h[i_minus_k]

        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)

        h = np.clip(h, 0.0, 1.0)
