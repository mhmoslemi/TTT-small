import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5

    def repair(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return np.zeros(n_points) + (n_points / 2) / n_points
        scale = (n_points / 2) / sum_h
        return h_clipped * scale

    def objective(h):
        h = repair(h)
        return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n_points)

    best_h = repair(initial_h)
    best_c5 = objective(best_h)
    start_time = time.time()

    for _ in range(5):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=50,
            tol=1e-5,
            mutation=(0.5, 1.0),
            recombination=0.8,
            polish=True,
            disp=False,
            seed=seed + _
        )
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
        if time.time() - start_time > budget_s:
            break

    best_h = repair(best_h)
    final_c5 = objective(best_h)
    return best_h, final_c5, n_points
