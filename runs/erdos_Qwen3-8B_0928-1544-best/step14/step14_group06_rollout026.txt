import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)
    h = h * (ones_count / np.sum(h)) if np.sum(h) != 0 else np.random.rand(n_points) * (ones_count / 0.5)
    
    def objective(x):
        x = np.clip(x, 0.0, 1.0)
        x_sum = np.sum(x)
        if x_sum == 0: return float('inf')
        x_scaled = x * (ones_count / x_sum)
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint(x):
        return np.sum(np.clip(x, 0.0, 1.0)) - ones_count
    
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    for _ in range(5):
        x0 = best_h.copy()
        res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5, options={'maxiter': 100})
        if res.success:
            current_h = res.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        if time.time() - start_time > budget_s:
            break
    
    best_h = np.clip(best_h, 0.0, 1.0)
    final_h = best_h * (ones_count / np.sum(best_h)) if np.sum(best_h) != 0 else h
    final_c5 = objective(final_h)
    
    return final_h, final_c5, n_points
