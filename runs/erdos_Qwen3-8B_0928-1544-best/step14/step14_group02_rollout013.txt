        initial_h = np.random.rand(n_points) * 0.5
    initial_h = initial_h * (ones_count / np.sum(initial_h))
    population_size = 20
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5
        population.append(h * (ones_count / h_sum))
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for _ in range(1000):
        if time.time() - start_time >= budget_s:
            break
        # Select the best individual
        current_best = max(population, key=compute_c5)
        current_c5 = compute_c5(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
        # Generate a new candidate
        idx = np.random.choice(len(population))
        parent = population[idx].copy()
        # Find the worst shift
        corr = np.correlate(parent, 1 - parent, mode='full')
        max_idx = np.argmax(corr)
        # Adjust parent to reduce this shift
        if max_idx < len(parent):
            i = max_idx
            j = (i + max_idx) % len(parent)
            if parent[i] > parent[j]:
                parent[i] -= 0.05
            else:
                parent[j] -= 0.05
        # Ensure values are in [0,1] and sum is correct
        parent = np.clip(parent, 0.0, 1.0)
        parent_sum = np.sum(parent)
        if parent_sum == 0:
            parent = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
        else:
            parent = parent * (ones_count / parent_sum)
        # Replace the worst candidate in population
        worst_idx = np.argmax([compute_c5(h) for h in population])
        population[worst_idx] = parent.copy()
    
    return best_h, best_c5, n_points
