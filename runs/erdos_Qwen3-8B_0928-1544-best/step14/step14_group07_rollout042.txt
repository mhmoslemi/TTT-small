import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = 35
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * (ones_count / 0.5)
        else:
            h = h * (ones_count / h_sum)
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()

    def objective(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / n_points)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [({'type': 'eq', 'fun': lambda h: np.sum(h) - ones_count})]

    try:
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-5, 'eps': 1e-5}
        )
        if result.success:
            best_c5 = result.fun
            best_h = result.x
    except:
        pass

    if best_c5 < 0.380876:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
