import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    n_population = 150
    F_start, F_end = 0.75, 0.95
    CR_start, CR_end = 0.9, 1.0
    mutation_rate = 0.3
    shrink_rate = 0.995

    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.6
    initial_h_sum = np.sum(initial_h)
    initial_h_scaled = initial_h * (ones_count / initial_h_sum) if initial_h_sum != 0 else np.random.rand(n_points) * 0.6 * (ones_count / 0.6)
    population = [initial_h_scaled]
    for _ in range(n_population - 1):
        h = np.random.rand(n_points) * 0.6
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.6
        population.append(h * (ones_count / h_sum))

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = objective(population[0])
    start_time = time.time()
    F = F_start
    CR = CR_start
    stagnation = 0

    population = [np.clip(h, 0.0, 1.0) for h in population]
    for i in range(len(population)):
        population[i] = population[i] * (ones_count / np.sum(population[i])) if np.sum(population[i]) != 0 else np.random.rand(n_points) * 0.6 * (ones_count / 0.6)

    while time.time() - start_time < budget_s:
        for _ in range(n_population):
            a, b, c = np.random.choice(n_population, 3, replace=False)
            a_vec, b_vec, c_vec = population[a], population[b], population[c]
            v = a_vec + F * (b_vec - c_vec)
            u = np.copy(population[_])
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = v[j]
            u = np.clip(u, 0.0, 1.0)
            u_sum = np.sum(u)
            if u_sum == 0:
                continue
            u_scaled = u * (ones_count / u_sum)
            new_c5 = objective(u_scaled)
            if new_c5 < objective(population[_]):
                population[_] = u_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = u_scaled.copy()
                    stagnation = 0
            else:
                stagnation += 1
            if stagnation > 50:
