import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70  # Optimal for efficiency and precision
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    initial_h_sum = np.sum(initial_h)
    initial_h_scaled = initial_h * (ones_count / initial_h_sum) if initial_h_sum != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
    
    def objective(h):
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            return float('inf')
        h_scaled = h * (ones_count / h_sum)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = initial_h_scaled.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Multi-restart differential evolution
    for restart in range(5):
        result = opt.differential_evolution(
            objective,
            [(0.0, 1.0)] * n_points,
            strategy='best1bin',
            maxiter=200,
            popsize=150,
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.9,
            polish=True,
            disp=False,
            seed=seed + restart
        )
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
        if time.time() - start_time > budget_s:
            break
    
    # Final safeguard and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h_sum = np.sum(best_h)
    best_h_scaled = best_h * (ones_count / best_h_sum) if best_h_sum != 0 else initial_h_scaled
