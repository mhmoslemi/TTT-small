    while time.time() - start_time < budget_s and iteration < max_iterations:
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(cross_corr)
        max_index = np.argmax(cross_corr)
        shift_samples = max_index - (n_points - 1)

        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + shift_samples
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx

        h = h - eta * grad
        h = np.clip(h, 0.0, 1.0)

        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * (ones_count / 0.5)
        else:
            scaling = ones_count / h_sum
            h = h * scaling

        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            noise = np.random.normal(0, perturb_std, n_points)
            h = h + noise
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            if h_sum == 0:
                h = np.random.rand(n_points) * (ones_count / 0.5)
            else:
                scaling = ones_count / h_sum
                h = h * scaling
            patience = 0

        eta *= decay_rate
        iteration += 1

    return best_h, best_c5, n_points
