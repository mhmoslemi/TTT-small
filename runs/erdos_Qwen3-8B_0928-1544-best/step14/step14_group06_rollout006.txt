import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5

    # Scale initial solution to satisfy the integral constraint
    h_sum = np.sum(h)
    h = h * (ones_count / h_sum) if h_sum != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)

    def repair(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clipped)
        scale = (ones_count) / sum_h if sum_h != 0 else 1.0
        return h_clipped * scale

    def objective(h):
        h = repair(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Adaptive differential evolution with exploration phase
    for _ in range(4):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=150,
            popsize=50,
            tol=1e-6,
            mutation=(0.6, 1.0),
            recombination=0.9,
            polish=True,
            disp=False,
            seed=seed + _
        )
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
        if time.time() - start_time > budget_s:
            break

    # Gradient refinement with repair
    def grad_objective(h):
        h = repair(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    result = opt.minimize(
        grad_objective,
        best_h,
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        method='L-BFGS-B',
        jac=None,
        options={'maxiter': 100, 'ftol': 1e-8}
    )

    best_h = repair(result.x)
    final_c5 = objective(best_h)
    return best_h, final_c5, n_points
