import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    pop_size = 100
    max_gen = 400
    elite_size = 10
    mutation_rate = 0.15
    crossover_rate = 0.8
    decay_rate = 0.995
    local_iter = 3

    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.6
    def scale_h(h):
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        if total == 0:
            return np.random.rand(n_points) * 0.6 * (ones_count / 0.6)
        return h * (ones_count / total)

    population = [scale_h(initial_h) if i == 0 else scale_h(np.random.rand(n_points) * 0.6) for i in range(pop_size)]

    def calc_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = population[0].copy()
    best_c5 = calc_c5(best_h)
    time_start = time.time()

    for gen in range(max_gen):
        if time.time() - time_start > budget_s:
            break
        if gen < 100:
            mutation_rate = max(0.1, mutation_rate)
        else:
            mutation_rate *= decay_rate

        elites = sorted(population, key=calc_c5)[:elite_size]
        new_pop = elites.copy()

        for _ in range(pop_size - elite_size):
            a, b, c = np.random.choice(len(population), 3, replace=False)
            parent1, parent2, parent3 = population[a], population[b], population[c]
            if np.random.rand() < crossover_rate:
                idx = np.random.randint(1, n_points-1)
                child = np.concatenate([parent1[:idx], parent2[idx:]])
            else:
                child = np.random.rand(n_points) * 0.6
            child = scale_h(child)

            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[j] += np.random.randn() * 0.2
                    child[j] = np.clip(child[j], 0.0, 1.0)

            child = scale_h(child)
            new_pop.append(child)

        population = new_pop
        current_c5 = calc_c5(population[0])
        if current_c5 < best_c5:
            best_h = population[0].copy()
            best_c5 = current_c5

        for idx, h in enumerate(population):
            if calc_c5(h) < calc_c5(population[0]):
                population[0] = h.copy()

    if time.time() - time_start < budget_s:
        try:
            for _ in range(local_iter):
                h = np.copy(best_h)
                cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ones_count}]
                bounds = [(0.0, 1.0) for _ in range(n_points)]
                res = opt.minimize(lambda x: calc_c5(x), h, bounds=bounds, constraints=cons, method='L-BFGS-B')
                if res.success:
                    best_h = res.x
                    best_c5 = calc_c5(best_h)
        except:
            pass

    return best_h, best_c5, n_points
