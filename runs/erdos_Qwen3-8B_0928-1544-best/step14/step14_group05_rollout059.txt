import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 0.5
    h_sum = np.sum(h)
    if h_sum == 0:
        h = np.random.rand(n_points) * 0.5
    h = h * (ones_count / h_sum)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    learning_rate = 0.1
    max_iterations = 1500
    perturbation_freq = 200
    perturbation_size = 0.01
    convergence_threshold = 1e-6
    start_time = time.time()

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag_k = max_index - (n_points - 1)

        grad = np.zeros(n_points)
        for i in range(n_points):
            x_plus_k = i + lag_k
            if x_plus_k < n_points:
                grad[i] += (1 - h[x_plus_k]) * dx
            x_minus_k = i - lag_k
            if x_minus_k >= 0:
                grad[i] -= h[x_minus_k] * dx

        h = h - learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (ones_count / h_sum)
        else:
            h = np.random.rand(n_points) * 0.5
            h = h * (ones_count / np.sum(h))

        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        else:
            if iteration % perturbation_freq == 0:
                noise = np.random.normal(0, perturbation_size, n_points)
                h = np.clip(h + noise, 0.0, 1.0)
                h_sum = np.sum(h)
                if h_sum != 0:
                    h = h * (ones_count / h_sum)
                else:
                    h = np.random.rand(n_points) * 0.5
                    h = h * (ones_count / np.sum(h))

        if iteration % 100 == 0:
            learning_rate *= 0.95

        if abs(current_c5 - best_c5) < convergence_threshold:
            break

    return best_h, best_c5, n_points
