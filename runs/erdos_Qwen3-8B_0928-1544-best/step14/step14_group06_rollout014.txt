import numpy as np
import time
import cma
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    ones_count = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h_sum = np.sum(initial_h)
        if initial_h_sum != 0:
            initial_h = initial_h * (ones_count / initial_h_sum)
        else:
            initial_h = np.zeros(n_points) + (ones_count / n_points)

    def repair(x):
        h_clipped = np.clip(x, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return np.zeros_like(x) + (ones_count / n_points)
        scale = ones_count / sum_h
        return h_clipped * scale

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    initial_h_repaired = repair(initial_h)
    best_solution = initial_h_repaired.copy()
    best_c5 = compute_c5(best_solution)

    start_time = time.time()
    num_runs = 3
    for _ in range(num_runs):
        if time.time() - start_time > budget_s:
            break
        es = cma.CMAEvolutionStrategy(initial_h_repaired, 0.5, {'popsize': 50, 'bounds': [(0.0, 1.0)] * n_points})
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            solutions = es.ask()
            for x in solutions:
                h_repaired = repair(x)
                c5_val = compute_c5(h_repaired)
                es.tell(x, [c5_val])
            if es.stop():
                break
        current_best = es.best_x
        current_c5 = compute_c5(current_best)
        if current_c5 < best_c5:
            best_solution = current_best.copy()
            best_c5 = current_c5

    def objective_grad(x):
        h_repaired = repair(x)
        return compute_c5(h_repaired)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(objective_grad, best_solution, method='L-BFGS-B', bounds=bounds, options={'ftol': 1e-8, 'maxiter': 100})

    final_h = repair(result.x)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, n_points
