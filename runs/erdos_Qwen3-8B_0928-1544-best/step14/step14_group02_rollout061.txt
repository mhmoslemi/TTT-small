    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    initial_h = initial_h * (ones_count / np.sum(initial_h)) if np.sum(initial_h) != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h, n_points)

    start_time = time.time()
    for _ in range(10000):
        if time.time() - start_time >= budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = best_h.copy()
        max_corr_idx = np.argmax(corr)
        # Adjust the positions involved in the max shift
        shift = max_corr_idx
        if shift >= n_points:
            shift -= n_points
        if shift < 0:
            shift += n_points
        # Perturb positions in the max shift
        for i in range(n_points):
            if i + shift < n_points:
                best_h[i] -= np.random.uniform(0.001, 0.01)
                best_h[i] = np.clip(best_h[i], 0.0, 1.0)
            if i - shift >= 0:
                best_h[i] -= np.random.uniform(0.001, 0.01)
                best_h[i] = np.clip(best_h[i], 0.0, 1.0)
        # Maintain sum constraint
        sum_h = np.sum(best_h)
        if sum_h != 0:
            best_h = best_h * (ones_count / sum_h)
    return best_h, best_c5, n_points

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n_points)
