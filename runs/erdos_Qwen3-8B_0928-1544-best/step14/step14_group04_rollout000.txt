        grad = compute_gradient(h, contrib_shifts)
        h += alpha * grad
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * 0.5 * (target_sum / 0.5)
        else:
            h = h * (target_sum / h_sum)
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_max = np.max(new_corr) * (2.0 / n_points)
        if new_max < best_c5:
            best_c5 = new_max
            best_h = h.copy()
            alpha = 0.1
        else:
            alpha *= 0.95
        if alpha < 1e-6:
            break
    return best_h, best_c5, n_points
