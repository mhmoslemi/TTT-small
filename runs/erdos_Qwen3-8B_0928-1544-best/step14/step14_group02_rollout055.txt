def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
    h_sum = np.sum(h)
    h = h * (ones_count / h_sum) if h_sum != 0 else np.random.rand(n_points) * (ones_count / 0.5)
    best_h = h.copy()
    best_c5 = 0.0
    
    def compute_c5(h, n_points):
        max_overlap = 0.0
        best_k = 0
        for k in range(n_points):
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlap *= (2.0 / n_points)
            if overlap > max_overlap:
                max_overlap = overlap
                best_k = k
        return max_overlap, best_k
    
    current_c5, best_k = compute_c5(h, n_points)
    best_c5 = current_c5
    start_time = time.time()
    iterations = 0
    
    while time.time() - start_time < budget_s:
        current_c5, best_k = compute_c5(h, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        contributions = []
        for i in range(n_points):
            j = (i + best_k) % n_points
            contributions.append(h[i] * (1 - h[j]))
        top_indices = np.argsort(contributions)[::-1][:int(n_points * 0.1)]
        
        for i in top_indices:
            j = (i + best_k) % n_points
            if h[i] > 0.5:
                h[i] -= 0.1
            else:
                h[j] += 0.1
        
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.random.rand(n_points) * (ones_count / 0.5)
        else:
            h = h * (ones_count / h_sum)
        h = np.clip(h, 0.0, 1.0)
        
        if iterations % 50 == 0:
            perturbation = np.random.rand(n_points) * 0.15
            h += perturbation
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            if h_sum != 0:
                h = h * (ones_count / h_sum)
        
        iterations += 1
    
    return best_h, best_c5, n_points
