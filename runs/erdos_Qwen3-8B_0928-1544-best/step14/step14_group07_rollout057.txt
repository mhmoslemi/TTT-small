import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 50
    N = n_points
    ones_count = N / 2
    swarm_size = 200

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=np.float64)
        particles = np.tile(initial_h, (swarm_size, 1)) + np.random.rand(swarm_size, N) * 0.1
    else:
        particles = np.random.rand(swarm_size, N) * (ones_count / np.sum(particles, axis=1, keepdims=True))

    velocities = np.random.rand(swarm_size, N) * 0.1
    pbest_positions = particles.copy()
    pbest_values = [compute_c5(p) for p in particles]
    gbest_pos, gbest_val = particles[0].copy(), pbest_values[0]
    best_c5, best_h = gbest_val, gbest_pos

    w = 0.7
    c1, c2 = 1.5, 1.5
    start_time = time.time()
    no_improvement = 0

    while time.time() - start_time < budget_s:
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest_positions[i] - particles[i]) + c2 * r2 * (gbest_pos - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_p = np.sum(particles[i])
            if sum_p == 0:
                particles[i] = np.random.rand(N) * (ones_count / 0.5)
            else:
                particles[i] *= ones_count / sum_p

        current_values = [compute_c5(p) for p in particles]
        for i in range(swarm_size):
            if current_values[i] < pbest_values[i]:
                pbest_positions[i], pbest_values[i] = particles[i].copy(), current_values[i]

        gbest_idx = np.argmin(current_values)
        if current_values[gbest_idx] < gbest_val:
            gbest_pos, gbest_val = particles[gbest_idx].copy(), current_values[gbest_idx]
            if gbest_val < best_c5:
                best_c5, best_h = gbest_val, gbest_pos

        if best_c5 < 0.38085:
            break
        if np.isclose(gbest_val, best_c5) and no_improvement >= 100:
            break
        no_improvement = 0

    return best_h, best_c5, n_points
