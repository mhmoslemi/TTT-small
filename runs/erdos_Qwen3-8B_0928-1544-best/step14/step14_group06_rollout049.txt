import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5

    def scale_h(h):
        h_sum = np.sum(h)
        if h_sum == 0:
            return np.random.rand(n_points) * 0.5
        return h * (ones_count / h_sum)

    initial_h = scale_h(initial_h)

    def objective(h):
        h_clip = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h_clip)
        if h_sum == 0:
            return float('inf')
        h_scaled = h_clip * (ones_count / h_sum)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for _ in range(5):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=150,
            popsize=50,
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.8,
            polish=True,
            disp=False,
            seed=seed + _
        )
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
        if time.time() - start_time > budget_s:
            break

    result = opt.minimize(
        objective,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        jac=None,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    best_h = result.x
    final_h = scale_h(best_h)
    final_c5 = objective(final_h)

    return final_h, final_c5, n_points
