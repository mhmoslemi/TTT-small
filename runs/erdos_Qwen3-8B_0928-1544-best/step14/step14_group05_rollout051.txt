    required_sum = n_points / 2.0

    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 0.5
    h_sum = np.sum(h)
    h = h * (required_sum / h_sum) if h_sum != 0 else np.random.rand(n_points) * 0.5 * (required_sum / 0.5)

    # Parameters
    learning_rate = 0.1
    decay_factor = 0.995
    perturbation_rate = 0.05
    stagnation_threshold = 100
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx

        # Find optimal lag
        max_index = np.argmax(corr)
        k_opt = max_index - (n_points - 1)

        # Compute gradient for this lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_opt
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) - h[i]

        # Update h
        h -= learning_rate * grad

        # Clip to [0, 1]
        h = np.clip(h, 0.0, 1.0)

        # Scale to required sum
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (required_sum / h_sum)
        else:
            # Avoid division by zero
            h = np.random.rand(n_points) * 0.5 * (required_sum / 0.5)

        # Check if current C5 is better
        corr_current = np.correlate(h, 1 - h, mode='full')
        current_c5_current = np.max(corr_current) * dx
        if current_c5_current < best_c5:
            best_c5 = current_c5_current
            best_h = h.copy()
        else:
            # Stagnation: reduce learning rate
            learning_rate *= decay_factor

        # Optional perturbation
        if np.random.rand() < perturbation_rate:

        # Check for early termination
        if best_c5 < 0.38085:
            break

    return best_h, best_c5, n_points
