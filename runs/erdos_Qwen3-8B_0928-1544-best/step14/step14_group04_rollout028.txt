def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        h = np.random.rand(n_points) * 0.5
    h = h * (ones_count / np.sum(h)) if np.sum(h) != 0 else np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
    h = np.clip(h, 0.0, 1.0)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    alpha = 0.1
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        idx = np.argmax(corr)
        k = idx - (n_points - 1)
        
        idxs = (np.arange(n_points) + k) % n_points
        gradient = (1 - h[idxs]) * dx
        
        h -= alpha * gradient
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
        else:
            h = h * (ones_count / sum_h)
        
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_max = np.max(new_corr) * dx
        
        if new_max < best_c5:
            best_c5 = new_max
            best_h = h.copy()
            alpha = 0.1
        else:
            alpha *= 0.95
        
        if alpha < 1e-6:
            break
    
    return best_h, best_c5, n_points
