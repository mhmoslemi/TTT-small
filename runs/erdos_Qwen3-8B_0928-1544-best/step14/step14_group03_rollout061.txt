import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    target_sum = n_points / 2
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 0.5
        h = h * (target_sum / np.sum(h))
    
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    eps = 1e-5

    def normalize(h):
        s = np.sum(h)
        if s == 0:
            return h
        return h * (target_sum / s)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        return max_val * dx

    def gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = normalize(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')

            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = normalize(h_minus)

            val_minus = corr_minus[k_max] * dx
            grad[i] = (val_plus - val_minus) / (2 * eps)
        return grad

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=gradient, options={'ftol': 1e-8, 'maxiter': 1000})

    best_c5 = objective(best_h)
