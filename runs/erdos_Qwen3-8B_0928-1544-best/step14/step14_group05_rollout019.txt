def run(seed=42, budget_s=1100, **kwargs):

        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:

            best_h = h.copy()
        m = np.argmax(corr)
        kmax = m - (n_points - 1)
        grad = np.zeros_like(h)
        for x in range(n_points):
            y1 = x + kmax
            term1 = 0.0
