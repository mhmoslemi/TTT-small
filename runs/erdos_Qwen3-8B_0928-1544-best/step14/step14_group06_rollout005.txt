import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        h = np.random.rand(n_points)
    
    # Scale initial solution to meet the integral constraint
    h_sum = np.sum(h)
    if h_sum == 0:
        h = np.random.rand(n_points) * 0.5 * (ones_count / 0.5)
    else:
        h = h * (ones_count / h_sum)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # Differential Evolution for global search
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_de = opt.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=50,
        tol=1e-5,
        mutation=(0.5, 1.0),
        recombination=0.8,
        polish=True,
        disp=False,
        seed=seed
    )
    best_h = result_de.x
    
    # SLSQP refinement for local optimization
    def constraint_sum(h_flat):
        return np.sum(h_flat) - ones_count
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    result_slsqp = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        constraints=cons,
        jac=None,
        tol=1e-6,
        options={'maxiter': 100}
    )
    best_h = result_slsqp.x
    
    # Final validation and scaling
    best_h = np.clip(best_h, 0.0, 1.0)
    h_sum = np.sum(best_h)
    if h_sum != 0:
        best_h = best_h * (ones_count / h_sum)
    else:
        best_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points) * 0.5
    
    return best_h, objective(best_h), n_points
