        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            alpha *= learning_rate_decay

        d_current = np.sum(h) / n_points
        if current_c5 > best_c5:
            d_current += alpha * (current_c5 - best_c5)
        else:
            d_current -= alpha * (best_c5 - current_c5)
        d_current = np.clip(d_current, 0.2, 0.8)

        # Adjust h to match new duty cycle
        h = h * (n_points / 2) / np.sum(h)
        h = h * d_current * n_points / (n_points / 2)
        h = np.clip(h, 0, 1)

    return best_h, best_c5, n_points
