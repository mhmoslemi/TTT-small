def run(seed=42, budget_s=1100, **kwargs):

    h = np.clip(h, 0.0, 1.0)

    def update_h(h, k):
        grad = np.zeros(n_points)
        for i in range(n_points):

            i_minus_k = (i - k) % n_points
            grad[i] += h[i_plus_k] + h[i_minus_k]
        return grad

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        idx = np.argmax(corr)
        k = idx - (n_points - 1)
        max_k_indices = np.where(corr == max_corr)[0]

        gradients = np.zeros(n_points)
        for k_val in max_k:
            grad = update_h(h, k_val)
            gradients += grad
        grad_norm = np.linalg.norm(gradients)
        if grad_norm > 1e-8:

        h = np.clip(h, 0.0, 1.0)
