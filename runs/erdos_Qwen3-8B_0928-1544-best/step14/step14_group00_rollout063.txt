def run(seed=42, budget_s=1100, **kwargs):

    np.random.seed(seed)
    n_points = 70

    # Initial guess
    if 'initial_h_values' in globals():

    else:
        initial_h = np.random.rand(n_points) * 0.5
        initial_h_sum = np.sum(initial_h)
        if initial_h_sum == 0:
            initial_h = np.random.rand(n_points) * 0.5
        initial_h = initial_h * (ones_count / initial_h_sum)
    h = initial_h.copy()
    # Compute initial C5
    corr = np.correlate(h, 1 - h, mode='full')
    valid_corr = corr[n_points-1 : 2*n_points -1]
    initial_c5 = np.max(valid_corr) * (2.0 / n_points)
    best_c5 = initial_c5
    best_h = h.copy()
    # Learning rate
    learning_rate = 0.1
    # Stagnation counter
    stagnation = 0
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        valid_corr = corr[n_points-1 : 2*n_points -1]
        k_max = np.argmax(valid_corr)
        # Compute gradient for shift k_max
        grad = np.zeros(n_points)
        k = k_max
        for j in range(n_points):
            if j + k < n_points:
                grad[j] += (1 - h[j + k])
            if j >= k:
                grad[j] += -h[j - k]
        # Update h
        h_new = h - learning_rate * grad
        # Clip to [0, 1]
        h_clipped = np.clip(h_new, 0.0, 1.0)
        # Normalize to sum 35
        h_sum = np.sum(h_clipped)
        if h_sum > 0:
            h = h_clipped * (ones_count / h_sum)
        else:
            # Random restart
            h = np.random.rand(n_points) * 0.5
            h_sum = np.sum(h)
            h = h * (ones_count / h_sum)
        # Compute new C5
        corr_new = np.correlate(h, 1 - h, mode='full')
        valid_corr_new = corr_new[n_points-1 : 2*n_points -1]
        current_c5 = np.max(valid_corr_new) * (2.0 / n_points)
        # Update best

            best_h = h.copy()
            stagnation = 0

        # Adjust learning rate based on stagnation

            learning_rate *= 0.5
            stagnation = 0
        # Check if time is up
        if time.time() - start_time >= budget_s:

    return best_h, best_c5, n_points
