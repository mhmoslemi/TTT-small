import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    start_time = time.time()
    n = 70
    dx = 2.0 / n
    initial_guess = initial_h_values if 'initial_h_values' in globals() else np.full(n, 0.5)
    best_h = initial_guess.copy()
    best_c5 = 1.0
    
    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Optimize with multiple restarts and better tolerance
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        # Use a more aggressive optimization setup
        result = minimize(
            fun=compute_c5,
            x0=initial_guess,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n)],
            constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - n/2},
            tol=1e-8,
            options={'maxiter': 200, 'ftol': 1e-9, 'gtol': 1e-7}
        )
        if result.success:
            current_h = result.x
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
            # Use result.x as new initial guess for next iteration
            initial_guess = current_h
        else:
            # Fallback to previous best
            initial_guess = best_h
    
    return best_h, best_c5, n
