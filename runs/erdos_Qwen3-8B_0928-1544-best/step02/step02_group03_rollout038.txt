            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            tol=1e-8,
            options={'maxiter': maxiter_per_step, 'ftol': 1e-9}
        )
        
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
                elapsed_time = time.time() - start_time
                if elapsed_time >= budget_s:
                    break
    
    return best_h, best_c5, n_points
