    h_initial = np.array(initial_h_values)

    target_sum = n_points / 2

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_C5_with_penalty(h):
        c5 = compute_C5(h)
        sum_h = np.sum(h)
        penalty = 1000.0 * (sum_h - target_sum) ** 2
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    elapsed_time = 0.0
    start_time = time.time()

    result = differential_evolution(

        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )

    best_h = result.x
    best_c5 = compute_C5(best_h)

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for restart in range(5):

            random_h = np.clip(random_h, 0.0, 1.0)

            result = differential_evolution(
