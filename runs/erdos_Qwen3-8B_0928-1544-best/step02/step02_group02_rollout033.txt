from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    
    # Check if initial_h_values is available
    def get_initial_guess(n):
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            return initial_h_values.copy()
        return np.full(n, 0.5)
    
    n = 70
    initial_guess = get_initial_guess(n)
    dx = 2.0 / n
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    result = minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 2000, 'ftol': 1e-10, 'gtol': 1e-10}
    )
    
    if result.success:
        c5_bound = result.fun
        h_values = result.x
    else:
        c5_bound = objective(initial_guess)
        h_values = initial_guess
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        return h_values, c5_bound, n
    
    return h_values, c5_bound, n
