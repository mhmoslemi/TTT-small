import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points
    particles = np.random.normal(0, 0.1, (30, n_points))
    particles = np.clip(h_initial + particles, 0, 1)
    particles = particles * (n_points / 2) / np.sum(particles, axis=1, keepdims=True)
    velocities = np.zeros((30, n_points))
    pbest = particles.copy()
    gbest = particles[0].copy()
    best_c5 = np.inf
    best_h = gbest.copy()
    c1, c2, w = 2.0, 2.0, 0.7
    start_time = time.time()
    while time.time() - start_time < budget_s:
        for i in range(30):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            curr_c5 = np.max(corr) * dx
            pcorr = np.correlate(pbest[i], 1 - pbest[i], mode='full')
            pcurr_c5 = np.max(pcorr) * dx
            if curr_c5 < pcurr_c5:
                pbest[i] = particles[i].copy()
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    best_h = particles[i].copy()
                    gbest = best_h.copy()
        if np.random.rand() < 0.1:
            c1 += 0.1
            c2 -= 0.1
            if c1 > 3: c1, c2 = 3, 0.5
            elif c2 < 0.5: c1, c2 = 0.5, 3
    return best_h, best_c5, n_points
