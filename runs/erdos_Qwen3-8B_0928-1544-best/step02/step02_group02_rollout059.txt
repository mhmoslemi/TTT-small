import numpy as np
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    
    n = 70
    dx = 2.0 / n
    best_h = np.full(n, 0.5)
    best_c5 = float('inf')
    
    if 'initial_h_values' in globals():
        initial_guess = initial_h_values.copy()
    else:
        initial_guess = np.full(n, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    # Multi-start optimization with perturbations
    num_perturbations = 5
    for _ in range(num_perturbations):
        if 'initial_h_values' in globals():
            p = initial_guess + np.random.normal(0, 0.1, n)
        else:
            p = np.full(n, 0.5) + np.random.normal(0, 0.1, n)
        p = np.clip(p, 0, 1)
        
        result = minimize(
            fun=objective,
            x0=p,
            method='L-BFGS-B',

            tol=1e-8,

        )
        
        if result.success:
            curr_c5 = result.fun
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = result.x
        else:
            curr_c5 = objective(p)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
