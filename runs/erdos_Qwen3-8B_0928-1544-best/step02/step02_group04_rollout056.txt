import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        n_initial = len(initial_h)
        new_n_points = 256
        x_initial = np.linspace(0, 2, n_initial)
        f = np.interp(np.linspace(0, 2, new_n_points), x_initial, initial_h)
        f = np.clip(f, 0, 1)
        current_sum = np.sum(f)
        scale_factor = (new_n_points / 2) / current_sum
        initial_h = f * scale_factor
    else:
        initial_h = np.ones(256) * (256 / 2) / 256
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    def enforce_constraints(h, target):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return h * (target / s)

    def generate_initial():
        pop = [initial_h.copy()]
        for _ in range(49):

                h = np.ones(n_points) * (n_points / 2) / n_points

                h = np.clip(h, 0, 1)

        return pop

    population = generate_initial()
