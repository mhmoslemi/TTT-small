        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        current_h *= scale
        new_c5 = compute_C5(current_h)
        if new_c5 < best_c5 - 1e-6:
            best_h = current_h.copy()
            best_c5 = new_c5
            learning_rate *= 1.1
        else:
            learning_rate *= 0.9
        if iter_count % perturb_freq == 0:
            noise = np.random.normal(0, perturb_std, n_points)
            current_h += noise
            current_h = np.clip(current_h, 0, 1)
            sum_h = np.sum(current_h)
            scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
            current_h *= scale

    return best_h, best_c5, n_points
