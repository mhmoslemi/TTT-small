        sum_clipped = np.sum(clipped)

            scaled = np.zeros(n_points)
        else:
            scaled = clipped * (n_points / 2) / sum_clipped
        initial_population.append(scaled)
    current_population = [(h, compute_C5(h)) for h in initial_population]

        if c5 < best_c5:
