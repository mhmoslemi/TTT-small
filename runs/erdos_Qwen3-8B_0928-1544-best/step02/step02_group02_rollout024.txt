import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    n = 70
    dx = 2.0 / n
    h = initial_h_values.copy()
    
    def objective(h_flat):
        h = h_flat.reshape(n)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n / 2
    
    bounds = Bounds(0.0, 1.0)
    
    result = minimize(
        fun=objective,
        x0=h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 10000}
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    if time.time() - start_time < budget_s - 500:
        perturb = np.random.normal(0, 0.1, n)
        perturb = np.clip(perturb, -0.3, 0.3)
        result = minimize(
            fun=objective,
            x0=(h + perturb),
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 5000}
        )
        best_h = result.x
        best_c5 = min(best_c5, result.fun)
    
    if time.time() - start_time < budget_s - 200:
        perturb = np.random.normal(0, 0.05, n)
        perturb = np.clip(perturb, -0.2, 0.2)

            fun=objective,

            options={'maxiter': 3000}
        )
        best_h = result.x
        best_c5 = min(best_c5, result.fun)
    
    return best_h, best_c5, n
