        delta_sum = (n_points / 2) - np.sum(candidate)
        candidate[-1] += delta_sum
        candidate = np.clip(candidate, 0, 1)
        new_c = compute_c5(candidate)
        if new_c < best_c5 or np.random.rand() < np.exp((best_c5 - new_c) / T):
            current_h = candidate.copy()
            best_c5 = new_c
        T *= alpha
        iteration += 1

    h_values = best_h
    if np.abs(np.sum(h_values) - n_points / 2) > 1e-6:
        h_values = initial_h
        best_c5 = compute_c5(h_values)
    return h_values, best_c5, n_points
