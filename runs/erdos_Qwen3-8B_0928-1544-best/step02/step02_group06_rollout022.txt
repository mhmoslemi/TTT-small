    target_sum = n_points / 2

                h = np.zeros(n_points)

                h[start2:start2+64] = 1.0
                guesses.append(h)
        guesses.append(np.full(n_points, target_sum / n_points))
        return guesses
    
    initial_guesses = gen_initial_guesses()

    num_fireflies = 50

        fireflies.append(h)

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:

        obj_vals = [np.max(np.correlate(h, 1 - h, mode='full') * dx) for h in fireflies]
        sorted_idx = np.argsort(obj_vals)
        sorted_h = [fireflies[i] for i in sorted_idx]
