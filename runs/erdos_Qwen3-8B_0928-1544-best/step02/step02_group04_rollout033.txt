import numpy as np
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 512
    dx = 2.0 / n_points

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    def clip_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        s = np.sum(h_clipped)
        if s == 0:
            return np.zeros_like(h)
        return h_clipped * (n_points / 2) / s

    if 'initial_h_values' in globals():
        x_initial = np.linspace(0, 2, len(initial_h_values))
        x_new = np.linspace(0, 2, n_points)
        f = np.interp(x_new, x_initial, initial_h_values)
        f *= (n_points / 2) / np.sum(f)
        initial_h = f
    else:
        initial_h = np.full(n_points, 0.5)

    population_size = 50
    population = [np.copy(initial_h) * 0.9 + np.random.rand(n_points) * 0.2 for _ in range(population_size)]

    def gen_triangle():
        x = np.linspace(0, 2, n_points)
        return np.minimum(x / 2, (2 - x) / 2)

    def gen_sine():
        x = np.linspace(0, 2, n_points)
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * x), 0, 1)

    for h in [gen_triangle(), gen_sine()]:
        population.append(h)

    population = [np.clip(np.random.rand(n_points) * (n_points / 2) / np.random.rand(), 0, 1) for _ in range(population_size - len(population))]

    best_h = np.copy(initial_h)
    best_c5 = evaluate(best_h)

    start = time.time()
    while time.time() - start < budget_s:
        fit = np.array([evaluate(h) for h in population])
        idx = np.argsort(fit)
        top = population[idx[:population_size // 2]]
        new_pop = []
        for i in range(0, len(top), 2):
            a, b = top[i], top[i+1]
            split = np.random.randint(1, n_points - 1)
            c1 = np.concatenate([a[:split], b[split:]])
            c2 = np.concatenate([b[:split], a[split:]])
            for c in [c1, c2]:
                for j in range(n_points):
                    if np.random.rand() < 0.05:
                        c[j] = np.random.rand()
                c = clip_rescale(c)
                new_pop.append(c)
        population = new_pop
        for h in population:
            if h is None:
                continue
            c = evaluate(h)
            if c < best_c5:
                best_c5 = c
                best_h = np.copy(h)

    best_h = clip_rescale(best_h)
    return best_h, best_c5, n_points
