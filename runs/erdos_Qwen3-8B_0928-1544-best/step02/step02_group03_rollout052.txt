import numpy as np
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    population_size = 50
    best_c5 = float('inf')
    best_h = None

    # Initialize population
    population = []
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        population.append(initial_h_values.copy())
    for _ in range(population_size - 1):
        h = np.random.rand(70)
        h = h * 35 / h.sum()
        h = np.clip(h, 0, 1)
        h = h * 35 / h.sum()
        population.append(h)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / 70)

    # Evaluate initial population
    for h in population:
        current_c5 = evaluate(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Tournament selection
        tournament = random.sample(population, 5)
        parent1 = min(tournament, key=evaluate)
        tournament2 = random.sample(population, 5)
        parent2 = min(tournament2, key=evaluate)
        # Crossover
        crossover = random.randint(1, 68)
        child1 = np.concatenate((parent1[:crossover], parent2[crossover:]))
        child2 = np.concatenate((parent2[:crossover], parent1[crossover:]))
        # Mutate
        def mutate(child):
            for i in range(len(child)):
                if np.random.rand() < 0.1:
                    child[i] = np.random.rand()
            child = np.clip(child, 0, 1)
            s = child.sum()
            if s != 0:
                child *= 35 / s
            return child
        child1 = mutate(child1)
        child2 = mutate(child2)
        # Replace worst
        evaluations = [evaluate(h) for h in population]
        worst_indices = np.argpartition(evaluations, -2)[-2:]
        population[worst_indices[0]] = child1
        population[worst_indices[1]] = child2
        # Update best
        for h in population:
            current_c5 = evaluate(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
