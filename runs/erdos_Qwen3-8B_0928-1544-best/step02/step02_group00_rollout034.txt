from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):

    best_c5 = np.inf

    def objective(h_flat):
        h = h_flat.reshape(n_points)

        max_overlap = np.max(corr) * dx
        return max_overlap

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    for _ in range(200):
        # Evaluate max overlap
        h_flat = best_h.copy()
        corr = correlate(h_flat, 1 - h_flat, mode='full')
        max_overlap = np.max(corr) * dx
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h_flat

        # Early exit if budget is nearing
        if _ > 10 and best_c5 < 0.38085:
            break
        
        # Perturb h for diversity
        perturb = np.random.normal(0, 0.05, n_points)
        h_flat = np.clip(best_h + perturb, 0, 1)
        h_flat = h_flat * (n_points / 2) / np.sum(h_flat)
        
        # Optimize
        res = minimize(

            h_flat,
            method='SLSQP',
