import numpy as np
from scipy.optimize import basinhopping, minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.full(70, 0.5)
    n_points = 70
    dx = 2.0 / n_points

    def objective_basin(h):
