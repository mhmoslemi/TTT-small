import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points_initial = len(initial_h_values)
    initial_h = initial_h_values.copy()
    np.random.seed(seed)
    
    # Define objective once
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr)
    
    # Generate perturbations
    perturbations = [initial_h.copy()]
    for _ in range(4):
        noise = np.random.normal(0, 0.05, n_points_initial)
        perturb = initial_h + noise
        perturb = np.clip(perturb, 0, 1)
        sum_perturb = np.sum(perturb)
        scaling = (n_points_initial / 2) / sum_perturb

        perturbations.append(perturb)
    
    best_c5 = float('inf')

    best_n_points = n_points_initial
    start_time = time.time()
    
    # Optimize each perturbation
    for perturb in perturbations:
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points_initial / 2})
        bounds = [(0.0, 1.0) for _ in range(n_points_initial)]
