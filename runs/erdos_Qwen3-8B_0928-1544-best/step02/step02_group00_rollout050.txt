        return np.sum(h_flat) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    # Check if we can afford to run more iterations
    max_iterations = 200
    for _ in range(max_iterations):
        # Compute current C5
        current_c5 = compute_c5(best_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
        else:
            # No improvement, try to break early
            break
        
        if elapsed_time >= budget_s:
            break
        
        # Optimize
        res = minimize(
            fun=lambda x: compute_c5(x),
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-8}
        )
        
        elapsed_time += res.message[-3:]  # Placeholder for actual timing logic
        
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        
        # Early exit if budget is nearing
        if _ > 10 and best_c5 < 0.38085:
            break
        
        # Update h for next iteration
        h = res.x
    
    # Ensure final solution meets constraints
    final_h = best_h
    final_h[final_h < 0] = 0
    final_h[final_h > 1] = 1
    final_sum = np.sum(final_h)
    if np.abs(final_sum - n_points / 2) > 1e-6:
        final_h = final_h / final_sum * (n_points / 2)
    
    return final_h, best_c5, n_points
