import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    start_time = time.time()

    num_restarts = 30
    max_iter_per_restart = 150
    np.random.seed(seed)

    for _ in range(num_restarts):
        random_h = np.random.rand(n_points)
        sum_rand = np.sum(random_h)
        random_h = random_h * (n_points / 2) / sum_rand
        random_h = np.clip(random_h, 0, 1)

        sum_rand = np.sum(random_h)
        if abs(sum_rand - n_points / 2) > 1e-6:
            delta = (n_points / 2 - sum_rand)
            random_h[-1] += delta
            random_h = np.clip(random_h, 0, 1)

        result = minimize(
            compute_C5,
            random_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': max_iter_per_restart}
        )
        if result.success:
            current_h = result.x
            current_c5 = compute_C5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                print(f"New best C5: {best_c5}")

        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break

    return best_h, best_c5, n_points
