import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(70, 0.5)
    sum_h = np.sum(h)
    if np.abs(sum_h - 35) > 1e-6:
        h *= 35.0 / sum_h
    
    def obj_func(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / 70)
    
    def cons_func(h):
        return np.sum(h) - 35
    
    bounds = [(0.0, 1.0)] * 70
    constraints = {'type': 'eq', 'fun': cons_func}
    
    best_h = h.copy()
    best_c = obj_func(best_h)
    
    for _ in range(50):
        pert = np.random.normal(0, 0.05, 70)
        perturbed = np.clip(best_h + pert, 0, 1)
        sum_p = np.sum(perturbed)
        perturbed = perturbed * (35.0 / sum_p)
        res = minimize(obj_func, perturbed, bounds=bounds, constraints=constraints, method='SLSQP', options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            curr_c = res.fun
            if curr_c < best_c:
                best_c = curr_c
                best_h = res.x
    
    final_res = minimize(obj_func, best_h, bounds=bounds, constraints=constraints, method='SLSQP', options={'maxiter': 300, 'ftol': 1e-9})
    if final_res.success:
        best_c = final_res.fun
        best_h = final_res.x
    
    return best_h, best_c, 70
