import numpy as np
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    initial_h = initial_h_values.copy()
    h = initial_h.copy()
    n = n_points
    dx = 2.0 / n

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def project(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        scale = (n / 2) / s
        return h * scale

    def create_population(base, size):
        return [np.clip(base + np.random.normal(0, 0.05, n), 0, 1) * (n / 2) / np.sum(np.clip(base + np.random.normal(0, 0.05, n), 0, 1)) for _ in range(size)]

    def tournament(pop, k=3):
        return min(np.random.choice(pop, k), key=evaluate)

    population = create_population(initial_h, 50)
    best_h = initial_h
    best_c5 = evaluate(best_h)
    t0 = time.time()

    while time.time() - t0 < budget_s:

        for _ in range(len(population) // 2):

            child = (a + b) / 2
            for i in range(n):
