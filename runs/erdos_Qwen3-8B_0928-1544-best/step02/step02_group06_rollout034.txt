import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 256
    dx = 2.0 / n_points
    np.random.seed(seed)
    initial_guesses = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_guesses.append(initial_h_values)
    else:
        num_initial = 30
        for _ in range(num_initial):
            block1_len = np.random.randint(32, 96)
            start1 = np.random.randint(0, n_points - block1_len)
            block2_len = np.random.randint(32, 96)
            start2 = np.random.randint(0, n_points - block2_len)
            h = np.zeros(n_points, dtype=np.float64)
            h[start1:start1 + block1_len] = 1.0
            h[start2:start2 + block2_len] = 1.0
            total_area = block1_len + block2_len
            if total_area != n_points / 2:
                scale = (n_points / 2) / total_area
                h = h * scale
            initial_guesses.append(h)
        initial_guesses.append(np.full(n_points, (n_points / 2) / n_points))
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx

        'type': 'eq',
        'fun': lambda h: np.sum(h) - n_points / 2,
        'jac': lambda h: np.ones_like(h)
    }]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = None
    best_c5 = float('inf')
    
    for guess in initial_guesses:
        result = minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',

            current_c5 = result.fun
        else:
            current_h = guess

        if np.abs(np.sum(current_h) - n_points / 2) > 1e-6:
            current_h = guess
            current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    if best_h is not None:
        result_refine = minimize(
            fun=objective,
