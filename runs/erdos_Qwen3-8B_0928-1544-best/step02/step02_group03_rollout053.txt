    max_gen = 220
    for gen in range(max_gen):
        evals = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / 70)
            evals.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        if time() - start_time > budget_s:
            break
        tourn_size = int(len(population) * 0.1)
        tourn_idx = np.random.choice(len(population), tourn_size, replace=False)
        tourn_evals = [evals[i] for i in tourn_idx]
        top_idx = sorted(tourn_idx, key=lambda i: evals[i])[:2]
        p1, p2 = population[top_idx[0]], population[top_idx[1]]
        off = []
        for _ in range(len(population) // 2):
            cp = np.random.randint(1, 70)
            c1, c2 = np.concatenate((p1[:cp], p2[cp:])), np.concatenate((p2[:cp], p1[cp:]))
            if np.random.rand() < 0.1:
                for i in range(70):
                    if np.random.rand() < 0.1:
                        c1[i] = np.random.rand()
            c1 = np.clip(c1, 0, 1)
            s = np.sum(c1)
            c1 = c1 * (35.0 / s) if s != 0 else np.full(70, 0.5)
            if np.random.rand() < 0.1:
                for i in range(70):
                    if np.random.rand() < 0.1:
                        c2[i] = np.random.rand()
            c2 = np.clip(c2, 0, 1)
            s = np.sum(c2)
            c2 = c2 * (35.0 / s) if s != 0 else np.full(70, 0.5)
            off.extend([c1, c2])
        population = [population[i] for i in sorted(range(len(population)), key=lambda x: evals[x])[:len(population)//2]] + off
    return best_h, best_c5, 70
