import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.optimize import Bounds

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h
    best_c5 = np.inf

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)

    for _ in range(400):
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-10,
            options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        h = res.x

    final_h = np.clip(best_h, 0.0, 1.0)
    final_sum = np.sum(final_h)
    if not np.isclose(final_sum, n_points / 2, atol=1e-6):
        final_h = final_h / final_sum * (n_points / 2)
    
    return final_h, best_c5, n_points
