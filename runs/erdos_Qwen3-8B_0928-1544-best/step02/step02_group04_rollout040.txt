        population[i] *= (n_points / 2) / np.sum(population[i])
        pop_fitness[i] = compute_c5(population[i])

    best_idx = np.argmin(pop_fitness)
    best_h = population[best_idx]
    best_c5 = pop_fitness[best_idx]

    start_time = time.time()
    mutation_rate = 0.05
    crossover_rate = 0.7
    while time.time() - start_time < budget_s:
        idx = np.argsort(pop_fitness)[:pop_size//2]
        selected = [population[i] for i in idx]
        new_pop = []
        for _ in range(pop_size):
            if np.random.rand() < crossover_rate:
                p1, p2 = np.random.choice(selected, 2, replace=False)
                pt = np.random.randint(1, len(p1))
                child = np.concatenate((p1[:pt], p2[pt:]))
            else:
                child = np.random.choice(selected)
                for i in range(len(child)):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.uniform(-0.1, 0.1)
                child = np.clip(child, 0, 1)
                child *= (n_points / 2) / np.sum(child)
            new_pop.append(child)

        for i in range(pop_size):
            new_pop[i] = np.clip(new_pop[i], 0, 1)
            new_pop[i] *= (n_points / 2) / np.sum(new_pop[i])
            pop_fitness[i] = compute_c5(new_pop[i])

        population = new_pop
        best_idx = np.argmin(pop_fitness)
        best_c5 = pop_fitness[best_idx]
        best_h = population[best_idx]

    return best_h, best_c5, n_points
