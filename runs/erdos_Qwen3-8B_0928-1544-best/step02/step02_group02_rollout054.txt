import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    
    # Define initial guess
    def get_initial_guess(n):
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            return initial_h_values.copy()
        return np.full(n, 0.5)
    
    n_points = 70
    initial_guess = get_initial_guess(n_points)
    dx = 2.0 / n_points
    
    # Objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Constraint: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    # Optimization setup
    result = minimize(
        fun=objective,
        x0=initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-7,
        options={'maxiter': 1000, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    # Check for success
    if result.success:
        current_h = result.x
        current_c5 = result.fun
    else:
        current_h = initial_guess
        current_c5 = objective(current_h)
    
    # Check elapsed time
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return current_h, current_c5, n_points
    
    # Early exit if time is up
    return current_h, current_c5, n_points
