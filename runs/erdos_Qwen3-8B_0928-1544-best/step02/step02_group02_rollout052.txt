import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 70
    num_particles = 20
    max_iter = 2000
    dx = 2.0 / n

    # Initialize positions
    initial_h = None
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        initial_h = initial_h_values

    particles = []
    velocities = []
    personal_best = []
    personal_best_cost = []
    global_best = None
    global_best_cost = float('inf')

    for _ in range(num_particles):
        if initial_h is not None and np.random.rand() < 0.5:
            h = initial_h.copy()
            noise = np.random.normal(0, 0.1, h.shape)
            h += noise
            h = np.clip(h, 0, 1)
            scale = (n / 2) / np.sum(h)
            h *= scale
        else:
            h = np.random.rand(n)
            h = np.clip(h, 0, 1)
            scale = (n / 2) / np.sum(h)
            h *= scale
        particles.append(h)
        velocities.append(np.random.uniform(-0.5, 0.5, n))
        # Compute initial cost
        corr = np.correlate(h, 1 - h, mode='full')
        cost = np.max(corr) * dx
        personal_best.append(h)
        personal_best_cost.append(cost)

    # Set initial global best
    for i in range(num_particles):
        if personal_best_cost[i] < global_best_cost:
            global_best = np.copy(particles[i])
            global_best_cost = personal_best_cost[i]

    # PSO main loop
    for iter in range(max_iter):
