import numpy as np
import time
import random
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        n_points = 140
        initial_h = np.ones(n_points) * 0.5

    best_c5 = correlate(best_h, 1 - best_h, mode='full').max() * dx
    T = 1000.0
    alpha = 0.99
    start_time = time.time()

        elapsed = time.time() - start_time
        if elapsed > budget_s:

        h_candidate = best_h.copy()
        for _ in range(5):

            h_candidate[idx] += np.random.uniform(-0.1, 0.1)
        h_candidate = np.clip(h_candidate, 0, 1)
        current_sum = h_candidate.sum()
        delta_sum = target_sum - current_sum
        if delta_sum != 0:
            idx = np.random.randint(n_points)
            h_candidate[idx] += delta_sum
