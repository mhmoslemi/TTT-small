            p1, p2 = selected[np.random.randint(len(selected))], selected[np.random.randint(len(selected))]
            idx = np.random.randint(1, len(p1)-1)
            child = np.concatenate([p1[:idx], p2[idx:]])
            for i in range(len(child)):
                if np.random.rand() < 0.05:
                    child[i] += np.random.uniform(-0.1, 0.1)
            child = np.clip(child, 0, 1)
            sum_val = np.sum(child)
            if sum_val > 0:
                child *= (n_points / 2) / sum_val
            new_population.append(child)
        population = new_population
        for h in population:
            current_c5 = evaluate(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        generations += 1

    best_h = np.clip(best_h, 0, 1)
    sum_val = np.sum(best_h)
    if sum_val > 0:
        best_h *= (n_points / 2) / sum_val
    return best_h, best_c5, n_points
