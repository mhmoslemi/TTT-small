        perturbed = best_h + np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        # Adjust to meet sum constraint
        sum_perturbed = np.sum(perturbed)
        if abs(sum_perturbed - n_points / 2) > 1e-6:
            diff = (n_points / 2) - sum_perturbed
            perturbed[-1] += diff
            perturbed = np.clip(perturbed, 0.0, 1.0)

        # Optimize from perturbed guess
        result = minimize(
            compute_C5,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': max_iter_per_restart, 'ftol': 1e-8, 'gtol': 1e-8}
        )

        if result.success and result.fun < best_c5:
            current_h = result.x
            current_c5 = compute_C5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                print(f"New best C5: {best_c5}")

        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break

    return best_h, best_c5, n_points
