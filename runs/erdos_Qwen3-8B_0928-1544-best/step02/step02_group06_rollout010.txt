    if result.success:
        best_h = result.x
    else:
        best_h = best_h
    
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    if np.abs(sum_h - target_sum) > 1e-6:
        best_h = initial_h_values if initial_h_values is not None else population[0]
        best_c5 = objective(best_h)
    
    current_time = time.time()
    if current_time - start_time > budget_s:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
