import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 200
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h
    best_c5 = np.inf
    max_iter = 100
    shift_res = 4 * n_points
    shifts = np.linspace(-2, 2, shift_res)
    
    for _ in range(max_iter):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        if best_c5 < 0.38085:
            break
        
        # Find shift with maximum overlap
        worst_k_index = np.argmax(corr)
        worst_shift = (worst_k_index - n_points) * dx
        k = worst_shift
        
        # Optimize for specific shift
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            corr_k = 0
            for i in range(n_points):
                j = int((i + k) / dx)
                if 0 <= j < n_points:
                    corr_k += h[i] * h[j]
            return corr_k * dx
        
        def constraint_sum(h_flat):
            return np.sum(h_flat) - n_points / 2
        
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum})
        
        if res.success:
            h = res.x
            h[h < 0] = 0
            h[h > 1] = 1
            total = np.sum(h)
            if abs(total - n_points / 2) > 1e-6:
                h = h / total * (n_points / 2)
    
    final_h = best_h
    final_h[final_h < 0] = 0
    final_h[final_h > 1] = 1
    final_sum = np.sum(final_h)
    if np.abs(final_sum - n_points / 2) > 1e-6:
        final_h = final_h / final_sum * (n_points / 2)
    
    return final_h, best_c5, n_points
