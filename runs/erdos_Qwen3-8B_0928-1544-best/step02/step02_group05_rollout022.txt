    best_c5 = float('inf')
    
    # First round: optimize multiple perturbations
    for h_start in perturbations:
        h_clipped = np.clip(h_start, 0, 1)

        scaling = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h_scaled = h_clipped * scaling
        
        def objective(h):
            corr = correlate(h, 1 - h, mode='full')
            return np.max(corr)
        
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result = minimize(
            objective,
            h_scaled,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 2000, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        if result.success and result.fun * dx < best_c5:

            best_h = result.x
    
    # Second round: upsample best candidate if time remains

    if remaining_time > 200 and n_points < 280:
        new_n_points = 280

        scaling = target_sum / sum_h
        new_h_scaled = new_h * scaling
        new_h_clipped = np.clip(new_h_scaled, 0, 1)
        
        def objective2(h):
            corr = correlate(h, 1 - h, mode='full')
            return np.max(corr)
        
        bounds_new = [(0.0, 1.0) for _ in range(new_n_points)]
        cons_new = ({'type': 'eq', 'fun': lambda h: np.sum(h) - new_n_points / 2})
