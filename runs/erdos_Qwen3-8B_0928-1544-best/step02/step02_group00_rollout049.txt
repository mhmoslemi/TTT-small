import numpy as np
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    sum_target = n_points / 2
    dx = 2.0 / n_points
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:

    else:

        h_initial = h_initial / np.sum(h_initial) * sum_target
    
    def create_candidate(h):

        h_sum = np.sum(h)
        if h_sum == 0:
            return np.zeros_like(h) + 1.0 / len(h) * sum_target
        h = h * sum_target / h_sum
        return h
    
    population = [create_candidate(h_initial)]
    for _ in range(49):
        perturbation = np.random.normal(0, 0.05, n_points)
        candidate = h_initial + perturbation
        population.append(create_candidate(candidate))
    
    best_h = population[0]
    best_c5 = np.inf
    
    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    for _ in range(200):
        fitness = [compute_c5(h) for h in population]
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:

            break
        
        top_indices = np.argsort(fitness)[:25]
