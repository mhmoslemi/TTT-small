        h = np.zeros(n_coarse)
        h[start:start+32] = 1.0
        h = run_optimize(h, n_coarse, 0.2, seed + 1)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Medium level
    h_medium = upscale(best_h)
    h_medium = perturb(h_medium, 0.1, seed + 2)
    h_medium = run_optimize(h_medium, n_points, 0.15, seed + 3)
    c5 = compute_c5(h_medium)
    if c5 < best_c5:
        best_h = h_medium.copy()
        best_c5 = c5
    
    # Fine level
    h_fine = upscale(best_h)
    h_fine = perturb(h_fine, 0.05, seed + 4)
    h_fine = run_optimize(h_fine, n_points, 0.1, seed + 5)
    c5 = compute_c5(h_fine)
    if c5 < best_c5:
        best_h = h_fine.copy()
        best_c5 = c5
    
    best_h = np.clip(best_h, 0.0, 1.0)
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = initial_h_values
    
    if best_c5 < 0.380876:
        return (best_h, best_c5, n_points)
    return (best_h, best_c5, n_points)
