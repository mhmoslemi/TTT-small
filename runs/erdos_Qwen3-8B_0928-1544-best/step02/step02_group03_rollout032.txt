import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = None
    n_points = 70
    dx = 2.0 / n_points
    
    # Initial guess if provided
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.full(n_points, 0.5)
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Bounds
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    # Initial optimization with SLSQP with higher iterations
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        tol=1e-9,
        options={'maxiter': 500, 'ftol': 1e-10}
    )
    
    if result.success:
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    # Additional optimization using differential evolution for global search
    # Convert to bounds for DE
    de_bounds = [(0, 1) for _ in range(n_points)]
    result_de = differential_evolution(
        fun=objective,
        bounds=de_bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-9,
        mutation=(0.5, 1),
        recombination=0.8,
        disp=False,
        func_tol=1e-9,
        x0=h_initial
    )
    
    if result_de.success:
        de_h = result_de.x
        if np.sum(de_h) * dx != 1:
            de_h = np.clip(de_h, 0, 1)
            de_h = de_h / np.sum(de_h) * n_points / 2
        de_c5 = np.max(np.correlate(de_h, 1 - de_h, mode='full')) * dx
        if de_c5 < best_c5:
            best_c5 = de_c5
            best_h = de_h
    
    # Final constraint check
    best_h = np.clip(best_h, 0, 1)
    if np.sum(best_h) * dx != 1:
        best_h = best_h / np.sum(best_h) * n_points / 2
    
    return best_h, best_c5, n_points
