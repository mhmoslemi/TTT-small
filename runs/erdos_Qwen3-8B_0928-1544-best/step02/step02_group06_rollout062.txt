    if 'initial_h_values' in globals():
        initial_guess = initial_h_values.copy()
    else:
        initial_guess = np.random.rand(n_points)
        initial_guess = initial_guess * (n_points / 2) / np.sum(initial_guess)
    
    num_fireflies = 50
    fireflies = [initial_guess.copy() for _ in range(num_fireflies)]
    for i in range(num_fireflies):
        fireflies[i] = fireflies[i] * (n_points / 2) / np.sum(fireflies[i])
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    alpha = 0.5
    beta = 1.0
    gamma = 0.1
    max_iter = 1000
    kernel_width = 5
    kernel = np.hanning(kernel_width)
    kernel = kernel / np.sum(kernel)
    
    best_h = initial_guess.copy()
    best_c5 = objective(initial_guess)
    
    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        brightnesses = [objective(f) for f in fireflies]
        sorted_fireflies = sorted(zip(brightnesses, fireflies), key=lambda x: -x[0])
        for i in range(len(sorted_fireflies)):
            current_brightness, current_firefly = sorted_fireflies[i]
            if i == 0:
                continue
            brightest, brightest_firefly = sorted_fireflies[0]
            r = np.linalg.norm(current_firefly - brightest_firefly)
            delta = beta * np.exp(-gamma * r) * (brightest_firefly - current_firefly)
            noise = alpha * np.random.normal(size=n_points)
            new_firefly = current_firefly + delta + noise
            new_firefly = convolve(new_firefly, kernel, mode='same')
            new_firefly = np.clip(new_firefly, 0, 1)
            new_sum = np.sum(new_firefly)
            if new_sum > 1e-12:
                new_firefly = new_firefly * (n_points / 2) / new_sum
            fireflies[i] = new_firefly
    
    for h in fireflies:
        curr_c5 = objective(h)
        if curr_c5 < best_c5:
            best_h = h.copy()
            best_c5 = curr_c5
    
    cons = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - n_points / 2,
        'jac': lambda h: np.ones_like(h)
    }]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
    )
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_h = best_h
        best_c5 = objective(best_h)
    
    best_h = np.clip(best_h, 0.0, 1.0)
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = initial_guess
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
