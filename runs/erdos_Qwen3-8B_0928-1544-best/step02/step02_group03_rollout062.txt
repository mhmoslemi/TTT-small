import numpy as np
from scipy.optimize import basinhopping, Bounds, minimize

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h = initial_h_values.copy()
    else:
        h = np.full(70, 0.5)
    
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * (2.0 / 70)
    
    def constraint_sum(x):
        return np.sum(x) - 35.0
    
    bounds = Bounds([0.0]*70, [1.0]*70)
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    def project_h(x):
        x_clipped = np.clip(x, 0, 1)
        if np.sum(x_clipped) == 0:
            return x_clipped
        return x_clipped * 35.0 / np.sum(x_clipped)
    
    result = basinhopping(
        fun=objective,
        x0=h,
        minimizer_kwargs={
            'method': 'SLSQP',
            'bounds': bounds,
            'constraints': cons,
            'tol': 1e-8,
            'options': {'maxiter': 200, 'ftol': 1e-9}
        },
        niter=20,
        T=1.0,
        accept_probability='exponential'
    )
    
    best_h = project_h(result.x)
    best_c5 = result.fun
    
    result_refine = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-8,
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    
    if result_refine.success:
        best_c5 = result_refine.fun
        best_h = result_refine.x
    
    best_h = project_h(best_h)
    return best_h, best_c5, len(best_h)
