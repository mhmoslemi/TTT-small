import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    initial_h = initial_h_values.copy()
    
    def clamp_and_scale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return h
        scale = (n_points / 2) / total
        return h * scale
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    pop_size = 50
    mutation_rate = 0.15
    tournament_size = 3
    max_gen = 100
    start_time = time.time()
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        p = initial_h + np.random.normal(0, 0.1, n_points)
        p = clamp_and_scale(p)
        population.append(p)
    
    best_h = initial_h
    best_c5 = evaluate(best_h)
    
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate and track better solutions
        fitness = np.array([evaluate(ind) for ind in population])
        idx = np.argmin(fitness)
        if fitness[idx] < best_c5:
            best_c5 = fitness[idx]
            best_h = population[idx].copy()
        
        new_pop = []
        for _ in range(pop_size):
            # Tournament selection
            tour = np.random.choice(population, tournament_size, replace=False)
            best_t = min(tour, key=lambda x: evaluate(x))
            new_pop.append(best_t)
        
        for i in range(pop_size):
            # Crossover
            p1 = new_pop[i]
            p2 = np.random.choice(population)
            alpha = np.random.rand()
            child = p1 * alpha + p2 * (1 - alpha)
            
            # Apply mutation
            if np.random.rand() < mutation_rate:
                mask = np.random.rand(n_points) < 0.1
                child[mask] += np.random.normal(0, 0.1, np.sum(mask))
                child = clamp_and_scale(child)
            
            new_pop[i] = clamp_and_scale(child)
        
        population = new_pop
    
    return (best_h, best_c5, n_points)
