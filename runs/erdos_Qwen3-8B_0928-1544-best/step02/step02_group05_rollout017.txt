import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def project_to_feasible(h):
        total = np.sum(h)
        target = n_points / 2
        if total == 0:

    alpha = 0.995
