import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    # Initialize with 100 points
    n_points = 100
    dx = 2.0 / n_points
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        prev_x = np.linspace(0, 2, 70)
        new_x = np.linspace(0, 2, n_points)
        initial_h = np.interp(new_x, prev_x, initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    # Objective with L1 regularization to encourage sparsity
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx + 0.01 * np.sum(np.abs(h))
    
    # Run Nelder-Mead optimization
    result = minimize(
        fun=objective,
        x0=initial_h,
        method='Nelder-Mead',
        options={'xatol': 1e-9, 'fatol': 1e-9, 'maxiter': 400, 'adaptive': True}
    )
    
    # Final projection to ensure feasibility
    h_proj = np.clip(result.x, 0.0, 1.0)
    sum_h = np.sum(h_proj)
    if sum_h != 0:
        h_proj *= 50.0 / sum_h
    else:
        h_proj = np.full(n_points, 0.5)
    
    # Compute C5 value
    corr = np.correlate(h_proj, 1 - h_proj, mode='full')
    c5 = np.max(corr) * dx
    
    return h_proj, c5, n_points
