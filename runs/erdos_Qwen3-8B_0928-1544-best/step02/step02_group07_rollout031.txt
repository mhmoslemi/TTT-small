        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial best solution
    best_h = h_initial.copy()

    max_iter_per_restart = 200

            diff = (n_points / 2) - sum_random

        # Optimize from random guess
        result = minimize(
