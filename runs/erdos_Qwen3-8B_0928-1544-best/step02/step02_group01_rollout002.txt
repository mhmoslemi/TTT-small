import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    n_particles = 30
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    particles = np.tile(initial_h, (n_particles, 1)) + np.random.normal(0, 0.1, (n_particles, n_points)) * 0.5

    particles = particles * scale_factors

        return np.max(corr) * dx
    pbest = particles.copy()
    gbest = particles.copy()

    for i in range(n_particles):
        current_c5 = compute_C5(particles[i])
        if current_c5 < best_c5:
            best_c5 = current_c5
            gbest = particles[i].copy()
    best_c5_list = []
    start_time = time.time()

        if elapsed > budget_s:

            velocities[i] = w * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
