    # Bounds: each h[i] must be between 0 and 1
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial best solution
    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    elapsed_time = 0.0
    start_time = time.time()

    # Optimization parameters
    num_restarts = 50  # Increased number of restarts
    max_iter_per_restart = 100
    np.random.seed(seed)

    for _ in range(num_restarts):
        # Generate random initial guess
        if _ == 0:  # Use initial_h_values as the first guess
            random_h = best_h
        else:
            random_h = np.random.rand(n_points)
            sum_random = np.sum(random_h)
            scale = (n_points / 2) / sum_random
            random_h = random_h * scale
            random_h = np.clip(random_h, 0.0, 1.0)

        # Adjust to meet sum constraint (ensure precision)
        sum_random = np.sum(random_h)
        if abs(sum_random - n_points / 2) > 1e-6:
            diff = (n_points / 2) - sum_random
            random_h[-1] += diff
            random_h = np.clip(random_h, 0.0, 1.0)

        # Optimize from random guess
        result = minimize(
            compute_C5,
            random_h,
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': max_iter_per_restart}
        )

        if result.success:
            current_h = result.x
            current_c5 = compute_C5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                print(f"New best C5: {best_c5}")

        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break

    return best_h, best_c5, n_points
