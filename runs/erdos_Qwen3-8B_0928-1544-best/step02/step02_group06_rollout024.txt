import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 256
    dx = 2.0 / n_points
    initial_guesses = []
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_guesses.append(initial_h_values)
    else:
        initial_guesses.append(np.full(n_points, n_points/2) / n_points)
    
    for _ in range(2):
        h = np.zeros(n_points)
        remaining = n_points / 2
        for _ in range(3):
            start = np.random.randint(0, n_points - 20)
            size = np.random.randint(10, 30)
            if remaining < size:
                size = remaining
            h[start:start+size] = 1.0
            remaining -= size
        initial_guesses.append(h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    cons = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - n_points / 2,
        'jac': np.ones_like
    }]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = None
    best_c5 = float('inf')
    
    for guess in initial_guesses:
        res = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 500})
        if res.success:
            current_h = res.x
        else:
            current_h = guess
        current_h = np.clip(current_h, 0, 1)
        if np.abs(np.sum(current_h) - n_points / 2) > 1e-6:
            current_h = guess
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    if best_h is not None:
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 300})
        best_h = res.x if res.success else best_h
        best_h = np.clip(best_h, 0, 1)
    
    return (best_h, best_c5, n_points)
