        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2, 'jac': np.ones(n_points)}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        best_curr = None
        best_c = float('inf')
        for guess in initial_guesses:
            res = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 200 if level == 0 else 300})
            if res.success:
                curr_h = res.x
                curr_c = res.fun
            else:
                curr_h = guess
                curr_c = objective(curr_h)
            curr_h = np.clip(curr_h, 0, 1)
            if np.abs(np.sum(curr_h) - n_points / 2) > 1e-6:
                curr_h = guess
                curr_c = objective(curr_h)
            if curr_c < best_c:
                best_curr = curr_h
                best_c = curr_c
        h_prev = best_curr.copy()
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    return (h_prev, best_c, 256)
