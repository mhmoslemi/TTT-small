import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()
    else:
        n = 100
        h = np.random.rand(n)
        h = h / np.sum(h) * (n / 2)
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h
    best_c5 = np.inf
    max_iterations = 200
    iteration = 0

    def evaluate_c5(h_vals):
        h = h_vals.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    def objective(h_flat):
        return evaluate_c5(h_flat)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)

    while iteration < max_iterations and budget_s > 0:
        current_c5 = evaluate_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        if best_c5 < 0.38085:
            break

        corr = correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        k_max_val = corr[k_max] * dx
        if k_max_val < best_c5:
            break

        # Add constraint: overlap(k_max) <= best_c5
        def constraint_k(h_flat):
            h = h_flat.reshape(n_points)
            # Compute overlap at k_max for current h
            # k_max is index in the full correlation output
            # The shift in terms of x is (k_max - n_points + 1) * dx
            # The overlap is corr[k_max] * dx, but we want it to be <= best_c5
            return corr[k_max] * dx - best_c5

        constraints = [{'type': 'ineq', 'fun': constraint_k}]
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints + [{'type': 'eq', 'fun': constraint_sum}],
            tol=1e-8,
            options={'maxiter': 100, 'ftol': 1e-12}
        )

        if res.success:
            best_h = res.x
            iteration += 1
            if best_c5 > res.fun and res.fun < 0.38085:
                best_c5 = res.fun
        else:
            break

        # Early exit if budget is nearing
        if iteration > 10 and best_c5 < 0.38085:
            break

    # Ensure final solution meets constraints
    final_h = best_h
    final_h[final_h < 0] = 0
    final_h[final_h > 1] = 1
    final_sum = np.sum(final_h)
    if np.abs(final_sum - n_points / 2) > 1e-6:
        final_h = final_h / final_sum * (n_points / 2)

    return final_h, best_c5, n_points
