from scipy.optimize import minimize
from scipy.signal import correlate
import time

    bounds = [(0.0, 1.0) for _ in range(n_points_initial)]
    
    start_time = time.time()
    time_used = 0.0
    
    result_initial = minimize(
        objective, initial_h, method='SLSQP',
        bounds=bounds, constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-9, 'eps': 1e-8}
    )
    best_h = result_initial.x
    best_c5 = result_initial.fun * dx_initial
    time_used = time.time() - start_time

        scaling = (n_points_initial / 2) / sum_clipped
        h_perturbed_scaled = h_clipped * scaling

    for h_start in perturbations:
