import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    n = 100
    dx = 2.0 / n
    initial_h = np.full(n, 0.5)
    
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        x_ref = np.linspace(0, 2, 70)
        x_new = np.linspace(0, 2, n)
        initial_h = np.interp(x_new, x_ref, initial_h_values)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx + 0.01 * np.sum(np.abs(h))
    
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2}
    bounds = [(0.0, 1.0) for _ in range(n)]
    
    result = minimize(
        fun=objective,
        x0=initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-8,
        options={'maxiter': 500, 'ftol': 1e-9}
    )
