import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    np.random.seed(seed)
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points

        return np.max(corr) * dx

        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:

        return h_clipped * scale

        perturb = np.random.normal(0, 0.1, n_points)
        h_perturbed = h_initial + perturb
        h_perturbed = project_feasible(h_perturbed)
        population.append(h_perturbed)

    best_h = h_initial.copy()
    best_c5 = compute_C5(h_initial)

    start_time = time.time()
    mutation_std = 0.1
    generation = 0

    while time.time() - start_time < budget_s:
        next_population = []
        for h in population:
            perturb = np.random.normal(0, mutation_std, n_points)
            h_mut = h + perturb
            h_mut = project_feasible(h_mut)

                best_h = h.copy()

        if generation % 100 == 0 and mutation_std > 0.01:

    return best_h, best_c5, n_points
