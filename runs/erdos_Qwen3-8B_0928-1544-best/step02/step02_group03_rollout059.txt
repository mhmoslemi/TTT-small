import numpy as np
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    # Initialize with provided initial_h_values if available
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        n = 100

        h_initial = np.full(n, 0.5)

    λ = 0.01  # Regularization parameter for L1 sparsity

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8,

    )
