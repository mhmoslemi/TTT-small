import numpy as np
from scipy.optimize import Bounds
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    population_size = 50
    pop = [initial_h]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (35.0 / np.sum(h))
        pop.append(h)

    best_h = initial_h
    best_c5 = float('inf')

    generation = 0
    no_improvement = 0
    max_generations = int(budget_s / (population_size * 0.1))

    while generation < max_generations and no_improvement < 100:
        generation += 1
        scores = []
        for h in pop:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            scores.append((c5, h))
        scores.sort()
        current_best_c5, current_best_h = scores[0]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5

            no_improvement = 0

            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_pop.extend([child1, child2])
        pop = new_pop[:population_size]

    return best_h, best_c5, n_points

def tournament_selection(pop, scores, tournament_size):
    selected = random.sample(list(range(len(pop))), tournament_size)
    best = min(selected, key=lambda i: scores[i][0])
    return pop[best].copy(), pop[best].copy()

def crossover(a, b):
    point = np.random.randint(1, len(a)-1)
    return np.concatenate((a[:point], b[point:])), np.concatenate((b[:point], a[point:]))

def mutate(h):
    h = h.copy()
    for i in range(len(h)):
        if np.random.rand() < 0.1:
            h[i] = np.random.rand()
    h = h * (35.0 / np.sum(h))
    h = np.clip(h, 0, 1)
    return h
