from scipy.optimize import minimize
import random

    gamma = 0.1
    kernel_type = 'gaussian'

    fireflies = [np.copy(initial_guess) for _ in range(num_fireflies)]
    for f in fireflies:
        np.random.shuffle(f)
        np.cumsum(f, out=f)
        f /= np.sum(f)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    def smooth_h(h, sigma):
        kernel = np.exp(-0.5 * (np.arange(-20, 21) ** 2) / (sigma**2))
        kernel /= np.sum(kernel)
        return np.convolve(h, kernel, mode='same')
    
    def fease(h):

        scale = (n_points / 2) / np.sum(h)
        return h * scale
    
    for iter in range(max_iter):
        brightness = [objective(f) for f in fireflies]

                    delta = beta * np.exp(-gamma * dist)
                    fireflies[i] += delta * (fireflies[j] - fireflies[i]) + alpha * np.random.randn(n_points)

    best_h = fireflies[0]
