import numpy as np
import time
import random
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points

    pop_size = 200
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        p = initial_h + np.random.normal(0, 0.1, n_points)
        p = np.clip(p, 0, 1)
        s = np.sum(p)
        if s == 0:
            p = np.zeros_like(p)
        else:
            p = p * (n_points / 2) / s
        population.append(p)

    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def rescale(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / s

    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        parents = []
        for _ in range(2):
            t = random.sample(population, 5)
            winner = min(t, key=evaluate)
            parents.append(winner)

        child = (parents[0] + parents[1]) / 2
        child = rescale(child)

        if np.random.rand() < 0.1:
            m = np.random.normal(0, 0.05, n_points)
            child = rescale(child + m)

        child_c5 = evaluate(child)
        worst_idx = np.argmax([evaluate(ind) for ind in population])
        population[worst_idx] = child

        if child_c5 < best_c5:
            best_h = child.copy()
            best_c5 = child_c5

    return (best_h, best_c5, n_points)
