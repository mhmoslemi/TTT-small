import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    from scipy.optimize import minimize
    import time

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_C5(h):
        h = np.array(h)
        n = n_points
        fft_h = np.fft.fft(h)
        fft_1mh = np.fft.fft(1.0 - h)
        cross = np.fft.ifft(fft_h * np.conj(fft_1mh)).real
        max_corr = np.max(np.abs(cross))
        return max_corr * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    elapsed = 0.0
    start = time.time()

    np.random.seed(seed)
    num_restarts = 40
    max_iter = 150

    for _ in range(num_restarts):
        # Perturbation and clipping
        pert = np.random.normal(0, 0.1, n_points)
        h_guess = h_initial + pert
        h_guess = np.clip(h_guess, 0.0, 1.0)
        sum_h = np.sum(h_guess)
        if abs(sum_h - n_points / 2) > 1e-6:
            diff = (n_points / 2) - sum_h
            h_guess[-1] += diff
            h_guess = np.clip(h_guess, 0.0, 1.0)
        # Optimize
        res = minimize(compute_C5, h_guess,
                       method='L-BFGS-B',
                       bounds=bounds,
                       constraints={'type': 'eq', 'fun': constraint_sum},
                       options={'maxiter': max_iter, 'ftol': 1e-6})
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h = res.x.copy()
                best_c5 = current_c5
                print(f"New best C5: {best_c5}")
        elapsed = time.time() - start
        if elapsed > budget_s:

    return best_h, best_c5, n_points
