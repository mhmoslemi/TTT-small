    prob = cp.Problem(cp.Minimize(obj), constraints)
    prob.solve(solver='CVXOPT', verbose=False)
    best_h = h.value
    best_c5 = np.max(correlate(best_h, 1 - best_h, mode='full')) * dx
    current_n_points = n_points
    iterations = 0
    
    # Iterative refinement
    while iterations < 10 and budget_s > 100:
        overlaps = correlate(best_h, 1 - best_h, mode='full') * dx
        k_max = np.argmax(overlaps)
        # Construct quadratic constraint
        h_roll = np.roll(best_h, k_max)
        quad_terms = np.sum(best_h * h_roll) * dx
        C = best_c5
        # Formulate QCQP
        h = cp.Variable(n_points)
        obj = cp.Minimize(C)
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        quad_expr = cp.quad_form(h, np.outer(best_h, best_h))
        constraints.append(quad_expr >= 1 - C)
        prob = cp.Problem(cp.Minimize(C), constraints)
        try:
            prob.solve(solver='CVXOPT', verbose=False)
            best_h = h.value
            best_c5 = np.max(correlate(best_h, 1 - best_h, mode='full')) * dx
            iterations += 1
        except:
            break
    
    # Upsample if time remains
    if budget_s > 1000:
        new_n_points = 140
        original_h = best_h
        new_h = np.interp(np.linspace(0, 2, new_n_points), np.linspace(0, 2, n_points), original_h)
        sum_h = np.sum(new_h)
        target_sum = new_n_points / 2
        scaling = target_sum / sum_h
        new_h_scaled = new_h * scaling
        new_h_clipped = np.clip(new_h_scaled, 0, 1)
        
        h = cp.Variable(new_n_points)
        obj = cp.Minimize(1.0)
        constraints = [cp.sum(h) == new_n_points / 2, 0 <= h, h <= 1]
        overlaps = correlate(new_h_clipped, 1 - new_h_clipped, mode='full') * (2.0 / new_n_points)
        k_max = np.argmax(overlaps)
        h_roll = np.roll(new_h_clipped, k_max)
        quad_terms = np.sum(new_h_clipped * h_roll) * (2.0 / new_n_points)
        constraints.append(cp.quad_form(h, np.outer(new_h_clipped, new_h_clipped)) >= 1 - 1.0)
        prob = cp.Problem(cp.Minimize(1.0), constraints)
        prob.solve(solver='CVXOPT', verbose=False)
        best_h2 = h.value
        best_c52 = np.max(correlate(best_h2, 1 - best_h2, mode='full')) * (2.0 / new_n_points)
        if best_c52 < best_c5:
            best_h = best_h2
            best_c5 = best_c52
            current_n_points = new_n_points
    
    return (best_h, best_c5, current_n_points)
