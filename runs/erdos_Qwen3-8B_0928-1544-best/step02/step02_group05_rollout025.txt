import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        n_points = 140
        h = np.full(n_points, 0.5)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    sum_h = np.sum(h)
    scaling = target_sum / sum_h if sum_h > 0 else 1.0
    h = h * scaling

    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)
    temperature = 1000.0
    cooling_rate = 0.99
    max_iter = int(budget_s / 0.1)
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        perturbed = h.copy()
        num_perturb = max(1, int(n_points * 0.05))
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        for i in indices:
            perturbed[i] = np.random.uniform(0.0, 1.0)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        sum_perturbed = np.sum(perturbed)
        scale = target_sum / sum_perturbed
        perturbed *= scale
        current_c5 = compute_c5(h)
        new_c5 = compute_c5(perturbed)
        delta = new_c5 - current_c5
        if np.random.rand() < np.exp(-delta / temperature):
            h = perturbed.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        temperature *= cooling_rate

    return (best_h, best_c5, n_points)
