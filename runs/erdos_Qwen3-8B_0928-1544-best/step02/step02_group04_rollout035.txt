    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        n_initial = len(initial_h)
        new_n_points = 256
        x_initial = np.linspace(0, 2, n_initial)
        f = interp1d(x_initial, initial_h, kind='linear', fill_value='extrapolate')
        x_new = np.linspace(0, 2, new_n_points)
        initial_h_resampled = f(x_new)
        current_sum = np.sum(initial_h_resampled)
        scale_factor = (new_n_points / 2) / current_sum
        initial_h_resampled *= scale_factor
        initial_h = initial_h_resampled
    else:
        new_n_points = 256
        initial_h = np.ones(new_n_points) * (new_n_points / 2) / new_n_points

    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = initial_h.copy()

    # Compute initial correlations
    corr = np.zeros(n_points)
    for k in range(n_points):
        h_rolled = np.roll(h, k)
        corr[k] = np.dot(h, 1 - h_rolled) * dx

    # Optimization loop
    for _ in range(100):
        for i in range(n_points):
            old_val = h[i]
            low, high = 0.0, 1.0
            for _ in range(20):
                m1 = low + (high - low)/3
                m2 = high - (high - low)/3
                # Evaluate f(m1) and f(m2)
                def f(m):
                    delta = m - old_val
                    i_plus_k = (np.arange(n_points) + i) % n_points
                    i_minus_k = (np.arange(n_points) - i) % n_points
                    term = 1.0 - h[i_plus_k] - h[i_minus_k]
                    delta_corr = delta * term
                    return np.max(corr + delta_corr)
                f1 = f(m1)
                f2 = f(m2)
                if f1 < f2:
                    high = m2
                else:
                    low = m1
            new_val = (low + high) / 2
            delta = new_val - old_val
            h[i] = new_val
            i_plus_k = (np.arange(n_points) + i) % n_points
            i_minus_k = (np.arange(n_points) - i) % n_points
            term = 1.0 - h[i_plus_k] - h[i_minus_k]
            corr += delta * term

        # Maintain sum constraint
        sum_h = np.sum(h)
        if sum_h != 0:
            scale_factor = (n_points / 2.0) / sum_h
            h *= scale_factor
        else:
            h[:] = 0.5

        c5 = np.max(corr)
        if c5 < 0.380876:
            break

    # Final checks
    sum_h = np.sum(h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        h = initial_h
        c5 = np.max(np.zeros(n_points))
        for k in range(n_points):
            h_rolled = np.roll(h, k)
            c5 = max(c5, np.dot(h, 1 - h_rolled) * dx)
    else:
        c5 = np.max(corr)

    h = np.clip(h, 0, 1)
    return h, c5, n_points
