        total = np.sum(h)

            return np.full_like(h, 0.5)
        scale = (n_points / 2) / total
        return np.clip(h * scale, 0, 1)
    
    h = project_feasible(h)
    
    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = current_c5
    
    T = 1000.0
    alpha = 0.995
    iter_count = 0

    while iter_count < max_iter and T > 1e-6:

        T *= alpha
