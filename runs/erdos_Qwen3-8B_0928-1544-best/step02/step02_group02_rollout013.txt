import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n = 70
    if 'initial_h_values' in locals() and len(initial_h_values) == n:
        initial = initial_h_values
    else:
        initial = np.full(n, 0.5)
    num_particles = 20
    particles = []
    velocities = []
    for _ in range(num_particles):
        if np.random.rand() < 0.5:
            pos = initial.copy()
        else:
            pos = np.random.rand(n)
            pos = pos * (n/2) / np.sum(pos)
        particles.append(pos)
        velocities.append(np.random.uniform(-0.5, 0.5, n))
    def compute_c5(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    personal_best = [p.copy() for p in particles]
    personal_best_values = [compute_c5(p) for p in particles]
    global_best = min(personal_best, key=lambda x: compute_c5(x))
    global_best_value = compute_c5(global_best)
    w, c1, c2 = 0.7, 2.0, 2.0
    iterations = 0
    best_c5 = global_best_value
    best_h = global_best.copy()
    while time.time() - start_time < budget_s:
        iterations += 1
        for i in range(num_particles):
            pos = particles[i]
            vel = velocities[i]
            r1, r2 = np.random.rand(), np.random.rand()
            cognitive = c1 * r1 * (personal_best[i] - pos)
            social = c2 * r2 * (global_best - pos)
            new_vel = w * vel + cognitive + social
            velocities[i] = new_vel
            new_pos = pos + new_vel
            new_pos = np.clip(new_pos, 0, 1)
            sum_new = np.sum(new_pos)
            if sum_new > 0:
                scale = (n/2) / sum_new
                new_pos *= scale
            else:
                new_pos = np.full(n, 0.5)
            particles[i] = new_pos
            current_c5 = compute_c5(particles[i])
            if current_c5 < personal_best_values[i]:
                personal_best[i] = particles[i].copy()
                personal_best_values[i] = current_c5
                if current_c5 < global_best_value:
                    global_best_value = current_c5
                    global_best = particles[i].copy()
                    best_c5 = global_best_value
                    best_h = global_best.copy()
        if iterations % 100 == 0:
            for _ in range(int(num_particles * 0.1)):
                idx = np.random.randint(num_particles)
                particles[idx] += np.random.normal(0, 0.01, n)
                particles[idx] = np.clip(particles[idx], 0, 1)
                sum_p = np.sum(particles[idx])
                if sum_p > 0:
                    scale = (n/2) / sum_p
                    particles[idx] *= scale
                else:
                    particles[idx] = np.full(n, 0.5)
    return (best_h, best_c5, n)
