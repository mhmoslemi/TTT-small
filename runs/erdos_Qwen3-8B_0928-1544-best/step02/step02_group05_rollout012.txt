            options={'maxiter': 800, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        best_h2 = result2.x
        best_c52 = result2.fun * (2.0 / new_n_points)
        if best_c52 < best_c5:
            best_c5 = best_c52
            best_h = best_h2
            current_n_points = new_n_points
    
    return (best_h, best_c5, current_n_points)
