from scipy.optimize import minimize

import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 3000, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    best_h = result.x
    best_c5 = result.fun
    current_n_points = n_points
