import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    from scipy.optimize import minimize
    import time

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_initial_guess():
        h = np.random.rand(n_points)
        scale = (n_points / 2) / np.sum(h)
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if abs(sum_h - n_points / 2) > 1e-6:
            h[-1] += (n_points / 2 - sum_h)
            h = np.clip(h, 0.0, 1.0)
        return h

    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    elapsed_time = 0.0
    start_time = time.time()
    num_restarts = 50
    max_iter_per_restart = 50

    for _ in range(num_restarts):
        random_h = generate_initial_guess()
        result = minimize(
            compute_C5,
            random_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': max_iter_per_restart, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        if result.success:
            current_h = result.x
            current_c5 = compute_C5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break

    return best_h, best_c5, n_points
