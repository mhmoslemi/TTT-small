    dx = 2.0 / n_points

    def evaluate_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    corr_initial = correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr_initial) * dx
    best_h = h.copy()

    for iteration in range(200):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap_idx = np.argmax(corr)
        k = max_overlap_idx - (n_points - 1)
        contributions = np.array([h[i] * (1 - h[i + k]) if 0 <= i + k < n_points else 0.0 for i in range(n_points)])
        top_indices = np.argsort(contributions)[::-1][:5]
        decrease_indices = top_indices
        overlapping = [i for i in range(n_points) if 0 <= i + k < n_points]
        non_overlapping = [i for i in range(n_points) if i not in overlapping]
        if len(non_overlapping) > 0:
            increase_indices = np.random.choice(non_overlapping, size=5, replace=False)
        else:
            increase_indices = np.random.choice(range(n_points), size=5, replace=False)
        delta = 0.01
        h[decrease_indices] -= delta
        h[increase_indices] += delta
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        if not np.isclose(current_sum, n_points / 2, atol=1e-6):
            h = h / current_sum * (n_points / 2)
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.38085:
            break
    
    final_h = best_h
    final_sum = np.sum(final_h)
    if not np.isclose(final_sum, n_points / 2, atol=1e-6):
        final_h = final_h / final_sum * (n_points / 2)
    final_h = np.clip(final_h, 0.0, 1.0)
    return final_h, best_c5, n_points
