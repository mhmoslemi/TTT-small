    dx = 2.0 / n_points
    
    pop_size = 50
    def evaluate_c5(h):
        h_fft = fft(h)
        g_fft = fft(1 - h)
        corr_fft = ifft(h_fft * np.conj(g_fft))
        return np.max(np.real(corr_fft) * dx)
    
    def clamp_rescale(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        scale = (n_points / 2) / s
        return h * scale
    
    def generate_initial():
        pop = [initial_h.copy()]
        x = np.linspace(0, 2, n_points)
        tri = np.minimum(x, 2 - x)
        tri = tri / np.trapz(tri, x)
        pop.append(tri)
        sin = 0.5 * (1 + np.sin(np.pi * x))
        sin = sin / np.trapz(sin, x)
        pop.append(sin)
        uni = np.ones(n_points) * (n_points / 2) / n_points
        pop.append(uni)
        np.random.seed(seed)
        for _ in range(pop_size - 4):
            h = np.random.rand(n_points)
            h = clamp_rescale(h)
            pop.append(h)
        return pop[:pop_size]
    
    population = generate_initial()
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate_c5(h) for h in population]
        indices = np.argsort(fitness)[:int(len(population) * 0.5)]
        selected = [population[i] for i in indices]
        new_pop = []
        while len(new_pop) < pop_size:
            p1, p2 = np.random.choice(selected, 2, replace=False)
            c = np.concatenate([p1[:np.random.randint(1, n_points)], p2[np.random.randint(1, n_points):]])
            m = np.random.rand(n_points) < 0.05
            c[m] = np.random.rand(np.sum(m))
            c = clamp_rescale(c)
            new_pop.append(c)
        population = new_pop
        curr_best = population[np.argmin(fitness)]
        if evaluate_c5(curr_best) < best_c5:
            best_c5 = evaluate_c5(curr_best)
            best_h = curr_best
    
    return best_h, best_c5, n_points
