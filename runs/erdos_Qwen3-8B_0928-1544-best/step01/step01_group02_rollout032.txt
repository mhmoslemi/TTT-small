import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    if n_points == 0:
        return np.array([]), 1.0, 0
    dx = 2.0 / n_points
    initial_guess = np.array(initial_h_values, dtype=np.float64)
    sum_initial = np.sum(initial_guess)

                if i not in indices:

    best_h = initial_guess.copy()

    for guess in initial_guesses:
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            cross_corr = np.correlate(h, 1 - h, mode='full')

        def constraint_sum(h_flat):
            return np.sum(h_flat) - n_points / 2

        cons = [{'type': 'eq', 'fun': constraint_sum}]
        
        import time
