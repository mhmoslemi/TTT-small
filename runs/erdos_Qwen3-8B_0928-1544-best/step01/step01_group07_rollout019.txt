    if initial_h is not None:
        current_sum = np.sum(initial_h)
        scale = target_sum / current_sum
        initial_h = np.clip(initial_h * scale, 0, 1)
        population.append(initial_h)
        for _ in range(population_size - 1):
            pert = np.random.normal(0, 0.05, n_points)
            new_h = initial_h + pert
            new_h = np.clip(new_h, 0, 1)
            current_sum_new = np.sum(new_h)
            scale_new = target_sum / current_sum_new
            new_h = np.clip(new_h * scale_new, 0, 1)
            population.append(new_h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            sum_rest = np.sum(h[:-1])
            last = target_sum - sum_rest
            last = np.clip(last, 0, 1)
            h = np.append(h[:-1], last)
            population.append(h)

    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h) for h in population]
        new_pop = []
        best_idx = np.argmin(fitness)
        new_pop.append(population[best_idx])
        for _ in range(population_size - 1):
            tournament = np.random.choice(range(population_size), 3, replace=False)
            best_t = tournament[np.argmin([fitness[i] for i in tournament])]
            parent1 = population[best_t]
            parent2 = population[np.random.choice(tournament, 1, replace=False)[0]]
            if np.random.rand() < 0.8:
                alpha = np.random.rand()
                child = parent1 * alpha + parent2 * (1 - alpha)
            else:
                child = parent1
            if np.random.rand() < 0.2:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
            current_sum = np.sum(child)
            scale_fac = target_sum / current_sum
            child = np.clip(child * scale_fac, 0, 1)
            new_pop.append(child)
        population = new_pop
        current_fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(current_fitness)
        if current_fitness[best_idx] < best_c5:
            best_c5 = current_fitness[best_idx]
            best_h = population[best_idx]

    return best_h, best_c5, n_points
