def run(seed=42, budget_s=1100, **kwargs):

    import time

    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    n_points = 45 if initial_h is None else len(initial_h)
    if initial_h is None:
        h_values = np.zeros(n_points)
        h_values[:n_points // 2] = 1
        h_values = h_values * (n_points / 2) / n_points
    else:
        h_values = initial_h.copy()

    dx = 2.0 / n_points
    target_sum = n_points / 2
    current_sum = np.sum(h_values)
    scale = target_sum / current_sum if abs(current_sum - target_sum) > 1e-6 else 1.0
    h_values *= scale

    def compute_C5(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    best_h = h_values.copy()
    best_c5 = compute_C5(best_h)
    learning_rate = 0.1
    steps = 0
    no_improvement = 0
    start_time = time.time()
    while steps < 10000 and time.time() - start_time < budget_s:
        cross_corr = np.correlate(h_values, 1 - h_values, mode='full')
        i_max = np.argmax(cross_corr)
        k = i_max - (n_points - 1)
        n_points_h = len(h_values)
        valid_x = np.arange(n_points_h)
        mask = (valid_x + k >= 0) & (valid_x + k < n_points_h)
        grad = np.zeros(n_points_h)
        grad[mask] = 1 - h_values[valid_x[mask] + k]
        # Update h using gradient descent
        h_values -= learning_rate * grad
        # Clip to [0, 1]
        h_values = np.clip(h_values, 0, 1)
        # Scale to meet the sum constraint
        current_sum = np.sum(h_values)
        scale = target_sum / current_sum
        h_values *= scale
