import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population_size = 30
    population = [initial_h]
    for _ in range(population_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        population.append(h)
    population = np.array(population)
    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    # Differential Evolution
    mutation_factor = 0.7
    crossover_prob = 0.8
    max_generations = 200
    stagnation_gen = 50
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([compute_c5(p) for p in population])
        current_best = np.min(fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin(fitness)]
        if gen > 0 and abs(current_best - best_c5) < 1e-8:
            if gen > stagnation_gen:
                break
        new_population = []
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < crossover_prob:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            trial = trial * (n_points / 2) / (trial_sum + 1e-8)
            new_population.append(trial)
        population = np.array(new_population)

    # SLSQP Refinement
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - (n_points / 2)})
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = minimize(compute_c5, best_h, method='SLSQP', bounds=bounds, constraints=cons,

    final_h = result.x if result.success else best_h

    # Fallback to initial solution
    initial_c5 = compute_c5(initial_h)
    if final_c5 > initial_c5:
        final_h = initial_h
        final_c5 = initial_c5

    return (final_h, final_c5, n_points)
