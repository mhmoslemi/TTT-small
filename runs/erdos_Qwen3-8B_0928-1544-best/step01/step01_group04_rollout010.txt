import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    h = np.array(initial_h_values)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    s = np.max(corr) * dx
    lambda_k = np.zeros(2 * n - 1)
    rho = 1.0
    max_iter = 1000
    tol = 1e-4
    best_h = h.copy()
    best_c5 = s
    start_time = time.time()
    
    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr) * dx
        if abs(current_max - s) < tol:
            break
        residuals = corr - s / dx
        lambda_k += rho * residuals
        
        g = np.zeros(n)
        for k in range(len(lambda_k)):
            A = np.zeros(n)
            for i in range(n):
                if i + k < n:
                    A[i] += lambda_k[k] * (1 - h[i + k])
                if i - k >= 0:
                    A[i] += lambda_k[k] * h[i - k]
            g += A
        h = h - 0.01 * g
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        scale = (n / 2) / h_sum if h_sum != 0 else 1.0
        h *= scale
        
        s = current_max
        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break
    
    return h, s, n
