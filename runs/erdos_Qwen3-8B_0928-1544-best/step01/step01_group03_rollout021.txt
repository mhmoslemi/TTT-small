import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points

    best_c5 = float('inf')
    best_h = h_initial
    n_starts = 5
    start_time = time.time()

    for _ in range(n_starts):
        noise_scale = 0.1 * np.sqrt(1.0 / n_points)
        noise = np.random.normal(0, noise_scale, n_points)
        perturbed = np.clip(h_initial + noise, 0, 1)
        scale = (n_points / 2.0) / np.sum(perturbed)
        perturbed *= scale

        def obj(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        def cons(h):
            return np.sum(h) - n_points / 2.0

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        res = minimize(obj, perturbed, method='COBYLA', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': 800, 'rhobeg': 0.1, 'rhoend': 1e-5, 'disp': False})

        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
