from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        n = 70
        initial_h = np.random.rand(n)

    result = minimize(
        fun=objective,
        x0=initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 100, 'ftol': 1e-9}

    best_h = result.x

    for _ in range(5):
