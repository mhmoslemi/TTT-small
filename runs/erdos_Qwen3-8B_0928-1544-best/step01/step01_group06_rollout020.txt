def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

        h = np.clip(h, 0, 1)
        total = np.sum(h)
        h *= (45/2) / total

    best_idx = np.argmin(fitness)
    best_h = h_pop[best_idx].copy()
    best_c5 = fitness[best_idx]
    start_time = time.time()
    while time.time() - start_time < budget_s:
        parents = []

            best_t = t[np.argmin(fitness[t])]
            parents.append(h_pop[best_t].copy())
