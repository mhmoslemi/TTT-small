            best_score = current_scores[best_idx]
            g_best = particles[best_idx].copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= turbulence_freq:
            for i in range(n_pop):
                if np.random.rand() < 0.1:
                    particles[i] = np.clip(np.random.rand(n_points), 0, 1)
                    current_sum = np.sum(particles[i])
                    target_sum = n_points / 2.0
                    if current_sum == 0:
                        particles[i] = np.zeros_like(particles[i]) + target_sum / n_points
                    else:
                        particles[i] *= target_sum / current_sum
            no_improve = 0
    
    h_values = g_best
    c5_bound = best_score
    return h_values, c5_bound, n_points
