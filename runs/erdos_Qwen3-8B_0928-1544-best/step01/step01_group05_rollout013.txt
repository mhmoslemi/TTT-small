import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()

    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx  # Corrected to include dx directly

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})

    result = minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={
            'maxiter': 500,
            'ftol': 1e-8,
            'gtol': 1e-8,
            'eps': 1e-8,
            'disp': False
        }
    )

    best_h = result.x
    best_c5 = result.fun
    return (best_h, best_c5, n_points)
