        return np.max(corr)

    pop_size = 100
    generations = 500
    mutation_rate = 0.1
    tournament_size = 5

            population[i] = h0
        else:

            block_len = np.random.randint(1, n_points // 2)
            start = np.random.randint(0, n_points - block_len)
            h = np.zeros(n_points)
            h[start:start+block_len] = 1.0
            h = np.clip(h, 0, 1)
            # Renormalize
            total = np.sum(h)
            if total > 0:
                h = h * (n_points / 2) / total
            population[i] = h

    best_h = h0
    best_score = fitness(h0)
    best_gen = 0
    no_improvement = 0

    for gen in range(generations):
        # Evaluate fitness
        scores = np.array([fitness(ind) for ind in population])
        best_idx = np.argmin(scores)
        current_best = scores[best_idx]

        # Early stopping if no improvement
        if current_best < best_score:
            best_score = current_best

                break

        # Selection via tournament
        parents = []

            parents.append(population[winner])

        # Crossover
        offspring = []
        for i in range(0, pop_size, 2):
            if np.random.rand() < 0.8:
                p1, p2 = parents[i], parents[i+1]
