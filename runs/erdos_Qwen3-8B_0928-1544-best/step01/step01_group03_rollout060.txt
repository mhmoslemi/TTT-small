import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 95
    dx = 2.0 / n_points
    h_initial = initial_h_values
    
    def objective(h_vector):
        corr = np.correlate(h_vector, 1 - h_vector, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vector):
        return np.sum(h_vector) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    n_starts = 5  # Adjusted for balance between exploration and time
    
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for _ in range(n_starts):
        noise = np.random.normal(0, 0.1 * np.sqrt(1 / n_points), n_points)
        h_perturbed = h_initial + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        if sum_perturbed == 0:
            continue
        scale = (n_points / 2) / sum_perturbed
        h_rescaled = h_perturbed * scale
        
        result = minimize(
            fun=objective,
            x0=h_rescaled,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 800, 'rhobeg': 0.1, 'rhoend': 1e-5}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    return (best_h, best_c5, n_points)
