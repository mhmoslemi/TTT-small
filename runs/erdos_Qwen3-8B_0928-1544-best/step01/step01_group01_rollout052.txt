import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        initial = initial_h_values
    else:
        initial = np.full(80, 0.5)
    n_points = 80
    dx = 2.0 / n_points
    population_size = 50
    max_iterations = 500

    c2 = 2.0

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    particles = []
    velocities = []
    pbest = []
    pbest_fitness = []

    for _ in range(population_size):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h.copy())
        pbest_fitness.append(evaluate(h))

    gbest = particles[np.argmin(pbest_fitness)]
    gbest_fitness = np.min(pbest_fitness)

    for iteration in range(max_iterations):
        for i in range(population_size):
            h = particles[i]
            vel = velocities[i]
            r1, r2 = np.random.rand(2)
            vel = w * vel + c1 * r1 * (pbest[i] - h) + c2 * r2 * (gbest - h)
            new_h = h + vel
            new_h = np.clip(new_h, 0, 1)
            current_sum = np.sum(new_h)
            if current_sum != 0:
                scale = (n_points / 2) / current_sum
                new_h = np.clip(new_h * scale, 0, 1)
            particles[i] = new_h
            velocities[i] = vel
        for i in range(population_size):
            current_fit = evaluate(particles[i])
            if current_fit < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = current_fit
        current_gbest = np.min(pbest_fitness)
        if current_gbest < gbest_fitness:
            gbest = pbest[np.argmin(pbest_fitness)]
            gbest_fitness = current_gbest
        if iteration > 50 and np.abs(gbest_fitness - np.min(pbest_fitness)) < 1e-6:
            break

    h_values = gbest
    c5_bound = gbest_fitness
    return h_values, c5_bound, n_points
