        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        population.append(h_rand)
    
    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    sigma = 0.1
    max_generations = 200
    elapsed = 0
    start_time = time.time()
    
    for gen in range(max_generations):
        fitnesses = np.array([-compute_C5(h) for h in population])
        total_fitness = np.sum(fitnesses)
        probabilities = fitnesses / total_fitness if total_fitness != 0 else np.ones_like(fitnesses) * 1.0 / population_size
        
        new_population = []
        for _ in range(population_size):
            idx1 = np.random.choice(range(population_size), p=probabilities)
            idx2 = np.random.choice(range(population_size), p=probabilities)
            parent1 = population[idx1]
            parent2 = population[idx2]
            
            split = np.random.randint(1, n_points)
            child = np.zeros(n_points)
            child[:split] = parent1[:split]
            child[split:] = parent2[split:]
            scale = (n_points / 2) / np.sum(child)
            child *= scale
            
            noise = np.random.normal(0, sigma, n_points)
            mutant = child + noise
            mutant = np.clip(mutant, 0, 1)
            sum_mutant = np.sum(mutant)
            scale = (n_points / 2) / sum_mutant if sum_mutant != 0 else 1.0
            mutant *= scale
            new_population.append(mutant)
        
        population = new_population
        for candidate in population:
            current_c5 = compute_C5(candidate)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = candidate.copy()
        
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    return best_h, best_c5, n_points
