import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    n_points = 45 if initial_h is None else len(initial_h)
    if initial_h is None:
        h_values = np.zeros(n_points)
        h_values[:n_points // 2] = 1
        h_values = h_values * (n_points / 2) / n_points
    else:
        h_values = initial_h.copy()

    dx = 2.0 / n_points
    target_sum = n_points / 2
    current_sum = np.sum(h_values)
    scale = target_sum / current_sum if abs(current_sum - target_sum) > 1e-6 else 1.0
    h_values *= scale

    def compute_C5_and_k(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(cross_corr)
        k = max_index - (n_points - 1)
        max_corr = cross_corr[max_index] * dx
        return max_corr, k

    def compute_gradient(h, k):
        n = len(h)
        grad = np.zeros(n)
        dx_val = 2.0 / n
        for i in range(n):
            term = 0.0
            if i + k < n:
                term += (1 - h[i + k]) * dx_val
            if i - k >= 0:
                term -= h[i - k] * dx_val
            grad[i] = term
        return grad

    best_h = h_values.copy()
    best_c5 = compute_C5_and_k(h_values)[0]
    learning_rate = 0.1
    steps = 0
    no_improvement = 0
    start_time = time.time()
    while steps < 10000 and time.time() - start_time < budget_s:
        current_c5, k = compute_C5_and_k(h_values)
        grad = compute_gradient(h_values, k)
        h_values += -learning_rate * grad
        h_values = np.clip(h_values, 0, 1)
        current_sum = np.sum(h_values)
        scale = target_sum / current_sum
        h_values *= scale

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > 500:
                learning_rate *= 0.5
                no_improvement = 0

        if steps % 250 == 0:
            noise = np.random.normal(0, 1e-4, n_points)
            h_values += noise
            h_values = np.clip(h_values, 0, 1)
            current_sum = np.sum(h_values)
            scale = target_sum / current_sum
            h_values *= scale
        steps += 1
    return (best_h, best_c5, n_points)
