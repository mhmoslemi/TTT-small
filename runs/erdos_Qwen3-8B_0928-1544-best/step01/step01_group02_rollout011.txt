    n_points = len(h)
    indices = np.random.choice(n_points, size=num_perturb, replace=False)
    h_perturbed = h.copy()
    for i in indices:

        delta = np.random.uniform(-0.1, 0.1)

        h_perturbed[i] = np.clip(h_perturbed[i], 0.0, 1.0)
    # Scale to maintain sum T
    current_sum = np.sum(h_perturbed)
    scale = T / current_sum if current_sum != 0 else 1.0
    h_perturbed *= scale
    return h_perturbed
