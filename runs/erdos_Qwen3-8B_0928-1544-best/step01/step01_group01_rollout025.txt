        result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
        if result.success:
            c5 = result.fun
            if c5 < best_c5:
                best_c5 = c5
                best_h = result.x
        else:
            c5 = obj(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    best_h = np.clip(best_h, 0, 1)
    if np.abs(np.sum(best_h) - n_points/2) > 1e-6:
        best_h = initial_guesses[0]
        best_c5 = obj(best_h)
    return best_h, best_c5, n_points
