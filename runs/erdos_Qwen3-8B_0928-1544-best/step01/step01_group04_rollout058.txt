import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    from scipy.optimize import minimize

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    corr = np.correlate(h_initial, 1 - h_initial, mode='full')
    s = np.max(corr) * dx
    corr_values = corr.copy()

    lambda_k = np.zeros_like(corr_values)
    rho = 1.0
    max_iter = 1000
    tol = 1e-4
    best_h = h_initial.copy()
    best_C5 = s
    start_time = time.time()

    for iter in range(max_iter):
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break

        # Primal update: h
        grad = np.zeros(n_points)
        for k in range(len(corr_values)):
            residual = corr_values[k] - s / dx
            for i in range(n_points):
                if i + k < n_points:
                    grad[i] += residual * (1 - h_initial[i + k])
                if i - k >= 0:
                    grad[i] -= residual * h_initial[i - k]
        h_new = h_initial - 0.1 * grad
        h_new = np.clip(h_new, 0, 1)
        sum_h = np.sum(h_new)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h_new *= scale
        h_initial = h_new

        # Slack update
        corr = np.correlate(h_initial, 1 - h_initial, mode='full')
        new_s = np.max(corr) * dx
        if new_s < s:
            s = new_s
        if abs(new_s - s) < tol:
            break

        # Dual update
        for k in range(len(corr_values)):
            residual = corr_values[k] - s / dx
            lambda_k[k] += rho * residual

        if np.max(np.abs(lambda_k)) < tol:
            break

    final_corr = np.correlate(h_initial, 1 - h_initial, mode='full')
    final_c5 = np.max(final_corr) * dx
    return h_initial, final_c5, n_points
