        np.random.seed(seed + start_idx)  # to have reproducible results
        noise = np.random.normal(0, noise_std, n_points)
        h_perturbed = h_initial + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)

        # define the objective function
        def objective(h_vector):
            corr = np.correlate(h_vector, 1 - h_vector, mode='full')
            return np.max(corr) * dx

        # define constraint
        def constraint_sum(h_vector):

        bounds = [(0.0, 1.0) for _ in range(n_points)]

        # run the optimization
        result = minimize(

            x0=h_perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],

                'maxiter': 200,
                'rhobeg': 0.1,
