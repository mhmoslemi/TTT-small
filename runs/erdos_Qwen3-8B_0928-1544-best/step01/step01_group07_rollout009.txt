def run(seed=42, budget_s=1100, **kwargs):
    # Initial guess
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n_points = len(h)
    else:
        n_points = 70
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimize directly at n=70
    result = minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 1000, 'ftol': 1e-9}
    )
    current_c5 = result.fun
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = result.x.copy()
    elapsed = time.time() - start_time
    
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    # Upscale to n=140 if time permits
    if elapsed < budget_s - 200:
        new_n = 140
        new_h = np.repeat(best_h, 2)
        dx_new = 2.0 / new_n
        result = minimize(
            fun=lambda h_flat: objective(h_flat),
            x0=new_h,
            method='SLSQP',
            bounds=Bounds([0.0]*new_n, [1.0]*new_n),
            constraints={'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat) - new_n / 2},
            tol=1e-8,
            options={'maxiter': 500, 'ftol': 1e-9}
        )
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x.copy()
        elapsed = time.time() - start_time
    
    return best_h, best_c5, len(best_h)
