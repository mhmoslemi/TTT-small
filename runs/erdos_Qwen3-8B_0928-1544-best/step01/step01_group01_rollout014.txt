import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        initial_guess = initial_h_values
    else:
        n_points = 256
        initial_guess = np.zeros(n_points)
        initial_guess[:] = 0.5
    n_points = len(initial_guess)

        'jac': lambda h: np.ones_like(h)
    }]

    bounds = [(0, 1) for _ in range(n_points)]

    result = minimize(
        fun=objective,
        x0=initial_guess,
        method='L-BFGS-B',
        bounds=bounds,

        options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
    )

    if result.success:
        h_values = result.x
        c5_bound = result.fun

        h_values = initial_guess
        c5_bound = objective(h_values)

    h_values = np.clip(h_values, 0, 1)

        c5_bound = objective(h_values)
