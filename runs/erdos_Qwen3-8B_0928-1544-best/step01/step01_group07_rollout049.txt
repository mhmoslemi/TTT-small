        h_ref = np.full(N_ref, 0.5)
    sum_h = np.sum(h_ref)
    if np.abs(sum_h - N_ref / 2) > 1e-6:
        delta = (N_ref / 2 - sum_h) / N_ref
        h_ref += delta * np.ones_like(h_ref)

    res = minimize(objective, h_ref, method='SLSQP', bounds=Bounds([0.]*len(h_ref), [1.]*len(h_ref)), constraints={'type': 'eq', 'fun': constraint_sum})
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    else:
        best_h = h_ref
        best_c5 = objective(h_ref)

    for i in range(1, len(ref_steps)):
        N_new = ref_steps[i]
        N_prev = ref_steps[i-1]
        new_x = np.linspace(0, 2, N_new)
        prev_x = np.linspace(0, 2, N_prev)
        h_upsampled = np.interp(new_x, prev_x, best_h)
        sum_h = np.sum(h_upsampled)
        delta = (N_new / 2 - sum_h) / N_new
        h_upsampled += delta * np.ones_like(h_upsampled)
        elapsed = time() - start_time
        if elapsed > budget_s:
            break
        h_ref = h_upsampled
        res_new = minimize(objective, h_ref, method='SLSQP', bounds=Bounds([0.]*len(h_ref), [1.]*len(h_ref)), constraints={'type': 'eq', 'fun': constraint_sum}, options={'maxiter': 100})
        if res_new.success:
            current_c5 = res_new.fun
            if current_c5 < best_c5:
                best_h = res_new.x
                best_c5 = current_c5
        remaining = budget_s - (time() - start_time)
        if remaining < 0:
            break

    return best_h, best_c5, len(best_h)
