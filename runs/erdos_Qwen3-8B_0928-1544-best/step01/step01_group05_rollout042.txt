from scipy.optimize import minimize
from scipy.signal import correlate

    def objective(h):
        corr = correlate(h, 1.0 - h, mode='full')
        return corr.max()

    def repair_feasibility(h):
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scale = (n_points / 2.0) / total
        h = h * scale
        if np.abs(np.sum(h) - n_points / 2.0) > 1e-6:
            error = np.sum(h) - n_points / 2.0
            if abs(error) > 1e-6:
                h += error * (h / np.max(h) if np.max(h) > 1e-8 else np.ones_like(h))
        return h

    def de_step(population, best_h, seed):
        np.random.seed(seed)
        pop_size = len(population)

            mutant = population[a] + np.random.rand() * (population[b] - population[c])
            trial = np.clip(mutant, 0.0, 1.0)
            trial = repair_feasibility(trial)
