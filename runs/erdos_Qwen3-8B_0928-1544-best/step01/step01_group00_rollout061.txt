import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum

    def compute_c5(h_vals):
        ones_minus_h = 1 - h_vals
        corr = np.correlate(h_vals, ones_minus_h, mode='full')
        dx = 2.0 / n_points
        return np.max(corr * dx)

    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    start_time = time.time()
    T = 1.0
    alpha = 0.999
    max_iter = 100000
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        perturb_indices = np.random.choice(n_points, size=5, replace=False)
        delta = np.random.uniform(-0.05, 0.05, size=n_points)
        delta[perturb_indices] = np.random.uniform(-0.1, 0.1, size=5)
        h_perturb = h + delta
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb_sum = np.sum(h_perturb)
        h_perturb = h_perturb * (n_points / 2) / h_perturb_sum
        new_c5 = compute_c5(h_perturb)
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
