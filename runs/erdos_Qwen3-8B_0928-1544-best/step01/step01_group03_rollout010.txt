import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 95
    dx = 2.0 / n_points
    np.random.seed(seed)
    x_vals = np.linspace(0, 2, n_points)
    
    def generate_individual():
        return (
            np.random.uniform(0, 0.5, 3),
            np.random.randint(1, 5, 3),
            np.random.uniform(0, 2*np.pi, 3)
        )
    
    def evaluate(ind):
        h = np.zeros(n_points)
        for a, f, p in zip(ind[0], ind[1], ind[2]):
            h += a * np.sin(2 * np.pi * f * x_vals + p)
        h += 0.5
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    population = [generate_individual() for _ in range(50)]
    fitness = np.array([evaluate(ind) for ind in population])
    best_idx = np.argmin(fitness)
    best_fit = fitness[best_idx]
    
    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        parents = np.argsort(fitness)[:20]
        new_pop = []
        for _ in range(50):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            a = (population[p1][0] + population[p2][0]) * 0.5
            f = np.round((population[p1][1] + population[p2][1]) * 0.5).astype(int)
            f = np.clip(f, 1, 5)
            p = (population[p1][2] + population[p2][2]) * 0.5
            a += np.random.normal(0, 0.05, 3)
            a = np.clip(a, 0, 0.5)
            p += np.random.normal(0, 0.1, 3)
            p = np.clip(p, 0, 2*np.pi)
            new_pop.append((a, f, p))
        fitness = np.array([evaluate(ind) for ind in new_pop])
        population = new_pop
        best_idx = np.argmin(fitness)
        best_fit = fitness[best_idx]
    
    best_ind = population[best_idx]
    h_vals = np.zeros(n_points)
    for a, f, p in zip(best_ind[0], best_ind[1], best_ind[2]):
        h_vals += a * np.sin(2 * np.pi * f * x_vals + p)
    h_vals += 0.5
    h_vals = np.clip(h_vals, 0, 1)
    return (h_vals, best_fit, n_points)
