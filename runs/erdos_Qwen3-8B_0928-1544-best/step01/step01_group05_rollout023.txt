import numpy as np
from scipy.optimize import differential_evolution
import math

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 50
    dx = 2.0 / n_points

    # Initial guess: all first n_points-1 variables set to 0.5
    initial_guess = np.full(n_points - 1, 0.5)

    def objective(h):
        last = 25.0 - np.sum(h)
        if not (0 <= last <= 1):
            return float('inf')
        h_full = np.concatenate([h, [last]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points - 1)]

    # Adjust differential evolution parameters for better convergence
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=150,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        disp=False
    )

    best_h = result.x
    last = 25.0 - np.sum(best_h)
    best_h_full = np.concatenate([best_h, [last]])
    best_c5 = result.fun

    return (best_h_full, best_c5, n_points)
