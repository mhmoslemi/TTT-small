import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    if n_points == 0:
        return np.array([]), 1.0, 0
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64)
    sum_h = np.sum(h)
    assert np.isclose(sum_h, n_points / 2), "Initial h must have sum = n_points / 2"
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr * dx)
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h
    
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        h_perturb = initial_h_values.copy()
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturb += perturb
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        result = minimize(objective, h_perturb, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-9, 'maxiter': 300})

    cross_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(cross_corr * dx)
