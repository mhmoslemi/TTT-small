import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    from scipy.optimize import minimize
    import time

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    pop_size = 50
    population = [h_initial]
    for _ in range(pop_size - 1):
        h_rand = np.random.rand(n_points)
        h_scaled = h_rand * (n_points / 2) / np.sum(h_rand)
        population.append(h_scaled)

    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    elapsed = 0.0
    start_time = time.time()

    for _ in range(200):
        fitness = [-compute_C5(h) for h in population]
        total_fitness = np.sum(fitness)
        probs = fitness / total_fitness
        parents = np.random.choice(population, size=pop_size, p=probs)
        new_population = []

        for i in range(0, pop_size, 2):
            p1, p2 = parents[i], parents[i+1] if i+1 < pop_size else parents[i]
            if np.random.rand() < 0.8:
                point = np.random.randint(1, n_points)
                c1 = np.concatenate([p1[:point], p2[point:]])
                c2 = np.concatenate([p2[:point], p1[point:]])
                c1 = c1 * (n_points / 2) / np.sum(c1)
                c2 = c2 * (n_points / 2) / np.sum(c2)
            else:
                c1, c2 = p1.copy(), p2.copy()

            if np.random.rand() < 0.3:
                c1 += np.random.normal(0, 0.1, n_points)
                c1 = np.clip(c1, 0, 1)
                c1 = c1 * (n_points / 2) / np.sum(c1)
            if np.random.rand() < 0.3:
                c2 += np.random.normal(0, 0.1, n_points)
                c2 = np.clip(c2, 0, 1)
                c2 = c2 * (n_points / 2) / np.sum(c2)
            new_population.extend([c1, c2])
        population = new_population

        for h in population:
            current_c5 = compute_C5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    return best_h, best_c5, n_points
