            h[-1] = 0.5
        else:
            h[0] = 0.5
        initial_h_values = h
    h = np.array(initial_h_values, dtype=np.float64)
    assert np.isclose(np.sum(h), n_points / 2), "Initial h sum not correct"
    dx = 2.0 / n_points

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr * dx)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'ftol': 1e-9, 'maxiter': 800})
    
    if not result.success:
        return h, np.max(np.correlate(h, 1 - h, mode='full') * dx), n_points
    
    h = result.x
    cross_corr = np.correlate(h, 1 - h, mode='full')
    c5_value = np.max(cross_corr * dx)
    
    return h, c5_value, n_points
