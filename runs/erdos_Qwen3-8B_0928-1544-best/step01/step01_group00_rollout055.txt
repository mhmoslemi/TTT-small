import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    def compute_c5(h_vals):
        ones_minus_h = 1 - h_vals
        corr = np.correlate(h_vals, ones_minus_h, mode='full')
        dx = 2.0 / n_points
        return np.max(corr * dx)

        eps = 1e-5
        grads = np.zeros_like(h_vals)
        orig_c5 = compute_c5(h_vals)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')

        for i in range(len(h_vals)):

            h_perturb[i] += eps

            h_sum_perturb = np.sum(h_perturb)
            h_perturb = h_perturb * (n_points / 2) / h_sum_perturb
            new_c5 = compute_c5(h_perturb)
            grads[i] = (new_c5 - orig_c5) / eps
        return grads
