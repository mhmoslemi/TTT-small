import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 95
    dx = 2.0 / n_points
    h_initial = initial_h_values

    def objective(h_vector):
        corr = np.correlate(h_vector, 1 - h_vector, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vector):
        return np.sum(h_vector) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (h_initial, np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx, n_points)

    h_optimized = result.x
    sum_h = np.sum(h_optimized)
    if abs(sum_h - n_points / 2) > 1e-6:
        h_optimized = h_optimized * (n_points / 2) / sum_h
    c5_bound = result.fun

    return (h_optimized, c5_bound, n_points)
