            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h *= (n_points / 2) / sum_h
            h_pop.append(h)
    
    velocities = [np.random.rand(n_points) * 0.1 for _ in range(n_particles)]
    pbest_pop = h_pop.copy()
    pbest_fitness = [evaluate_erdos_solution(h) for h in h_pop]
    gbest_index = np.argmin(pbest_fitness)
    gbest = pbest_pop[gbest_index]
    gbest_fitness = pbest_fitness[gbest_index]
    best_fitness = gbest_fitness
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * w + c1 * r1 * (pbest_pop[i] - h_pop[i]) + c2 * r2 * (gbest - h_pop[i])
            velocities[i] = np.clip(vel, -0.5, 0.5)
            
            h = h_pop[i] + velocities[i]
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h *= (n_points / 2) / sum_h
            
            h_pop[i] = h.copy()
            current_fitness = evaluate_erdos_solution(h)
            if current_fitness < pbest_fitness[i]:
                pbest_pop[i] = h.copy()
                pbest_fitness[i] = current_fitness
        
        current_gbest_fitness = np.min(pbest_fitness)
        if current_gbest_fitness < gbest_fitness:
            gbest_index = np.argmin(pbest_fitness)
            gbest = pbest_pop[gbest_index]
            gbest_fitness = current_gbest_fitness
            best_fitness = gbest_fitness
    
    h_values = gbest
    c5_bound = best_fitness
    return h_values, c5_bound, n_points
