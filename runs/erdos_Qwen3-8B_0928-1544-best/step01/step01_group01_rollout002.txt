        if no_improvement >= 100:
            break
        # Select parents via tournament
        parents = []
        for _ in range(population_size):
            # Select tournament members
            tournament = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament]
            winner_idx = tournament[np.argmin(tournament_fitness)]
            parents.append(population[winner_idx])

        for i in range(0, population_size, 2):

            parent2 = parents[i+1] if i+1 < population_size else parents[i]

            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                # Normalize to ensure sum is n_points/2
                child1 = child1 / np.sum(child1) * (n_points / 2)
                child2 = child2 / np.sum(child2) * (n_points / 2)
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            # Mutation for child1
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, 0.05, size=n_points)
                child1 += mutation
                child1 = np.clip(child1, 0, 1)
                child1 = child1 / np.sum(child1) * (n_points / 2)
            # Mutation for child2
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, 0.05, size=n_points)
                child2 += mutation
                child2 = np.clip(child2, 0, 1)
                child2 = child2 / np.sum(child2) * (n_points / 2)
            offspring.append(child1)
            offspring.append(child2)
        # Replace worst individuals
        # Evaluate offspring fitness
        offspring_fitness = []
        for h in offspring:
            corr = np.correlate(h, 1 - h, mode='full') * dx
            c5 = np.max(corr)
            offspring_fitness.append(c5)
        # Combine population and offspring
        combined = population + offspring
        # Sort by fitness
        combined_sorted = sorted(zip(combined, offspring_fitness), key=lambda x: x[1])
        # Take the best population_size individuals
        new_population = [x[0] for x in combined_sorted[:population_size]]
        population = new_population

    # After GA loop, evaluate the best individual
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)
    # Ensure constraints are met

    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:

        best_h = initial_h.copy()
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)

    return best_h, best_c5, n_points
