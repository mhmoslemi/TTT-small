import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    # Initialize with provided initial H values or default
    initial_h = initial_h_values.copy() if 'initial_h_values' in globals() else np.zeros(70)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Objective function: compute max overlap integral correctly
    def objective(h):
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode='full')
        lags = np.arange(-n_points + 1, n_points)
        lag_in_range = np.abs(lags * dx) <= 2.0
        valid_corr = corr[lag_in_range]
        max_overlap = np.max(valid_corr) * dx
        return max_overlap
    
    # Constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    # Optimization setup with L-BFGS-B for better convergence
    result = minimize(
        fun=objective,
        x0=initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-6,
        options={'maxiter': 300, 'ftol': 1e-9, 'gtol': 1e-9}
    )
    
    # Return the best solution found
    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
