import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:

        if s == 0:

        h = np.clip(h, 0, 1)

    tournament_size = 3
    mutation_rate = 0.1
    crossover_rate = 0.8

    population = population[:pop_size]
