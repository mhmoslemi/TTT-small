        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Perform optimization
        result = minimize(
            objective,
            new_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-8}
        )
        
        # Check time
        if time.time() - start_time > budget_s:
            break
        
        if result.success:
            h = result.x
            current_c5 = result.fun * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # If optimization fails, retain the previous best_h
            pass
    
    return (best_h, best_c5, n_points)
