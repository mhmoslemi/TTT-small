import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    initial_h = initial_h_values.copy()

        corr = correlate(h, 1 - h, mode='full')

        return max_corr * dx

    CROSSOVER_PROB = 0.8

    STAGNATION = 50

    population = [initial_h.copy()]
    for _ in range(POP_SIZE - 1):
        h = np.random.rand(n_points)
        h = h * 35.0 / np.sum(h)
        h = np.clip(h, 0.0, 1.0)
        population.append(h)

    best_c5 = float('inf')
    best_h = initial_h.copy()
    stagnation_count = 0

    for generation in range(MAX_GENERATIONS):
        current_time = time.time()
        if current_time - start_time > MAX_TIME:
            break
        fitness = [evaluate_fitness(ind) for ind in population]
        best_idx = np.argmin(fitness)
        current_best = fitness[best_idx]
        if abs(current_best - best_c5) < 1e-6:
            stagnation_count += 1
        else:
            best_c5 = current_best
            stagnation_count = 0
        if stagnation_count >= STAGNATION:
            break
        new_population = []
        for i in range(POP_SIZE):
            a, b, c = np.random.choice(population, 3, replace=False)

            trial = np.clip(trial, 0.0, 1.0)
            trial_sum = np.sum(trial)
            trial = trial * (35.0 / trial_sum)

                new_population.append(trial)
            else:
                new_population.append(population[i])
        population = new_population

    fitness_values = [evaluate_fitness(ind) for ind in population]
    best_idx = np.argmin(fitness_values)
    best_h = population[best_idx]
    best_c5 = fitness_values[best_idx]

    def objective_SLSQP(h):

        max_corr = np.max(corr)
