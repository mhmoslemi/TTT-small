import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum

    def compute_c5(h_vals):
        ones_minus_h = 1 - h_vals
        corr = np.correlate(h_vals, ones_minus_h, mode='full')
        dx = 2.0 / n_points
        return np.max(corr * dx)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    T = 1.0
    alpha = 0.995
    max_iter = 20000
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Perturb
        selected = np.random.choice(n_points, size=5, replace=False)
        for i in selected:
            h[i] += np.random.uniform(-0.1, 0.1)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        scaling = (n_points / 2.0) / current_sum
        h = h * scaling
        h = np.clip(h, 0, 1)
        # Evaluate
        new_c5 = compute_c5(h)
        # Accept
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            if np.random.rand() < np.exp(-(new_c5 - best_c5) / T):
                best_c5 = new_c5
                best_h = h.copy()
        T *= alpha
    return (best_h, best_c5, n_points)
