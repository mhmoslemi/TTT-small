                    initial[0] += (n//2 - (n//2 + 1)) / n
                current_h = initial
            else:
                m = len(current_h)
                if n % m != 0:
                    repeat_times = n // m
                    remainder = n % m
                    upsampled = np.repeat(current_h, repeat_times)
                    if remainder > 0:
                        upsampled = np.concatenate([upsampled, current_h[:remainder]])
                    current_h = upsampled
                else:
                    current_h = np.repeat(current_h, n // m)
        
        n_points = n
        dx = 2.0 / n_points
        
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        result = minimize(
            fun=objective,
            x0=current_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 500, 'ftol': 1e-9}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    final_n = len(best_h) if best_h is not None else 14
    return best_h, best_c5, final_n
