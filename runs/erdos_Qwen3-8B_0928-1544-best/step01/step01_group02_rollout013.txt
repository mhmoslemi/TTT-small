import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import random

    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def calculate_C5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr * dx)

    if n_points == 0:
        return np.array([]), 1.0, 0

    M = 20
    population = [np.array(initial_h_values, dtype=np.float64)]
    for _ in range(M - 1):
        h = np.array(initial_h_values) + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h = h * (target_sum / current_sum)
        population.append(h)

    best_h = np.array(initial_h_values)
    best_c5 = calculate_C5(best_h)

    F = 0.8
    CR = 0.9
    max_gens = 200
    start_time = time.time()

    for gen in range(max_gens):
        if time.time() - start_time > budget_s:
            break
        for i in range(M):
            a, b, c = random.sample(range(M), 3)
            h_a = population[a]
            h_b = population[b]
            h_c = population[c]
            mutant = h_a + F * (h_b - h_c)

            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]

            trial = np.clip(trial, 0.0, 1.0)
            current_sum = np.sum(trial)
            if current_sum != 0:
                trial = trial * (target_sum / current_sum)

            c5_trial = calculate_C5(trial)

                population[i] = trial
                if c5_trial < best_c5:
                    best_h = trial.copy()

    return best_h, best_c5, n_points
