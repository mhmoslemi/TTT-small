import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 95
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        perturb = perturb * scale
        
        res = minimize(obj, perturb, method='COBYLA', bounds=bounds, constraints=[{'type': 'eq', 'fun': cons_sum}])
        if res.success:
