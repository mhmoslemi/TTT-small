import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import math

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    if 'initial_h_values' in locals():
        h_initial = np.array(initial_h_values, dtype=float)
    else:
        h_initial = np.full(70, 1.0 / 70)
    best_h = h_initial.copy()
    best_c5 = float('inf')
    best_n = 70

    def evaluate(h):
        corr = correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * 2.0 / len(h)

    def objective(h):
        sum_h = np.sum(h)
        penalty = 1e6 * abs(sum_h - len(h) / 2)
        return evaluate(h) + penalty

    def refine(h):
        bounds = [(0.0, 1.0) for _ in range(len(h))]
        res = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            disp=False,
            seed=seed
        )
        return res.x

    # Phase 1: Initial refinement
    h_refined = refine(h_initial)
    c5 = evaluate(h_refined)
    if c5 < best_c5:
        best_h = h_refined
        best_c5 = c5
        best_n = len(h_refined)
