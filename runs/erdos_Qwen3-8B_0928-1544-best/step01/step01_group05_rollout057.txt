import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    required_sum = n_points / 2.0
    best_c5 = float('inf')
    best_h = initial_h_values.copy()

    def objective_de(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.inf
        scaling = required_sum / current_sum
        h_proj = np.clip(h * scaling, 0, 1)
        corr = correlate(h_proj, 1 - h_proj, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    try:
        result = differential_evolution(
            objective_de, bounds, strategy='best1bin', popsize=30,
            maxiter=100, tol=1e-6, seed=seed)
        best_solution = result.x
        current_c5 = objective_de(best_solution)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_solution
    except:
        pass

    def objective_slsqp(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - required_sum})
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    try:
        result_slsqp = minimize(
            objective_slsqp, best_h, method='SLSQP',
            bounds=bounds_slsqp, constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-6, 'eps': 1e-8, 'disp': False})
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun
    except:
        pass

    return (best_h, best_c5, n_points)
