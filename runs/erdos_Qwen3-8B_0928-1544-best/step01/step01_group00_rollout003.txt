import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    population_size = 50
    elite_count = 2
    mutation_rate = 0.1
    crossover_rate = 0.8
    dx = 2.0 / n_points

    def normalize_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / h_sum

    def compute_c5(h_vals):
        ones_minus_h = 1 - h_vals
        corr = np.correlate(h_vals, ones_minus_h, mode='full')
        return np.max(corr * dx)

    population = []
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values, dtype=float)
        h_initial = normalize_h(h_initial)
        population.append(h_initial)
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = normalize_h(h)
        population.append(h)

    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h) for h in population]
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_c5 = fitness[current_best]
            best_h = population[current_best].copy()
        parents = np.argsort(fitness)[:int(population_size * 0.2)]
        new_population = [population[p] for p in parents[:elite_count]]
        while len(new_population) < population_size:
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = population[p1].copy()
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points)
                child[split:] = population[p2][split:]
            if np.random.rand() < mutation_rate:
                mutate_count = max(1, int(n_points * 0.1))
                indices = np.random.choice(n_points, mutate_count, replace=False)
                child[indices] = np.random.uniform(0, 1)
            child = normalize_h(child)
            new_population.append(child)
        population = new_population

    return (best_h, best_c5, n_points)
