    pop_size = 50
    pop = [initial_h]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        pop.append(h)
    
    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr * (2.0 / n_points))
    
    elite_size = 2
    crossover_rate = 0.8
    max_gen = 0
    
    best_h = pop[0].copy()
    best_c5 = compute_c5(pop[0])
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        c5s = [compute_c5(h) for h in pop]
        c5s_idx = np.argsort(c5s)
        sorted_pop = [pop[i] for i in c5s_idx]
        elites = sorted_pop[:elite_size]
        remaining = sorted_pop[elite_size:]
        
        new_candidates = []
        for _ in range(pop_size - elite_size):
            if np.random.rand() < crossover_rate:
                p1 = np.random.choice(remaining)
                p2 = np.random.choice(remaining)
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:split], p2[split:]])
            else:
                p = np.random.choice(remaining)
                child = p.copy()
                idx = np.random.choice(n_points, int(n_points * 0.1), replace=False)
                child[idx] += np.random.uniform(-0.1, 0.1, size=len(idx))
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            new_candidates.append(child)
        
        pop = elites + new_candidates
        current_best = np.argmin(c5s)
        if c5s[current_best] < best_c5:
            best_c5 = c5s[current_best]
            best_h = pop[current_best].copy()
    
    return (best_h, best_c5, n_points)
