        dx = 2.0 / n_points
        return np.max(corr * dx)
    
    current_h = h.copy()
    best_h = current_h.copy()
    best_c5 = compute_c5(current_h)
    start_time = time.time()
    T = 0.5
    alpha = 0.995
    max_iter = 200000
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        num_perturb = max(1, int(0.1 * n_points))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        new_h = current_h.copy()
        delta_perturb = T * 0.1
        for i in indices:
            perturb = np.random.choice([-delta_perturb, delta_perturb])
            new_h[i] += perturb
            new_h[i] = np.clip(new_h[i], 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum == 0:
            new_h = new_h * (n_points / 2) / (1e-10)
        else:
            new_h = new_h * (n_points / 2) / new_h_sum
        
        new_c5 = compute_c5(new_h)
        current_c5 = compute_c5(current_h)
        delta_c5 = new_c5 - current_c5
        
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            current_h = new_h.copy()
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = current_h.copy()
        
        T *= alpha
    
    return (best_h, best_c5, n_points)
