            corr = np.correlate(best_h, 1 - best_h, mode='full')
            overlap = np.max(corr) * dx
            overlaps.append(overlap)
        max_overlap = np.max(overlaps)
        chosen_k = sampled_shifts[np.argmax(overlaps)]

        # Compute gradient of max_overlap for chosen_k using finite differences
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += 1e-5
            h_plus = np.clip(h_plus, 0, 1)
            sum_plus = np.sum(h_plus)
            if sum_plus != 0:
                h_plus = h_plus * (n_points / 2) / sum_plus
            overlap_plus = compute_overlap(h_plus, chosen_k)
            h_minus = best_h.copy()
            h_minus[i] -= 1e-5
            h_minus = np.clip(h_minus, 0, 1)
            sum_minus = np.sum(h_minus)
            if sum_minus != 0:
                h_minus = h_minus * (n_points / 2) / sum_minus
            overlap_minus = compute_overlap(h_minus, chosen_k)
            grad[i] = (overlap_plus - overlap_minus) / (2 * 1e-5)

        # Update h with gradient descent
        h_update = best_h - learning_rate * grad
        h_update = np.clip(h_update, 0, 1)
        sum_update = np.sum(h_update)
        if sum_update != 0:
            h_update = h_update * (n_points / 2) / sum_update

        # Check if update improves the solution
        current_c5 = compute_C5(h_update)
        if current_c5 < best_c5:
            best_h = h_update.copy()
            best_c5 = current_c5
            print(f"New best C5: {best_c5}")
        else:
            # Early stopping if improvement is minimal
            if np.abs(current_c5 - best_c5) < early_stopping_threshold:
                break

        # Check time budget
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break

    return best_h, best_c5, n_points
