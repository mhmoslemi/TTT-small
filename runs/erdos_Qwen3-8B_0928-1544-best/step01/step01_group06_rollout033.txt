def run(seed=42, budget_s=1100, **kwargs):

    else:
        h_values = initial_h.copy()

    dx = 2.0 / n_points
    target_sum = n_points / 2
    current_sum = np.sum(h_values)
    scale = target_sum / current_sum if abs(current_sum - target_sum) > 1e-6 else 1.0
    h_values *= scale

    def compute_C5(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    def compute_gradient(h):
        n = len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)

            h_plus[i] += 1e-4
            h_minus = h.copy()
            h_minus[i] -= 1e-4

            grad[i] = g
