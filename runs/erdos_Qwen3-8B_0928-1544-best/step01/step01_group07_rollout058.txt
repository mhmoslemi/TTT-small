    best_c5 = objective(initial_h)

    for gen in range(generations):
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx]

        parents = []
        for _ in range(population_size // 2):
            tournament = np.random.choice(range(population_size), size=3, replace=False)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            parents.append(population[winner])

        children = []
        for i in range(len(parents) // 2):
            p1, p2 = parents[2*i], parents[2*i+1]
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, 0.1, size=n_points)
                child += mutation
                child = np.clip(child, 0.0, 1.0)
            child = adjust_sum(child, target_sum)
            children.append(child)

        population = children

    return (best_h, best_c5, n_points)
