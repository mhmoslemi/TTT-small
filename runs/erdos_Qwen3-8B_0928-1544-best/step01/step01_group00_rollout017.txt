    t = cp.Variable()

    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]

    for k in range(2 * n_points - 1):
        constraint_sum = 0.0
        for i in range(n_points):
            j = i + k
            if j < n_points:
                constraint_sum += h_var[i] * (1 - h_var[j])
        constraints.append(constraint_sum * dx <= t)

    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except:
        return (h, np.inf, n_points)

    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INFEASIBLE]:
        return (h, np.inf, n_points)

    h_opt = h_var.value
    c5 = t.value
    return (h_opt, c5, n_points)
