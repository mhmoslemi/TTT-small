import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        h_initial = np.zeros(n_points)
        h_initial[:int(n_points/2)] = 1.0

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        penalty = 1e6 * abs(np.sum(h) - target_sum)
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    result = differential_evolution(obj, bounds, strategy='best1bin', maxiter=50, popsize=50, tol=1e-6, mutation=(0.5, 1.0), recombination=0.7, seed=seed, disp=False)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        if 'initial_h_values' in globals():
            c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx
            return (h_initial, c5, n_points)
        return (h_initial, 1.0, n_points)
    h_opt = result.x
    final_c5 = result.fun - 1e6 * abs(np.sum(h_opt) - target_sum)
    return (h_opt, final_c5, n_points)
