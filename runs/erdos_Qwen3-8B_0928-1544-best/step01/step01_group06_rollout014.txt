def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    n_points = 45 if initial_h is None else len(initial_h)
    if initial_h is None:
        h_values = np.zeros(n_points)
        h_values[:n_points // 2] = 1
        h_values = h_values * (n_points / 2) / n_points
    else:
        h_values = initial_h.copy()

    dx = 2.0 / n_points
    target_sum = n_points / 2
    current_sum = np.sum(h_values)
    scale = target_sum / current_sum if abs(current_sum - target_sum) > 1e-6 else 1.0
    h_values *= scale

    def compute_C5(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        cross_corr_valid = cross_corr[n_points-1 : 2*n_points-1]
        return np.max(cross_corr_valid) * dx

    def compute_gradient(h):
        n = len(h)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        cross_corr_valid = cross_corr[n-1 : 2*n-1]
        k_max = np.argmax(cross_corr_valid)
        mask = (np.arange(n) + k_max) < n
        grad = np.zeros(n)
        grad[mask] = -(1 - h[np.arange(n) + k_max][mask])
        return grad

    best_h = h_values.copy()
    best_c5 = compute_C5(best_h)
    learning_rate = 0.1
    steps = 0
    no_improvement = 0
    start_time = time.time()
    while steps < 10000 and time.time() - start_time < budget_s:
        grad = compute_gradient(h_values)
        h_values += -learning_rate * grad
        h_values = np.clip(h_values, 0, 1)
        current_sum = np.sum(h_values)
        scale = target_sum / current_sum
        h_values *= scale

        current_c5 = compute_C5(h_values)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > 500:
                learning_rate *= 0.5
                no_improvement = 0

        if steps % 250 == 0:
            noise = np.random.normal(0, 1e-3, n_points)
            h_values += noise

            current_sum = np.sum(h_values)
            scale = target_sum / current_sum
            h_values *= scale
        steps += 1
    return (best_h, best_c5, n_points)
