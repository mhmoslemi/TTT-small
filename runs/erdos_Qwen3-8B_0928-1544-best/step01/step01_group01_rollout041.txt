        h_resampled = np.interp(x_new, x_original, initial_h_values)
        initial_guesses.append(h_resampled)
        for _ in range(19):
            start = np.random.randint(0, 129)
            h = np.zeros(n_points)
            h[start:start+128] = 1.0
            initial_guesses.append(h)
    else:
        for _ in range(20):
            start = np.random.randint(0, 129)
            h = np.zeros(n_points)
            h[start:start+128] = 1.0
            initial_guesses.append(h)
    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = initial_guesses[0]
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2, 'jac': np.ones(n_points)}]
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    for guess in initial_guesses:
        result = minimize(
            fun=objective, x0=guess, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            current_c5 = objective(guess)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = guess
    result_final = minimize(
        fun=objective, x0=best_h, method='SLSQP',
        bounds=bounds, constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
    )
    if result_final.success:
        best_c5 = result_final.fun
        best_h = result_final.x
    best_h = np.clip(best_h, 0, 1)
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = initial_guesses[0]
        best_c5 = objective(best_h)
    return best_h, best_c5, n_points
