        A += np.random.normal(0, 0.05, M)
        A = np.clip(A, 0, 0.5)
        f += np.random.normal(0, 0.1, M)
        f = np.clip(f, 0.5, 2.0)
        phi += np.random.normal(0, 0.1, M)
        phi = np.clip(phi, 0, 2*np.pi)
        return (A, f, phi)

    pop = [gen_ind() for _ in range(pop_size)]
    best_h = None
    best_c5 = float('inf')
    start = time.time()

    for _ in range(gens):
        if time.time() - start > budget_s:
            break
        fits = [eval_ind(ind) for ind in pop]
        fit_pop = sorted(zip(fits, pop), key=lambda x: x[0])
        best_f, best_ind = fit_pop[0]
        if best_f < best_c5:
            best_c5 = best_f
            best_h = h_func(*best_ind)
        new_pop = []
        for _ in range(pop_size):
            p1, p2 = np.random.choice(pop, 2, replace=False)
            child = crossover(p1, p2)
            if np.random.rand() < mut_rate:
                child = mutate(child)
            new_pop.append(child)
        pop = new_pop

    if best_h is None:
        best_ind = pop[0]
        best_h = h_func(*best_ind)
        best_c5 = eval_ind(best_ind)
    return best_h, best_c5, n_points
