import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    h = h * (n_points / 2) / h_sum

    def compute_c5(h_vals):
        ones_minus_h = 1 - h_vals
        corr = np.correlate(h_vals, ones_minus_h, mode='full')
        dx = 2.0 / n_points
        return np.max(corr * dx)

    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    best_c5 = current_c5
    best_h = current_h.copy()
    start_time = time.time()
    current_temp = 1.0
    alpha = 0.995
    max_iter = 100000

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        indices = np.random.choice(n_points, size=5, replace=False)
        delta = np.random.uniform(-0.15, 0.15, size=len(indices))
        h_new = current_h.copy()
        h_new[indices] += delta
        h_new = np.clip(h_new, 0, 1)
        total = np.sum(h_new)
