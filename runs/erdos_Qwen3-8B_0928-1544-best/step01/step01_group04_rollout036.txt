import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    pop_size = 50
    mutation_rate = 0.3
    max_generations = 200

    population = [h_initial.copy()]
    for _ in range(pop_size - 1):
        h = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h)
        scale = (n_points / 2.0) / sum_h if sum_h != 0 else 1.0
        h *= scale
        population.append(h)

    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    start_time = time.time()

    for gen in range(max_generations):
        fitness = np.array([-compute_C5(h) for h in population])
        total_fitness = np.sum(fitness)
        probabilities = fitness / total_fitness
        parent_idx = np.random.choice(len(population), size=2, p=probabilities)
        parent1, parent2 = population[parent_idx[0]], population[parent_idx[1]]

        point = np.random.randint(0, n_points)
        child = np.zeros(n_points)
        child[:point] = parent1[:point]
        child[point:] = parent2[point:]
        sum_child = np.sum(child)
        scale = (n_points / 2.0) / sum_child if sum_child != 0 else 1.0
        child *= scale

        if np.random.rand() < mutation_rate:
            noise = np.random.normal(0, 0.1, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            sum_child = np.sum(child)
            scale = (n_points / 2.0) / sum_child if sum_child != 0 else 1.0
            child *= scale

        min_idx = np.argmin(fitness)
        population[min_idx] = child

        current_c5 = compute_C5(child)
        if current_c5 < best_c5:
            best_h = child.copy()
            best_c5 = current_c5

        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    return best_h, best_c5, n_points
