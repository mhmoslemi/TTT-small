        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr) * dx

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    while time.time() - start_time < budget_s:
        fitness = [compute_C5(h) for h in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = np.copy(population[best_idx])
        new_pop = []
        for _ in range(pop_size):
            t1 = np.random.choice(population, size=5, replace=False)
            p1 = t1[np.argmin([compute_C5(h) for h in t1])]
            t2 = np.random.choice(population, size=5, replace=False)
            p2 = t2[np.argmin([compute_C5(h) for h in t2])]
            split = np.random.randint(1, n_points - 1)
            child = np.concatenate([p1[:split], p2[split:]])
            if np.random.rand() < 0.1:
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child[i] += np.random.normal(0, 0.02)
                        child[i] = np.clip(child[i], 0, 1)
            sum_child = np.sum(child)
            child = child * (n_points / 2) / sum_child
            new_pop.append(child)
        population = new_pop

    return (best_h, best_c5, n_points)
