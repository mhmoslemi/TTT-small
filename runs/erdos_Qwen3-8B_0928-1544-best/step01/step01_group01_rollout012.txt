                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        else:
            h = np.zeros(n_points)
            h[:n_points//2] = 1
            sum_h = n_points // 2
            scale = (n_points / 2) / sum_h
            h = h * scale
            h = np.clip(h, 0, 1)
        population.append(h)
        velocities.append(np.random.uniform(-velocity_scale, velocity_scale, n_points))

    personal_best = [np.copy(p) for p in population]
    global_best = np.copy(population[0])
    best_fitness = float('inf')

    for _ in range(max_iter):
        for i in range(population_size):
            pos = population[i]
            vel = velocities[i]
            r1, r2 = random.random(), random.random()
            vel = w * vel + c1 * r1 * (personal_best[i] - pos) + c2 * r2 * (global_best - pos)
            vel = np.clip(vel, -max_velocity, max_velocity)
            new_pos = pos + vel
            new_pos = np.clip(new_pos, 0, 1)
            sum_new = np.sum(new_pos)
            if sum_new == 0:
                new_pos = np.zeros_like(new_pos)
            else:
                scale = (n_points / 2) / sum_new
                new_pos = new_pos * scale
            population[i], velocities[i] = new_pos, vel

        for i in range(population_size):
            pos = population[i]
            corr = np.correlate(pos, 1 - pos, mode='full') * dx
            current_fitness = np.max(corr)
            if current_fitness < np.max(np.correlate(personal_best[i], 1 - personal_best[i], mode='full') * dx):
                personal_best[i] = np.copy(pos)
                if current_fitness < best_fitness:
                    best_fitness = current_fitness
                    global_best = np.copy(pos)

        if iter > 50 and best_fitness < 1e-6:
            break

    return global_best, best_fitness, n_points
