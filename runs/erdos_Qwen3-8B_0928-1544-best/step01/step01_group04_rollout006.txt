    # Bounds: each h[i] must be between 0 and 1
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial best solution
    best_h = h_initial.copy()
    best_c5 = compute_C5(best_h)
    elapsed_time = 0.0
    start_time = time.time()

    # Optimization parameters
    num_restarts = 20
    max_iter_per_restart = 150
    perturbation_scale = 0.1

    # Generate multiple perturbations and optimize
    for _ in range(num_restarts):
        # Perturb initial guess
        perturbation = np.random.normal(0, perturbation_scale, n_points)
        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        scale = (n_points / 2) / sum_perturbed if sum_perturbed != 0 else 1.0
        h_perturbed *= scale

        # Optimize from perturbed guess
        result = minimize(
            compute_C5,
            h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': max_iter_per_restart, 'ftol': 1e-6, 'gtol': 1e-6}
        )

        if result.success:
            current_h = result.x
            current_c5 = compute_C5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                print(f"New best C5: {best_c5}")
                h_initial = current_h  # Update seed for next restart

        # Check time budget
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break

    return best_h, best_c5, n_points
