from scipy.optimize import minimize
from scipy.signal import correlate

    def objective(h):

        max_corr = np.max(corr)
        return max_corr

    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    
    result = minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-9}
    )
    
    # Refinement pass
    best_h = result.x
    best_c5 = result.fun * dx
    for _ in range(2):
        # Refined optimization
        result = minimize(

            options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-9}
        )
