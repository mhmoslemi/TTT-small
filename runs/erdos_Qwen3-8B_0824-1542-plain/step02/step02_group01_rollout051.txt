import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported helper functions
# evaluate_erdos_solution(h_values) computes C5
# initial_h_values is a starting point with n_points samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Starting with a slightly larger number of points for better resolution
    n_points = 100
    h_initial = initial_h_values
    
    # Function to evaluate the C5 value
    def evaluate_C5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        c5 = max_corr * dx
        return c5
    
    # Function to compute the integral constraint
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization function with SLSQP
    def optimize_h(h_start):
        result = optimize.minimize(
            evaluate_C5,
            h_start,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": constraint_sum}],
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 50}
        )
        return result.x, evaluate_C5(result.x)
    
    # Initial optimization
    best_h, best_c5 = optimize_h(h_initial)
    
    # Try multiple different initializations
    for _ in range(3):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = np.clip(best_h + perturbation, 0.0, 1.0)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        h_perturbed, c5_perturbed = optimize_h(h_perturbed)
        if c5_perturbed < best_c5:
            best_h, best_c5 = h_perturbed, c5_perturbed
    
    # Try increasing the number of points if time permits
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.5:
        n_points_new = int(1.2 * n_points)
        dx_new = 2.0 / n_points_new
        
        # Resample to new grid
        x = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, n_points_new)
        h_new = np.interp(x_new, x, best_h)
        
        # Enforce sum constraint
        h_new = h_new * (n_points_new / 2) / np.sum(h_new)
        
        # Evaluate and optimize with new points
        def evaluate_C5_new(h):
            corr = np.correlate(h, 1 - h, mode="full")
            max_corr = np.max(corr)
            c5 = max_corr * dx_new
            return c5
        
        result_new = optimize.minimize(
            evaluate_C5_new,
            h_new,
            bounds=[(0.0, 1.0) for _ in range(n_points_new)],
            constraints=[{"type": "eq", "fun": lambda h: np.sum(h) - n_points_new / 2}],
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 50}
        )
        
        best_h = result_new.x
        best_c5 = evaluate_C5_new(best_h)
        n_points = n_points_new
    
    return best_h, best_c5, n_points
