import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    best_c5 = 1.0
    best_h = initial_h.copy()
    best_n_points = n_points
    
    def evaluate_c5(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Candidate number of points
    n_points_list = [50, 70, 100, 120, 150, 200]
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        
        # Create an initial guess with piecewise constant segments
        # Starting with a simple oscillatory pattern
        h_initial = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                h_initial[i] = 1.0
            elif i % 3 == 1:
                h_initial[i] = 0.5
            else:
                h_initial[i] = 0.0
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Bounds for h
        bounds = Bounds(0, 1)
        
        # Objective function
        def objective_function(x):
            return evaluate_c5(x)
        
        # Constraint for integral
        def constraint_sum(x):
            return np.sum(x) - n_points / 2
        
        # Optimization settings
        start_time = time.time()
        result = minimize(
            objective_function,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'disp': False}
        )
        elapsed_time = time.time() - start_time
        
        if elapsed_time < budget_s:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
                best_n_points = n_points
        else:
            # If we run out of time, just use the best so far
            break
    
    # Try differential evolution for global optimization
    if budget_s > 10:
        start_time = time.time()
        # Define bounds for each h[i]
        bounds_de = [(0, 1) for _ in range(best_n_points)]
        
        # Define objective function for DE
        def objective_de(x):
            return evaluate_c5(x)
        
        # Optimization
        result_de = differential_evolution(
            objective_de,
            bounds_de,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.6,
            disp=False,
            polish=True,
            vectorized=True
        )
        
        elapsed_time = time.time() - start_time
        if elapsed_time < budget_s:
            current_c5 = result_de.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_de.x
                best_n_points = len(bounds_de)
    
    return (best_h, best_c5, best_n_points)
