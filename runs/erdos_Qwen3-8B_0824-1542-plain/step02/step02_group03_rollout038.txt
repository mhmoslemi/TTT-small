import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with a lower number of points for faster initial exploration
    n_points = 100
    dx = 2.0 / n_points
    
    # Use the initial_h_values if available to start the search
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    else:
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h):
        h = h.reshape(n_points)
        return np.sum(h) - (n_points / 2)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)}
    
    # Optimization with more iterations and tighter tolerances
    start_time = time.time()
    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, 
                      options={'ftol': 1e-9, 'maxiter': 200})
    elapsed_time = time.time() - start_time
    
    # If time is exceeded, return early
    if elapsed_time > budget_s:
        return (initial_h, 0.38092, n_points)
    
    h_values = result.x
    c5_bound = objective(h_values)
    return (h_values, c5_bound, n_points)
