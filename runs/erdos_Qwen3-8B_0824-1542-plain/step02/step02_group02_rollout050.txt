import numpy as np

from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=60, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    best_c5 = 1.0
    best_h = initial_h.copy()

    def evaluate_c5(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Try varying numbers of points
    n_points_list = [50, 70, 100, 120, 150, 200]
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        # Initialize with a combination of sinusoidal and piecewise constant patterns
        # Sinusoidal pattern with normalization
        h = np.sin(np.linspace(0, np.pi, n_points)) ** 2
        h = (h - np.min(h)) / (np.max(h) - np.min(h))
        # Piecewise constant pattern to create sharp transitions
        h = np.concatenate([h[:n_points//3], np.ones(n_points//3), h[n_points//3:2*n_points//3], np.zeros(n_points//3), h[2*n_points//3:]])
        h = h * (n_points / 2) / np.sum(h)
        
        # Bounds for h
        bounds = Bounds(0, 1)
        
        def objective_function(h_values):
            return evaluate_c5(h_values)
        
        def constraint_sum(h_values):
            return np.sum(h_values) - n_points / 2
        
        # Optimization settings
        start_time = time.time()
        result = minimize(
            objective_function,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'disp': False}
        )
        elapsed_time = time.time() - start_time
        
        if elapsed_time < budget_s:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
                best_n_points = n_points
        else:
            # If we run out of time, just use the best so far
            break
    
    return (best_h, best_c5, best_n_points)
