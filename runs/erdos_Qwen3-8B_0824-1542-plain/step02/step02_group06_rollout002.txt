    def objective_function(h_values):
        return evaluate_c5(h_values)

    best_c5 = evaluate_c5(initial_h)
    best_h = initial_h.copy()
    best_n_points = n_points
    time_left = budget_s

    for n_points in [100, 150, 200, 250, 300, 350, 400]:

        h = np.zeros(n_points)
        # Fill with 0.5 in the middle to avoid overlap
        h[n_points // 2 - 5 : n_points // 2 + 5] = 0.5
        # Create a more complex initial guess with random patterns
        for i in range(n_points):
            if np.random.rand() < 0.3:
                h[i] = np.random.uniform(0.2, 0.8)
        # Normalize to ensure integral is 1
        h = h * (n_points / 2) / np.sum(h)

        bounds = Bounds(0, 1)

        start_time = time.time()
        result = minimize(
            objective_function,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},

        )

            current_c5 = result.fun
            if current_c5 < best_c5:

                best_h = result.x.copy()

            break

    for _ in range(3):

        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        # Normalize to ensure integral is 1
        perturbed_h = perturbed_h * (best_n_points / 2) / np.sum(perturbed_h)
