import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize, Bounds
from scipy.optimize import Bounds, LinearConstraint

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Start with 200 points for a balance between accuracy and speed
    dx = 2.0 / n_points
    
    # Use the initial_h_values if available to start the search
    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    else:
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h):
        h = h.reshape(n_points)
        return np.sum(h) - (n_points / 2)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)}
    
    # Optimization with more iterations and tighter tolerances
    start_time = time.time()
    
    # Use a hybrid approach with SLSQP and then a local search for fine-tuning
    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, 
                      options={'ftol': 1e-9, 'maxiter': 200})
    elapsed_time = time.time() - start_time
    
    # If time is exceeded, return early
    if elapsed_time > budget_s:
        return (initial_h, 0.38092, n_points)
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Perform a local search with a different method for refinement
    result_local = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons,
                            options={'ftol': 1e-9, 'maxiter': 100})
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        return (best_h, best_c5, n_points)
    
    refined_h = result_local.x
    refined_c5 = objective(refined_h)
    
    # Check if the refined solution is better
    if refined_c5 < best_c5:
        best_h = refined_h
        best_c5 = refined_c5
    
    return (best_h, best_c5, n_points)
