import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from scipy.sparse import csr_matrix
from scipy.sparse.csgraph import min_weight_full_bipartite_matching
from scipy.optimize import LinearConstraint, Bounds

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reduced number of points for faster optimization
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    # Initial construction: evenly spaced pulses
    h[::n_points//5] = 1.0
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h *= scale
    h = np.clip(h, 0, 1)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        # Compute all shifts
        max_shift = int(np.ceil(2.0 / dx))
        corr = np.zeros(max_shift)
        for k in range(1, max_shift):
            # h shifted by k*dx
            h_shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - h_shifted))
            corr[k-1] = overlap
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h):
        h = h.reshape(n_points)
        return np.sum(h) - (n_points / 2)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)}
    
    # Optimization settings
    options = {
        'ftol': 1e-9,
        'maxiter': 100,
        'disp': False
    }
    
    # Use a hybrid approach: first with SLSQP, then with Nelder-Mead
    start_time = time.time()
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options=options)
    elapsed_time = time.time() - start_time
    
    if elapsed_time > budget_s:
        return (h, 0.38092, n_points)
    
    h_values = result.x
    # Refine with Nelder-Mead
    result = minimize(objective, h_values, method='Nelder-Mead', bounds=bounds, constraints=cons, options=options)
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        return (h_values, objective(h_values), n_points)
    
    h_values = result.x
    c5_bound = objective(h_values)
    
    # Final refinement with simulated annealing
    def objective_sa(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        # Compute all shifts
        max_shift = int(np.ceil(2.0 / dx))
        corr = np.zeros(max_shift)
        for k in range(1, max_shift):
            # h shifted by k*dx
            h_shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - h_shifted))
            corr[k-1] = overlap
        c5 = np.max(corr) * dx
        return c5
    
    # Simulated annealing
    def simulated_annealing(h_init, T=100.0, cooling_rate=0.995, max_iter=500):
        h = h_init.copy()
        best_h = h.copy()
        best_c5 = objective_sa(h)
        current_c5 = best_c5
        for step in range(max_iter):
            T *= cooling_rate
            # Perturb the solution
            h_perturbed = h.copy()
            i = np.random.randint(n_points)
            delta = np.random.uniform(-0.1, 0.1)
            h_perturbed[i] = np.clip(h_perturbed[i] + delta, 0, 1)
            # Check constraint
            if np.abs(np.sum(h_perturbed) - (n_points / 2)) > 1e-6:
                continue
            current_c5 = objective_sa(h_perturbed)
            # Acceptance probability
            if current_c5 < best_c5 or np.random.rand() < np.exp((current_c5 - best_c5) / T):
                h = h_perturbed.copy()
                best_c5 = current_c5
        return h, best_c5
    
    h_values, c5_bound = simulated_annealing(h_values)
    return (h_values, c5_bound, n_points)
