import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize, Bounds
import copy

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with a reasonable number of points
    n_points = 200
    dx = 2.0 / n_points
    
    # Use the initial_h_values if available to start the search
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    else:
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h):
        h = h.reshape(n_points)
        return np.sum(h) - (n_points / 2)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)}
    
    # Optimization with more iterations and tighter tolerances
    start_time = time.time()
    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, 
                      options={'ftol': 1e-8, 'maxiter': 200})
    elapsed_time = time.time() - start_time
    
    # If time is exceeded, return early
    if elapsed_time > budget_s:
        return (initial_h, 0.38092, n_points)
    
    h_values = result.x
    c5_bound = objective(h_values)
    
    # Try refining the solution with a different starting point
    # Create a perturbed version of the solution
    perturbed_h = h_values.copy()
    perturbed_h += np.random.normal(0, 0.01, size=n_points)
    perturbed_h = np.clip(perturbed_h, 0, 1)
    perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
    
    # Run a second optimization
    result_perturbed = minimize(objective, perturbed_h, method='SLSQP', bounds=bounds, constraints=cons, 
                                options={'ftol': 1e-8, 'maxiter': 200})
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        return (h_values, c5_bound, n_points)
    
    h_values_perturbed = result_perturbed.x
    c5_bound_perturbed = objective(h_values_perturbed)
    
    # Choose the better solution
    if c5_bound_perturbed < c5_bound:
        h_values = h_values_perturbed
        c5_bound = c5_bound_perturbed
    
    # Try with more points if time allows
    if elapsed_time < budget_s - 10:
        n_points = 300
        dx = 2.0 / n_points
        initial_h = h_values.copy()
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        
        def objective_high_res(h):
            h = h.reshape(n_points)
            h = np.clip(h, 0, 1)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        def constraint_high_res(h):
            h = h.reshape(n_points)
            return np.sum(h) - (n_points / 2)
        
        bounds_high_res = [(0, 1) for _ in range(n_points)]
        cons_high_res = {'type': 'eq', 'fun': constraint_high_res, 'jac': lambda h: np.ones(n_points)}
        
        result_high_res = minimize(objective_high_res, initial_h, method='SLSQP', bounds=bounds_high_res, constraints=cons_high_res, 
                                   options={'ftol': 1e-8, 'maxiter': 200})
        elapsed_time = time.time() - start_time
        if elapsed_time < budget_s:
            h_values = result_high_res.x
            c5_bound = objective_high_res(h_values)
    
    return (h_values, c5_bound, n_points)
