import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with a moderate number of points for efficiency
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize h with a structured starting point to avoid local optima
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        initial_h = np.clip(initial_h, 0, 1)
    else:
        # Create a simple starting point
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                initial_h[i] = 1.0
            elif i % 10 == 1 or i % 10 == 2:
                initial_h[i] = 0.5
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    def objective(h_values):
        h = h_values.reshape(n_points)
        h = np.clip(h, 0, 1)
        # Compute the overlap integral for all shifts
        corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h_values):
        h = h_values.reshape(n_points)
        return np.sum(h) - (n_points / 2)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)}
    
    # Optimization settings
    options = {
        'ftol': 1e-9,
        'gtol': 1e-9,
        'maxiter': 200,
        'disp': False,
        'iprint': -1
    }
    
    start_time = time.time()
    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, 
                      options=options)
    elapsed_time = time.time() - start_time
    
    # Return early if time is exceeded
    if elapsed_time > budget_s:
        return (initial_h, 0.38092, n_points)
    
    h_values = result.x
    c5_bound = objective(h_values)
    return (h_values, c5_bound, n_points)
