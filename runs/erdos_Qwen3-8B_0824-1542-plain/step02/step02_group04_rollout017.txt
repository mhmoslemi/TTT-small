import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize, Bounds
from scipy.optimize.optimize import OptimizeResult

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Start with 200 points for better resolution
    dx = 2.0 / n_points
    # Initialize with a pattern that spreads the weight evenly
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 0.5 * (1 + np.cos(np.pi * i / n_points))
    initial_h = np.clip(initial_h, 0, 1)
    # Ensure the integral constraint is satisfied
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    h = np.copy(initial_h)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint(h):
        h = np.clip(h, 0, 1)
        return np.sum(h) - (n_points / 2)

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)}

    # Optimization settings
    options = {
        'ftol': 1e-8,
        'maxiter': 150,
        'disp': False
    }

    start_time = time.time()
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options=options)
    elapsed_time = time.time() - start_time

    if elapsed_time > budget_s:
        return (initial_h, 0.38092, n_points)

    h_values = result.x
    c5_bound = objective(h_values)
    return (h_values, c5_bound, n_points)
