                h[i] = 1.0
            elif i % 5 == 1:
                h[i] = 0.5
            elif i % 5 == 2:
                h[i] = 0.3
            elif i % 5 == 3:
                h[i] = 0.2

                h[i] = 0.1
        # Normalize to ensure integral is 1
        h = h * (n_points / 2) / np.sum(h)

        bounds = Bounds(0, 1)

        start_time = time.time()
        result = minimize(

            h,
