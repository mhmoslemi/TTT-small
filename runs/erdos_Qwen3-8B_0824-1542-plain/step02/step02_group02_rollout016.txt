import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    # Initial construction from initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    best_c5 = 1.0
    best_h = initial_h.copy()
    best_n_points = n_points
    
    def evaluate_c5(h_values):
        h = np.array(h_values)
        # Use correlate with mode 'full' to get all shifts
        corr = correlate(h, 1 - h, mode="full")
        # Extract the maximum value and multiply by dx
        c5 = np.max(corr) * dx
        return c5
    
    # Try different numbers of points in a more refined way
    n_points_list = [100, 120, 140, 160, 180, 200]
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        # Generate a good starting guess based on the initial construction
        # We use a piecewise constant function with a small number of segments
        # to avoid getting stuck in local optima
        starting_segments = 8
        segment_size = 2.0 / starting_segments
        h_start = np.zeros(n_points)
        for i in range(starting_segments):
            start_idx = int(i * segment_size / dx)
            end_idx = int((i + 1) * segment_size / dx)
            h_start[start_idx:end_idx] = 1.0 if i % 2 == 0 else 0.0
        # Normalize to meet the integral constraint
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        
        # Bounds for h
        bounds = Bounds(0, 1)
        
        def objective_function(h_values):
            return evaluate_c5(h_values)
        
        def constraint_sum(h_values):
            return np.sum(h_values) - n_points / 2
        
        # Optimization settings
        start_time = time.time()
        result = minimize(
            objective_function,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'disp': False}
        )
        elapsed_time = time.time() - start_time
        
        if elapsed_time < budget_s:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
                best_n_points = n_points
        else:
            # If we run out of time, just use the best so far
            break
    
    return (best_h, best_c5, best_n_points)
