import random
import math
from deap import base, creator, tools, algorithms
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    best_c5 = 1.0
    best_h = initial_h.copy()
    
    def evaluate_c5(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Setup genetic algorithm
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)

    # Genetic algorithm parameters
    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0.0, 1.0)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    
    toolbox.register("evaluate", evaluate_c5)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0.0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Initial population with the initial guess
    population = [initial_h]
    
    # Set up the algorithm
    stats = tools.Statistics(lambda ind: ind.fitness.values)
    stats.register("avg", np.mean)
    stats.register("std", np.std)
    stats.register("min", np.min)
    stats.register("max", np.max)
    
    start_time = time.time()
    for gen in range(100):  # Max generations
        if time.time() - start_time > budget_s:
            break
        offspring = algorithms.varAnd(population, toolbox, cxpb=0.5, mutpb=0.1)
        fits = [toolbox.evaluate(ind) for ind in offspring]
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        population = tools.selBest(population + offspring, len(population))
        best_ind = tools.selBest(population, 1)[0]
        if best_ind.fitness.values[0] < best_c5:
            best_c5 = best_ind.fitness.values[0]
            best_h = best_ind
    
    return (best_h, best_c5, n_points)
