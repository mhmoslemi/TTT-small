import numpy as np
import scipy.optimize as optimize
import time
import random
import copy

# Pre-imported helper functions
# evaluate_erdos_solution(h_values) computes C5
# initial_h_values is a starting point with n_points samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial n_points and h
    n_points = 141
    h = initial_h_values.copy()
    
    def evaluate_C5(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def optimize_h(h_start):
        result = optimize.minimize(
            evaluate_C5,
            h_start,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": constraint_sum}],
            method="L-BFGS-B",
            options={"ftol": 1e-9, "maxiter": 50}
        )
        return result.x, evaluate_C5(result.x)
    
    best_h, best_c5 = optimize_h(h)
    
    for _ in range(5):
        # Perturb the current best solution
        perturbation = np.random.normal(0, 0.05, size=n_points)
        h_perturbed = np.clip(best_h + perturbation, 0.0, 1.0)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        h_perturbed, c5_perturbed = optimize_h(h_perturbed)
        if c5_perturbed < best_c5:
            best_h, best_c5 = h_perturbed, c5_perturbed
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.5:
        # Try with double the number of points
        n_points_new = int(1.5 * n_points)
        dx_new = 2.0 / n_points_new
        
        # Resample the current best solution to the new grid
        x = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, n_points_new)
        h_new = np.interp(x_new, x, best_h)
        
        # Ensure the sum constraint is satisfied
        h_new = h_new * (n_points_new / 2) / np.sum(h_new)
        
        def evaluate_C5_new(h_values):
            h = np.array(h_values)
            corr = np.correlate(h, 1 - h, mode="full")
            max_corr = np.max(corr)
            return max_corr * dx_new
        
        def constraint_sum_new(h_values):
            return np.sum(h_values) - n_points_new / 2
        
        bounds_new = [(0.0, 1.0) for _ in range(n_points_new)]
        
        result_new = optimize.minimize(
            evaluate_C5_new,
            h_new,
            bounds=bounds_new,
            constraints=[{"type": "eq", "fun": constraint_sum_new}],
            method="L-BFGS-B",
            options={"ftol": 1e-9, "maxiter": 50}
        )
        
        best_h = result_new.x
        best_c5 = evaluate_C5_new(best_h)
        n_points = n_points_new
    
    return best_h, best_c5, n_points
