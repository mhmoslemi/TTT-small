import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize, Bounds
from scipy.optimize import Bounds, LinearConstraint

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try different numbers of points for speed and accuracy balance
    n_points_list = [200, 300, 400]
    best_c5 = float('inf')
    best_h = None
    best_n = 200
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        
        # Use initial_h_values if available
        if 'initial_h_values' in globals():
            initial_h = initial_h_values.copy()
            initial_h = np.clip(initial_h, 0, 1)
            initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        else:
            initial_h = np.random.rand(n_points)
            initial_h = np.clip(initial_h, 0, 1)
            initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        
        def objective(h):
            h = h.reshape(n_points)
            h = np.clip(h, 0, 1)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        def constraint(h):
            h = h.reshape(n_points)
            return np.sum(h) - (n_points / 2)
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)}
        
        # Try different optimization methods
        start_time = time.time()
        try:
            result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, 
                              options={'ftol': 1e-9, 'maxiter': 200})
        except:
            result = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                              options={'ftol': 1e-9, 'maxiter': 200})
        
        elapsed_time = time.time() - start_time
        
        if elapsed_time > budget_s:
            continue
        
        h_values = result.x
        c5_bound = objective(h_values)
        
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_values
            best_n = n_points
    
    return (best_h, best_c5, best_n)
