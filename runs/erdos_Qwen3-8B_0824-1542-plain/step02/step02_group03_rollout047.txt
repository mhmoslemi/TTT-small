        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * 2.0 / n_points  # dx = 2.0 / n_points
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - (n_points / 2)
    
    # Bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)}
    
    # Initial attempt with SLSQP
    start_time = time.time()
    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons,

    elapsed_time = time.time() - start_time
    
    # If time is exceeded, return early
    if elapsed_time > budget_s:

    # If improvement is possible, try a more diverse search with differential evolution
    if elapsed_time < budget_s * 0.7:
