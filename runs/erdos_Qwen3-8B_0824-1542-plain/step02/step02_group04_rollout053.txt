import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)

    best_n = 200
    
    for n_points in n_points_list:
        # Initial guess
        if n_points == 200:
            h = np.array(initial_h_values, dtype=np.float64)
        else:
            # For larger n_points, create a refined version of the initial guess
            # by interpolation
            x = np.linspace(0, 2, len(initial_h_values))
            f = interp1d(x, initial_h_values, kind='linear', fill_value='extrapolate')
            h = f(np.linspace(0, 2, n_points))
            h = np.clip(h, 0, 1)
        
        dx = 2.0 / n_points
        h = np.clip(h, 0, 1)
        
        # Define objective function
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0, 1)
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        # Define constraint
        def constraint(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) - (n_points / 2)
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)}
        
        # Optimization settings
        options = {
            'ftol': 1e-8,
            'maxiter': 150,
            'disp': False
        }
        
        start_time = time.time()
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options=options)
        elapsed_time = time.time() - start_time
        
        if elapsed_time > budget_s:
            continue
        
        h_values = result.x
        h_values = np.clip(h_values, 0, 1)
        c5_bound = objective(h_values)
        
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_values
            best_n = n_points
    
    return (best_h, best_c5, best_n)
