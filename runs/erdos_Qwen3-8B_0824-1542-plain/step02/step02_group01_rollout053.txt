import numpy as np
import scipy.optimize as optimize
import time

import copy
import cvxpy as cp

# Pre-imported helper functions
# evaluate_erdos_solution(h_values) computes C5
# initial_h_values is a starting point with n_points samples

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Try with different numbers of points
    n_points_list = [100, 150, 200]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 100
    
    for n_points in n_points_list:
        # Initialize h with uniform distribution
        h_initial = np.random.uniform(0, 1, n_points)
        # Rescale to satisfy the integral constraint
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Define bounds: h[i] must be in [0, 1]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Function to evaluate the C5 value
        def evaluate_C5(h):
            h = np.array(h)
            # Compute the correlation
            corr = np.correlate(h, 1 - h, mode="full")
            # Get the maximum value
            max_corr = np.max(corr)
            # Multiply by dx (2/n_points)
            dx = 2.0 / n_points
            c5 = max_corr * dx
            return c5
        
        # Function to compute the integral constraint
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Define the optimization problem
        h = cp.Variable(n_points, lb=0.0, ub=1.0)
        objective = cp.Minimize(evaluate_C5(h))
        constraints = [cp.sum(h) == n_points / 2]
        
        # Solve the optimization problem
        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve(solver=cp.SCS, max_iter=1000)
            if result is not None:
                current_h = h.value
                current_c5 = evaluate_C5(current_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
                    best_n_points = n_points
        except:
            pass
        
        # Try with multiple initializations
        for _ in range(3):
            # Create random perturbations
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            h_perturbed = np.clip(h_initial + perturbation, 0.0, 1.0)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            # Define the optimization problem
            h = cp.Variable(n_points, lb=0.0, ub=1.0)

                result = prob.solve(solver=cp.SCS, max_iter=1000)

                    current_h = h.value
                    current_c5 = evaluate_C5(current_h)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = current_h
                        best_n_points = n_points
            except:
                pass
    
    # Check if we have time left to try a higher resolution
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.5:
        # Try with double the number of points
        n_points_new = int(1.5 * best_n_points)
        dx_new = 2.0 / n_points_new
        
        # Resample the current best solution to the new grid
        x = np.linspace(0, 2, best_n_points)
        x_new = np.linspace(0, 2, n_points_new)
        h_new = np.interp(x_new, x, best_h)
        
        # Ensure the sum constraint is satisfied
        h_new = h_new * (n_points_new / 2) / np.sum(h_new)
        
        # Function to evaluate the C5 value
        def evaluate_C5_new(h):
            corr = np.correlate(h, 1 - h, mode="full")
            max_corr = np.max(corr)
            c5 = max_corr * dx_new
            return c5
        
        # Define the optimization problem
        h = cp.Variable(n_points_new, lb=0.0, ub=1.0)
        objective = cp.Minimize(evaluate_C5_new(h))
        constraints = [cp.sum(h) == n_points_new / 2]
        
        # Solve the optimization problem
        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve(solver=cp.SCS, max_iter=1000)
            if result is not None:
                current_h = h.value
                current_c5 = evaluate_C5_new(current_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
                    best_n_points = n_points_new
        except:
            pass
    
    # Return the best solution found
    return best_h, best_c5, best_n_points
