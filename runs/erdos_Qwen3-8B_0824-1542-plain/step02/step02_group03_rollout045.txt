import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize, Bounds
from scipy.optimize import Bounds, LinearConstraint
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Start with 200 points for a balance between accuracy and speed
    dx = 2.0 / n_points

    # Use the initial_h_values if available to start the search
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    else:
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        h = h.reshape(n_points)
        return np.sum(h) - (n_points / 2)

    # Bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones(n_points)}

    # Try different optimization methods with different settings
    start_time = time.time()
    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons_sum,
                      options={'ftol': 1e-9, 'maxiter': 150})
    elapsed_time = time.time() - start_time

    # If time is exceeded, return early
    if elapsed_time > budget_s:
        return (initial_h, 0.38092, n_points)

    h_values = result.x
    c5_bound = objective(h_values)

    # Try a different starting point with a modified initial guess
    if elapsed_time < budget_s * 0.7:
        initial_h_new = np.zeros(n_points)
        n_segments = 5
        segment_length = n_points // n_segments
        for i in range(n_segments):
            start = i * segment_length
            end = (i + 1) * segment_length
            initial_h_new[start:end] = 1.0
        initial_h_new = initial_h_new * (n_points / 2) / np.sum(initial_h_new)
        initial_h_new = np.clip(initial_h_new, 0, 1)

        result = minimize(objective, initial_h_new, method='SLSQP', bounds=bounds, constraints=cons_sum,
                          options={'ftol': 1e-9, 'maxiter': 100})
        elapsed_time = time.time() - start_time
        if elapsed_time < budget_s:
            h_values = result.x
            c5_bound = objective(h_values)

    return (h_values, c5_bound, n_points)
