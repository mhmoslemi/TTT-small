        c5 = np.max(corr) * dx
        return c5

    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)}

    }

    start_time = time.time()
    result_slsqp = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options=options)
    elapsed_time = time.time() - start_time

    if elapsed_time > budget_s:
        return (initial_h, 0.38092, n_points)

    h_values_slsqp = result_slsqp.x
    c5_bound_slsqp = objective(h_values_slsqp)

    # Try differential evolution for global search
    def objective_de(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
