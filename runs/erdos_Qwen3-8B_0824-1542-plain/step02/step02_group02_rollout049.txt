        # Bounds for h
        bounds = Bounds(0, 1)
        
        def objective_function(h_values):
            return evaluate_c5(h_values)
        
        def constraint_sum(h_values):
            return np.sum(h_values) - n_points / 2
        
        # Optimization settings
        start_time = time.time()
        result = minimize(
            objective_function,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'disp': False}
        )

        if elapsed_time < budget_s - best_time:
            current_c5 = result.fun

                best_c5 = current_c5
                best_h = result.x.copy()
                best_n_points = n_points
