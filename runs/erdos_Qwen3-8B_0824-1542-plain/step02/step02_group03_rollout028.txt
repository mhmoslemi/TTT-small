        corr = np.correlate(h, 1 - h, mode="full")

        return c5

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)}

    start_time = time.time()
    
    # Set up for differential evolution
    # Ensure we respect the constraints and bounds
    def objective_de(h):
        h = np.clip(h, 0, 1)

        h = np.clip(h, 0, 1)
