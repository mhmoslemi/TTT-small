        # Normalize to ensure integral is 1
        h = h * (n_points / 2) / np.sum(h)

        bounds = Bounds(0, 1)

        # Add constraints for smooth transitions
        def smoothness_constraint(h_values):
            h = np.array(h_values)
            return np.sum(np.gradient(h))  # Penalize sharp changes

        smoothness_constraint = LinearConstraint(np.eye(n_points), -1e3, 1e3)

        start_time = time.time()
        result = minimize(
            objective_function,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=[
                {'type': 'eq', 'fun': constraint_sum},
                smoothness_constraint
            ],

            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
                best_n_points = n_points
