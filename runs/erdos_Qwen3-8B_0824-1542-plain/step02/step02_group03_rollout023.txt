    np.random.seed(seed)
    n_points = 150  # Use fewer points for faster computation while maintaining accuracy
    
    # Use the initial_h_values if available to start the search
    if 'initial_h_values' in globals():

        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint(h):
        h = h.reshape(n_points)
        return np.sum(h) - (n_points / 2)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)}
    
    # Optimization with more iterations and tighter tolerances
    start_time = time.time()
    result = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                      options={'ftol': 1e-9, 'maxiter': 200})
    elapsed_time = time.time() - start_time
    
    # If time is exceeded, return early
    if elapsed_time > budget_s:
        return (initial_h, 0.38092, n_points)
    
    h_values = result.x
    c5_bound = objective(h_values)
    
    # Further refinement using a simple gradient descent
    for _ in range(10):
        grad = np.zeros(n_points)
        for i in range(n_points):
            h = h_values.copy()

            c1 = objective(h)
            h[i] -= 2e-4
            c2 = objective(h)
