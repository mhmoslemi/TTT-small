import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize, Bounds
from scipy.optimize import Bounds
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h, dtype=np.float64)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint(h):
        h = h.reshape(n_points)
        return np.sum(h) - (n_points / 2)

    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    options = {
        'ftol': 1e-8,
        'maxiter': 150,
        'disp': False
    }

    # Try different initializations and optimization strategies
    def optimize_with_method(h_initial, method):
        result = minimize(objective, h_initial, method=method, bounds=bounds, constraints={'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)}, options=options)
        return result.x, objective(result.x)

    # Try different methods and initializations
    best_h = h
    best_c5 = objective(h)
    best_n = n_points

    # Try the initial construction
    initial_h_values = initial_h
    h_initial = np.array(initial_h_values, dtype=np.float64)
    h_values, c5 = optimize_with_method(h_initial, 'SLSQP')
    if c5 < best_c5:
        best_h = h_values
        best_c5 = c5
        best_n = n_points

    # Try random initialization
    h_initial = np.random.rand(n_points)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    h_values, c5 = optimize_with_method(h_initial, 'SLSQP')
    if c5 < best_c5:
        best_h = h_values
        best_c5 = c5
        best_n = n_points

    # Try differential evolution
    def differential_evolution_objective(x):
        h = x.reshape(n_points)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def differential_evolution_constraint(x):
        h = x.reshape(n_points)
        return np.sum(h) - (n_points / 2)

    bounds_de = [(0, 1) for _ in range(n_points)]
    result_de = differential_evolution(differential_evolution_objective, bounds_de, constraints=differential_evolution_constraint, strategy='best1bin', popsize=15, maxiter=50, tol=1e-8, mutation=(0.5, 1), recombination=0.8, seed=seed)
    h_values = result_de.x
    c5 = differential_evolution_objective(h_values)
    if c5 < best_c5:
        best_h = h_values
        best_c5 = c5
        best_n = n_points

    # Try a smaller number of points
    n_points_small = 100
    dx_small = 2.0 / n_points_small

    h_values_small = h_initial_small.copy()
    h_values_small = np.clip(h_values_small, 0, 1)
    corr_small = np.correlate(h_values_small, 1 - h_values_small, mode="full")
    c5_small = np.max(corr_small) * dx_small

    # Optimize with SLSQP for small n_points
    def objective_small(h):
        h = h.reshape(n_points_small)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx_small
        return c5

    def constraint_small(h):
        h = h.reshape(n_points_small)
        return np.sum(h) - (n_points_small / 2)

    bounds_small = [(0, 1) for _ in range(n_points_small)]
    cons_small = {'type': 'eq', 'fun': constraint_small, 'jac': lambda h: np.ones(n_points_small)}
    result_small = minimize(objective_small, h_values_small, method='SLSQP', bounds=bounds_small, constraints=cons_small, options=options)

    c5_small = objective_small(h_values_small)
    if c5_small < best_c5:
        best_h = h_values_small
        best_c5 = c5_small
        best_n = n_points_small

    # Check if we have enough time left to try more
    elapsed_time = time.time() - time.time()
    if elapsed_time < budget_s:
        # Try another random initialization
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        h_values, c5 = optimize_with_method(h_initial, 'SLSQP')
        if c5 < best_c5:
            best_h = h_values
            best_c5 = c5
            best_n = n_points

        # Try another differential evolution run
        result_de = differential_evolution(differential_evolution_objective, bounds_de, constraints=differential_evolution_constraint, strategy='best1bin', popsize=15, maxiter=50, tol=1e-8, mutation=(0.5, 1), recombination=0.8, seed=seed + 1)
        h_values = result_de.x
        c5 = differential_evolution_objective(h_values)
        if c5 < best_c5:
            best_h = h_values
            best_c5 = c5
            best_n = n_points

    return (best_h, best_c5, best_n)
