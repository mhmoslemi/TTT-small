import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import DifferentialEvolution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial construction with n=120 samples
    initial_h_values = np.load('initial_h_values.npy')  # Assuming initial_h_values.npy is available
    n_points = len(initial_h_values)
    best_c5 = 1.0
    best_h = initial_h_values.copy()
    best_n_points = n_points
    
    def evaluate_c5(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / len(h))
        return c5
    
    def optimize_with_de(n_points, seed):
        dx = 2.0 / n_points
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        
        def objective_function(h_values):
            return evaluate_c5(h_values)
        
        def constraint_sum(h_values):
            return np.sum(h_values) - n_points / 2
        
        bounds = Bounds(0, 1)
        result = minimize(
            objective_function,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'disp': False}
        )
        return result.x, result.fun
    
    # Try different numbers of points
    n_points_list = [100, 120, 150, 200]
    for n_points in n_points_list:
        start_time = time.time()
        h, current_c5 = optimize_with_de(n_points, seed)
        elapsed_time = time.time() - start_time
        
        if elapsed_time < budget_s:
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                best_n_points = n_points
        else:
            # If we run out of time, just use the best so far
            break
    
    return (best_h, best_c5, best_n_points)
