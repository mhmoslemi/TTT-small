import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
from scipy.optimize.slsqp import minimize_slsqp

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess with a structured pattern
    n_points = 100
    initial_h_values = np.zeros(n_points)
    initial_h_values[::2] = 1.0
    np.random.shuffle(initial_h_values)
    initial_h_values = (initial_h_values * n_points / 2).astype(np.float64)
    initial_h_values = initial_h_values / (np.sum(initial_h_values) * 2.0 / n_points)
    
    def evaluate_c5(h):
        h = h.reshape(-1)
        dx = 2.0 / len(h)
        conv = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(conv) * dx
        return max_overlap
    
    def optimize_h(initial_h, n_points, max_iterations=100):
        best_h = initial_h.copy()
        best_c5 = evaluate_c5(best_h)
        
        for _ in range(max_iterations):
            # Perturb the current best solution
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h = best_h + perturbation
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2.0)
            h = h / (np.sum(h) * 2.0 / n_points)
            
            # Use SLSQP to refine the solution
            x0 = h.copy()
            bounds = Bounds(0.0, 1.0)
            constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
            
            result = minimize_slsqp(evaluate_c5, x0=x0, bounds=bounds, constraints=constraint)
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        
        return best_h, best_c5
    
    # Start with the initial guess
    start_time = time.time()
    best_h, best_c5 = optimize_h(initial_h_values, n_points)
    
    # If time allows, try a few more iterations
    if time.time() - start_time < budget_s:
        for _ in range(5):
            # Perturb the current best and refine
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h = best_h + perturbation
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2.0)
            h = h / (np.sum(h) * 2.0 / n_points)
            
            # Optimize
            result = minimize_slsqp(evaluate_c5, x0=h, bounds=Bounds(0.0, 1.0),
                                   constraints=LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2))
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    
    return (best_h, best_c5, n_points)
