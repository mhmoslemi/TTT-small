from scipy.optimize import differential_evolution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Start with 200 points for a balance between accuracy and speed
    
    # Use the initial_h_values if available to start the search
    if 'initial_h_values' in globals():

        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    else:
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * 2.0 / n_points  # dx = 2 / n_points
        return c5
    
    def constraint(h):
        h = h.reshape(n_points)
        return np.sum(h) - (n_points / 2)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)}
    
    # Try a more global optimization approach with differential evolution
    def objective_diff_ev(h):
        return objective(h)
    
    # Define bounds for differential evolution
    bounds_diff_ev = [(0, 1) for _ in range(n_points)]
    
    # Run differential evolution with a smaller population and step size
    start_time = time.time()
    result = differential_evolution(objective_diff_ev, bounds_diff_ev, strategy='best1bin', 
                                   popsize=20, tol=1e-6, mutation=(0.5, 1), recombination=0.8, 
                                   polish=True, disp=False, maxiter=100)
    elapsed_time = time.time() - start_time
    
    # If time is exceeded, return early
    if elapsed_time > budget_s:
        return (initial_h, 0.38092, n_points)
    
    h_values = result.x
    c5_bound = objective(h_values)
    
    # Further refine with local optimization
    def objective_local(h):
        return objective(h)
    
    # Define bounds for local optimization
    bounds_local = [(0, 1) for _ in range(n_points)]
    
    # Run local optimization with SLSQP
    result_local = minimize(objective_local, h_values, method='SLSQP', bounds=bounds_local,

    elapsed_time += time.time() - start_time
    if elapsed_time > budget_s:
        return (h_values, c5_bound, n_points)
    
    h_values = result_local.x
    c5_bound = objective(h_values)
    
    return (h_values, c5_bound, n_points)
