        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

        h = h.reshape(n_points)
        return np.sum(h) - (n_points / 2)
