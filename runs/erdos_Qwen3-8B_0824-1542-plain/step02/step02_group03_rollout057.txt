import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with a moderate number of points for speed
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h = np.clip(initial_h, 0, 1)
        # Normalize to meet the integral constraint
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    else:
        # Random initial guess
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h):
        h = h.reshape(n_points)
        return np.sum(h) - (n_points / 2)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)}
    
    # Optimization settings
    options = {
        'ftol': 1e-9,
        'maxiter': 200,
        'disp': False
    }
    
    start_time = time.time()
    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, options=options)
    elapsed_time = time.time() - start_time
    
    if elapsed_time > budget_s:
        return (initial_h, 0.38092, n_points)
    
    h_values = result.x
    c5_bound = objective(h_values)
    
    # Try a few additional refinements
    for _ in range(3):
        if elapsed_time + 1 > budget_s:
            break
        # Perturb the solution slightly
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = h_values + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        # Evaluate and compare
        perturbed_c5 = objective(perturbed_h)
        if perturbed_c5 < c5_bound:
            h_values = perturbed_h
            c5_bound = perturbed_c5
    
    return (h_values, c5_bound, n_points)
