import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize, Bounds
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Start with 200 points for a balance between accuracy and speed
    dx = 2.0 / n_points
    
    # Use the initial_h_values if available to start the search
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    else:
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h):
        h = h.reshape(n_points)
        return np.sum(h) - (n_points / 2)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization with differential evolution for global search
    def differential_evolution_run():
        start_time = time.time()
        result = differential_evolution(
            objective, 
            bounds, 
            strategy='best1bin',
            popsize=15,
            maxiter=50,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9,
            seed=seed,
            disp=False
        )
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            return (initial_h, 0.38092, n_points)
        h_values = result.x
        c5_bound = objective(h_values)
        return (h_values, c5_bound, n_points)
    
    # Try differential evolution first
    h_values, c5_bound, n_points = differential_evolution_run()
    
    # If time allows, try local optimization with SLSQP
    if time.time() - time.time() < budget_s:
        start_time = time.time()
        result = minimize(
            objective, 
            h_values, 
            method='SLSQP', 
            bounds=bounds, 
            constraints={'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)},
            options={'ftol': 1e-9, 'maxiter': 50}
        )
        elapsed_time = time.time() - start_time
        if elapsed_time < budget_s:
            h_values = result.x
            c5_bound = objective(h_values)
    
    return (h_values, c5_bound, n_points)
