import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def evaluate_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def objective_function(h_values):
        return evaluate_c5(h_values)

    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2

    # Initialize best solution
    best_c5 = evaluate_c5(initial_h)
    best_h = initial_h.copy()
    best_n_points = n_points
    time_left = budget_s

    # Parameters for optimization
    n_point_list = [100, 120, 150, 200, 250, 300, 350, 400]
    max_iterations = 100
    ftol = 1e-8

    for n_points in n_point_list:
        dx = 2.0 / n_points
        # Create initial guess with more informative distribution
        h = np.zeros(n_points)
        # Create a smooth transition in the middle with more variation
        h[n_points // 4 : 3 * n_points // 4] = 0.5
        h[n_points // 2 - 10 : n_points // 2 + 10] = 0.3
        h[n_points // 2 - 30 : n_points // 2 - 10] = 0.7
        h[n_points // 2 + 10 : n_points // 2 + 30] = 0.7
        # Normalize
        h = h * (n_points / 2) / np.sum(h)

        bounds = Bounds(0, 1)
        start_time = time.time()

        result = minimize(
            objective_function,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': max_iterations, 'ftol': ftol, 'disp': False}
        )

        elapsed_time = time.time() - start_time
        if elapsed_time < budget_s:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
                best_n_points = n_points
        else:
            # If we run out of time, just use the best so far
            break

    return (best_h, best_c5, best_n_points)
