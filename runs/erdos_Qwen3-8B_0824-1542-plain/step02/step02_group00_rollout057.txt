import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with the initial guess
    initial_h_values = np.load('initial_h_values.npy')  # Assuming this file exists with a good initial guess
    n_points = len(initial_h_values)
    
    def evaluate_c5(h):
        h = h.reshape(-1)
        dx = 2.0 / len(h)
        conv = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(conv) * dx
        return max_overlap
    
    # Function to enforce the integral constraint
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Optimization setup
    def optimize_h(initial_h, n_points, max_iterations=50, perturbation_range=0.1):
        best_h = initial_h.copy()
        best_c5 = evaluate_c5(best_h)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = best_h * (n_points / 2.0) / (np.sum(best_h) * 2.0 / n_points)
        
        for _ in range(max_iterations):
            # Perturb the current best solution
            perturbation = np.random.uniform(-perturbation_range, perturbation_range, size=n_points)
            h = best_h + perturbation
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2.0) / (np.sum(h) * 2.0 / n_points)
            
            # Use SLSQP with more precise settings
            x0 = h.copy()
            bounds = Bounds(0.0, 1.0)
            constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
            
            result = optimize.minimize(evaluate_c5, x0=x0, bounds=bounds, constraints=constraint, method='SLSQP')
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = optimize_h(initial_h_values, n_points)
    
    # If time allows, perform additional refinement
    if time.time() - start_time < budget_s:
        for _ in range(5):
            # Perturb the current best and refine
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            h = best_h + perturbation
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2.0) / (np.sum(h) * 2.0 / n_points)
            
            # Optimize
            result = optimize.minimize(evaluate_c5, x0=h, bounds=Bounds(0.0, 1.0),
                                       constraints=LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2),
                                       method='SLSQP')
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    
    # Ensure the final solution meets the constraint and is within bounds
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (n_points / 2.0) / (np.sum(best_h) * 2.0 / n_points)
    
    return (best_h, best_c5, n_points)
