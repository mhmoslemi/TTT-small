import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize, Bounds
from scipy.optimize.slsqp import SLSQP
from scipy.optimize import Bounds

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Start with 200 points for a balance between accuracy and speed
    
    # Use the initial_h_values if available to start the search
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    else:
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    dx = 2.0 / n_points
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - (n_points / 2)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda h_flat: np.ones(n_points)}
    
    # Use a more robust optimization method with tighter tolerances and a better initial guess
    start_time = time.time()
    
    # Try multiple restarts to avoid local minima
    best_h = initial_h
    best_c5 = objective(initial_h)
    
    for _ in range(3):  # 3 restarts
        try:
            result = minimize(
                objective, 
                initial_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons, 
                options={'ftol': 1e-9, 'maxiter': 150, 'eps': 1e-8, 'disp': False}
            )
            elapsed_time = time.time() - start_time
            if elapsed_time > budget_s:
                break
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        except:
            pass
    
    # After the optimization, evaluate with the best found solution
    h_values = best_h
    c5_bound = best_c5
    
    return (h_values, c5_bound, n_points)
