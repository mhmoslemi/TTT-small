import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize, Bounds
from scipy.optimize import DifferentialEvolution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with a smaller number of points for faster evaluation
    n_points = 150  # Lower n_points for faster computation
    dx = 2.0 / n_points
    
    # Use the initial_h_values if available
    initial_h = np.zeros(n_points)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    else:
        # Random initial guess
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Objective function: maximize the overlap integral
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Constraint: integral of h must be 1 (sum(h)*dx = 1)
    def constraint(h):
        h = h.reshape(n_points)
        return np.sum(h) - (n_points / 2)
    
    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization: Use a more global search method
    def differential_evolution_run():
        result = DifferentialEvolution(
            objective, 
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=50,
            tol=1e-6,
            polish=True,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        return result.x
    
    # Start with a local optimization
    result = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, 
                      constraints={'type': 'eq', 'fun': constraint},
                      options={'ftol': 1e-9, 'maxiter': 100})
    h_values = result.x
    
    # If local optimization improves, use that
    best_h = h_values
    best_c5 = objective(h_values)
    
    # If time allows, perform a global search using differential evolution
    start_time = time.time()
    if time.time() - start_time < budget_s:
        try:
            de_result = differential_evolution_run()
            de_h = de_result
            de_c5 = objective(de_h)
            if de_c5 < best_c5:
                best_h = de_h
                best_c5 = de_c5
        except:
            pass
    
    # Return the best found solution
    return (best_h, best_c5, n_points)
