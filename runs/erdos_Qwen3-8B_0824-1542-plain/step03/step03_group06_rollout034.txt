import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
from scipy.signal import correlate
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Predefined initial guess (n=100 samples)
    initial_h_values = np.zeros(100)
    initial_h_values[::2] = 1.0
    np.random.shuffle(initial_h_values)
    initial_h_values = (initial_h_values * 100 / 2).astype(np.float64)
    
    # Try different n_points configurations
    n_points_list = [100, 128, 160, 180, 200]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 100
    
    for n_points in n_points_list:
        if time.time() > budget_s - 5:
            break
        
        # Ensure total integral is 1
        h_values = initial_h_values.copy()
        h_values = h_values * (n_points / 2.0) / (np.sum(h_values) * 2.0 / n_points)
        
        def evaluate_c5(h):
            h = h.reshape(-1)
            dx = 2.0 / len(h)
            conv = correlate(h, 1 - h, mode="full")
            max_overlap = np.max(conv) * dx
            return max_overlap
        
        # Define bounds and constraints
        bounds = Bounds(0.0, 1.0)
        constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
        
        # Use differential evolution for global optimization
        result = differential_evolution(
            evaluate_c5,
            bounds,
            constraints=constraint,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            disp=False
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                best_n_points = n_points
    
    return (best_h, best_c5, best_n_points)
