import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported helper functions
# evaluate_erdos_solution(h_values) computes C5
# initial_h_values is a starting point with n_points samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = np.array(initial_h_values)
    total = np.sum(h)
    h = h * (n_points / 2) / total  # Ensure the integral constraint is met
    
    def evaluate_C5(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        dx = 2.0 / len(h)
        return max_corr * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - len(h_values) / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial optimization
    result = optimize.minimize(
        evaluate_C5,
        h,
        bounds=bounds,
        constraints=[{"type": "eq", "fun": constraint_sum}],
        method="SLSQP",
        options={"ftol": 1e-10, "maxiter": 100}
    )
    best_h = result.x
    best_c5 = evaluate_C5(best_h)
    
    # Try multiple random perturbations
    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = np.clip(best_h + perturbation, 0.0, 1.0)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        result_perturbed = optimize.minimize(
            evaluate_C5,
            h_perturbed,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": constraint_sum}],
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 100}
        )
        if evaluate_C5(result_perturbed.x) < best_c5:
            best_h = result_perturbed.x
            best_c5 = evaluate_C5(best_h)
    
    # If time remains, try increasing the number of points
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.5:
        n_points_new = int(1.5 * n_points)
        dx_new = 2.0 / n_points_new
        
        # Interpolate current solution to new grid
        x = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, n_points_new)
        h_new = np.interp(x_new, x, best_h)
        h_new = h_new * (n_points_new / 2) / np.sum(h_new)
        
        def evaluate_C5_new(h_values):
            corr = np.correlate(h_values, 1 - h_values, mode="full")
            max_corr = np.max(corr)
            return max_corr * dx_new
        
        def constraint_sum_new(h_values):
            return np.sum(h_values) - n_points_new / 2
        
        bounds_new = [(0.0, 1.0) for _ in range(n_points_new)]
        
        result_new = optimize.minimize(
            evaluate_C5_new,
            h_new,
            bounds=bounds_new,
            constraints=[{"type": "eq", "fun": constraint_sum_new}],
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 100}
        )
        
        best_h = result_new.x
        best_c5 = evaluate_C5_new(best_h)
        n_points = n_points_new
    
    return best_h, best_c5, n_points
