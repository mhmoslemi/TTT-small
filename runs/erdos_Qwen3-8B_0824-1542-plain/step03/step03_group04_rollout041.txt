import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported helper functions
# evaluate_erdos_solution(h_values) computes C5
# initial_h_values is a starting point with n_points samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    # Start with a low number of points to save time
    initial_n_points = 100
    n_points = initial_n_points
    dx = 2.0 / n_points
    
    # Initial optimization with the provided starting point
    h_initial = initial_h_values
    h_initial = h_initial[:n_points]  # Ensure it fits the current n_points
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    def evaluate_C5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        c5 = max_corr * dx
        return c5
    
    constraints = [{"type": "eq", "fun": lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial optimization
    result = optimize.minimize(
        evaluate_C5,
        h_initial,
        bounds=bounds,
        constraints=constraints,
        method="SLSQP",
        options={"ftol": 1e-9, "maxiter": 100}
    )
    
    best_h = result.x
    best_c5 = evaluate_C5(best_h)
    
    # Try a few different perturbations and improvements
    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0.0, 1.0)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        result_perturbed = optimize.minimize(
            evaluate_C5,
            perturbed_h,
            bounds=bounds,
            constraints=constraints,
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 50}
        )
        
        if evaluate_C5(result_perturbed.x) < best_c5:
            best_h = result_perturbed.x
            best_c5 = evaluate_C5(best_h)
    
    # Try a few different local search strategies
    for _ in range(5):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0.0, 1.0)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        result_perturbed = optimize.minimize(
            evaluate_C5,
            perturbed_h,
            bounds=bounds,
            constraints=constraints,
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 30}
        )
        
        if evaluate_C5(result_perturbed.x) < best_c5:
            best_h = result_perturbed.x
            best_c5 = evaluate_C5(best_h)
    
    # Try increasing the number of points if time allows
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.5:
        n_points_new = int(1.5 * n_points)  # increase by 50%
        dx_new = 2.0 / n_points_new
        
        # Resample the current best solution to the new grid
        x = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, n_points_new)
        h_new = np.interp(x_new, x, best_h)
        
        # Ensure the sum constraint is satisfied
        h_new = h_new * (n_points_new / 2) / np.sum(h_new)
        
        # Evaluate the new solution
        def evaluate_C5_new(h):
            corr = np.correlate(h, 1 - h, mode="full")
            max_corr = np.max(corr)
            c5 = max_corr * dx_new
            return c5
        
        constraints_new = [{"type": "eq", "fun": lambda h: np.sum(h) - n_points_new / 2}]
        bounds_new = [(0.0, 1.0) for _ in range(n_points_new)]
        
        result_new = optimize.minimize(
            evaluate_C5_new,
            h_new,
            bounds=bounds_new,
            constraints=constraints_new,
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 50}
        )
        
        new_c5 = evaluate_C5_new(result_new.x)
        if new_c5 < best_c5:
            best_h = result_new.x
            best_c5 = new_c5
            n_points = n_points_new
    
    return best_h, best_c5, n_points
