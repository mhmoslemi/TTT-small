import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported helper functions
# evaluate_erdos_solution(h_values) computes C5
# initial_h_values is a starting point with n_points samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    # Initial parameters
    n_points = 211  # Starting with the initial n_points from the provided initial_h_values
    h_initial = initial_h_values
    
    def evaluate_C5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def optimize_h(h_start):
        result = optimize.minimize(
            evaluate_C5,
            h_start,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": constraint_sum}],
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 150}
        )
        return result.x, evaluate_C5(result.x)
    
    # Initial optimization
    h_start = initial_h_values
    h_start = np.array(h_start)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    best_h, best_c5 = optimize_h(h_start)
    
    # Try multiple different initializations to avoid local optima
    for _ in range(5):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = np.clip(best_h + perturbation, 0.0, 1.0)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        h_perturbed, c5_perturbed = optimize_h(h_perturbed)
        if c5_perturbed < best_c5:
            best_h, best_c5 = h_perturbed, c5_perturbed
    
    # Try different starting points
    for _ in range(3):
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        h_rand, c5_rand = optimize_h(h_rand)
        if c5_rand < best_c5:
            best_h, best_c5 = h_rand, c5_rand
    
    # If time remains, try increasing number of points
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.5:
        n_points_new = int(1.5 * n_points)
        dx_new = 2.0 / n_points_new
        
        # Resample the current best solution to the new grid
        x = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, n_points_new)
        h_new = np.interp(x_new, x, best_h)
        
        # Ensure the sum constraint is satisfied
        h_new = h_new * (n_points_new / 2) / np.sum(h_new)
        
        # Define new evaluation function
        def evaluate_C5_new(h):
            corr = np.correlate(h, 1 - h, mode="full")
            max_corr = np.max(corr)
            c5 = max_corr * dx_new
            return c5
        
        # Optimization for the new number of points
        result_new = optimize.minimize(
            evaluate_C5_new,
            h_new,
            bounds=[(0.0, 1.0) for _ in range(n_points_new)],
            constraints=[{"type": "eq", "fun": lambda h: np.sum(h) - n_points_new / 2}],
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 100}
        )
        
        # Update the best solution
        new_h = result_new.x
        new_c5 = evaluate_C5_new(new_h)
        if new_c5 < best_c5:
            best_h, best_c5 = new_h, new_c5
        n_points = n_points_new
    
    return best_h, best_c5, n_points
