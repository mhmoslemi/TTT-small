import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported helper functions
# evaluate_erdos_solution(h_values) computes C5
# initial_h_values is a starting point with n_points samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = np.array(initial_h_values)
    dx = 2.0 / n_points
    
    def evaluate_C5(h_array):
        h = np.array(h_array)
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function for optimization
    def constraint_sum(h_array):
        return np.sum(h_array) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial optimization with more iterations
    result = optimize.minimize(
        lambda h: evaluate_C5(h),
        h,
        bounds=bounds,
        constraints=[{"type": "eq", "fun": constraint_sum}],
        method="SLSQP",
        options={"ftol": 1e-9, "maxiter": 200}
    )
    
    best_h = result.x
    best_c5 = result.fun
    best_n_points = n_points
    
    # Try a few different perturbations and improvements
    for _ in range(5):
        # Perturb the current solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0.0, 1.0)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        # Evaluate the perturbed solution
        result_perturbed = optimize.minimize(
            lambda h: evaluate_C5(h),
            perturbed_h,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": constraint_sum}],
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 100}
        )
        
        if result_perturbed.fun < best_c5:
            best_h = result_perturbed.x
            best_c5 = result_perturbed.fun
    
    # Try a few different local search strategies
    for _ in range(5):
        # Define a small perturbation
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0.0, 1.0)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        # Evaluate the perturbed solution
        result_perturbed = optimize.minimize(
            lambda h: evaluate_C5(h),
            perturbed_h,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": constraint_sum}],
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 50}
        )
        
        if result_perturbed.fun < best_c5:
            best_h = result_perturbed.x
            best_c5 = result_perturbed.fun
    
    # Check if we have time for a higher resolution
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.5:
        # Try with double the number of points
        n_points_new = int(2 * n_points)
        dx_new = 2.0 / n_points_new
        
        # Resample the current best solution to the new grid
        x = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, n_points_new)
        h_new = np.interp(x_new, x, best_h)
        
        # Ensure the sum constraint is satisfied
        h_new = h_new * (n_points_new / 2) / np.sum(h_new)
        
        # Evaluate the new solution
        def evaluate_C5_new(h):
            corr = np.correlate(h, 1 - h, mode="full")
            max_corr = np.max(corr)
            return max_corr * dx_new
        
        # Optimization for the new number of points
        result_new = optimize.minimize(
            evaluate_C5_new,
            h_new,
            bounds=[(0.0, 1.0) for _ in range(n_points_new)],
            constraints=[{"type": "eq", "fun": lambda h: np.sum(h) - n_points_new / 2}],
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 50}
        )
        
        # Update the best solution
        best_h = result_new.x
        best_c5 = result_new.fun
        best_n_points = n_points_new
    
    # Return the best solution found
    return best_h, best_c5, best_n_points
