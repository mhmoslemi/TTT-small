import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported helper functions
# evaluate_erdos_solution(h_values) computes C5
# initial_h_values is a starting point with n_points samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    # Start with the initial point
    n_points_initial = len(initial_h_values)
    dx_initial = 2.0 / n_points_initial
    
    # Initial evaluation
    def evaluate_C5(h, n_points, dx):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        c5 = max_corr * dx
        return c5
    
    def constraint_sum(h, n_points):
        return np.sum(h) - n_points / 2
    
    # Initial optimization
    bounds = [(0.0, 1.0) for _ in range(n_points_initial)]
    result = optimize.minimize(
        lambda h: evaluate_C5(h, n_points_initial, dx_initial),
        initial_h_values,
        bounds=bounds,
        constraints=[{"type": "eq", "fun": lambda h: constraint_sum(h, n_points_initial)}],
        method="SLSQP",
        options={"ftol": 1e-9, "maxiter": 100}
    )
    best_h = result.x
    best_c5 = evaluate_C5(best_h, n_points_initial, dx_initial)
    best_n_points = n_points_initial
    
    # Try a few different strategies
    for _ in range(5):
        # Perturb and optimize
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points_initial)
        perturbed_h = np.clip(best_h + perturbation, 0.0, 1.0)
        perturbed_h = perturbed_h * (n_points_initial / 2) / np.sum(perturbed_h)
        result_perturbed = optimize.minimize(
            lambda h: evaluate_C5(h, n_points_initial, dx_initial),
            perturbed_h,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": lambda h: constraint_sum(h, n_points_initial)}],
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 50}
        )
        if evaluate_C5(result_perturbed.x, n_points_initial, dx_initial) < best_c5:
            best_h = result_perturbed.x
            best_c5 = evaluate_C5(best_h, n_points_initial, dx_initial)
    
    # Try to increase resolution
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.5:
        n_points_new = int(2 * n_points_initial)
        dx_new = 2.0 / n_points_new
        x = np.linspace(0, 2, n_points_initial)
        x_new = np.linspace(0, 2, n_points_new)
        h_new = np.interp(x_new, x, best_h)
        h_new = h_new * (n_points_new / 2) / np.sum(h_new)
        
        # Optimize for new resolution
        bounds_new = [(0.0, 1.0) for _ in range(n_points_new)]
        result_new = optimize.minimize(
            lambda h: evaluate_C5(h, n_points_new, dx_new),
            h_new,
            bounds=bounds_new,
            constraints=[{"type": "eq", "fun": lambda h: constraint_sum(h, n_points_new)}],
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 50}
        )
        best_h = result_new.x
        best_c5 = evaluate_C5(best_h, n_points_new, dx_new)
        best_n_points = n_points_new
    
    return best_h, best_c5, best_n_points
