import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Use the initial construction
    initial_h_values = np.load('initial_h_values.npy')  # Load the initial construction if available
    
    # Try different n_points values - start with 100, then 200 if time allows
    n_points_list = [100, 200]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 100
    
    for n_points in n_points_list:
        if time.time() > budget_s:
            break
        
        # Initialize h with a structured pattern and slightly perturb
        if n_points == 100:
            h = initial_h_values.copy()
        else:
            h = np.zeros(n_points)
            h[::2] = 1.0
            h = np.random.permutation(h)
            h = (h * n_points / 2).astype(np.float64)
        
        # Normalize to ensure the integral is 1
        h = h / (np.sum(h) * 2.0 / n_points) * (n_points / 2.0)
        
        def evaluate_c5(h):
            h = h.reshape(-1)
            dx = 2.0 / len(h)
            conv = correlate(h, 1 - h, mode="full")
            max_overlap = np.max(conv) * dx
            return max_overlap
        
        def optimize_h(initial_h, n_points, max_iterations=100, perturbation_scale=0.1):
            best_h = initial_h.copy()
            best_c5 = evaluate_c5(best_h)
            
            for _ in range(max_iterations):
                # Perturb the current best solution
                perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
                h = best_h + perturbation
                h = np.clip(h, 0.0, 1.0)
                h = h * (n_points / 2.0)
                h = h / (np.sum(h) * 2.0 / n_points)
                
                # Use SLSQP to refine the solution
                x0 = h.copy()
                bounds = Bounds(0.0, 1.0)
                constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
                
                result = optimize.minimize(evaluate_c5, x0=x0, bounds=bounds, constraints=constraint, method='SLSQP')
                
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x
            
            return best_h, best_c5
        
        start_time = time.time()
        current_h, current_c5 = optimize_h(h, n_points)
        
        # If time allows, try a few more iterations
        if time.time() - start_time < budget_s:
            for _ in range(5):
                # Perturb the current best and refine
                perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
                h = current_h + perturbation
                h = np.clip(h, 0.0, 1.0)
                h = h * (n_points / 2.0)
                h = h / (np.sum(h) * 2.0 / n_points)
                
                # Optimize
                result = optimize.minimize(evaluate_c5, x0=h, bounds=Bounds(0.0, 1.0),
                                           constraints=LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2),
                                           method='SLSQP')
                
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        current_h = result.x
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            best_n_points = n_points
    
    return (best_h, best_c5, best_n_points)
