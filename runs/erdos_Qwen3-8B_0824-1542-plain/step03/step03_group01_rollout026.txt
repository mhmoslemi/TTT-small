import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported helper functions
# evaluate_erdos_solution(h_values) computes C5
# initial_h_values is a starting point with n_points samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 211  # Start with the initial n_points from initial_h_values
    
    # Function to evaluate C5
    def evaluate_C5(h, n_points):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    # Function to compute the integral constraint
    def constraint_sum(h, n_points):
        return np.sum(h) - n_points / 2
    
    # Initial optimization
    def optimize_h(h_start, n_points):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = optimize.minimize(
            lambda h: evaluate_C5(h, n_points),
            h_start,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": lambda h: constraint_sum(h, n_points)}],
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 150}
        )
        return result.x, evaluate_C5(result.x, n_points)
    
    # Initial optimization with initial_h_values
    h_initial = initial_h_values
    best_h, best_c5 = optimize_h(h_initial, best_n_points)
    
    # Try multiple initializations with random perturbations
    for _ in range(5):
        perturbation = np.random.uniform(-0.05, 0.05, size=best_n_points)
        h_perturbed = np.clip(best_h + perturbation, 0.0, 1.0)
        h_perturbed = h_perturbed * (best_n_points / 2) / np.sum(h_perturbed)
        h_perturbed, c5_perturbed = optimize_h(h_perturbed, best_n_points)
        if c5_perturbed < best_c5:
            best_h, best_c5 = h_perturbed, c5_perturbed
    
    # Try increasing n_points if time remains
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.5:
        n_points_new = int(1.5 * best_n_points)
        dx_new = 2.0 / n_points_new
        
        # Resample current best solution to new grid
        x = np.linspace(0, 2, best_n_points)
        x_new = np.linspace(0, 2, n_points_new)
        h_new = np.interp(x_new, x, best_h)
        h_new = h_new * (n_points_new / 2) / np.sum(h_new)
        
        # Optimize for new n_points
        h_new, c5_new = optimize_h(h_new, n_points_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
            best_n_points = n_points_new
    
    return best_h, best_c5, best_n_points
