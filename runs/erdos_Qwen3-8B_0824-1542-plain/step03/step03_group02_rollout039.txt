import numpy as np
import scipy.optimize as optimize
import time
import random

# evaluate_erdos_solution(h_values) computes C5
# initial_h_values is a starting point with n_points samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    # Try different numbers of points
    for n_points in [211, 300, 400, 500]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Initial guess
        if n_points == 211:
            h_initial = initial_h_values
        else:
            # Resample the best solution from previous points
            # For simplicity, we use a uniform distribution
            h_initial = np.random.rand(n_points)
            h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Define the function to evaluate C5
        def evaluate_C5(h):
            h = np.array(h)
            corr = np.correlate(h, 1 - h, mode="full")
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            c5 = max_corr * dx
            return c5
        
        # Define the constraint function
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Define bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization setup
        result = optimize.minimize(
            evaluate_C5,
            h_initial,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": constraint_sum}],
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 150}
        )
        
        # Update the best solution
        current_c5 = evaluate_C5(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
            best_n_points = n_points
        
        # Try a few perturbations to avoid local optima
        for _ in range(3):
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h_perturbed = np.clip(result.x + perturbation, 0.0, 1.0)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            result = optimize.minimize(
                evaluate_C5,
                h_perturbed,
                bounds=bounds,
                constraints=[{"type": "eq", "fun": constraint_sum}],
                method="SLSQP",
                options={"ftol": 1e-10, "maxiter": 100}
            )
            current_c5 = evaluate_C5(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                best_n_points = n_points
    
    return best_h, best_c5, best_n_points
