    def evaluate_solution(h_values):
        h = h_values.reshape(n_points)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h_values):

        return np.sum(h) - (n_points / 2)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)}
    
    def objective(h_values):
        return evaluate_solution(h_values)
    
    def optimize_with_basin_hopping(h_initial):
        result = basinhopping(objective, h_initial, niter=50, T=1.0, stepsize=0.5, minimizer_kwargs={"bounds": bounds})
        return result.x, result.fun
    
    def optimize_with_differential_evolution(h_initial):
        result = differential_evolution(objective, bounds, strategy='best1bin', 
                                       popsize=15, mutation=(0.5, 1), recombination=0.9, 
                                       tol=1e-6, maxiter=100, constraints=cons)
        return result.x, result.fun
    
    def optimize_with_slsqp(h_initial):
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, 
                          constraints=cons, options={'ftol': 1e-8, 'maxiter': 150})
        return result.x, result.fun
    
    # Start with the initial solution
    best_h = h_initial
    best_c5 = evaluate_solution(best_h)
    best_n_points = n_points

        # Try different optimization methods
        try:
            # Basin hopping
            start_time = time.time()
            h_basin, c5_basin = optimize_with_basin_hopping(h)
            elapsed += time.time() - start_time
            if c5_basin < best_c5:
                best_c5 = c5_basin
