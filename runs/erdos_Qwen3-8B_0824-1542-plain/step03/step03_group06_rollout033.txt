import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess with a structured pattern
    initial_h_values = np.zeros(200)
    initial_h_values[::4] = 1.0
    np.random.shuffle(initial_h_values)
    initial_h_values = (initial_h_values * 200 / 2).astype(np.float64)
    
    # Try different n_points values for efficiency and performance
    n_points_list = [100, 200, 150]
    best_c5 = float('inf')
    best_h_values = np.zeros(100)
    
    for n_points in n_points_list:
        # Ensure the initial guess is scaled properly
        initial_h = np.zeros(n_points)
        if n_points == 200:
            initial_h[::4] = 1.0
        elif n_points == 150:
            initial_h[::3] = 1.0
        else:  # n_points == 100
            initial_h[::2] = 1.0
        initial_h = np.random.permutation(initial_h)
        initial_h = (initial_h * n_points / 2).astype(np.float64)
        initial_h = initial_h / (np.sum(initial_h) * 2.0 / n_points)
        initial_h = initial_h * (n_points / 2.0)
        
        def evaluate_c5(h):
            h = h.reshape(-1)
            dx = 2.0 / len(h)
            conv = correlate(h, 1 - h, mode="full")
            max_overlap = np.max(conv) * dx
            return max_overlap
        
        def optimize_h(initial_h, n_points, max_iterations=100, perturbation_scale=0.1):
            best_h = initial_h.copy()
            best_c5 = evaluate_c5(best_h)
            
            for _ in range(max_iterations):
                # Perturb the current best solution
                perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
                h = best_h + perturbation
                h = np.clip(h, 0.0, 1.0)
                h = h * (n_points / 2.0)
                h = h / (np.sum(h) * 2.0 / n_points)
                
                # Use SLSQP to refine the solution
                x0 = h.copy()
                bounds = Bounds(0.0, 1.0)
                constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
                
                result = optimize.minimize(evaluate_c5, x0=x0, bounds=bounds, constraints=constraint, method='SLSQP')
                
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x
            
            return best_h, best_c5
        
        # Execute the optimization
        start_time = time.time()
        best_h, current_c5 = optimize_h(initial_h, n_points)
        
        # If time allows, do more refinement
        if time.time() - start_time < budget_s:
            for _ in range(5):
                # Perturb the current best and refine
                perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
                h = best_h + perturbation
                h = np.clip(h, 0.0, 1.0)
                h = h * (n_points / 2.0)
                h = h / (np.sum(h) * 2.0 / n_points)
                
                # Optimize
                result = optimize.minimize(evaluate_c5, x0=h, bounds=Bounds(0.0, 1.0),
                                           constraints=LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2),
                                           method='SLSQP')
                
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x
        
        # Update the best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_values = best_h
    
    return (best_h_values, best_c5, 100)
