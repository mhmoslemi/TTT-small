import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Use a more refined initial guess based on a periodic pattern
    n_points = 100
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            initial_h_values[i] = 1.0
        elif i % 3 == 1:
            initial_h_values[i] = 0.5
        else:
            initial_h_values[i] = 0.0
    initial_h_values = initial_h_values * (n_points / 2)
    initial_h_values = initial_h_values / (np.sum(initial_h_values) * 2.0 / n_points)
    
    def evaluate_c5(h):
        h = h.reshape(-1)
        dx = 2.0 / len(h)
        conv = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(conv) * dx
        return max_overlap
    
    def optimize_h(initial_h, n_points, max_iterations=50):
        best_h = initial_h.copy()
        best_c5 = evaluate_c5(best_h)
        
        for _ in range(max_iterations):
            # Perturb the current best solution
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h = best_h + perturbation
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2.0)
            h = h / (np.sum(h) * 2.0 / n_points)
            
            # Use a hybrid optimization approach: SLSQP and gradient descent
            x0 = h.copy()
            bounds = Bounds(0.0, 1.0)
            constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
            
            # First, use SLSQP for refinement
            result_slsqp = optimize.minimize(evaluate_c5, x0=x0, bounds=bounds, constraints=constraint, method='SLSQP')
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
            else:
                # If SLSQP fails, fall back to gradient descent
                result_gd = optimize.minimize(evaluate_c5, x0=x0, bounds=bounds, constraints=constraint, method='L-BFGS-B')
                if result_gd.success:
                    current_c5 = result_gd.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result_gd.x
        
        return best_h, best_c5
    
    # Start with the initial guess
    start_time = time.time()
    best_h, best_c5 = optimize_h(initial_h_values, n_points)
    
    # If time allows, try a few more iterations with different perturbations
    if time.time() - start_time < budget_s:
        for _ in range(5):
            # Perturb the current best and refine
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h = best_h + perturbation
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2.0)
            h = h / (np.sum(h) * 2.0 / n_points)
            
            # Optimize
            result = optimize.minimize(evaluate_c5, x0=h, bounds=Bounds(0.0, 1.0),
                                       constraints=LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2),
                                       method='SLSQP')
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    
    return (best_h, best_c5, n_points)
