import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess with a structured pattern
    initial_h_values = np.zeros(200)
    initial_h_values[::3] = 1.0
    np.random.shuffle(initial_h_values)
    initial_h_values = (initial_h_values * 200 / 2).astype(np.float64)
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2  # Ensure the integral condition is met
    
    # Function to evaluate C5
    def evaluate_c5(h):
        h = h.reshape(-1)
        conv = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(conv) * dx
        return max_overlap
    
    # Function to normalize h
    def normalize_h(h):
        h = h.copy()
        current_sum = np.sum(h)
        h = h * target_sum / current_sum
        return h
    
    # Function to optimize h using SLSQP
    def optimize_h(initial_h, n_points, max_iterations=50):
        best_h = initial_h.copy()
        best_c5 = evaluate_c5(best_h)
        best_h = normalize_h(best_h)
        
        for _ in range(max_iterations):
            # Perturb the current best solution
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h = best_h + perturbation
            h = np.clip(h, 0.0, 1.0)
            h = normalize_h(h)
            
            # Use SLSQP to refine the solution
            x0 = h.copy()
            bounds = Bounds(0.0, 1.0)
            constraint = LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum)
            
            result = optimize.minimize(evaluate_c5, x0=x0, bounds=bounds, constraints=constraint, method='SLSQP')
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        
        return best_h, best_c5
    
    # Start with the initial guess
    start_time = time.time()
    best_h, best_c5 = optimize_h(initial_h_values, n_points)
    
    # If time allows, try a few more iterations
    if time.time() - start_time < budget_s:
        for _ in range(5):
            # Perturb the current best and refine
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h = best_h + perturbation
            h = np.clip(h, 0.0, 1.0)
            h = normalize_h(h)
            
            # Optimize
            result = optimize.minimize(evaluate_c5, x0=h, bounds=Bounds(0.0, 1.0),
                                       constraints=LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum),
                                       method='SLSQP')
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    
    return (best_h, best_c5, n_points)
