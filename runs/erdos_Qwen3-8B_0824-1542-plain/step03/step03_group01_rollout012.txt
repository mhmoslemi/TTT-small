import numpy as np
import scipy.optimize as optimize
import time
import random
import matplotlib.pyplot as plt

# Pre-imported helper functions
# evaluate_erdos_solution(h_values) computes C5
# initial_h_values is a starting point with n_points samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 211  # Initial number of points from the provided initial_h_values

    # Initial parameters
    n_points = 211
    h_initial = initial_h_values

    def evaluate_C5(h, n_points, dx):
        h = np.array(h)
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Get the maximum value
        max_corr = np.max(corr)
        # Multiply by dx (2/n_points)
        c5 = max_corr * dx
        return c5

    def constraint_sum(h, n_points):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1 => sum(h) = n_points / 2

    def optimize_h(h_start, n_points, dx, initial_c5):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = optimize.minimize(
            lambda h: evaluate_C5(h, n_points, dx),
            h_start,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": lambda h: constraint_sum(h, n_points)}],
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 150}
        )
        return result.x, evaluate_C5(result.x, n_points, dx)

    def perturb_h(h, n_points, perturbation_factor=0.05):
        perturbation = np.random.uniform(-perturbation_factor, perturbation_factor, size=n_points)
        h_perturbed = np.clip(h + perturbation, 0.0, 1.0)
        # Ensure the sum constraint is satisfied
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        return h_perturbed

    # Initial optimization
    h_initial = np.array(initial_h_values)
    dx_initial = 2.0 / n_points
    best_h, best_c5 = optimize_h(h_initial, n_points, dx_initial, evaluate_C5(h_initial, n_points, dx_initial))

    # Try multiple different initializations to avoid local optima
    for _ in range(5):
        h_perturbed = perturb_h(best_h, n_points)
        h_perturbed, c5_perturbed = optimize_h(h_perturbed, n_points, dx_initial, evaluate_C5(h_perturbed, n_points, dx_initial))
        if c5_perturbed < best_c5:
            best_h, best_c5 = h_perturbed, c5_perturbed

    # If time remains, try increasing number of points
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.5:
        n_points_new = int(1.5 * n_points)
        dx_new = 2.0 / n_points_new

        # Resample the current best solution to the new grid
        x = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, n_points_new)
        h_new = np.interp(x_new, x, best_h)

        # Ensure the sum constraint is satisfied
        h_new = h_new * (n_points_new / 2) / np.sum(h_new)

        # Evaluate the new solution
        h_new, c5_new = optimize_h(h_new, n_points_new, dx_new, evaluate_C5(h_new, n_points_new, dx_new))
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
            n_points = n_points_new

    return best_h, best_c5, n_points
