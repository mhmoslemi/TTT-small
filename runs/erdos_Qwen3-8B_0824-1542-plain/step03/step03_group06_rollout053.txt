import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess based on a simple structured pattern
    n_points = 100
    initial_h_values = np.zeros(n_points)
    initial_h_values[::3] = 1.0  # Place 1s periodically
    initial_h_values[1::3] = 0.5  # Alternate with 0.5s
    initial_h_values = (initial_h_values * n_points / 2).astype(np.float64)
    initial_h_values = np.clip(initial_h_values, 0.0, 1.0)
    
    def evaluate_c5(h):
        dx = 2.0 / len(h)
        conv = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(conv) * dx
        return max_overlap
    
    def optimize_h(initial_h, n_points, max_iterations=20, perturbation_scale=0.05):
        best_h = initial_h.copy()
        best_c5 = evaluate_c5(best_h)
        best_h = best_h / (np.sum(best_h) * 2.0 / n_points)
        best_h = best_h * (n_points / 2.0)
        
        for _ in range(max_iterations):
            # Perturb the current best solution
            h = best_h + np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2.0)
            h = h / (np.sum(h) * 2.0 / n_points)
            
            # Use SLSQP to refine the solution
            x0 = h.copy()
            bounds = Bounds(0.0, 1.0)
            constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
            
            result = optimize.minimize(evaluate_c5, x0=x0, bounds=bounds, constraints=constraint, method='SLSQP')
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = optimize_h(initial_h_values, n_points)
    
    # If time allows, try a few more iterations
    if time.time() - start_time < budget_s:
        for _ in range(5):
            # Perturb the current best and refine
            h = best_h + np.random.uniform(-0.05, 0.05, size=n_points)
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2.0)
            h = h / (np.sum(h) * 2.0 / n_points)
            
            # Optimize
            result = optimize.minimize(evaluate_c5, x0=h, bounds=Bounds(0.0, 1.0),
                                       constraints=LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2),
                                       method='SLSQP')
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    
    return (best_h, best_c5, n_points)
