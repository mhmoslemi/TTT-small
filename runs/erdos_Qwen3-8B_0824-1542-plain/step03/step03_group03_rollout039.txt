import numpy as np
import scipy.optimize as optimize
import time
import random
import copy

# initial_h_values is a starting point with n_points samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    remaining_time = budget_s - (time.time() - start_time)
    
    # Start with a high-resolution initial guess
    n_points = 376
    h_initial = initial_h_values.copy()
    dx = 2.0 / n_points
    
    # Function to evaluate the C5 value
    def evaluate_C5(h):
        h_array = np.array(h)
        corr = np.correlate(h_array, 1 - h_array, mode="full")
        max_corr = np.max(corr)
        c5 = max_corr * dx
        return c5
    
    # Function to compute the integral constraint
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds: h[i] must be in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial optimization
    result = optimize.minimize(
        evaluate_C5,
        h_initial,
        bounds=bounds,
        constraints=[{"type": "eq", "fun": constraint_sum}],
        method="SLSQP",
        options={"ftol": 1e-10, "maxiter": 200, "eps": 1e-8}
    )
    
    best_h = result.x
    best_c5 = evaluate_C5(best_h)
    
    # Try a few different perturbations and improvements
    for _ in range(5):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0.0, 1.0)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        result_perturbed = optimize.minimize(
            evaluate_C5,
            perturbed_h,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": constraint_sum}],
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 100, "eps": 1e-8}
        )
        
        if evaluate_C5(result_perturbed.x) < best_c5:
            best_h = result_perturbed.x
            best_c5 = evaluate_C5(best_h)
    
    # Try increasing resolution if time allows
    if remaining_time > 5.0:
        n_points_new = int(1.5 * n_points)
        dx_new = 2.0 / n_points_new
        
        # Resample the current best solution to the new grid
        x = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, n_points_new)
        h_new = np.interp(x_new, x, best_h)
        
        # Ensure the sum constraint is satisfied
        h_new = h_new * (n_points_new / 2) / np.sum(h_new)
        
        # Define new bounds and constraints
        new_bounds = [(0.0, 1.0) for _ in range(n_points_new)]
        new_constraint = {"type": "eq", "fun": lambda h: np.sum(h) - n_points_new / 2}
        
        # Optimization for the new number of points
        result_new = optimize.minimize(
            lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * dx_new,
            h_new,
            bounds=new_bounds,
            constraints=[new_constraint],
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 150, "eps": 1e-8}
        )
        
        # Update the best solution
        best_h = result_new.x
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx_new
        n_points = n_points_new
    
    # Final refinement
    for _ in range(3):
        perturbation = np.random.uniform(-0.02, 0.02, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0.0, 1.0)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        result_perturbed = optimize.minimize(
            evaluate_C5,
            perturbed_h,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": constraint_sum}],
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 50, "eps": 1e-8}
        )
        
        if evaluate_C5(result_perturbed.x) < best_c5:
            best_h = result_perturbed.x
            best_c5 = evaluate_C5(best_h)
    
    return best_h, best_c5, n_points
