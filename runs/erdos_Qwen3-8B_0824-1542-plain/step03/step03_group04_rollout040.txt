        constraints=[{"type": "eq", "fun": lambda h: constraint_sum(h, n_points)}],

        options={"ftol": 1e-9, "maxiter": 200}
    )
    
    best_h = result.x

    def genetic_algorithm(population_size=30, generations=50, mutation_rate=0.1, elite_fraction=0.2):
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * (n_points / 2) / np.sum(h) for h in population]
        
        for generation in range(generations):
            # Evaluate fitness
            fitness = [evaluate_C5(h, n_points, dx) for h in population]

            elite_indices = np.argsort(fitness)[:int(elite_fraction * population_size)]

            parent_indices = np.random.choice(len(population), size=population_size - len(elites), replace=False)

            offspring = []

                        child[j] = p1[j]
                    else:
                        child[j] = p2[j]
                offspring.append(child)

            for i in range(len(offspring)):
                for j in range(n_points):

        result_new = optimize.minimize(

        best_c5 = evaluate_C5_new(best_h)
        n_points = n_points_new
