    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Function to compute the C5 value
    def compute_c5(h_values):
        h = h_values.copy()
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - (n_points / 2)
    
    # Objective function to minimize
    def objective(h):
        return compute_c5(h)
    
    # Bounds for each value in h
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    options = {
        'ftol': 1e-8,
        'maxiter': 500,
        'disp': False
    }
    
    # Optimization using SLSQP
    def optimize_h(h_initial):
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, 
                          constraints={'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)}, 
                          options=options)
        if result.success:
            return result.x
        else:
            return h_initial
    
    # Function to generate perturbations
    def perturb_h(h, perturbation=0.05):
        h_perturbed = h.copy()
        for i in range(len(h_perturbed)):
            h_perturbed[i] += np.random.uniform(-perturbation, perturbation)
            h_perturbed[i] = np.clip(h_perturbed[i], 0, 1)
        return h_perturbed
    
    # Function to generate random initial points
    def generate_random_h():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Function to generate a more structured initial guess
    def generate_structured_h():
        # Create a periodic structure with alternating regions
        h = np.zeros(n_points)
        for i in range(n_points):
            if (i % 4 == 0) or (i % 4 == 1):
                h[i] = 1.0
            else:
                h[i] = 0.0
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Initial strategies
    n_tries = 5
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    
    # Try initial strategies
    for _ in range(n_tries):
        h_perturbed = perturb_h(best_h)
        h_optimized = optimize_h(h_perturbed)
        c5 = compute_c5(h_optimized)
        if c5 < best_c5:
            best_h = h_optimized

    structured_h = generate_structured_h()

        best_c5 = c5

    random_h = generate_random_h()

            best_c5 = c5
