    def evaluate_C5(h, dx):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute the integral constraint
    def constraint_sum(h, n_points):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1 => sum(h) = n_points / 2
    
    # Define bounds: h[i] must be in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # First optimization to find a good initial solution
    result = optimize.minimize(
        lambda h: evaluate_C5(h, dx),
        h_initial,
        bounds=bounds,
        constraints=[{"type": "eq", "fun": lambda h: constraint_sum(h, n_points)}],
        method="SLSQP",
        options={"ftol": 1e-9, "maxiter": 300}
    )
    
    best_h = result.x
    best_c5 = evaluate_C5(best_h, dx)
    best_n_points = n_points
    
    # Try different perturbations to escape local minima
    for _ in range(3):
        # Perturb the current solution
        perturbation = np.random.uniform(-0.15, 0.15, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0.0, 1.0)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        # Evaluate the perturbed solution
        result_perturbed = optimize.minimize(
            lambda h: evaluate_C5(h, dx),
            perturbed_h,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": lambda h: constraint_sum(h, n_points)}],
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 150}
        )
        
        if evaluate_C5(result_perturbed.x, dx) < best_c5:
            best_h = result_perturbed.x
            best_c5 = evaluate_C5(best_h, dx)
    
    # Try a different starting point by resampling
    # Create a new starting point by uniformly sampling the interval
    new_h = np.random.uniform(0, 1, size=n_points)
    new_h = new_h * (n_points / 2) / np.sum(new_h)
    
    result_resampled = optimize.minimize(
        lambda h: evaluate_C5(h, dx),
        new_h,
        bounds=bounds,
        constraints=[{"type": "eq", "fun": lambda h: constraint_sum(h, n_points)}],
        method="SLSQP",
        options={"ftol": 1e-9, "maxiter": 200}
    )

        best_h = result_resampled.x
        best_c5 = evaluate_C5(best_h, dx)

            corr = np.correlate(h, 1 - h, mode="full")
            max_corr = np.max(corr)
            return max_corr * dx_new

            evaluate_C5_new,

            options={"ftol": 1e-9, "maxiter": 100}

            best_h = result_new.x
            best_c5 = evaluate_C5_new(best_h)
            best_n_points = n_points_new
    
    return best_h, best_c5, best_n_points
