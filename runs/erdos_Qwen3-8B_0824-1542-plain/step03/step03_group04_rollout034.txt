import numpy as np
import scipy.optimize as optimize
import time
import random

    np.random.seed(seed)

    # Function to evaluate the C5 value
    def evaluate_C5(h, n_points, dx):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        c5 = max_corr * dx
        return c5
    
    # Function to compute the integral constraint
    def constraint_sum(h, n_points):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1 => sum(h) = n_points / 2
    
    # Define bounds: h[i] must be in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial optimization
    dx = 2.0 / n_points
    
    # Initial optimization with more iterations
    result = optimize.minimize(
        lambda h: evaluate_C5(h, n_points, dx),
        h_initial,
        bounds=bounds,
        constraints=[{"type": "eq", "fun": lambda h: constraint_sum(h, n_points)}],
        method="SLSQP",
        options={"ftol": 1e-9, "maxiter": 200}
    )
    
    best_h = result.x
    best_c5 = evaluate_C5(best_h, n_points, dx)
    
    # Local search with a different approach
    for _ in range(5):
        # Define a new perturbation strategy
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0.0, 1.0)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        # Evaluate the perturbed solution
        result_perturbed = optimize.minimize(
            lambda h: evaluate_C5(h, n_points, dx),
            perturbed_h,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": lambda h: constraint_sum(h, n_points)}],
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 100}
        )
        
        if evaluate_C5(result_perturbed.x, n_points, dx) < best_c5:
            best_h = result_perturbed.x
            best_c5 = evaluate_C5(best_h, n_points, dx)
    
    # Refinement with genetic algorithm
    def genetic_optimize(population_size=50, generations=20, mutation_rate=0.1):
        population = [np.random.uniform(0, 1, n_points) * (n_points / 2) / np.sum(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
        best_individual = None
        best_c5 = float('inf')

            fitness = [evaluate_C5(ind, n_points, dx) for ind in population]

            selected = [best_individual] + [np.random.choice(population) for _ in range(population_size - 1)]

            new_population = []

                        child[i] += np.random.uniform(-0.1, 0.1)

            best_c5 = evaluate_C5(best_h, n_points, dx)

    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.5:
        # Try with double the number of points
        n_points_new = int(2 * n_points)
        dx_new = 2.0 / n_points_new
        
        # Resample the current best solution to the new grid
        x = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, n_points_new)
        h_new = np.interp(x_new, x, best_h)

        h_new = h_new * (n_points_new / 2) / np.sum(h_new)
        
        # Evaluate the new solution
        def evaluate_C5_new(h):

            return c5
        
        # Optimization for the new number of points
        result_new = optimize.minimize(
            evaluate_C5_new,
            h_new,
            bounds=[(0.0, 1.0) for _ in range(n_points_new)],
            constraints=[{"type": "eq", "fun": lambda h: np.sum(h) - n_points_new / 2}],
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 50}
        )
        
        # Update the best solution
        best_h = result_new.x
        best_c5 = evaluate_C5_new(best_h)
        n_points = n_points_new
    
    # Return the best solution found
    return best_h, best_c5, n_points
