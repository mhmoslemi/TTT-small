import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=60, **kwargs):

    # Function to evaluate C5
    def evaluate_C5(h, n_points, dx):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        c5 = max_corr * dx
        return c5
    
    # Function to compute the integral constraint
    def constraint_sum(h, n_points):
        return np.sum(h) - n_points / 2
    
    # Function to optimize using SLSQP
    def optimize_solution(h_initial, n_points, dx, seed):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = optimize.minimize(
            lambda h: evaluate_C5(h, n_points, dx),
            h_initial,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": lambda h: constraint_sum(h, n_points)}],
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 200}
        )
        return result.x
    
    # Function to generate perturbed solutions
    def perturb_solution(h, n_points):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = np.clip(h + perturbation, 0.0, 1.0)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        return perturbed_h
    
    # Function to resample and optimize for higher resolution
    def upscale_solution(h, n_points):
        x = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, 2 * n_points)
        h_new = np.interp(x_new, x, h)
        h_new = h_new * (2 * n_points / 2) / np.sum(h_new)
        return h_new
    
    # Initial optimization
    dx = 2.0 / n_points
    best_h = optimize_solution(h_initial, n_points, dx, seed)
    best_c5 = evaluate_C5(best_h, n_points, dx)
    
    # Perturbation and local search
    for _ in range(5):
        perturbed_h = perturb_solution(best_h, n_points)
        result = optimize.minimize(
            lambda h: evaluate_C5(h, n_points, dx),
            perturbed_h,
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{"type": "eq", "fun": lambda h: constraint_sum(h, n_points)}],
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 100}
        )
        if evaluate_C5(result.x, n_points, dx) < best_c5:
            best_h = result.x
            best_c5 = evaluate_C5(best_h, n_points, dx)
    
    # Local search
    for _ in range(5):
        perturbed_h = perturb_solution(best_h, n_points)
        result = optimize.minimize(
            lambda h: evaluate_C5(h, n_points, dx),
            perturbed_h,
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{"type": "eq", "fun": lambda h: constraint_sum(h, n_points)}],
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 50}
        )
        if evaluate_C5(result.x, n_points, dx) < best_c5:
            best_h = result.x
            best_c5 = evaluate_C5(best_h, n_points, dx)
    
    # Check if we have time to upscale
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.5:
        n_points_new = 2 * n_points
        dx_new = 2.0 / n_points_new
        h_upscaled = upscale_solution(best_h, n_points)
        result = optimize.minimize(
            lambda h: evaluate_C5(h, n_points_new, dx_new),
            h_upscaled,
            bounds=[(0.0, 1.0) for _ in range(n_points_new)],
            constraints=[{"type": "eq", "fun": lambda h: constraint_sum(h, n_points_new)}],
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 50}
        )
        if evaluate_C5(result.x, n_points_new, dx_new) < best_c5:
            best_h = result.x
            best_c5 = evaluate_C5(best_h, n_points_new, dx_new)
            n_points = n_points_new
    
    return best_h, best_c5, n_points
