import numpy as np
import scipy.optimize as optimize
import time

# initial_h_values is a starting point with n_points samples

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Start with the initial construction
    h_initial = initial_h_values
    n_points_initial = len(h_initial)
    dx_initial = 2.0 / n_points_initial
    
    # Function to evaluate the C5 value
    def evaluate_C5(h, dx):
        h = np.array(h)
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Get the maximum value
        max_corr = np.max(corr)
        # Multiply by dx
        c5 = max_corr * dx
        return c5
    
    # Function to compute the integral constraint
    def constraint_sum(h, n_points):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1 => sum(h) = n_points / 2
    
    # Define bounds: h[i] must be in [0, 1]
    def get_bounds(n_points):
        return [(0.0, 1.0) for _ in range(n_points)]
    
    # Function to perform optimization
    def optimize_h(h_start, dx, n_points):
        result = optimize.minimize(
            lambda h: evaluate_C5(h, dx),
            h_start,
            bounds=get_bounds(n_points),
            constraints=[{"type": "eq", "fun": lambda h: constraint_sum(h, n_points)}],
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 200}
        )
        return result.x, evaluate_C5(result.x, dx)
    
    # Initial optimization
    best_h, best_c5 = optimize_h(h_initial, dx_initial, n_points_initial)
    
    # Try multiple different initializations to avoid local optima
    for _ in range(5):
        # Create random perturbations
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points_initial)
        h_perturbed = np.clip(best_h + perturbation, 0.0, 1.0)
        # Adjust to maintain sum constraint
        h_perturbed = h_perturbed * (n_points_initial / 2) / np.sum(h_perturbed)
        h_perturbed, c5_perturbed = optimize_h(h_perturbed, dx_initial, n_points_initial)
        if c5_perturbed < best_c5:
            best_h, best_c5 = h_perturbed, c5_perturbed
    
    # If time remains, consider increasing the number of points
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.5:
        # Try with double the number of points
        n_points_new = int(1.5 * n_points_initial)
        dx_new = 2.0 / n_points_new
        
        # Resample the current best solution to the new grid
        x = np.linspace(0, 2, n_points_initial)
        x_new = np.linspace(0, 2, n_points_new)
        h_new = np.interp(x_new, x, best_h)
        
        # Ensure the sum constraint is satisfied
        h_new = h_new * (n_points_new / 2) / np.sum(h_new)
        
        # Optimization for the new number of points
        result_new = optimize.minimize(
            lambda h: evaluate_C5(h, dx_new),
            h_new,
            bounds=get_bounds(n_points_new),
            constraints=[{"type": "eq", "fun": lambda h: constraint_sum(h, n_points_new)}],
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 100}
        )
        
        # Update the best solution
        best_h = result_new.x
        best_c5 = evaluate_C5(best_h, dx_new)
        n_points_initial = n_points_new
    
    # Return the best solution found
    return best_h, best_c5, n_points_initial
