```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
from scipy.signal import correlate
from scipy.optimize import minimize, Bounds, LinearConstraint, SLSQP

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess with a structured pattern
    initial_h_values = np.zeros(100)
    initial_h_values[::2] = 1.0
    np.random.shuffle(initial_h_values)
    initial_h_values = (initial_h_values * 100 / 2).astype(np.float64)
    
    n_points = 100
    h_values = initial_h_values.copy()
    
    def evaluate_c5(h):
        h = h.reshape(-1)
        dx = 2.0 / len(h)
        conv = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(conv) * dx
        return max_overlap
    
    def perturb_and_refine(h, n_points, n_iterations=10):
        best_h = h.copy()
        best_c5 = evaluate_c5(best_h)
        for _ in range(n_iterations):
            # Perturb
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            h_perturbed = np.clip(best_h + perturbation, 0.0, 1.0)
            h_perturbed = h_perturbed * (n_points / 2.0)
            h_perturbed = h_perturbed / (np.sum(h_perturbed) * 2.0 / n_points)
            
            # Optimize
            x0 = h_perturbed.copy()
            bounds = Bounds(0.0, 1.0)
            constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
            result = minimize(evaluate_c5, x0=x0, bounds=bounds, constraints=constraint, method='SLSQP')
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        return best_h, best_c5
    
    # Start with the initial guess
    start_time = time.time()
    best_h, best_c5 = perturb_and_refine(h_values, n_points)
    
    # If time allows, try a few more iterations with different perturbations
    if time.time() - start_time < budget_s:
        for _ in range(5):
            # Perturb
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h_perturbed = np.clip(best_h + perturbation, 0.0, 1.0)
            h_perturbed = h_perturbed * (n_points / 2.0)
            h_perturbed = h_perturbed / (np.sum(h_perturbed) * 2.0 / n_points)
            
            # Optimize
            x0 = h_perturbed.copy()
            bounds = Bounds(0.0, 1.0)
            constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
            result = minimize(evaluate_c5, x0=x0, bounds=bounds, constraints=constraint, method='SLSQP')
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    
    return (best_h, best_c5, n_points)
```
