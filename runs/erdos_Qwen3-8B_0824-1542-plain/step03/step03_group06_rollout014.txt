import numpy as np
import scipy.optimize as optimize
import time
from scipy.signal import correlate
from scipy.optimize import Bounds, LinearConstraint, minimize

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Use more points for better accuracy
    dx = 2.0 / n_points
    
    # Initial construction based on a structured pattern
    initial_h = np.zeros(n_points)
    initial_h[::4] = 1.0
    initial_h[1::4] = 0.5
    initial_h = np.random.permutation(initial_h)
    
    def evaluate_c5(h):
        conv = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(conv) * dx
        return max_overlap
    
    def optimize_h(initial_h, n_points, max_iterations=50):
        best_h = initial_h.copy()
        best_c5 = evaluate_c5(best_h)
        
        for _ in range(max_iterations):
            # Apply a small random perturbation
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h = best_h + perturbation
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2.0)
            h = h / (np.sum(h) * 2.0 / n_points)
            
            # Set up optimization problem
            bounds = Bounds(0.0, 1.0)
            constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
            
            # Use SLSQP to minimize the maximum overlap
            result = minimize(evaluate_c5, x0=h, bounds=bounds, constraints=constraint, method='SLSQP')
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = optimize_h(initial_h, n_points)
    
    # Final refinement
    if time.time() - start_time < budget_s:
        # Apply a small random perturbation
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2.0)
        h = h / (np.sum(h) * 2.0 / n_points)
        
        # Final optimization
        bounds = Bounds(0.0, 1.0)
        constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
        result = minimize(evaluate_c5, x0=h, bounds=bounds, constraints=constraint, method='SLSQP')
        
        if result.success:
            best_c5 = result.fun
            best_h = result.x
    
    return (best_h, best_c5, n_points)
