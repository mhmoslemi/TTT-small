import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported helper functions
# evaluate_erdos_solution(h_values) computes C5
# initial_h_values is a starting point with n_points samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with a slightly higher resolution than the initial 376 samples
    n_points = 400
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()  # Use the initial construction
    
    # Resample the initial solution to the new grid
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    h_new = np.interp(x_new, x_initial, h_initial)
    h_new = h_new * (n_points / 2) / np.sum(h_new)
    
    # Function to evaluate C5
    def evaluate_C5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute integral constraint
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial optimization
    result = optimize.minimize(
        evaluate_C5,
        h_new,
        bounds=bounds,
        constraints=[{"type": "eq", "fun": constraint_sum}],
        method="SLSQP",
        options={"ftol": 1e-10, "maxiter": 200, "eps": 1e-8}
    )
    
    best_h = result.x
    best_c5 = evaluate_C5(best_h)
    
    # Try a few local improvements with random perturbations
    for _ in range(10):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0.0, 1.0)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        result_perturbed = optimize.minimize(
            evaluate_C5,
            perturbed_h,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": constraint_sum}],
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 100, "eps": 1e-8}
        )
        
        if evaluate_C5(result_perturbed.x) < best_c5:
            best_h = result_perturbed.x
            best_c5 = evaluate_C5(best_h)
    
    # Check if we have time to try a higher resolution
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1.0:
        n_points_new = int(1.5 * n_points)
        dx_new = 2.0 / n_points_new
        
        # Resample the current best solution to the new grid
        x_current = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, n_points_new)
        h_new = np.interp(x_new, x_current, best_h)
        h_new = h_new * (n_points_new / 2) / np.sum(h_new)
        
        # Define new bounds and constraints
        new_bounds = [(0.0, 1.0) for _ in range(n_points_new)]
        new_constraint = {"type": "eq", "fun": lambda h: np.sum(h) - n_points_new / 2}
        
        # Evaluate the new solution
        def evaluate_C5_new(h):
            corr = np.correlate(h, 1 - h, mode="full")
            max_corr = np.max(corr)
            return max_corr * dx_new
        
        # Optimization for the new number of points
        result_new = optimize.minimize(
            evaluate_C5_new,
            h_new,
            bounds=new_bounds,
            constraints=[new_constraint],
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 150, "eps": 1e-8}
        )
        
        # Update the best solution
        best_h = result_new.x
        best_c5 = evaluate_C5_new(best_h)
        n_points = n_points_new
    
    return best_h, best_c5, n_points
