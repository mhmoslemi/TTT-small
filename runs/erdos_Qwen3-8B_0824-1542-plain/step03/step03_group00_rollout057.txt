    def compute_c5(h_values):

        h = np.clip(h, 0, 1)

        c5 = np.max(corr) * dx
        return c5

        return np.sum(h) - (n_points / 2)
    
    # Objective function to minimize
    def objective(h):
        return compute_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]

    options = {

        'maxiter': 150,
        'disp': False
    }
    
    # Optimization using SLSQP
    def optimize_h(h_initial):
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, 
                          constraints={'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)}, 
                          options=options)
        if result.success:
            return result.x
        else:
            return h_initial

    def perturb_h(h, perturbation=0.05):
        h_perturbed = h.copy()
        for i in range(len(h_perturbed)):
            h_perturbed[i] += np.random.uniform(-perturbation, perturbation)
            h_perturbed[i] = np.clip(h_perturbed[i], 0, 1)
