import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
from scipy.interpolate import interp1d

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Use the provided initial construction
    initial_h_values = np.load('initial_h_values.npy')  # Assuming this file exists with the initial construction
    n_points = len(initial_h_values)
    
    def evaluate_c5(h):
        h = h.reshape(-1)
        dx = 2.0 / len(h)
        conv = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(conv) * dx
        return max_overlap
    
    def optimize_h(initial_h, n_points, max_iterations=50):
        best_h = initial_h.copy()
        best_c5 = evaluate_c5(best_h)
        
        for _ in range(max_iterations):
            # Perturb the current best solution
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            h = best_h + perturbation
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2.0)
            h = h / (np.sum(h) * 2.0 / n_points)
            
            # Use differential evolution for global optimization
            bounds = Bounds(0.0, 1.0)
            constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
            result = optimize.differential_evolution(evaluate_c5, bounds, constraints=constraint, 
                                                     args=(n_points,), popsize=15, iterations=20, 
                                                     tol=1e-6, seed=seed)
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        
        return best_h, best_c5
    
    # Start with the initial guess
    best_h, best_c5 = optimize_h(initial_h_values, n_points)
    
    # If time allows, try a few more iterations
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Perturb the current best and refine
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2.0)
        h = h / (np.sum(h) * 2.0 / n_points)
        
        # Optimize
        result = optimize.differential_evolution(evaluate_c5, Bounds(0.0, 1.0), 
                                                 LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2), 
                                                 args=(n_points,), popsize=15, iterations=20, 
                                                 tol=1e-6, seed=seed + 1)
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return (best_h, best_c5, n_points)
