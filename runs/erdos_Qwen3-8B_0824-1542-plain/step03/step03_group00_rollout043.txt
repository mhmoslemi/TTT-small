    def compute_c5(h_values):

        c5 = np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]

    options = {

        'disp': False
    }
    
    # Optimization using SLSQP with improved perturbation and exploration
    def optimize_h(h_initial):
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, 
                          constraints={'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)},

        if result.success:
            return result.x
        else:
            return h_initial

    def perturb_h(h, perturbation=0.05, num_perturbations=5):
        perturbations = []
