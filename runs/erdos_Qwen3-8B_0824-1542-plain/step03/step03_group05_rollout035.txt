import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess with a structured pattern
    initial_h_values = np.zeros(100)
    initial_h_values[::2] = 1.0
    np.random.shuffle(initial_h_values)
    initial_h_values = (initial_h_values * 100 / 2).astype(np.float64)
    
    n_points = 100
    h_values = initial_h_values.copy()
    
    def evaluate_c5(h):
        h = h.reshape(-1)
        dx = 2.0 / len(h)
        conv = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(conv) * dx
        return max_overlap
    
    def optimize_h(initial_h, n_points, max_iterations=50):
        best_h = initial_h.copy()
        best_c5 = evaluate_c5(best_h)
        
        for _ in range(max_iterations):
            # Use differential evolution to explore the space
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
            
            result = differential_evolution(evaluate_c5, bounds, constraints=constraint, 
                                           popsize=15, strategy='best1bin', mutation=0.8, 
                                           recombination=0.9, tol=1e-6, seed=seed, 
                                           maxiter=20, disp=False)
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = optimize_h(h_values, n_points)
    
    # If time allows, perform additional refinement
    if time.time() - start_time < budget_s:
        for _ in range(5):
            # Use local optimization to refine the solution
            bounds = Bounds(0.0, 1.0)
            constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
            
            result = optimize.minimize(evaluate_c5, x0=best_h, bounds=bounds, 
                                       constraints=constraint, method='L-BFGS-B')
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    
    # Scale to ensure the integral is exactly 1
    best_h = best_h * (n_points / 2.0) / (np.sum(best_h) * 2.0 / n_points)
    
    return (best_h, best_c5, n_points)
