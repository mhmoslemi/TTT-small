            constraints=[{"type": "eq", "fun": lambda h: constraint_sum(h, n_points)}],
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 100, "eps": 1e-8}
        )
        return res.x, evaluate_C5(res.x, n_points, dx)
