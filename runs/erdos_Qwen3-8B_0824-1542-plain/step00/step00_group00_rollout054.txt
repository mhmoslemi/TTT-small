import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 128
    # Adjust n_points for speed and accuracy
    n_points = min(128, int(budget_s * 10))
    
    # Define initial guess based on provided initial_h_values
    h_initial = np.zeros(n_points)
    h_initial[:len(initial_h_values)] = initial_h_values[:n_points]
    h_initial = np.pad(h_initial, (0, n_points - len(h_initial)), mode='constant')

    # Normalize to meet the integral constraint
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

    # Convert to variables
    h = cp.Variable(n_points, nonneg=True, boolean=False)
    
    # Objective function: minimize the maximum overlap integral
    # We use a convex relaxation to approximate the max function
    overlaps = []
    for k in range(1, n_points):
        overlap = cp.multiply(h, 1 - h[k:])
        overlaps.append(overlap)
    
    # Compute the maximum overlap
    max_overlap = cp.Max(overlaps)
    
    # Constraint: integral of h is 1
    constraint = cp.sum(h) == n_points / 2
    
    # Optimization problem
    problem = cp.Problem(cp.Minimize(max_overlap), [constraint])
    
    # Solve the problem
    try:
        problem.solve(solver=cp.SCS, eps=1e-4, time_limit=budget_s - (time.time() - start_time))
    except:
        pass
    
    # Extract the solution
    h_values = h.value
    if h_values is None:
        h_values = h_initial
    
    # Evaluate the solution
    c5_bound = evaluate_erdos_solution(h_values)
    
    return h_values, c5_bound, n_points
