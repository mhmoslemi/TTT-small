def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with fewer points for faster computation
    dx = 2.0 / n_points
    h = np.array(initial_h_values[:n_points], dtype=np.float64)
    h = h / (dx * n_points) * n_points / 2  # Normalize to meet the integral constraint
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        h = np.pad(h, (0, len(h)), 'constant')  # Pad to handle full convolution
        correlation = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlation)
    
    # Function to compute the integral constraint
    def integral_constraint(h):
        return np.sum(h) * dx - 1.0
    
    # Function to evaluate the C5 value
    def objective_func(h):
        return compute_c5(h)
    
    # Function to enforce integral constraint
    def constraint_func(h):
        return integral_constraint(h)
    
    # Initial solution
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Optimization parameters
    max_iterations = 50
    tolerance = 1e-5
    
    # Try different numbers of points from 100 to 400
    for n_points in [100, 200, 300, 400]:
        if time.time() - start_time > budget_s:
            break
        
        dx = 2.0 / n_points
        h = np.random.rand(n_points)  # Random initial guess
        h = h * (n_points / 2)  # Scale to meet integral constraint
        h = h / (dx * n_points) * n_points / 2  # Normalize to meet the integral constraint
        
        # Run optimization
        for _ in range(max_iterations):
            if time.time() - start_time > budget_s:
                break
            
            # Use a simple gradient descent with constraints
            h = h.copy()
            h = h / (dx * n_points) * n_points / 2  # Ensure integral is met

            if c5 < best_c5:

                best_h = h.copy()
        
        # After optimization for this n_points, update best solution
        if compute_c5(best_h) < best_c5:
            best_c5 = compute_c5(best_h)
