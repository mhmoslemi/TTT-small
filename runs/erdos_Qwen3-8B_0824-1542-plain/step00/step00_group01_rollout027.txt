    def evaluate_c5(h):
        h = np.array(h)
        h = np.concatenate([h, h])  # For full convolution
        shifted_h = np.roll(h, shift=1)
        overlap = np.correlate(h, 1 - shifted_h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Function to calculate the cost (C5) and constraints
    def objective(h_values):
        return evaluate_c5(h_values)
    
    # Constraints: sum(h) == n_points / 2

    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    
    # Setup optimization
    h = cp.Variable(n_points)
    objective = cp.Minimize(evaluate_c5(h))
    constraints = [cp.sum(h) == n_points / 2]

    h_initial_opt = h_initial.copy()
    
    # Optimization loop
    start_time = time.time()
    best_h = h_initial_opt
    best_c5 = evaluate_c5(h_initial_opt)

    for _ in range(5):
        # Solve with CVXPY
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.SCS, warm_start=True, verbose=False, max_iter=100)
            if prob.status == 'optimal':
                h_values = np.round(h.value, 5)
                c5 = evaluate_c5(h_values)
                if c5 < best_c5:
                    best_h = h_values
                    best_c5 = c5
            else:
                print(f"Optimization failed with status: {prob.status}")
                break

            print("Error during optimization")

        elapsed = time.time() - start_time
