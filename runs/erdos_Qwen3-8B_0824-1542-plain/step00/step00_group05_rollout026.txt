import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-loaded functions
# initial_h_values = ... (n=70 samples)
# evaluate_erdos_solution = ... (function to compute C_5 from h_values)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 200  # number of samples
    dx = 2.0 / n_points
    h_initial = np.zeros(n_points)
    
    # Start from the initial construction (rescaled if needed)
    if hasattr(initial_h_values, '__iter__') and len(initial_h_values) == n_points:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.random.rand(n_points)
    
    # Rescale to satisfy the integral constraint
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Objective function: minimize the maximum overlap
    def objective(h):
        h = np.clip(h, 0, 1)
        conv = np.correlate(h, 1 - h, mode="full")
        overlap = conv * dx
        return np.max(overlap)
    
    # Optimization: minimize the maximum overlap
    # Use a genetic algorithm with local refinement
    def optimize_h(h_initial):
        # Genetic algorithm
        pop_size = 50
        num_generations = 50
        mutation_rate = 0.1
        elite_size = 5
        
        population = np.random.rand(pop_size, n_points)
        population = np.clip(population * (n_points / 2), 0, n_points / 2)
        population = population / (n_points / 2)
        
        for gen in range(num_generations):
            fitness = np.array([objective(ind) for ind in population])
            elite_indices = np.argsort(fitness)[:elite_size]
            elite = population[elite_indices]
            
            # Crossover
            new_population = []
            for _ in range(pop_size - elite_size):
                parent1 = np.random.choice(elite)
                parent2 = np.random.choice(elite)
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                new_population.append(child)
            
            # Mutation
            for i in range(len(new_population)):
                child = new_population[i]
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[j] = np.random.rand()
                new_population[i] = np.clip(child, 0, 1)
            
            population = np.vstack([elite, new_population])
        
        best_h = population[np.argmin(fitness)]
        return best_h
    
    # Optimize
    best_h = optimize_h(h_initial)
    best_c5 = objective(best_h)
    
    # Refine with local optimization
    def local_optimize(h):
        h = h.copy()
        h = np.clip(h, 0, 1)
        result = opt.minimize(lambda x: objective(x), x0=h, method='L-BFGS-B', bounds=(0, 1))
        return result.x
    
    best_h = local_optimize(best_h)
    best_c5 = objective(best_h)
    
    # Check time
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return (best_h, best_c5, n_points)
    
    # Final refinement
    for _ in range(5):
        best_h = local_optimize(best_h)
        best_c5 = objective(best_h)
        if elapsed > budget_s - 1:
            break
    
    return (best_h, best_c5, n_points)
