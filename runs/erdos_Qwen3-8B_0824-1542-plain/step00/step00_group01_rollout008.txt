import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Initial setup
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    dx = 2.0 / n_points_initial
    sum_h_initial = np.sum(h_initial) * dx
    if not np.isclose(sum_h_initial, 1.0):
        raise ValueError("Initial h_values doesn't satisfy the integral constraint")
    
    # Try with n_points_initial
    n_points = n_points_initial
    h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # Define function to compute C5
    def compute_c5(h_values):
        h = h_values.copy()
        dx = 2.0 / len(h)
        h = np.repeat(h, 2)  # Pad to handle wrap-around
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    # Define function to compute gradient for optimization
    def compute_gradient(h_values):
        h = h_values.copy()
        dx = 2.0 / len(h)
        h = np.repeat(h, 2)  # Pad to handle wrap-around
        conv = np.correlate(h, 1 - h, mode="full")
        grad = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            for i in range(len(h) - k - 1):
                grad[i] -= (1 - h[i + k]) * dx
                grad[i + k] += h[i] * dx
        return grad
    
    # Define function to compute the integral constraint
    def constraint(h_values):
        return np.sum(h_values) * (2.0 / len(h_values)) - 1.0
    
    # Optimization parameters
    max_iterations = 100
    learning_rate = 0.01
    tolerance = 1e-5
    elapsed_time = 0
    
    # Start timer
    start_time = time.time()
    
    for _ in range(max_iterations):
        # Compute current C5
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient
        grad = compute_gradient(h)
        
        # Update h using gradient descent
        h = h - learning_rate * grad
        
        # Ensure h stays within [0, 1]
        h = np.clip(h, 0.0, 1.0)
        
        # Check constraint
        constraint_val = constraint(h)
        if not np.isclose(constraint_val, 0.0, atol=1e-6):
            # If constraint is not satisfied, adjust h to meet it
            h = h * (1.0 / np.sum(h) * (len(h) / 2.0))
        
        # Check time budget
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    # Evaluate best solution
    best_c5 = compute_c5(best_h)
    
    # Return the best solution and C5 bound
    return (best_h, best_c5, n_points)
