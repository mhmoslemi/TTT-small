    h = np.random.rand(n_points) * 2 - 1  # Initial random guess
    h = np.clip(h, 0, 1)

    c5_candidate = evaluate_erdos_solution(h)

    def objective(h_flat):
        h = h_flat.reshape((n_points, 1))
        integral = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(integral)

    h_opt = np.clip(h_opt, 0, 1)

    bounds = [(0, 1) for _ in range(n_points)]

    result = optimize.minimize(

        h_opt.flatten(),
        bounds=bounds,
        method="L-BFGS-B",

    if result.success:
