import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros(2 * n_points)
    shifted_h[n_points:] = h
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def initial_h_values():
    # Returns an initial construction for n_points = 45

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with the initial h_values
    h_initial = initial_h_values()
    n_points = len(h_initial)
    
    # Function to evaluate the C5 value
    def evaluate(h):
        dx = 2.0 / n_points
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Function to compute the integral constraint
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization with simulated annealing
    def optimize_h():
        h = h_initial.copy()
        best_c5 = evaluate(h)
        best_h = h.copy()
        
        # Parameters for simulated annealing
        temp = 1.0
        cooling_rate = 0.995
        iterations = 10000
        for _ in range(iterations):
            if temp < 0.001:
                break
            # Perturb the h values
            idx = np.random.randint(0, n_points)
            h[idx] += np.random.uniform(-0.1, 0.1)
            h = np.clip(h, 0.0, 1.0)
            
            # Compute current c5
            current_c5 = evaluate(h)
            
            # Compute the previous c5
            prev_c5 = evaluate(best_h)
            
            # Acceptance probability
            if current_c5 < prev_c5:
                best_h = h.copy()
                best_c5 = current_c5
            else:
                prob = np.exp((prev_c5 - current_c5) / temp)
                if np.random.rand() < prob:
                    best_h = h.copy()
                    best_c5 = current_c5
            
            temp *= cooling_rate
        
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = optimize_h()
    
    # If we have time, try to refine using local optimization
    if time.time() - start_time < budget_s - 5:
        # Local optimization with constraints
        h = cp.Variable(n_points)
        constraints = [constraint_sum(h) == 0]
        objective = cp.Minimize(evaluate(h))
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.SCS)
            best_h = h.value
            best_c5 = evaluate(best_h)
        except:
            pass
    
    return best_h, best_c5, n_points
