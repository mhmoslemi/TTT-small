def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 42
    
    # Initial guess
    h = initial_h_values.copy()
    
    # Convert to float
    h = h.astype(float)
    
    # Initial evaluation
    c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = c5
    
    # Try increasing number of points up to a maximum of 1000
    max_points = 1000
    for n_points in range(42, max_points + 1, 10):
        # Initialize h with some randomness
        h = np.random.rand(n_points)
        # Normalize to satisfy the integral constraint
        h = h * (n_points / 2.0) / np.sum(h)
        # Ensure h is in [0, 1]
        h = np.clip(h, 0.0, 1.0)
        
        # Convert to float
        h = h.astype(float)
        
        # Optimization using CVXPY
        try:
            # Define variables
            x = cp.Variable(n_points)
            
            # Objective: minimize the maximum overlap
            # We aim to find a sequence that minimizes the maximum overlap
            # So we need to minimize the maximum of the overlaps
            # To do this, we can use a disciplined convex programming approach
            # First, compute the overlaps for all possible k (shifts)
            
            # Since the function is periodic, we consider k from 0 to n_points-1
            # The overlap for k is the integral of h(x) * (1 - h(x + k mod n_points)) dx
            # We approximate this using the discrete samples
            # Since the function is defined on [0, 2], and we have n_points points, dx = 2 / n_points
            # So the overlap for each k is sum(h[i] * (1 - h[(i + k) % n_points])) * dx
            
            # Evaluate the overlaps for all k
            overlaps = np.zeros(n_points)
            for k in range(n_points):
                overlaps[k] = np.dot(h, 1 - h[(np.arange(n_points) + k) % n_points]) * (2.0 / n_points)
            
            # Define the maximum overlap
            max_overlap = cp.Max(overlaps)
            
            # Define the objective: minimize the maximum overlap
            obj = cp.Minimize(max_overlap)
            
            # Define constraints
            constraints = [
                cp.sum(x) == n_points / 2.0,
                0 <= x,
                x <= 1
            ]
            
            # Solve the problem
            problem = cp.Problem(obj, constraints)
            start_time = time.time()
            result = problem.solve(solver=cp.MOSEK, verbose=False)
            elapsed = time.time() - start_time
            
            if elapsed > budget_s:
                break
            
            if result is not None and result < best_c5:
                best_c5 = result
                best_h = x.value.copy()
        
        except:
            pass
    
    # After optimization, convert the best_h to a numpy array
    best_h = best_h.astype(float)
    
    return best_h, best_c5, len(best_h)
