    n_points = 100  # Choose a manageable number of points for the search
    h_values = np.zeros(n_points)
    
    # Initial guess: use the provided initial_h_values scaled to fit n_points
    if n_points < len(initial_h_values):
        h_values = initial_h_values[:n_points]
    elif n_points > len(initial_h_values):
        h_values = np.tile(initial_h_values, n_points // len(initial_h_values) + 1)[:n_points]
    
    # Normalize to satisfy the constraint sum(h) * dx = 1
    dx = 2.0 / n_points
    h_values = h_values / np.sum(h_values) * (n_points / 2)
    
    # Optimization using gradient descent
    learning_rate = 0.01
    num_iterations = 1000
    for _ in range(num_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate current overlap
        current_c5 = evaluate_erdos_solution(h_values)
        
        # Compute gradients for each point
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Compute contribution to C₅ from h[i]
            for k in range(n_points):
                overlap = np.dot(h_values, 1 - h_values[k: k + n_points])
                grad[i] += (overlap - current_c5) * (1 - h_values[k: k + n_points]) * dx
        
        # Update h values
        h_values -= learning_rate * grad
        
        # Ensure h remains in [0, 1]
        h_values = np.clip(h_values, 0, 1)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h_values)
    return h_values, final_c5, n_points
