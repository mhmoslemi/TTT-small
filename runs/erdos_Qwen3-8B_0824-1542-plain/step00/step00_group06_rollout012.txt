```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Placeholder, actual data would be here

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with initial solution
    n_points = len(initial_h_values)
    h_values = np.copy(initial_h_values)
    best_c5 = evaluate_erdos_solution(h_values)
    best_h = h_values
    
    # Parameters for optimization
    max_iterations = 100
    step_size = 0.1
    cooling_rate = 0.995
    temperature = 100.0
    
    for iteration in range(max_iterations):
        start_time = time.time()
        elapsed = 0
        
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h_values)
        
        # Generate a neighboring solution
        perturbation = np.random.normal(0, step_size, size=n_points)
        h_values += perturbation
        np.clip(h_values, 0, 1, out=h_values)
        h_values = np.round(h_values, 5)  # Reduce floating point error
        
        # Enforce integral constraint
        total_integral = np.sum(h_values) * (2.0 / n_points)
        if abs(total_integral - 1.0) > 1e-6:
            # Adjust to maintain integral constraint
            h_values = h_values / total_integral * 1.0
            h_values = np.round(h_values, 5)
        
        # Evaluate the perturbed solution
        perturbed_c5 = evaluate_erdos_solution(h_values)
        
        # Acceptance probability
        if perturbed_c5 < current_c5:
            current_c5 = perturbed_c5
            h_values = np.copy(h_values)
        else:
            prob = np.exp((current_c5 - perturbed_c5) / temperature)
            if np.random.rand() < prob:
                current_c5 = perturbed_c5
                h_values = np.copy(h_values)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(h_values)
        
        # Update temperature
        temperature *= cooling_rate
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    return (best_h, best_c5, n_points)
```
