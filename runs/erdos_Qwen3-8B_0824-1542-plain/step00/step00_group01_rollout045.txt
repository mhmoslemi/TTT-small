```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp
import random

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode="full")) * (2 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # Assume this is a numpy array of shape (n_points,)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for faster evaluation
    max_points = 200  # Maximum points to consider

    # Define a function to evaluate the C5 value
    def evaluate_c5(h):
        nonlocal n_points
        if len(h) != n_points:
            h = np.resize(h, n_points)
        c5 = evaluate_erdos_solution(h)
        return c5

    # Initialize with the given initial_h_values
    initial_h = np.array(initial_h_values)
    if len(initial_h) > max_points:
        initial_h = initial_h[:max_points]
    initial_h = np.resize(initial_h, max_points)
    initial_h = initial_h / (max_points / 2)  # Normalize to satisfy the integral constraint

    # Try starting from the initial solution
    h = initial_h.copy()
    c5 = evaluate_c5(h)
    best_h = h
    best_c5 = c5
    print(f"Initial C5: {best_c5:.6f}")

    # Try improving the solution with local search
    def local_search(h):
        nonlocal best_c5, best_h, n_points
        for _ in range(100):  # Number of iterations
            if time.time() - start_time > budget_s - 1:
                break
            idx = np.random.randint(0, len(h))
            # Perturb the value
            perturbation = np.random.uniform(-0.2, 0.2)
            h[idx] += perturbation
            h[idx] = np.clip(h[idx], 0, 1)
            # Ensure the integral constraint is satisfied
            total_integral = np.sum(h) * (2 / len(h))
            if abs(total_integral - 1) > 1e-5:
                # Adjust the perturbation to satisfy the constraint
                diff = (1 - total_integral) * len(h) / 2
                h[idx] = np.clip(h[idx] + diff, 0, 1)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        return best_h, best_c5

    best_h, best_c5 = local_search(h)

    # Try increasing the number of points
    for n_points in range(100, max_points + 1, 50):
        if time.time() - start_time > budget_s - 1:
            break
        # Create a new h with n_points
        h = np.linspace(0, 2, n_points)
        h_vals = np.random.rand(n_points)
        h_vals = np.clip(h_vals, 0, 1)
        # Normalize to satisfy the integral constraint
        h_vals = h_vals / (n_points / 2)
        c5 = evaluate_c5(h_vals)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_vals.copy()

    # Try using a more sophisticated optimization approach
    # Use convex optimization to find a better h
    def optimize_h(n_points):
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1]
        constraints.append(cp.sum(h) == n_points / 2)
        c5 = cp.Maximize(cp.max(signal.correlate(h.value, 1 - h.value, mode="full") * (2 / n_points)))
        problem = cp.Problem(c5, constraints)
        problem.solve(solver=cp.ECOS)
        if problem.status != 'optimal':
            return None
        return h.value

    for n_points in range(150, max_points + 1, 50):
        if time.time() - start_time > budget_s - 1:
            break
        h_vals = optimize_h(n_points)
        if h_vals is not None:
            c5 = evaluate_c5(h_vals)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_vals.copy()

    print(f"Best C5: {best_c5:.6f}")
    return best_h, best_c5, len(best_h)
```
