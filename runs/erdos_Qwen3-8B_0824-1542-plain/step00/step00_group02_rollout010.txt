```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Placeholder for initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    n_points_initial = len(initial_h_values)
    h = initial_h_values.copy()
    
    # Try varying number of points to find a good balance
    n_points_list = [n_points_initial, n_points_initial - 5, n_points_initial + 5]
    best_c5 = float('inf')
    best_h = h
    
    for n_points in n_points_list:
        if n_points < 10:
            continue  # Too small for meaningful results
        if n_points > 1000:
            continue  # Too large for time constraints
        
        # Ensure h has the correct number of points
        if len(h) != n_points:
            h = np.zeros(n_points)
            h[:n_points_initial] = initial_h_values[:n_points_initial]
            h[n_points_initial:] = np.random.rand(n_points - n_points_initial)
        
        # Set up the optimization problem
        N = n_points
        dx = 2.0 / N
        
        # Optimization variables
        h_var = cp.Variable(N, nonneg=True, boolean=True)
        obj = cp.Minimize(0)  # Objective is to minimize the maximum overlap
        constraints = []
        
        # Constraint: Integral of h over [0, 2] must be 1
        constraints.append(cp.sum(h_var) * dx == 1)
        
        # Objective: Maximize the overlap for all shifts k
        # We use a max constraint by introducing a new variable
        max_overlap = cp.Variable()
        constraints.append(max_overlap >= cp.sum(h_var * (1 - h_var[(np.arange(N) + k) % N])) * dx for k in range(N - 1))
        obj = cp.Minimize(max_overlap)
        
        # Solve the problem
        try:
            problem = cp.Problem(obj, constraints)
            problem.solve(solver=cp.ECOS, verbose=False)
            if problem.status == 'optimal':
                h_opt = h_var.value
                c5 = max_overlap.value
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
        except:
            pass
        
        # Early termination if time is up
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_h, best_c5, len(best_h))
```
