```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points, can adjust if needed
    dx = 2.0 / n_points
    h = initial_h_values.copy()[:n_points]  # Start with initial values
    
    # Ensure the initial h satisfies the constraint sum(h) == n_points / 2
    h = np.array(h, dtype=np.float64)
    h = h / np.sum(h) * (n_points / 2)
    
    def objective_function(h_values):
        h = h_values.reshape(-1)
        # Compute the overlap integral
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def compute_c5(h_values):
        h = h_values.reshape(-1)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Function to optimize h
    def optimize_h(h, n_points, budget_s):
        start_time = time.time()
        best_h = h.copy()
        best_c5 = compute_c5(best_h)
        
        # Use L-BFGS-B for optimization
        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(
            objective_function,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'gtol': 1e-6, 'maxiter': 1000}
        )
        
        if result.success:
            best_h = result.x
            best_c5 = compute_c5(best_h)
        
        # Check if we have time left for further refinement
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time > 0.1:
            # Use a more refined optimization with fewer steps
            result_refined = opt.minimize(
                objective_function,
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                options={'gtol': 1e-8, 'maxiter': 100}
            )
            if result_refined.success:
                best_h = result_refined.x
                best_c5 = compute_c5(best_h)
        
        return best_h, best_c5
    
    # Run the optimization
    best_h, best_c5 = optimize_h(h, n_points, budget_s)
    
    # Ensure the final h satisfies the constraint sum(h) == n_points / 2
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    return best_h, best_c5, n_points
```
