```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 45
    h = initial_h_values.copy()
    
    # Define the overlap integral function
    def evaluate_overlap(h):
        dx = 2.0 / n_points
        h_shifted = np.zeros((n_points, n_points))
        for i in range(n_points):
            h_shifted[i, :] = h[(i + 1) % n_points:]
        overlap = np.sum(np.multiply(h, 1 - h_shifted), axis=1)
        return np.max(overlap) * dx
    
    # Define the objective function for optimization
    def objective(h):
        return evaluate_overlap(h)
    
    # Define constraints
    def constraints(h):
        return np.sum(h) - n_points / 2
    
    # Optimization setup
    opt_h = cp.Variable(n_points)
    objective_func = cp.Minimize(objective(opt_h))
    constraints_func = [cp.sum(opt_h) == n_points / 2, 0 <= opt_h, opt_h <= 1]
    
    # Solve the optimization problem
    start_time = time.time()
    prob = cp.Problem(objective_func, constraints_func)
    try:
        prob.solve(solver=cp.SCS, eps=1e-4, max_iter=1000)
    except:
        prob.solve(solver=cp.SCS, eps=1e-3, max_iter=500)
    
    if prob.status == 'optimal':
        h_opt = np.round(opt_h.value, 5)
    else:
        h_opt = h
    
    # Evaluate the overlap
    c5_bound = evaluate_overlap(h_opt)
    
    # Check if we can improve with more points
    if time.time() - start_time < budget_s - 5:
        # Try with 100 points
        n_points = 100
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = np.random.permutation(h)
        h = h / np.sum(h) * n_points / 2
        
        def evaluate_overlap_large(h):
            h_shifted = np.zeros((n_points, n_points))
            for i in range(n_points):
                h_shifted[i, :] = h[(i + 1) % n_points:]
            overlap = np.sum(np.multiply(h, 1 - h_shifted), axis=1)
            return np.max(overlap) * dx
        
        def objective_large(h):
            return evaluate_overlap_large(h)
        
        def constraints_large(h):
            return np.sum(h) - n_points / 2
        
        opt_h_large = cp.Variable(n_points)
        objective_func_large = cp.Minimize(objective_large(opt_h_large))
        constraints_func_large = [cp.sum(opt_h_large) == n_points / 2, 0 <= opt_h_large, opt_h_large <= 1]
        
        prob_large = cp.Problem(objective_func_large, constraints_func_large)
        try:
            prob_large.solve(solver=cp.SCS, eps=1e-4, max_iter=1000)
        except:
            prob_large.solve(solver=cp.SCS, eps=1e-3, max_iter=500)
        
        if prob_large.status == 'optimal':
            h_opt_large = np.round(opt_h_large.value, 5)
        else:
            h_opt_large = h
        
        # Evaluate the overlap for large n_points
        c5_bound_large = evaluate_overlap_large(h_opt_large)
        c5_bound = min(c5_bound, c5_bound_large)
        h_opt = h_opt_large
    
    return h_opt, c5_bound, n_points
```
