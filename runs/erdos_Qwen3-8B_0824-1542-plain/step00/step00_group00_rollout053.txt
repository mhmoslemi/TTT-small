import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint

def run(seed=42, budget_s=60, **kwargs):
    # Initial setup
    np.random.seed(seed)
    n_points = 100  # Use 100 points for efficiency and reasonable accuracy
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    
    # If initial_h_values is provided, use it; otherwise, start with a uniform distribution
    if "initial_h_values" in globals():
        initial_h_values = np.array(initial_h_values)
        if len(initial_h_values) != n_points:
            initial_h_values = np.zeros(n_points)
        initial_h_values = np.clip(initial_h_values, 0, 1)
    
    # Create initial guess
    h_initial = initial_h_values.copy()
    sum_h_initial = np.sum(h_initial)
    if abs(sum_h_initial - n_points / 2) > 1e-6:
        h_initial = np.random.uniform(0, 1, n_points)
        sum_h_initial = np.sum(h_initial)
        scale = (n_points / 2) / sum_h_initial
        h_initial *= scale
        h_initial = np.clip(h_initial, 0, 1)

    # Define the objective function
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h.reshape(n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)

    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Bounds for h
    bounds = Bounds(0, 1)

    # Optimization settings
    max_iter = 100
    tol = 1e-6

    # Start timing
    start_time = time.time()

    # Optimization using differential evolution
    result = optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=max_iter,
        popsize=15,
        tol=tol,
        mutation=(0.5, 1),
        recombination=0.8,
        disp=False,
        x0=h_initial
    )

    # Extract best solution
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_c5 = objective(best_h)

    # Ensure we stay within the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s - 1:
        return (best_h, best_c5, n_points)

    # If we still have time, refine with local optimization
    result_local = optimize.minimize(
        objective,
        best_h,
        bounds=bounds,
        method='L-BFGS-B',
        tol=tol,
        options={'maxiter': 50}
    )
    best_h = result_local.x
    best_h = np.clip(best_h, 0, 1)
    best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
