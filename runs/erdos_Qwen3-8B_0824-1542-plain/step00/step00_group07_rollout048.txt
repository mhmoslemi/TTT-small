def run(seed=42, budget_s=60, **kwargs):

    # Try different n_points values

    best_c5 = float('inf')

            h = h * (n_points / 2) / np.sum(h)

        # Initial guess from previous best

        constraints = [
            cp.Minimize(0),
            cp.sum(h_var) == n_points / 2,
            0 <= h_var <= 1
        ]
        
        # Objective function to minimize the max overlap
        overlap = cp.Maximize(cp.sum(cp.multiply(h_var, 1 - h_var)))

        prob.solve(solver=cp.CVXOPT, verbose=False)

    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h, best_c5, n_points)
