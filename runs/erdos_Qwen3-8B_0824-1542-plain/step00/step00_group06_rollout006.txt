import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
initial_h_values = np.random.rand(95) * 2 - 1  # Placeholder for initial values
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:] = h_values
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use 100 points for a balance between speed and accuracy
    dx = 2.0 / n_points
    h = np.random.rand(n_points)
    h = h * 2 - 1  # Normalize to [-1, 1], then scale to [0, 1]
    h = (h + 1) / 2  # Scale to [0, 1]
    h = h * (n_points / 2)  # Ensure the integral is 1
    h = h / (n_points / 2)  # Normalize again
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()
    
    # Use a gradient-based optimizer
    def objective(x):
        x = x.reshape(n_points)
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1 - x, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(x):
        x = x.reshape(n_points)
        return np.sum(x) - n_points / 2
    
    # Setup optimization problem
    x = cp.Variable(n_points, nonneg=True, boolean=False)
    objective_func = cp.Max(cp.Max(cp.convolution(x, 1 - x, mode="full")) * dx)
    constraints = [cp.sum(x) == n_points / 2]
    prob = cp.Problem(cp.Minimize(objective_func), constraints)
    
    # Run the optimization
    try:
        result = prob.solve(solver=cp.MOSEK, verbose=False)
        if result is not None:
            best_h = x.value
            best_c5 = result
    except:
        pass
    
    # Fallback to local optimization
    if best_c5 is None or best_c5 > evaluate_erdos_solution(best_h):
        result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)], tol=1e-4)
        best_h = result.x
        best_c5 = result.fun
    
    # Refine the solution with local search
    for _ in range(5):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        c5 = evaluate_erdos_solution(perturbed_h)
        if c5 < best_c5:
            best_h = perturbed_h
            best_c5 = c5
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure we stop before the budget runs out
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 1:
        # Further refinement if time allows
        result = optimize.minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)], tol=1e-4)
        best_h = result.x
        final_c5 = result.fun
    
    return best_h, final_c5, n_points
