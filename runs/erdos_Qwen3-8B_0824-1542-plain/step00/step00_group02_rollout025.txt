import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    h_initial = initial_h_values.copy()
    n_initial = len(h_initial)
    dx_initial = 2.0 / n_initial
    
    # Try improving with a smaller number of points for faster evaluation
    n_points = 100  # Start with 100 points for faster optimization
    dx = 2.0 / n_points
    
    # Initialize h as a uniform distribution satisfying the integral constraint
    h = np.random.rand(n_points) * 0.5 * 2  # Scale to [0, 1]
    h = h / (np.sum(h) / n_points) * (n_points / 2)  # Normalize to satisfy the integral constraint
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    best_time = time.time() - start_time
    
    # Optimization loop
    for _ in range(100):  # Max 100 iterations
        # Convert h to a numpy array
        h = best_h.copy()
        h = h.astype(float)
        
        # Define variables
        x = cp.Variable(n_points, boolean=True)
        
        # Objective: minimize the max overlap
        # Note: This is a non-convex problem; we use a heuristic approach
        # Compute the cross-correlation and find the maximum value
        h_correlation = np.zeros(n_points * 2)
        for i in range(n_points):
            for j in range(n_points):
                if i + j < n_points:
                    h_correlation[i + j] += h[i] * (1 - h[j])
                else:
                    h_correlation[i + j] += h[i] * (1 - h[j - n_points])
        max_overlap = np.max(h_correlation) * dx
        
        # Try small perturbations to improve the result
        for _ in range(10):  # Try 10 perturbations
            # Randomly select a point to perturb
            idx = np.random.randint(n_points)
            h[idx] = np.random.uniform(0, 1)
            
            # Normalize to satisfy the integral constraint
            h = h / (np.sum(h) / n_points) * (n_points / 2)
            
            # Evaluate the overlap
            h_correlation = np.zeros(n_points * 2)
            for i in range(n_points):
                for j in range(n_points):
                    if i + j < n_points:
                        h_correlation[i + j] += h[i] * (1 - h[j])
                    else:
                        h_correlation[i + j] += h[i] * (1 - h[j - n_points])
            current_c5 = np.max(h_correlation) * dx
            
            # Update best solution if improvement
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                best_time = time.time() - start_time
                if best_c5 < 0.380876:  # Target bound
                    break
            else:
                # Revert back to previous h if no improvement
                h = best_h.copy()
        
        # Check time limit
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    h = best_h.copy()
    h = h.astype(float)
    h_correlation = np.zeros(n_points * 2)
    for i in range(n_points):
        for j in range(n_points):
            if i + j < n_points:
                h_correlation[i + j] += h[i] * (1 - h[j])
            else:
                h_correlation[i + j] += h[i] * (1 - h[j - n_points])
    c5_bound = np.max(h_correlation) * dx
    
    return best_h, c5_bound, n_points
