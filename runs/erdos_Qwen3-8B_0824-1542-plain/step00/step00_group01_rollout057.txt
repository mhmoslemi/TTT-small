```python
import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as sla
import time
import cvxpy as cp
import random

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Placeholder for initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    n_points = 100  # Start with 100 points for faster evaluation

    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        nonlocal n_points
        if len(h) != n_points:
            h = np.resize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(overlap)

    # Function to find a good initial guess using genetic algorithm
    def genetic_algorithm(n_points, max_iter=10):
        population_size = 50
        mutation_rate = 0.1
        num_generations = max_iter

        # Initialize population
        def create_individual():
            return np.random.uniform(0, 1, n_points)

        population = [create_individual() for _ in range(population_size)]
        population = [h for h in population if np.sum(h) * (2.0 / n_points) >= 0.95 and np.sum(h) * (2.0 / n_points) <= 1.05]

        for generation in range(num_generations):
            # Evaluate fitness
            fitness = [evaluate_overlap(ind) for ind in population]
            best_idx = np.argmin(fitness)
            best_fitness = fitness[best_idx]
            best_h = population[best_idx].copy()

            # Select parents
            parents = np.random.choice(population, size=population_size, replace=True)
            # Crossover
            offspring = []
            for i in range(population_size):
                parent1 = parents[i]
                parent2 = parents[(i + 1) % population_size]
                child = np.zeros(n_points)
                for j in range(n_points):
                    if np.random.random() < 0.5:
                        child[j] = parent1[j]
                    else:
                        child[j] = parent2[j]
                # Mutation
                for j in range(n_points):
                    if np.random.random() < mutation_rate:
                        child[j] = np.random.uniform(0, 1)
                # Normalize and check constraints
                total = np.sum(child) * (2.0 / n_points)
                if total < 0.95 or total > 1.05:
                    child = np.random.uniform(0, 1, n_points)
                offspring.append(child)
            population = offspring

            # Update best solution
            if best_fitness < best_c5:
                best_c5 = best_fitness
                best_h = best_h.copy()

        return best_h

    # Function to find a good solution using convex optimization
    def convex_optimization(n_points):
        h = cp.Variable(n_points)
        constraints = [cp.sum(h) * (2.0 / n_points) == 1.0]
        for k in range(1, n_points):
            overlap_k = cp.sum(h * (1 - h[k:])) * (2.0 / n_points)
            constraints.append(overlap_k <= cp.Variable())
        obj = cp.Minimize(cp.Max(overlap_k for overlap_k in [cp.sum(h * (1 - h[k:])) * (2.0 / n_points) for k in range(1, n_points)]))
        problem = cp.Problem(obj, constraints)
        result = problem.solve(solver=cp.SCS, verbose=False)
        if result is not None:
            return h.value
        return np.random.uniform(0, 1, n_points)

    # Initial guess from provided initial_h_values
    initial_guess = initial_h_values
    initial_guess = np.resize(initial_guess, n_points)
    initial_c5 = evaluate_overlap(initial_guess)
    best_h = initial_guess.copy()
    best_c5 = initial_c5

    # Try multiple starting points
    for _ in range(3):
        if time.time() - start_time > budget_s - 5:
            break
        # Genetic algorithm
        h = genetic_algorithm(n_points)
        c5 = evaluate_overlap(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Convex optimization
        h = convex_optimization(n_points)
        c5 = evaluate_overlap(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Try increasing number of points if time allows
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        # Genetic algorithm
        h = genetic_algorithm(n_points)
        c5 = evaluate_overlap(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Convex optimization
        h = convex_optimization(n_points)
        c5 = evaluate_overlap(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Final check with current best
    best_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate_overlap(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, n_points)
```
