```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different number of points to find the best solution
    for n_points in [256, 128, 64, 32]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Initial guess
        h_values = initial_h_values.copy()
        h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_values)), h_values)
        
        # Ensure the sum of h is correct
        h_values = h_values * (n_points / 2) / (np.sum(h_values))
        
        # Optimization problem
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        objective = cp.Minimize(0)
        
        # Optimization
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, max_iter=100, verbose=False)
            if prob.status == 'optimal':
                h_values = h.value
        except:
            pass
        
        # Compute the overlap integral
        h_values = np.clip(h_values, 0, 1)
        dx = 2.0 / n_points
        h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_values)), h_values)
        h_values = h_values * (n_points / 2) / (np.sum(h_values))
        
        # Compute the overlap
        shifted_h = np.zeros((n_points, n_points))
        for i in range(n_points):
            shifted_h[i, :] = h_values[i:] + h_values[:n_points - i]
        
        overlap = np.max(np.sum(h_values[:, np.newaxis] * (1 - shifted_h), axis=1)) * dx
        
        if overlap < 0.380876:
            # If we find a better solution, keep it
            best_h_values = h_values
            best_c5 = overlap
            best_n_points = n_points
            print(f"Found better C5 bound: {best_c5} with n_points={best_n_points}")
    
    # If no better solution found, use the initial one
    if 'best_h_values' not in locals():
        best_h_values = initial_h_values
        best_n_points = len(initial_h_values)
        best_c5 = evaluate_erdos_solution(best_h_values)
    
    return best_h_values, best_c5, best_n_points
```
