```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
import matplotlib.pyplot as plt

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # Start with 100 samples; can be adjusted if needed
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()  # Start from the initial construction
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    h = h_initial.copy()
    best_c5 = float('inf')
    best_h = h.copy()
    
    try:
        while time.time() - start_time < budget_s:
            # Convert h to a list for optimization
            h_list = h.tolist()
            
            # Define the objective function: maximize the overlap integral
            def objective(x):
                h_temp = np.array(x)
                overlap = np.correlate(h_temp, 1 - h_temp, mode="full")
                c5 = np.max(overlap) * dx
                return c5
            
            # Define constraints
            constraints = [
                cp.Minimize(cp.sum(h_temp) - n_points / 2),  # Enforce ∫h(x)dx = 1
                cp.Minimize(cp.sum(cp.maximum(h_temp, 1) - 1)),  # Ensure h_temp ≤ 1
                cp.Minimize(cp.sum(cp.maximum(0 - h_temp, 0)))   # Ensure h_temp ≥ 0
            ]
            
            # Create a problem and solve
            h_temp = cp.Variable(n_points)
            prob = cp.Problem(cp.Minimize(objective(h_temp)))
            
            # Use ECOS solver since it's fast for small problems
            prob.solve(solver=cp.ECOS, eps=1e-4, max_iter=100)
            
            if prob.status == 'OPTIMAL':
                h = h_temp.value
                c5 = objective(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            
            # Check for convergence or early stopping
            if np.max(np.abs(h - best_h)) < 1e-4:
                break
        
        # Final evaluation and return result
        return (best_h, best_c5, n_points)
    
    except Exception as e:
        print("Error during optimization:", e)
        return (best_h, best_c5, n_points)
```
