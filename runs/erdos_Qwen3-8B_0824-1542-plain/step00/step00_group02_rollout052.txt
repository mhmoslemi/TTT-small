import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Define initial guess
    n_points = 45
    h_initial = initial_h_values.copy()
    h_initial = h_initial[:n_points]
    
    # Function to compute the overlap integral
    def compute_c5(h):
        dx = 2.0 / n_points
        h = np.pad(h, (0, 1), mode='constant')  # pad for full convolution
        conv = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(conv * dx)
        return c5
    
    # Function to optimize the overlap integral
    def optimize_h(initial_h, n_points, budget_s):
        start_time = time.time()
        best_h = initial_h.copy()
        best_c5 = compute_c5(best_h)
        best_iter = 0
        
        # Use a local optimization method
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            return compute_c5(h)
        
        def constraints(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) - n_points / 2  # Ensure the integral is 1
        
        # Initial optimization using L-BFGS-B
        result = optimize.minimize(
            objective,
            x0=initial_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            jac=None,
            options={'ftol': 1e-8, 'maxiter': 1000}
        )
        
        # Update best solution
        if result.success:
            best_h = result.x
            best_c5 = compute_c5(best_h)
            best_iter = result.nit
        
        # Refine using local search
        for _ in range(5):
            if time.time() - start_time > budget_s - 1:
                break
            perturb = np.random.uniform(-0.1, 0.1, size=n_points)
            perturb = np.clip(perturb, 0, 1)
            candidate = best_h + perturb
            candidate = np.clip(candidate, 0, 1)
            candidate = candidate * (n_points / 2) / np.sum(candidate)  # Ensure the integral is 1
            
            c5 = compute_c5(candidate)
            if c5 < best_c5:
                best_h = candidate
                best_c5 = c5
        
        return best_h, best_c5
    
    # Optimize the initial guess
    h_optimized, c5_bound = optimize_h(h_initial, n_points, budget_s)
    
    # Return the result
    return h_optimized, c5_bound, n_points
