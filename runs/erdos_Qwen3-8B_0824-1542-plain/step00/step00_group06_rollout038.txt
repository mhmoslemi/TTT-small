def run(seed=42, budget_s=60, **kwargs):

    # Start timing
    start_time = time.time()
    
    # Initial number of points
    n_points = 100  # Start with 100 points for faster initial evaluation
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h_values):
        h = np.array(h_values)
        dx = 2.0 / len(h)
        h_correlation = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(h_correlation) * dx
        return max_overlap
    
    # Initial guess

    if len(h_initial) != n_points:
        h_initial = np.random.rand(n_points)
        h_initial = (h_initial * 2 - 1) * 0.5 + 0.5  # Normalize to [0,1]
    
    # Optimization setup: we are trying to minimize the maximum overlap
    # We'll use a gradient-based optimizer with a penalty on the max overlap
    # and constraints on the integral of h
    def objective_function(h):
        # Ensure h is in [0,1]
        h_clamped = np.clip(h, 0, 1)
        # Compute the overlap integral
        dx = 2.0 / len(h_clamped)
        h_correlation = np.correlate(h_clamped, 1 - h_clamped, mode='full')
        max_overlap = np.max(h_correlation) * dx
        # Penalize the max overlap
        return max_overlap
    
    def constraint_function(h):
        # Enforce the integral constraint
        dx = 2.0 / len(h)
        return np.sum(h) * dx - 1

    max_iter = 100

    best_n_points = n_points

    current_c5 = evaluate_overlap(h_initial)

        best_c5 = current_c5
        best_h = h_initial.copy()

    for step in range(max_iter):

        h = best_h.copy()
        h = np.clip(h, 0, 1)
        dx = 2.0 / len(h)
        h_correlation = np.correlate(h, 1 - h, mode='full')
