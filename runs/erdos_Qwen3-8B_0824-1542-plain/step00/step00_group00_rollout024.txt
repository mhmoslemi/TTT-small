def run(seed=42, budget_s=60, **kwargs):

        # Compute the overlap integral
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def evaluate_and_store(h, elapsed_time, best_c5):
        if elapsed_time > budget_s:
            return best_c5, best_h
        c5 = objective_function(h)

            best_c5 = c5
            best_h = h.copy()
        return best_c5, best_h

    best_h = h.copy()
    start_time = time.time()

    # Local search with perturbations
    for _ in range(100):
        # Generate a random perturbation
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed_sum = np.sum(h_perturbed)
        h_perturbed = h_perturbed * (n_points / 2) / h_perturbed_sum
        current_c5 = objective_function(h_perturbed)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_perturbed.copy()

        # Check time
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s - 1:
            break

    # If not enough time, try refinement
    if elapsed_time < budget_s - 1:
        # Try improving with a small step size
        for _ in range(50):

            h_perturbed = np.clip(h_perturbed, 0, 1)

            current_c5 = objective_function(h_perturbed)
            if current_c5 < best_c5:

            elapsed_time = time.time() - start_time
