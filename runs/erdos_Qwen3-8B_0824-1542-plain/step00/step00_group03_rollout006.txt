```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 70  # Start with 70 samples
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64)  # Start from the initial construction
    
    # Ensure the initial h satisfies the constraint sum(h) == n_points / 2
    if not np.isclose(np.sum(h) * dx, 1.0):
        h = np.array(initial_h_values, dtype=np.float64)
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Try improving with different n_points
    for n_points in [70, 100, 120, 150, 180, 200]:
        start_time = time.time()
        dx = 2.0 / n_points
        h = np.random.rand(n_points) * 0.5  # Start with random values in [0, 0.5]
        h = h / np.sum(h) * (n_points / 2)
        
        # Define variables
        x = cp.Variable(n_points, name='x')
        constraints = [
            x >= 0,
            x <= 1,
            cp.sum(x) == n_points / 2
        ]
        
        # Define the objective function to minimize the maximum overlap
        def objective(x, dx):
            overlap = np.zeros(n_points - 1)
            for k in range(1, n_points):
                overlap[k - 1] = np.sum(x[:n_points - k] * (1 - x[k:])) * dx
            return cp.Max(overlap)
        
        # Optimization
        prob = cp.Problem(cp.Minimize(objective(x, dx)), constraints)
        try:
            result = prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s - (time.time() - start_time))
            if result is not None and result < best_c5:
                best_h = x.value
                best_c5 = result
        except:
            pass
        
        # Evaluate the current best
        if best_c5 is not None:
            current_c5 = evaluate_erdos_solution(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, len(best_h)
```
