```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')  # Assuming this is provided
evaluate_erdos_solution = lambda h: max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess
    n_points = 100  # Start with 100 points for efficiency
    h = np.copy(initial_h_values[:n_points])
    h /= np.sum(h) * 2.0  # Normalize to satisfy the integral constraint
    h = np.clip(h, 0, 1)
    
    # Optimization function to minimize the maximum overlap
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        overlaps = signal.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_overlap = np.max(overlaps)
        return max_overlap
    
    # Optimization bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use L-BFGS-B for optimization
    result = optimize.minimize(
        objective,
        x0=h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h /= np.sum(best_h) * 2.0  # Ensure integral constraint is satisfied
    
    # Evaluate the best solution
    c5_bound = evaluate_erdos_solution(best_h)
    
    # If we have time, try increasing n_points for better accuracy
    if time.time() - start_time < budget_s - 5:
        n_points = 200
        h = np.random.uniform(0, 1, n_points)
        h /= np.sum(h) * 2.0  # Normalize to satisfy the integral constraint
        h = np.clip(h, 0, 1)
        
        # Optimization function
        def objective_large(h_flat):
            h = h_flat.reshape(n_points)
            overlaps = signal.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            max_overlap = np.max(overlaps)
            return max_overlap
        
        # Optimize
        result = optimize.minimize(
            objective_large,
            x0=h,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h /= np.sum(best_h) * 2.0  # Ensure integral constraint is satisfied
        
        c5_bound = evaluate_erdos_solution(best_h)
    
    return best_h, c5_bound, n_points
```
