def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial parameters
    n_points = 80  # Starting with 80 points
    dx = 2.0 / n_points
    h_initial = np.array(initial_h_values)  # Use initial construction

    if np.abs(np.sum(h_initial) - n_points / 2) > 1e-6:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    
    # Define a function to compute the overlap integral
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h_pad = np.concatenate([h, np.zeros_like(h)])
        h_shifted = np.concatenate([np.zeros_like(h), h[:n_points - 1]])
        overlap = np.correlate(h_pad, h_shifted, mode="full")
        c5_values = []
        for k in range(1, n_points + 1):
            if k >= len(overlap):
                c5_values.append(0.0)
            else:
                c5_values.append(overlap[k] * dx)
        return np.max(c5_values)
    
    # Objective function for optimization
    def objective(h):
        return compute_c5(h)

    h = h_initial.copy()

            break

        current_c5 = compute_c5(h)

            h_plus[i] += 1e-5
            c5_plus = compute_c5(h_plus)
            h_minus = h.copy()
            h_minus[i] -= 1e-5
            c5_minus = compute_c5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)
        
        # Update h
        h -= learning_rate * grad
        
        # Ensure h stays in [0, 1] and integral is preserved
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
