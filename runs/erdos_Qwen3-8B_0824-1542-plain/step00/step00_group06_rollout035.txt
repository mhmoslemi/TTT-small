```python
import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points, can increase if time allows
    dx = 2.0 / n_points
    h = initial_h_values.copy()[:n_points]  # Use the provided initial construction

    # Ensure the initial h has the correct integral
    h = h / np.sum(h) * (n_points / 2)

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h[:n_points]
        # Compute the overlap integral
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def optimize_h():
        result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        return result.x

    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    # Try a few optimization steps
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        current_h = optimize_h()
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5

    # Optionally try increasing n_points if time remains
    if time.time() - start_time < budget_s - 1:
        n_points = 150
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        h[:n_points // 2] = 1.0
        h = h / np.sum(h) * (n_points / 2)
        current_h = optimize_h()
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5

    return (best_h, best_c5, n_points)
```
