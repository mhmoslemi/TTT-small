```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported helper functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h_values = np.array(initial_h_values, dtype=float)
    best_c5 = evaluate_erdos_solution(h_values)
    best_h = h_values.copy()
    
    # Try a few variations with different n_points
    for n_points in [100, 200, 400]:
        if time.time() - start_time > budget_s - 5:
            break
        
        # Scale the initial solution to the new n_points
        h_scaled = np.repeat(h_values, n_points // len(h_values))
        h_scaled = h_scaled[:n_points]
        h_scaled = h_scaled / np.sum(h_scaled) * (n_points / 2)
        
        # Set up optimization variables
        h = cp.Variable(n_points, nonneg=True, integer=False)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        
        # Define the overlap integral for all shifts
        dx = 2.0 / n_points
        overlap = np.zeros(n_points * 2 + 1)
        for k in range(1, n_points + 1):
            overlap[k:n_points + k] = h * (1 - h[k:n_points + k])
        
        # Objective is to minimize the maximum overlap
        objective = cp.Minimize(cp.Max(overlap * dx))
        
        # Solve the optimization problem
        if time.time() - start_time > budget_s - 5:
            break
        
        try:
            prob = cp.Problem(objective, constraints)
            result = prob.solve(solver=cp.SCS, eps=1e-4, verbose=False)
            if result is not None and prob.status == 'optimal':
                h_values = h.value
                c5 = evaluate_erdos_solution(h_values)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_values.copy()
        except:
            pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, len(best_h))
```
