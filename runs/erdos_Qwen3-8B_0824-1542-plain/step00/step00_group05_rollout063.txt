import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    h_initial = initial_h_values
    n_points_initial = len(h_initial)
    dx = 2.0 / n_points_initial
    sum_h = np.sum(h_initial) * dx
    if np.abs(sum_h - 1.0) > 1e-6:
        h_initial = h_initial * (1.0 / sum_h) * 2.0 / n_points_initial * n_points_initial / 2
    
    # Try a smaller number of points for faster optimization
    n_points = 100  # Adjust based on performance
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5 + 0.25  # Initial guess in [0.25, 0.75]
    h = h * (1.0 / np.sum(h) * dx) * n_points / 2  # Normalize to satisfy the integral constraint
    
    # Define the overlap integral function
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h_shifted = np.roll(h, 1)
        overlap = np.correlate(h, 1 - h_shifted, mode='full')
        c5 = np.max(overlap) * dx
        return c5
    
    # Optimization function: minimize c5
    def objective_func(h_flat):
        h = h_flat.reshape(n_points)
        return compute_c5(h)
    
    # Optimization constraints
    def constraints_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * dx - 1.0  # Ensure the integral constraint is satisfied
    
    # Use L-BFGS-B optimizer for faster convergence
    result = opt.minimize(
        fun=objective_func,
        x0=h.ravel(),
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        jac=None,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    # Check if we have a feasible solution within the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (h_initial, evaluate_erdos_solution(h_initial), n_points_initial)
    
    # Extract the optimized h values
    h_opt = result.x.reshape(n_points)
    c5_value = compute_c5(h_opt)
    
    # Refine further if time allows
    for _ in range(2):
        if time.time() - start_time > budget_s - 1:
            break
        result = opt.minimize(
            fun=objective_func,
            x0=h_opt,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            jac=None,
            options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        h_opt = result.x.reshape(n_points)
        c5_value = compute_c5(h_opt)
    
    return (h_opt, c5_value, n_points)
