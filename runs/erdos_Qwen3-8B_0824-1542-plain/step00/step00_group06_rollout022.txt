import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=60, **kwargs):
    # Predefined initial solution with n_points=95
    initial_h_values = np.load('initial_h.npy')  # Assume this file exists with the initial solution

    # Convert initial solution to a function for faster evaluation
    def evaluate_c5(h):
        # Compute the convolution of h with (1 - h)
        conv = np.correlate(h, 1 - h, mode="full")
        # Ensure we're considering the relevant part of the convolution
        # For overlap with shift k, we need the convolution at index k
        # Since the correlation is full, the relevant part is from index 0 to len(h) - 1
        # So we take the maximum of the convolution values
        c5 = np.max(conv[:len(h)])
        return c5

    # Define the optimization problem
    def optimize_h(n_points, initial_guess=None):
        # Define variables
        h = cp.Variable(n_points)
        # Constraints
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        # Objective: minimize the maximum overlap integral
        # We approximate the maximum by taking the maximum of the correlation values
        # We flatten the h and compute the convolution
        h_flat = cp.Variable(n_points)
        h_flat = h
        conv = np.correlate(h_flat.value, 1 - h_flat.value, mode="full")
        # Approximate with a small epsilon to avoid issues with zero
        # Since we're using a numerical solver, we will use an approximation
        # Objective: minimize the maximum of the convolutions
        # We use the current h value as an approximation for optimization
        objective = cp.Minimize(cp.Max(conv[:n_points]))
        # Solve the optimization problem
        prob = cp.Problem(objective, constraints)
        # Try to solve the problem
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iter=1000)
            if prob.status == 'optimal':
                return h.value
            else:
                return h.value  # fallback
        except:
            return h.value  # fallback

    # Try different n_points to find the best solution
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    # Start with the initial n_points
    n_points_list = [95, 100, 120, 150, 180, 200]
    for n_points in n_points_list:
        # Generate initial guess
        if n_points == 95:
            initial_guess = initial_h_values
        else:
            # If not initial, use a simple periodic function
            # We use a simple pattern of 0 and 1 to spread the integral
            x = np.linspace(0, 2, n_points)
            initial_guess = np.zeros(n_points)
            for i in range(n_points):
                if i % 3 == 0:
                    initial_guess[i] = 1
            # Normalize to meet the integral constraint
            initial_guess = initial_guess * (n_points / 2) / np.sum(initial_guess)

        # Optimize the function
        start_time = time.time()
        h_optimized = optimize_h(n_points, initial_guess)
        elapsed = time.time() - start_time

        # Evaluate the solution
        c5 = evaluate_c5(h_optimized)

        # Update the best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_optimized
            best_n_points = n_points

        # If we are over the budget, break and return
        if elapsed > budget_s:
            break

    return best_h, best_c5, best_n_points
