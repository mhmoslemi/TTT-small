def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial number of points
    n_points = 100
    max_n_points = 200
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / len(h))
        return np.max(overlap)
    
    # Function to generate a random initial guess
    def generate_initial_guess(n_points):
        h = np.random.rand(n_points)
        h = (h * 2.0 - 1.0) * 0.5 + 0.5  # Scale to [0, 1]
        total = np.sum(h) * (2.0 / n_points)
        h = h * (1.0 / total) * (n_points / 2.0)
        return h

    initial_h = np.copy(initial_h_values[:n_points])
    total = np.sum(initial_h) * (2.0 / n_points)
    initial_h = initial_h * (1.0 / total) * (n_points / 2.0)

    best_c5 = evaluate_overlap(best_h)
    
    # Iterative refinement
    for n_points in range(100, max_n_points + 1, 10):
        # Generate a new initial guess
        h = generate_initial_guess(n_points)
        
        # Optimization setup: minimize the maximum overlap using CVXPY
        x = cp.Variable(n_points)
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= 1
        ]
        overlap = np.correlate(x.value, 1 - x.value, mode="full") * (2.0 / n_points)
        max_overlap = cp.Max(overlap)
        
        # Solve the optimization problem
        prob = cp.Problem(cp.Minimize(max_overlap), constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-6, max_iter=1000, time_limit=10.0)
        except:
            pass

        if prob.status == 'optimal':
            h = x.value

    # Final evaluation for the best solution
    final_c5 = evaluate_overlap(best_h)
    
    # Return the best solution and the C5 bound
    return (best_h, final_c5, len(best_h))
