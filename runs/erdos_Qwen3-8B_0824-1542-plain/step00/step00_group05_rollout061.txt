```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(70) * 2 - 1
initial_h_values = np.clip(initial_h_values, 0, 2)
initial_h_values = np.interp(initial_h_values, (initial_h_values.min(), initial_h_values.max()), (0, 2))
initial_h_values = np.interp(initial_h_values, (0, 2), (0, 1))
initial_h_values = initial_h_values * (70 / 2)  # Ensure integral is 1
initial_h_values = initial_h_values / np.sum(initial_h_values)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for faster computation
    
    # Initial guess from the provided values
    h = initial_h_values[:n_points].copy()
    h = np.interp(h, (h.min(), h.max()), (0, 2))
    h = np.interp(h, (0, 2), (0, 1))
    h = h * (n_points / 2)
    h = h / np.sum(h)
    
    # Function to evaluate C5
    def evaluate_c5(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        corr = signal.correlate(h_vals, 1 - h_vals, mode="full")
        return np.max(corr) * (2.0 / len(h_vals))
    
    # Function to optimize h_vals
    def optimize_h_vals(h_vals, n_points):
        x = cp.Variable(n_points)
        constraints = [
            0 <= x,
            x <= 1,
            cp.sum(x) == n_points / 2
        ]
        objective = cp.Minimize(evaluate_c5(x))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, max_iter=1000, time_limit=10)
            if prob.status == 'optimal':
                return x.value
            else:
                return h_vals
        except:
            return h_vals
    
    # Main optimization loop
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h
    
    for _ in range(5):
        h = optimize_h_vals(h, n_points)
        if time.time() - start_time > budget_s - 5:
            break
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Try increasing n_points if time allows
    if time.time() - start_time < budget_s - 5:
        n_points = min(200, int(budget_s * 10))
        h = np.random.rand(n_points)
        h = np.interp(h, (h.min(), h.max()), (0, 2))
        h = np.interp(h, (0, 2), (0, 1))
        h = h * (n_points / 2)
        h = h / np.sum(h)
        
        for _ in range(3):
            h = optimize_h_vals(h, n_points)
            if time.time() - start_time > budget_s - 5:
                break
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    
    return best_h, best_c5, n_points
```
