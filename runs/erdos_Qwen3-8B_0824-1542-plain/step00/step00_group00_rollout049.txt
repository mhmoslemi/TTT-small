```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for speed
    
    # Initial guess
    h_initial = initial_h_values.copy()
    h_initial = np.resize(h_initial, (n_points,))
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Function to compute C5
    def compute_c5(h):
        h = np.resize(h, (n_points,))
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        k_values = np.arange(-n_points + 1, n_points)
        overlaps = []
        for k in k_values:
            h_shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - h_shifted)) / n_points
            overlaps.append(overlap)
        return np.max(overlaps)
    
    # Define objective and constraints for optimization
    def optimize_h(n_points):
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        objective = cp.Maximize(cp.sum(h * (1 - cp.roll(h, k)) for k in range(n_points)))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver='CVXOPT')
        return h.value
    
    # Try different n_points
    for n_points in [100, 150, 200]:
        if time.time() - start_time > budget_s - 5:
            break
        
        # Initial guess
        h_initial = initial_h_values.copy()
        h_initial = np.resize(h_initial, (n_points,))
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Optimization
        h_opt = optimize_h(n_points)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
    
    # Try local search around best solution
    if best_h is not None and time.time() - start_time < budget_s - 5:
        for _ in range(10):
            perturb = np.random.normal(0, 0.05, size=len(best_h))
            h_perturb = np.clip(best_h + perturb, 0, 1)
            h_perturb = h_perturb * (len(best_h) / 2) / np.sum(h_perturb)
            c5 = compute_c5(h_perturb)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_perturb
    
    # Final check with initial guess
    h_initial = initial_h_values.copy()
    h_initial = np.resize(h_initial, (n_points,))
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    c5 = compute_c5(h_initial)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_initial
    
    return (best_h, best_c5, len(best_h))
```
