```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define function to compute C5 from h_values
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        shifted = np.zeros_like(h)
        for k in range(1, len(h)):
            shifted[k:] = h[:-k]
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Define objective function for optimization
    def objective(h):
        return compute_c5(h)
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - len(h) / 2
    
    # Initial guess from provided starting point
    n_points = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    
    # Optimization settings
    max_iter = 100
    tol = 1e-6
    time_limit = budget_s
    
    start_time = time.time()
    
    # Use scipy.optimize with constraints
    result = optimize.minimize(
        objective,
        h_initial,
        method='SLSQP',
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': max_iter, 'ftol': tol}
    )
    
    # If optimization is not successful, use initial guess
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_h = h_initial
        best_c5 = compute_c5(h_initial)
    
    # Return the best solution found
    return best_h, best_c5, n_points
```
