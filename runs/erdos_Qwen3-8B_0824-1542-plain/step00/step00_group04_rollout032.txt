```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    random.seed(seed)
    start_time = time.time()
    
    # Define initial number of points
    n_points = 100
    max_n_points = 1000  # Maximum allowed n_points to stay under 60s
    
    # Initial solution
    h_initial = initial_h_values.copy()
    h_initial = np.resize(h_initial, (n_points,))
    h_initial = (h_initial * n_points / 2.0).astype(float)  # Adjust to sum to n_points / 2
    
    def compute_c5(h):
        h = np.resize(h, (n_points,))
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0)  # Normalize to ensure sum is n_points / 2
        h = h / (n_points / 2.0)  # Rescale back to [0,1]
        shifted = np.zeros_like(h)
        for k in range(1, n_points // 2 + 1):
            shifted[k:] = h[:-k]
        overlap = np.correlate(h, 1 - h, mode="full")
        dx = 2.0 / n_points
        c5 = np.max(overlap * dx)
        return c5
    
    def objective_func(h):
        return compute_c5(h)
    
    def optimize_solution(h_initial, n_points, budget_s):
        h = h_initial.copy()
        best_h = h.copy()
        best_c5 = compute_c5(h)
        best_n_points = n_points
        
        # Try optimizing with different n_points
        for n_points in range(n_points, min(max_n_points, 2 * n_points), 10):
            h = np.resize(h, (n_points,))
            h = np.clip(h, 0, 1)
            h = h * n_points / 2.0
            h = h / (n_points / 2.0)
            
            # Use optimization to find better solution
            x = cp.Variable(n_points)
            constraints = [
                0 <= x,
                x <= 1,
                cp.sum(x) == n_points / 2.0
            ]
            objective = cp.Minimize(cp.max(cp.multiply(x, 1 - cp.roll(x, 1, axis=0))))
            prob = cp.Problem(objective, constraints)
            
            try:
                prob.solve(solver=cp.SCS, eps=1e-6, verbose=False)
                if prob.status == 'optimal':
                    h_opt = np.array(x.value)
                    c5_opt = compute_c5(h_opt)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                        best_n_points = n_points
            except:
                pass
        
        return best_h, best_c5, best_n_points
    
    h_opt, c5_opt, n_points_opt = optimize_solution(h_initial, n_points, budget_s)
    
    # Final evaluation
    final_c5 = compute_c5(h_opt)
    
    # Check if we have time to try higher resolution
    if time.time() - start_time < budget_s - 10:
        h_opt_higher = np.resize(h_opt, (n_points_opt * 2,))
        h_opt_higher = np.clip(h_opt_higher, 0, 1)
        h_opt_higher = h_opt_higher * (n_points_opt * 2) / 2.0
        h_opt_higher = h_opt_higher / ((n_points_opt * 2) / 2.0)
        
        def optimize_higher(h):
            x = cp.Variable(len(h))
            constraints = [
                0 <= x,
                x <= 1,
                cp.sum(x) == len(x) / 2.0
            ]
            objective = cp.Minimize(cp.max(cp.multiply(x, 1 - cp.roll(x, 1, axis=0))))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.SCS, eps=1e-6, verbose=False)
            return np.array(x.value)
        
        h_opt_higher = optimize_higher(h_opt_higher)
        final_c5 = compute_c5(h_opt_higher)
    
    return h_opt, final_c5, n_points_opt
```
