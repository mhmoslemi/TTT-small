```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Placeholder for actual file

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Set default number of points
    n_points = 100  # Start with 100 points for speed and efficiency
    max_iter = 100  # Maximum number of iterations
    tolerance = 1e-4  # Tolerance for stopping criteria
    
    # Initialize h with uniform distribution
    h = np.random.rand(n_points) * 2.0  # Scale to [0, 2]
    h = np.clip(h, 0, 2)
    h /= np.sum(h) * (2.0 / n_points)  # Normalize to have integral 1
    h = np.clip(h, 0, 1)  # Ensure h is in [0, 1]
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    best_n_points = n_points
    
    start_time = time.time()
    
    for _ in range(max_iter):
        # Evaluate the current solution
        c5 = evaluate_erdos_solution(h)
        
        # Check if current solution is better
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n_points = n_points
        
        # Try increasing n_points if time permits
        if time.time() - start_time > budget_s * 0.8:
            break
        
        # Attempt to increase n_points for finer resolution
        n_points += 20
        if n_points > 1000:
            break
        
        # Re-initialize h with a better starting point
        h = np.random.rand(n_points) * 2.0
        h = np.clip(h, 0, 2)
        h /= np.sum(h) * (2.0 / n_points)
        h = np.clip(h, 0, 1)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, best_n_points
```
