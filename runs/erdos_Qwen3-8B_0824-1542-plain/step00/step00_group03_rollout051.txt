```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 100  # Start with 100 points for faster computation
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points])  # Use initial construction
    h = h / np.sum(h) * (n_points / 2)  # Normalize to satisfy the integral constraint
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Parameters for optimization
    max_iter = 50
    step_size = 0.05
    tolerance = 1e-4
    num_workers = 1  # Limited to 1 CPU
    num_candidates = 10
    
    for _ in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break
        
        # Generate candidate perturbations
        perturbations = np.random.uniform(-step_size, step_size, size=n_points)
        
        # Evaluate candidates
        candidates = []
        for i in range(num_candidates):
            h_candidate = h + perturbations[i]
            h_candidate = np.clip(h_candidate, 0, 1)
            sum_h = np.sum(h_candidate)
            if abs(sum_h - n_points / 2) > tolerance:
                h_candidate = h_candidate * (n_points / 2) / sum_h
            c5 = evaluate_erdos_solution(h_candidate, dx)
            candidates.append((c5, h_candidate))
        
        # Select best candidate
        candidates.sort()
        best_candidate = candidates[0][1]
        current_c5 = candidates[0][0]
        
        if current_c5 < best_c5:
            best_h = best_candidate
            best_c5 = current_c5
            print(f"New best C5: {best_c5:.6f}")
        
        h = best_h
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    print(f"Final C5: {final_c5:.6f}")
    
    return best_h, final_c5, n_points
```
