import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial setup
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    target_sum = n_points / 2
    
    # Function to compute C5 for a given h
    def compute_c5(h):
        h = np.array(h)
        h = np.concatenate([h, h[:n_points - 1]]).reshape(2 * n_points)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Function to evaluate and return (h, c5, n_points)
    def evaluate(h):
        c5 = compute_c5(h)
        return h, c5, n_points
    
    # Function to optimize h using gradient descent
    def optimize_h(h_initial):
        h = cp.Variable(n_points)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == target_sum
        ]
        
        # Objective function: minimize max overlap
        # We use a convex relaxation by minimizing the maximum overlap
        # This is an approximation as it's not directly convex
        # We instead use a heuristic with gradient descent
        # For a more accurate approach, we could use a bilevel optimization framework
        
        # We optimize using a gradient-based method with a penalty
        # Define the objective as the max overlap
        # This is complex to directly minimize, so we use a numerical approach
        
        # Use a numerical optimization with a penalty term
        # We minimize the maximum overlap by using a penalty function
        # Define a function to compute the max overlap
        def objective_func(h_values):
            h = np.array(h_values)
            h = np.concatenate([h, h[:n_points - 1]]).reshape(2 * n_points)
            overlap = np.correlate(h, 1 - h, mode="full")
            return np.max(overlap) * dx
        
        # Use L-BFGS-B to minimize the objective
        result = optimize.minimize(
            objective_func,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            jac=None,
            options={'maxiter': 1000, 'ftol': 1e-6}
        )
        
        return result.x

    # Run the optimization
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    
    # Perform a few iterations of optimization
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Optimize with the current h
        optimized_h = optimize_h(best_h)
        
        # Evaluate the optimized h
        current_c5 = compute_c5(optimized_h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = optimized_h
            best_c5 = current_c5
        
        # Random perturbations to avoid local minima
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_c5 = compute_c5(perturbed_h)
        
        if perturbed_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = perturbed_c5
    
    # Final evaluation
    final_h = np.clip(best_h, 0, 1)
    final_c5 = compute_c5(final_h)
    
    # Return the best solution
    return final_h, final_c5, n_points
