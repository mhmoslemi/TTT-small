```python
import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

# Global constants
MAX_ITERATIONS = 200
EPSILON = 1e-3
TIME_BUDGET = 60

def run(seed=42, budget_s=TIME_BUDGET, **kwargs):
    np.random.seed(seed)
    
    n_points = 45
    h_values = initial_h_values.copy()
    
    # Ensure sum(h_values) == n_points / 2
    if not np.isclose(np.sum(h_values), n_points / 2):
        h_values = (np.random.rand(n_points) * 2 - 1) * 0.5 + 0.5
        h_values = np.clip(h_values, 0, 1)
        h_values = h_values * (n_points / 2) / np.sum(h_values)
    
    best_c5 = evaluate_erdos_solution(h_values)
    best_h = h_values.copy()
    best_n_points = n_points
    
    start_time = time.time()
    
    for _ in range(MAX_ITERATIONS):
        if time.time() - start_time > budget_s:
            break
        
        # Perturb the current solution
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbation = np.clip(perturbation, -0.1, 0.1)
        h_perturbed = h_values + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        c5_perturbed = evaluate_erdos_solution(h_perturbed)
        
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed.copy()
            best_n_points = n_points
        
        # Consider reducing n_points
        if n_points > 20:
            n_points_new = n_points - 5
            h_values_new = np.random.rand(n_points_new) * 2 - 1
            h_values_new = np.clip(h_values_new, 0, 1)
            h_values_new = h_values_new * (n_points_new / 2) / np.sum(h_values_new)
            c5_new = evaluate_erdos_solution(h_values_new)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_values_new.copy()
                best_n_points = n_points_new
    
    return best_h, best_c5, best_n_points
```
