import numpy as np
import scipy.optimize as optimize
import time
import functools

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define the initial solution
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_h = np.sum(h_initial) * dx
    if np.abs(sum_h - 1.0) > 1e-6:
        h_initial = np.array(h_initial, dtype=float)
        h_initial = h_initial / (sum_h / 1.0) * 1.0
    
    # Objective function to minimize overlap integral
    def objective(h):
        h = np.array(h, dtype=float)
        sum_h = np.sum(h) * dx
        if np.abs(sum_h - 1.0) > 1e-6:
            return np.inf
        # Compute the overlap integral for all possible k
        conv = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(conv) * dx
        return c5
    
    # Gradient function (approximate using finite differences)
    def grad_objective(h):
        h = np.array(h, dtype=float)
        sum_h = np.sum(h) * dx
        if np.abs(sum_h - 1.0) > 1e-6:
            return np.zeros_like(h)
        h0 = h.copy()
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h1 = h0.copy()
            h1[i] += 1e-6
            c5_1 = objective(h1)
            h1[i] -= 2e-6
            c5_2 = objective(h1)
            grad[i] = (c5_1 - c5_2) / (2e-6)
        return grad
    
    # Optimization settings
    max_iter = 100
    tol = 1e-5
    n_points = len(h_initial)
    
    # Initial solution
    h_current = h_initial.copy()
    best_h = h_current.copy()
    best_c5 = objective(h_current)
    best_time = 0
    
    # Optimize with L-BFGS-B
    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Perform a local optimization
        result = optimize.minimize(
            objective,
            h_current,
            method='L-BFGS-B',
            jac=grad_objective,
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 10}
        )
        
        h_current = result.x
        current_c5 = objective(h_current)
        
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
        
        # Check for convergence
        if np.abs(current_c5 - best_c5) < tol:
            break
    
    # Evaluate the best solution
    final_c5 = objective(best_h)
    return (best_h, final_c5, n_points)
