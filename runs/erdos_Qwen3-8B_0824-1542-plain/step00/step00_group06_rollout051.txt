import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution
    n_points = 95
    h_initial = initial_h_values
    h_values = h_initial.copy()
    
    # Function to compute the overlap integral (C₅)
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        # Compute correlation
        corr = signal.correlate(h, 1 - h, mode="full")
        # Compute max overlap integral
        c5 = np.max(corr) * dx
        return c5
    
    # Function to optimize for lower C₅
    def optimize_h(h_guess):
        h = cp.Variable(len(h_guess))
        constraints = [
            cp.sum(h) == len(h_guess) / 2,  # Ensure the integral is 1
            cp.Minimize(cp.max(signal.correlate(h.value, 1 - h.value, mode="full") * (2.0 / len(h.value))))
        ]
        # Use a solver that can handle large problems
        prob = cp.Problem(cp.Minimize(cp.max(signal.correlate(h.value, 1 - h.value, mode="full") * (2.0 / len(h.value)))), constraints)
        prob.solve(solver='ECOS', verbose=False)
        if prob.status != 'OPTIMAL':
            return h_guess
        return h.value
    
    # Optimization loop
    best_h = h_values
    best_c5 = compute_c5(best_h)
    elapsed = 0
    
    while elapsed < budget_s - 1:
        # Optimize the current guess
        h_optimized = optimize_h(best_h)
        # Check if this is better
        c5_optimized = compute_c5(h_optimized)
        if c5_optimized < best_c5:
            best_h = h_optimized
            best_c5 = c5_optimized
        # Check if we can increase resolution
        n_points = len(best_h)
        if n_points < 1000:
            n_points *= 2
            # Reconstruct the function with more points
            best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        elapsed = time.time() - start_time
    
    return best_h, best_c5, len(best_h)
