import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions and initial solution
initial_h_values = np.load("initial_h.npy")  # Assuming this is provided

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    initial_h_values = np.load("initial_h.npy")
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    
    # Function to compute the C5 value
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h_shifted = np.roll(h, shift=1, axis=0)
        h_shifted = np.concatenate((h_shifted, np.zeros_like(h_shifted[:1])), axis=0)
        overlap = np.correlate(h, 1 - h_shifted, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    # Function to evaluate the objective
    def objective_func(h):
        h = np.clip(h, 0, 1)
        h_shifted = np.roll(h, shift=1, axis=0)
        h_shifted = np.concatenate((h_shifted, np.zeros_like(h_shifted[:1])), axis=0)
        overlap = np.correlate(h, 1 - h_shifted, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    # Function to evaluate the constraint
    def constraint_func(h):
        h = np.clip(h, 0, 1)
        return np.sum(h) - n_points / 2
    
    # Function to perform a local search
    def local_search(h_initial, iterations=100, step_size=0.01):
        h = h_initial.copy()
        best_h = h.copy()
        best_c5 = compute_c5(h)
        for _ in range(iterations):
            perturbation = np.random.uniform(-step_size, step_size, size=len(h))
            h_perturbed = np.clip(h + perturbation, 0, 1)
            c5_perturbed = compute_c5(h_perturbed)
            if c5_perturbed < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = c5_perturbed
            h = best_h.copy()
        return best_h, best_c5
    
    # Function to perform a genetic algorithm
    def genetic_algorithm(population_size=50, generations=50, mutation_rate=0.1):
        population = np.random.uniform(0, 1, size=(population_size, n_points))
        population = np.clip(population, 0, 1)
        population = (population * n_points / 2).astype(np.float64)
        population = np.clip(population, 0, n_points / 2)
        population = population / (n_points / 2)
        population = np.clip(population, 0, 1)
        best_h = population[0]
        best_c5 = compute_c5(best_h)
        for _ in range(generations):
            fitness = np.array([compute_c5(h) for h in population])
            indices = np.argsort(fitness)
            selected = population[indices[:population_size // 2]]
            offspring = []
            for h1, h2 in zip(selected[::2], selected[1::2]):
                child = np.random.rand(n_points)
                child = np.random.choice([h1, h2], p=[0.5, 0.5], size=n_points)
                child = np.clip(child, 0, 1)
                child = np.random.rand(n_points) < mutation_rate
                child = np.where(child, np.random.uniform(0, 1, size=n_points), child)
                child = np.clip(child, 0, 1)
                offspring.append(child)
            population = np.array(offspring)
            population = np.clip(population, 0, 1)
            population = (population * n_points / 2).astype(np.float64)
            population = np.clip(population, 0, n_points / 2)
            population = population / (n_points / 2)
            population = np.clip(population, 0, 1)
            best_h = population[0]
            best_c5 = compute_c5(best_h)
        return best_h, best_c5
    
    # Initial search
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    best_h, best_c5 = local_search(best_h)
    
    # Genetic algorithm search
    best_h, best_c5 = genetic_algorithm()
    
    # Optimization with constraints
    h = cp.Variable(n_points)
    objective = cp.Minimize(best_c5)
    constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
    problem = cp.Problem(objective, constraints)
    try:
        problem.solve(solver=cp.GUROBI, time_limit=budget_s - (time.time() - start_time))
        if problem.status == "optimal":
            best_h = h.value
            best_c5 = compute_c5(best_h)
    except:
        pass
    
    # Final refinement
    best_h, best_c5 = local_search(best_h)
    
    # Ensure the solution is within budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
