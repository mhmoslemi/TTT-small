```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from functools import partial

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Placeholder for initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h)
    
    best_h = h
    best_c5 = current_c5
    
    # Try a few refinements
    for _ in range(3):
        # Try a few random perturbations
        for _ in range(5):
            # Randomly modify a few points
            indices = np.random.choice(n_points, size=5, replace=False)
            h[indices] = np.random.uniform(0, 1, size=5)
            
            # Ensure the integral condition is satisfied
            h_norm = np.sum(h) * dx
            if np.abs(h_norm - 1) > 1e-6:
                h = initial_h_values.copy()
                continue
            
            # Evaluate and update
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Try a more systematic optimization
        h = best_h.copy()
        # Use a simple gradient descent approach for the overlap integral
        c5 = evaluate_erdos_solution(h)
        for _ in range(20):
            # Compute the overlap integral for all shifts
            corr = np.correlate(h, 1 - h, mode="full")
            overlap = corr * dx
            max_overlap = np.max(overlap)
            
            # Compute the derivative (approximate) for each point
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-4
                h_minus = h.copy()
                h_plus[i] -= 1e-4
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode="full")
                grad[i] = (np.max(corr_plus * dx) - np.max(corr_minus * dx)) / (2 * 1e-4)
            
            # Project the gradient to respect the constraints
            # Here, we use a simple projected gradient descent
            step_size = 0.1
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            
            # Ensure the integral condition is satisfied
            h_norm = np.sum(h_new) * dx
            if np.abs(h_norm - 1) > 1e-6:
                h_new = initial_h_values.copy()
            
            c5_new = evaluate_erdos_solution(h_new)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    
    # Final check
    if time.time() - start_time > budget_s - 1:
        return (best_h, best_c5, n_points)
    
    # Further refinement using a simple optimization
    h = best_h.copy()
    for _ in range(10):
        # Use a simple optimization with constraints
        x = cp.Variable(n_points)
        constraints = [cp.sum(x) == n_points / 2]
        objective = cp.Maximize(cp.Max(cp.correlate(x, 1 - x, mode="full") * dx))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.OSQP, max_iter=100)
            if prob.status == 'optimal':
                h = np.clip(x.value, 0, 1)
                c5 = evaluate_erdos_solution(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        except:
            pass
    
    # Final evaluation
    c5 = evaluate_erdos_solution(best_h)
    return (best_h, c5, n_points)
```
