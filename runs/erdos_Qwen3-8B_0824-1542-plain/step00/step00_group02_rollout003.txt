```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Define parameters
    max_points = 1000  # Maximum number of points
    n_points = 100  # Start with 100 points
    max_iterations = 200  # Max number of iterations
    
    # Initial guess from provided initial_h_values
    initial_h = initial_h_values.copy()
    initial_h = np.array(initial_h[:n_points])  # Ensure it has n_points samples
    
    # Check if initial solution is feasible
    if np.sum(initial_h) != n_points / 2:
        initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    
    best_solution = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_solution)
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Increase the number of points gradually
        n_points = min(n_points + 20, max_points)
        dx = 2.0 / n_points
        
        # Generate random perturbations
        perturbation = np.random.rand(n_points) * 0.2 - 0.1
        
        # Create a new candidate solution
        candidate_h = initial_h + perturbation
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Ensure the integral constraint is satisfied
        sum_h = np.sum(candidate_h)
        scale_factor = (n_points / 2) / sum_h
        candidate_h = candidate_h * scale_factor
        
        # Evaluate the solution
        current_c5 = evaluate_erdos_solution(candidate_h)
        
        # Update the best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_solution = candidate_h.copy()
        
        # Update the initial guess for the next iteration
        initial_h = candidate_h
    
    return best_solution, best_c5, n_points
```
