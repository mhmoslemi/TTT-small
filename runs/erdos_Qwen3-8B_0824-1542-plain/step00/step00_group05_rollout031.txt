```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial sample size for faster optimization
    n_points = 100  # Start with 100 points for faster computation
    dx = 2.0 / n_points
    
    # Start with the initial solution
    h = initial_h_values[:n_points].copy()
    h = np.clip(h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h_extended = np.concatenate([h, np.zeros_like(h)])
        correlation = signal.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(correlation) * dx
        return max_overlap
    
    # Function to compute the objective for optimization
    def objective(h):
        return compute_c5(h)
    
    # Optimization: Use scipy.optimize to minimize the overlap
    def optimize_h(h_initial):
        h = cp.Variable(h_initial.shape)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        prob = cp.Problem(cp.Minimize(objective(h)), constraints)
        result = prob.solve(solver='ECOS', max_iter=1000)
        if result == cp.OPTIMAL:
            return h.value
        else:
            return h_initial
    
    # Run optimization
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    print(f"Initial C5: {best_c5}")
    
    # Try a few more iterations with increasing points
    for n in [150, 200, 250]:
        if time.time() - start_time > budget_s - 1:
            break
        dx = 2.0 / n
        h = np.zeros(n)
        # Start with uniform distribution
        h[:int(n / 2)] = 1.0
        h = optimize_h(h)
        h = np.clip(h, 0, 1)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        print(f"n={n}, C5={current_c5}")
    
    # Final check with the current best
    best_h = np.clip(best_h, 0, 1)
    best_c5 = compute_c5(best_h)
    print(f"Final C5: {best_c5}")
    
    return best_h, best_c5, len(best_h)
```
