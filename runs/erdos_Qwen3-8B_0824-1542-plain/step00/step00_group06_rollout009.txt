import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    # Initial guess based on given construction
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values, dtype=np.float64)
    h_initial = h_initial / np.sum(h_initial) * (n_points_initial / 2)

    # Try different n_points starting from the initial one, up to a reasonable limit
    n_points_list = [n_points_initial, n_points_initial * 2, n_points_initial * 4]
    best_c5 = float('inf')
    best_h = None
    best_n_points = n_points_initial

    for n_points in n_points_list:
        if n_points < 10:
            continue  # We need enough samples to compute the correlation

        # Define the grid
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        h = cp.Variable(n_points)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]

        # Objective: Minimize the maximum overlap integral
        # We approximate the maximum using a convex relaxation
        overlap = []
        for k in range(1, n_points):
            if k >= n_points:
                break
            h_shifted = np.roll(h, k)
            overlap_val = cp.sum(h * (1 - h_shifted)) * dx
            overlap.append(overlap_val)
        overlap = cp.vstack(overlap)
        max_overlap = cp.Max(overlap)
        objective = cp.Minimize(max_overlap)

        # Solve the optimization problem
        start_time = time.time()
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == 'optimal':
                h_values = h.value
                c5_bound = max_overlap.value
                if c5_bound < best_c5:
                    best_c5 = c5_bound
                    best_h = h_values
                    best_n_points = n_points
        except:
            pass

        # Check time budget
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, best_n_points)
