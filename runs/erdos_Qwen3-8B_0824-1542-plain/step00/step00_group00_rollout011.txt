```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import sys

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial solution
    n_points = 45
    h_initial = initial_h_values()
    h_values = h_initial.copy()
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        n = len(h)
        dx = 2.0 / n
        h_flat = np.concatenate([h, np.zeros(n)])
        one_minus_h = 1.0 - h
        corr = signal.correlate(h_flat, one_minus_h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Function to compute the objective and constraints
    def objective(x):
        return compute_c5(x)
    
    # Constraint: sum(h) == n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Bounds: 0 <= h[i] <= 1 for all i
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization
    result = optimize.minimize(
        objective,
        h_values,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        method="SLSQP",
        options={"ftol": 1e-10, "maxiter": 100}
    )
    
    # Evaluate best solution found
    best_h = result.x
    c5_bound = compute_c5(best_h)
    
    # Early stopping if time is up
    if time.time() - start_time > budget_s - 1:
        return (best_h, c5_bound, n_points)
    
    # Try a few more optimizations with different starting points
    for _ in range(3):
        # Perturb the solution slightly
        perturb = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        
        # Reoptimize
        result = optimize.minimize(
            objective,
            perturbed_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 50}
        )
        
        # Update best solution
        best_h = result.x
        c5_bound = compute_c5(best_h)
        
        # Check time
        if time.time() - start_time > budget_s - 1:
            return (best_h, c5_bound, n_points)
    
    # Final evaluation
    return (best_h, c5_bound, n_points)
```
