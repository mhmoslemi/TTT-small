import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # Assuming it's preloaded
n_points_initial = len(initial_h_values)

# Helper function to evaluate the overlap integral for a given h
def evaluate_c5(h):
    return max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

# Strategy: Use a combination of heuristic initialization and convex optimization
# to refine the step function h to minimize the overlap integral

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = n_points_initial  # Start with initial number of samples

    # Function to optimize
    def objective_function(h_values):
        # Convert to numpy array and evaluate the overlap integral
        h = np.array(h_values, dtype=float)
        c5 = evaluate_c5(h)
        return c5

    # Function to generate an initial guess
    def generate_initial_guess(n_points, seed):
        np.random.seed(seed)
        h = np.random.rand(n_points) * 2
        h = np.clip(h, 0, 2)  # Ensure h stays within [0, 2]
        # Normalize to ensure the integral is 1
        h = (h - np.min(h)) / (np.max(h) - np.min(h))
        h = np.clip(h, 0, 1)
        # Ensure the integral is exactly 1
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Define the optimization problem
    def optimize_h(n_points):
        np.random.seed(seed)
        h = generate_initial_guess(n_points, seed)
        # Define variables
        x = cp.Variable(n_points)
        # Objective: min max_k (integral of h(x)(1 - h(x + k)))
        # Since this is non-convex, we use a heuristic approach
        # Evaluate all possible shifts
        h_values = x.value
        c5 = evaluate_c5(h_values)
        # Perform gradient descent
        for _ in range(100):
            # Compute gradients
            # This is a heuristic approach and may need refinement
            # For the purpose of the problem, we use a simple gradient descent
            # Compute the overlap integrals for all shifts
            shifts = np.correlate(h_values, 1 - h_values, mode="full")
            # Find the maximum shift
            max_shift_index = np.argmax(shifts)
            # Compute the derivative of the overlap integral with respect to h
            # This is a heuristic and may not be fully accurate
            # For the purpose of this problem, we use a simplified gradient
            derivative = np.zeros(n_points)
            for i in range(n_points):
                # Compute the contribution of h[i] to the max shift
                # This is a simplified approximation
                derivative[i] = np.sum((1 - h_values) * np.roll(h_values, -i))
            # Update h_values
            h_values -= 0.01 * derivative
            # Ensure h_values stays within [0, 1]
            h_values = np.clip(h_values, 0, 1)
            # Normalize to ensure the integral is 1
            h_values = h_values * (n_points / 2) / np.sum(h_values)
        # Evaluate the final c5
        final_c5 = evaluate_c5(h_values)
        return h_values, final_c5

    # Try multiple numbers of points
    for n_points in [n_points_initial, n_points_initial - 10, n_points_initial - 20, n_points_initial - 30]:
        if n_points < 10:
            continue  # Too few points
        # Run the optimization
        h, c5 = optimize_h(n_points)
        # Check if we found a better solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            # Print progress
            elapsed = time.time() - start_time
            remaining = budget_s - elapsed
            print(f"n_points={n_points}, c5={c5:.6f}, elapsed={elapsed:.2f}s, remaining={remaining:.2f}s")

        # Check if budget is exceeded
        if time.time() - start_time > budget_s:
            break

    # Return the best found solution
    return best_h, best_c5, len(best_h)
