import numpy as np
import scipy.optimize as opt
import time
import copy

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Load initial construction
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        # Ensure h is within [0,1] and sum is correct
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf  # Penalize for incorrect total integral
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx

    # Function to optimize
    def objective(h_flat):
        h = h_flat.copy()
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        # Enforce the constraint sum(h) = n_points / 2
        if np.abs(np.sum(h) - n_points / 2) > 1e-6:
            return np.inf
        # Compute the overlap integral
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx

    # Initial guess
    h_initial = initial_h_values.copy()

    # Optimization settings
    max_iter = 100
    time_limit = budget_s
    start_time = time.time()

    # Use L-BFGS-B optimizer
    result = opt.minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        jac=False,
        options={'disp': False, 'maxiter': max_iter}
    )

    # Extract the best solution found
    best_h = result.x
    best_c5 = compute_c5(best_h)
    best_n_points = n_points

    # If the optimization didn't find a better solution, try perturbing the initial guess
    if best_c5 >= 0.38092:
        perturbations = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        perturbed_c5 = compute_c5(perturbed_h)
        if perturbed_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = perturbed_c5

    # Track time and return the best solution
    elapsed_time = time.time() - start_time
    if elapsed_time < time_limit:
        # Try another perturbation if time allows
        perturbations = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        perturbed_c5 = compute_c5(perturbed_h)
        if perturbed_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = perturbed_c5

    return (best_h, best_c5, best_n_points)
