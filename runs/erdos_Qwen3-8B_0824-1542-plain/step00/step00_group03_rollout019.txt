import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import copy
import math

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with a reasonable number of samples

    # Try multiple starting points
    for _ in range(3):
        # Initialize with initial_h_values or random
        if _ == 0:
            h = initial_h_values.copy()
        else:
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)  # Normalize

        # Use a local optimization to refine the solution
        def objective(x):
            h = x.reshape((n_points,))
            # Compute the overlap integral
            h_extended = np.concatenate([h, h])
            corr = signal.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * (2.0 / n_points)
            return c5

        # Constraints: sum(h) == n_points / 2
        def constraint(x):
            return np.sum(x) - n_points / 2

        cons = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0, 1) for _ in range(n_points)]

        # Use L-BFGS-B for optimization
        result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)

        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        else:
            print("Optimization failed.")

        # Try increasing the number of points for better resolution
        if time.time() - start_time < budget_s - 5:
            n_points += 50
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            # Try another optimization with the new n_points
            def objective_new(x):
                h = x.reshape((n_points,))
                h_extended = np.concatenate([h, h])
                corr = signal.correlate(h, 1 - h, mode="full")
                c5 = np.max(corr) * (2.0 / n_points)
                return c5

            def constraint_new(x):
                return np.sum(x) - n_points / 2

            cons_new = [{'type': 'eq', 'fun': constraint_new}]
            bounds_new = [(0, 1) for _ in range(n_points)]

            result_new = opt.minimize(objective_new, h, method='L-BFGS-B', bounds=bounds_new, constraints=cons_new, tol=1e-5)

            if result_new.success:
                current_h_new = result_new.x
                current_c5_new = objective_new(current_h_new)
                if current_c5_new < best_c5:
                    best_c5 = current_c5_new
                    best_h = current_h_new

    # Final check with the best_h
    if best_h is not None:
        # Evaluate the best solution
        best_c5 = evaluate_erdos_solution(best_h)
        return (best_h, best_c5, len(best_h))
    else:
        return (initial_h_values, float('inf'), len(initial_h_values))
