import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    
    # Initial construction
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h)
    
    # Ensure the sum is correct
    sum_h = np.sum(h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        h = h / sum_h * (n_points / 2)
    
    def compute_c5(h_vals):
        h_vals = np.array(h_vals)
        n_points = len(h_vals)
        dx = 2.0 / n_points
        h = h_vals
        # Compute the overlap integral
        # We convolve h with (1 - h)
        # The full convolution is of size 2n - 1
        # We need to evaluate the max over all shifted overlaps
        # Since the original h is on [0, 2], the overlap for shift k is from x=0 to x=2-k
        # We will use convolution to compute it
        conv = np.convolve(h, 1 - h, mode='full')
        # The max value corresponds to the largest overlap integral
        # The values in conv represent dx * integral, so we normalize by dx
        c5 = np.max(conv) * dx
        return c5
    
    def objective_function(x):
        h_vals = x
        return compute_c5(h_vals)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    def constraint_bounds(x):
        return np.all((0 <= x) & (x <= 1))
    
    # Optimization settings
    num_iterations = 50
    best_c5 = float('inf')
    best_h = h.copy()
    
    for _ in range(num_iterations):
        # Use a local optimization method to improve the current solution
        # Convert h to a variable
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(objective_function(h_var))
        constraints = [
            cp.sum(h_var) == n_points / 2,
            0 <= h_var,
            h_var <= 1
        ]
        problem = cp.Problem(objective, constraints)
        try:
            result = problem.solve(solver=cp.SCS, verbose=False)
            if result is not None:
                h_new = np.array(h_var.value)
                # Check bounds
                h_new = np.clip(h_new, 0, 1)
                # Compute the new c5
                c5 = compute_c5(h_new)
                # Update the best solution
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
        except:
            # If optimization fails, fallback to a random perturbation
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h_new = h + perturbation
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
    
    # Final check
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    # Final optimization step with gradient descent
    h_opt = best_h.copy()
    learning_rate = 0.01
    for _ in range(50):
        # Compute gradients
        grad = np.gradient(objective_function(h_opt))
        # Update
        h_opt = h_opt - learning_rate * grad
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    # Ensure we stay within time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h.tolist(), best_c5, len(best_h))
    
    return (best_h.tolist(), best_c5, len(best_h))
