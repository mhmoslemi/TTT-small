import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Initial guess
    n_points = 100
    h_values = initial_h_values.copy()
    h_values = h_values[:n_points]
    h_values = np.array(h_values) * (n_points / 2) / np.sum(h_values)
    h_values = np.clip(h_values, 0, 1)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) * dx) * (n_points / 2)
        h = np.clip(h, 0, 1)
        kernel = 1 - h
        corr = np.correlate(h, kernel, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Function to optimize h_values
    def optimize_h_values(h_values_initial):
        h_values = h_values_initial.copy()
        h_values = np.clip(h_values, 0, 1)
        h_values = h_values / (np.sum(h_values) * dx) * (n_points / 2)
        h_values = np.clip(h_values, 0, 1)
        dx = 2.0 / n_points
        result = opt.minimize(
            compute_c5,
            h_values,
            bounds=[(0, 1) for _ in range(n_points)],
            method='L-BFGS-B',
            options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 1000}
        )
        return result.x
    
    best_c5 = float('inf')
    best_h = h_values.copy()
    
    # Try multiple starting points
    for _ in range(5):
        # Start from a random perturbation of the initial guess
        h_values = initial_h_values.copy()
        h_values = h_values[:n_points]
        h_values = np.array(h_values) * (n_points / 2) / np.sum(h_values)
        h_values = np.clip(h_values, 0, 1)
        h_values = h_values + np.random.normal(0, 0.05, n_points)
        h_values = np.clip(h_values, 0, 1)
        h_values = h_values / (np.sum(h_values) * dx) * (n_points / 2)
        h_values = np.clip(h_values, 0, 1)
        
        # Run the optimizer
        start_time = time.time()
        optimized_h = optimize_h_values(h_values)
        elapsed = time.time() - start_time
        
        if elapsed > budget_s:
            continue
        
        c5 = compute_c5(optimized_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = optimized_h
    
    # Further refinement with convex optimization
    n_points = 200
    dx = 2.0 / n_points
    h = cp.Variable(n_points)
    constraints = [
        cp.sum(h) == n_points / 2,
        0 <= h,
        h <= 1
    ]
    objective = cp.Minimize(0)
    prob = cp.Problem(objective, constraints)
    # Use a heuristic to construct a starting point based on best_h
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / (np.sum(best_h) * dx) * (n_points / 2)
    best_h = np.clip(best_h, 0, 1)
    h0 = np.array(best_h)
    h0 = h0 / np.max(h0) * 1
    h0 = h0 * np.max(h0) * (n_points / 2) / np.sum(h0)
    h0 = np.clip(h0, 0, 1)
    
    # Run the exact optimization
    start_time = time.time()
    prob.solve(solver=cp.SCS, max_iter=1000)
    elapsed = time.time() - start_time
    
    if elapsed > budget_s:
        pass
    else:
        h_values = h.value
    
    c5 = compute_c5(h_values)
    best_c5 = c5
    best_h = h_values
    
    # Final evaluation
    evaluate_erdos_solution(best_h, n_points, dx)
    
    return (best_h, best_c5, n_points)
