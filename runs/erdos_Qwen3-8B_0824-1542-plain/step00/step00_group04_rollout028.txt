```python
import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Try a range of n_points
    n_points_list = [200, 300, 400, 500, 600, 700, 800, 900, 1000]
    best_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Generate initial guess based on previous solution
        initial_guess = np.zeros(n_points)
        dx = 2.0 / n_points
        for i in range(len(best_h)):
            # Map the previous solution to the new grid
            x_prev = i * dx
            x_new = np.searchsorted(np.linspace(0, 2, n_points), x_prev)
            if 0 <= x_new < n_points:
                initial_guess[x_new] = best_h[i]
        
        # Ensure the sum is correct
        initial_guess /= (np.sum(initial_guess) / (n_points / 2))
        
        # Define the function to minimize
        def objective(h):
            # Rescale to [0, 2]
            x = np.linspace(0, 2, n_points)
            h_func = np.interp(x, np.linspace(0, 2, len(h)), h)
            
            # Compute the overlap integral
            h_func = np.clip(h_func, 0, 1)
            cor = np.correlate(h_func, 1 - h_func, mode="full")
            c5 = np.max(cor) * dx
            
            return c5
        
        # Define the constraints
        def constraint_sum(h):
            return np.sum(h) - (n_points / 2)
        
        # Use L-BFGS-B for optimization
        result = optimize.minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint_sum}]
        )
        
        # Get the best solution for this n_points
        h_candidate = result.x
        c5_candidate = evaluate_erdos_solution(h_candidate)
        
        # Update the best solution
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure the final solution meets the constraints
    dx = 2.0 / len(best_h)
    final_sum = np.sum(best_h) * dx
    if not np.isclose(final_sum, 1.0):
        best_h = np.clip(best_h, 0, 1)
        best_h /= (np.sum(best_h) / (len(best_h) / 2))
    
    return (best_h, final_c5, len(best_h))
```
