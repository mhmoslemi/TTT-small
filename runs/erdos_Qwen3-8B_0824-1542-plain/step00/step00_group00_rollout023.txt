import numpy as np
import scipy.optimize
import scipy.optimize.optimize
import scipy.optimize.minimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from provided initial construction
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Ensure the sum is exactly n_points / 2
    if abs(sum_h - n_points / 2) > 1e-10:
        initial_h[0] = (n_points / 2 - sum_h) / dx
        initial_h[0] = np.clip(initial_h[0], 0, 1)
    
    # Function to compute C5
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = np.concatenate([h, h])  # To handle full convolution
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Function to evaluate the objective (minimize C5)
    def objective(h):
        c5 = compute_c5(h)
        return c5
    
    # Function to compute the constraints
    def constraints(h):
        sum_h = np.sum(h)
        return [sum_h - n_points / 2]
    
    # Bounds for h values
    bounds = Bounds(0, 1)
    
    # Optimization
    start_time = time.time()
    result = scipy.optimize.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraints},
        options={'ftol': 1e-10, 'maxiter': 100}
    )
    
    # Check time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        # If time is up, return the best found so far
        return (initial_h, compute_c5(initial_h), n_points)
    
    h_opt = result.x
    c5_bound = result.fun
    return (h_opt, c5_bound, n_points)
