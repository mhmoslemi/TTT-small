import scipy.optimize
import time
from scipy.optimize import minimize

    dx = 2.0 / n_points

    # Define the function to compute C5
    def compute_c5(h_values):
        h = np.array(h_values)
        # Correlate h with (1 - h)
        corr = correlate(h, 1 - h, mode="full")
        # Compute max overlap integral
        max_overlap = np.max(corr * dx)
        return max_overlap

    # Define the objective function to minimize C5
    def objective(h_values):
        return compute_c5(h_values)

    # Define the constraint: sum(h) == n_points / 2
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2

    # Optimization settings
    options = {'maxiter': 100, 'disp': False}
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Optimization loop to improve the solution
    start_time = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Initial optimization
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=constraints, options=options)
    if result.success:
        h = result.x
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    # Additional refinement if time allows
    if time.time() - start_time < budget_s - 5:
        # Try a different initial guess
        h_new = np.random.uniform(0, 1, n_points)

        result = minimize(objective, h_new, method='L-BFGS-B', bounds=bounds, constraints=constraints, options=options)

            h = result.x
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:

    final_c5 = compute_c5(best_h)
