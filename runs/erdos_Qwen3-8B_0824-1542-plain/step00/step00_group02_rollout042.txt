import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial construction
    n_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    
    # Try several starting points around the initial one
    starting_points = [h_initial]
    for i in range(3):
        # Perturb the initial solution by small random changes
        perturbed = h_initial + np.random.uniform(-0.1, 0.1, size=n_initial)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed = (perturbed * n_initial) / np.sum(perturbed)
        starting_points.append(perturbed)
    
    best_c5 = float('inf')
    best_h = h_initial
    
    # Try different number of points (we'll use a smaller number to fit within the time budget)
    n_points_list = [50, 60, 70, 80, 90, 100]
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate a grid over [0, 2]
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Initial guess based on the starting points
        for h_initial in starting_points:
            # Resample to the current number of points
            h_initial = h_initial[:n_points] if len(h_initial) > n_points else h_initial
            h_initial = np.pad(h_initial, (0, n_points - len(h_initial)), 'constant')
            h_initial = h_initial[:n_points]
            h_initial = (h_initial * n_points) / np.sum(h_initial)
            h_initial = np.clip(h_initial, 0.0, 1.0)
            
            # Define variables
            h = cp.Variable(n_points)
            
            # Objective: minimize the maximum overlap integral
            # We'll use a convex relaxation to find a good approximation
            overlap = cp.Variable(n_points + 1)
            constraints = []
            for k in range(1, n_points + 1):
                if k > n_points:
                    break
                for i in range(n_points - k + 1):
                    overlap[i] += h[i] * (1 - h[i + k])
            
            # Max overlap
            max_overlap = cp.Max(overlap)
            
            # Constraints
            constraints.append(h >= 0)
            constraints.append(h <= 1)
            constraints.append(cp.sum(h) == n_points / 2)
            
            # Problem
            problem = cp.Problem(cp.Minimize(max_overlap), constraints)
            
            # Solve
            try:
                problem.solve(solver=cp.SCS, eps=1e-4)
                if problem.status == 'optimal' or problem.status == 'infeasible':
                    h_values = h.value
                    h_values = np.clip(h_values, 0.0, 1.0)
                    h_values = (h_values * n_points) / np.sum(h_values)
                    
                    # Compute the overlap integral
                    c5 = evaluate_erdos_solution(h_values, dx)
                    
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_values.copy()
                        print(f"Found better solution with n_points={n_points}, C5={c5:.6f}")
            except:
                pass
    
    # Final result
    return best_h, best_c5, len(best_h)
