def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 400  # We choose 400 points for a balance between accuracy and speed
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initial values (starting from the provided initial_h_values)
    h = initial_h_values.copy()
    h = h[:n_points]  # Use only the first n_points samples

    # Function to evaluate the overlap integral for the current h
    def evaluate_overlap(h):
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Function to compute the objective for optimization: maximize the overlap
    def objective_func(h):
        return evaluate_overlap(h)

    # Optimization: use a local optimizer to improve the initial solution
    # We use a limited number of iterations to fit within the budget
    # We use a combination of scipy.optimize.minimize with L-BFGS-B and some manual adjustments

    # Create bounds for h
    bounds = Bounds(0, 1)

    # Initial optimization pass using L-BFGS-B
    result = minimize(objective_func, h, method='L-BFGS-B', bounds=bounds, jac=None, options={'maxiter': 100})
    h = result.x

    # Ensure the sum of h is exactly target_sum
    # This is important to satisfy the constraint ∫₀² h(x) dx = 1
    # We can perform a small adjustment to enforce the constraint
    # This is a heuristic and might not always work, but it helps
    current_sum = np.sum(h) * dx
    error = current_sum - 1.0
    if abs(error) > 1e-6:
        # Make a simple adjustment to correct the sum
        # This is a heuristic and could be more sophisticated
        delta = error / (n_points * dx)
        h[-1] += delta

    # Now perform a few local searches around the current best
    for _ in range(3):
        # Small perturbation around the current h
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = np.clip(h + perturbation, 0, 1)
        # Check if sum of perturbed h is close to target_sum
        current_sum = np.sum(perturbed_h) * dx
        error = current_sum - 1.0
        if abs(error) > 1e-6:
            delta = error / (n_points * dx)
            perturbed_h[-1] += delta
        # Evaluate the overlap
        c5_current = evaluate_overlap(perturbed_h)
        if c5_current < evaluate_overlap(h):
            h = perturbed_h

    # Final evaluation
    c5_bound = evaluate_overlap(h)

    # Ensure we do not exceed the budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
