import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial h_values
    h = np.copy(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    
    # Check if the initial h satisfies the integral constraint
    if not np.isclose(sum_h, n_points / 2):
        h = h / sum_h * (n_points / 2)
    
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        n = len(h)
        dx = 2.0 / n
        h_extended = np.concatenate([h, np.zeros(n)])
        corr = np.correlate(h_extended, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def evaluate_and_update(h):
        c5 = compute_c5(h)
        return c5
    
    # Gradient-based optimization using scipy
    def objective(x):
        return evaluate_and_update(x)
    
    # Constraints: sum(h) == n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    # Optimization
    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        tol=1e-6
    )
    
    best_h = result.x
    best_c5 = evaluate_and_update(best_h)
    
    # If we have time, perform further refinement
    if time.time() - start_time < budget_s - 5:
        # Use a local search around the best solution
        for _ in range(5):
            perturb = np.random.uniform(-0.1, 0.1, size=n_points)
            perturb = np.clip(perturb, -0.1, 0.1)
            new_h = best_h + perturb
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            c5 = evaluate_and_update(new_h)
            if c5 < best_c5:
                best_h = new_h
                best_c5 = c5
    
    # Final evaluation
    final_c5 = evaluate_and_update(best_h)
    
    # If we still have time, perform a convex optimization
    if time.time() - start_time < budget_s - 5:
        # Create a convex optimization problem
        x = cp.Variable(n_points)
        objective = cp.Max(cp.sum(cp.multiply(x, cp.multiply(1 - x, cp.eye(n_points)))) / 2)
        constraints = [cp.sum(x) == n_points / 2, 0 <= x, x <= 1]
        prob = cp.Problem(cp.Minimize(objective), constraints)
        prob.solve(solver=cp.ECOS)
        if prob.status == 'optimal':
            best_h = x.value
            best_c5 = evaluate_and_update(best_h)
    
    # Final check
    final_c5 = evaluate_and_update(best_h)
    
    return best_h, final_c5, n_points
