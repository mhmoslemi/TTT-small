import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h_values, n_points: np.max(signal.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / n_points)
initial_h_values = np.load("initial_h_values.npy")  # Assuming this is a precomputed initial guess

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define the initial point
    n_points = len(initial_h_values)
    h = np.copy(initial_h_values)
    
    # Optimization parameters
    max_iterations = 100
    learning_rate = 0.01
    tolerance = 1e-4
    best_c5 = evaluate_erdos_solution(h, n_points)
    best_h = np.copy(h)
    
    # Function to compute the overlap integral
    def compute_c5(h_values):
        return evaluate_erdos_solution(h_values, len(h_values))
    
    # Function to compute the gradient of the overlap integral
    def compute_gradient(h_values):
        n_points = len(h_values)
        dx = 2.0 / n_points
        shift = np.arange(-n_points + 1, n_points)
        c = np.correlate(h_values, 1 - h_values, mode="full")
        gradient = np.zeros(n_points)
        for i in range(n_points):
            for k in range(n_points - 1):
                if k < n_points:
                    delta = np.zeros(n_points)
                    delta[k] = 1
                    shifted_h = np.roll(h_values, k)
                    c_k = np.correlate(shifted_h, 1 - h_values, mode="full")
                    grad_i = (c_k[i] - c[i]) / dx
                    gradient[i] += grad_i
        return gradient
    
    # Gradient descent optimization
    for _ in range(max_iterations):
        grad = compute_gradient(h)
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)  # Ensure h remains within [0, 1]
        
        # Ensure the integral constraint is satisfied
        total = np.sum(h) * (2.0 / n_points)
        if abs(total - 1) > tolerance:
            h = h * (1 / total) * (2.0 / n_points)
        
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(h)
        
        if time.time() - start_time > budget_s - 1:
            break
    
    # Further refinement using a convex optimization approach
    n_points = len(best_h)
    dx = 2.0 / n_points
    
    # Define variables
    h_var = cp.Variable(n_points)
    constraints = [cp.sum(h_var) * dx == 1, 0 <= h_var, h_var <= 1]
    
    # Objective: minimize the maximum overlap
    shift = np.arange(-n_points + 1, n_points)
    c = cp.Variable(n_points * (2 * n_points - 1))
    for i in range(n_points):
        for k in range(n_points - 1):
            if k < n_points:
                shifted_h = cp.reshape(cp.roll(h_var, k), (n_points,))
                corr = cp.sum(shifted_h * (1 - h_var))
                c[i * (2 * n_points - 1) + k] = corr
    
    objective = cp.Minimize(cp.Max(c))
    problem = cp.Problem(objective, constraints)
    
    try:
        problem.solve(solver=cp.MOSEK, solver_options={"MSK_IPAR_NUM_THREADS": 1})
        if problem.status == "OPTIMAL":
            best_h = np.array(h_var.value)
    except:
        pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
