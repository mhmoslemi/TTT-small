import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    n_points = len(initial_h_values)
    h_values = np.array(initial_h_values, dtype=np.float64)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        if len(h) != n_points:
            raise ValueError("h must have the same length as initial_h_values")
        dx = 2.0 / n_points
        h = np.array(h, dtype=np.float64)
        shifted_h = np.roll(h, 1)
        overlap = np.correlate(h, 1 - shifted_h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    # Function to compute the objective and constraints
    def objective_func(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Attempt to optimize using scipy optimization
    def optimize_h(h_initial):
        result = optimize.minimize(
            fun=objective_func,
            x0=h_initial,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        best_h = result.x
        return best_h, result.fun
    
    # Run optimization
    h_opt, c5_opt = optimize_h(h_values)
    
    # Check if we have time to do more iterations
    elapsed = time.time() - start_time
    if elapsed < budget_s - 5:
        # Try a few more iterations with different starting points
        for _ in range(2):
            # Perturb the current solution
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            h_perturbed = np.clip(h_opt + perturbation, 0, 1)
            h_perturbed, c5_perturbed = optimize_h(h_perturbed)
            if c5_perturbed < c5_opt:
                h_opt, c5_opt = h_perturbed, c5_perturbed
    
    # Now try to refine the solution using convex optimization
    def cvxpy_objective(h):
        return compute_c5(h)
    
    h_cvxpy = cp.Variable(n_points)
    objective = cp.Minimize(cvxpy_objective(h_cvxpy))
    constraints = [cp.sum(h_cvxpy) == n_points / 2, 0 <= h_cvxpy, h_cvxpy <= 1]
    
    # Use ECOS as the solver (requires installation)
    try:
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == 'optimal':
            h_opt = h_cvxpy.value
            c5_opt = compute_c5(h_opt)
    except:
        pass
    
    # Final check to ensure we have a valid solution
    h_opt = np.clip(h_opt, 0, 1)
    c5_opt = compute_c5(h_opt)
    
    # Early return if time is up
    if time.time() - start_time > budget_s - 1:
        return (h_opt, c5_opt, n_points)
    
    # Final evaluation to ensure it's valid
    final_c5 = evaluate_erdos_solution(h_opt)
    
    return (h_opt, final_c5, n_points)
