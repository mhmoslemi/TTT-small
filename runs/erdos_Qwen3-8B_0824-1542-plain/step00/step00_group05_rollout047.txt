import numpy as np

import scipy.signal as signal

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100  # Starting with 100 points for faster exploration
    dx = 2.0 / n_points
    max_iterations = 100  # Max iterations for optimization
    tolerance = 1e-4  # Tolerance for stopping criteria

    # Initial guess
    h = initial_h_values.copy()
    h = h[:n_points]  # Ensure it's the correct length
    h = h / np.sum(h) * (n_points / 2)  # Normalize to meet the constraint
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        if len(h) != n_points:
            h = np.resize(h, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)  # Ensure the integral constraint is met
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Function to optimize h using gradient descent
    def optimize_h(h_initial):
        h = h_initial.copy()
        best_h = h.copy()
        best_c5 = compute_c5(h)
        start_time = time.time()
        
        for _ in range(max_iterations):
            if time.time() - start_time > budget_s:
                break
            
            # Compute the overlap integral
            c5 = compute_c5(h)
            
            # Compute gradient (approximate using finite differences)
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-4
                h_minus = h.copy()
                h_minus[i] -= 1e-4
                grad[i] = (compute_c5(h_plus) - compute_c5(h_minus)) / (2 * 1e-4)
            
            # Update h using gradient descent
            h -= 0.01 * grad
            
            # Ensure h stays within [0, 1] and satisfies the integral constraint
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            
            # Update best solution
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        
        return best_h, best_c5
    
    # Run the optimization
    best_h, best_c5 = optimize_h(h)
    
    # Refine the solution with a higher number of points if time permits
    if time.time() - time.time() < budget_s - 10:
        n_points = 200
        dx = 2.0 / n_points
        h_refined = np.zeros(n_points)
        h_refined[:len(best_h)] = best_h
        h_refined = h_refined / np.sum(h_refined) * (n_points / 2)
        
        # Use a more sophisticated optimization method
        def objective(x):
            x = x.reshape(n_points)

            x = x / np.sum(x) * (n_points / 2)
            corr = signal.correlate(x, 1 - x, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        # Use a simple gradient-based optimization
        x0 = h_refined.copy()
        result = optimize.minimize(objective, x0, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        best_h = result.x
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
