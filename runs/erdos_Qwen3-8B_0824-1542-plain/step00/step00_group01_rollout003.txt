import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Use the initial h_values as a starting point
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    
    # Precompute dx
    dx = 2.0 / n_points
    
    # Optimization variables
    x = cp.Variable(n_points)
    constraints = [
        cp.sum(x) == n_points / 2,  # Ensure integral is 1
        0 <= x,
        x <= 1
    ]
    
    # Define the objective function to minimize the maximum overlap
    # To handle the max, we use a convex reformulation with auxiliary variables
    # We minimize the maximum value of the correlation, which is equivalent to minimizing the max overlap
    
    # Compute the correlation in the optimization
    # Since we're optimizing, this needs to be done in a loop or with a custom function
    
    # Since we can't directly optimize over max(correlation), we use a simple iterative optimization approach
    # We optimize the h_values to minimize the maximum overlap
    
    # Optimization loop
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try optimizing with different starting points and parameter settings
    for _ in range(5):
        # Use a local optimizer to minimize the maximum overlap
        def objective_func(h_values):
            h = h_values.reshape(n_points)
            dx = 2.0 / n_points
            corr = np.correlate(h, 1 - h, mode="full")
            max_overlap = np.max(corr) * dx
            return max_overlap
        
        # Use L-BFGS-B to minimize the max overlap
        result = opt.minimize(objective_func, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)], tol=1e-8)
        h = result.x
        
        # Check if the solution is valid
        if np.all(0 <= h) and np.all(h <= 1):
            # Compute the overlap integral
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            
            # Update the best solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Check if we have used up the budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Return the best solution found
    return best_h, best_c5, n_points
