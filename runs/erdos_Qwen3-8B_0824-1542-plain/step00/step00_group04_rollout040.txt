    # Ensure h_values stays within [0,1]
    h_values = np.clip(h_values, 0, 1)

    # Convert to numpy array
    h = np.array(h_values, dtype=np.float64)
    dx = 2.0 / n_points

    # Function to compute the maximum overlap integral
    def compute_c5(h):
        h_extended = np.concatenate([h, np.zeros_like(h)])

        return c5

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Optimization setup
    h_var = cp.Variable(n_points, low_bound=0, up_bound=1)
    objective = cp.Minimize(cp.Max(cp.multiply(cp.sum(cp.multiply(h_var, 1 - h_var)), dx)))
    
    # Constraint: integral of h must be 1
    constraint = [cp.sum(h_var) == n_points / 2]
    
    # Solve the problem using ECOS (which is available in CVXOPT)
    try:
        problem = cp.Problem(objective, constraint)

        h_opt = h_var.value
        c5_bound = compute_c5(h_opt)
    except:
        # If ECOS fails, fall back to a local optimization
        # Local optimization: Use scipy.optimize to minimize the maximum overlap
        def objective_func(h):
            h = np.clip(h, 0, 1)
            overlap = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(overlap) * dx
            return c5

        def constraint_func(h):
            return np.sum(h) - n_points / 2

        # Bounds for each variable
        bounds = [(0, 1) for _ in range(n_points)]

        # Initial guess
        h_initial = h_values.copy()

        # Optimization with L-BFGS-B
