```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points_init = len(initial_h_values)
    h_values = np.array(initial_h_values, dtype=np.float64)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        dx = 2.0 / len(h)
        h_flat = h.flatten()
        conv = signal.correlate(h_flat, 1 - h_flat, mode='full')
        c5 = np.max(conv) * dx
        return c5
    
    # Optimization function that minimizes the max overlap
    def objective(x):
        h = x.reshape(-1)
        if np.any(h < 0) or np.any(h > 1):
            return np.inf
        dx = 2.0 / len(h)
        if np.abs(np.sum(h) * dx - 1.0) > 1e-8:
            return np.inf
        conv = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(conv) * dx
        return c5
    
    # Optimization settings
    max_iter = 100
    n_points = n_points_init
    best_c5 = np.inf
    best_h = h_values.copy()
    
    # Try different n_points to find a good balance between speed and accuracy
    for n_points in [200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Initialize with the initial configuration or random
        if n_points == n_points_init:
            h_initial = h_values.copy()
        else:
            h_initial = np.random.rand(n_points) * 2 - 1
            h_initial = np.clip(h_initial, 0, 1)
            h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Bounds for optimization
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization
        result = optimize.minimize(
            objective,
            h_initial,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-8, 'disp': False}
        )
        
        if result.success:
            h_candidate = result.x
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
        
        # Reduce the number of iterations for smaller n_points
        if n_points < 300:
            max_iter = 50
    
    # Evaluate the best found solution
    final_h = best_h
    final_c5 = compute_c5(final_h)
    
    # Return the best solution found
    return (final_h, final_c5, len(final_h))
```
