def run(seed=42, budget_s=60, **kwargs):

    # Define the number of points (n_points) - we'll try with n=100 to keep it fast
    n_points = 100
    dx = 2.0 / n_points

    if len(initial_h_values) != n_points:
        initial_h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)

    # Function to evaluate C5
    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Optimization function using L-BFGS-B
    def optimize_h(h):
        def objective(x):
            h_ = x.reshape(-1)
            return evaluate_c5(h_)
        
        def gradient(x):
            h_ = x.reshape(-1)
            grad = np.zeros_like(h_)
            corr = signal.correlate(h_, 1 - h_, mode="full")
            max_index = np.argmax(corr)
            # Compute derivative of the max correlation with respect to h
            # We'll use finite differences for simplicity
            perturb = 1e-6
            for i in range(len(h_)):
                h_perturb = h_.copy()
                h_perturb[i] += perturb
                corr_perturb = signal.correlate(h_perturb, 1 - h_perturb, mode="full")
                grad[i] = (np.max(corr_perturb) - np.max(corr)) / perturb
            return grad.ravel()
        
        result = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            jac=gradient,
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        
        # Check if we've exceeded the time budget
        if time.time() - start_time > budget_s - 1:
            return result

                break

        return result

    # Run the optimization

    # Final solution
