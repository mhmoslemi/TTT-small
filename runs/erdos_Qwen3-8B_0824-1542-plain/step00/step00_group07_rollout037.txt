```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(signal.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load("initial_h_values.npy")  # Assume this is provided

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 100  # Start with 100 points, which is under 1k and fast to optimize
    dx = 2.0 / n_points

    # Initial guess based on the provided initial values, scaled and truncated to fit n_points
    initial_h = np.zeros(n_points)
    for i in range(len(initial_h_values)):
        # Map the initial values to the new grid
        x_initial = i * (2.0 / len(initial_h_values))
        x_new = x_initial * n_points / 2.0
        x_new = int(round(x_new))
        if 0 <= x_new < n_points:
            initial_h[x_new] = initial_h_values[i]

    # Normalize to meet the integral constraint
    total = np.sum(initial_h) * dx
    scaling = (1.0 / total) if total != 0 else 1.0
    initial_h *= scaling
    initial_h = np.clip(initial_h, 0, 1)

    # Objective function to minimize the maximum overlap
    def objective(h):
        h_values = h.reshape(-1)
        overlap = signal.correlate(h_values, 1 - h_values, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraints: h in [0, 1], sum(h) = n_points / 2
    bounds = Bounds(0, 1)
    constraint_sum = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)

    # Optimization using L-BFGS-B with initial guess
    result = opt.minimize(
        objective,
        initial_h,
        method="L-BFGS-B",
        bounds=bounds,
        constraints=constraint_sum,
        options={"ftol": 1e-8, "maxiter": 1000}
    )

    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    elapsed = time.time() - start_time

    # If time remains, try a finer grid (200 points) for better accuracy
    if elapsed < budget_s - 5:
        n_points = 200
        dx = 2.0 / n_points
        initial_h = np.zeros(n_points)
        for i in range(len(initial_h_values)):
            x_initial = i * (2.0 / len(initial_h_values))
            x_new = x_initial * n_points / 2.0
            x_new = int(round(x_new))
            if 0 <= x_new < n_points:
                initial_h[x_new] = initial_h_values[i]

        total = np.sum(initial_h) * dx
        scaling = (1.0 / total) if total != 0 else 1.0
        initial_h *= scaling
        initial_h = np.clip(initial_h, 0, 1)

        bounds = Bounds(0, 1)
        constraint_sum = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)

        result = opt.minimize(
            objective,
            initial_h,
            method="L-BFGS-B",
            bounds=bounds,
            constraints=constraint_sum,
            options={"ftol": 1e-8, "maxiter": 1000}
        )

        best_h = result.x
        best_c5 = evaluate_erdos_solution(best_h)

    return best_h, best_c5, n_points
```
