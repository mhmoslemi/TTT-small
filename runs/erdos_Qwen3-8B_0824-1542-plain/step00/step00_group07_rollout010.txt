import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions and variables
evaluate_erdos_solution = lambda h_values: max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load("initial_h.npy")  # Assuming initial_h.npy is available

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Predefine parameters
    max_n_points = 500  # Try up to 500 samples
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    # Try different numbers of points
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute dx
        dx = 2.0 / n_points
        
        # Initialize h with uniform distribution
        h = np.random.uniform(0, 1, n_points)
        total = h.sum()
        h = h * (n_points / 2) / total
        
        # Define the objective and constraints
        h_var = cp.Variable(n_points, nonneg=True)
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
        
        # Objective: minimize the maximum overlap integral
        # Compute all possible shifts
        overlap = []
        for k in range(1, n_points):
            shifted = np.roll(h_var.value, k)
            overlap.append(cp.multiply(h_var, 1 - shifted))
        
        # Compute the maximum overlap
        max_overlap = cp.Max(overlap)
        objective = cp.Minimize(max_overlap)
        
        # Solve the optimization problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, max_iter=1000, eps=1e-6)
            if prob.status == 'optimal':
                h_values = h_var.value
                c5 = evaluate_erdos_solution(h_values)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_values
                    best_n_points = n_points
        except:
            pass
        
        # Also try optimizing from initial_h_values if available
        if hasattr(initial_h_values, '__iter__') and len(initial_h_values) == n_points:
            h = np.copy(initial_h_values)
            total = h.sum()
            h = h * (n_points / 2) / total
            
            # Define the objective and constraints
            h_var = cp.Variable(n_points, nonneg=True)
            constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
            
            # Objective: minimize the maximum overlap integral
            # Compute all possible shifts
            overlap = []
            for k in range(1, n_points):
                shifted = np.roll(h_var, k)
                overlap.append(cp.multiply(h_var, 1 - shifted))
            
            # Compute the maximum overlap
            max_overlap = cp.Max(overlap)
            objective = cp.Minimize(max_overlap)
            
            # Solve the optimization problem
            prob = cp.Problem(objective, constraints)
            try:
                prob.solve(solver=cp.SCS, max_iter=1000, eps=1e-6)
                if prob.status == 'optimal':
                    h_values = h_var.value
                    c5 = evaluate_erdos_solution(h_values)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_values
                        best_n_points = n_points
            except:
                pass
    
    # Return the best solution
    return (best_h, best_c5, best_n_points)
