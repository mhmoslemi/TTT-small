```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
from scipy import sparse
from scipy.sparse import csr_matrix
import cvxpy as cp

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros(2 * n_points)
    shifted_h[n_points:] = h
    shifted_h[:n_points] = h
    shifted_1_minus_h = 1 - shifted_h
    correlate = np.correlate(h, shifted_1_minus_h, mode="full")
    c5 = np.max(correlate) * dx
    return c5

def initial_h_values():
    # Example initial construction with n=95 samples
    n_points = 95
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 0.5
        else:
            h[i] = 0.5
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values()
    n_points = len(best_h)
    dx = 2.0 / n_points

    # Initial evaluation
    current_c5 = evaluate_erdos_solution(best_h)
    best_c5 = current_c5
    print(f"Initial C5: {best_c5}")

    # Define optimization variables
    h = cp.Variable(n_points, nonneg=True, boolean=True)
    constraints = [cp.sum(h) == n_points / 2]

    # Objective function: minimize the maximum overlap integral across all shifts
    max_overlap = 0
    for k in range(1, n_points):
        shifted_h = np.roll(h.value, k)
        current_overlap = cp.sum(h * (1 - shifted_h)) * dx
        max_overlap = cp.Max(max_overlap, current_overlap)
    objective = max_overlap
    # Use a quadratic penalty to approximate the max function
    penalty = 0
    for k in range(1, n_points):
        shifted_h = np.roll(h, k)
        current_overlap = cp.sum(h * (1 - shifted_h)) * dx
        penalty += (current_overlap - cp.Variable())**2
    problem = cp.Problem(cp.Minimize(penalty), constraints)
    result = problem.solve(solver=cp.ECOS, max_iter=100)
    if result is not None:
        h_vals = h.value
        c5 = evaluate_erdos_solution(h_vals)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_vals
            print(f"New best C5: {best_c5}")

    # Refine with local search
    for _ in range(5):
        if time.time() - start_time > budget_s - 5:
            break
        h_vals = np.array(best_h)
        # Perturb the solution
        indices = np.random.choice(n_points, size=5, replace=False)
        for i in indices:
            h_vals[i] = 1 - h_vals[i] if np.random.rand() < 0.5 else h_vals[i]
        h_vals = np.clip(h_vals, 0, 1)
        if np.sum(h_vals) * dx < 1 - 1e-5 or np.sum(h_vals) * dx > 1 + 1e-5:
            continue
        c5 = evaluate_erdos_solution(h_vals)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_vals
            print(f"New best C5: {best_c5}")

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    print(f"Final C5: {final_c5}")
    return best_h, final_c5, n_points
```
