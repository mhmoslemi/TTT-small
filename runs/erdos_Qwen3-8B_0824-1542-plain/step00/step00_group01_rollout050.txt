```python
import numpy as np
import scipy.optimize as optimize
import time
import copy
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Example initial solution (n=47)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # Start with 100 points for speed
    while time.time() - start_time < budget_s:
        # Generate candidate h_values
        h = np.random.uniform(0, 1, n_points)
        total = np.sum(h)
        h = h * (n_points / 2.0) / total  # Normalize to meet the constraint

        # Compute C5
        c5 = evaluate_erdos_solution(h)
        if c5 < 0.38092:
            print(f"Found better solution with C5 = {c5:.6f} at {n_points} points")
            # Try increasing n_points for better precision
            n_points = min(n_points * 2, 1000)  # Max 1000 points to stay within budget
        else:
            # If not better, try a different random sample
            h = np.random.uniform(0, 1, n_points)
            total = np.sum(h)
            h = h * (n_points / 2.0) / total  # Normalize to meet the constraint

        # Update best solution
        current_best = h
        best_c5 = c5

    return current_best, best_c5, len(current_best)
```
