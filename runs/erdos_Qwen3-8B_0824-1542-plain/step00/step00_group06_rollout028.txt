import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

# Global constants
TARGET_C5 = 0.380876
MAX_TIME = 60

def run(seed=42, budget_s=MAX_TIME, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    
    # Define parameters
    n_points = 95  # Starting with the initial sample size
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    
    # Objective: minimize the maximum overlap
    # We'll optimize the h function to reduce the maximum overlap
    # We'll use a simple gradient-based optimization with bounds

    # Initial guess
    h = initial_h

    # Optimization function to compute C5
    def compute_c5(h):
        h = np.array(h)
        h = np.concatenate([h, h])  # To handle the full convolution
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    # Optimization function to minimize the maximum overlap
    def objective(x):
        h = x
        c5 = compute_c5(h)
        return c5

    # Bounds on h: [0,1]
    bounds = Bounds(0, 1)

    # Use L-BFGS-B solver for optimization
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        tol=1e-6,
        options={'maxiter': 1000}
    )

    # Update h with the optimized values
    h_opt = result.x
    c5_opt = result.fun

    # Track time
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed

    # If time remains, try increasing the number of points for better resolution
    if remaining > 10:
        n_points = int(120)  # Try a slightly larger number of samples
        dx = 2.0 / n_points
        h = np.random.uniform(0, 1, n_points)
        # Set initial guess as a more uniform distribution
        h = np.repeat(np.linspace(0, 1, 5), n_points // 5)
        h = h[:n_points]
        h = h / np.sum(h) * (n_points / 2)

        # Optimization function to compute C5
        def compute_c5_large(h):
            h = np.array(h)
            h = np.concatenate([h, h])  # To handle the full convolution
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5

        # Optimization function to minimize the maximum overlap
        def objective_large(x):
            h = x
            c5 = compute_c5_large(h)
            return c5

        # Bounds on h: [0,1]
        bounds = Bounds(0, 1)

        # Use L-BFGS-B solver for optimization
        result = optimize.minimize(
            objective_large,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            tol=1e-6,
            options={'maxiter': 1000}
        )

        # Update h with the optimized values
        h_opt = result.x
        c5_opt = result.fun
        n_points = 120

    # Evaluate the solution
    best_c5 = c5_opt
    return h_opt, best_c5, n_points
