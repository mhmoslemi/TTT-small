```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_47.npy")  # Example initial solution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # Start with 100 points, can adjust if needed
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    # Enforce the integral constraint sum(h) * dx = 1 => sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # Optimization loop
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Random perturbation
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = np.clip(h + perturb, 0, 1)
        h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
        c5 = evaluate_erdos_solution(h_perturb)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturb.copy()
    
    # Try a more structured optimization
    # Define variables
    h_var = cp.Variable(n_points, nonneg=True)
    constraints = [cp.sum(h_var) == n_points / 2]
    
    # Objective: minimize the max correlation
    # This is non-convex and difficult to optimize directly
    # We approximate by minimizing the maximum correlation using a convex relaxation
    # For each shift, we minimize the integral of h(x)(1-h(x+k)) dx
    # We use a convex relaxation by using a convex hull approach
    
    # Define the correlation function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Use a local optimization method for final refinement
    h_refined = best_h.copy()
    result = optimize.minimize(
        fun=objective,
        x0=h_refined,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)]
    )
    h_refined = result.x
    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
    
    final_c5 = evaluate_erdos_solution(h_refined)
    
    return h_refined, final_c5, n_points
```
