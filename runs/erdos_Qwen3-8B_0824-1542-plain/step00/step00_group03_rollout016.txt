```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr
from scipy.optimize import Bounds, LinearConstraint
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial number of points
    n_points = 100  # Start with 100 points to keep optimization fast
    
    # Check if we can increase n_points if time allows
    while time.time() - start_time < budget_s - 2:
        n_points += 50
        if n_points > 300:
            break
    
    # Compute dx
    dx = 2.0 / n_points
    
    # Initial guess from provided values or random
    if len(initial_h_values) == n_points:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Define the function to minimize
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = signal.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Perform optimization
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    
    # Extract the best solution
    h_values = result.x
    c5_bound = evaluate_erdos_solution(h_values)
    
    # If time allows, attempt a local refinement
    if time.time() - start_time < budget_s - 2:
        # Small perturbation to avoid local minima
        h_values += np.random.rand(n_points) * 0.05
        h_values = np.clip(h_values, 0, 1)
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        c5_bound = evaluate_erdos_solution(h_values)
    
    # Return the best solution
    return h_values, c5_bound, n_points
```
