import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try different numbers of points
    n_candidates = [200, 300, 400, 500]
    
    best_h = None
    best_c5 = float('inf')
    
    for n_points in n_candidates:
        if time.time() - start_time > budget_s:
            break
        
        # Calculate dx
        dx = 2.0 / n_points
        
        # Initialize h with a base pattern
        h = np.array(initial_h_values) / (len(initial_h_values) / n_points)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), h)
        
        # Ensure h is in [0,1] and normalized
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Define objective function
        def objective(x):
            h_new = x.reshape(n_points)
            h_new = np.clip(h_new, 0, 1)
            integral = np.correlate(h_new, 1 - h_new, mode="full") * dx
            return np.max(integral)
        
        # Define constraints
        def constraints(x):
            h_new = x.reshape(n_points)
            return np.sum(h_new) - n_points / 2
        
        # Define bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use optimization to minimize the maximum overlap
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraints},
            options={'ftol': 1e-6, 'maxiter': 100}
        )
        
        if result.success:
            h_optimized = result.x
            h_optimized = np.clip(h_optimized, 0, 1)
            h_optimized = h_optimized * (n_points / 2) / np.sum(h_optimized)
            
            # Evaluate C5
            c5 = evaluate_erdos_solution(h_optimized)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_optimized
        
    # If no solution found, use the initial one
    if best_h is None:
        best_h = initial_h_values / (len(initial_h_values) / 400)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, len(best_h))
