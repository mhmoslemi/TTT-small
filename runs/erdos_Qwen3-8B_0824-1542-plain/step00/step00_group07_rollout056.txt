import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timer
    start_time = time.time()
    
    # Initial configuration
    n_points = 80
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    # Ensure sum(h) * dx = 1
    scaling_factor = 1.0 / (sum_h * dx)
    h *= scaling_factor
    
    # Function to compute C5
    def compute_c5(h_values):
        n_points = len(h_values)
        dx = 2.0 / n_points
        h = h_values.copy()
        # Ensure sum(h) * dx = 1
        sum_h = np.sum(h)
        scaling_factor = 1.0 / (sum_h * dx)
        h *= scaling_factor
        
        # Compute the overlap integral
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Function to optimize h with constraints
    def optimize_h(h_values):
        n_points = len(h_values)
        dx = 2.0 / n_points
        h = cp.Variable(n_points)
        obj = cp.Minimize(0)
        constraints = [
            cp.sum(h) == n_points / 2,  # Ensure sum(h) * dx = 1
            0 <= h,
            h <= 1
        ]
        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver='MOSEK', quiet=True)
        if result is None:
            return h_values  # Fallback to current values
        return h.value
    
    # Function to improve the solution
    def improve_solution(h_values, iterations=10, n_points=80):
        for _ in range(iterations):
            # Optimize using convex optimization
            h_opt = optimize_h(h_values)
            # Compute current C5
            c5 = compute_c5(h_opt)
            # Update solution
            h_values = h_opt
            print(f"Iteration {_+1}: C5 = {c5:.6f}")
            # Check if time is up
            if time.time() - start_time > budget_s - 1:
                break
        return h_values, compute_c5(h_values)
    
    # Run the optimization
    h_values, c5_bound = improve_solution(h, iterations=50, n_points=80)
    
    # Return the best solution found
    return h_values, c5_bound, 80
