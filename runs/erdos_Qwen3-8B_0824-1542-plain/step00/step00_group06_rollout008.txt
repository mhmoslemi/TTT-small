import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess from provided starting point
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_h = sum(h_initial) * dx
    assert np.isclose(sum_h, 1), "Initial h_values does not satisfy the constraint"
    
    # Define objective function for optimization
    def evaluate(h_values):
        h = np.array(h_values)
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Integrate by multiplying with dx
        integrated = corr * dx
        # Take maximum as the objective
        c5 = np.max(integrated)
        return c5
    
    # Optimization function
    def optimize_h():
        # Use a smaller number of points for faster optimization (e.g., 100)
        n_points = 100
        dx = 2.0 / n_points
        
        # Define variable
        h = cp.Variable(n_points, lb=0, ub=1)
        
        # Define objective: minimize the maximum correlation
        corr = np.correlate(h.value, 1 - h.value, mode="full")
        # Wait, we can't use h.value here. Let's define it symbolically
        # We need to maximize over all shifts k, so we need to take the max over all k
        # We'll approximate this by evaluating for all possible k in the range
        
        # Compute all possible shifts (k)
        max_shift = int(2 * n_points)
        max_correlation = 0.0
        
        # Define symbolic expressions for shifts
        for k in range(1, max_shift):
            shifted_h = cp.reshape(h, (n_points, 1)) @ cp.reshape(1 - h, (1, n_points))
            corr_k = shifted_h[k - 1, :n_points - k + 1]
            max_correlation = cp.Max(max_correlation, cp.Max(corr_k))
        
        # Objective is to minimize the maximum correlation
        objective = cp.Minimize(max_correlation)
        
        # Constraints
        constraints = [cp.sum(h) == n_points / 2]
        
        # Solve the optimization problem
        prob = cp.Problem(objective, constraints)
        result = prob.solve(solver=cp.SCS, verbose=False)
        
        if result is not None and prob.status == cp.OPTIMAL:
            optimized_h = h.value
            return optimized_h
        else:
            return h_initial
    
    # Run optimization for a limited time
    optimized_h = optimize_h()
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h_initial, evaluate(h_initial), len(h_initial))
    
    # Return the best solution found
    return (optimized_h, evaluate(optimized_h), len(optimized_h))
