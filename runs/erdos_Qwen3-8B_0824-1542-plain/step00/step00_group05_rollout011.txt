```python
import numpy as np
import scipy.optimize as optimize
import matplotlib.pyplot as plt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    
    # Initial number of points
    n_points = 100  # Starting with 100 points for optimization
    
    # Initial guess: use provided initial_h_values
    h_initial = initial_h_values.copy() / np.max(initial_h_values) * 2
    h_initial = h_initial[:n_points]  # Ensure it's the same length
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)  # Normalize to meet the integral constraint
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Ensure integral constraint is met
        # Compute the overlap integral with convolution
        result = np.correlate(h, 1 - h, mode="full")
        result = result * (2.0 / n_points)  # Scale by dx
        return np.max(result)
    
    # Function to optimize
    def optimize_h(initial_guess):
        h = cp.Variable(n_points)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        # Objective: minimize the maximum overlap
        obj = cp.Minimize(cp.Max(cp.multiply(h, 1 - h)))
        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver=cp.SCS, max_iters=1000, eps=1e-4, time_limit=10)
        if result is None:
            return initial_guess
        return h.value
    
    # Optimization loop
    best_h = h_initial.copy()
    best_c5 = evaluate_overlap(best_h)
    elapsed = 0
    
    # Try with different numbers of points up to budget
    for n_points in [100, 200, 300, 400, 500]:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Initialize with the best solution from previous size
        h = best_h.copy()
        h = np.resize(h, n_points)
        h = h * (n_points / 2) / np.sum(h)  # Normalize to meet the integral constraint
        h = np.clip(h, 0, 1)
        
        # Try a few optimization steps
        for _ in range(3):
            h = optimize_h(h)
            c5 = evaluate_overlap(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        
        # Check if we can do a better job with a smaller number of points
        if n_points < 100:
            continue
        
        # Evaluate the current best solution
        c5 = evaluate_overlap(best_h)
        best_c5 = c5
    
    # Final check with the best solution
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)  # Normalize
    final_c5 = evaluate_overlap(final_h)
    
    return final_h, final_c5, n_points
```
