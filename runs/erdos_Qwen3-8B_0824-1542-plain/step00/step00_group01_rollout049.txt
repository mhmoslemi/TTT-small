```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Assuming the initial h is loaded here

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define the number of points; use initial h to determine size
    n_points_initial = len(initial_h_values)
    n_points = n_points_initial  # Start with same number of points as initial
    
    # Objective: minimize the maximum overlap integral
    # Constraints: sum(h) = n_points / 2 and 0 <= h[i] <= 1

    # Attempt to improve with more points
    for n_points in [n_points_initial, n_points_initial * 2, n_points_initial * 3]:
        if n_points > 2000:
            continue
        h = np.random.rand(n_points) * 2 - 1
        h = (h + 1) / 2  # Initial guess between 0 and 1
        h = h * (n_points / 2) / np.sum(h)  # Normalize to meet constraint

        # Define variables
        x = cp.Variable(n_points)

        # Objective: minimize the maximum overlap integral
        # Compute correlation with 1 - h for all shifts
        # We use a reduced form since full correlation is expensive
        # Only evaluate for a subset of shifts for efficiency

        # To speed up the optimization, compute only a subset of shifts
        shifts = np.random.choice(n_points, size=10, replace=False)
        overlap_energies = []
        for shift in shifts:
            corr = np.correlate(h, 1 - h, mode="full")
            overlap_energies.append(corr[shift + n_points - 1])

        obj = cp.Minimize(cp.Max(overlap_energies))

        # Constraints
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= 1
        ]

        # Solve the problem
        prob = cp.Problem(obj, constraints)
        try:
            start_time = time.time()
            result = prob.solve(solver=cp.ECOS, max_iter=1000, abstol=1e-8, reltol=1e-8)
            elapsed = time.time() - start_time
            if elapsed > budget_s:
                break
            if result is not None:
                h_opt = x.value
                h_opt = np.clip(h_opt, 0, 1)
                c5_bound = evaluate_erdos_solution(h_opt)
                return (h_opt, c5_bound, n_points)
        except:
            pass

    # If no improvement found, return initial h
    h_opt = initial_h_values
    c5_bound = evaluate_erdos_solution(h_opt)
    return (h_opt, c5_bound, len(h_opt))
```
