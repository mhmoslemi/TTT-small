```python
import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import minimize, Bounds
from scipy.linalg import block_diag
from scipy.signal import correlate

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial number of points
    n_points = 100  # Start with 100 points, can increase if needed
    max_iterations = 100  # Max number of iterations for optimization
    initial_h_values = np.array(initial_h_values) if 'initial_h_values' in globals() else np.random.rand(n_points) * 2 - 1
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    
    best_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    
    for _ in range(max_iterations):
        # Ensure the integral constraint is satisfied for the current h
        current_sum = np.sum(best_h)
        if np.abs(current_sum - n_points / 2) > 1e-6:
            # Adjust h to satisfy the constraint
            scale_factor = (n_points / 2) / current_sum
            best_h = best_h * scale_factor
        
        # Define the optimization problem
        def objective(h):
            return evaluate_erdos_solution(h)
        
        bounds = Bounds(0, 1)
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, jac=None, options={'maxiter': 10})
        
        if result.success:
            best_h = result.x
            best_c5 = result.fun
        
        # Check if we are out of time
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final adjustment to ensure the integral constraint is satisfied
    final_sum = np.sum(best_h)
    scale_factor = (n_points / 2) / final_sum
    best_h = best_h * scale_factor
    
    # Evaluate the final result
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, n_points
```
