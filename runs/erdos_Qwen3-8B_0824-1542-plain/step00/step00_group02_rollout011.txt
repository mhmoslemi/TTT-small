import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 45  # Start with the initial n_points
    
    # Initial guess from provided initial_h_values
    h = np.array(initial_h_values)
    if len(h) != n_points:
        h = np.linspace(0, 1, n_points)
        h = np.concatenate((h, h[::-1]))  # Mirror to make it symmetric
        h = h[:n_points]
    h = h / np.sum(h) * (n_points / 2)  # Ensure the sum is exactly n_points / 2
    
    # Function to compute the overlap integral
    def compute_c5(h):
        dx = 2.0 / n_points
        # Compute the correlation
        corr = signal.correlate(h, 1 - h, mode="full")
        # The overlap integral is max of the correlation times dx
        return np.max(corr) * dx
    
    # Objective function for optimization (minimize C5)
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        return compute_c5(h)
    
    # Constraints for optimization
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h_flat = h.copy()
    
    # Run optimization
    start_time = time.time()
    result = optimize.minimize(
        objective,
        h_flat,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    # Check if optimization was successful
    if result.success:
        h_opt = result.x
    else:
        h_opt = h  # Use initial guess if optimization failed
    
    # Compute the final C5 value
    c5_bound = compute_c5(h_opt)
    
    # If we have time, try increasing the number of points
    if time.time() - start_time < budget_s - 10:
        n_points = 60
        h = np.zeros(n_points)
        # Distribute mass evenly to start
        h[:int(n_points / 2)] = 1
        h[int(n_points / 2):] = 0
        # Convert to a more balanced initial guess
        h = np.concatenate((h, h[::-1]))[:n_points]
        h = h / np.sum(h) * (n_points / 2)
        
        # Redefine the compute_c5 function with new n_points
        def compute_c5_new(h):
            dx = 2.0 / n_points
            corr = signal.correlate(h, 1 - h, mode="full")
            return np.max(corr) * dx
        
        def objective_new(h_flat):
            h = h_flat.reshape(n_points)
            return compute_c5_new(h)
        
        def constraint_sum_new(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) - n_points / 2
        
        # Run optimization again with new n_points
        result_new = optimize.minimize(
            objective_new,
            h.copy(),
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint_sum_new}],
            options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        if result_new.success:
            h_opt = result_new.x
        else:
            h_opt = h  # Use initial guess if optimization failed
        
        c5_bound = compute_c5_new(h_opt)
        h_opt = h_opt.reshape(n_points)
    
    # Final check to ensure the sum is correct
    h_opt = h_opt.reshape(n_points)
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt / np.sum(h_opt) * (n_points / 2)
    
    return h_opt, c5_bound, n_points
