```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    h_shifted = np.concatenate([np.zeros(n_points), h[:-1]])
    overlap = np.correlate(h, 1 - h_shifted, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def initial_h_values():
    # A sample initial solution
    n_points = 70
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            h[i] = 1.0
        elif i % 3 == 1:
            h[i] = 0.5
        else:
            h[i] = 0.0
    h = h / (np.sum(h) * 2.0 / n_points)
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try different n_points values
    n_points_list = [100, 120, 140, 160, 180, 200]
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0

    for n_points in n_points_list:
        start_time = time.time()
        
        # Initialize h
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                h[i] = 1.0
            elif i % 3 == 1:
                h[i] = 0.5
            else:
                h[i] = 0.0
        h = h / (np.sum(h) * 2.0 / n_points)
        
        # Refine h using gradient descent
        h = h.tolist()
        
        # Optimization
        for _ in range(200):
            # Convert to numpy array
            h_arr = np.array(h)
            
            # Compute current C5
            dx = 2.0 / n_points
            h_shifted = np.concatenate([np.zeros(n_points), h_arr[:-1]])
            overlap = np.correlate(h_arr, 1 - h_shifted, mode="full")
            current_c5 = np.max(overlap) * dx
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            # Simple gradient descent step
            dx = 2.0 / n_points
            h_shifted = np.concatenate([np.zeros(n_points), h_arr[:-1]])
            overlap = np.correlate(h_arr, 1 - h_shifted, mode="full")
            max_overlap_idx = np.argmax(overlap)
            h_shifted = np.concatenate([np.zeros(n_points), h_arr[:-1]])
            overlap = np.correlate(h_arr, 1 - h_shifted, mode="full")
            weights = np.zeros(n_points)
            for i in range(n_points):
                for k in range(n_points):
                    if max_overlap_idx == i + k:
                        if i < n_points:
                            weights[i] += h_shifted[k]
            
            # Update h
            for i in range(n_points):
                if h[i] < 1.0 and weights[i] > 0:
                    h[i] += 0.01
                elif h[i] > 0.0 and weights[i] < 0:
                    h[i] -= 0.01
                h[i] = np.clip(h[i], 0.0, 1.0)
            
            h_arr = np.array(h)
            if np.sum(h_arr) * dx < 1.0 - 1e-6 or np.sum(h_arr) * dx > 1.0 + 1e-6:
                break
        
        elapsed_time += time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    if best_h is None:
        best_h = initial_h_values()
        best_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, best_c5, len(best_h)
```
