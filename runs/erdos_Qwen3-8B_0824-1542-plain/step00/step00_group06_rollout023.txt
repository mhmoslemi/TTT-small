```python
import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)

    # Initial parameters
    n_points = 100  # Start with 100 points, adjust if needed
    dx = 2.0 / n_points
    h = initial_h_values.copy()[:n_points]  # Use initial construction
    h = np.clip(h, 0, 1)  # Ensure values are in [0,1]
    h = h * (n_points / 2) / np.sum(h)  # Normalize to meet the integral constraint

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    time_remaining = budget_s
    start_time = time.time()

    # Optimization settings
    max_iterations = 100
    for _ in range(max_iterations):
        if time_remaining <= 0:
            break
        time_remaining = budget_s - (time.time() - start_time)

        # Perturb the current solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)

        # Evaluate the perturbed solution
        c5 = evaluate_erdos_solution(h_perturbed)

        # Update best solution if improvement
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturbed.copy()

        # Update current solution
        h = best_h.copy()

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, n_points)
```
