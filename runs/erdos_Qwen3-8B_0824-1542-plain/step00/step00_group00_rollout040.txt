def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros(2 * n_points)
    shifted_h[:n_points] = h
    shifted_h[n_points:] = h
    result = np.correlate(h, 1 - h, mode="full")
    result = result * dx
    return np.max(result)

def run(seed=42, budget_s=60, **kwargs):
    # Define the initial configuration
    h_initial = initial_h_values()
    n_points_initial = len(h_initial)
    dx_initial = 2.0 / n_points_initial
    sum_h_initial = np.sum(h_initial) * dx_initial
    # Ensure the initial configuration meets the integral constraint
    if not np.isclose(sum_h_initial, 1):
        h_initial = h_initial * (1 / sum_h_initial)
    
    # Try different n_points
    n_points_list = [n_points_initial, n_points_initial - 10, n_points_initial - 20, n_points_initial - 30, n_points_initial - 40]
    n_points_list = [n for n in n_points_list if n > 20]
    
    best_c5 = float('inf')
    best_h = h_initial
    best_n_points = n_points_initial
    
    # Try each n_points in the list
    for n_points in n_points_list:
        if n_points < 2:
            continue
        
        # Define the problem variables
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1]
        constraints.append(cp.sum(h) == n_points / 2)
        
        # Define the objective function
        # We aim to minimize the maximum overlap integral
        # This is a minimax problem, which is challenging to optimize directly
        # We instead use a heuristic that tries to minimize the maximum of the integral over k
        
        # We evaluate the overlap integral for different lags and choose the maximum
        def objective(h_vec):
            h = h_vec
            dx = 2.0 / n_points
            shifted_h = np.zeros(2 * n_points)
            shifted_h[:n_points] = h
            shifted_h[n_points:] = h
            result = np.correlate(h, 1 - h, mode="full")
            result = result * dx
            return np.max(result)
        
        # Use a local optimization approach to minimize the maximum overlap integral
        # Start from the initial configuration
        h_start = np.zeros(n_points)
        h_start[:len(h_initial)] = h_initial
        h_start = h_start[:n_points]
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        h_start = np.clip(h_start, 0, 1)
        
        # Run a local optimization
        start_time = time.time()
        result = optimize.minimize(objective, h_start, method='L-BFGS-B', bounds=[(0, 1)] * n_points, tol=1e-6)
        elapsed = time.time() - start_time
        
        if elapsed > budget_s:
            continue
        
        h_opt = result.x
        c5 = objective(h_opt)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            best_n_points = n_points
    
    # Return the best solution found
    return (best_h, best_c5, best_n_points)
