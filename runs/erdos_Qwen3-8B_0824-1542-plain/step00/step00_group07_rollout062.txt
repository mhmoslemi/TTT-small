```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Placeholder for actual initial solution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points, can increase if needed
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5  # Initial guess with uniform distribution
    h = h / np.sum(h) * (n_points / 2)  # Normalize to meet integral constraint

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Use a gradient-based optimizer with constraints
        x = cp.Variable(n_points)
        constraints = [
            0 <= x,
            x <= 1,
            cp.sum(x) == n_points / 2
        ]
        objective = cp.Maximize(cp.max(cp.sum(cp.multiply(x, 1 - cp.roll(x, 1)), axis=0)))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-4, max_iters=100)
            if prob.status == 'optimal':
                h_candidate = x.value
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass  # Skip if solver fails

        # Perturb the solution slightly to escape local optima
        h = best_h + 0.01 * np.random.randn(n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)

    return best_h, best_c5, n_points
```
