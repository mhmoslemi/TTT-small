```python
import numpy as np
import scipy.optimize as opt
import time
import sys
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Parameters
    max_n_points = 512  # Max number of points to try
    start_n_points = 64  # Start with smaller number of points
    n_points = start_n_points  # Initial number of points
    
    # Function to compute the overlap integral
    def compute_c5(h_values):
        dx = 2.0 / len(h_values)
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Function to optimize h_values
    def optimize_h(n_points):
        dx = 2.0 / n_points
        h = cp.Variable(n_points, bounds=(0, 1))
        constraint = cp.sum(h) == n_points / 2
        objective = cp.Minimize(0)  # Just optimize the constraints
        
        problem = cp.Problem(objective, [constraint])
        problem.solve(solver=cp.SCS, eps=1e-4, max_iters=100)
        
        if problem.status not in ['OPTIMAL', 'PRIMAL_INFEASIBLE', 'DUAL_INFEASIBLE']:
            return None
        
        h_values = h.value
        c5 = compute_c5(h_values)
        return h_values, c5
    
    # Function to perform local search
    def local_search(h, n_points):
        dx = 2.0 / n_points
        h = np.array(h)
        best_h = h.copy()
        best_c5 = compute_c5(best_h)
        
        for _ in range(100):
            # Random perturbation
            idx = np.random.randint(n_points)
            h[idx] += np.random.uniform(-0.1, 0.1)
            h = np.clip(h, 0, 1)
            while np.sum(h) < n_points / 2 - 1e-6 or np.sum(h) > n_points / 2 + 1e-6:
                h[idx] += np.random.uniform(-0.1, 0.1)
                h = np.clip(h, 0, 1)
            
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        
        return best_h, best_c5
    
    # Initial solution
    h_initial = initial_h_values
    if len(h_initial) != start_n_points:
        h_initial = np.interp(np.linspace(0, 2, start_n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    h_initial = np.clip(h_initial, 0, 1)
    
    best_h = h_initial
    best_c5 = compute_c5(best_h)
    best_n_points = start_n_points
    
    start_time = time.time()
    elapsed = 0
    
    # Try increasing number of points
    while elapsed < budget_s - 1 and n_points <= max_n_points:
        print(f"Trying {n_points} points")
        # Try optimization
        h_opt, c5_opt = optimize_h(n_points)
        if h_opt is not None:
            h_opt, c5_opt = local_search(h_opt, n_points)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                best_n_points = n_points
        
        # Try local search
        h_ls, c5_ls = local_search(best_h, n_points)
        if c5_ls < best_c5:
            best_h = h_ls
            best_c5 = c5_ls
        
        # Update elapsed time
        elapsed = time.time() - start_time
        n_points *= 2
    
    # Final check with the best solution
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, best_n_points
```
