```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define parameters
    n_points = 300  # Number of points for the function h
    dx = 2.0 / n_points
    target_C5 = 0.380876  # Target bound to beat
    max_time = budget_s  # Time budget in seconds
    
    # Initialize with the initial guess
    h = np.copy(initial_h_values)
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), h)
    
    # Ensure the integral is 1
    h = h * (n_points / 2) / np.sum(h)
    
    # Function to evaluate the C5 value
    def evaluate_C5(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        shifted_h = np.zeros((n_points, n_points))
        for i in range(n_points):
            shifted_h[i, :] = h
        kernel = 1 - h
        correlation = signal.correlate(h, kernel, mode='full')
        c5 = np.max(correlation * dx)
        return c5
    
    # Function to perform local optimization around the current h
    def optimize_local(h_current):
        h = cp.Variable(n_points)
        constraint = cp.sum(h) == n_points / 2
        objective = cp.Minimize(cp.max(cp.multiply(h, 1 - h[cp.diag_indices(n_points)])))
        prob = cp.Problem(objective, [constraint])
        try:
            result = prob.solve()
            if result is not None:
                h_opt = h.value
                h_opt = np.clip(h_opt, 0, 1)
                return h_opt
        except:
            pass
        return h_current
    
    # Function to perform global search using random perturbations
    def perturb_h(h, magnitude=0.1):
        h_perturbed = h + magnitude * np.random.randn(n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        return h_perturbed
    
    # Main optimization loop
    start_time = time.time()
    best_h = h
    best_c5 = evaluate_C5(h)
    
    for _ in range(300):  # Run for a fixed number of iterations
        if time.time() - start_time > max_time:
            break
        
        # Local optimization
        h_opt = optimize_local(best_h)
        c5_opt = evaluate_C5(h_opt)
        
        # Perturb and evaluate
        h_perturbed = perturb_h(best_h)
        c5_perturbed = evaluate_C5(h_perturbed)
        
        # Choose the better of the two
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
        if c5_perturbed < best_c5:
            best_h = h_perturbed
            best_c5 = c5_perturbed
        
        # Early stopping if we've beat the target
        if best_c5 < target_C5:
            break
    
    # Final evaluation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_C5(final_h)
    
    return final_h, final_c5, n_points
```
