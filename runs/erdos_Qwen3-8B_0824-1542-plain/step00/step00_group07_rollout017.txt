```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial solution as a starting point
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    
    # Evaluate the initial solution
    c5_bound = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = c5_bound
    
    # Try optimizing with different n_points values
    for n_points in [80, 160, 320, 640]:
        if time.time() - start_time > budget_s - 10:
            break
        
        # Ensure the number of points is even for symmetry
        if n_points % 2 != 0:
            n_points += 1
        
        # Initialize h with a uniform distribution
        h = np.random.rand(n_points) * 2 - 1  # Random in [-1, 1]
        h = (h + 1) / 2  # Scale to [0, 1]
        h = h * (n_points / 2) / (np.sum(h))  # Normalize to meet the integral constraint
        
        # Optimization with simulated annealing
        def objective(x):
            h_vals = x.copy()
            h_vals = np.clip(h_vals, 0, 1)
            # Compute the overlap integral
            dx = 2.0 / n_points
            corr = signal.correlate(h_vals, 1 - h_vals, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        def constraint(x):
            return np.sum(x) - n_points / 2
        
        constraints = ({'type': 'eq', 'fun': constraint})
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use SLSQP with bounds and constraints
        result = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'disp': False}
        )
        
        h_vals = result.x
        c5 = evaluate_erdos_solution(h_vals)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_vals.copy()
        
        print(f"n_points={n_points}, c5={c5}")
    
    # Final evaluation with best solution
    best_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, best_c5, len(best_h)
```
