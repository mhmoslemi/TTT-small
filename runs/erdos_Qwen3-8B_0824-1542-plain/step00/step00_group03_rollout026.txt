import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(70) * 2 - 1  # Placeholder for initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # Start with a manageable number of points
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 2 - 1  # Random initial values between -1 and 1
    h = np.clip(h, 0, 1)  # Clip to [0, 1]
    h = h * (n_points / 2) / np.sum(h)  # Normalize to satisfy the integral constraint

    # Initial evaluation
    c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = c5

    # Optimization loop
    for _ in range(100):  # Limit iterations to stay within time budget
        if time.time() - start_time > budget_s - 1:
            break

        # Define optimization variables
        x = cp.Variable(n_points, nonneg=True)
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= 1
        ]

        # Define objective: minimize the maximum overlap integral
        shifted_h = np.roll(x.value, 1) if x.value is not None else np.roll(h, 1)
        overlap = np.correlate(x.value, 1 - shifted_h, mode="full") * dx
        max_overlap = cp.Max(overlap)

        # Use a convex relaxation if needed or use a more refined approach
        # For simplicity, we'll use a local optimization approach
        # Here, we'll use scipy.optimize to minimize the max overlap
        # We'll optimize for h values that reduce the max overlap

        # Define a function to compute the max overlap
        def objective_func(h_values):
            shifted_h = np.roll(h_values, 1)
            overlap = np.correlate(h_values, 1 - shifted_h, mode="full") * dx
            return np.max(overlap)

        # Use L-BFGS-B for optimization
        result = opt.minimize(
            objective_func,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            jac=None,
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )

        h = result.x
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

        # Evaluate
        c5 = evaluate_erdos_solution(h)

        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Optional: reduce n_points if needed to stay within time
        if n_points > 50 and time.time() - start_time > budget_s - 5:
            n_points = 50
            dx = 2.0 / n_points
            h = best_h[:n_points]
            h = h * (n_points / 2) / np.sum(h)

    return best_h, best_c5, n_points
