```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Example initial construction

def run(seed=42, budget_s=60, **kwargs):
    # Parameters
    n_points = 100  # Initial number of points
    max_points = 200  # Maximum number of points to try
    max_iterations = 100  # Max iterations for optimization
    initial_h = initial_h_values
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = len(best_h)

    # Start with the initial solution
    print(f"Starting with n={len(best_h)} points, C5={best_c5:.6f}")
    start_time = time.time()

    for n in range(len(best_h) + 1, max_points + 1):
        if time.time() - start_time > budget_s - 5:
            break

        # Resample the initial solution to n points
        h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h)), initial_h)
        h = np.clip(h, 0, 1)
        h = h * (n / 2)  # Ensure integral is 1
        h = h / np.mean(h)  # Normalize to ensure the integral is 1

        # Use a heuristic to refine the function
        for _ in range(10):
            # Evaluate the overlap
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n_points = n

            # Try optimizing with a small perturbation
            perturb = np.random.normal(0, 0.05, n)
            h_perturbed = np.clip(h + perturb, 0, 1)
            h_perturbed = h_perturbed * (n / 2)  # Ensure integral is 1
            h_perturbed = h_perturbed / np.mean(h_perturbed)

            c5_perturbed = evaluate_erdos_solution(h_perturbed)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
                best_n_points = n

    # Final evaluation
    best_c5 = evaluate_erdos_solution(best_h)
    return best_h, best_c5, best_n_points
```
