import numpy as np

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Start with 200 points for balance between speed and accuracy
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()  # Use the initial construction
    initial_h /= np.sum(initial_h) * dx  # Normalize to ensure ∫h(x)dx = 1
    initial_h = np.clip(initial_h, 0, 1)  # Ensure values are in [0, 1]

    # Initial evaluation
    h_values = initial_h
    best_c5 = compute_c5(h_values, dx)

    # Simple greedy improvement: adjust h(x) to reduce overlap
    for _ in range(50):  # Max iterations to avoid infinite loops
        start_time = time.time()
        if time.time() - start_time > budget_s:
            break

        # Estimate overlap for each shift (we only care about shifts up to 1)
        shifts = np.arange(0, 1.5, 0.1)
        overlap = []
        for k in shifts:
            shifted_h = np.roll(h_values, int(k * n_points / dx))
            shifted_h = np.clip(shifted_h, 0, 1)
            current_overlap = np.sum(h_values * (1 - shifted_h)) * dx
            overlap.append(current_overlap)

        # Find the shift with maximum overlap (we take the worst-case shift)
        max_overlap_idx = np.argmax(overlap)
        max_overlap_shift = shifts[max_overlap_idx]

        # Adjust h(x) to reduce overlap at the worst shift
        # Find the shift in terms of index
        k_idx = int(max_overlap_shift * n_points / dx)
        shifted_h = np.roll(h_values, k_idx)
        adjustment = h_values * (1 - shifted_h)
        adjustment = np.clip(adjustment, 0, 1)

        # Perturb h_values to reduce overlap
        perturbation = np.random.normal(0, 0.05, n_points)
        h_values += perturbation
        h_values = np.clip(h_values, 0, 1)
        h_values /= np.sum(h_values) * dx  # Maintain ∫h(x)dx = 1

        # Evaluate new overlap
        new_c5 = compute_c5(h_values, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            h_values = h_values.copy()

    # Try a more precise optimization with cvxpy
    # Convert h_values to a variable
    h = cp.Variable(n_points, nonneg=True)
    constraints = [cp.sum(h) == n_points / 2]  # ∫h(x)dx = 1
    objective = cp.Minimize(cp.Max([cp.sum(h * (1 - cp.roll(h, k_idx))) * dx for k_idx in range(1, 100)]))

    # Run optimization within budget
    start_time = time.time()
    while time.time() - start_time < budget_s:
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == 'optimal':
                h_values = np.clip(h.value, 0, 1)
                h_values /= np.sum(h_values) * dx
                best_c5 = compute_c5(h_values, dx)
                break
        except:
            break

    return h_values, best_c5, n_points

def compute_c5(h, dx):
    # Compute the overlap integral for all shifts
    shifts = np.arange(0, 2, dx)
    max_overlap = 0
    for k in shifts:
        shifted_h = np.roll(h, int(k * len(h) / dx))
        shifted_h = np.clip(shifted_h, 0, 1)
        overlap = np.sum(h * (1 - shifted_h)) * dx
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap
