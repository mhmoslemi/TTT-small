```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import os
import random
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 100  # Start with 100 points for speed, can increase if needed
    dx = 2.0 / n_points
    max_iterations = 10  # Limit iterations to stay within time budget
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h_np = np.array(h)
        corr = signal.correlate(h_np, 1 - h_np, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Function to compute the objective and constraints
    def objective_function(h):
        return compute_c5(h)
    
    # Function to compute the constraint (integral of h is 1)
    def constraint_function(h):
        return np.sum(h) - n_points / 2
    
    # Function to compute the bounds for optimization
    def bounds_function(h):
        return [(0, 1) for _ in range(n_points)]
    
    # Start from initial solution
    h_initial = initial_h_values.copy()
    h_initial = h_initial[:n_points]
    
    # Initial C5 value
    best_c5 = compute_c5(h_initial)
    best_h = h_initial.copy()
    
    # Try different starting points if needed
    for _ in range(max_iterations):
        # Create a new random starting point
        h_new = np.random.uniform(0, 1, n_points)
        # Enforce the integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        c5_new = compute_c5(h_new)
        
        # Use scipy.optimize to find a better solution
        h_opt = cp.Variable(n_points)
        obj = cp.Minimize(objective_function(h_opt))
        constr = [cp.sum(h_opt) == n_points / 2]
        prob = cp.Problem(obj, constr)
        
        # Solve the optimization problem
        start_time = time.time()
        try:
            prob.solve(solver='ECOS', verbose=False)
            if prob.status == 'OPTIMAL':
                h_new = h_opt.value
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
        except:
            pass
        
        # If within time budget and improvement is found, increase n_points
        if time.time() - start_time > budget_s * 0.5 and best_c5 < 0.3808:
            n_points *= 2
            dx = 2.0 / n_points
            h_initial = np.random.uniform(0, 1, n_points)
            h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
            best_h = h_initial.copy()
            best_c5 = compute_c5(h_initial)
    
    # Final evaluation
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
```
