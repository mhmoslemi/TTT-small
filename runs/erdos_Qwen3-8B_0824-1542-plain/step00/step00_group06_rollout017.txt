import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial configuration
    n_points = 95  # Starting with the initial n_points
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / sum_h

    # Function to compute C5
    def compute_c5(h):
        n = len(h)
        dx = 2.0 / n
        h_extended = np.zeros(2 * n - 1)
        h_extended[:n] = h
        h_shifted = np.zeros(2 * n - 1)
        h_shifted[n-1:] = h
        corr = np.correlate(h_extended, 1 - h_shifted, mode="full")
        return np.max(corr) * dx

    # Function to evaluate the loss
    def evaluate_loss(h):
        return compute_c5(h)

    # Function to perform a single optimization step with gradient descent
    def optimize_h(h_initial, max_iter=100, step_size=0.01, tol=1e-6):
        h = h_initial.copy()
        losses = []
        for _ in range(max_iter):
            current_loss = evaluate_loss(h)
            losses.append(current_loss)
            if current_loss < tol:
                break
            # Compute gradient numerically
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                loss_plus = evaluate_loss(h_plus)
                h_minus = h.copy()
                h_minus[i] -= 1e-5
                loss_minus = evaluate_loss(h_minus)
                grad[i] = (loss_plus - loss_minus) / (2 * 1e-5)
            # Update h
            h -= step_size * grad
            # Ensure h stays in [0, 1]
            h = np.clip(h, 0.0, 1.0)
            # Normalize sum(h) to n_points / 2
            sum_h = np.sum(h)
            if sum_h != 0:
                h = h * (n_points / 2) / sum_h
        return h, losses

    # Run the optimization
    h_optimized, losses = optimize_h(h, max_iter=100, step_size=0.01, tol=1e-6)
    best_c5 = compute_c5(h_optimized)
    
    # Refine with a grid search over small perturbations
    for _ in range(5):
        perturbations = np.random.uniform(-0.05, 0.05, size=len(h_optimized))
        h_perturbed = np.clip(h_optimized + perturbations, 0.0, 1.0)
        sum_h_perturbed = np.sum(h_perturbed)
        if sum_h_perturbed != 0:
            h_perturbed = h_perturbed * (n_points / 2) / sum_h_perturbed
        current_c5 = compute_c5(h_perturbed)
        if current_c5 < best_c5:
            best_c5 = current_c5
            h_optimized = h_perturbed
    
    # Check time budget
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (h_optimized, best_c5, n_points)
    
    # Try increasing n_points if time allows
    remaining_time = budget_s - elapsed_time
    if remaining_time > 0.5:
        n_points_new = int(n_points * 1.5)
        h_new = np.zeros(n_points_new)
        # Map the optimized h to the new n_points
        for i in range(n_points_new):
            idx = int(i * n_points / n_points_new)
            h_new[i] = h_optimized[idx] if idx < len(h_optimized) else h_optimized[-1]
        # Normalize to sum(n_points_new / 2)
        sum_h_new = np.sum(h_new)
        if sum_h_new != 0:
            h_new = h_new * (n_points_new / 2) / sum_h_new
        # Optimize the new h
        h_new_optimized, _ = optimize_h(h_new, max_iter=50, step_size=0.05, tol=1e-6)
        new_c5 = compute_c5(h_new_optimized)
        if new_c5 < best_c5:
            best_c5 = new_c5
            h_optimized = h_new_optimized
            n_points = n_points_new
    
    return (h_optimized, best_c5, n_points)
