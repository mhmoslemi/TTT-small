```python
import numpy as np
import scipy.optimize as optimize
import copy
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    
    # Define function to compute C_5 for a given h_values
    def compute_c5(h):
        n_points = len(h)
        dx = 2.0 / n_points
        h = np.array(h)
        shifted = np.zeros_like(h)
        for k in range(1, n_points):
            shifted[k:] = h[:-k]
        overlap = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(overlap) * dx
        return c5
    
    # Define function to compute the objective and constraints
    def objective(x):
        return compute_c5(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Function to generate initial guess
    def generate_initial_guess(n_points):
        h = np.zeros(n_points)
        if n_points <= 20:
            # For small n, use equidistant 1s and 0s
            h[:n_points // 2] = 1
        else:
            # For larger n, start with the initial guess
            h = copy.deepcopy(initial_h_values)
            h = np.array(h)
            if len(h) != n_points:
                h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        return h
    
    # Main optimization loop
    best_h = None
    best_c5 = float('inf')
    
    # Try different numbers of points
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate initial guess
        h = generate_initial_guess(n_points)
        h = np.clip(h, 0, 1)
        
        # Set up bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        
        # Use L-BFGS-B optimizer
        result = optimize.minimize(
            objective,
            h,
            bounds=bounds,
            constraints=cons,
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        
        # Evaluate the result
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_c5 = compute_c5(current_h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        
        # Optionally, try perturbations
        for _ in range(3):
            perturb = np.random.normal(0, 0.05, n_points)
            perturb = np.clip(perturb, -0.1, 0.1)
            perturbed_h = current_h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_c5 = compute_c5(perturbed_h)
            if perturbed_c5 < best_c5:
                best_c5 = perturbed_c5
                best_h = perturbed_h
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    # Return the best solution
    return (best_h.tolist(), final_c5, len(best_h))
```
