import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(signal.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load("initial_h.npy")  # Assuming initial_h.npy contains the initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial number of points
    n_points = len(initial_h_values)
    if n_points < 100:
        # Start with a larger number of points to allow finer adjustments
        n_points = 200
    
    # Initial guess for h_values
    h_values = initial_h_values.copy()
    if len(h_values) != n_points:
        h_values = np.random.rand(n_points)
        h_values = h_values / np.sum(h_values) * (n_points / 2)
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        # Normalize h_values to ensure sum(h_values) == n_points / 2
        h_values = h_values / np.sum(h_values) * (n_points / 2)
        # Compute the correlation
        corr = signal.correlate(h_values, 1 - h_values, mode="full")
        # Return the maximum value of the correlation
        return np.max(corr) * (2.0 / n_points)
    
    # Function to update h_values in a way that respects the constraints
    def update_h(h_values):
        # Normalize h_values to ensure sum(h_values) == n_points / 2
        h_values = h_values / np.sum(h_values) * (n_points / 2)
        # Ensure h_values are within [0, 1]
        h_values = np.clip(h_values, 0, 1)
        return h_values
    
    # Optimization loop
    best_c5 = np.inf
    best_h = h_values.copy()
    
    # Use a simple gradient descent approach with random perturbations
    for _ in range(100):
        # Random perturbation
        perturbation = np.random.rand(n_points) * 0.1 - 0.05
        h_values = h_values + perturbation
        h_values = update_h(h_values)
        
        # Evaluate the objective
        current_c5 = objective(h_values)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()
        
        # Check if we have exceeded the time budget
        if time.time() - start_time > budget_s:
            break
    
    # Use CVXPY to find a better solution with constraints
    h = cp.Variable(n_points)
    constraints = [
        cp.sum(h) == n_points / 2,
        0 <= h,
        h <= 1
    ]
    objective = cp.Maximize(cp.multiply(signal.correlate(h, 1 - h, mode="full"), 2.0 / n_points))
    prob = cp.Problem(objective, constraints)
    
    # Solve the problem
    try:
        prob.solve(solver=cp.SCS, max_iter=100)
        if prob.status == "optimal":
            best_h = h.value
            best_c5 = objective.value
    except:
        pass
    
    # Ensure h_values are within [0, 1] and sum to n_points / 2
    best_h = update_h(best_h)
    
    # Final evaluation
    best_c5 = objective(best_h)
    
    # Return the best solution
    return (best_h, best_c5, n_points)
