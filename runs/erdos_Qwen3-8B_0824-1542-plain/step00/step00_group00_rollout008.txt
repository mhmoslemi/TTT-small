import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Assuming initial construction is in a file

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try starting from initial_h_values
    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points
    c5_bound = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = c5_bound
    
    # Try a few other initial points
    for _ in range(3):
        # Generate random h with sum(h) == n_points / 2
        h = np.random.rand(n_points)
        h = (h - np.min(h)) / (np.max(h) - np.min(h))  # Normalize to [0,1]
        h = h - np.min(h)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Now optimize with more refined approach
    h = best_h.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Try a few different point counts
    for n in [n_points, n_points - 5, n_points + 5]:
        if n < 10:
            continue
        h = np.random.rand(n)
        h = (h - np.min(h)) / (np.max(h) - np.min(h))  # Normalize to [0,1]
        h = h - np.min(h)
        h = h / np.sum(h) * (n / 2)
        h = np.clip(h, 0, 1)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            n_points = n
    
    # Perform a more structured optimization for best_h
    h = best_h.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Define the objective function
    def objective(x):
        h = x.reshape(n_points)
        c5 = evaluate_erdos_solution(h)
        return c5
    
    # Define the constraint: sum(h) = n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    # Set up the optimization problem
    x = cp.Variable(n_points)
    obj = cp.Minimize(objective(x))
    cons = [cp.sum(x) == n_points / 2]
    prob = cp.Problem(obj, cons)
    
    # Solve the optimization problem
    try:
        result = prob.solve(solver=cp.SCS, max_iter=100)
        if result is not None:
            h_opt = x.value
            c5_opt = evaluate_erdos_solution(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    except:
        pass
    
    # Return the best result
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 1:
        # Try a few more iterations with different initial guesses
        for _ in range(3):
            h = np.random.rand(n_points)
            h = (h - np.min(h)) / (np.max(h) - np.min(h))  # Normalize to [0,1]
            h = h - np.min(h)
            h = h / np.sum(h) * (n_points / 2)
            h = np.clip(h, 0, 1)
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return (best_h, best_c5, n_points)
