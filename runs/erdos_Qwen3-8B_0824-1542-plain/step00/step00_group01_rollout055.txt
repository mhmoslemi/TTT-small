```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions and initial solution
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 47  # Starting with the initial solution
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    # Function to compute C5
    def compute_c5(h_values):
        h_array = np.array(h_values)
        overlap = np.correlate(h_array, 1 - h_array, mode="full")
        return np.max(overlap) * dx
    
    # Function to optimize h
    def optimize_h(h_values, max_iter=100, tolerance=1e-5):
        n_points = len(h_values)
        dx = 2.0 / n_points
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1]
        constraints.append(cp.sum(h) == n_points / 2)
        
        # Objective: minimize the maximum overlap integral
        # We use a convex relaxation by minimizing the maximum of linear functions
        # Using a more efficient approach with a binary search on the upper bound
        best_c5 = np.inf
        best_h = h_values.copy()
        
        # Binary search over possible C5 values
        low = 0.0
        high = 1.0
        for _ in range(100):
            mid = (low + high) / 2
            # Define the constraints for this mid value
            constraints_c5 = [cp.sum(cp.multiply(h, 1 - h[k:])) <= mid * dx for k in range(n_points)]
            prob = cp.Problem(cp.Minimize(0), constraints + constraints_c5)
            try:
                prob.solve(solver=cp.SCS, max_iters=1000, eps=1e-6, verbose=False)
                if prob.status == 'optimal':
                    if mid < best_c5:
                        best_c5 = mid
                        best_h = h.value
            except:
                pass
            # Adjust the search bounds based on the result
            if prob.status == 'optimal':
                low = mid
            else:
                high = mid
        
        return best_h, best_c5
    
    # Run the optimization
    start_time = time.time()
    best_h, best_c5 = optimize_h(h)
    
    # Evaluate the best solution
    best_c5 = compute_c5(best_h)
    print(f"Found C5 = {best_c5}")
    
    # Return the result
    return best_h, best_c5, len(best_h)
```
