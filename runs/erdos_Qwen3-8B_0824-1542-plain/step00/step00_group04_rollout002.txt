```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Try different n_points values starting from the initial one and working up
    n_points_list = [42, 60, 80, 100, 120, 140, 160, 180, 200, 220, 240, 260, 280, 300]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Compute dx
        dx = 2.0 / n_points
        
        # Initialize h with the initial construction (scaled to fit the current n_points)
        if n_points == 42:
            h = initial_h_values.copy()
        else:
            # Interpolate the initial h_values linearly to the new n_points
            # For simplicity, we'll just duplicate the initial values and rescale
            # This is a crude interpolation, but works for demonstration
            h = np.linspace(0, 2, n_points)
            h_initial = initial_h_values.copy()
            h_initial = np.interp(h, np.linspace(0, 2, len(h_initial)), h_initial)
            h = np.interp(h, np.linspace(0, 2, n_points), h_initial)
        
        # Rescale to meet the integral constraint
        h = h / np.sum(h) * (n_points / 2)
        
        # Initial attempt
        current_c5 = evaluate_erdos_solution(h, dx)
        
        # If this is better, update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            best_n_points = n_points
        
        # If we're below the target, return early
        if best_c5 <= 0.380876:
            break
        
        # Try to refine with optimization
        x = cp.Variable(n_points)
        constraints = [0 <= x, x <= 1]
        constraints.append(cp.sum(x) == n_points / 2)
        
        # Define the objective: minimize the maximum overlap
        # To avoid maximizing directly, we use a trick with binary variables
        # We minimize a variable t such that t >= overlap for all k
        t = cp.Variable(n_points)
        constraints += [t >= 0]
        
        # Define the overlap for each k
        for k in range(1, n_points):
            overlap = cp.sum(x * (1 - x[k:])) / (n_points * dx)
            constraints += [t >= overlap]
        
        # Objective: minimize t
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            result = prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if result is not None and result < best_c5:
                best_c5 = result
                best_h = x.value.copy()
                best_n_points = n_points
        except:
            pass
        
        # If we're below the target, return early
        if best_c5 <= 0.380876:
            break
    
    # Final check to ensure we have a valid solution
    if best_h is None:
        best_h = initial_h_values.copy()
        best_c5 = evaluate_erdos_solution(best_h, 2.0 / len(best_h))
    
    return best_h, best_c5, best_n_points
```
