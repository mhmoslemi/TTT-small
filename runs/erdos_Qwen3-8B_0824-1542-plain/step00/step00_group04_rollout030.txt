import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for faster computation
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    # Use initial_h_values provided if available
    if 'initial_h_values' in globals():
        initial_h_values = np.array(initial_h_values)
        if len(initial_h_values) != n_points:
            initial_h_values = np.zeros(n_points)
    else:
        initial_h_values = np.zeros(n_points)
    # Make sure the initial_h_values sum to n_points / 2
    initial_h_values = np.tile([0.5, 0.5], (n_points // 2 + 1, 1))[:n_points].flatten()
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)

    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        # Compute the max overlap integral
        overlap_integral = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap_integral)

    # Optimization function: minimize the max overlap integral
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        overlap_integral = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap_integral)

    # Use the initial guess
    initial_guess = initial_h_values.copy()

    # Run optimization
    start_time = time.time()
    result = optimize.minimize(
        objective,
        initial_guess,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'ftol': 1e-6, 'gtol': 1e-6, 'maxiter': 100}
    )
    elapsed_time = time.time() - start_time

    # If time is still left, try increasing n_points
    if elapsed_time < budget_s:
        n_points = 200
        dx = 2.0 / n_points
        initial_h_values = np.zeros(n_points)
        # Use initial_h_values provided if available
        if 'initial_h_values' in globals():
            initial_h_values = np.array(initial_h_values)

            h = np.clip(h, 0, 1)

            # Compute the max overlap integral
            overlap_integral = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(overlap_integral)

        def objective_new(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)

            return np.max(overlap_integral)

        # Use the best solution from the previous step as initial guess
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h / np.sum(best_h) * (n_points / 2)
        best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)

        initial_guess_new = best_h.copy()

        # Run optimization for n_points = 200
        result_new = optimize.minimize(
            objective_new,

            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],

        # Compare the results and choose the best one
        c5_old = result.fun
        c5_new = result_new.fun
        if c5_new < c5_old:
            best_result = result_new
        else:
            best_result = result
    else:
        best_result = result

    # Extract the best solution
    best_h = best_result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)

    return (best_h, best_result.fun, n_points)
