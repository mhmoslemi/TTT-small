import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial guess with n_points = 45 (from initial_h_values)
    n_points = 45
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        h = np.pad(h, (0, n_points), 'constant')  # Pad to full length
        result = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(result)
    
    # Objective function for optimization
    def objective(h):
        return compute_c5(h)
    
    # Function to check if h is valid
    def is_valid(h):
        return np.all(0 <= h) and np.all(h <= 1) and np.isclose(np.sum(h) * dx, 1.0)
    
    # Optimization using gradient descent with momentum
    def optimize_h(h, max_iter=1000, learning_rate=0.01, momentum=0.9, eps=1e-6):
        h = np.array(h, dtype=np.float64)
        best_h = h.copy()
        best_c5 = compute_c5(h)
        velocity = np.zeros_like(h)
        for i in range(max_iter):
            # Ensure h is within [0, 1]
            h = np.clip(h, 0, 1)
            if not is_valid(h):
                break
            
            # Compute the overlap integral
            c5 = compute_c5(h)
            
            # Update velocity and h
            grad = np.zeros_like(h)
            for k in range(1, n_points + 1):
                for i in range(n_points):
                    if i + k < n_points:
                        grad[i] += (1 - h[i + k]) * dx
                    grad[i] -= h[i] * dx
            
            velocity = momentum * velocity - learning_rate * grad
            h += velocity
            h = np.clip(h, 0, 1)
            
            # Update best solution
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
            if abs(best_c5 - compute_c5(best_h)) < eps:
                break
        return best_h, best_c5
    
    # Run optimization
    h_opt, c5_opt = optimize_h(h_initial)
    
    # Try a few different starting points to escape local optima
    for _ in range(3):
        perturbed_h = h_opt + np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (np.sum(perturbed_h) / np.sum(h_opt))
        h_opt_new, c5_opt_new = optimize_h(perturbed_h)
        if c5_opt_new < c5_opt:
            h_opt = h_opt_new
            c5_opt = c5_opt_new
    
    # Try a smaller number of points for faster evaluation
    n_points = 30
    dx = 2.0 / n_points
    h_small = h_opt * (n_points / 45)
    h_small = np.clip(h_small, 0, 1)
    h_small = h_small * (np.sum(h_small) / np.sum(h_small))
    h_small = np.random.permutation(h_small)
    
    # Final evaluation
    final_c5 = compute_c5(h_small)
    
    # If time allows, run a more sophisticated solver
    if time.time() - start_time < budget_s - 5:
        # Use CVXPY to construct a more optimal solution
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
        
        # Objective is to minimize the maximum overlap
        overlap = cp.Maximize(cp.Max(cp.sum(cp.multiply(h, 1 - h[k:]))) for k in range(n_points))
        
        prob = cp.Problem(overlap, constraints)
        prob.solve(solver=cp.SCS, eps=1e-4)
        
        if prob.status == 'OPTIMAL':
            h_opt = h.value
            final_c5 = compute_c5(h_opt)
    
    # Return the best solution found
    return h_small, final_c5, n_points
