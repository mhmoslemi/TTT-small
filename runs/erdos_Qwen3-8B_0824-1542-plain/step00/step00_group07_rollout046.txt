import numpy as np

import scipy.signal as signal
import random
import time
import cvxpy as cp

# Pre-imported functions (simulated)
initial_h_values = np.random.rand(80)  # Placeholder for initial samples

def evaluate_erdos_solution(h_values):

    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start from initial_h_values
    h_initial = initial_h_values.copy()
    h_initial = h_initial / np.sum(h_initial) * (len(h_initial) / 2)
    h = h_initial.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    # Objective function for optimization
    def objective(x):
        h = x
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Constraints: sum(h) == n_points / 2
    def constraints(x):
        return np.sum(x) - n_points / 2

    # Optimization loop
    best_c5 = float('inf')
    best_h = h.copy()
    best_n = n_points

    # Initial evaluation
    current_c5 = evaluate_erdos_solution(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
        best_n = n_points

    # Try reducing n_points
    for n in range(40, n_points, -1):
        dx_new = 2.0 / n
        h_new = np.zeros(n)
        # Distribute mass to cover the interval [0, 2]
        h_new[0] = 1  # Start at the beginning
        for i in range(1, n):
            h_new[i] = np.random.uniform(0, 1)
        h_new = h_new / np.sum(h_new) * (n / 2)
        
        # Evaluate
        current_c5 = evaluate_erdos_solution(h_new)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            best_n = n

    # Try increasing n_points
    for n in range(n_points, 160):

        h_new[0] = 1  # Start at the beginning

            h_new[i] = np.random.uniform(0, 1)
        h_new = h_new / np.sum(h_new) * (n / 2)
        
        # Evaluate
        current_c5 = evaluate_erdos_solution(h_new)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            best_n = n

    # Use optimization to improve the current best
    x = cp.Variable(best_n)
    obj = cp.Minimize(objective(x))
    constraints = [cp.sum(x) == best_n / 2]
    prob = cp.Problem(obj, constraints)
    try:
        result = prob.solve(solver=cp.SCS, max_iter=500, time_limit=budget_s - (time.time() - start_time))
        if result is not None:
            best_h = x.value
            current_c5 = evaluate_erdos_solution(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
    except:
        pass

    for n in range(20, best_n, -1):

        h_new = h_new / np.sum(h_new) * (n / 2)
        
        # Evaluate
        current_c5 = evaluate_erdos_solution(h_new)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            best_n = n

    # Final check with random initialization
    h_rand = np.random.rand(best_n)
    h_rand = h_rand / np.sum(h_rand) * (best_n / 2)
    current_c5 = evaluate_erdos_solution(h_rand)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_rand.copy()
        best_n = best_n

    # Return the best found solution
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print("Time exceeded, returning best found")
    return (best_h, best_c5, best_n)
