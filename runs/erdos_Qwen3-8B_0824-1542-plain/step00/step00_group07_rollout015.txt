```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(signal.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.random.rand(80) * 2 - 1  # Dummy initial values for demonstration

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for a balance between speed and accuracy

    # Initial guess from the pre-imported initial construction
    initial_h = initial_h_values[:n_points].copy()
    # Normalize to satisfy the integral constraint
    initial_h = initial_h / np.mean(initial_h) * (n_points / 2)
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    # Try increasing the number of points up to a maximum of 500
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 1:
            break

        # Define the optimization variables
        h = cp.Variable(n_points, nonneg=True)
        # Objective: minimize the maximum overlap integral
        # We use a convex relaxation by minimizing the maximum of the integrals
        # Convert to a convex optimization problem by minimizing the maximum via a linear constraint
        # We use a bilevel approach to minimize the maximum overlap integral

        # Compute the correlation (overlap integral)
        correlation = signal.correlate(h.value, 1 - h.value, mode="full") * (2.0 / n_points) if h.value is not None else np.zeros(2 * n_points - 1)
        max_correlation = np.max(correlation)

        # If initial guess is already better, skip optimization
        if max_correlation < best_c5:
            best_c5 = max_correlation
            best_h = h.value.copy()
            continue

        # Define the optimization problem
        obj = cp.Minimize(cp.Max(correlation))
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]

        # Solve the optimization problem
        try:
            problem = cp.Problem(obj, constraints)
            problem.solve(solver=cp.SCS, max_iter=1000, eps=1e-6)
            if problem.status == 'optimal':
                current_h = h.value.copy()
                current_c5 = evaluate_erdos_solution(current_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
        except:
            pass

    return best_h, best_c5, n_points
```
