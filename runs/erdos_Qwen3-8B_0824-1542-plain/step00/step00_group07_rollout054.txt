```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial number of points
    n_points = 80  # Starting with the initial construction
    dx = 2.0 / n_points
    
    # Start with the initial h_values
    h_values = initial_h_values.copy()
    
    # Evaluate the initial solution
    c5_bound = evaluate_erdos_solution(h_values, n_points, dx)
    
    # Try to improve the solution by increasing n_points if time permits
    max_points = 200  # Maximum points to consider
    
    for n_points in range(80, max_points + 1):
        dx = 2.0 / n_points
        h_values = np.random.rand(n_points) * 2 - 1  # Random initialization for exploration
        h_values = np.clip(h_values, 0, 1)
        h_values = h_values * (n_points / 2) / np.sum(h_values)  # Normalize to meet the integral constraint
        
        # Define the overlap integral
        def overlap_integral(h):
            h_shift = np.roll(h, 1)
            overlap = np.sum(h * (1 - h_shift)) * dx
            return overlap
        
        # Objective: minimize the maximum overlap integral
        def objective(h):
            return np.max([overlap_integral(h), overlap_integral(np.roll(h, 2)), overlap_integral(np.roll(h, 3))])
        
        # Use COBYLA for optimization
        result = optimize.minimize(
            objective,
            h_values,
            method='COBYLA',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100, 'disp': False}
        )
        
        # Update h_values
        h_values = result.x
        
        # Evaluate the solution
        c5_bound = evaluate_erdos_solution(h_values, n_points, dx)
        
        # Check if we are within the time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Return the best solution found
    return (h_values, c5_bound, n_points)
```
