import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

initial_h_values = np.load('initial_h_values.npy')  # Assuming this is provided

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 500  # Start with 500 points for reasonable performance
    dx = 2.0 / n_points
    max_iter = 100  # Max number of iterations to avoid infinite loops
    tolerance = 1e-5  # Tolerance for convergence
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    # Initial guess
    h = np.copy(initial_h_values[:n_points])
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    for _ in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Compute the overlap integral
        h_reshaped = np.resize(h, (n_points, 1))
        shifted_h = np.zeros((n_points, n_points))
        for i in range(n_points):
            shifted_h[i, :] = h_reshaped[:(n_points - i), 0]
        one_minus_h = 1 - h
        overlap = np.zeros(n_points)
        for i in range(n_points):
            overlap[i] = np.sum(h_reshaped[i, 0] * one_minus_h[i:])
        c5 = np.max(overlap) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Try perturbing the h values to improve the solution
        perturbation = np.random.normal(0, 0.05, n_points)
        h += perturbation
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Return the best solution found
    return (best_h, best_c5, n_points)
