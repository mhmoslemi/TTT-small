```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode='full')) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Assuming initial_h.npy is preloaded

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial number of points
    n_points = 100  # Start with 100 points, can increase if needed
    
    # Initial guess (using provided initial_h_values or random if not available)
    if initial_h_values is None or len(initial_h_values) != n_points:
        h = np.random.rand(n_points) * 2 - 1  # Random values in [-1, 1], then scale to [0, 1]
        h = (h + 1) / 2
    else:
        h = initial_h_values.copy()
    
    # Normalize to satisfy the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization loop
    for _ in range(100):  # Max 100 iterations
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.5:
            break
        
        # Convert to CVXPY variables
        h_var = cp.Variable(n_points, lower=0, upper=1)
        constraints = [cp.sum(h_var) == n_points / 2]
        
        # Objective: minimize the max correlation
        # Note: CVXPY does not directly handle max over a large set, so we use a trick
        # by adding a large epsilon and minimizing the upper bound of the max
        # This is an approximation and may require iterative refinement
        
        # Compute all shifted versions of h
        h_shifted = []
        for k in range(1, n_points):
            shifted = np.roll(h_var.value, k)
            h_shifted.append(shifted)
        
        # Compute overlaps
        overlaps = [cp.sum(h_var * (1 - shifted)) for shifted in h_shifted]
        max_overlap = cp.Max(overlaps)
        
        # Objective: minimize max overlap
        problem = cp.Problem(cp.Minimize(max_overlap), constraints)
        problem.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8, time_limit=0.5)
        
        if problem.status != 'OPTIMAL':
            break
        
        h = h_var.value
        current_c5 = evaluate_erdos_solution(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Early stopping if improvement is minimal
        if abs(current_c5 - best_c5) < 1e-6:
            break
    
    # Final check for best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, n_points
```
