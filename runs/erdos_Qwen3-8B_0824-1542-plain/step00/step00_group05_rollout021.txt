```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.array([h[i:] for i in range(n_points)])
    shifted = np.concatenate([shifted, np.zeros((n_points, 1))], axis=1)
    one_minus_h = 1.0 - h
    overlap = np.dot(h, one_minus_h)
    overlap = np.max(overlap) * dx
    return overlap

def initial_h_values():
    return np.load('initial_h.npy').tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)

    # Start timing
    start_time = time.time()

    # Initial solution
    n_points = 100
    h_initial = initial_h_values()
    h_initial = h_initial[:n_points]
    h_initial = np.array(h_initial)
    if len(h_initial) != n_points:
        h_initial = np.random.rand(n_points)
        h_initial = (h_initial * 2.0 - 1.0) * 0.5 + 0.5
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        h_initial = h_initial.tolist()
    h = np.array(h_initial)

    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()

    # Optimization
    n_points = 100
    h = np.zeros(n_points)
    num_vars = n_points
    h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)

    # Optimize
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break

        # Objective function
        def objective(x):
            h = np.reshape(x, (n_points,))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            c5 = evaluate_erdos_solution(h)
            return c5

        # Optimization
        result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        h = result.x
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Final solution
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    return best_h.tolist(), best_c5, n_points
```
