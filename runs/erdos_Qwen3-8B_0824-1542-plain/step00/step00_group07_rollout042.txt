import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial solution from the given starting point
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_h = np.sum(h_initial) * dx
    # Ensure the integral is exactly 1
    if abs(sum_h - 1) > 1e-6:
        h_initial = initial_h_values.copy()
        # Normalize to ensure the integral is 1
        h_initial = h_initial * (1 / (np.sum(h_initial) * dx))
    
    def compute_c5(h):
        h = np.array(h)
        n = len(h)
        dx = 2.0 / n
        # Compute (1 - h) for convolution
        one_minus_h = 1 - h
        # Perform full convolution
        conv = np.correlate(h, one_minus_h, mode='full')
        # Multiply by dx to get the integral
        conv = conv * dx
        # Find the maximum value of the convolution
        return np.max(conv)
    
    def objective(x):
        # x is the h values
        h = np.array(x)
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) * (2.0 / len(h)) - 1
    
    def constraint_bounds(h):
        return np.all((h >= 0) & (h <= 1))
    
    # Try a few different numbers of points, starting with a small number
    best_c5 = float('inf')
    best_h = h_initial
    best_n_points = len(h_initial)
    
    # Try reducing the number of points to speed up the optimization
    for n_points in [100, 80, 60, 50, 40, 30]:
        if time.time() - start_time > budget_s:
            break
        # Initialize h with the initial values, scaled for the new n_points
        if n_points != len(h_initial):
            # Scale the initial h to fit the new n_points
            h_initial_scaled = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
            # Ensure the integral is 1
            h_initial_scaled = h_initial_scaled * (1 / (np.sum(h_initial_scaled) * (2.0 / n_points)))
        
        # Use a local optimization method with bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = optimize.minimize(
            objective,
            h_initial_scaled,
            bounds=bounds,
            method='L-BFGS-B',
            options={'ftol': 1e-7, 'maxiter': 100}
        )
        
        if time.time() - start_time > budget_s:
            break
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                best_n_points = n_points
    
    # Evaluate the best solution found
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n_points
