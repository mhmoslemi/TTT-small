import numpy as np
import scipy.optimize as optimize
import time
import os
import sys
import math
import random
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial setup
    n_points = 100  # Start with 100 points
    dx = 2.0 / n_points
    h = np.array(initial_h_values[:n_points])  # Use initial values and then refine
    h = h / (dx * n_points) * (n_points / 2)  # Normalize to ensure sum(h) * dx = 1

    def compute_c5(h_values):
        n = len(h_values)
        dx = 2.0 / n
        h = h_values.copy()
        # Ensure values are within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute 1 - h
        h1 = 1 - h
        # Compute cross-correlation
        corr = np.correlate(h, h1, mode="full")
        # Compute max overlap
        c5 = np.max(corr) * dx
        return c5

    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        return compute_c5(h)

    def constraint_sum(x):
        return np.sum(x) - (n_points / 2)

    def constraint_bounds(x):
        return np.all((0 <= x) & (x <= 1))

    # Initial optimization: refine starting point with local search
    def refine_solution(h_initial, max_iter=100, max_eval=1000):
        h = h_initial.copy()
        best_h = h.copy()
        best_c5 = compute_c5(h)
        for _ in range(max_iter):
            # Use local optimization to refine the solution
            x = cp.Variable(h.shape)
            obj = cp.Minimize(objective(x))
            constraints = [constraint_sum(x) == 0, constraint_bounds(x)]
            prob = cp.Problem(obj, constraints)
            try:
                prob.solve(solver=cp.SCS, max_iter=100, verbose=False)
                if prob.status == 'optimal':
                    h = x.value
                    current_c5 = compute_c5(h)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
                else:
                    break
            except:
                break
        return best_h, best_c5

    # Refine the initial solution
    refined_h, c5 = refine_solution(h)
    best_h = refined_h
    best_c5 = c5

    # Try increasing n_points for better resolution
    for n_points in [150, 200, 250, 300]:
        if time.time() - start_time > budget_s - 1:
            break
        dx = 2.0 / n_points
        # Initialize with the best solution so far and interpolate
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = best_h[int(i * len(best_h) / n_points)]
        h = h / (dx * n_points) * (n_points / 2)
        h = np.clip(h, 0, 1)

        # Use local optimization to refine
        x = cp.Variable(h.shape)
        obj = cp.Minimize(objective(x))
        constraints = [constraint_sum(x) == 0, constraint_bounds(x)]
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, max_iter=100, verbose=False)
            if prob.status == 'optimal':
                h = x.value
                current_c5 = compute_c5(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        except:
            pass

    # Final check with a small perturbation
    h_perturb = best_h + 0.01 * np.random.randn(len(best_h))
    h_perturb = np.clip(h_perturb, 0, 1)
    h_perturb = h_perturb / (dx * n_points) * (n_points / 2)
    current_c5 = compute_c5(h_perturb)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_perturb.copy()

    # Final evaluation
    final_c5 = compute_c5(best_h)
    print(f"Final C5 bound: {final_c5}")
    
    # Check time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        print("Time budget exceeded, returning best found solution")
    else:
        print("Time budget not exceeded, continuing optimization")

    # Final output
    return (best_h.tolist(), final_c5, len(best_h))
