```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    n_points = 100  # Start with 100 points, adjust if needed
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()[:n_points]
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2)  # Ensure the integral is 1
    initial_h = initial_h / (n_points / 2)  # Normalize to [0, 1]

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    def objective(x):
        nonlocal best_c5
        h = x.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2)
        h = h / (n_points / 2)
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    # Set up optimization problem
    x = cp.Variable(n_points)
    constraints = [cp.sum(x) == n_points / 2, 0 <= x, x <= 1]
    objective_func = cp.Max(cp.multiply(cp.sum(cp.multiply(x, 1 - x)), 1))  # Placeholder, needs refinement

    # Use Sequential Least Squares Programming (SLSQP) for optimization
    start_time = time.time()
    result = optimize.minimize(objective, initial_h, method='SLSQP',
                               bounds=[(0, 1) for _ in range(n_points)],
                               constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2},
                               options={'ftol': 1e-9, 'maxiter': 100})
    
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return best_h, best_c5, n_points

    h_opt = result.x
    best_h = h_opt
    best_c5 = evaluate_erdos_solution(best_h)

    return best_h, best_c5, n_points
```
