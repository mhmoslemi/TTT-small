import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial H values
    h_values = initial_h_values()
    n_points = len(h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(h_values)
    if abs(sum_h - n_points / 2) > 1e-6:
        h_values = (h_values * (n_points / 2)) / sum_h
    
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        h_extended = np.concatenate([h, np.zeros_like(h)])
        corr = np.correlate(h, 1 - h, mode="full")
        overlap = np.max(corr * dx)
        return overlap

    def objective_func(h):
        return compute_c5(h)

    def constraint_func(h):
        return np.sum(h) - n_points / 2

    def optimize_h(h_initial):
        # Setup optimization problem
        h = cp.Variable(n_points, nonneg=True)
        constraints = [cp.sum(h) == n_points / 2]
        objective = cp.Minimize(objective_func(h.value if h_initial is None else h_initial))
        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve(solver=cp.SCS, max_iter=1000, eps=1e-4, verbose=False)
            if result is not None:
                return h.value
        except:
            pass
        return h_initial

    # Optimization loop
    best_c5 = np.inf
    best_h = h_values.copy()
    for _ in range(3):
        h_values = optimize_h(best_h)
        c5 = compute_c5(h_values)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_values
        # Early stopping if time is up
        if time.time() - start_time > budget_s - 1:
            break

    # Final evaluation
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
