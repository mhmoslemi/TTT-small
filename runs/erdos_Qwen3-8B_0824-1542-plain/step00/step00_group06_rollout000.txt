def run(seed=42, budget_s=60, **kwargs):

    # Define the objective function to minimize C5
    def objective_function(h):
        h = np.array(h)
        # Ensure h is in [0,1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        h = h.reshape(-1, 1)
        ones_minus_h = 1 - h
        # Compute cross-correlation (full mode)
        corr = np.correlate(h.flatten(), ones_minus_h.flatten(), mode="full")
        dx = 2.0 / len(h)
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization: Use basin-hopping to escape local minima
    def optimize_h():
        # Define the optimization problem with constraints
        # Use convex optimization with constraints
        n = len(h_values)
        # Create a variable
        h = cp.Variable(n)
        # Objective: minimize the maximum overlap integral
        # To handle the max, we use a reformulation with auxiliary variables
        # This is a nonlinear optimization problem and may be computationally intensive
        # So we use a heuristic approach with simulated annealing and random restarts
        
        # We use a simulated annealing approach as a heuristic
        # Run simulated annealing for a limited number of steps
        def simulate_annealing(h0):
            h = np.copy(h0)
            best_h = np.copy(h)
            best_c5 = objective_function(h)
            T = 1.0
            steps = 1000
            for step in range(steps):
                # Perturb h
                perturbation = np.random.normal(0, 0.05, size=n)
                h_perturbed = np.clip(h + perturbation, 0, 1)
                # Ensure the integral constraint is satisfied
                h_perturbed = h_perturbed * (n / 2) / np.sum(h_perturbed)
                c5_perturbed = objective_function(h_perturbed)
                # Acceptance probability
                if np.random.rand() < np.exp((best_c5 - c5_perturbed) / T):
                    h = h_perturbed
                    best_h = h_perturbed
                    best_c5 = c5_perturbed
                T *= 0.995
            return best_h, best_c5
        
        # Run simulated annealing with random restarts
        best_h = np.copy(h_values)
        best_c5 = objective_function(h_values)
        for _ in range(10):

            h_restart, c5_restart = simulate_annealing(h_restart)
            if c5_restart < best_c5:
                best_h = h_restart
                best_c5 = c5_restart
        return best_h, best_c5
