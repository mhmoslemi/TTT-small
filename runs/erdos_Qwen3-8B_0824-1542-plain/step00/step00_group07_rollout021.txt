import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)

    # Initial guess from provided initial_h_values
    initial_h_values = np.array(initial_h_values)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    if sum_h != n_points / 2:
        initial_h_values = initial_h_values * (n_points / 2) / sum_h

    # Function to compute the overlap integral C5
    def compute_c5(h):
        h = np.array(h)
        h = np.concatenate([h, np.zeros_like(h)])  # Pad for full convolution
        result = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(result)

    # Function to compute the overlap C5
    def objective(h):
        return compute_c5(h)

    # Bounds on h: [0, 1]
    bounds = Bounds(0, 1)

    # Optimizer settings
    max_iter = 1000
    tol = 1e-6

    # Start time tracking
    start_time = time.time()

    # Initial guess
    h_initial = initial_h_values.copy()
    h_initial = np.clip(h_initial, 0, 1)  # Ensure values are within bounds

    # Optimization loop with early stopping
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    best_n_points = n_points

    # Try optimization for a limited time
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        try:
            # Optimization with L-BFGS-B
            result = opt.minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, tol=tol, maxiter=max_iter)
            if result.success:
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                c5_candidate = compute_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                    best_n_points = len(h_candidate)
                    h_initial = best_h.copy()
            else:
                break
        except:
            break

    # Try a few random perturbations
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturbation = np.random.uniform(-0.1, 0.1, size=len(best_h))
        h_candidate = best_h + perturbation
        h_candidate = np.clip(h_candidate, 0, 1)
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
            best_n_points = len(h_candidate)

    return best_h, best_c5, best_n_points
