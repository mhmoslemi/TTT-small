import numpy as np

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to evaluate C5
    def evaluate_c5(h):
        h = np.array(h)
        # Ensure h is within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute (1 - h) as the complementary function
        one_minus_h = 1 - h
        # Compute the full convolution
        corr = np.correlate(h, one_minus_h, mode='full')
        # Only consider the relevant part of the convolution
        # The overlap integral is max_k ∫ h(x)(1 - h(x+k)) dx
        # The relevant values are from the convolution between -1 to +1
        # We need to adjust indices to get the correct overlap integrals
        # For each shift k, the overlap integral is the convolution at position k
        # But we only need to consider k from 0 to n_points-1
        # The result of convolution has length 2n_points - 1
        # The relevant part is from index n_points-1 to 2n_points-2
        relevant_corr = corr[n_points-1:2*n_points-1]
        # Multiply by dx to get the integral
        max_overlap = np.max(relevant_corr * dx)
        return max_overlap
    
    # Function to optimize the h values
    def optimize_h(h_initial):
        h = h_initial.copy()
        # Perform optimization
        def objective(x):
            return evaluate_c5(x)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2
        
        # Bounds for h
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use Sequential Least Squares Programming (SLSQP)
        result = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        # Return the optimized h values
        return result.x
    
    # Run the optimization
    start_time = time.time()
    optimized_h = optimize_h(initial_h)
    elapsed_time = time.time() - start_time
    
    # Evaluate the result
    c5_bound = evaluate_c5(optimized_h)
    n_points = len(optimized_h)
    
    # If we have time left, try a few more iterations with slight perturbations
    if elapsed_time < budget_s - 10:
        # Perturb the solution
        perturbations = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = np.clip(optimized_h + perturbations, 0, 1)
        # Ensure the sum constraint is satisfied
        sum_perturbed = np.sum(perturbed_h)
        scale_factor = (n_points / 2) / sum_perturbed if sum_perturbed != 0 else 1
        perturbed_h *= scale_factor
        # Evaluate the perturbed solution
        c5_perturbed = evaluate_c5(perturbed_h)
        if c5_perturbed < c5_bound:
            optimized_h = perturbed_h
            c5_bound = c5_perturbed
    
    return optimized_h, c5_bound, n_points
