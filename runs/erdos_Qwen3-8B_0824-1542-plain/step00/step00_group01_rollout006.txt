import numpy as np
import scipy.optimize as opt

# evaluate_erdos_solution is not used directly, but the structure of the problem is known
# initial_h_values is a starting point for h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 100  # Start with 100 points, can be increased if needed
    dx = 2.0 / n_points
    max_iterations = 100  # Max number of iterations for optimization
    tolerance = 1e-4  # Tolerance for stopping criterion
    
    # Initial guess from provided values
    h_initial = np.array(initial_h_values)
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    h_initial = np.clip(h_initial, 0, 1)
    
    best_c5 = float('inf')
    best_h = h_initial.copy()
    
    start_time = time.time()
    
    # Optimization loop
    for iteration in range(max_iterations):
        # Convert h to a variable for optimization
        h = cp.Variable(n_points, bounds=(0, 1))
        constraint = cp.sum(h) == n_points / 2
        
        # Compute all shifts (k) we need to check
        shifts = np.arange(-n_points + 1, n_points)
        k_values = np.unique(shifts)
        n_shifts = len(k_values)
        
        # Objective: minimize the maximum overlap integral across all shifts
        # We use a convex relaxation by minimizing the sum of max values
        objective = cp.Minimize(cp.sum(cp.maximum(0, cp.sum(h * (1 - h[k_values % n_points])) for k in k_values)) / n_shifts)
        
        # Solve the problem
        problem = cp.Problem(objective, [constraint])
        try:
            result = problem.solve(solver='SCIPY', eps=1e-5, verbose=False)
            if result is not None:
                h_values = h.value
                # Evaluate the overlap integral for each shift
                overlap_values = []
                for k in k_values:
                    shifted_idx = (np.arange(n_points) + k) % n_points
                    overlap = np.sum(h_values * (1 - h_values[shifted_idx])) * dx
                    overlap_values.append(overlap)
                c5 = np.max(overlap_values)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_values.copy()
                if time.time() - start_time > budget_s - 1:
                    break
        except:
            pass
        
        # Early stopping if no improvement
        if iteration > 0 and np.abs(best_c5 - np.inf) < tolerance:
            break
    
    # Final check with the best solution
    h_values = best_h
    # Evaluate the overlap integral for each shift
    shifts = np.arange(-n_points + 1, n_points)
    k_values = np.unique(shifts)
    overlap_values = []
    for k in k_values:
        shifted_idx = (np.arange(n_points) + k) % n_points
        overlap = np.sum(h_values * (1 - h_values[shifted_idx])) * dx
        overlap_values.append(overlap)
    c5_final = np.max(overlap_values)
    
    return (h_values, c5_final, n_points)
