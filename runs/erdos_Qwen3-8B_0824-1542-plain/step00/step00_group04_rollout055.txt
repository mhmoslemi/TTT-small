import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint, minimize
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import spsolve
import sys

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(42)  # Placeholder for initial construction

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Initial guess
    n_points = 1000  # Start with 1000 samples
    h_initial = np.random.rand(n_points)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)  # Ensure integral is 1
    
    # Objective function: minimize max overlap
    def objective(x):
        h = x.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    # Constraints: sum(h) == n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Bounds: 0 <= h[i] <= 1
    bounds = Bounds(0, 1)
    
    # Linear constraint: sum(h) == n_points / 2
    cons = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    
    # Optimization
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    
    # Time check
    elapsed = time.time() - start_time
    if elapsed >= budget_s - 1:
        return (result.x, result.fun, n_points)
    
    # Try larger n_points if time allows
    if elapsed < budget_s - 5:
        n_points = 2000
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        def objective_large(x):
            h = x.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * (2.0 / n_points)
            return c5
        
        cons_large = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        result_large = minimize(objective_large, h_initial, method='SLSQP', bounds=bounds, constraints=cons_large, tol=1e-6)
        
        if result_large.fun < result.fun:
            result = result_large
    
    # Final result
    return (result.x, result.fun, n_points)
