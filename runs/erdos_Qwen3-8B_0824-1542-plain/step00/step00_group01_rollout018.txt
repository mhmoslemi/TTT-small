import numpy as np
import scipy.optimize as opt
import time
import copy
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timer
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # starting with 100 points
    max_iterations = 100  # limit iterations to fit within budget
    n_points_limit = 400  # upper limit on number of points to keep optimization feasible
    
    # Initial guess from provided initial_h_values
    h = initial_h_values.copy()
    n_points_initial = len(h)
    if n_points_initial > n_points_limit:
        n_points_initial = n_points_limit
    
    # Function to compute C5
    def compute_c5(h_values):
        h = np.array(h_values)
        n = len(h)
        dx = 2.0 / n
        # Compute 1 - h
        one_minus_h = 1.0 - h
        # Compute the full correlation
        corr = np.correlate(h, one_minus_h, mode="full")
        # Find the maximum value
        max_corr = np.max(corr)
        # Multiply by dx
        c5 = max_corr * dx
        return c5
    
    # Function to optimize h
    def optimize_h(h_initial, n_points):
        h = h_initial.copy()
        n = len(h)
        if n < n_points:
            # Interpolate to n_points
            x = np.linspace(0, 2, n)
            h_new = np.interp(np.linspace(0, 2, n_points), x, h)
        elif n > n_points:
            # Resample to n_points
            h_new = np.zeros(n_points)
            for i in range(n_points):
                idx = int(i * n / n_points)
                h_new[i] = h[idx]
        else:
            h_new = h.copy()
        
        # Ensure the integral condition is satisfied
        sum_h = np.sum(h_new)
        dx = 2.0 / n_points
        target_sum = n_points / 2
        if abs(sum_h - target_sum) > 1e-6:
            # Adjust h to meet the integral condition
            h_new = h_new / sum_h * target_sum
        
        # Prepare variables for optimization
        h_var = cp.Variable(n_points)
        constraints = [0 <= h_var, h_var <= 1]
        # Objective function: minimize the maximum overlap (C5)
        # We can't directly minimize the max, so we use a convex relaxation
        # by minimizing the maximum overlap over all k
        # We use a convex optimization approach with a scalar variable to represent the maximum
        max_overlap = cp.Variable()
        constraints.append(max_overlap >= 0)
        for k in range(n_points):
            overlap = cp.sum(h_var * (1 - h_var[k:])) / (2.0 / n_points)
            constraints.append(max_overlap >= overlap)
        
        # Objective: minimize the maximum overlap
        prob = cp.Problem(cp.Minimize(max_overlap), constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-4)
            if prob.status == "optimal":
                h_new = h_var.value
        except:
            pass
        
        return h_new
    
    # Optimizer function that handles the trade-off between points and performance
    def optimize_h_with_points(h_initial, max_points=400):
        best_h = h_initial.copy()
        best_c5 = compute_c5(best_h)
        best_points = len(best_h)
        
        # Try different numbers of points
        for n in range(len(h_initial), max_points + 1):
            h_opt = optimize_h(h_initial, n)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5
                best_points = n
        
        return best_h, best_c5, best_points
    
    # Run the optimizer
    best_h, best_c5, best_points = optimize_h_with_points(initial_h_values, max_points=n_points_limit)
    
    # Check if we can do more iterations with a smaller number of points
    for _ in range(max_iterations):
        # Try reducing the number of points
        n_points_red = max(int(best_points * 0.8), 50)
        h_opt = optimize_h(best_h, n_points_red)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5
            best_points = n_points_red
        # Try increasing the number of points
        n_points_inc = min(int(best_points * 1.2), n_points_limit)
        h_opt = optimize_h(best_h, n_points_inc)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5
            best_points = n_points_inc
        
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    current_time = time.time()
    elapsed_time = current_time - start_time
    remaining_time = budget_s - elapsed_time
    
    # Try more iterations if time permits
    for _ in range(int(remaining_time / 1.0)):
        # Try reducing the number of points
        n_points_red = max(int(best_points * 0.8), 50)
        h_opt = optimize_h(best_h, n_points_red)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5
            best_points = n_points_red
        # Try increasing the number of points
        n_points_inc = min(int(best_points * 1.2), n_points_limit)
        h_opt = optimize_h(best_h, n_points_inc)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5
            best_points = n_points_inc
    
    # Return the final result
    return best_h, best_c5, best_points
