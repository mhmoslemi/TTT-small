import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    n_points = 100  # Start with 100 samples for faster evaluation
    dx = 2.0 / n_points
    h = initial_h_values[:n_points].copy()  # Start with initial construction
    h = h / np.sum(h) * (n_points / 2)  # Ensure sum(h) == n_points / 2

    # Function to compute the maximum overlap integral
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        h_shifted = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            h_shifted[k:] = h[:-k]
        overlap = np.correlate(h, 1 - h_shifted, mode="full") * dx
        return np.max(overlap)

    # Function to optimize h
    def optimize_h(h_initial):
        h = cp.Variable(n_points)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        objective = cp.Minimize(cp.Max(cp.correlate(h.value, 1 - h.value, mode="full") * dx))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.SCS, max_iter=100, eps=1e-5, verbose=False)
        return h.value

    # Optimization loop
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()

    # Try multiple starting points
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        # Random perturbation
        h = best_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

        # Local optimization
        h_opt = optimize_h(h)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Final check with full optimization
    h_opt = optimize_h(best_h)
    c5_opt = compute_c5(h_opt)
    if c5_opt < best_c5:
        best_h = h_opt
        best_c5 = c5_opt

    # Final check with larger n_points
    n_points = 200
    dx = 2.0 / n_points
    h = best_h * (n_points / 2) / (n_points / 2)  # Normalize to new n_points
    h = h[:n_points]
    h = h / np.sum(h) * (n_points / 2)
    h_opt = optimize_h(h)
    c5_opt = compute_c5(h_opt)
    if c5_opt < best_c5:
        best_h = h_opt
        best_c5 = c5_opt

    return best_h, best_c5, n_points
