```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    n_points_initial = len(initial_h_values)
    h_initial = initial_h_values.copy()
    
    # Determine the number of points to use
    n_points = n_points_initial  # Start with the initial number of points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        # Compute the convolution
        corr = np.correlate(h, 1 - h, mode="full")
        # Take the maximum value and multiply by dx
        c5 = np.max(corr) * dx
        return c5
    
    # Function to optimize h to minimize C5
    def objective(h):
        return compute_c5(h)
    
    # Function to check the constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization settings
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    h_initial = np.array(h_initial)
    
    # Run the optimization
    result = minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    # Check if we have enough time to try a larger n_points
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 5:
        # Try with more points
        n_points = int(n_points_initial * 1.5)
        h_initial = np.random.rand(n_points) * 0.5  # Random initial guess for larger n_points
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
    
    # Evaluate the best solution found
    best_h = result.x
    c5_bound = compute_c5(best_h)
    
    # Return the best solution found
    return (best_h.tolist(), c5_bound, len(best_h))
```
