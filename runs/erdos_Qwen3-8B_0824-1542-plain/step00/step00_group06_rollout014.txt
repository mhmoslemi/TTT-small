```python
import numpy as np
import scipy.optimize as optimize
import time
import sys
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

# Global constants
MAX_ITERATIONS = 100
MAX_TIME = 60  # seconds
n_points = 95  # initial n_points
dx = 2.0 / n_points

def compute_c5(h_values):
    h = np.array(h_values)
    h_shifted = np.concatenate((h[-1:], h[:-1]))
    overlap = np.correlate(h, 1 - h_shifted, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with initial h_values
    h_values = initial_h_values.copy()
    best_h = h_values.copy()
    best_c5 = compute_c5(h_values)
    best_n_points = len(h_values)
    
    # Try increasing n_points slightly to see if it helps
    for n_points in range(95, 150):
        dx = 2.0 / n_points
        h_values = np.random.rand(n_points)
        h_values = h_values * (n_points / 2) / np.sum(h_values)  # normalize to have integral 1
        
        start_time = time.time()
        for _ in range(MAX_ITERATIONS):
            # Compute the current C5
            current_c5 = compute_c5(h_values)
            
            # If we're over time, break
            if time.time() - start_time > budget_s:
                break
            
            # Use gradient descent to minimize C5
            def objective(x):
                h = x.copy()
                h_shifted = np.concatenate((h[-1:], h[:-1]))
                overlap = np.correlate(h, 1 - h_shifted, mode="full")
                c5 = np.max(overlap) * dx
                return c5
            
            # Set up optimization
            x = cp.Variable(n_points)
            constraints = [
                cp.sum(x) == n_points / 2,
                0 <= x,
                x <= 1
            ]
            prob = cp.Problem(cp.Minimize(objective(x)), constraints)
            
            # Solve the optimization
            try:
                result = prob.solve(solver=cp.SCIP, verbose=False)
                if result is not None:
                    h_values = x.value
            except:
                pass
            
        # Update best solution
        current_c5 = compute_c5(h_values)
        if current_c5 < best_c5:
            best_h = h_values.copy()
            best_c5 = current_c5
            best_n_points = n_points
    
    return best_h, best_c5, best_n_points
```
