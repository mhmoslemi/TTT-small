import numpy as np

# Pre-imported functions

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try different numbers of points to find the best balance between accuracy and speed
    n_points_list = [100, 200, 300, 400]
    
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Start with initial guess or random guess
        if n_points == 100:
            h_initial = initial_h_values.copy()
            h_initial = np.interp(np.linspace(0, 2, n_points), x, h_initial)
            h_initial = np.clip(h_initial, 0, 1)
        else:
            h_initial = np.random.uniform(0, 1, n_points)
        
        # Ensure the integral is exactly 1
        h_initial = h_initial / (np.sum(h_initial) * dx) * dx * n_points / 2
        
        # Optimization using gradient descent with momentum
        h = h_initial.copy()
        learning_rate = 0.01
        momentum = 0.9
        decay = 0.99
        max_iter = 2000
        tol = 1e-4
        start_time = time.time()
        
        for iter in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Compute the current overlap integral
            h_correlated = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(h_correlated) * dx
            
            # If we already have a better solution, stop early
            if c5 >= best_c5:
                break
            
            # Compute the derivative of the overlap integral with respect to h
            # This is a simplified heuristic based on the maximum overlap
            overlap = h_correlated
            max_overlap_idx = np.argmax(overlap)
            shift = max_overlap_idx - n_points + 1
            derivative = np.zeros(n_points)
            
            # Compute the derivative at the max overlap
            for i in range(n_points):
                # Compute the derivative of the overlap at position i
                # This is a numerical approximation
                h_plus = h.copy()
                h_plus[i] += 1e-6
                h_minus = h.copy()
                h_minus[i] -= 1e-6
                overlap_plus = np.correlate(h_plus, 1 - h_plus, mode="full")[max_overlap_idx]
                overlap_minus = np.correlate(h_minus, 1 - h_minus, mode="full")[max_overlap_idx]
                derivative[i] = (overlap_plus - overlap_minus) / (2 * 1e-6)
            
            # Update using gradient descent with momentum
            gradient = -derivative * learning_rate
            h = h + momentum * gradient - (momentum * gradient - gradient) * decay

            h = np.clip(h, 0, 1)
            
            # Ensure the integral is exactly 1
            h = h / (np.sum(h) * dx) * dx * n_points / 2
            
            # Track the best solution

                best_c5 = c5
                best_h = h.copy()
                best_n_points = n_points
        
        # Evaluate the best solution so far
        if best_c5 < best_c5:
            best_c5 = evaluate_erdos_solution(best_h)
