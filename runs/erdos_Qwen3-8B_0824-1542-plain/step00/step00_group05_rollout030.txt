import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions and initial solution
initial_h_values = np.array([...])  # Placeholder for the actual initial solution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    max_points = 1000
    optimal_n_points = 100  # Try a smaller number for faster optimization
    
    # Initial guess
    h = initial_h_values[:optimal_n_points].copy()
    h /= np.sum(h)  # Normalize to ensure sum * dx = 1
    h *= optimal_n_points / 2  # Ensure sum * dx = 1
    
    # Optimization function to minimize the maximum overlap
    def objective(x):
        h_vals = x.reshape(-1)
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * optimal_n_points / 2
        dx = 2.0 / optimal_n_points
        shifted = np.zeros_like(h_vals)
        for k in range(1, optimal_n_points):
            shifted[k:] = h_vals[:-k]
        overlap = np.maximum(np.convolve(h_vals, 1 - h_vals, mode='full') * dx)
        return np.max(overlap)
    
    # Optimization using differential evolution
    bounds = [(0, 1) for _ in range(optimal_n_points)]
    result = optimize.differential_evolution(objective, bounds, 
                                             strategy='best1bin', 
                                             popsize=15, 
                                             maxiter=100, 
                                             tol=1e-6,
                                             seed=seed,
                                             disp=False)
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * optimal_n_points / 2
    dx = 2.0 / optimal_n_points
    shifted = np.zeros_like(best_h)
    for k in range(1, optimal_n_points):
        shifted[k:] = best_h[:-k]
    c5 = np.max(np.convolve(best_h, 1 - best_h, mode='full') * dx)
    
    return (best_h.tolist(), c5, optimal_n_points)
