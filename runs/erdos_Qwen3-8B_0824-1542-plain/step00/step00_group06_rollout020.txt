import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Predefined initial solution
    initial_n_points = len(initial_h_values)
    initial_h_values = np.array(initial_h_values)
    
    # Try different n_points starting from initial_n_points
    n_points_candidates = [initial_n_points, initial_n_points // 2 * 2, initial_n_points // 2 * 3, initial_n_points // 2 * 4]
    n_points_candidates = list(set(n_points_candidates))
    n_points_candidates.sort()
    
    best_h = initial_h_values
    best_c5 = evaluate_erdos_solution(initial_h_values)
    best_n_points = initial_n_points
    
    start_time = time.time()
    
    for n_points in n_points_candidates:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Define the grid
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Objective: minimize max_k [integral(h(x) * (1 - h(x + k))) dx]
        # We approximate the integral using correlation
        
        # Optimization variables
        h = cp.Variable(n_points, nonneg=True, bool=False)
        constraints = [cp.sum(h) == n_points / 2]
        
        # Objective function: minimize the maximum correlation
        # We use a convex relaxation by introducing a variable for the max value
        max_corr = cp.Variable()
        constraints.append(max_corr >= 0)
        
        # Compute the correlation for all k
        for k in range(1, n_points):
            if k >= n_points:
                break
            h_shifted = np.roll(h, k)
            h_shifted = np.clip(h_shifted, 0, 1)
            corr = correlate(h, 1 - h_shifted, mode='full')
            corr = corr[n_points - 1:]
            corr = corr * dx
            constraints.append(max_corr >= corr)
        
        # Solve the optimization problem
        try:
            prob = cp.Problem(cp.Minimize(max_corr), constraints)
            prob.solve(solver='ECOS', eps=1e-5)
            
            if prob.status == 'OPTIMAL':
                h_values = np.round(h.value, 4)
                h_values = np.clip(h_values, 0, 1)
                c5 = evaluate_erdos_solution(h_values)
                
                if c5 < best_c5:
                    best_h = h_values
                    best_c5 = c5
                    best_n_points = n_points
        except:
            pass
    
    return best_h, best_c5, best_n_points
