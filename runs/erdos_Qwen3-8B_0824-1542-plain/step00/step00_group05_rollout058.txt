import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

# Strategy:
# 1. Start with the given initial solution and try to refine it.
# 2. Use a combination of gradient-based optimization and simulated annealing to explore the space.
# 3. Use the max-correlation as the objective to minimize, which directly corresponds to C₅.
# 4. Use constraints to maintain the integral of h(x) equal to 1.
# 5. Limit the number of samples to under 1000 to ensure the computation stays within time budget.
# 6. Evaluate the solution periodically and return the best found within the time limit.

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Start with the initial sample configuration
    n_points = 100
    h_values = initial_h_values.copy()
    h_values = h_values[:n_points]  # Limit to 100 points
    
    # Normalize to satisfy the integral constraint
    h_values = h_values * (n_points / 2) / np.sum(h_values)
    
    # Function to compute C₅
    def compute_c5(h):
        dx = 2.0 / len(h)
        h_array = np.array(h)
        corr = np.correlate(h_array, 1 - h_array, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Define objective function for optimization
    def objective_function(h_flat):
        h = h_flat.reshape(-1)
        return compute_c5(h)
    
    # Define constraints for optimization
    def constraint(h_flat):
        h = h_flat.reshape(-1)
        return np.sum(h) - (n_points / 2)
    
    # Initial bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use scipy's optimization with bounds and constraints
    result = opt.minimize(
        fun=objective_function,
        x0=h_values,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        method='L-BFGS-B',
        jac=None,
        options={'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Use simulated annealing to refine the solution
    def simulated_annealing():
        temp = 100.0
        cooling_rate = 0.995
        iterations = 1000
        best_h = best_h.copy()
        best_c5 = best_c5
        
        for _ in range(iterations):
            # Generate a perturbed solution
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            new_h = np.clip(best_h + perturbation, 0.0, 1.0)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            # Evaluate the new solution
            new_c5 = compute_c5(new_h)
            
            # Accept the new solution with probability
            if np.random.rand() < np.exp((best_c5 - new_c5) / temp):
                best_h = new_h
                best_c5 = new_c5
            
            # Cool down
            temp *= cooling_rate
        
        return best_h, best_c5
    
    best_h, best_c5 = simulated_annealing()
    
    # Evaluate the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Return the best solution found
    return (best_h, final_c5, n_points)
