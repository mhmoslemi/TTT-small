def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 45  # Start with the current construction
    h = initial_h_values.copy()
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = h.astype(float)
        correlate = np.correlate(h, 1 - h, mode="full")
        return np.max(correlate) * (2.0 / len(h))
    
    # Function to optimize h using simulated annealing
    def optimize_h(h_initial, budget_s):
        start_time = time.time()
        
        def objective(x):
            x = np.clip(x, 0, 1)
            return compute_c5(x)
        
        # Use Simulated Annealing
        result = optimize.anneal(objective, h_initial, maxiter=1000, T=1.0, schedule='geometric',
                                 accept=optimize.accept_func_boltzmann, seed=seed)
        
        # Extract the best solution
        best_h = result[0]
        best_c5 = compute_c5(best_h)
        
        # Return the best solution found within the budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            return best_h, best_c5, len(best_h)
        
        # Use a more refined optimization if time allows
        if elapsed < budget_s - 1:
            # Use a smaller step size for local refinement
            result = optimize.minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)] * len(best_h),
                                       tol=1e-5, options={'maxiter': 100})
            best_h = result.x
            best_c5 = compute_c5(best_h)
        
        return best_h, best_c5, len(best_h)
    
    # Run the optimization
    best_h, best_c5, n_points = optimize_h(h, budget_s)
    
    return best_h, best_c5, n_points
