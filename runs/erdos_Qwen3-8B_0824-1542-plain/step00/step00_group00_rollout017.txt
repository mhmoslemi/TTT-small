```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Starting with the initial construction
    n_points = len(initial_h_values)
    h_values = np.copy(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(h_values) * dx
    if not np.isclose(sum_h, 1.0):
        h_values = initial_h_values * (1.0 / sum_h) * dx
    c5_bound = evaluate_erdos_solution(h_values)
    
    # Try optimizing with different numbers of points
    for n in [n_points // 2, n_points, n_points * 2, n_points * 3]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Ensure n is at least 40 to avoid numerical issues
        if n < 40:
            continue
        
        # Generate initial samples
        x = np.linspace(0, 2, n)
        h = np.random.rand(n)
        h = h * (1.0 / np.sum(h)) * (n / 2)  # Normalize to sum to n/2
        
        # Define the optimization variables
        h_var = cp.Variable(n)
        constraints = [0 <= h_var, h_var <= 1]
        constraints.append(cp.sum(h_var) == n / 2)
        
        # Define the objective function
        def objective(h_var):
            # Compute the overlap integral
            h = np.array(h_var)
            overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n)
            max_overlap = np.max(overlap)
            return max_overlap
        
        # Use a minimax formulation to minimize the maximum overlap integral
        max_overlap = cp.Maximize(cp.minimize(objective(h_var)))
        
        # Solve the optimization problem
        try:
            prob = cp.Problem(max_overlap, constraints)
            prob.solve(solver=cp.SCS, eps=1e-4, max_iter=1000)
            if prob.status == 'OPTIMAL':
                h_values = h_var.value
                c5_bound = evaluate_erdos_solution(h_values)
                n_points = n
        except:
            pass
        
    return h_values, c5_bound, n_points
```
