import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.random.rand(70) * 2 - 1  # Placeholder, actual initial values would be defined elsewhere

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial guess
    n_points = 100  # Use 100 points for faster evaluation
    h_values = np.random.rand(n_points) * 2 - 1
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values * (n_points / 2) / np.sum(h_values)  # Adjust to meet the integral constraint
    best_c5 = evaluate_erdos_solution(h_values)
    best_h_values = h_values.copy()
    
    # Optimization parameters
    num_iterations = 100
    learning_rate = 0.01
    decay_rate = 0.99
    
    for i in range(num_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h_values)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_values = h_values.copy()
        
        # Compute gradients for all shifts (k = 1 to n_points - 1)
        gradients = np.zeros_like(h_values)
        for k in range(1, n_points):
            # Compute overlap integral derivative
            overlap = np.correlate(h_values, 1 - h_values, mode="full")
            overlap_shifted = np.roll(overlap, k)
            overlap_integral = overlap_shifted[n_points - 1]
            
            # Compute derivative of the overlap integral with respect to h_values
            derivative = np.zeros_like(h_values)
            for i in range(n_points):
                # For each x in h_values, compute derivative of the overlap integral
                # at x + k
                if i + k < n_points:
                    derivative[i] += (1 - h_values[i + k]) * (1 - h_values[i]) * (2.0 / n_points)
                if i - k >= 0:
                    derivative[i] += (1 - h_values[i - k]) * (1 - h_values[i]) * (2.0 / n_points)
            
            # Compute gradient for current shift (k)
            gradient = -derivative * (2.0 / n_points) * overlap_integral
            gradients += gradient
        
        # Update h_values using gradient descent
        h_values -= learning_rate * gradients
        h_values = np.clip(h_values, 0, 1)
        h_values = h_values * (n_points / 2) / np.sum(h_values)  # Adjust to meet the integral constraint
        
        # Decay learning rate
        learning_rate *= decay_rate
    
    # Return best solution found
    return best_h_values, best_c5, n_points
