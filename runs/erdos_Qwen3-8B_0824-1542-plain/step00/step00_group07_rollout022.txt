def run(seed=42, budget_s=60, **kwargs):
    import random
    import time
    import numpy as np

    np.random.seed(seed)

    # Number of points
    n_points = 1000
    if n_points < 500:
        n_points = 500
    elif n_points > 1000:
        n_points = 1000

    # Initial solution
    h = np.copy(initial_h_values)
    h = np.resize(h, (n_points,))
    h = h[:n_points]
    h = np.pad(h, (0, n_points - len(h)), 'constant')  # ensure length matches

    # Normalize to satisfy the integral constraint
    sum_h = np.sum(h)
    if sum_h == 0:
        h = np.ones(n_points) * (n_points / 2) / n_points
    else:
        h = h * (n_points / 2) / sum_h

    # Function to evaluate C5
    def evaluate(h):
        h = np.resize(h, (n_points,))
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * (2.0 / n_points)

    # Function to compute the overlap for a given h
    def compute_c5(h):
        return evaluate(h)

    # Function to compute the integral constraint
    def constraint(h):
        return np.sum(h) - (n_points / 2)

    # Optimization settings
    def optimize_h(h_initial):
        h = cp.Variable(n_points)
        obj = cp.Minimize(cp.norm(cp.multiply(h, 1 - h)))
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.SCS, eps=1e-6)
        if prob.status != 'optimal':
            print("Optimization failed")
            return h_initial
        return h.value

    start_time = time.time()
    best_h = h
    best_c5 = compute_c5(h)

    # Perform iterations to improve the solution
    for _ in range(10):
        h = optimize_h(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5
        if time.time() - start_time > budget_s - 1:
            break

    # Refine using a local optimizer
    def objective(h):
        return compute_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]
    result = optimize.minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, jac=None, tol=1e-6)
    best_h = result.x
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_points
