```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    n_points = n_points_initial
    
    best_solution = h_initial
    best_c5 = evaluate_erdos_solution(h_initial)
    
    # Try reducing n_points to optimize speed
    for n_points in [n_points_initial // 2, n_points_initial // 3, n_points_initial // 4]:
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Initial approximation
        h = np.random.rand(n_points)
        h = h * 2 - 1  # Random guess within [0, 1]
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Ensure integral is 1
        
        # Optimization: minimize max correlation
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            # Compute 1 - h shifted by k
            shifted = np.zeros((n_points, n_points))
            for k in range(1, n_points):
                shifted[:, k] = np.roll(h, k)
                shifted[:, k] = 1 - shifted[:, k]
            # Compute overlap integrals
            c5 = np.max(np.sum(h[:, np.newaxis] * shifted, axis=1) * dx)
            return c5
        
        def objective_func_with_grad(h_flat):
            h = h_flat.reshape(n_points)
            # Compute 1 - h shifted by k
            shifted = np.zeros((n_points, n_points))
            for k in range(1, n_points):
                shifted[:, k] = np.roll(h, k)
                shifted[:, k] = 1 - shifted[:, k]
            # Compute overlap integrals
            c5 = np.max(np.sum(h[:, np.newaxis] * shifted, axis=1) * dx)
            # Gradient
            grad = np.zeros(n_points)
            for k in range(1, n_points):
                shifted_k = np.roll(h, k)
                shifted_k = 1 - shifted_k
                overlap = np.sum(h[:, np.newaxis] * shifted_k, axis=1)
                indices = np.argmax(overlap)
                grad += (overlap[indices] * dx) * np.roll(np.ones(n_points), k)
            # Normalize gradient
            if np.max(grad) > 0:
                grad = grad / np.max(grad)
            return c5, grad.flatten()
        
        # Optimization using L-BFGS-B
        result = optimize.minimize(
            objective_func,
            h,
            method='L-BFGS-B',
            jac=objective_func_with_grad,
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        # Check if the solution improves the bound
        h_optimized = result.x
        h_optimized = h_optimized.reshape(n_points)
        h_optimized = np.clip(h_optimized, 0, 1)
        h_optimized = h_optimized * (n_points / 2) / np.sum(h_optimized)
        c5_val = evaluate_erdos_solution(h_optimized)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_solution = h_optimized
        
        if time.time() - start_time > budget_s:
            break
    
    # Evaluate final solution
    final_c5 = evaluate_erdos_solution(best_solution)
    return (best_solution.tolist(), final_c5, len(best_solution))
```
