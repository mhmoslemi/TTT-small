```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Initial parameters
    n_points = 42  # Starting with the initial sample size
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(h, n_points, dx)
    best_h = h.copy()
    best_n_points = n_points
    best_time = 0.0
    
    # Try increasing the number of points up to a maximum
    max_n_points = 100
    total_time = 0.0
    
    for n_points in range(42, max_n_points + 1):
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        # Use a simple initialization (e.g. uniform) to avoid local optima
        h[:(n_points // 2)] = 1.0
        h[(n_points // 2):] = 0.0
        # Enforce the constraint sum(h) = n_points / 2
        h = h / (n_points / 2) * (n_points / 2)
        
        # Define the variables
        x = cp.Variable(n_points, bounds=(0.0, 1.0))
        # Enforce the constraint sum(x) = n_points / 2
        constraint = cp.sum(x) == n_points / 2
        
        # Objective: minimize the maximum overlap
        # Construct the overlap integral for all k
        max_overlap = 0.0
        for k in range(1, n_points):
            if k >= n_points:
                break
            overlap_k = cp.sum(x * (1 - x[k:])) * dx
            max_overlap = cp.maximum(max_overlap, overlap_k)
        
        # Define the problem
        prob = cp.Problem(cp.Minimize(max_overlap), [constraint])
        
        # Solve the problem
        start_time = time.time()
        try:
            prob.solve(solver=cp.SCS, eps=1e-6, max_iter=1000)
        except:
            prob.solve(solver=cp.CVXOPT, eps=1e-6, max_iter=1000)
        end_time = time.time()
        total_time += end_time - start_time
        
        # Check if we've exceeded the budget
        if total_time >= budget_s:
            break
        
        # Extract the solution
        h = x.value
        current_c5 = evaluate_erdos_solution(h, n_points, dx)
        
        # Update the best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            best_n_points = n_points
    
    # After the loop, return the best solution
    return best_h, best_c5, best_n_points
```
