```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 45  # Starting with 45 points as per initial construction

    # Try increasing the number of points up to 200 (as per budget)
    for n_points in [45, 60, 80, 100, 120, 140, 160, 180, 200]:
        if time.time() - start_time > budget_s - 1:
            break

        # Generate initial h values
        h = np.zeros(n_points)
        h[:] = initial_h_values[:]  # Initial construction
        h = h[:n_points]  # Ensure it fits the number of points
        h /= np.sum(h) * 2  # Normalize to integral 1

        # Optimization: minimize the maximum overlap
        # We use a convex optimization approach with CVXPY
        h_var = cp.Variable(n_points, nonneg=True)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            0 <= h_var,
            h_var <= 1
        ]

        # Objective is to minimize the maximum overlap
        # Use a convex approximation with a parameter 'rho'
        rho = 0.1
        obj = cp.Minimize(cp.sum(cp.maximum(0, cp.multiply(h_var, 1 - h_var[1:]))) + rho * cp.sum(h_var))
        prob = cp.Problem(obj, constraints)

        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iter=100)
            if prob.status == 'optimal':
                h_opt = np.array(h_var.value)
                c5 = evaluate_erdos_solution(h_opt, n_points)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
        except:
            pass

        # Try a gradient-based optimization
        def objective(h):
            dx = 2.0 / n_points
            overlap = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(overlap)

        def objective_grad(h):
            dx = 2.0 / n_points
            overlap = np.correlate(h, 1 - h, mode="full") * dx
            max_idx = np.argmax(overlap)
            grad = np.zeros_like(h)
            # Compute derivative of overlap at max_idx
            # Using finite differences
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                h_minus = h.copy()
                h_minus[i] -= 1e-5
                overlap_plus = np.correlate(h_plus, 1 - h_plus, mode="full") * dx
                overlap_minus = np.correlate(h_minus, 1 - h_minus, mode="full") * dx
                grad[i] = (overlap_plus[max_idx] - overlap_minus[max_idx]) / (2 * 1e-5)
            return grad

        # Use L-BFGS-B for optimization
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            jac=objective_grad,
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-6}
        )

        if result.success:
            h_opt = result.x
            c5 = evaluate_erdos_solution(h_opt, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt

    # Return the best result found
    return best_h, best_c5, n_points
```
