```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Initial solution (n_points = 80)
    n_points = 80
    h_initial = initial_h_values.copy()
    
    # Compute initial C₅
    initial_c5 = evaluate_erdos_solution(h_initial)
    
    # Define the objective function to minimize
    def objective(h):
        # Ensure h has the correct number of points
        h = np.array(h).reshape(-1)
        if len(h) != n_points:
            raise ValueError(f"Expected {n_points} points, got {len(h)}")
        # Compute overlap integral
        overlap = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * (2.0 / n_points))
        return c5
    
    # Define constraints: sum(h) == n_points / 2
    def constraint_sum(h):
        h = np.array(h).reshape(-1)
        return np.sum(h) - n_points / 2
    
    # Define bounds: 0 <= h[i] <= 1
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Set up the optimization problem
    x0 = h_initial.copy()
    
    # Use SLSQP method for optimization
    result = optimize.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'disp': False, 'maxiter': 100}
    )
    
    # Extract best solution
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Check if we can try a larger n_points (up to 1000) for potentially better results
    # If time allows, we'll try a larger n_points and see if we can get a better C₅
    if budget_s > 10:
        # Try with n_points = 1000
        n_points = 1000
        dx = 2.0 / n_points
        h_new = np.random.uniform(0, 1, n_points)
        h_new = h_new - np.min(h_new)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Compute initial C₅ for the new h
        overlap = signal.correlate(h_new, 1 - h_new, mode="full")
        c5_new = np.max(overlap * dx)
        
        # If the new C₅ is better, proceed with optimization
        if c5_new < best_c5:
            # Optimization for the new n_points
            def objective_new(h):
                h = np.array(h).reshape(-1)
                if len(h) != n_points:
                    raise ValueError(f"Expected {n_points} points, got {len(h)}")
                overlap = signal.correlate(h, 1 - h, mode="full")
                c5 = np.max(overlap * dx)
                return c5
            
            def constraint_sum_new(h):
                h = np.array(h).reshape(-1)
                return np.sum(h) - n_points / 2
            
            bounds_new = [(0.0, 1.0) for _ in range(n_points)]
            x0_new = h_new.copy()
            
            result_new = optimize.minimize(
                objective_new,
                x0_new,
                method='SLSQP',
                bounds=bounds_new,
                constraints={'type': 'eq', 'fun': constraint_sum_new},
                options={'disp': False, 'maxiter': 50}
            )
            
            best_h = result_new.x
            best_c5 = objective_new(best_h)
    
    return best_h, best_c5, n_points
```
