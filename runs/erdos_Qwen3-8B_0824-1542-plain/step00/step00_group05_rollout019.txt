```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    n_points = 100  # Start with 100 points, which is under 1k

    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        dx = 2.0 / n_points
        h_array = np.array(h)
        shifted_h = np.concatenate([np.zeros(n_points), h_array[:-1]])
        overlap = signal.correlate(h_array, 1 - h_array, mode="full") * dx
        return np.max(overlap)

    # Function to optimize h using gradient descent with constraints
    def optimize_h(h_initial):
        h = cp.Variable(n_points)
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        objective = cp.Minimize(evaluate_overlap(h))
        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve(solver=cp.SCS, max_iter=1000, eps=1e-5)
            if result is not None and np.isfinite(result):
                return h.value
        except:
            pass
        return h_initial

    # Try different n_points to find the best solution within budget
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s:
            break
        h_initial = initial_h_values.copy()[:n_points]
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = optimize_h(h_initial)
        h_initial = np.clip(h_initial, 0, 1)
        c5 = evaluate_overlap(h_initial)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_initial

    return best_h, best_c5, n_points
```
