```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import cvxpy as cp
import random

# Pre-imported utilities
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    best_n_points = n_points

    # Function to evaluate the C5 value
    def evaluate(h, dx):
        h = np.array(h, dtype=float)
        h = np.concatenate((h, h[:1]))  # Double the array to handle wrap-around
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    # Function to optimize with CVXPY for a given number of points
    def optimize_solution(n_points):
        dx = 2.0 / n_points
        h = cp.Variable(n_points, nonneg=True)
        constraints = [cp.sum(h) == n_points / 2.0]  # Ensure ∫h(x)dx = 1

        # Objective is to minimize the maximum overlap
        # We use the max function with a constraint to approximate the minmax
        # Define the correlation for all shifts in [0, 2]
        def get_overlap(h_vec):
            h = np.concatenate((h_vec, h_vec[:1]))  # Double for wrap-around
            corr = signal.correlate(h, 1 - h_vec, mode='full') * dx
            return np.max(corr)

        # Approximate the max overlap with constraints
        t = cp.Variable()
        constraints += [t >= get_overlap(h)]
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve(solver=cp.SCS, eps=1e-4, max_iter=100)
            if result is not None:
                h_values = h.value
                h_values[h_values < 0] = 0
                h_values[h_values > 1] = 1
                return h_values, get_overlap(h_values)
        except:
            pass
        return None, float('inf')

    # Try a series of point counts, starting from initial_n_points
    for n_points in [len(initial_h_values), 50, 100, 200]:
        if time.time() - start_time > budget_s - 5:
            break
        
        print(f"Trying n_points = {n_points}")
        h_opt, c5_opt = optimize_solution(n_points)
        if h_opt is not None and c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            best_n_points = n_points

    # Try local optimization with gradient descent on current solution
    def objective(h):
        return evaluate(h, 2.0 / len(h))

    result = opt.minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)] * len(best_h), tol=1e-6)
    if result.success:
        best_h = result.x
        best_c5 = evaluate(best_h, 2.0 / len(best_h))

    # Final verification
    final_c5 = evaluate(best_h, 2.0 / len(best_h))
    print(f"Final C5 bound: {final_c5:.6f}")
    return best_h, final_c5, best_n_points
```
