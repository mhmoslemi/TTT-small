```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial h_values
    n_points = len(initial_h_values)
    h = np.array(initial_h_values)
    
    # Function to compute C5
    def compute_c5(h_values):
        h = h_values
        dx = 2.0 / len(h)
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Find maximum overlap integral
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Function to evaluate and update the best solution
    def update_best_solution(current_h, current_c5, best_h, best_c5):
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
        return best_h, best_c5

    # Define the objective function for optimization
    def objective(h_values):
        return compute_c5(h_values)

    # Define the constraints
    def constraints(h_values):
        # Constraint 1: Integral of h is 1
        dx = 2.0 / len(h_values)
        integral = np.sum(h_values) * dx
        return [integral - 1]

    # Optimization setup
    n_points = len(h)
    h_var = cp.Variable(n_points)
    objective_func = cp.Minimize(compute_c5(h_var))
    constraints_func = [cp.sum(h_var) == n_points / 2]

    # Run optimization
    prob = cp.Problem(objective_func, constraints_func)
    try:
        prob.solve(solver=cp.SCS, max_iter=1000)
        if prob.status == 'optimal':
            h_opt = h_var.value
            c5_opt = compute_c5(h_opt)
        else:
            h_opt = h.copy()
            c5_opt = compute_c5(h_opt)
    except:
        h_opt = h.copy()
        c5_opt = compute_c5(h_opt)

    # Refine the solution with local optimization
    result = optimize.minimize(
        objective,
        h_opt,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        tol=1e-6,
        options={'maxiter': 100}
    )
    
    h_opt = result.x
    c5_opt = compute_c5(h_opt)
    
    # Check if the solution is valid
    dx = 2.0 / len(h_opt)
    if np.sum(h_opt) * dx != 1:
        h_opt = h.copy()
        c5_opt = compute_c5(h_opt)
    
    # Best solution
    best_h = h_opt
    best_c5 = c5_opt
    
    # Early exit if time is up
    if time.time() - start_time > budget_s - 1:
        return (best_h, best_c5, len(best_h))
    
    # Further refinement
    for _ in range(3):
        result = optimize.minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(len(best_h))],
            tol=1e-6,
            options={'maxiter': 50}
        )
        best_h = result.x
        best_c5 = compute_c5(best_h)
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_h, best_c5, len(best_h))
```
