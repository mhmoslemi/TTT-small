```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Import pre-defined initial solution
from evaluate_erdos_solution import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 70
    h_values = initial_h_values.copy()
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        dx = 2.0 / len(h)
        return np.max(corr * dx)
    
    # Function to enforce integral constraint
    def integral_constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization setup
    n_points = len(h_values)
    h = cp.Variable(n_points)
    objective = cp.Minimize(cp.Max(cp.sum(cp.multiply(h, 1 - h[1:]))) + cp.sum(cp.multiply(h, 1 - h[-1])) / 2)
    constraints = [h >= 0, h <= 1, integral_constraint(h) == 0]
    
    # Try optimization with different solvers
    solvers = ['ECOS', 'SCS', 'MOSEK', 'CVXOPT', 'GLPK', 'CBC']
    best_c5 = np.inf
    best_h = h_values
    
    for solver in solvers:
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=solver, verbose=False)
            if prob.status == 'OPTIMAL':
                h_sol = h.value
                if h_sol is not None:
                    c5 = compute_c5(h_sol)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_sol
        except:
            continue
        
        # Early exit if time is up
        if time.time() - start_time > budget_s - 1:
            break
    
    # Fallback to initial solution if no improvement
    if best_c5 == np.inf:
        best_c5 = compute_c5(h_values)
        best_h = h_values
    
    # Refine with local optimization
    h_refined = best_h.copy()
    for _ in range(5):
        # Local search
        for i in range(n_points):
            h_refined[i] = np.clip(h_refined[i] + np.random.uniform(-0.1, 0.1), 0, 1)
            if np.sum(h_refined) < n_points / 2 - 0.1:
                h_refined[i] = np.clip(h_refined[i] + 0.1, 0, 1)
            elif np.sum(h_refined) > n_points / 2 + 0.1:
                h_refined[i] = np.clip(h_refined[i] - 0.1, 0, 1)
        h_refined = np.clip(h_refined, 0, 1)
        np.random.shuffle(h_refined)
        h_refined = np.clip(h_refined, 0, 1)
        c5 = compute_c5(h_refined)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_refined
    
    # Final check to respect integral constraint
    best_h = np.clip(best_h, 0, 1)
    while np.sum(best_h) < n_points / 2 - 0.1:
        best_h[np.random.randint(n_points)] += 0.1
    while np.sum(best_h) > n_points / 2 + 0.1:
        best_h[np.random.randint(n_points)] -= 0.1
    best_h = np.clip(best_h, 0, 1)
    
    # Final evaluation
    c5_final = compute_c5(best_h)
    return best_h, c5_final, n_points
```
