import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions and initial values
initial_h_values = np.array([...])  # Assume this is a valid initial construction

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Initial number of points - start with 100 points
    n_points = 100
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        nonlocal n_points
        h = np.array(h)
        # Normalize to ensure the integral is 1
        h_normalized = h / (np.sum(h) / n_points)
        # Compute the overlap integrals
        overlap = np.correlate(h_normalized, 1 - h_normalized, mode="full")
        # Multiply by dx (2 / n_points)
        overlap *= 2 / n_points
        # Take the maximum overlap integral
        return np.max(overlap)
    
    # Function to optimize with constraints
    def optimize_h(n_points):
        # Define the variables
        h = cp.Variable(n_points, lowBound=0, upBound=1)
        # Constraint: the integral of h must be 1
        constraint = cp.sum(h) == n_points / 2
        # Objective: minimize the maximum overlap integral
        # We use a semi-infinite optimization approach by minimizing the maximum value
        # We use a convex relaxation by bounding the maximum overlap
        # For simplicity, we'll use a grid-based approach to approximate the maximum
        max_overlap = cp.Variable()
        constraints = []
        for k in range(1, n_points):
            # We compute the overlap for k=1 to k=n_points-1
            # For each k, compute the overlap integral
            # Since we're using a finite number of points, we'll approximate the integral
            # by taking a weighted average and using the dx
            # We approximate the integral by using the correlation
            # We need to handle the shift properly
            # Create h_shifted for the shift k
            h_shifted = np.roll(h, k)
            # Compute the overlap integral
            overlap = np.sum(h * (1 - h_shifted)) * (2 / n_points)
            # Add the constraint that the maximum overlap is at least this
            constraints.append(max_overlap >= overlap)
        # Objective: minimize the maximum overlap
        objective = cp.Minimize(max_overlap)
        # Solve the problem
        prob = cp.Problem(objective, constraints + [constraint])
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == "optimal":
                return h.value
            else:
                print("Optimization failed, using initial guess")
                return initial_h_values[:n_points]
        except:
            print("Optimization failed, using initial guess")
            return initial_h_values[:n_points]
    
    # Start with initial n_points
    best_h = initial_h_values[:n_points]
    best_c5 = evaluate_overlap(best_h)
    best_n_points = n_points
    
    # Try increasing the number of points to improve the bound
    for n_points in [200, 300, 400, 500]:
        start_time = time.time()
        h = optimize_h(n_points)
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break  # Stop if we exceed the budget
        c5 = evaluate_overlap(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
            best_n_points = n_points
    
    return best_h, best_c5, best_n_points
