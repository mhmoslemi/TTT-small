import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Placeholder, should be replaced with actual initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Determine number of points based on budget
    # We'll use n_points = 400 as it's a good balance between speed and accuracy
    n_points = 400
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    
    # Start from the initial construction
    if len(initial_h_values) == n_points:
        initial_h = initial_h_values.copy()
    else:
        # If initial is not available, use a random construction
        np.random.seed(seed)
        initial_h = np.random.rand(n_points)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Function to evaluate the overlap integral
    def evaluate(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Function to minimize the maximum overlap integral
    def optimization_objective(h_vec):
        h = h_vec.copy()
        # Ensure h stays within [0,1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integrals for all shifts
        corr = np.correlate(h, 1 - h, mode="full")
        # Return the maximum overlap integral
        return np.max(corr) * dx
    
    # Function to constrain the integral of h to be 1
    def constraint_function(h_vec):
        return np.sum(h_vec) * dx - 1
    
    # Define the optimization variables
    h = cp.Variable(n_points)
    objective = cp.Minimize(optimization_objective(h))
    
    # Add constraint for the integral of h to be 1
    constraints = [constraint_function(h) == 0]
    
    # Solve the optimization problem
    start_time = time.time()
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except:
        # If ECOS fails, try another solver
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
        except:
            prob.solve(solver=cp.SCS, verbose=False)
    
    # Check if the optimization was successful
    if prob.status != 'OPTIMAL':
        # If not, use the initial guess
        h_values = initial_h
    else:
        h_values = h.value
    
    # Clip the values to [0,1]
    h_values = np.clip(h_values, 0, 1)
    
    # Ensure the integral constraint is satisfied
    if np.abs(np.sum(h_values) * dx - 1) > 1e-6:
        # If it's not satisfied, use the initial guess
        h_values = initial_h
    
    # Evaluate the best solution
    c5_bound = evaluate(h_values)
    
    return h_values, c5_bound, n_points
