        h = np.clip(h, 0, 1)
        h = np.concatenate([h, h[:n_points//2]])  # To handle wrap-around
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def evaluate_and_update(h_values, n_points):
        h = np.array(h_values)
        h = np.clip(h, 0, 1)
        h = np.concatenate([h, h[:n_points//2]])
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Initial guess

    # Initial guess: uniform distribution
    h_values = np.full(n_points, 1.0 / n_points * sum_h)
    
    # Optimization loop
    best_h = h_values.copy()
    best_c5 = evaluate_and_update(best_h, n_points)
    elapsed = 0.0

    # Use a simple gradient descent with momentum
    learning_rate = 0.1
    momentum = 0.9
    gradient = np.zeros(n_points)
    for _ in range(1000):
        start_time = time.time()
        h = np.clip(best_h, 0, 1)

        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        
        # Compute gradient approximation using finite differences
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += 1e-5
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = np.concatenate([h_plus, h_plus[:n_points//2]])
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
            max_overlap_plus = np.max(corr_plus) * dx
            grad[i] = (max_overlap_plus - max_overlap) / 1e-5

        gradient = momentum * gradient + grad
        best_h = best_h - learning_rate * gradient
        best_h = np.clip(best_h, 0, 1)

            best_h = best_h.copy()
        
        elapsed = time.time() - start_time

            break
