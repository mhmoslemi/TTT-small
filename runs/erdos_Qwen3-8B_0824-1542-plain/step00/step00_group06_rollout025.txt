import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial solution
    n_points = len(initial_h_values)
    h = np.array(initial_h_values)
    dx = 2.0 / n_points
    
    # Function to compute C5 value
    def compute_c5(h):
        h = np.array(h)
        h_expanded = np.concatenate([h, np.zeros_like(h)])
        h_shifted = np.zeros_like(h_expanded)
        for k in range(1, len(h_expanded) - 1):
            h_shifted[k:] = h[:-k]
        h1_minus_h = 1 - h_expanded
        cross_corr = np.correlate(h_expanded, h1_minus_h, mode="full")
        c5 = np.max(cross_corr) * dx
        return c5
    
    # Function to compute the objective and constraints for optimization
    def optimize_h(h_initial):
        h = cp.Variable(h_initial.shape)
        dx = 2.0 / len(h_initial)
        objective = cp.Minimize(cp.Max(cp.multiply(h, 1 - h)))
        constraints = [
            cp.sum(h) == len(h_initial) / 2,
            0 <= h,
            h <= 1
        ]
        problem = cp.Problem(objective, constraints)
        result = problem.solve(solver=cp.SCS, verbose=False)
        if result is None:
            return h_initial
        return h.value
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_n_points = n_points
    
    # Try reducing n_points for faster evaluation if possible
    for n_points in [512, 256, 128, 64, 32, 16, 8]:
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            best_n_points = n_points
    
    # Try optimizing the initial solution
    h_optimized = optimize_h(best_h)
    current_c5 = compute_c5(h_optimized)
    if current_c5 < best_c5:
        best_h = h_optimized
        best_c5 = current_c5
        best_n_points = len(h_optimized)
    
    # Try a few more random samples to escape local minima
    for _ in range(3):
        h = np.random.rand(len(best_h)) * 2 - 1
        h = np.clip(h, 0, 1)
        h = h * (len(best_h) / 2) / np.sum(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            best_n_points = len(h)
    
    # Final check to ensure it's under the budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h.tolist(), best_c5, best_n_points)
    
    return (best_h.tolist(), best_c5, best_n_points)
