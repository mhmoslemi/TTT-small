def run(seed=42, budget_s=60, **kwargs):
    n_points = 100  # Start with a moderate number of points
    dx = 2.0 / n_points
    
    # Initial guess from pre-imported values
    h_values = initial_h_values[:n_points].copy()
    # Ensure the initial guess satisfies the integral constraint
    total = np.sum(h_values) * dx
    scale = 1.0 / total
    h_values *= scale
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = h.copy()
        h = np.concatenate((h, h[:n_points//2]))  # Extend for full convolution
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to check constraints and compute objective
    def objective_func(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        total = np.sum(h) * dx
        h = h * (1.0 / total)  # Normalize to satisfy integral constraint
        return compute_c5(h)
    
    # Function to compute the value of C5
    def compute_c5_value(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        total = np.sum(h) * dx
        h = h * (1.0 / total)
        return compute_c5(h)
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    x0 = h_values.copy()
    
    # Optimization options
    options = {
        'maxiter': 100,
        'ftol': 1e-6,
        'gtol': 1e-6,
        'disp': False
    }
    
    # Run optimization
    start_time = time.time()
    result = optimize.minimize(objective_func, x0, method='L-BFGS-B', bounds=bounds, options=options)
    
    # Check elapsed time
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        # Return the best solution found so far
        return (h_values, compute_c5_value(x0), n_points)
    
    # Extract the best solution
    best_h = result.x.copy()
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h) * dx
    best_h = best_h * (1.0 / total)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n_points)
