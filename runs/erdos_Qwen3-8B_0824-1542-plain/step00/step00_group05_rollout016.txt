import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial construction from initial_h_values
    initial_h_values = np.array(initial_h_values)
    n_points_initial = len(initial_h_values)
    dx = 2.0 / n_points_initial
    sum_initial = np.sum(initial_h_values) * dx
    # Adjust initial construction to satisfy the integral constraint
    initial_h_values *= (1 / sum_initial) * (n_points_initial / 2)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    
    # Start with the initial construction
    h = initial_h_values.copy()
    n_points = n_points_initial
    best_c5 = np.inf
    
    # Try reducing the number of points to find a faster solution
    for n_points in [100, 200, 300, 400, 500]:
        # Define function to compute C₅
        def compute_c5(h_values):
            dx = 2.0 / n_points
            h = h_values.reshape(n_points)
            # Compute the convolution (full mode) of h and (1 - h)
            correlation = np.correlate(h, 1 - h, mode="full")
            # Take the maximum value over all lags
            max_overlap = np.max(correlation) * dx
            return max_overlap
        
        # Define bounds for h[x] in [0,1] for all x
        bounds = Bounds(np.zeros(n_points), np.ones(n_points))
        
        # Initial guess is the current h_values
        initial_guess = h.copy()
        
        # Start timer
        start_time = time.time()
        
        # Use L-BFGS-B optimizer for speed and effectiveness
        result = minimize(
            compute_c5,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 1000, 'ftol': 1e-8}
        )
        
        # Check if the optimization completed within the budget
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
        
        # Update best solution
        current_h = result.x
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            h = current_h
        
    # Ensure the final h satisfies the integral constraint
    dx = 2.0 / n_points
    sum_h = np.sum(h) * dx
    h *= (1 / sum_h) * (n_points / 2)
    h = np.clip(h, 0, 1)
    
    return h, best_c5, n_points
