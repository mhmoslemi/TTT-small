import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import sys
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    elapsed_time = 0
    
    # Define parameters
    n_points = 100  # Start with 100 points, can be increased if time allows
    
    # Initial guess
    h = initial_h_values.copy()
    h = h[:n_points]  # Truncate to n_points
    h = h / (n_points / 2)  # Normalize to satisfy the integral constraint
    
    # Function to evaluate the C5 value
    def evaluate_C5(h_values):
        h = np.array(h_values)
        # Ensure it's within [0, 1]
        h = np.clip(h, 0, 1)
        
        # Compute the overlap integral
        corr = signal.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        c5 = np.max(corr) * dx
        
        return c5
    
    # Optimization function
    def optimize_C5(h_values):
        nonlocal elapsed_time
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            return np.inf
        
        h = np.array(h_values)
        # Ensure it's within [0, 1]
        h = np.clip(h, 0, 1)
        
        # Compute the overlap integral
        corr = signal.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        c5 = np.max(corr) * dx
        
        # Enforce the integral constraint using a penalty
        integral_penalty = 1e4 * np.abs(np.sum(h) * dx - 1)
        
        return c5 + integral_penalty
    
    # Bounds for each point: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial optimization using L-BFGS-B
    result = optimize.minimize(
        optimize_C5,
        h,
        bounds=bounds,
        method='L-BFGS-B',
        jac=None,
        options={'maxiter': 100, 'disp': False}
    )
    
    # Best solution found
    best_h = result.x
    best_c5 = evaluate_C5(best_h)
    
    # Optional: Refine with a local search
    if elapsed_time < budget_s - 5:
        # Local search around the best solution
        for _ in range(5):
            # Perturb the solution
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            perturbed_h = best_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0, 1)
            
            # Evaluate
            c5_perturbed = evaluate_C5(perturbed_h)
            
            # Update best solution if better
            if c5_perturbed < best_c5:
                best_h = perturbed_h
                best_c5 = c5_perturbed
    
    # Final evaluation
    final_c5 = evaluate_C5(best_h)
    
    # Return the result
    return (best_h, final_c5, n_points)
