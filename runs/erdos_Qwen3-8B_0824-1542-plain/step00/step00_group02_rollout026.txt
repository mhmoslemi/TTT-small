import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions and variables
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial guess
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        return np.max(result * dx)
    
    # Function to compute the overlap for a given h
    def evaluate(h):
        return compute_c5(h)
    
    # Initial evaluation
    c5_initial = evaluate(h_initial)
    best_h = h_initial
    best_c5 = c5_initial
    
    # Try fine-tuning with gradient descent
    def objective(x):
        return evaluate(x)
    
    # Optimization with gradient descent
    def gradient_descent(h0, max_iter=100, learning_rate=0.01, tol=1e-6):
        h = h0.copy()
        for i in range(max_iter):
            grad = np.zeros_like(h)
            for shift in range(1, len(h)):
                overlap = np.correlate(h, 1 - h, mode="full")
                grad -= np.gradient(overlap * dx, shift)
            grad /= np.max(np.abs(grad)) + 1e-10
            h -= learning_rate * grad
            h = np.clip(h, 0, 1)
            if np.abs(evaluate(h) - evaluate(h_initial)) < tol:
                break
        return h
    
    h_tuned = gradient_descent(h_initial)
    c5_tuned = evaluate(h_tuned)
    
    if c5_tuned < best_c5:
        best_h = h_tuned
        best_c5 = c5_tuned
    
    # Try another starting point with random noise
    h_random = np.random.rand(n_points)
    h_random = (h_random * 2 - 1) * 0.5 + 0.5
    h_random = h_random * (n_points / 2) / np.sum(h_random)
    h_random = np.clip(h_random, 0, 1)
    c5_random = evaluate(h_random)
    
    if c5_random < best_c5:
        best_h = h_random
        best_c5 = c5_random
    
    # Try a few more small perturbations
    for _ in range(5):
        perturbation = np.random.randn(n_points) * 0.05
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        c5_perturbed = evaluate(h_perturbed)
        if c5_perturbed < best_c5:
            best_h = h_perturbed
            best_c5 = c5_perturbed
    
    # Try small n_points for faster evaluation
    n_points_small = 100
    dx_small = 2.0 / n_points_small
    h_small = np.random.rand(n_points_small)
    h_small = (h_small * 2 - 1) * 0.5 + 0.5
    h_small = h_small * (n_points_small / 2) / np.sum(h_small)
    h_small = np.clip(h_small, 0, 1)
    c5_small = evaluate(h_small)
    
    if c5_small < best_c5:
        best_h = h_small
        best_c5 = c5_small
    
    # Try convex optimization for better results
    h = cp.Variable(n_points_small)
    objective = cp.Minimize(cp.max(cp.correlate(h, 1 - h, mode="full") * dx_small))
    constraints = [cp.sum(h) == n_points_small / 2, 0 <= h, h <= 1]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)
    if prob.status == "optimal":
        h_opt = h.value
        c5_opt = evaluate(h_opt)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    # Ensure we finish before the time limit
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Try a few more iterations if time allows
        for _ in range(5):
            perturbation = np.random.randn(n_points_small) * 0.05
            h_perturbed = best_h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points_small / 2) / np.sum(h_perturbed)
            c5_perturbed = evaluate(h_perturbed)
            if c5_perturbed < best_c5:
                best_h = h_perturbed
                best_c5 = c5_perturbed
    
    # Final check
    best_c5 = np.min([best_c5, evaluate(best_h)])
    
    return (best_h, best_c5, n_points_small)
