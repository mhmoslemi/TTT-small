import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    h_shifted = np.zeros_like(h)
    for k in range(1, n_points):
        h_shifted = np.roll(h, k)
        overlap = correlate(h, 1 - h_shifted, mode="full")
        c5 = np.max(overlap) * dx
    return c5

def initial_h_values():
    return [0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
            0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
            0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
            0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
            0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5]

def run(seed=42, budget_s=60, **kwargs):
    # Initial parameters
    n_points = 100  # Start with 100 points
    initial_h = initial_h_values()
    initial_h = initial_h[:n_points]  # Truncate to n_points
    initial_h = np.array(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h) * dx
    # Adjust initial_h to satisfy the integral constraint
    if abs(sum_h - 1) > 1e-6:
        scaling = 1.0 / sum_h
        initial_h *= scaling
    initial_h = np.clip(initial_h, 0.0, 1.0)

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(initial_h)
    best_n_points = n_points
    best_time = 0.0

    # Start timer
    start_time = time.time()

    # Try reducing n_points
    for n_points in [80, 60, 50, 40, 30, 20]:
        if time.time() - start_time > budget_s - 1:
            break
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h) * dx
        if abs(sum_h - 1) > 1e-6:
            scaling = 1.0 / sum_h
            h *= scaling
        h = np.clip(h, 0.0, 1.0)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n_points = n_points
        print(f"Try n_points={n_points}: c5={c5}")

    # Try refining with a more sophisticated approach
    n_points = best_n_points
    h = best_h.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h) * dx
    if abs(sum_h - 1) > 1e-6:
        scaling = 1.0 / sum_h
        h *= scaling
    h = np.clip(h, 0.0, 1.0)
    c5 = evaluate_erdos_solution(h)
    if c5 < best_c5:
        best_h = h.copy()
        best_c5 = c5

    # Refine further with optimization
    if time.time() - start_time < budget_s - 2:
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) * dx == 1,
            0 <= h,
            h <= 1
        ]
        # Objective: minimize the max overlap
        # To handle this, we can minimize the maximum overlap by introducing a variable
        # that bounds the overlap
        overlap_max = cp.Variable()
        constraints.append(overlap_max >= cp.multiply(h, 1 - cp.roll(h, k)) for k in range(1, n_points))
        objective = cp.Minimize(overlap_max)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver='ECOS')
            if prob.status == 'OPTIMAL':
                h_opt = h.value
                h_opt = np.clip(h_opt, 0.0, 1.0)
                c5_opt = evaluate_erdos_solution(h_opt)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
        except:
            pass

    # Final check
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), best_c5, best_n_points)

    return (best_h.tolist(), best_c5, best_n_points)
