```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(80) * 2 - 1  # Example initial values for h (needs to be scaled and normalized)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between accuracy and speed
    dx = 2.0 / n_points
    max_iterations = 100  # Max optimization iterations
    tolerance = 1e-4  # Tolerance for convergence
    best_c5 = float('inf')
    best_h = None

    # Initial guess based on provided initial_h_values
    h_initial = np.zeros(n_points)
    h_initial[:len(initial_h_values)] = initial_h_values
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    h_initial = np.clip(h_initial, 0, 1)

    start_time = time.time()

    for _ in range(max_iterations):
        # Objective: minimize the max overlap integral
        h = cp.Variable(n_points, nonneg=True)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        # Compute the overlap integrals for all shifts
        overlap = []
        for k in range(1, n_points):
            shifted_h = np.roll(h.value, k)
            overlap.append(np.max(np.dot(h.value, 1 - shifted_h) * dx))
        # Since we can't directly optimize the max, we use an epsilon-based approach
        epsilon = 1e-3
        for k in range(1, n_points):
            shifted_h = np.roll(h.value, k)
            overlap_k = np.dot(h.value, 1 - shifted_h) * dx
            if overlap_k > best_c5 + epsilon:
                overlap.append(overlap_k)
        # Use max(overlap) as the objective
        c5 = cp.Max(overlap)
        prob = cp.Problem(cp.Minimize(c5), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iter=500, eps=1e-5)
            if prob.status == 'optimal':
                h_val = h.value
                c5_val = prob.objective.value
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_val
        except:
            # If solver fails, use a gradient descent approach
            h_val = h_initial.copy()
            for _ in range(100):
                c5_list = []
                for k in range(1, n_points):
                    shifted_h = np.roll(h_val, k)
                    c5_list.append(np.dot(h_val, 1 - shifted_h) * dx)
                c5_val = np.max(c5_list)
                # Gradient descent step
                grad = np.zeros(n_points)
                for k in range(1, n_points):
                    shifted_h = np.roll(h_val, k)
                    grad += np.dot(1 - shifted_h, np.eye(n_points)[range(n_points)] - np.roll(np.eye(n_points), k, axis=0))
                grad *= dx
                h_val -= 0.01 * grad
                h_val = np.clip(h_val, 0, 1)
                h_val = h_val * (n_points / 2) / np.sum(h_val)
                h_val = np.clip(h_val, 0, 1)
                if np.max(np.abs(h_val - h_initial)) < tolerance:
                    break
            if np.max(np.abs(h_val - h_initial)) < tolerance:
                best_h = h_val
                best_c5 = np.max(np.dot(h_val, 1 - np.roll(h_val, k)) * dx for k in range(1, n_points))
            h_initial = h_val.copy()

        # Early stopping if budget is exhausted
        if time.time() - start_time > budget_s - 1:
            break

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
```
