def run(seed=42, budget_s=60, **kwargs):

    n_points = 100  # Adjust for speed and precision balance

    h = h * (n_points / 2) / np.sum(h)
    
    def objective_function(h):
        # Compute the overlap integral for the current h
        h = np.concatenate([h, np.zeros(n_points)])
        h_shifted = np.concatenate([np.zeros(n_points), h[:-n_points]])
        overlap = np.correlate(h, 1 - h_shifted, mode="full")
        c5 = np.max(overlap) * (2 / n_points)
        return c5
    
    def objective_function_and_grad(h):
        # Compute the overlap integral and its gradient
        h = np.concatenate([h, np.zeros(n_points)])
        h_shifted = np.concatenate([np.zeros(n_points), h[:-n_points]])
        overlap = np.correlate(h, 1 - h_shifted, mode="full")
        c5 = np.max(overlap) * (2 / n_points)
        
        # Gradient with respect to h (simplified approximation)
        grad = np.zeros_like(h)

            h_test = h.copy()

    # Optimization
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()

    for _ in range(100):

        result = optimize.minimize(
