import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial construction from the current best solution
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    dx = 2.0 / n_points_initial
    
    # Function to evaluate the C_5 bound
    def evaluate_c5(h):
        h = np.array(h)
        n = len(h)
        dx = 2.0 / n
        # Compute the overlap integral for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        # Take the maximum value scaled by dx
        c5 = np.max(overlap) * dx
        return c5
    
    # Start with the initial solution
    best_h = h_initial.copy()
    best_c5 = evaluate_c5(best_h)
    best_n = n_points_initial
    
    # Try smaller n_points if possible
    for n_points in range(40, n_points_initial, -1):
        dx = 2.0 / n_points
        # Generate initial guess for h
        h = np.zeros(n_points)
        # Spread the area uniformly
        h = np.repeat(np.linspace(0, 1, n_points), 1)
        h = np.clip(h, 0, 1)
        # Add some noise to avoid local optima
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        # Ensure the integral equals 1
        h = h * (n_points / 2) / np.sum(h)
        
        # Define optimization variables
        h_var = cp.Variable(n_points, nonneg=True)
        constraints = [cp.sum(h_var) == n_points / 2, 0 <= h_var, h_var <= 1]
        # Objective: minimize the maximum overlap
        overlap = correlate(h_var.value, 1 - h_var.value, mode="full")
        max_overlap = cp.Max(overlap)
        problem = cp.Problem(cp.Minimize(max_overlap), constraints)
        
        # Try to optimize
        start_time = time.time()
        try:
            result = problem.solve(solver=cp.ECOS, max_iter=100)
            if time.time() - start_time > budget_s:
                continue
            if result is not None:
                h_opt = h_var.value
                c5 = evaluate_c5(h_opt)
                if c5 < best_c5:
                    best_h = h_opt
                    best_c5 = c5
                    best_n = n_points
        except:
            pass
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, best_n
