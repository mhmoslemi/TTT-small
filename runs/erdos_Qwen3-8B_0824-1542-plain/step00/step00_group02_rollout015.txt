import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial construction
    n_points = 45
    h = np.array(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    # Adjust to ensure sum(h) * dx = 1 (i.e., sum(h) = n_points / 2)
    if abs(sum_h - n_points / 2) > 1e-6:
        h = h / sum_h * (n_points / 2)
    
    # Function to compute C5
    def compute_c5(h_values):
        n = len(h_values)
        dx = 2.0 / n
        h = np.array(h_values)
        # Compute h(x) * (1 - h(x+k)) for all lags
        # Use numpy convolution with mode="full"
        shifted = np.zeros_like(h)
        for k in range(1, n):
            shifted = np.roll(h, k)
            overlap = h * (1 - shifted)
            overlap_integral = np.sum(overlap) * dx
            if overlap_integral > c5:
                c5 = overlap_integral
        return c5
    
    # Initial C5
    c5 = compute_c5(h)
    
    # Optimization parameters
    max_iterations = 100
    improvement_threshold = 1e-5
    n_points = 45  # Start with 45 points, can increase if needed
    
    for _ in range(max_iterations):
        # Time check
        if time.time() - start_time > budget_s - 5:
            break
        
        # Define variables
        h = cp.Variable(n_points)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        
        # Objective: minimize the maximum overlap
        # Use a convex relaxation by minimizing the maximum of the overlaps
        overlapping_integrals = []
        for k in range(1, n_points):
            shifted = cp.roll(h, k)
            overlap = h * (1 - shifted)
            overlapping_integrals.append(overlap)
        
        # Objective is to minimize the maximum of the overlapping integrals
        # We use a convex relaxation by minimizing the maximum of the integrals
        objective = cp.Minimize(cp.Max([cp.sum(overlap) * (2.0 / n_points) for overlap in overlapping_integrals]))
        
        # Solve the problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-4, max_iter=100)
        except:
            # If SCS fails, try another solver
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, eps=1e-4, max_iter=100)
        
        # Update h with the solution
        h_values = h.value
        if h_values is None:
            break
        
        # Compute new C5
        new_c5 = compute_c5(h_values)
        
        # Check for improvement
        if abs(new_c5 - c5) < improvement_threshold:
            break
        c5 = new_c5
    
    # Return the best result found
    return h_values, c5, n_points
