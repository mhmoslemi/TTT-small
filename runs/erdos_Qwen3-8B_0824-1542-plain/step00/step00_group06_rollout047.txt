```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * 2.0 / len(h)
initial_h_values = np.load('initial_h.npy')  # Assuming this is a pre-loaded initial solution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points to keep computation manageable
    max_iter = 100  # Max number of iterations
    tolerance = 1e-4  # Stopping tolerance
    
    # Initial guess
    h = initial_h_values[:n_points]
    h = np.clip(h, 0, 1)
    
    # Normalize the initial guess to satisfy the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break
        
        # Compute current C5
        current_c5 = evaluate_erdos_solution(h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Random perturbation
        perturbation = np.random.normal(0, 0.05, n_points)
        h += perturbation
        h = np.clip(h, 0, 1)
        
        # Enforce integral constraint
        h = h * (n_points / 2) / np.sum(h)
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
```
