import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_h = np.sum(h_initial)
    # Ensure the initial sum matches the constraint
    if np.abs(sum_h - n_points / 2) > 1e-10:
        h_initial = h_initial / sum_h * (n_points / 2)
    
    # Define the objective function to minimize the max overlap
    def objective(h):
        h = h.reshape(-1)
        dx = 2.0 / len(h)
        h = np.clip(h, 0, 1)
        # Compute the convolution
        conv = np.correlate(h, 1 - h, mode="full")
        # Compute the max overlap
        c5 = np.max(conv) * dx
        return c5

    # Define a function to evaluate and return the best result
    def evaluate_solution(h):
        nonlocal start_time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            return (h, np.inf, len(h))
        try:
            c5 = objective(h)
            return (h, c5, len(h))
        except:
            return (h, np.inf, len(h))
    
    # Perform a descent step starting from the initial solution
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    best_n = len(best_h)
    
    # Try a few descent steps
    for _ in range(5):
        # Perturb the current solution slightly
        h_perturbed = best_h.copy()
        idx = np.random.randint(len(h_perturbed))
        h_perturbed[idx] += np.random.normal(0, 0.05)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Normalize to ensure the sum constraint
        h_perturbed = h_perturbed / np.sum(h_perturbed) * (len(h_perturbed) / 2)
        c5_perturbed = objective(h_perturbed)
        if c5_perturbed < best_c5:
            best_h = h_perturbed
            best_c5 = c5_perturbed
            best_n = len(best_h)
    
    # Try a few local optimization steps
    for _ in range(3):
        # Create a local optimization problem with constraints
        n = len(best_h)
        h = np.zeros(n)
        h[:] = best_h[:]
        A = np.zeros((n + 1, n))
        A[0, :] = 1.0
        b = np.array([n / 2.0])
        lb = np.zeros(n)
        ub = np.ones(n)
        
        # Define the objective function
        def objective_func(x):
            x = x.reshape(-1)
            dx = 2.0 / n
            conv = np.correlate(x, 1 - x, mode="full")
            c5 = np.max(conv) * dx
            return c5
        
        # Local optimization using SLSQP
        result = optimize.minimize(
            objective_func,
            h,
            bounds=list(zip(lb, ub)),
            method='SLSQP',
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2}]
        )
        
        if result.success:
            best_h = result.x
            best_c5 = objective(best_h)
            best_n = n
    
    # Check if we can improve with fewer points
    for n in range(40, len(best_h), -1):
        if n < 40:
            break
        h = np.zeros(n)
        h[:] = best_h[:n]
        h = h / np.sum(h) * (n / 2)
        c5 = objective(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
            best_n = n
    
    # Final evaluation
    result = evaluate_solution(best_h)
    return result
