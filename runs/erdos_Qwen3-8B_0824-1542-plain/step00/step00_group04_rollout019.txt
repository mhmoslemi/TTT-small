def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Choose a starting number of points
    n_points = 100  # Start with 100 points, can increase if needed
    
    # Initial guess
    h_initial = initial_h_values()
    h_initial = h_initial[:n_points]  # Truncate to n_points samples
    
    # Ensure the initial guess satisfies the integral constraint
    dx = 2.0 / n_points
    h_initial = h_initial * (n_points / 2) / (np.sum(h_initial) * dx)
    
    # Define the objective function to minimize
    def objective(h):
        # Scale h to [0, 2] interval
        h_scaled = h * dx
        # Compute the overlap integral
        corr = signal.correlate(h_scaled, 1 - h_scaled, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Define the constraints
    def constraints(h):
        # Integral constraint
        integral = np.sum(h) * dx
        return [integral - 1.0]
    
    # Optimization
    try:
        # Use bounded optimization with constraints
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraints},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        h_optimized = result.x
    except:
        # If optimization fails, return initial guess
        h_optimized = h_initial

    c5_bound = evaluate_erdos_solution(h_optimized)
