import numpy as np
import scipy.optimize as optimize
import copy
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Hyperparameters
    n_points = 500  # Number of points to use for the function h
    max_iterations = 100  # Max number of iterations for optimization
    learning_rate = 0.1  # Learning rate for gradient descent
    tolerance = 1e-5  # Tolerance for stopping the optimization
    
    # Initial guess: use the provided initial h_values and interpolate to n_points
    initial_h = initial_h_values.copy()
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    
    h_values = initial_h.copy()
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Compute the current overlap integral
        h = h_values
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        
        # Compute the gradient to minimize the maximum overlap
        # Approximate the gradient using finite differences
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_minus = h.copy()
            h_minus[i] -= 1e-5
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode="full")
            grad[i] = (np.max(corr_plus * dx) - np.max(corr_minus * dx)) / (2 * 1e-5)
        
        # Update h_values using gradient descent
        h_values -= learning_rate * grad
        
        # Ensure h_values stays within [0, 1] and satisfies the integral constraint
        h_values = np.clip(h_values, 0, 1)
        # Normalize to satisfy the integral constraint
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        
        # Early stopping if we meet the target C5 bound
        if c5 < 0.380876:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h_values, n_points)
    return (h_values, final_c5, n_points)
