```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Assuming a precomputed initial solution

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Set up optimization parameters
    max_points = 100  # Max number of samples to use
    n_points = min(max_points, len(initial_h_values))  # Use initial values if they are shorter
    
    # Convert initial values to a numpy array
    h_initial = np.array(initial_h_values[:n_points])
    
    # Ensure the initial h_initial satisfies the constraints
    if np.sum(h_initial) * (2.0 / n_points) != 1.0:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2.0)
    
    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Define the objective and constraints
    def objective_func(h):
        # Compute the overlap integral for each shift
        c5_values = []
        for k in range(len(h) - 1):
            overlap = np.sum(h * (1 - h[k+1:])) * (2.0 / len(h))
            c5_values.append(overlap)
        return max(c5_values)
    
    # Optimization settings
    max_iterations = 100
    tolerance = 1e-4
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        # Define the optimization problem
        h = cp.Variable(n_points)
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2.0
        ]
        obj = cp.Minimize(objective_func(h))
        prob = cp.Problem(obj, constraints)
        
        # Solve the problem
        try:
            prob.solve(solver=cp.SCS, eps=tolerance, max_iter=100)
        except:
            # Fallback to local optimization if SCS fails
            h_opt = cp.Variable(n_points)
            constraints = [
                h_opt >= 0,
                h_opt <= 1,
                cp.sum(h_opt) == n_points / 2.0
            ]
            obj = cp.Minimize(objective_func(h_opt))
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.SCS, eps=tolerance, max_iter=100)
        
        if prob.status != 'optimal':
            break
        
        # Convert to numpy array and evaluate
        h_opt_np = np.array(h.value)
        
        # Check if the solution satisfies the constraints
        if np.any(h_opt_np < 0) or np.any(h_opt_np > 1):
            continue
        
        c5 = evaluate_erdos_solution(h_opt_np)
        
        # Update best solution
        if c5 < best_c5:
            best_h = h_opt_np.copy()
            best_c5 = c5
        
        # Early stopping if we're within the time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Return the best solution and its C5 value
    return (best_h, best_c5, n_points)
```
