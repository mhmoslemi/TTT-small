```python
import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for speed

    # Start with the initial solution
    h_initial = initial_h_values
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    c5 = evaluate_erdos_solution(h_initial, n_points)
    best_c5 = c5
    best_h = h_initial.copy()

    # Try increasing n_points up to 500 with some budget
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 1:
            break

        # Generate a random initial guess
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)

        # Optimization using scipy.optimize
        def objective(x):
            h = x.reshape(n_points)
            overlap = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap) * (2.0 / n_points)
            return c5

        def constraints(x):
            return np.sum(x) - n_points / 2

        # Setup for optimization
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraints}
        result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)

        if result.success:
            h_opt = result.x
            c5_opt = evaluate_erdos_solution(h_opt, n_points)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Early exit if time is up
    if time.time() - start_time > budget_s - 1:
        return (best_h, best_c5, n_points)

    # Try a few more random starting points with the best n_points
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        c5 = evaluate_erdos_solution(h, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    return (best_h, best_c5, n_points)
```
