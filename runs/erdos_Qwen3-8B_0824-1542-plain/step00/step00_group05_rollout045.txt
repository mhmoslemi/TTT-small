import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    overlap = correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def initial_h_values():
    # This is a placeholder for the initial construction, which is supposed
    # to be imported from the environment. For the purpose of this example,
    # we'll return a dummy initial guess.
    n_points = 70
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1
        else:
            h[i] = 0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define the initial number of points and the target C5 value
    n_points = 100  # Start with a moderate number of points
    target_c5 = 0.380876
    best_c5 = float('inf')
    best_h = initial_h_values()
    
    # Initial check
    h = np.array(best_h)
    dx = 2.0 / n_points
    overlap = correlate(h, 1 - h, mode="full")
    current_c5 = np.max(overlap) * dx
    best_c5 = current_c5
    best_h = h.tolist()
    
    # Optimization loop
    start_time = time.time()
    
    # Try different numbers of points up to a maximum of 200
    for n_points in range(100, 200, 10):
        dx = 2.0 / n_points
        # Generate initial guess based on uniform distribution
        h_initial = np.random.rand(n_points)
        h_initial = (h_initial * 2 - 1) * 0.5 + 0.5  # Scale to [0.5, 1.0]
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        h_initial = h_initial.tolist()
        
        # Define optimization variables
        h = cp.Variable(n_points, nonneg=True)
        constraints = [cp.sum(h) == n_points / 2]
        objective = cp.Minimize(0)
        result = cp.Problem(objective, constraints)
        
        # Try to solve the problem
        try:
            result.solve()
            h_values = h.value.tolist()
            overlap = correlate(h_values, 1 - h_values, mode="full")
            c5 = np.max(overlap) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_values
        except:
            pass
        
        # Early stopping
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Final check
    h = np.array(best_h)
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    final_c5 = np.max(overlap) * dx
    
    return best_h, final_c5, len(best_h)
