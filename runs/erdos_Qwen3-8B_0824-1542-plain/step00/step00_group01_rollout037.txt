```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
import functools

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Set initial n_points
    n_points = 47
    h_initial = initial_h_values.copy()
    
    # Try with a few different n_points to escape local optima
    for n_points in [47, 64, 81, 100, 121, 144, 169, 196, 225]:
        if time.time() - start_time > budget_s - 1:
            break

        # Ensure h_initial is of length n_points
        if len(h_initial) > n_points:
            h_initial = h_initial[:n_points]
        elif len(h_initial) < n_points:
            h_initial = np.pad(h_initial, (0, n_points - len(h_initial)), 'constant')

        # Ensure the integral constraint is satisfied
        integral = np.sum(h_initial) * (2.0 / n_points)
        if np.abs(integral - 1) > 1e-6:
            h_initial = np.random.rand(n_points)
            while np.abs(np.sum(h_initial) * (2.0 / n_points) - 1) > 1e-6:
                h_initial = np.random.rand(n_points)

        # Convert to a function for optimization
        h = cp.Variable(n_points)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]

        # Define the overlap integral for all shifts
        max_overlap = 0
        for k in range(1, n_points):
            # Compute the overlap integral for shift k
            h_shifted = np.roll(h.value, k) if h.value is not None else np.roll(h, k)
            overlap = np.sum(h.value * (1 - h_shifted)) * (2.0 / n_points)
            max_overlap = max(max_overlap, overlap)

        # Objective: minimize the maximum overlap
        # Use a convex relaxation with a tolerance
        tolerance = 1e-3
        objective = cp.Minimize(cp.Max(cp.sum(h * (1 - np.roll(h, k))) * (2.0 / n_points) for k in range(1, n_points)))
        problem = cp.Problem(objective, constraints)
        try:
            result = problem.solve(solver=cp.ECOS, verbose=False)
            if result is not None and result < max_overlap + tolerance:
                h_opt = h.value
                c5_bound = result
                break
        except:
            pass

        # If optimization failed, try a local search
        if h.value is not None:
            h_opt = h.value
            c5_bound = max_overlap
        else:
            h_opt = h_initial
            c5_bound = max_overlap

        # Final evaluation
        final_c5 = evaluate_erdos_solution(h_opt, n_points)
        if final_c5 < 0.380876:
            return (h_opt, final_c5, n_points)

    return (h_initial, evaluate_erdos_solution(h_initial, n_points), n_points)
```
