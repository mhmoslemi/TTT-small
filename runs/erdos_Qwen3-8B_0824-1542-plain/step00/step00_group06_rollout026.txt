```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Example initial solution (n=95 points)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Choose a starting point (could be initial_h_values or a random one)
    n_points = 100  # Start with 100 points
    h = np.random.rand(n_points) * 2 - 1  # Random initial guess
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Normalize to meet the integral constraint
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization loop
    for _ in range(50):  # 50 iterations
        if time.time() - start_time > budget_s:
            break
        
        # Convert to a list for optimization
        h_list = h.tolist()
        
        # Create a CVXPY model
        x = cp.Variable(n_points)
        constraints = [
            cp.sum(x) == n_points / 2,  # Integral constraint
            cp.Minimize(cp.max([cp.sum(cp.multiply(x, 1 - x[k:])) for k in range(1, n_points)]))
        ]
        
        try:
            prob = cp.Problem(cp.Minimize(cp.max([cp.sum(cp.multiply(x, 1 - x[k:])) for k in range(1, n_points)])), constraints)
            prob.solve(solver='ECOS', eps=1e-6)
            
            if prob.status == 'OPTIMAL':
                h = np.clip(x.value, 0, 1)
                c5 = evaluate_erdos_solution(h)
                
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
        except:
            pass
        
        # Try a local search with perturbations
        for _ in range(10):
            perturb = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + perturb, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            c5 = evaluate_erdos_solution(h)
            
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    
    # Try increasing number of points if time allows
    if time.time() - start_time < budget_s - 5:
        n_points = 200
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        for _ in range(20):
            if time.time() - start_time > budget_s:
                break
            
            h_list = h.tolist()
            x = cp.Variable(n_points)
            constraints = [
                cp.sum(x) == n_points / 2,
                cp.Minimize(cp.max([cp.sum(cp.multiply(x, 1 - x[k:])) for k in range(1, n_points)]))
            ]
            
            try:
                prob = cp.Problem(cp.Minimize(cp.max([cp.sum(cp.multiply(x, 1 - x[k:])) for k in range(1, n_points)])), constraints)
                prob.solve(solver='ECOS', eps=1e-6)
                
                if prob.status == 'OPTIMAL':
                    h = np.clip(x.value, 0, 1)
                    c5 = evaluate_erdos_solution(h)
                    
                    if c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = c5
            except:
                pass
            
            for _ in range(5):
                perturb = np.random.normal(0, 0.05, n_points)
                h = np.clip(h + perturb, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                c5 = evaluate_erdos_solution(h)
                
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
    
    return best_h, best_c5, n_points
```
