```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import random
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(42) * 2 - 1  # Placeholder for initial H values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # Start with 100 points for better resolution
    max_iter = 100  # Limit iterations to stay within budget
    best_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Function to evaluate C5
    def evaluate(h):
        return evaluate_erdos_solution(h)
    
    # Simple gradient descent with momentum to optimize the solution
    def optimize_h(h_start):
        h = h_start.copy()
        learning_rate = 0.01
        momentum = 0.9
        gradient = np.zeros_like(h)
        for _ in range(max_iter):
            grad = np.gradient(evaluate(h))
            gradient = momentum * gradient - learning_rate * grad
            h += gradient
            h = np.clip(h, 0, 1)
            # Ensure integral constraint is met
            h /= (np.sum(h) / (n_points / 2))
            current_c5 = evaluate(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            if time.time() - start_time > budget_s - 1:
                break
        return best_h, best_c5
    
    # Run optimization
    best_h, best_c5 = optimize_h(best_h)
    
    # Refinement: try a few random perturbations
    for _ in range(5):
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = best_h + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb /= (np.sum(h_perturb) / (n_points / 2))
        current_c5 = evaluate(h_perturb)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_perturb
    
    # Final refinement: optimize with a small learning rate
    for _ in range(20):
        grad = np.gradient(evaluate(best_h))
        best_h -= 0.001 * grad
        best_h = np.clip(best_h, 0, 1)
        best_h /= (np.sum(best_h) / (n_points / 2))
        current_c5 = evaluate(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
    
    # Ensure the solution meets the constraint
    best_h /= (np.sum(best_h) / (n_points / 2))
    
    # Ensure we don't exceed the budget
    if time.time() - start_time > budget_s - 1:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
```
