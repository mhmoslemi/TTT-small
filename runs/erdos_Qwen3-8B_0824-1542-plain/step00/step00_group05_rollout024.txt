import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial h values
    n_points = 100  # Start with 100 samples
    h = initial_h_values[:n_points]  # Use initial construction for starting point
    
    # Ensure sum(h) * dx = 1, where dx = 2.0 / n_points
    dx = 2.0 / n_points
    h = np.array(h, dtype=np.float64)
    h /= (h.sum() * dx)  # Normalize to ensure the integral is 1
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        h = np.concatenate([h, np.zeros_like(h)])  # Extend for full convolution
        correlation = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlation) * dx
        return c5
    
    # Function to compute the overlap integral for a given h
    def objective(h):
        return compute_c5(h)
    
    # Optimization using L-BFGS-B
    def optimize_h(h_initial, max_iter=100):
        h = h_initial.copy()
        for _ in range(max_iter):
            # Compute gradient numerically
            grad = np.zeros_like(h)
            eps = 1e-6
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_minus = h.copy()
                h_minus[i] -= eps
                grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
            
            # Gradient descent step
            h -= 0.01 * grad
            
            # Ensure constraints
            h = np.clip(h, 0, 1)
            h /= (h.sum() * dx)  # Normalize to ensure the integral is 1
            
            # Check if we're within budget
            if time.time() - start_time > budget_s - 1:
                return h
        
        return h
    
    # Run the optimizer
    best_h = optimize_h(h)
    best_c5 = compute_c5(best_h)
    
    # Try increasing the number of points for better accuracy
    n_points = min(200, int(1000 * (best_c5 / 0.380876)))
    h = best_h.copy()
    
    # Re-normalize for the new number of points
    dx = 2.0 / n_points
    h = np.linspace(0, 2, n_points)
    h = np.interp(h, np.linspace(0, 2, len(best_h)), best_h)
    h = np.clip(h, 0, 1)
    h /= (h.sum() * dx)  # Normalize to ensure the integral is 1
    
    # Refine using convex optimization
    n_points = 200
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    h = cp.Variable(n_points)
    constraints = [cp.sum(h * dx) == 1, 0 <= h, h <= 1]
    c5 = cp.Maximize(cp.sum(cp.multiply(h, 1 - cp.roll(h, 1)) * dx))
    
    # Solve the optimization problem
    prob = cp.Problem(c5, constraints)
    prob.solve(solver='ECOS', verbose=False)
    
    # Check if we're within budget
    if time.time() - start_time > budget_s - 1:
        return best_h.tolist(), best_c5, len(best_h)
    
    # Evaluate the final result
    final_c5 = compute_c5(h.value)
    
    # Return the best solution
    return h.value.tolist(), final_c5, n_points
