        h = cp.Variable(n_points, lower=0, upper=1)

        constraints = [cp.sum(h) == n_points / 2]
        
        # Objective function: minimize the maximum overlap integral
        # We'll use a convex relaxation by minimizing the maximum over all possible k
        overlap = []

            # Compute overlap for shift k
            h_shift = np.roll(h.value, k) if hasattr(h, 'value') else np.roll(h, k)

            overlap.append(overlap_val)

        # We add a constraint that all overlap integrals are <= c
        c = cp.Variable()
        constraints += [cp.sum(h * (1 - np.roll(h, k))) * dx <= c for k in range(1, n_points)]
        objective = cp.Minimize(c)

        problem = cp.Problem(objective, constraints)
