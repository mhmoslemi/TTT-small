import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(70) * 2 - 1  # Placeholder for initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Define number of points
    n_points = 100  # A reasonable number for speed and accuracy
    
    # Define dx
    dx = 2.0 / n_points
    
    # Initial guess
    h_initial = np.zeros(n_points)
    for i in range(n_points):
        h_initial[i] = 0.5  # Start with a flat function
    
    # Define the objective function
    def objective_function(h):
        # Compute the overlap integral for all k
        corr = np.correlate(h, 1 - h, mode="full")
        # Compute the maximum value
        c5 = np.max(corr) * dx
        return c5
    
    # Define constraints
    def constraints_function(h):
        # Constraint: integral of h is 1
        integral = np.sum(h) * dx
        return integral - 1
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Set up the optimization problem
    # We use a local optimization method (L-BFGS-B) for speed
    # And use a wrapper to evaluate the function and constraints
    
    # Run the optimization
    result = opt.minimize(
        fun=objective_function,
        x0=h_initial,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    # If optimization did not converge, try a different method
    if result.success:
        h_opt = result.x
    else:
        # Use a global optimization method if needed (e.g., differential evolution)
        result = opt.differential_evolution(
            objective_function,
            bounds=bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-8
        )
        h_opt = result.x
    
    # Evaluate the final C5 value
    c5_bound = evaluate_erdos_solution(h_opt)
    
    # Ensure the integral is 1
    integral = np.sum(h_opt) * dx
    if abs(integral - 1) > 1e-6:
        # Adjust the function slightly to satisfy the constraint
        scaling_factor = 1 / integral
        h_opt = h_opt * scaling_factor
    
    # Final evaluation
    c5_bound = evaluate_erdos_solution(h_opt)
    
    # Stop if time is up
    if time.time() - start_time > budget_s - 1:
        return h_opt, c5_bound, n_points
    
    # Optionally, refine further with a local optimizer
    result_refined = opt.minimize(
        fun=objective_function,
        x0=h_opt,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    h_opt = result_refined.x
    c5_bound = evaluate_erdos_solution(h_opt)
    integral = np.sum(h_opt) * dx
    if abs(integral - 1) > 1e-6:
        scaling_factor = 1 / integral
        h_opt = h_opt * scaling_factor
    c5_bound = evaluate_erdos_solution(h_opt)
    
    return h_opt, c5_bound, n_points
