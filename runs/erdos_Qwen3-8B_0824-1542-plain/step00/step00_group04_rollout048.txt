```python
import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Assuming this is preloaded

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial number of points
    n_points = 100
    max_n_points = 500  # We can increase if time permits
    
    # Initial h based on provided values
    h = initial_h_values.copy()
    h = h[:n_points]  # Ensure we don't exceed n_points
    
    # Normalize to satisfy the integral constraint
    h /= (np.sum(h) / (n_points / 2))
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Function to evaluate and return the C5 value
    def evaluate(h):
        h = h.copy()
        h /= (np.sum(h) / (n_points / 2))
        return evaluate_erdos_solution(h)
    
    # Optimization: use differential evolution to search for better h
    bounds = [(0, 1) for _ in range(n_points)]
    result = optimize.differential_evolution(evaluate, bounds, tol=1e-5, maxiter=100, disp=False)
    
    # Update best_h if improvement is found
    if result.fun < best_c5:
        best_h = result.x
        best_c5 = result.fun
    
    # Try increasing n_points if within budget
    if time.time() - start_time < budget_s - 5:
        n_points = min(max_n_points, int(n_points * 1.5))
        h = np.random.rand(n_points)
        h /= (np.sum(h) / (n_points / 2))
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5
    
    # Final check with the best solution
    best_h /= (np.sum(best_h) / (n_points / 2))
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
```
