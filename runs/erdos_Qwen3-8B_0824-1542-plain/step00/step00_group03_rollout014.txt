import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial construction from the provided variable
    initial_h_values = np.array(initial_h_values)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def evaluate_c5(h):
        h = h.reshape(-1)
        overlap = correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def optimize_h(n_points):
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        objective = cp.Minimize(0)  # We will evaluate the objective separately
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, max_iter=100, time_limit=10)
        except:
            prob.solve(solver=cp.ECOS, max_iter=100, time_limit=10)
        if prob.status not in ["OPTIMAL", "OPTIMAL_INACCURATE"]:
            return None
        h_values = h.value
        return h_values
    
    best_h = initial_h_values.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Try optimizing with different n_points
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 5:
            break
        try:
            h_opt = optimize_h(n_points)
            if h_opt is not None:
                current_c5 = evaluate_c5(h_opt)
                if current_c5 < best_c5:
                    best_h = h_opt
                    best_c5 = current_c5
        except:
            continue
    
    return best_h, best_c5, len(best_h)
