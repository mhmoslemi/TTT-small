```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
from functools import partial
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))  # placeholder
initial_h_values = np.random.rand(47)  # placeholder, actual initial values would be provided

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for faster evaluation
    
    # Attempt to find a good starting point from the initial_h_values
    initial_h = initial_h_values.copy()
    initial_h = initial_h[:n_points]  # Trim or pad to match n_points
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2)  # Ensure integral is 1
    while True:
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(initial_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = initial_h.copy()
        
        # Try a few random perturbations
        for _ in range(5):
            perturbation = np.random.normal(0, 0.05, n_points)
            perturbed_h = initial_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (n_points / 2)  # Ensure integral is 1
            current_c5 = evaluate_erdos_solution(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = perturbed_h.copy()
        
        # Check if time is up
        if time.time() - start_time > budget_s - 1:
            break
        
        # Try increasing n_points if time allows and best_c5 hasn't improved
        if n_points < 500 and best_c5 >= 0.380876:
            n_points += 50
            initial_h = best_h.copy()
            initial_h = np.clip(initial_h, 0, 1)
            initial_h = initial_h * (n_points / 2)  # Ensure integral is 1
    
    # Use CVXPY to perform more structured optimization
    if n_points < 500:
        h = cp.Variable(n_points)
        objective = cp.Maximize(cp.norm(signal.correlate(h.value, 1 - h.value, mode="full")) * (2.0 / n_points))
        constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            best_h = h.value
            best_c5 = evaluate_erdos_solution(best_h)
        except:
            pass
    
    # Final check
    best_c5 = evaluate_erdos_solution(best_h)
    return best_h, best_c5, n_points
```
