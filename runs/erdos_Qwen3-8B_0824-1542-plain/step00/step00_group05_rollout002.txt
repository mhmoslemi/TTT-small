import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 200  # Start with 200 samples for balance between speed and resolution
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points])  # Start from the initial construction
    h = h[:n_points]  # Ensure it's of correct length
    h = np.clip(h, 0, 1)  # Ensure all values are within [0, 1]
    
    # Check initial constraint
    if np.abs(np.sum(h) * dx - 1) > 1e-8:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * n_points / 2
        h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    best_time = 0

    # Early stopping if initial solution is already under the target
    if best_c5 < 0.380876:
        return (best_h, best_c5, n_points)

    start_time = time.time()

    # Optimization: hill-climb with random perturbations
    for _ in range(100):  # Max 100 iterations
        if time.time() - start_time > budget_s - 1:  # Leave 1s buffer
            break

        # Perturb h
        perturb = np.random.rand(n_points) - 0.5
        perturb *= 0.1  # Small perturbation
        h = h + perturb
        h = np.clip(h, 0, 1)
        
        # Enforce constraint: sum(h) * dx = 1
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate
        c5 = evaluate_erdos_solution(h)
        
        # Update best solution
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # If we can afford to increase resolution, try with more points
    if time.time() - start_time < budget_s - 5:
        n_points = 400
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h / np.sum(h) * n_points / 2
        h = np.clip(h, 0, 1)
        
        best_c5 = evaluate_erdos_solution(h)
        best_h = h.copy()
        
        for _ in range(50):  # Max 50 iterations with higher resolution
            if time.time() - start_time > budget_s - 1:
                break
            
            perturb = np.random.rand(n_points) - 0.5
            perturb *= 0.1
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            c5 = evaluate_erdos_solution(h)
            
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

    # Final check
    if best_c5 < 0.380876:
        return (best_h, best_c5, n_points)
    else:
        return (best_h, best_c5, n_points)
