import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different numbers of points (smaller is faster)
    n_points_list = [100, 150, 200, 250, 300]
    best_h = None
    best_c5 = float('inf')
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s - 5:
            break
        
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Initialize h with the provided initial solution or uniform
        h = initial_h_values.copy()
        h = np.interp(x, np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        
        # Ensure the integral constraint is satisfied
        h_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-5:
            # Scale h to meet the constraint
            h = h * (target_sum / h_sum)
        
        # Function to evaluate the overlap integral
        def evaluate_overlap(h):
            h = np.clip(h, 0, 1)
            h = h.reshape(-1)
            # Compute 1 - h
            one_minus_h = 1.0 - h
            # Compute the convolution
            corr = np.correlate(h, one_minus_h, mode="full")
            # Take the maximum value
            max_corr = np.max(corr)
            # Multiply by dx to get the integral
            return max_corr * dx
        
        # Optimization: minimize the overlap integral subject to sum(h) = n_points/2
        # Use a gradient-based optimization method
        def objective_func(h_vec):
            h = h_vec.reshape(-1)
            return evaluate_overlap(h)
        
        def constraint_func(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        # Set up the optimization
        h_vec = np.copy(h)
        opt_result = optimize.minimize(
            objective_func,
            h_vec,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        
        # Get the best solution
        best_h_current = opt_result.x
        current_c5 = evaluate_overlap(best_h_current)
        
        # Update the best solution if this one is better
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h_current
        
        # Early stopping if we've met the target
        if best_c5 <= 0.380876:
            break
    
    # Evaluate the best solution
    final_c5 = evaluate_overlap(best_h)
    
    return (best_h, final_c5, len(best_h))
