import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')  # Example path, assumes initial_h_values.npy exists

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try multiple starting points to avoid local minima
    n_points_list = [45, 60, 80, 100]  # Try different numbers of samples
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    for n_points in n_points_list:
        # Check if we're over budget
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate initial guess based on initial_h_values and scale to n_points
        if n_points == len(initial_h_values):
            h = initial_h_values.copy()
        else:
            h = np.zeros(n_points)
            # Use initial_h_values as a template and resample
            # Map initial_h_values to [0, 2] range and interpolate
            initial_x = np.linspace(0, 2, len(initial_h_values))
            h_x = np.linspace(0, 2, n_points)
            h = np.interp(h_x, initial_x, initial_h_values)
        
        # Adjust to satisfy the integral constraint
        # Normalize h so that sum(h) * dx = 1 => sum(h) = n_points / 2
        target_sum = n_points / 2
        current_sum = np.sum(h)
        if current_sum != 0:
            h = h * (target_sum / current_sum)
        
        # Define the overlap integral
        def compute_c5(h):
            dx = 2.0 / n_points
            h_shifted = np.zeros(2 * n_points)
            h_shifted[:n_points] = h
            result = signal.correlate(h, 1 - h, mode="full")
            c5 = np.max(result) * dx
            return c5
        
        # Use a local optimization method to minimize c5
        def objective(x):
            h = x.reshape(n_points)
            dx = 2.0 / n_points
            h_shifted = np.zeros(2 * n_points)
            h_shifted[:n_points] = h
            result = signal.correlate(h, 1 - h, mode="full")
            c5 = np.max(result) * dx
            return c5
        
        # Bounds for optimization
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Initial guess
        x0 = h
        
        # Use L-BFGS-B for optimization
        result = optimize.minimize(
            objective,
            x0,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6},
            jac=None  # No analytical derivative, use numerical
        )
        
        # Apply the optimized h
        optimized_h = result.x
        optimized_h = np.clip(optimized_h, 0.0, 1.0)
        # Adjust to satisfy the integral constraint
        target_sum = n_points / 2
        current_sum = np.sum(optimized_h)
        if current_sum != 0:
            optimized_h = optimized_h * (target_sum / current_sum)
        
        # Compute the final C5 value
        c5 = compute_c5(optimized_h)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = optimized_h
            best_n = n_points
    
    # Return the best solution found
    return best_h, best_c5, best_n
