import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)

    # Initial solution from the provided starting point
    n_points_initial = len(initial_h_values)
    h_initial = initial_h_values.copy()
    dx_initial = 2.0 / n_points_initial
    sum_h_initial = np.sum(h_initial) * dx_initial
    if abs(sum_h_initial - 1.0) > 1e-6:
        h_initial = initial_h_values.copy()
        h_initial[0] = h_initial[0] + (1.0 - sum_h_initial) / dx_initial
        h_initial = np.clip(h_initial, 0.0, 1.0)

    # Try a few smaller n_points to find a better solution
    n_points_list = [95, 128, 192, 256, 384, 512, 768]
    best_c5 = float('inf')
    best_h = h_initial
    best_n_points = n_points_initial

    for n_points in n_points_list:
        # Compute dx
        dx = 2.0 / n_points

        # Generate initial guesses based on the initial solution
        h = np.zeros(n_points)
        for i in range(n_points):
            # Map initial h to current grid
            x_initial = i * dx_initial
            x_current = i * dx
            h[i] = np.interp(x_current, np.linspace(0, 2, n_points_initial), h_initial)

        # Normalize h to ensure the integral is 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0.0, 1.0)

        # Function to compute C5 value
        def compute_c5(h_vec):
            h = h_vec
            h_shifted = np.concatenate((h[-1:], h[:-1]))
            cross_corr = np.correlate(h, 1 - h_shifted, mode='full')
            c5 = np.max(cross_corr) * dx
            return c5

        # Optimization: try to minimize C5 by adjusting h
        # Use a simple optimization with gradient descent
        def optimize_h():
            nonlocal best_c5, best_h, best_n_points
            time_start = time.time()
            tolerance = 1e-4
            max_iterations = 500
            learning_rate = 0.01

            h = h.copy()
            for _ in range(max_iterations):
                # Compute current C5
                current_c5 = compute_c5(h)

                # Compute the gradient (approximate)
                grad = np.zeros_like(h)
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += 1e-4
                    c5_plus = compute_c5(h_plus)
                    h_minus = h.copy()
                    h_minus[i] -= 1e-4
                    c5_minus = compute_c5(h_minus)
                    grad[i] = (c5_plus - c5_minus) / (2e-4)

                # Update h
                h = h - learning_rate * grad
                h = np.clip(h, 0.0, 1.0)
                h = h * (n_points / 2) / np.sum(h)
                h = np.clip(h, 0.0, 1.0)

                # Check for convergence
                if current_c5 < best_c5 - tolerance:
                    best_c5 = current_c5
                    best_h = h.copy()
                    best_n_points = n_points
                    if best_c5 < 0.380876:
                        break

                # Stop if time is up
                if time.time() - time_start > budget_s - 1:
                    break

        optimize_h()

    # Check the best solution found
    final_h = best_h
    final_c5 = best_c5
    final_n_points = best_n_points

    # Final evaluation
    final_c5 = evaluate_erdos_solution(final_h, n_points=final_n_points)
    return final_h, final_c5, final_n_points
