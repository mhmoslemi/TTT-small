import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=60, **kwargs):
    # Initial solution from the provided initial_h_values
    initial_h = initial_h_values
    n_points_initial = len(initial_h)
    n_points = n_points_initial  # Start with the same number of points
    
    # Define a function to evaluate the overlap integral
    def evaluate_overlap(h_values, dx):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Function to optimize h_values
    def optimize_overlap(h_values, dx):
        h = cp.Variable(len(h_values), name='h')
        constraints = [cp.sum(h) == len(h_values) / 2]
        constraints += [0 <= h]
        constraints += [h <= 1]
        
        # Objective: minimize the maximum overlap
        corr = np.correlate(h.value if h.value is not None else np.zeros_like(h), 1 - h.value if h.value is not None else np.zeros_like(h), mode="full")
        max_overlap = cp.Max(corr * dx)
        
        # Instead of direct optimization over max_overlap, we minimize the maximum over all shifts
        # For simplicity, we'll minimize the maximum over a set of shifts
        # Here we compute the overlap for shifts 0 to len(h)-1
        corr_values = np.correlate(h, 1 - h, mode="full")
        max_overlap_expr = cp.Max(corr_values * dx)
        
        prob = cp.Problem(cp.Minimize(max_overlap_expr), constraints)
        prob.solve(solver=cp.CVXPY_MI, max_iters=100, time_limit=10)  # Use a solver that handles MI

        return h.value if prob.status == 'optimal' else h_values

    # Function to refine the solution using a grid search
    def refine_solution(h_values, dx, max_iterations=5):
        for _ in range(max_iterations):
            h = optimize_overlap(h_values.copy(), dx)
            current_c5 = evaluate_overlap(h, dx)
            if current_c5 < evaluate_overlap(h_values, dx):
                h_values = h
            else:
                break
        return h_values

    # Starting point
    h_values = initial_h.copy()
    n_points = len(h_values)
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Try a few refinements
    for _ in range(3):
        h_values = refine_solution(h_values, dx)
        current_c5 = evaluate_overlap(h_values, dx)
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    final_c5 = evaluate_overlap(h_values, dx)
    
    return h_values, final_c5, n_points
