import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Define the number of points (n_points)
    n_points = 100  # Start with 100 points, adjust if needed
    dx = 2.0 / n_points
    max_shift = int(2 / dx)  # Maximum shift to consider for the correlation

    def compute_overlap(h):
        h = np.array(h)
        # Normalize h to ensure the integral condition is met
        h_norm = h / np.sum(h) * n_points / 2
        # Compute (1 - h) for the convolution
        one_minus_h = 1 - h_norm
        # Compute the full convolution and get the max overlap
        conv = signal.correlate(h_norm, one_minus_h, mode='full')
        # Since we are interested in shifts k in [0, 2], we take the relevant part of the convolution
        # and multiply by dx to get the integral
        c5_candidate = np.max(conv[:max_shift + 1] * dx)
        return c5_candidate

    # Initial guess from the provided initial_h_values
    h_initial = initial_h_values.copy()
    h_initial = h_initial[:n_points]  # Ensure it's of the correct length

    # Optimization: minimize the maximum overlap integral using a gradient-based optimizer
    # We'll use the scipy.optimize.minimize function with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(
        compute_overlap,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        options={'ftol': 1e-8, 'maxiter': 500}
    )

    # Extract the best solution found
    best_h = result.x
    c5_bound = compute_overlap(best_h)

    # Return the best solution, the c5 bound, and the number of points
    return (best_h, c5_bound, n_points)
