```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    
    # Optimization parameters
    max_iterations = 50
    convergence_tolerance = 1e-3
    max_time = budget_s
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    for _ in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > max_time:
            break
        
        # Define variables for optimization
        x = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= 1
        ]
        
        # Objective function: minimize max correlation
        # We'll use a convex relaxation by minimizing the maximum over a set of linear expressions
        # Approximate max by optimizing over a set of correlation values
        # Evaluate all possible shifts k where the overlap is non-zero
        shifts = np.arange(-n_points + 1, n_points)
        corr_vals = []
        for k in shifts:
            if k < 0:
                shifted = np.roll(x, -k)
            else:
                shifted = np.roll(x, k)
            corr = cp.sum(x * (1 - shifted))
            corr_vals.append(corr)
        
        # Objective: minimize the maximum correlation
        obj = cp.Minimize(cp.Max(corr_vals))
        
        # Solve the optimization problem
        try:
            prob = cp.Problem(obj, constraints)
            prob.solve(solver='ECOS', eps=1e-6, verbose=False)
        except:
            # If ECOS fails, try another solver
            try:
                prob.solve(solver='SCIPY', eps=1e-6, verbose=False)
            except:
                prob.solve(solver='MOSEK', eps=1e-6, verbose=False)
        
        if prob.status != 'optimal':
            break
        
        # Update h
        h = x.value
        
        # Evaluate the current solution
        current_c5 = evaluate_erdos_solution(h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Check for convergence
        if abs(current_c5 - best_c5) < convergence_tolerance:
            break
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
```
