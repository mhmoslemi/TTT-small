```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Starting with 100 points for efficiency

    def evaluate(h):
        dx = 2.0 / n_points
        h = np.array(h).reshape(-1)
        h = np.pad(h, (0, n_points - len(h)), 'constant')  # Ensure length is n_points
        h = np.clip(h, 0, 1)
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def optimize_h():
        nonlocal best_c5, best_h
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Ensure the integral is 1
        for _ in range(100):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            current_c5 = evaluate(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                if best_c5 < 0.380876:
                    print(f"New best C5: {best_c5}")
                    # Early exit if we've found a good solution
                    if time.time() - start_time > budget_s - 10:
                        return

    optimize_h()

    # Try increasing n_points if time allows
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        n_points = 200
        print(f"Trying with {n_points} points...")
        def evaluate_higher_res(h):
            dx = 2.0 / n_points
            h = np.array(h).reshape(-1)
            h = np.pad(h, (0, n_points - len(h)), 'constant')
            h = np.clip(h, 0, 1)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5

        def optimize_higher_res():
            nonlocal best_c5, best_h
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            for _ in range(50):
                h = np.random.rand(n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                current_c5 = evaluate_higher_res(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    if best_c5 < 0.380876:
                        print(f"New best C5: {best_c5}")
                        if time.time() - start_time > budget_s - 10:
                            return

        optimize_higher_res()

    if best_h is None:
        best_h = initial_h_values.copy()
        best_c5 = evaluate(best_h)

    return best_h, best_c5, n_points
```
