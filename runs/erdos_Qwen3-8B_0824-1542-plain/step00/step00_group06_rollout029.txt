```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution
    h_vals = initial_h_values.copy()
    n_points = len(h_vals)
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h_vals, dx)
    best_h = h_vals.copy()
    best_c5 = current_c5
    
    # Try a few different starting points
    for i in range(3):
        # Perturb the initial solution slightly
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_vals = initial_h_values + perturbation
        h_vals = np.clip(h_vals, 0, 1)
        
        # Ensure the total integral is 1
        total_integral = np.sum(h_vals) * dx
        scaling_factor = 1.0 / total_integral
        h_vals = h_vals * scaling_factor
        
        # Evaluate
        current_c5 = evaluate_erdos_solution(h_vals, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h_vals.copy()
            best_c5 = current_c5
    
    # Try optimization with CVXOPT
    def objective(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization with bounds
    h = cp.Variable(n_points)
    constraints = [
        cp.sum(h) == n_points / 2,
        0 <= h,
        h <= 1
    ]
    prob = cp.Problem(cp.Minimize(objective(h)), constraints)
    
    # Try solving with different solvers
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status == 'optimal':
            h_vals = h.value
            current_c5 = evaluate_erdos_solution(h_vals, 2.0 / n_points)
            if current_c5 < best_c5:
                best_h = h_vals.copy()
                best_c5 = current_c5
    except:
        pass
    
    # Try with smaller n_points
    for n in [100, 200, 300]:
        h_vals = np.random.rand(n) * 2 - 1
        h_vals = np.clip(h_vals, 0, 1)
        total_integral = np.sum(h_vals) * (2.0 / n)
        scaling_factor = 1.0 / total_integral
        h_vals = h_vals * scaling_factor
        
        current_c5 = evaluate_erdos_solution(h_vals, 2.0 / n)
        if current_c5 < best_c5:
            best_h = h_vals.copy()
            best_c5 = current_c5
    
    # Check elapsed time and return
    elapsed = time.time() - start_time
    if elapsed >= budget_s - 1:
        return (best_h, best_c5, n_points)
    else:
        return (best_h, best_c5, n_points)
```
