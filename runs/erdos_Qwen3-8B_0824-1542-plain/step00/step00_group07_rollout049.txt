```python
import numpy as np
import scipy.optimize as optimize
import time
import sys
import copy
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points, faster to compute and optimize

    # Helper function to compute the overlap integral
    def compute_c5(h):
        h = np.concatenate([h, np.zeros(n_points)])
        kernel = 1 - h[:n_points]
        corr = np.correlate(h, kernel, mode="full")
        dx = 2.0 / n_points
        return np.max(corr * dx)

    # Initial solution
    h_initial = initial_h_values
    h_initial = h_initial[:n_points]
    current_c5 = compute_c5(h_initial)
    best_c5 = current_c5
    best_h = h_initial.copy()

    # Try increasing n_points to get better resolution
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 1:
            break
        print(f"Trying n_points = {n_points}")
        h = np.zeros(n_points)
        # Start from a uniform distribution but adjust to satisfy the integral constraint
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)

        # Optimization using scipy.optimize.minimize with bounds
        def objective(x):
            h = x.copy()
            dx = 2.0 / n_points
            h = np.concatenate([h, np.zeros(n_points)])
            kernel = 1 - h[:n_points]
            corr = np.correlate(h, kernel, mode="full")
            return np.max(corr * dx)

        def constraint(x):
            return np.sum(x) - (n_points / 2)

        x0 = h.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': constraint})
        result = optimize.minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-5)
        if result.success:
            h_result = result.x
            c5_result = compute_c5(h_result)
            if c5_result < best_c5:
                best_c5 = c5_result
                best_h = h_result.copy()
                print(f"Found better solution at n_points = {n_points}, C5 = {best_c5}")
        else:
            print(f"Optimization failed for n_points = {n_points}")

    # Try CVXPY optimization for better solution
    if time.time() - start_time < budget_s - 1:
        print("Trying CVXPY optimization")
        n_points = 500
        h = cp.Variable(n_points)
        dx = 2.0 / n_points
        # Objective function: maximize min overlap
        # We use a trick to minimize the maximum overlap by reformulating
        slack = cp.Variable(n_points * (n_points + 1) // 2)
        obj = cp.Minimize(cp.sum(slack))
        constraints = []
        for i in range(n_points):
            for j in range(i, n_points):
                offset = (j - i) % n_points
                overlap = cp.sum(cp.multiply(h[i:], 1 - h[(i + offset):]))
                constraints.append(slack[i * n_points + j] >= overlap)
                constraints.append(slack[i * n_points + j] >= 0)

        # Integral constraint
        constraints.append(cp.sum(h) == n_points / 2)
        # Bounding constraints
        constraints.append(cp.SOC(h >= 0))
        constraints.append(cp.SOC(h <= 1))

        # Solve
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.MOSEK, eps=1e-5)
            if prob.status == 'optimal':
                h_result = h.value
                c5_result = compute_c5(h_result)
                if c5_result < best_c5:
                    best_c5 = c5_result
                    best_h = h_result.copy()
                    print(f"Found better solution with CVXPY, C5 = {best_c5}")
        except:
            print("CVXPY optimization failed")

    # Final check using a different initial guess
    if time.time() - start_time < budget_s - 1:
        print("Trying a different initial guess")
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        result = optimize.minimize(objective, h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-5)
        if result.success:
            h_result = result.x
            c5_result = compute_c5(h_result)
            if c5_result < best_c5:
                best_c5 = c5_result
                best_h = h_result.copy()
                print(f"Found better solution with different initial guess, C5 = {best_c5}")

    # Final evaluation
    print(f"Final best C5 = {best_c5}")
    return (best_h, best_c5, n_points)
```
