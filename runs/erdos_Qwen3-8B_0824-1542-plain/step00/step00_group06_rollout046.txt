```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial guess
    n_points = len(initial_h_values)
    h_initial = initial_h_values.copy()
    h_initial = np.clip(h_initial, 0, 1)
    
    # Define the evaluation function
    def evaluate(h):
        h = np.clip(h, 0, 1)
        c5 = evaluate_erdos_solution(h)
        return c5
    
    # Define a function to minimize the maximum overlap
    def objective_func(h):
        return evaluate(h)
    
    # Define a function to return the current best
    def get_current_best(h):
        return h, evaluate(h)
    
    # Use the initial guess as the starting point
    best_h, best_c5 = get_current_best(h_initial)
    
    # Try a few random perturbations to escape local minima
    for _ in range(3):
        perturbation = np.random.normal(0, 0.1, n_points)
        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_c5 = evaluate(h_perturbed)
        if current_c5 < best_c5:
            best_h, best_c5 = h_perturbed, current_c5
    
    # Try a fine-tuning optimization
    def objective(h):
        return evaluate(h)
    
    # Set up the optimization problem
    h = cp.Variable(n_points)
    constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
    
    # Use a local solver for fast optimization
    prob = cp.Problem(cp.Minimize(objective(h)), constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iter=100, eps=1e-6, time_limit=0.5 * (budget_s - (time.time() - start_time)))
        if prob.status == 'optimal':
            best_h = h.value
            best_c5 = evaluate(best_h)
    except:
        pass
    
    # Check if we have a better solution
    if evaluate(best_h) < best_c5:
        best_c5 = evaluate(best_h)
    
    # Ensure the solution is valid
    best_h = np.clip(best_h, 0, 1)
    
    # Final check
    final_c5 = evaluate(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    # Output the result
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print(f"Time budget exceeded, returning best solution so far")
    
    return (best_h, best_c5, len(best_h))
```
