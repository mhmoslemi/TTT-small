import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

initial_h_values = np.load('initial_h_values.npy')  # Placeholder for initial values
n_points_initial = len(initial_h_values)

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    
    # Start with initial solution
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    current_c5 = compute_c5(h, dx)
    
    # Define objective function for optimization
    def objective_function(h):
        return compute_c5(h, dx)
    
    # Define constraints for optimization
    def constraints(h):
        return [np.sum(h) == n_points / 2]
    
    # Attempt to optimize with different starting points
    for _ in range(5):  # Try multiple starting points
        # Perturb the initial solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = np.clip(h + perturbation, 0, 1)
        
        # Compute starting c5
        c5_start = compute_c5(h_perturbed, dx)
        
        # If perturbed solution is better, use it
        if c5_start < current_c5:
            h = h_perturbed
            current_c5 = c5_start
        
        # Set up the optimization problem
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(objective_function(h_var))
        constraints = [cp.sum(h_var) == n_points / 2, 0 <= h_var, h_var <= 1]
        
        # Solve the optimization problem
        try:
            prob = cp.Problem(objective, constraints)
            result = prob.solve(solver='ECOS', verbose=False)
            if result is not None:
                h_opt = h_var.value
                c5_opt = compute_c5(h_opt, dx)
                if c5_opt < current_c5:
                    h = h_opt
                    current_c5 = c5_opt
        except:
            pass
        
        # Check if time is up
        if time.time() - start_time > budget_s:
            break
    
    # Return the best solution
    return (h, current_c5, n_points)

def compute_c5(h, dx):
    # Compute the maximum correlation
    corr = signal.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5
