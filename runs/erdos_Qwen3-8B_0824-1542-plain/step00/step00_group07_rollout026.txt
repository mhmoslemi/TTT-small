```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.correlate(h_values, 1 - h_values, mode="full") * (2.0 / len(h_values))
initial_h_values = np.array([0.5 if i % 2 == 0 else 0.0 for i in range(80)])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    n_points = len(best_h)
    elapsed = 0

    # Try improving from initial solution
    for _ in range(10):
        if elapsed >= budget_s - 1:
            break
        # Try random perturbation
        h = best_h.copy()
        idx = np.random.randint(0, len(h))
        h[idx] = np.random.uniform(0, 1)
        # Enforce integral constraint
        total = np.sum(h)
        h = h * (n_points / 2) / total
        # Evaluate
        c5 = evaluate_erdos_solution(h)
        max_c5 = np.max(c5)
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h
        elapsed = time.time() - start_time

    # Try a refined approach with gradient descent
    if elapsed < budget_s - 1:
        n_points = 100
        h = np.zeros(n_points)
        h[:n_points // 2] = 1.0
        h = h * (n_points / 2) / np.sum(h)
        h = h + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        best_h = h
        best_c5 = evaluate_erdos_solution(best_h)
        best_c5 = np.max(best_c5)
        n_points = len(best_h)
        elapsed = time.time() - start_time

    # Try a more systematic optimization with CVXPY
    if elapsed < budget_s - 1:
        n_points = 100
        h = cp.Variable(n_points)
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]
        # Objective is to minimize the maximum overlap
        overlap = cp.sum(cp.multiply(h, 1 - cp.roll(h, 1))) * (2.0 / n_points)
        obj = cp.Minimize(overlap)
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == 'optimal':
                best_h = np.round(h.value, 5)
                best_h = np.clip(best_h, 0, 1)
                best_h = best_h * (n_points / 2) / np.sum(best_h)
                best_c5 = evaluate_erdos_solution(best_h)
                best_c5 = np.max(best_c5)
                n_points = len(best_h)
        except:
            pass
        elapsed = time.time() - start_time

    # Final check
    if elapsed < budget_s - 1:
        h = best_h.copy()
        h = h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = evaluate_erdos_solution(h)
        max_c5 = np.max(c5)
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h
        n_points = len(best_h)
        elapsed = time.time() - start_time

    return best_h, best_c5, n_points
```
