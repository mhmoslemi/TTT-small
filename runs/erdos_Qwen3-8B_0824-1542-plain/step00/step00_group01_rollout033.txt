import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds
import copy

# Pre-imported functions
evaluate_erdos_solution = None
initial_h_values = None

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Adjusted to a value less than 1k for speed
    dx = 2.0 / n_points
    initial_h = copy.deepcopy(initial_h_values)
    if len(initial_h) != n_points:
        initial_h = np.random.rand(n_points)
        initial_h = (initial_h * n_points / 2).astype(np.float64)
        initial_h = initial_h / np.sum(initial_h) * n_points / 2

    # Objective function
    def objective(h):
        h = h.reshape(-1)
        h = np.clip(h, 0.0, 1.0)
        h = h / np.sum(h) * n_points / 2
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Constraints
    def constraint_sum(h):
        h = h.reshape(-1)
        return np.sum(h) - n_points / 2

    # Optimization
    bounds = Bounds(0.0, 1.0)
    result = optimize.minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}])

    # Evaluate and return best solution
    h_values = result.x
    c5_bound = objective(h_values)
