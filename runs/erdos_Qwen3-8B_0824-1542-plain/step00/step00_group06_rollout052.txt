```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Assume this is a pre-loaded numpy array with shape (n_points,)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    best_n_points = 0

    # Try different numbers of points
    for n_points in [100, 150, 200, 250, 300, 350, 400]:
        if time.time() - start_time > budget_s:
            break
        
        # Generate initial guess based on initial_h_values
        if n_points < len(initial_h_values):
            scaled_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        else:
            scaled_h = np.tile(initial_h_values, n_points // len(initial_h_values) + 1)[:n_points]
        
        # Normalize to satisfy the integral constraint
        scaled_h = scaled_h * (n_points / 2) / np.sum(scaled_h)
        
        # Define optimization variables
        h = cp.Variable(n_points, nonneg=True)
        constraints = [cp.sum(h) == n_points / 2]
        
        # Define objective function: minimize the maximum overlap
        # Convert h into a function for integral approximation
        def objective_func(h_vals):
            h_vals = np.array(h_vals)
            overlap = correlate(h_vals, 1 - h_vals, mode="full") * (2.0 / n_points)
            return np.max(overlap)
        
        # Use scipy optimization for faster evaluation
        def objective(x):
            return objective_func(x)
        
        # Use L-BFGS-B optimizer
        result = optimize.minimize(
            objective,
            scaled_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-6, 'disp': False}
        )
        
        h_vals = result.x
        current_c5 = evaluate_erdos_solution(h_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_vals
            best_n_points = n_points

    return best_h, best_c5, best_n_points
```
