import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import minimize
from scipy.integrate import simps
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with a moderate number of points
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy() if hasattr(initial_h_values, 'copy') else initial_h_values[:]
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h / (np.sum(initial_h) * dx) * (n_points / 2)

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) * dx) * (n_points / 2)
        h = h[:n_points]
        shift = np.arange(-n_points + 1, n_points)
        max_overlap = 0
        for k in shift:
            if k >= 0:
                corr = correlate(h, 1 - h, mode='full')[k:]
            else:
                corr = correlate(1 - h, h, mode='full')[-k:]
            overlap = simps(corr, dx=dx)
            max_overlap = max(max_overlap, overlap)
        return max_overlap

    def optimize_h(h0):
        result = minimize(objective, h0, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)], options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 100})
        return result.x

    def refine_n_points(n, h):
        new_n = n * 2
        new_dx = 2.0 / new_n
        x = np.linspace(0, 2, new_n)
        h_new = np.interp(x, np.linspace(0, 2, n), h)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / (np.sum(h_new) * new_dx) * (new_n / 2)
        return h_new

    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h

    for n in [100, 200, 400]:
        if time.time() - start_time > budget_s - 10:
            break
        h = optimize_h(initial_h)
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
        h_refined = refine_n_points(n, h)
        h_refined = optimize_h(h_refined)
        current_c5 = objective(h_refined)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_refined

    if time.time() - start_time < budget_s - 10:
        h_final = optimize_h(best_h)
        best_c5 = objective(h_final)
        best_h = h_final

    return best_h, best_c5, n_points
