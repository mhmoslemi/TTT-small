import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    start_time = time.time()
    
    # Define the initial guess with the provided initial_h_values
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if not np.isclose(sum_h, n_points / 2):
        initial_h = initial_h * (n_points / 2) / sum_h

    # Function to compute the overlap integral for a given h
    def compute_c5(h):
        h = np.array(h)
        h = np.concatenate([h, h[:n_points - 1]])  # Extend h for full convolution
        result = np.correlate(h, 1 - h[:n_points], mode="full")
        result = result * dx
        return np.max(result)

    # Optimization: minimize the overlap integral C₅
    # We use a gradient descent approach with simulated annealing
    # to find a better h that minimizes the maximum overlap
    def optimize_h():
        h = initial_h.copy()
        best_h = h.copy()
        best_c5 = compute_c5(h)
        temp = 1.0
        cooling_rate = 0.995
        iterations = 0
        max_iterations = 100000  # Maximum number of iterations
        while temp > 1e-5 and iterations < max_iterations:
            # Perturb the current h
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h = np.clip(h + perturbation, 0, 1)
            # Ensure the integral constraint is satisfied
            h = h * (n_points / 2) / np.sum(h)
            # Compute the C₅ value
            current_c5 = compute_c5(h)
            # Accept the new solution with a probability based on simulated annealing
            if current_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - current_c5) / temp):
                best_h = h.copy()
                best_c5 = current_c5
            # Cool down
            temp *= cooling_rate
            iterations += 1
            # Check elapsed time
            if time.time() - start_time > budget_s - 1:
                break
        return best_h, best_c5

    # Run the optimization
    best_h, best_c5 = optimize_h()
    
    # Evaluate the solution and return it
    return best_h, best_c5, len(best_h)
