def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.zeros_like(h)
    for k in range(1, n_points - 1):
        shifted[k:] = h[:-k]
    overlap_integral = np.max(np.correlate(h, 1 - h, mode="full") * dx)
    return overlap_integral
