import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100  # Start with 100 points for faster computation
    max_iter = 100
    tolerance = 1e-4
    
    # Initial guess from the provided starting point
    h = initial_h_values[:n_points].copy()
    h /= (2.0 / n_points)  # Normalize to satisfy the integral constraint
    h = np.clip(h, 0.0, 1.0)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        c5 = evaluate_erdos_solution(h)
        return c5
    
    # Function to update h with a gradient-based optimization step
    def update_h(h, learning_rate):
        # We use a simple gradient descent approach on a smoothed version of C5
        # Since the function is non-convex, we approximate the gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-6
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            grad[i] = (compute_c5(h_plus) - compute_c5(h_minus)) / (2 * eps)
        
        # Update h with a descent step
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        # Ensure the integral constraint is satisfied
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        return h_new
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Use a learning rate that decreases over time
        learning_rate = 0.1 / (1 + _ / 100)
        h = update_h(h, learning_rate)
        
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Refine with a smaller step size
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        
        learning_rate = 0.05 / (1 + _ / 100)
        h = update_h(h, learning_rate)
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Try perturbing the solution to escape local minima
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        
        perturbation = np.random.normal(0, 0.05, size=len(best_h))
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (n_points / 2.0) / np.sum(perturbed_h)
        
        current_c5 = compute_c5(perturbed_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = perturbed_h.copy()
    
    return best_h, best_c5, n_points
