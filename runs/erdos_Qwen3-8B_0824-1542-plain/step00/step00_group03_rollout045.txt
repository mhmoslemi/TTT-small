```python
import numpy as np
import scipy.optimize as optimize
import time
import multiprocessing
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
import sys
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load('initial_h_70.npy')  # Assuming initial_h_70.npy exists and contains the initial sample

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    
    # Define number of points and initial setup
    n_points = 100  # Start with 100 points
    dx = 2.0 / n_points
    
    # Initial guess: use the provided initial_h_values and scale to n_points
    if len(initial_h_values) != n_points:
        initial_h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    
    # Ensure the initial guess satisfies the constraint sum(h) = n_points / 2
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    
    # Objective function
    def objective(h_values):
        # Ensure h_values are within [0, 1]
        h_values = np.clip(h_values, 0, 1)
        
        # Ensure sum(h_values) = n_points / 2
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        
        # Compute the overlap integral
        correlation = correlate(h_values, 1 - h_values, mode="full")
        c5 = np.max(correlation) * dx
        return c5
    
    # Constraints
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Bounds
    bounds = Bounds(0, 1)
    
    # Optimization
    result = minimize(objective, initial_h_values, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum})
    h_values = result.x
    
    # Compute the final C5 value
    c5_bound = evaluate_erdos_solution(h_values)
    
    # Check if we can try a higher resolution
    if time.time() - start_time < budget_s - 5:
        n_points = 150
        dx = 2.0 / n_points
        initial_h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_values)), h_values)
        initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
        
        result = minimize(objective, initial_h_values, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum})
        h_values = result.x
        c5_bound = evaluate_erdos_solution(h_values)
    
    # Return the best solution found
    return h_values, c5_bound, n_points
```
