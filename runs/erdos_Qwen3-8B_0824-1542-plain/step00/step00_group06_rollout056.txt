import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 256  # We start with a reasonable number of points
    dx = 2.0 / n_points
    h = initial_h_values.copy()[:n_points]  # Use initial construction if available
    if len(h) < n_points:
        h = np.zeros(n_points)
        h[:len(initial_h_values)] = initial_h_values
        h = np.pad(h, (0, n_points - len(initial_h_values)), 'constant')
    
    # Ensure the integral constraint is satisfied
    total_integral = np.sum(h) * dx
    if abs(total_integral - 1.0) > 1e-6:
        # Adjust h to meet the integral constraint
        adjustment = (1.0 - total_integral) / dx
        h += adjustment / n_points
        h = np.clip(h, 0, 1)
    
    # Define the overlap integral function
    def compute_c5(h_values):
        h = h_values.reshape(-1)
        dx = 2.0 / len(h)
        h = np.clip(h, 0, 1)
        kernel = 1 - h
        corr = np.correlate(h, kernel, mode="full")
        return np.max(corr) * dx
    
    # Optimization function using a local optimization method
    def optimize_h(h_initial):
        h = cp.Variable(len(h_initial))
        constraints = [0 <= h, h <= 1]
        constraints.append(cp.sum(h) == len(h_initial) / 2)
        
        # Objective function: minimize the maximum overlap integral
        # Since we're minimizing the max, we can use a convex optimization approach
        # with a convex constraint on the max value
        
        # Define the overlap integral for a given h
        def objective(h_vals):
            h = h_vals.reshape(-1)
            dx = 2.0 / len(h)
            kernel = 1 - h
            corr = np.correlate(h, kernel, mode="full")
            return np.max(corr) * dx
        
        # Since our objective is to minimize the max, we can use a convex optimization
        # with the constraint that the max of the overlap is less than or equal to a variable
        # and then minimize that variable
        
        # Convert to a constrained optimization problem
        t = cp.Variable()
        constraints.append(t >= objective(h))
        obj = cp.Minimize(t)
        
        problem = cp.Problem(obj, constraints)
        result = problem.solve(solver=cp.CVXOPT)
        if result is None:
            return h_initial
        return h.value
    
    # Perform a local optimization
    best_h = optimize_h(h)
    best_c5 = compute_c5(best_h)
    
    # Try a few more optimizations with different starting points
    for _ in range(3):
        # Add small perturbations
        perturbation = np.random.normal(0, 0.05, len(best_h))
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = optimize_h(perturbed_h)
        current_c5 = compute_c5(perturbed_h)
        if current_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = current_c5
    
    # Further refine using a gradient-based method
    def objective_func(h_vals):
        return compute_c5(h_vals)
    
    result = optimize.minimize(
        objective_func,
        best_h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(len(best_h))],
        options={'maxiter': 100}
    )
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Check time remaining
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # If we have time, try more aggressive refinement
        result = optimize.minimize(
            objective_func,
            best_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(len(best_h))],
            options={'maxiter': 50}
        )
        best_h = result.x
        best_c5 = compute_c5(best_h)
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    total_integral = np.sum(best_h) * dx
    best_h = np.round(best_h, 5)
    if abs(total_integral - 1.0) > 1e-6:
        # Adjust h to meet the integral constraint
        adjustment = (1.0 - total_integral) / dx
        h = best_h + adjustment / len(best_h)
        h = np.clip(h, 0, 1)
        best_h = h
    
    # Return the best solution
    return (best_h, best_c5, len(best_h))
