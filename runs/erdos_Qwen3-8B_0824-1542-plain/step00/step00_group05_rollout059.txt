import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Predefined initial solution with n_points = 70
    n_points = 70
    h_initial = initial_h_values
    h_initial = np.array(h_initial)
    dx = 2.0 / n_points
    assert np.isclose(np.sum(h_initial) * dx, 1.0), "Initial solution does not satisfy the integral constraint"
    
    # Function to evaluate the overlap integral
    def evaluate_c5(h):
        h = np.array(h)
        if len(h) != n_points:
            h = np.pad(h, (0, n_points - len(h)), 'constant')
        elif len(h) > n_points:
            h = h[:n_points]
        h = np.clip(h, 0, 1)
        integral = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(integral)
    
    # Function to check the integral constraint
    def constraint(h):
        h = np.array(h)
        return np.sum(h) - n_points / 2
    
    # Optimization setup
    h_bounds = [(0, 1) for _ in range(n_points)]
    h_initial = h_initial.tolist()
    
    # Initial evaluation
    c5_initial = evaluate_c5(h_initial)
    best_h = h_initial.copy()
    best_c5 = c5_initial
    
    # Use the SLSQP method for optimization
    result = optimize.minimize(
        lambda h: evaluate_c5(h),
        h_initial,
        bounds=h_bounds,
        constraints={'type': 'eq', 'fun': constraint},
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Check if optimization succeeded
    if result.success:
        best_h = result.x.tolist()
        best_c5 = evaluate_c5(best_h)
    
    # Refine with a small grid search around the best solution
    for i in range(len(best_h)):
        for delta in [0.05, -0.05]:
            h_test = best_h.copy()
            h_test[i] += delta
            h_test = np.clip(h_test, 0, 1)
            if np.isclose(np.sum(h_test), n_points / 2):
                c5_test = evaluate_c5(h_test)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_test
    
    # Final check for best solution
    best_h = np.clip(best_h, 0, 1)
    best_c5 = evaluate_c5(best_h)
    
    # Record the best solution
    return best_h, best_c5, n_points
