import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        shifted_h = np.zeros(2 * n_points)
        shifted_h[n_points:] = h
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Function to compute the integral constraint
    def integral_constraint(h):
        return np.sum(h) * dx - 1.0
    
    # Function to evaluate and return the best solution
    def evaluate_solution(h):
        c5 = compute_c5(h)
        return c5, h
    
    # Initial evaluation
    best_c5, best_h = evaluate_solution(h_initial)
    best_h_values = best_h.copy()
    best_n_points = n_points
    
    # Try a few refinements
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Create a new h with some random noise for local search
        h = copy.deepcopy(best_h_values)
        h += np.random.uniform(-0.1, 0.1, size=n_points)
        h = np.clip(h, 0.0, 1.0)
        
        # Enforce integral constraint
        integral = np.sum(h) * dx
        adjustment = (1.0 - integral) / n_points
        h += adjustment
        h = np.clip(h, 0.0, 1.0)
        
        # Evaluate
        c5, h = evaluate_solution(h)
        
        # Update best if improved
        if c5 < best_c5:
            best_c5 = c5
            best_h_values = h.copy()
    
    # Try increasing the number of points (if time allows)
    if time.time() - start_time < budget_s - 5:
        n_points_new = min(2 * n_points, 1000)
        dx_new = 2.0 / n_points_new
        
        # Create a new initial guess with more points
        h_initial_new = np.zeros(n_points_new)
        h_initial_new[:n_points] = best_h_values / dx * dx_new
        h_initial_new = np.clip(h_initial_new, 0.0, 1.0)
        h_initial_new = h_initial_new + np.random.uniform(-0.05, 0.05, size=n_points_new)
        h_initial_new = np.clip(h_initial_new, 0.0, 1.0)
        
        # Enforce integral constraint
        integral = np.sum(h_initial_new) * dx_new
        adjustment = (1.0 - integral) / n_points_new
        h_initial_new += adjustment
        h_initial_new = np.clip(h_initial_new, 0.0, 1.0)
        
        # Evaluate the new solution
        c5_new, h_new = evaluate_solution(h_initial_new)
        
        # Update best if improved
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h_values = h_new.copy()
            best_n_points = n_points_new
    
    return best_h_values, best_c5, best_n_points
