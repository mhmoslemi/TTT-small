```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions and variables
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    # Initial attempt with the provided initial_h_values
    n_points = len(initial_h_values)
    h = np.array(initial_h_values)
    c5 = evaluate_erdos_solution(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        best_n_points = n_points

    # Try a few refinements with different n_points
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 1:
            break
        # Generate a random initial guess with the required sum
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        
        # Define the cost function
        def objective(x):
            h = x.reshape(n_points)
            # Compute the overlap integral
            conv = signal.correlate(h, 1 - h, mode='full')
            c5 = np.max(conv) * (2.0 / n_points)
            return c5

        # Define the constraints
        def constraints(x):
            h = x.reshape(n_points)
            return np.sum(h) - n_points / 2

        # Define the optimization problem
        x = cp.Variable(n_points)
        objective_func = cp.Minimize(objective(x))
        constraints_func = [constraints(x) == 0]

        # Solve the optimization problem
        problem = cp.Problem(objective_func, constraints_func)
        try:
            result = problem.solve(solver=cp.SCS, eps=1e-4, time_limit=budget_s - (time.time() - start_time))
            if result is not None:
                h_opt = x.value
                c5 = evaluate_erdos_solution(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                    best_n_points = n_points
        except:
            pass

    # Final evaluation
    return best_h, best_c5, best_n_points
```
