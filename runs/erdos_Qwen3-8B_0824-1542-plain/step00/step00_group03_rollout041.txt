evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

    def objective_function(h_values):
        # Ensure h_values is valid
        if not np.all((h_values >= 0) & (h_values <= 1)):

        total_area = np.sum(h) * (2.0 / n_points)

            scaling = 1.0 / total_area
