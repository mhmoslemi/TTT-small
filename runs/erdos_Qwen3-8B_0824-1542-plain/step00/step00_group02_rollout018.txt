import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial construction from initial_h_values
    n_points_initial = len(initial_h_values)
    h_initial = initial_h_values
    
    # Try with different n_points to find a good balance between resolution and speed
    n_points_list = [n_points_initial, 2 * n_points_initial]
    
    best_c5 = float('inf')
    best_h = h_initial
    best_n_points = n_points_initial
    
    for n_points in n_points_list:
        if n_points < 10:
            continue  # Skip too small n_points to ensure sufficient resolution
        
        # Compute dx
        dx = 2.0 / n_points
        
        # Create h as a vector with n_points samples
        h = np.random.rand(n_points)
        h = h * 2 - 1  # Random initial guess in [-1, 1]
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2)  # Scale to satisfy the integral constraint
        h = h / np.sum(h) * (n_points / 2)  # Ensure sum(h) == n_points / 2 exactly
        
        # Optimization
        start_time = time.time()
        iterations = 0
        max_iterations = 100
        while time.time() - start_time < budget_s and iterations < max_iterations:
            # Compute the overlap integral for all k
            overlap = np.zeros(n_points + 1)
            for k in range(-n_points // 2, n_points // 2 + 1):
                # Ensure k is within the range that makes sense for the convolution
                if k < 0:
                    h_k = np.roll(h, k)
                else:
                    h_k = h
                overlap_k = np.sum(h * (1 - h_k)) * dx
                overlap[abs(k)] = max(overlap[abs(k)], overlap_k)
            
            # Choose the maximum overlap
            max_overlap = np.max(overlap)
            
            # If we have already achieved the best result, break early
            if max_overlap <= best_c5:
                break
            
            # Define objective function for optimization
            def objective(x):
                h = x.reshape(n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                overlap = np.zeros(n_points + 1)
                for k in range(-n_points // 2, n_points // 2 + 1):
                    if k < 0:
                        h_k = np.roll(h, k)
                    else:
                        h_k = h
                    overlap_k = np.sum(h * (1 - h_k)) * dx
                    overlap[abs(k)] = max(overlap[abs(k)], overlap_k)
                return max(overlap)
            
            # Use L-BFGS-B for optimization
            result = opt.minimize(objective, h.flatten(), method='L-BFGS-B', tol=1e-6)
            h = result.x.reshape(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Update best result
            overlap = np.zeros(n_points + 1)
            for k in range(-n_points // 2, n_points // 2 + 1):
                if k < 0:
                    h_k = np.roll(h, k)
                else:
                    h_k = h
                overlap_k = np.sum(h * (1 - h_k)) * dx
                overlap[abs(k)] = max(overlap[abs(k)], overlap_k)
            current_c5 = np.max(overlap)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
                best_n_points = n_points
            iterations += 1
        
        # Evaluate the best solution
        evaluate_erdos_solution(best_h, best_n_points)
    
    return best_h, best_c5, best_n_points
