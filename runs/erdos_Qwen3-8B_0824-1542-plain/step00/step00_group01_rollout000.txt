```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load('initial_h_values.npy')  # Example initial construction (n=47)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different n_points
    n_points_list = [100, 150, 200]
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Initialize h with the initial construction if available and of the same length
        if len(initial_h_values) == n_points:
            h = np.copy(initial_h_values)
        else:
            h = np.random.rand(n_points)
        
        # Rescale to meet the integral constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Optimization: minimize max overlap
        # We use a heuristic approach due to the complexity of the problem
        
        # Define the overlap function
        def objective(h):
            h = np.array(h)
            h = h * (n_points / 2) / np.sum(h)
            overlap = np.correlate(h, 1 - h, mode="full")
            return np.max(overlap) * (2.0 / n_points)
        
        # Use simulated annealing
        result = opt.minimize(
            fun=objective,
            x0=h,
            method='anneal',
            options={'maxiter': 100, 'tol': 1e-4}
        )
        
        current_h = result.x
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
            best_n = n_points
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, best_n)
```
