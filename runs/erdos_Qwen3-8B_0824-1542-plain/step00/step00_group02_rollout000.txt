import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 45  # Start with the given initial construction
    dx = 2.0 / n_points

    # Initial solution
    h = np.array(initial_h_values)
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / sum_h
    c5 = evaluate_erdos_solution(h, dx)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Try increasing n_points up to 500 (max 1000 samples is a bit slow)
    for n_points in range(50, 500, 50):
        dx = 2.0 / n_points
        h = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h

        # Local optimization
        for _ in range(5):
            # Compute correlation
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr * dx)

            # Update best solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

            # Adjust h to reduce overlap
            for i in range(n_points):
                # Sample the current h around i
                window = h[max(0, i-5):min(n_points, i+6)]
                if np.sum(window) < 5:
                    continue  # Not enough mass to adjust
                # Find the peak of the overlap
                corr_window = np.correlate(window, 1 - window, mode="full")
                peak_idx = np.argmax(corr_window)
                peak_val = window[peak_idx]
                # Reduce peak by adjusting h[i]
                h[i] = max(0, h[i] - 0.01)
                h[i] = min(1, h[i])

        # Check if time is up
        if time.time() - start_time > budget_s - 1:
            break

    # Return the best solution found
    return (best_h, best_c5, len(best_h))
