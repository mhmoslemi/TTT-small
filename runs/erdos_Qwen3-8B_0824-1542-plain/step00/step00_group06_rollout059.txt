import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from provided construction
    initial_h_values = evaluate_erdos_solution()
    n_points_initial = len(initial_h_values)
    
    # Try starting with smaller n_points for faster optimization
    n_points_values = [128, 256, 512]
    best_c5 = float('inf')
    best_h = initial_h_values
    
    start_time = time.time()
    
    for n_points in n_points_values:
        if time.time() - start_time > budget_s:
            break
        
        # Generate a random starting point if we don't have one
        if n_points == n_points_initial:
            h = np.copy(initial_h_values)
        else:
            h = np.random.rand(n_points)
            h = (h * 2 - 1) * 0.5 + 0.5  # Rescale to [0, 1]
            h = h * (n_points / 2)  # Adjust sum to n_points / 2
        
        # Set up optimization problem
        x = cp.Variable(n_points)
        dx = 2.0 / n_points
        sum_h = cp.sum(x)
        objective = cp.Minimize(0)  # We minimize over the max overlap
        
        # Constraints
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= 1
        ]
        
        # Optimization
        problem = cp.Problem(objective, constraints)
        try:
            result = problem.solve(solver=cp.CBC, verbose=False)
            if result is not None and np.isfinite(result):
                # Compute overlap integral
                h_values = x.value
                h_values = np.clip(h_values, 0, 1)
                # Compute convolution
                conv = np.correlate(h_values, 1 - h_values, mode="full")
                c5 = np.max(conv) * dx
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_values
        except:
            pass  # Skip if optimization fails
        
        # Early stopping if we hit the target
        if best_c5 <= 0.380876:
            break
    
    return (best_h, best_c5, len(best_h))
