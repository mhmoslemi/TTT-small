import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Function to generate a step function h with n_points samples
    def generate_h(n_points):
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = (h * 2 - 1) * 0.5 + 0.5
        h = np.clip(h, 0, 1)
        h = h * n_points / 2
        return h

    # Function to compute the overlap integral
    def compute_c5(h):
        n_points = len(h)
        dx = 2.0 / n_points
        h_extended = np.concatenate([h, np.zeros(n_points)])
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5

    # Start from the initial solution
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    if np.sum(initial_h) != n_points / 2:
        initial_h = initial_h * n_points / 2 / np.sum(initial_h)

    best_h = copy.deepcopy(initial_h)
    best_c5 = compute_c5(best_h)
    best_n_points = n_points

    # Try with smaller n_points to explore more quickly
    for n_points in [100, 200, 300, 400, 500]:
        # Generate random initial guess
        h = generate_h(n_points)
        h = h * n_points / 2 / np.sum(h)
        
        def objective(x):
            h = x.reshape(n_points)
            c5 = compute_c5(h)
            return c5

        # Optimization: minimize C5
        result = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            tol=1e-5,
            options={'maxiter': 100}
        )
        
        if result.success:
            h_opt = result.x
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                best_n_points = n_points

    return (best_h, best_c5, best_n_points)
