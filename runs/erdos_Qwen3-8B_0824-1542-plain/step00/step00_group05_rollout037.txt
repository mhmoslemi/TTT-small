```python
import numpy as np
import scipy.optimize
import scipy.signal
import time
import cvxpy as cp
from functools import partial

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial construction
    n_points = 70
    h_values = initial_h_values.copy()
    
    # Evaluate current C5
    c5_bound = evaluate_erdos_solution(h_values)
    
    # Refinement parameters
    max_refinement = 5
    refinement_steps = 3
    best_h = h_values
    best_c5 = c5_bound
    
    # Try different refinement strategies
    for _ in range(max_refinement):
        # Try increasing n_points
        n_points = min(300, n_points * 2)
        h = np.zeros(n_points)
        
        # Distribute the integral constraint
        dx = 2.0 / n_points
        total_integral = 1.0
        h = np.random.rand(n_points)
        h = h / np.sum(h) * total_integral / dx
        
        # Optimize h for better C5
        def objective(h):
            h = h.reshape(-1)
            dx = 2.0 / len(h)
            corr = scipy.signal.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        def constraint(h):
            h = h.reshape(-1)
            dx = 2.0 / len(h)
            integral = np.sum(h) * dx
            return integral - 1.0
        
        # Use scipy.optimize to minimize C5
        result = scipy.optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}]
        )
        
        # Update best solution
        if result.success:
            h_values = result.x
            c5_bound = objective(h_values)
            if c5_bound < best_c5:
                best_h = h_values
                best_c5 = c5_bound
        
        # Check if we have exceeded the time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, len(best_h)
```
