    def calculate_c5(h):
        h = np.array(h)

        shifted_h = np.roll(h, shift=1, axis=0)
        overlap = correlate(h, 1 - shifted_h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    # Optimization function
    def optimize_h(h_values, max_iter=100, n_points=n_points, dx=dx):
        best_h = h_values.copy()
        best_c5 = calculate_c5(best_h)
        
        for _ in range(max_iter):
            # Use a local optimization approach with gradient descent
            h = np.array(best_h)
            h = h / np.sum(h) * (n_points / 2)  # Normalize to meet the integral constraint
            h = np.clip(h, 0, 1)
            
            # Use a small perturbation to avoid getting stuck in local optima
            perturbation = np.random.normal(0, 0.05, size=n_points)
            h_perturbed = h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed / np.sum(h_perturbed) * (n_points / 2)
            
            c5_perturbed = calculate_c5(h_perturbed)
            
            if c5_perturbed < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = c5_perturbed
        
        return best_h, best_c5
    
    # Try different numbers of points to find a better solution
    for n_points in [45, 60, 80, 100, 120, 150]:

        h_values = np.clip(h_values, 0, 1)

        best_h, best_c5 = optimize_h(h_values)
