def compute_c5(h, dx):
    # Compute the overlap integral for all shifts
    # Use numpy.correlate with mode='full'
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum value is the required C5 value
    c5 = np.max(corr) * dx
    return c5
