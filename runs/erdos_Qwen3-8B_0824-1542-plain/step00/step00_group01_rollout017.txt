```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions and variables
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h_values = np.array(initial_h_values, dtype=np.float64)
    best_c5 = evaluate_erdos_solution(h_values)
    best_h = h_values.copy()
    
    # Define parameters
    max_iterations = 100
    num_samples = n_points
    dx = 2.0 / num_samples
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.array(h, dtype=np.float64)
        h_pad = np.concatenate([h, np.zeros_like(h)])
        one_minus_h = 1 - h
        one_minus_h_pad = np.concatenate([one_minus_h, np.zeros_like(one_minus_h)])
        corr = signal.correlate(h_pad, one_minus_h_pad, mode="full")
        return np.max(corr * dx)
    
    # Function to evaluate the objective and constraints for optimization
    def objective_function(h):
        return evaluate_overlap(h)
    
    # Function to get the constraint sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using Nelder-Mead simplex algorithm
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Perform local optimization
        result = opt.minimize(
            objective_function,
            best_h,
            method='Nelder-Mead',
            options={'xatol': 1e-4, 'fatol': 1e-4, 'disp': False}
        )
        
        if result.success:
            h_candidate = result.x
            c5_candidate = evaluate_overlap(h_candidate)
            
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
        
        # Optional: try a random perturbation
        perturbation = np.random.uniform(-0.1, 0.1, size=num_samples)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        c5_perturbed = evaluate_overlap(perturbed_h)
        
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = perturbed_h.copy()
    
    # Final check
    final_c5 = evaluate_overlap(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    # Return the best solution found
    return (best_h.tolist(), best_c5, n_points)
```
