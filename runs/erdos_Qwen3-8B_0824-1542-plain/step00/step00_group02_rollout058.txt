import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 45  # Start with the initial construction
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    # Function to compute C5
    def compute_c5(h_values):
        h = np.array(h_values)
        h = np.concatenate([h, h[:n_points-1]])  # Extend for convolution
        result = np.correlate(h, 1 - h, mode="full")
        return np.max(result * dx)
    
    # Function to evaluate the objective
    def objective(h_values):
        return compute_c5(h_values)
    
    # Function to evaluate the constraint (integral equals 1)
    def constraint(h_values):
        return np.sum(h_values) * dx - 1.0
    
    # Optimization using gradient descent with adaptive learning rate
    def gradient_descent(n_iterations=1000, learning_rate=0.01):
        h = initial_h_values.copy()
        best_h = h.copy()
        best_c5 = compute_c5(h)
        
        for i in range(n_iterations):
            # Compute the gradient numerically
            grad = np.zeros_like(h)
            for k in range(len(h)):
                perturbation = np.zeros_like(h)
                perturbation[k] = 1e-5
                grad[k] = (compute_c5(h + perturbation) - compute_c5(h - perturbation)) / (2 * 1e-5)
            
            # Update h
            h -= learning_rate * grad
            
            # Enforce constraints
            h = np.clip(h, 0, 1)
            h = h / (np.sum(h) * dx) * 1.0  # Ensure the integral equals 1
            
            # Update best solution
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            # Early stopping if time is up
            if time.time() - start_time > budget_s - 1:
                break
        
        return best_h, best_c5
    
    # Run gradient descent
    best_h, best_c5 = gradient_descent()
    
    # Try a few random perturbations for diversity
    for _ in range(5):
        perturbed_h = best_h + np.random.uniform(-0.1, 0.1, size=len(best_h))
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h / (np.sum(perturbed_h) * dx) * 1.0
        current_c5 = compute_c5(perturbed_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = perturbed_h.copy()
    
    # Final refinement with CVXPY
    h = cp.Variable(n_points)
    obj = cp.Minimize(cp.Max(cp.multiply(cp.ones(n_points), cp.sum(cp.multiply(h, 1 - h)))))  # Simplified objective
    constraints = [cp.sum(h) * dx == 1.0, 0 <= h, h <= 1.0]
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iter=100)
        best_h = h.value
        best_c5 = compute_c5(best_h)
    except:
        pass
    
    # Final check and return
    n_points = len(best_h)
    dx = 2.0 / n_points
    best_c5 = compute_c5(best_h)
    
    # Ensure we stay within time budget
    if time.time() - start_time > budget_s - 1:
        pass
    
    return best_h, best_c5, n_points
