import numpy as np
import scipy.optimize as optimize

    initial_h = initial_h * (n_points / 2) / sum_h

    # Function to evaluate the C5 value
    def evaluate_c5(h):
        h = np.array(h)
        h = np.concatenate([h, h[:1]])  # Extend to handle circular shifts
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Function to optimize h with constraints
    def optimize_h(h_init, n_points, seed):
        h = cp.Variable(n_points)
        # Objective: minimize the maximum overlap
        # Since we can't directly minimize the max, we use a convex optimization approach
        # We want to minimize the maximum value of the overlap integral over all shifts
        # We use a convex relaxation by introducing auxiliary variables

        # Define the shift range (from -1 to 1, in terms of dx)
        shift_range = np.arange(-n_points + 1, n_points)
        corr = np.zeros(len(shift_range))
        for i, shift in enumerate(shift_range):
            corr[i] = np.correlate(h.value, 1 - h.value, mode="full")[shift + n_points - 1] * dx

        # We use a convex optimization with a relaxation
        # Instead of minimizing the max, we minimize a weighted average
        weights = np.exp(-np.abs(shift_range) / 10)  # Bias towards smaller shifts
        objective = cp.Minimize(cp.sum(weights * corr))

        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]

        # Solve the problem
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.MOSEK, verbose=False)

        if prob.status != 'OPTIMAL':
            print("Optimization failed. Trying another method.")
            # Fallback to local optimization
            def objective_func(h_vec):
                h = np.array(h_vec)
                h = np.concatenate([h, h[:1]])
                corr = np.correlate(h, 1 - h, mode="full")
                c5 = np.max(corr) * dx
                return c5

            result = optimize.minimize(
                objective_func,
                h_init,

                bounds=[(0, 1) for _ in range(n_points)],
                options={'maxiter': 1000, 'ftol': 1e-6}

    start_time = time.time()

    optimized_h = optimize_h(best_h, n_points, seed)
