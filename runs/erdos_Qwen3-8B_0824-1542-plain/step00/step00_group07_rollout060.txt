```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Assume this is pre-loaded with n=80 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 80
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    h_sum = np.sum(h)
    total_area = h_sum * dx
    if abs(total_area - 1.0) > 1e-6:
        h = h * (1.0 / total_area) * 2.0  # Normalize to ensure ∫h(x)dx = 1
    
    def objective(h_values):
        # Compute overlap integral for all shifts
        c5 = evaluate_erdos_solution(h_values)
        return c5
    
    def update_h(h_values):
        # Perform a small random perturbation to escape local minima
        perturb = np.random.normal(0, 0.05, size=len(h_values))
        h_perturbed = np.clip(h_values + perturb, 0, 1)
        # Normalize to maintain area constraint
        h_normalized = h_perturbed * (1.0 / (np.sum(h_perturbed) * dx)) * 2.0
        return h_normalized
    
    # Simple gradient descent with random restarts
    best_h = h
    best_c5 = objective(best_h)
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        h = update_h(h)
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5
    
    # Try a few different starting points to avoid local minima
    for _ in range(3):
        if time.time() - start_time > budget_s - 1:
            break
        # Create a random initial guess
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand * (1.0 / (np.sum(h_rand) * dx)) * 2.0
        current_c5 = objective(h_rand)
        if current_c5 < best_c5:
            best_h = h_rand
            best_c5 = current_c5
    
    # Try a more structured approach with CVXPY
    # Define variables
    h_vars = cp.Variable(n_points, name='h')
    dx = 2.0 / n_points
    constraints = [0 <= h_vars, h_vars <= 1]
    constraints.append(cp.sum(h_vars) * dx == 1)
    
    # Objective: minimize the maximum overlap
    # Compute all possible shifts
    shifts = np.arange(-n_points + 1, n_points)
    overlap = []
    for k in shifts:
        h_shifted = np.roll(h_vars, k)
        overlap_k = cp.multiply(h_vars, 1 - h_shifted)
        overlap.append(overlap_k)
    
    # Objective function is the maximum of the integrals
    obj = cp.Minimize(cp.Max([cp.sum(overlap_k) * dx for overlap_k in overlap]))
    
    # Solve the optimization problem
    try:
        problem = cp.Problem(obj, constraints)
        problem.solve(solver=cp.ECOS, verbose=False)
        if problem.status == 'OPTIMAL':
            best_h = h_vars.value
            best_c5 = evaluate_erdos_solution(best_h)
    
    except:
        pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
```
