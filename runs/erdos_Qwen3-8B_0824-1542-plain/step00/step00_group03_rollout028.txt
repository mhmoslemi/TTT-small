import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
# evaluate_erdos_solution() - evaluates the C5 value of a given h_values
# initial_h_values - initial construction (n=70 samples)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    
    # Initial parameters
    n_points = 70  # start with initial construction size
    max_points = 100  # maximum number of points to try
    
    # Initial h_values
    h_values = initial_h_values.copy()
    h_values = h_values[:n_points]
    
    best_c5 = evaluate_erdos_solution(h_values)
    best_h_values = h_values.copy()
    
    for n_points in range(70, max_points + 1):
        # Rescale the current h_values to fit the new number of points
        dx = 2.0 / n_points
        h_new = np.zeros(n_points)
        for i in range(n_points):
            h_new[i] = h_values[i * n_points // len(h_values)]
        
        # Normalize to ensure integral is 1
        h_new = h_new * (len(h_values) / n_points) / (np.sum(h_new) * dx)
        
        # Evaluate
        c5 = evaluate_erdos_solution(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h_values = h_new.copy()
        
        # Check time budget
        if time.time() - start_time > budget_s:
            break
    
    # Try optimization on the best found so far
    n_opt = len(best_h_values)
    dx = 2.0 / n_opt
    h_opt = cp.Variable(n_opt)
    
    # Constraints
    constraints = [
        cp.Minimize(0),
        cp.sum(h_opt) == n_opt / 2,
        0 <= h_opt,
        h_opt <= 1
    ]
    
    # Objective: minimize max_k ∫ h(x)(1 - h(x+k)) dx
    # We approximate this with a grid search over k
    max_k = 10  # Number of shifts to check
    objective = cp.Minimize(0)
    
    for k in range(1, max_k + 1):
        # Compute overlap integral for this k
        h_correlate = np.correlate(h_opt.value, 1 - h_opt.value, mode="full") * dx
        overlap = cp.Max(h_correlate)
        objective = cp.Minimize(cp.Max(overlap))
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, eps=1e-4, verbose=False)
        if prob.status == 'optimal':
            best_h_values = h_opt.value
    except:
        pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h_values)
    
    return best_h_values, final_c5, len(best_h_values)
