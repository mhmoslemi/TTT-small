import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Placeholder; actual code would load or generate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 100  # Reduced from 95 for faster optimization
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Start from a reasonable initial guess
    if initial_h_values.size == n_points:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Normalize to meet the integral constraint
    
    # Define optimization variables
    h_var = cp.Variable(n_points, name='h')
    
    # Define the objective
    def compute_c5(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        return np.max(corr) * dx
    
    # Define and solve optimization problem for a given h
    def optimize_h(h_0):
        h = h_0.copy()
        h[h == 0] = 1e-6  # Avoid division by zero in constraints
        h[h == 1] = 1 - 1e-6
        
        # Objective function
        def objective_func(h_vals):
            return compute_c5(h_vals)
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            0 <= h_var,
            h_var <= 1
        ]
        
        # Optimization problem
        prob = cp.Problem(cp.Minimize(objective_func(h_var)), constraints)
        result = prob.solve(solver=cp.SCS, max_iters=100)
        
        if result is not None:
            return h_var.value
        return h
        
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Optimization loop
    for _ in range(10):  # Limited iterations to stay within time budget
        h = optimize_h(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    return best_h, best_c5, n_points
