def evaluate_erdos_solution(h_vals):
    n_points = len(h_vals)
    dx = 2.0 / n_points
    h = np.array(h_vals, dtype=np.float64)
    shifted = np.zeros((n_points * 2 + 1), dtype=np.float64)
    shifted[n_points:] = 1.0 - h[:n_points]
    corr = np.correlate(h, shifted, mode="full")
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # Start with 100 points for speed
    n_points_max = 200  # Max points to try if time allows
    
    # Initial guess
    h = np.copy(initial_h_values[:n_points])
    h = h / np.sum(h) * (n_points / 2)
    
    best_h = h
    best_c5 = evaluate_erdos_solution(h)
    best_n = n_points
    
    # Try increasing n_points if time allows
    for n_points in range(n_points + 1, n_points_max + 1):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Initialize with a better starting guess
        if n_points <= len(initial_h_values):
            h = np.copy(initial_h_values[:n_points])
        else:
            h = np.zeros(n_points)
            h[::2] = 1.0
        h = h / np.sum(h) * (n_points / 2)
        
        # Set up optimization
        h_var = cp.Variable(n_points)
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Define the overlap integral
        def objective_func(h_vals):
            h = np.array(h_vals)
            shifted = np.zeros((n_points * 2 + 1), dtype=np.float64)
            shifted[n_points:] = 1.0 - h[:n_points]
            corr = np.correlate(h, shifted, mode="full")
            return np.max(corr) * (2.0 / n_points)
        
        # Use gradient descent to find a local minimum
        def objective(h_vals):
            return objective_func(h_vals)
        
        def gradient(h_vals):
            h = np.array(h_vals)
            shifted = np.zeros((n_points * 2 + 1), dtype=np.float64)
            shifted[n_points:] = 1.0 - h[:n_points]
            corr = np.correlate(h, shifted, mode="full")
            c5 = np.max(corr) * (2.0 / n_points)
            # Estimate the gradient using finite differences
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                grad_plus = objective_func(h_plus)
                h_minus = h.copy()
                h_minus[i] -= 1e-5
                grad_minus = objective_func(h_minus)
                grad[i] = (grad_plus - grad_minus) / (2 * 1e-5)
            return grad
        
        # Perform gradient descent
        h_opt = np.copy(h)
        learning_rate = 0.01
        for _ in range(100):
            grad = gradient(h_opt)
            h_opt -= learning_rate * grad
            # Project back to constraints
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt / np.sum(h_opt) * (n_points / 2)
            current_c5 = objective(h_opt)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt
                best_n = n_points
        
        # Check if current is better
        current_c5 = objective(h_opt)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt
            best_n = n_points
    
    # Convert best_h to a numpy array
    best_h = np.array(best_h, dtype=np.float64)
    return (best_h, best_c5, best_n)
