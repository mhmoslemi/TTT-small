import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for efficiency
    dx = 2.0 / n_points
    h = copy.deepcopy(initial_h_values)
    
    def compute_c5(h_values):
        h_array = np.array(h_values)
        shifted_h = np.zeros_like(h_array)
        for k in range(1, len(h_array)):
            shifted_h[k:] = h_array[:-k]
        one_minus_h = 1.0 - h_array
        correlation = np.correlate(h_array, one_minus_h, mode="full")
        c5 = np.max(correlation) * dx
        return c5
    
    def objective(x):
        h_values = x.copy()
        h_values = np.clip(h_values, 0, 1)
        return compute_c5(h_values)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    def constraint_bounds(x):
        return np.all((0 <= x) & (x <= 1))
    
    # Initial guess
    x0 = np.array(h)
    
    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    options = {
        'maxiter': 100,
        'ftol': 1e-6,
        'disp': False
    }
    
    # Solve the optimization problem
    start_time = time.time()
    result = optimize.minimize(objective, x0, bounds=bounds, method='L-BFGS-B', options=options)
    
    # Check if the optimization was successful
    if result.success:
        best_h = result.x
        best_c5 = compute_c5(best_h)
    else:
        best_h = x0
        best_c5 = compute_c5(best_h)
    
    # Check if we can increase the number of points for a better solution
    elapsed = time.time() - start_time
    if elapsed < budget_s - 5:
        n_points = min(200, int(2 * n_points))
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        # Initialize with a simple alternating pattern to keep the integral around 1
        for i in range(n_points):
            if i % 2 == 0:
                h[i] = 0.5
            else:
                h[i] = 0.5
        # Adjust to meet the integral constraint
        total = np.sum(h) * dx
        if total < 1:
            delta = (1 - total) / dx
            for i in range(n_points):
                h[i] += delta
        h = np.clip(h, 0, 1)
        
        def new_objective(x):
            h_array = x.copy()
            h_array = np.clip(h_array, 0, 1)
            shifted_h = np.zeros_like(h_array)
            for k in range(1, len(h_array)):
                shifted_h[k:] = h_array[:-k]
            one_minus_h = 1.0 - h_array
            correlation = np.correlate(h_array, one_minus_h, mode="full")
            c5 = np.max(correlation) * dx
            return c5
        
        def new_constraint_sum(x):
            return np.sum(x) - n_points / 2
        
        def new_constraint_bounds(x):
            return np.all((0 <= x) & (x <= 1))
        
        # New optimization
        new_x0 = h
        new_bounds = [(0, 1) for _ in range(n_points)]
        new_options = {
            'maxiter': 50,
            'ftol': 1e-6,
            'disp': False
        }
        new_result = optimize.minimize(new_objective, new_x0, bounds=new_bounds, method='L-BFGS-B', options=new_options)
        
        if new_result.success:
            best_h = new_result.x
            best_c5 = compute_c5(best_h)
        else:
            best_h = new_x0
            best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
