```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for faster computation
    dx = 2.0 / n_points

    def evaluate(h):
        h = np.array(h)
        integral = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(integral)
        return c5

    # Start with initial guess
    h = initial_h_values.copy()
    h = np.array(h)[:n_points]
    h = h / (np.sum(h) * dx) * (n_points / 2)
    c5 = evaluate(h)
    best_c5 = c5
    best_h = h

    # Try increasing n_points to improve accuracy
    for n_points in [100, 200, 300, 400, 500, 600, 700]:
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h / (np.sum(h) * dx) * (n_points / 2)
        h = np.clip(h, 0, 1)

        # Use L-BFGS-B to minimize C5
        def objective(x):
            x = x.reshape(n_points)
            x = np.clip(x, 0, 1)
            integral = np.correlate(x, 1 - x, mode="full") * dx
            return np.max(integral)

        result = optimize.minimize(objective, h, method='L-BFGS-B', jac=False, tol=1e-5, options={'maxiter': 100})
        if result.success:
            c5 = result.fun
            if c5 < best_c5:
                best_c5 = c5
                best_h = result.x.copy()
                print(f"n_points={n_points}, C5={best_c5}")

        # Check if time budget is exceeded
        if time.time() - start_time > budget_s - 1:
            break

    # Return the best found solution
    return (best_h, best_c5, n_points)
```
