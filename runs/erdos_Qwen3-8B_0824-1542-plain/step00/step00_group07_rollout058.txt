        corr = correlate(h, 1 - h, mode='full')

    def optimize_h(h):
        nonlocal best_c5, best_h
        start_time = time.time()
        lr = 0.1
        momentum = 0.9
        max_iter = 1000
        h = h.copy()

            eps = 1e-6

                h_minus[i] -= eps
