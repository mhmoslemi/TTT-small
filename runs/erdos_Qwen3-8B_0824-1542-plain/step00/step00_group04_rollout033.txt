import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.roll(h, shift=1, axis=0)
    overlap = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def initial_h_values():
    # Default initial guess (n=42)
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Choose initial n_points
    n_points = 100  # Start with 100 points for reasonable performance
    dx = 2.0 / n_points
    
    # Initialize h_values
    h_values = initial_h_values()
    h_values = h_values[:n_points]  # Trim to n_points
    h_values = np.pad(h_values, (0, n_points - len(h_values)), 'constant') if len(h_values) < n_points else h_values
    
    # Ensure total integral is 1
    total_integral = np.sum(h_values) * dx
    scaling_factor = 1.0 / total_integral
    h_values *= scaling_factor
    
    best_c5 = evaluate_erdos_solution(h_values)
    best_h = h_values.copy()
    best_n = n_points
    
    time_start = time.time()
    
    # Try a few different n_points to find a good setup
    for n_points in [60, 80, 100, 120, 140, 160, 180, 200]:
        if time.time() - time_start > budget_s - 5:
            break
        
        dx = 2.0 / n_points
        
        # Initialize h_values with a random pattern
        h_values = np.random.rand(n_points)
        h_values = (h_values - np.min(h_values)) / (np.max(h_values) - np.min(h_values))  # Normalize to [0,1]
        total_integral = np.sum(h_values) * dx
        scaling_factor = 1.0 / total_integral
        h_values *= scaling_factor
        
        # Perform gradient descent optimization
        def objective(h):
            h = h.reshape((n_points,))
            h = np.clip(h, 0, 1)
            dx = 2.0 / n_points
            shifted = np.roll(h, shift=1, axis=0)
            overlap = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(overlap)
        
        # Initial guess
        x0 = h_values.copy()
        result = opt.minimize(objective, x0, method='L-BFGS-B', bounds=[(0, 1)] * n_points, tol=1e-4)
        h_opt = result.x
        c5 = objective(h_opt)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
            best_n = n_points
    
    # Final evaluation
    best_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, best_n
