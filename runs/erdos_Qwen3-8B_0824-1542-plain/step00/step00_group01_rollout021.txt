import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    n_points = 100  # Start with 100 points, can be adjusted if needed
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points])  # Initial guess from provided values
    h = h[:n_points]  # Ensure it's of correct length
    h = h / np.sum(h) * (n_points / 2)  # Normalize to satisfy the integral constraint

    # Function to compute C5
    def compute_c5(h_values):
        h = h_values.copy()
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the convolution
        conv = np.correlate(h, 1 - h, mode="full")
        # Compute the maximum overlap integral
        c5 = np.max(conv) * dx
        return c5

    # Function to optimize h
    def optimize_h(h_initial):
        h = cp.Variable(n_points)
        # Constraints
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2  # Ensure the integral is 1
        ]
        # Objective: minimize the maximum overlap integral
        # Use a convex relaxation
        def objective(h_vals):
            h = h_vals.copy()
            # Ensure h is in [0, 1]
            h = np.clip(h, 0, 1)
            # Compute the convolution
            conv = np.correlate(h, 1 - h, mode="full")
            # Compute the maximum overlap integral
            c5 = np.max(conv) * dx
            return c5

        # Minimize the maximum overlap integral
        obj = cp.Minimize(cp.Minimize(objective(h)))
        problem = cp.Problem(obj, constraints)
        try:
            result = problem.solve(solver=cp.ECOS, verbose=False)
            if result is not None:
                return h.value
            else:
                return h_initial
        except:
            return h_initial

    # Simulated annealing approach
    def simulated_annealing(h_initial, seed, budget_s):
        np.random.seed(seed)
        h_current = h_initial.copy()
        best_h = h_current.copy()
        best_c5 = compute_c5(h_current)
        time_start = time.time()
        temp = 1.0
        cooling_rate = 0.995
        iterations = 0

        while time.time() - time_start < budget_s:
            # Generate a random perturbation
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            h_perturbed = h_current + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            # Ensure the integral is 1
            h_perturbed = h_perturbed / np.sum(h_perturbed) * (n_points / 2)
            c5_perturbed = compute_c5(h_perturbed)

            # Acceptance probability
            delta = c5_perturbed - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                h_current = h_perturbed
                if c5_perturbed < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_perturbed

            temp *= cooling_rate
            iterations += 1

        return best_h, best_c5

    # Run the optimization
    h_optimized, c5_bound = simulated_annealing(h, seed, budget_s)

    # Final evaluation
    final_c5 = compute_c5(h_optimized)

    return h_optimized, final_c5, n_points
