```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time
import sys

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Placeholder for the actual initial solution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different n_points values to find the best solution within budget
    n_points_list = [100, 200, 300]  # Test with different resolutions
    
    best_h = None
    best_c5 = float('inf')
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Compute dx
        dx = 2.0 / n_points
        
        # Initialize h with the initial solution or random values
        if n_points == len(initial_h_values):
            h = initial_h_values
        else:
            h = np.random.rand(n_points)
            # Scale to meet the integral constraint
            h = h * (n_points / 2) / np.sum(h)
        
        # Check if the initial solution is valid
        if n_points == len(initial_h_values):
            h = initial_h_values.copy()
            h = h[:n_points]
        
        # Perform optimization
        print(f"Trying with n_points = {n_points}")
        h = h.copy()
        
        # Ensure the sum of h is n_points / 2
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            scale_factor = (n_points / 2) / sum_h
            h = h * scale_factor
        
        # Create h as a variable
        x = cp.Variable(n_points)
        # Define the constraints
        constraints = [
            cp.Minimize(cp.max(cp.multiply(cp.reshape(x, (n_points, 1)), 1 - cp.reshape(x, (1, n_points)))[:, :n_points - 1].sum(axis=1))),
            x >= 0,
            x <= 1,
            cp.sum(x) == n_points / 2
        ]
        
        # Solve the optimization problem
        try:
            prob = cp.Problem(*constraints)
            result = prob.solve(solver=cp.SCS)
            if result is not None:
                h_opt = x.value
                c5 = evaluate_erdos_solution(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
        except:
            pass
        
        # Simple gradient descent as a fallback
        if best_h is None:
            h = h.copy()
            for _ in range(200):
                if time.time() - start_time > budget_s:
                    break
                # Compute current overlap
                c5_current = evaluate_erdos_solution(h)
                # Compute gradients for each shift
                grad = np.zeros_like(h)
                for k in range(1, n_points):
                    shift = np.roll(h, k)
                    overlap = np.sum(h * (1 - shift))
                    grad -= (overlap - c5_current) * np.roll(np.ones_like(h), k)
                grad /= np.linalg.norm(grad)
                # Update h
                h = h - 0.01 * grad
                # Ensure constraints
                h = np.clip(h, 0, 1)
                sum_h = np.sum(h)
                if np.abs(sum_h - n_points / 2) > 1e-6:
                    scale_factor = (n_points / 2) / sum_h
                    h = h * scale_factor
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    
    # Final check for best_h
    if best_h is not None:
        c5 = evaluate_erdos_solution(best_h)
        return (best_h, c5, len(best_h))
    else:
        return (initial_h_values, evaluate_erdos_solution(initial_h_values), len(initial_h_values))
```
