import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Choose a manageable number of points, less than 1k
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    
    # Start with a uniform distribution as an initial guess
    initial_h_values[:] = 0.5
    # Adjust to meet the integral constraint
    initial_h_values[0::2] = 0.5 + 0.5 / 2
    initial_h_values[1::2] = 0.5 - 0.5 / 2
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    
    def objective_function(h_values):
        h = h_values.reshape(-1)
        # Compute the overlap integral for all k in the range
        corr = correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(corr)
        return c5

    # Optimization using a local optimizer with constraint
    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(
        objective_function,
        initial_h_values,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 100}
    )

    h_values = result.x
    c5_bound = objective_function(h_values)
    return (h_values, c5_bound, n_points)
