```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points, can increase if needed
    dx = 2.0 / n_points
    max_iterations = 100  # Limit the number of iterations to stay within time budget

    # Initial guess based on available initial_h_values
    h = np.array(initial_h_values, dtype=np.float64)
    h = h[:n_points]  # Ensure it's the same length as n_points
    h = np.concatenate([h, np.zeros(n_points - len(h))])  # Pad if necessary
    h = h[:n_points]  # Trim if necessary
    h = h / (np.sum(h) * dx) * (n_points / 2)  # Normalize to satisfy the integral constraint

    best_c5 = np.inf
    best_h = h.copy()

    for iteration in range(max_iterations):
        start_time = time.time()
        if time.time() - start_time > budget_s:
            break

        # Objective function to minimize C5
        def objective_vectorized(h_vec):
            h = h_vec.reshape((n_points,))
            # Ensure h is in [0, 1]
            h_clamped = np.clip(h, 0, 1)
            # Compute the overlap integral
            h_correlate = np.correlate(h_clamped, 1 - h_clamped, mode="full")
            c5 = np.max(h_correlate) * dx
            return c5

        # Constraint: ∫h(x)dx = 1
        def constraint(h_vec):
            h = h_vec.reshape((n_points,))
            return np.sum(h) * dx - (n_points / 2)

        # Optimization problem
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(objective_vectorized(h_var))
        constraints = [cp.sum(h_var) * dx == n_points / 2, 0 <= h_var, h_var <= 1]
        prob = cp.Problem(objective, constraints)

        # Solve the problem
        try:
            result = prob.solve(solver=cp.SCS, eps=1e-6, max_iters=100)
            if result is not None:
                h_new = h_var.value
                c5_new = objective_vectorized(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
        except:
            pass  # Skip if solver fails

        # Early stopping if we've met the target
        if best_c5 <= 0.380876:
            break

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, n_points)
```
