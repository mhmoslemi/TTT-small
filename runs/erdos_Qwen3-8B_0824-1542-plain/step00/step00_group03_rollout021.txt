import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
# initial_h_values = ...  # Assume this is defined globally with n=70 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100  # Choose a manageable number of points
    dx = 2.0 / n_points
    max_iter = 100  # Limit number of iterations for time efficiency
    
    # Helper function to compute the overlap integral
    def compute_c5(h):
        # Ensure h is in [0,1]
        h = np.clip(h, 0, 1)
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Compute the max value of the integrals
        c5 = np.max(corr) * dx
        return c5

    # Initial guess based on provided initial_h_values
    if hasattr(initial_h_values, '__iter__') and len(initial_h_values) == n_points:
        h_initial = np.array(initial_h_values)
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    else:
        h_initial = np.random.rand(n_points)
        h_initial = (h_initial * 2 - 1) * 0.5 + 0.5  # Random initial guess in [0.5, 1.0]
        h_initial = np.clip(h_initial, 0, 1)
        # Normalize the integral to 1
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

    # Optimization using a local optimization method
    def objective(h_vec):
        h = h_vec.reshape(-1)
        c5 = compute_c5(h)
        return c5

    # Bounds for the variables
    bounds = [(0, 1) for _ in range(n_points)]

    # Perform a local optimization with L-BFGS-B
    result = optimize.minimize(
        objective,
        h_initial,
        method="L-BFGS-B",
        bounds=bounds,
        jac=None,
        options={'maxiter': max_iter, 'ftol': 1e-8, 'gtol': 1e-8}
    )

    # Extract the best solution
    h_values = result.x
    c5_bound = compute_c5(h_values)

    # Check time and return if within budget
    elapsed = time.time() - start_time
    if elapsed >= budget_s - 1:
        return (h_values, c5_bound, n_points)

    # Optional: Use a convex optimization approach for refinement
    # Define a convex optimization problem
    h = cp.Variable(n_points)
    objective = cp.Maximize(cp.sum((1 - h) * cp.reshape(h, (n_points, 1)) * np.eye(n_points)))
    constraints = [
        cp.sum(h) == n_points / 2,
        0 <= h,
        h <= 1
    ]
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iter=100)
        h_values = h.value
        c5_bound = compute_c5(h_values)
    except:
        pass

    return (h_values, c5_bound, n_points)
