def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=60, **kwargs):
    # Start timing
    start_time = time.time()
    
    # Starting point
    n_points = 100  # Start with 100 points
    h_initial = np.copy(initial_h_values[:n_points])
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2)  # Ensure the area is correct

    # Function to compute the maximum overlap for a given h_values
    def compute_c5(h_values):
        n_points = len(h_values)
        dx = 2.0 / n_points
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    # Optimization function using local search with simulated annealing
    def optimize_h_values(h_initial, n_points, budget_s):
        best_h = np.copy(h_initial)
        best_c5 = compute_c5(best_h)
        time_used = 0

        # Simulated Annealing parameters
        temp = 100.0
        cooling_rate = 0.995
        max_iter = 10000

        for i in range(max_iter):
            if time_used >= budget_s:
                break
            # Generate a perturbed version of h
            perturbation = np.random.normal(0, 0.05, n_points)
            h_perturbed = np.clip(best_h + perturbation, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2)  # Ensure the area is correct

            # Compute the new c5
            new_c5 = compute_c5(h_perturbed)

            # Acceptance probability
            if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5) / temp):
                best_h = h_perturbed
                best_c5 = new_c5

            # Update temperature
            temp *= cooling_rate
            time_used = time.time() - start_time

        return best_h, best_c5

    # Run the optimization
    optimized_h, optimized_c5 = optimize_h_values(h_initial, n_points, budget_s)
    return optimized_h, optimized_c5, n_points
