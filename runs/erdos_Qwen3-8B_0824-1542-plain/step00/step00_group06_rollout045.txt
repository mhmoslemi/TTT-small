import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timer
    start_time = time.time()
    
    # Initial guess
    n_points = 100  # Start with 100 samples
    h = initial_h_values.copy()  # Use the provided initial construction
    h = h[:n_points]  # Truncate to n_points
    
    # Ensure sum of h is n_points / 2
    h_sum = np.sum(h)
    if h_sum != n_points / 2:
        h += (n_points / 2 - h_sum) / n_points
    
    # Optimization parameters
    max_iter = 200
    tol = 1e-5
    best_c5 = np.inf
    best_h = h.copy()
    
    for _ in range(max_iter):
        # Evaluate C5
        dx = 2.0 / n_points
        h_correlate = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(h_correlate) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Check if we're under time budget
        if time.time() - start_time > budget_s - 1:
            break
        
        # Update h using gradient descent with a small learning rate
        learning_rate = 0.1
        h_new = h.copy()
        
        # Compute the gradient of C5 with respect to h
        # We approximate the gradient using finite differences
        # For each x in h, we compute how much changing h[x] affects the max correlation
        
        # Compute the correlation for each h
        def compute_c5(h_current):
            dx = 2.0 / n_points
            h_correlate = np.correlate(h_current, 1 - h_current, mode="full")
            return np.max(h_correlate) * dx
        
        # Compute directional derivative
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_plus = np.clip(h_plus, 0, 1)
            c5_plus = compute_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= 1e-5
            h_minus = np.clip(h_minus, 0, 1)
            c5_minus = compute_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)
        
        # Update h
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Ensure the integral constraint is satisfied
        h_sum = np.sum(h_new)
        if h_sum != n_points / 2:
            h_new += (n_points / 2 - h_sum) / n_points
        
        h = h_new
    
    # Final evaluation
    dx = 2.0 / n_points
    h_correlate = np.correlate(best_h, 1 - best_h, mode="full")
    c5 = np.max(h_correlate) * dx
    
    print(f"Best C5 found: {c5}")
    
    return best_h, c5, n_points
