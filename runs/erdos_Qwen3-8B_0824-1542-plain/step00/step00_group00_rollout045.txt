```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    n_points = 45  # Start with initial 45 samples
    
    # Try increasing n_points if time remains
    for n_points in [45, 60, 80, 100, 120, 150, 180, 200]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Construct initial guess (copy initial values)
        h = np.copy(initial_h_values)
        h = np.resize(h, n_points)
        h /= np.sum(h) * 2.0  # Normalize to ensure sum(h) * dx = 1
        
        # Optimization variables
        x = cp.Variable(n_points)
        dx = 2.0 / n_points
        sum_h = cp.sum(x)
        constraint = [sum_h == n_points / 2]  # Enforce ∫ h(x) dx = 1
        
        # Objective: Minimize the maximum of overlap integrals
        # We use a convex relaxation by minimizing the maximum of the integrals
        # using a convex optimization approach
        overlap = []
        for k in range(1, n_points):
            # Compute shifted h and (1-h) using wrap-around
            h_shifted = np.roll(x, k)
            overlap_k = cp.multiply(x, 1 - h_shifted) * dx
            overlap.append(overlap_k)
        
        # Maximize the overlap across all k
        max_overlap = cp.Max(overlap)
        objective = cp.Minimize(max_overlap)
        
        # Solve the optimization problem
        prob = cp.Problem(objective, constraint)
        try:
            prob.solve(solver=cp.SCS, eps=1e-5)  # Use SCS solver for efficiency
        except:
            # If optimization fails, use a greedy improvement
            h = np.copy(initial_h_values)
            h = np.resize(h, n_points)
            h /= np.sum(h) * 2.0
            for _ in range(100):
                # Randomly perturb the function
                idx = np.random.randint(n_points)
                h[idx] += np.random.uniform(-0.1, 0.1)
                h = np.clip(h, 0, 1)
                h /= np.sum(h) * 2.0  # Re-normalize
        
        # Evaluate the solution
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Return the best solution
    return (best_h, best_c5, n_points)
```
