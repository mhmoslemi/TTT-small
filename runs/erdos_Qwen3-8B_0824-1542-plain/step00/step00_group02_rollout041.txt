```python
import numpy as np
import scipy.optimize as optimize
import time
import copy
import cvxpy as cp

# Pre-imported functions and variables
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * dx
initial_h_values = np.load('initial_h_values.npy')  # Assume this is a numpy array of shape (n_points,)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 45  # Start with the initial sample size
    dx = 2.0 / n_points
    h = initial_h_values.copy()  # Start with the given initial solution
    
    # Optimization parameters
    max_iterations = 50
    time_limit = budget_s
    start_time = time.time()
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    for iteration in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > time_limit:
            break
        
        # Add some random perturbation to the function
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = np.clip(h + perturbation, 0, 1)
        
        # Ensure the integral constraint is satisfied
        integral = np.sum(h_perturbed) * dx
        if np.abs(integral - 1.0) > 1e-6:
            # Adjust to satisfy the integral constraint
            scaling_factor = 1.0 / integral
            h_perturbed *= scaling_factor
        
        # Evaluate the C5 value
        current_c5 = evaluate_erdos_solution(h_perturbed)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = current_c5
        
        # Check if we have improved sufficiently
        if np.abs(best_c5 - 0.380876) < 1e-5:
            break
        
        # Consider increasing the number of points for better resolution
        if iteration % 10 == 0 and n_points < 200:
            n_points += 10
            dx = 2.0 / n_points
            # Interpolate the current solution to the new number of points
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
            h = np.clip(h, 0, 1)
            # Ensure integral constraint is satisfied
            integral = np.sum(h) * dx
            scaling_factor = 1.0 / integral
            h *= scaling_factor
        
    return best_h, best_c5, n_points
```
