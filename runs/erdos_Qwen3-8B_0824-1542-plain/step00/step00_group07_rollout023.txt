```python
import numpy as np
import scipy.optimize as sco
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * 2.0 / len(h)
initial_h_values = np.load("initial_h_values.npy")  # Placeholder for actual data

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 80  # Start with 80 samples
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = h[:n_points]  # Ensure we're working with the correct length

    # Ensure the initial h satisfies the constraint sum(h) == n_points / 2
    if np.abs(np.sum(h) - n_points / 2) > 1e-6:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)

    def objective(h):
        # Ensure h is in [0,1]
        h_clipped = np.clip(h, 0, 1)
        # Correlation
        corr = np.correlate(h_clipped, 1 - h_clipped, mode="full")
        # Compute max overlap
        c5 = np.max(corr) * dx
        return c5

    def constraint(h):
        return np.sum(h) - n_points / 2

    # Optimization with bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    x0 = h

    # Use Sequential Least Squares Programming (SLSQP) optimizer
    result = sco.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="SLSQP",
        options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8}
    )

    best_h = result.x
    best_c5 = objective(best_h)
    elapsed = time.time() - start_time

    if elapsed < budget_s:
        # Try increasing n_points if time allows
        n_points = min(128, n_points + 32)
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)

        def objective_extended(h):
            h_clipped = np.clip(h, 0, 1)
            corr = np.correlate(h_clipped, 1 - h_clipped, mode="full")
            c5 = np.max(corr) * dx
            return c5

        def constraint_extended(h):
            return np.sum(h) - n_points / 2

        bounds_extended = [(0, 1) for _ in range(n_points)]
        x0_extended = h

        result_extended = sco.minimize(
            objective_extended,
            x0_extended,
            bounds=bounds_extended,
            constraints={"type": "eq", "fun": constraint_extended},
            method="SLSQP",
            options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8}
        )

        best_h = result_extended.x
        best_c5 = objective_extended(best_h)

    # Final check and return
    best_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
```
