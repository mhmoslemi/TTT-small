import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define initial h_values from provided initial_h_values
    initial_h_values = np.load("initial_h_values.npy")  # assuming initial_h_values.npy is available
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        initial_h_values = initial_h_values * (n_points / 2) / sum_h
    
    # Function to compute C_5 for a given h
    def compute_c5(h):
        h = np.concatenate([h, [0.0] * (n_points - len(h))])  # pad to ensure full convolution
        conv = np.correlate(h, 1 - h, mode="full")
        c5_values = conv[:n_points] * dx  # Only take the first n_points values
        return np.max(c5_values)
    
    # Function to compute the gradient of C_5 with respect to h
    # This is an approximation using finite differences
    def compute_gradient(h):
        grad = np.zeros_like(h)
        h_perturb = h.copy()
        h_perturb[0] += 1e-6
        grad[0] = (compute_c5(h_perturb) - compute_c5(h)) / 1e-6
        h_perturb = h.copy()
        h_perturb[0] -= 1e-6
        grad[0] = (compute_c5(h_perturb) - compute_c5(h)) / (-1e-6)
        
        for i in range(1, len(h)):
            h_perturb = h.copy()
            h_perturb[i] += 1e-6
            grad[i] = (compute_c5(h_perturb) - compute_c5(h)) / 1e-6
            h_perturb = h.copy()
            h_perturb[i] -= 1e-6
            grad[i] = (compute_c5(h_perturb) - compute_c5(h)) / (-1e-6)
        
        return grad
    
    # Optimization using gradient descent
    h = initial_h_values.copy()
    learning_rate = 0.01
    max_iterations = 1000
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for _ in range(max_iterations):
        grad = compute_gradient(h)
        h = h - learning_rate * grad
        
        # Clip h to [0, 1]
        h = np.clip(h, 0.0, 1.0)
        
        # Ensure the integral constraint is satisfied
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            h = h * (n_points / 2) / sum_h
        
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if time.time() - start_time > budget_s - 1:
            break
    
    # Further refinement using local search
    for _ in range(100):
        perturbation = np.random.normal(0, 0.05, size=n_points)
        h_perturb = np.clip(h + perturbation, 0.0, 1.0)
        sum_h = np.sum(h_perturb)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            h_perturb = h_perturb * (n_points / 2) / sum_h
        
        current_c5 = compute_c5(h_perturb)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_perturb.copy()
        
        if time.time() - start_time > budget_s - 1:
            break
    
    return best_h, best_c5, n_points
