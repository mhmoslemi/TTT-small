import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

# Pre-imported functions
# initial_h_values is a numpy array of shape (n_points,)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial guess from the given initial_h_values
    n_points_initial = initial_h_values.shape[0]
    
    # Try a few different n_points to find the best one within budget
    n_points_list = [n_points_initial, n_points_initial - 10, n_points_initial + 10]
    best_c5 = float('inf')
    best_h = None
    best_n_points = n_points_initial
    
    for n_points in n_points_list:
        if n_points <= 0:
            continue
        
        # Compute dx
        dx = 2.0 / n_points
        
        # Initial guess: interpolate the initial_h_values to n_points
        if n_points_initial != n_points:
            # Interpolate the initial_h_values to the new number of points
            x_initial = np.linspace(0, 2, n_points_initial)
            x_new = np.linspace(0, 2, n_points)
            h_initial = np.interp(x_new, x_initial, initial_h_values)
            h = h_initial.copy()
        else:
            h = initial_h_values.copy()
        
        # Ensure sum is correct
        sum_h = np.sum(h)
        required_sum = n_points / 2
        if abs(sum_h - required_sum) > 1e-6:
            h += (required_sum - sum_h) / n_points
        
        # Define the objective function
        def objective(h):
            h = h.reshape(-1)
            dx = 2.0 / len(h)
            # Compute 1 - h
            one_minus_h = 1 - h
            # Compute the convolution (full mode)
            conv = signal.correlate(h, one_minus_h, mode="full")
            # Multiply by dx
            conv *= dx
            # Take the maximum value
            return np.max(conv)
        
        # Define the constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Define the bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use the SLSQP method for optimization
        result = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        # Check if the optimization succeeded
        if result.success:
            h_opt = result.x
            c5 = objective(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n_points = n_points
        
        # Check time budget
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s - 1:
            break
    
    # Return the best solution found
    return (best_h, best_c5, best_n_points)
