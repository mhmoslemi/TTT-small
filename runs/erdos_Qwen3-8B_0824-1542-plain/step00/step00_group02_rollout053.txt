import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Define the initial number of points
    n_points = 45
    h = initial_h_values.copy()
    
    # Function to evaluate C5
    def evaluate_c5(h):
        h = np.array(h)
        h = np.concatenate([h, h[:n_points//2]])  # Extend for full correlation
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    # Function to optimize h
    def optimize_h(h_initial, n_points, budget_s):
        # Define variables
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,  # Ensure integral is 1
            0 <= h,
            h <= 1
        ]
        
        # Objective: minimize the maximum overlap integral
        # Since we cannot directly minimize the maximum, we minimize the worst-case value
        # We use a convex relaxation approach
        
        # Use a grid search to estimate the max correlation
        def objective(h_values):
            h_values = np.array(h_values)
            h_extended = np.concatenate([h_values, h_values[:n_points//2]])
            corr = signal.correlate(h_extended, 1 - h_values, mode="full")
            return np.max(corr) * (2.0 / n_points)
        
        # Use a gradient-based optimizer
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            objective,
            h_initial,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6},
            tol=1e-6
        )
        return result.x
    
    # Try a few different starting points
    best_c5 = float('inf')
    best_h = h.copy()
    
    # First, try the initial point
    h_optimized = optimize_h(h, n_points, budget_s)
    current_c5 = evaluate_c5(h_optimized)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_optimized.copy()
    
    # Try some random perturbations
    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = np.clip(h_optimized + perturbation, 0, 1)
        current_c5 = evaluate_c5(h_perturbed)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_perturbed.copy()
    
    # Try a few more iterations with more aggressive optimization
    for _ in range(3):
        h_optimized = optimize_h(best_h, n_points, budget_s)
        current_c5 = evaluate_c5(h_optimized)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_optimized.copy()
    
    # Ensure we do not exceed the time budget
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # If time remains, try a few more random perturbations
        for _ in range(3):
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            h_perturbed = np.clip(best_h + perturbation, 0, 1)
            current_c5 = evaluate_c5(h_perturbed)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed.copy()
    
    # Ensure the final h is valid
    best_h = np.clip(best_h, 0, 1)
    # Ensure the integral is correct
    integral = np.sum(best_h) * (2.0 / n_points)
    if abs(integral - 1) > 1e-6:
        print("Integral check failed:", integral)
    
    return best_h, best_c5, n_points
