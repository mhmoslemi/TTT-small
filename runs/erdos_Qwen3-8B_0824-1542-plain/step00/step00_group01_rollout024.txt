import numpy as np
import scipy.optimize as optimize
import time
import copy
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    max_attempts = 5  # Limit to prevent infinite loops
    attempts = 0

    # Initial guess from provided starting point
    initial_h = initial_h_values
    n_points_initial = len(initial_h)
    dx_initial = 2.0 / n_points_initial
    h_initial = np.array(initial_h, dtype=float)
    sum_initial = np.sum(h_initial) * dx_initial
    # Adjust initial guess to meet the constraint sum(h) * dx = 1
    if np.abs(sum_initial - 1) > 1e-6:
        scaling_factor = 1.0 / sum_initial
        h_initial = h_initial * scaling_factor
    # Ensure h is within [0, 1]
    h_initial = np.clip(h_initial, 0, 1)
    best_h = h_initial
    best_n_points = n_points_initial
    best_c5 = evaluate_erdos_solution(h_initial, n_points_initial, dx_initial)

    # Try with different n_points
    for n_points in [47, 64, 96, 128, 192, 256, 384, 512, 768]:
        if time.time() - start_time > budget_s - 5:
            break
        attempts += 1
        if attempts > max_attempts:
            break
        dx = 2.0 / n_points
        # Initial point near the provided h
        h0 = np.zeros(n_points)
        h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
        h0 = np.clip(h0, 0, 1)
        # Adjust to meet constraint
        sum_h0 = np.sum(h0) * dx
        if np.abs(sum_h0 - 1) > 1e-6:
            scaling_factor = 1.0 / sum_h0
            h0 = h0 * scaling_factor
        h0 = np.clip(h0, 0, 1)
        # Define the objective function
        def objective(x):
            h = x.copy()
            # Compute 1 - h(x + k) for all k (correlation)
            # We use a sliding window approach to compute the max overlap
            # This is a simplified approximation of the convolution
            # For a step function, we can compute the integral for each shift
            max_overlap = 0.0
            for shift in range(1, int(2 * dx) + 1):
                # Shift the function
                h_shifted = np.roll(h, shift)
                # Compute integral of h(x) * (1 - h(x + shift))
                # Since it's a step function, this is the sum over intervals
                # We use the actual value of dx for integration
                integral = 0.0
                for i in range(n_points):
                    x_i = i * dx
                    x_i_shifted = (i + shift) * dx
                    if x_i_shifted >= 2:
                        continue
                    # Find the index where x_i_shifted falls
                    idx_shifted = int(np.floor(x_i_shifted / dx))
                    if idx_shifted >= n_points:
                        continue
                    if idx_shifted < 0:
                        continue
                    value = h[i] * (1 - h_shifted[idx_shifted])
                    integral += value
                max_overlap = max(max_overlap, integral * dx)
            return max_overlap

        # Constraints: sum(h) * dx = 1
        sum_h = np.sum(h0) * dx
        constraint = LinearConstraint(np.ones(n_points), [1.0], [1.0])

        # Bounds for each h[i]
        bounds = Bounds(0, 1)

        # Use L-BFGS-B for optimization
        result = optimize.minimize(objective, h0, method='L-BFGS-B', bounds=bounds, constraints=constraint, tol=1e-6)

        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            sum_h = np.sum(h_opt) * dx
            # Adjust to meet constraint
            if np.abs(sum_h - 1) > 1e-6:
                scaling_factor = 1.0 / sum_h
                h_opt = h_opt * scaling_factor
            h_opt = np.clip(h_opt, 0, 1)
            # Compute C5
            c5 = evaluate_erdos_solution(h_opt, n_points, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n_points = n_points

    return (best_h, best_c5, best_n_points)
