import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2 / len(h_values))
initial_h_values = np.load("initial_h_values.npy")  # Example path, but in the actual problem it's pre-imported

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for faster computation
    max_iterations = 100
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Function to evaluate overlap integral
    def evaluate_overlap(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full") * (2 / len(h))
        return np.max(overlap)
    
    # Initial guess
    h_initial = initial_h_values[:n_points].copy()
    h_initial /= (2 / n_points)  # Normalize to sum to 1
    h_initial = np.clip(h_initial, 0, 1)
    
    # Optimization function
    def optimize_h(h):
        nonlocal best_c5, best_h
        h_val = np.array(h)
        h_val = np.clip(h_val, 0, 1)
        sum_h = np.sum(h_val) * (2 / n_points)
        if not np.isclose(sum_h, 1, atol=1e-6):
            return float('inf')
        c5 = evaluate_overlap(h_val)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_val.copy()
        return c5
    
    # Perform local optimization
    result = opt.minimize(
        fun=optimize_h,
        x0=h_initial,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': max_iterations}
    )
    
    # Try a few perturbations if needed
    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = np.clip(result.x + perturbation, 0, 1)
        sum_h_perturbed = np.sum(h_perturbed) * (2 / n_points)
        if np.isclose(sum_h_perturbed, 1, atol=1e-6):
            c5_perturbed = evaluate_overlap(h_perturbed)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
    
    # Try increasing n_points if time allows
    if time.time() - start_time < budget_s - 5:
        n_points = 150
        h_initial = np.random.rand(n_points)
        h_initial /= (2 / n_points)  # Normalize to sum to 1
        h_initial = np.clip(h_initial, 0, 1)
        
        # Optimization with increased points
        result = opt.minimize(
            fun=optimize_h,
            x0=h_initial,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': max_iterations}
        )
        
        # Perturbations again
        for _ in range(5):
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h_perturbed = np.clip(result.x + perturbation, 0, 1)
            sum_h_perturbed = np.sum(h_perturbed) * (2 / n_points)
            if np.isclose(sum_h_perturbed, 1, atol=1e-6):
                c5_perturbed = evaluate_overlap(h_perturbed)
                if c5_perturbed < best_c5:
                    best_c5 = c5_perturbed
                    best_h = h_perturbed.copy()
    
    # Try even higher resolution if time permits
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        h_initial = np.random.rand(n_points)
        h_initial /= (2 / n_points)  # Normalize to sum to 1
        h_initial = np.clip(h_initial, 0, 1)
        
        result = opt.minimize(
            fun=optimize_h,
            x0=h_initial,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': max_iterations}
        )
        
        for _ in range(5):
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h_perturbed = np.clip(result.x + perturbation, 0, 1)
            sum_h_perturbed = np.sum(h_perturbed) * (2 / n_points)
            if np.isclose(sum_h_perturbed, 1, atol=1e-6):
                c5_perturbed = evaluate_overlap(h_perturbed)
                if c5_perturbed < best_c5:
                    best_c5 = c5_perturbed
                    best_h = h_perturbed.copy()
    
    # Final check
    h_final = np.clip(best_h, 0, 1)
    sum_h_final = np.sum(h_final) * (2 / len(h_final))
    if not np.isclose(sum_h_final, 1, atol=1e-6):
        h_final = np.array([1.0] * len(h_final))
    
    # Ensure we don't exceed the time budget
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # For remaining time, try more sophisticated methods
        if len(h_final) < 300:
            n_points = 300
            h_initial = np.random.rand(n_points)
            h_initial /= (2 / n_points)  # Normalize to sum to 1
            h_initial = np.clip(h_initial, 0, 1)
            
            result = opt.minimize(
                fun=optimize_h,
                x0=h_initial,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                options={'maxiter': max_iterations}
            )
            
            for _ in range(5):
                perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
                h_perturbed = np.clip(result.x + perturbation, 0, 1)
                sum_h_perturbed = np.sum(h_perturbed) * (2 / n_points)
                if np.isclose(sum_h_perturbed, 1, atol=1e-6):
                    c5_perturbed = evaluate_overlap(h_perturbed)
                    if c5_perturbed < best_c5:
                        best_c5 = c5_perturbed
                        best_h = h_perturbed.copy()
    
    return (best_h, best_c5, len(best_h))
