```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions and variables
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Start timer
    start_time = time.time()
    
    # Set initial number of points and adjust if needed
    n_points = 100  # Start with 100 points
    max_points = 1000  # Maximum points allowed
    
    # Initial guess from the given initial_h_values
    initial_h = initial_h_values[:]  # Make a copy
    initial_h = np.array(initial_h)
    
    # If initial_h has different length, adjust it to the desired n_points
    if len(initial_h) != n_points:
        # Resample to n_points
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    
    # Ensure sum of h is correct
    dx = 2.0 / n_points
    total_sum = np.sum(initial_h) * dx
    if abs(total_sum - 1) > 1e-6:
        initial_h = initial_h / total_sum * 1
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Optimization parameters
    max_iterations = 100
    tolerance = 1e-5
    improvement_threshold = 1e-6
    
    # Optimization: Use a heuristic to adjust h values
    for iteration in range(max_iterations):
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s - 1:
            break
        
        # Try to improve h values
        h = best_h.copy()
        h = np.clip(h, 0, 1)  # Ensure h stays within [0, 1]
        dx = 2.0 / n_points

        # Compute all shifts and their overlaps
        shifts = np.arange(-n_points + 1, n_points)
        overlaps = []
        for k in shifts:
            if k < 0:
                shifted_h = np.roll(h, -k)
            else:
                shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            overlaps.append(overlap)
        
        # Find the maximum overlap
        max_overlap = np.max(overlaps)
        
        # Try to improve the solution
        # If the current solution is already good enough, return it
        if max_overlap < best_c5 + improvement_threshold:
            break
        
        # Perturb the solution
        for i in range(n_points):
            # Small random perturbation
            perturbation = np.random.uniform(-0.05, 0.05)
            h[i] += perturbation
            h[i] = np.clip(h[i], 0, 1)  # Ensure within bounds
        
        # Ensure the total sum is correct
        total_sum = np.sum(h) * dx
        if abs(total_sum - 1) > 1e-6:
            # Normalize the sum
            h = h / total_sum * 1
        
        # Evaluate the new solution
        new_c5 = evaluate_erdos_solution(h)
        
        # Update best solution
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
    
    # Evaluate the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Return the best solution, its C5 value, and the number of points
    return (best_h.tolist(), final_c5, n_points)
```
