    best_c5 = float('inf')
    best_h = h_initial
    best_n_points = n_points
    
    for h_start in start_points:
        # Try different n_points sizes to find the best one within the budget
        for n in [n_points, n_points - 5, n_points + 5, 100, 200, 300]:
            if n < 10:
                continue
            # Ensure n is even for symmetric sampling
            n = int(n)
            if n % 2 != 0:
                n += 1
            dx = 2.0 / n
            x = np.linspace(0, 2, n, endpoint=False)
            
            # Initialize h with the start point, scaled to the new n
            h = np.zeros(n)
            if h_start.shape[0] == n_points:
                h[:n_points] = h_start
            else:
                h = np.random.rand(n)
                h = np.clip(h, 0, 1)
                h = h * (n_points / n)  # Scale to maintain average value
            
            # Ensure the integral constraint is met
            total = np.sum(h) * dx
            scaling = 1.0 / total
            h = h * scaling
            
            # Define the objective: minimize max_k (integral(h(x)*(1-h(x+k)) dx)
            # Evaluate for all possible shifts k in [0, 2]
            # Use convolution to compute the correlation
            def evaluate_c5(h_values):
                # Convolve h with (1 - h)
                conv = np.correlate(h_values, 1 - h_values, mode="full")
                # Take the max value over the convolution result
                c5 = np.max(conv) * dx
                return c5
            
            # Use a local optimization method to refine the h
            # We'll use a bounded optimization with a gradient-based solver
            h_var = cp.Variable(n, bounds=(0, 1))
            obj = cp.Minimize(cp.max(cp.pnorm(cp.multiply(h_var, 1 - h_var), 1)))
            constraints = [cp.sum(h_var) * dx == 1.0]
            problem = cp.Problem(obj, constraints)
            try:
                start_time = time.time()
