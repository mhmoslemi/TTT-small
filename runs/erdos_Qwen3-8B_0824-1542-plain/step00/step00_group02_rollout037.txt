def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    one_minus_h = 1.0 - h
    # Compute the cross-correlation
    corr = np.correlate(h, one_minus_h, mode="full")
    # Normalize by dx
    corr /= dx
    # Find the maximum value
    return np.max(corr)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100  # Start with 100 points, can try increasing if time permits
    dx = 2.0 / n_points

    # Initial guess: start with the provided initial_h_values, scale to n_points
    # We'll interpolate or repeat the initial values to match n_points
    if len(initial_h_values) < n_points:
        # Interpolate the initial_h_values to n_points
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    elif len(initial_h_values) > n_points:
        # Downsample the initial_h_values to n_points
        h_initial = initial_h_values[::len(initial_h_values) // n_points]
    else:
        h_initial = initial_h_values.copy()

    # Ensure the initial_h_values sum to n_points / 2 (i.e., integral is 1)
    h_initial = (h_initial * n_points / 2) / np.sum(h_initial)
    
    # Optimization: try to minimize the maximum overlap integral
    # Convert to a problem of minimizing max_k corr_k
    # We'll use a convex optimization approach with a relaxation

    # Set up the optimization variables
    h = cp.Variable(n_points)
    constraints = [
        cp.sum(h) == n_points / 2,
        0 <= h,
        h <= 1
    ]
    # Objective: minimize the maximum correlation
    # We'll use a convex relaxation to approximate this
    # For simplicity, use a heuristic based on random samples of shifts
    shifts = np.random.choice(n_points, size=100, replace=False)
    
    # Define the objective as the maximum of the overlaps for the sampled shifts
    objective = cp.Minimize(cp.Max([cp.max(cp.multiply(h, 1 - cp.roll(h, shift))) for shift in shifts]))
    
    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iter=1000)
    except:
        # If ECOS fails, fall back to another solver
        prob.solve(solver=cp.CVXOPT, verbose=False, max_iter=1000)
    
    # Extract the optimized h_values
    h_optimized = h.value

    elapsed_time = time.time() - start_time

    # Ensure we return the best solution before time runs out
    if elapsed_time > budget_s - 1:
        # If time is nearly up, return the current best solution
        print(f"Time is up, returning current solution with C5 = {c5_bound}")
        return (h_optimized, c5_bound, n_points)

        print(f"Optimized solution with C5 = {c5_bound}")
