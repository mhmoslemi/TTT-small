import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.75, 0.75, 0.75, 0.75, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Start with a refined version of the initial guess
    n_points = 128
    h = initial_h_values.copy()
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    if not np.isclose(sum_h, n_points / 2):
        h = h / sum_h * (n_points / 2)
    
    def objective(h_values):
        h_values = np.clip(h_values, 0.0, 1.0)
        sum_h = np.sum(h_values)
        if not np.isclose(sum_h, n_points / 2):
            h_values = h_values / sum_h * (n_points / 2)
        return evaluate_erdos_solution(h_values)
    
    # Initial optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        h,
        method="L-BFGS-B",
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        options={"maxiter": 100, "ftol": 1e-6}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    remaining_time = budget_s - (time.time() - start_time)
    
    if remaining_time > 1:
        # Try a different starting point with random perturbation
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0.0, 1.0)
        
        result = opt.minimize(
            objective,
            h,
            method="L-BFGS-B",
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            options={"maxiter": 100, "ftol": 1e-6}
        )
        best_h = result.x
        best_c5 = objective(best_h)
    
    # If more time is available, try a larger grid
    if remaining_time > 3:
        n_points = 256
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0.0, 1.0)
        
        def objective_large(h_values):
            h_values = np.clip(h_values, 0.0, 1.0)
            sum_h = np.sum(h_values)
            if not np.isclose(sum_h, n_points / 2):
                h_values = h_values / sum_h * (n_points / 2)
            return evaluate_erdos_solution(h_values)
        
        result = opt.minimize(
            objective_large,
            h,
            method="L-BFGS-B",
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            options={"maxiter": 100, "ftol": 1e-6}
        )
        best_h = result.x
        best_c5 = objective_large(best_h)
    
    # Final check and constraint satisfaction
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    if not np.isclose(sum_h, n_points / 2):
        best_h = best_h / sum_h * (n_points / 2)
    
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
