    # Function to evaluate C₅ for a given h
    def evaluate_c5(h, n_points):
        dx = 2.0 / n_points
        h = np.concatenate([h, h[:2]])  # Extend to handle full convolution
        corr = np.correlate(h, 1 - h[:n_points], mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Function to optimize h to minimize C₅
    def optimize_h(initial_h, n_points, max_iter=50):
        h = cp.Variable(n_points)
        # Objective: minimize the maximum overlap integral
        overlap = cp.Maximize(cp.Variable(n_points + 1))
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        
        # Create a binary variable to track the maximum overlap
        max_overlap = cp.Variable(1)
        for i in range(n_points + 1):
            overlap_i = cp.sum(h[i:i + n_points] * (1 - h[i + n_points:]))
            constraints.append(overlap_i <= max_overlap)
        
        # Objective is to minimize the maximum overlap
        objective = cp.Minimize(max_overlap)
        prob = cp.Problem(objective, constraints)
        start_time = time.time()
        
        # Try different solvers if the initial one fails
        for solver in ['SCIPY', 'ECOS', 'CVXOPT', 'MOSEK', 'GUROBI']:
            try:
                prob.solve(solver=solver, max_iter=max_iter, verbose=False)
                if prob.status == 'optimal':
                    break
            except:
                continue
        
        if prob.status != 'optimal':
            return initial_h
        return h.value
    
    # Start with initial_h_values
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h, initial_n_points)
    
    # Try optimizing with different n_points to find a better solution
    n_points_list = [50, 40, 30, 25, 20, 15, 10]
    for n_points in n_points_list:
        start_time = time.time()
        h = optimize_h(best_h, n_points)
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        c5 = evaluate_c5(h, n_points)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
