    n_points = len(initial_h_values)
    h_values = initial_h_values.copy()
    
    def evaluate_c5(h):
        h = h.reshape(-1)
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(conv) * dx
        return max_overlap
    
    def optimize_h(n_points, h_initial):
        # Normalize the initial values to satisfy the integral constraint
        h = h_initial.copy()
        h = h / (np.sum(h) * 2.0 / n_points)

            return evaluate_c5(x)
        
        result = optimize.minimize(
            objective, 
            x0=h, 
            bounds=bounds, 
            constraints=constraint, 
            method='SLSQP'
        )
        return result.x
    
    def compute_c5(h):
        return evaluate_c5(h)
