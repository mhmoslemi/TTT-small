import numpy as np
import scipy.signal as signal
import time
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 500  # Increase resolution for better results
    dx = 2.0 / n_points
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values.npy' in locals() else np.random.rand(n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    initial_h_values = np.clip(initial_h_values, 0, 1)

    # Define the objective function
    def objective_function(individual):
        h = individual.reshape(-1)
        corr = signal.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(corr)
        return (c5,)

    # Set up genetic algorithm
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)

   toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0, 1)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)

    toolbox.register("evaluate", objective_function)
    toolbox.register("mate", tools.cxTwoPoint)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)

    # Start genetic algorithm
    population = toolbox.population(n=50)
    best_individual = None
    best_c5 = np.inf

    start_time = time.time()
    for gen in range(100):
        offspring = algorithms.varAnd(population, toolbox, cxpb=0.5, mutpb=0.1)
        fits = toolbox.map(toolbox.evaluate, offspring)
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        population = toolbox.select(offspring, len(population))
        best_ind = tools.selBest(population, 1)[0]
        if best_ind.fitness.values[0] < best_c5:
            best_c5 = best_ind.fitness.values[0]
            best_individual = best_ind
        if time.time() - start_time > budget_s:
            break

    h_values = best_individual
    return (h_values, best_c5, n_points)
