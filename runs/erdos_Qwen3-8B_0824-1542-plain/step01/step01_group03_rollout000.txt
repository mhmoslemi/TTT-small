import numpy as np

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.75, 0.75, 0.75, 0.75, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial guess with initial_h_values (n=128 samples)
    n_points = 128
    h = initial_h_values.copy()
    h = np.clip(h, 0.0, 1.0)  # Ensure within bounds
    sum_h = np.sum(h)
    if np.isclose(sum_h, n_points / 2):
        pass  # Already satisfies the constraint
    else:
        # Adjust to meet the constraint
        h = h / sum_h * (n_points / 2)
    
    # Function to evaluate C5
    def objective(h_values):
        h_values = np.clip(h_values, 0.0, 1.0)
        sum_h = np.sum(h_values)
        if not np.isclose(sum_h, n_points / 2):
            h_values = h_values / sum_h * (n_points / 2)
        c5 = evaluate_erdos_solution(h_values)
        return c5
    
    # Function to check constraint
    def constraint(h_values):
        h_values = np.clip(h_values, 0.0, 1.0)
        return np.sum(h_values) - n_points / 2
    
    # Optimization settings
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Try increasing the number of points if time allows
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1:
        n_points_new = 256
        h_new = np.random.rand(n_points_new)
        h_new = h_new / np.sum(h_new) * (n_points_new / 2)  # Ensure constraint is satisfied
        
        def objective_new(h_values):
            h_values = np.clip(h_values, 0.0, 1.0)
            sum_h = np.sum(h_values)
            if not np.isclose(sum_h, n_points_new / 2):
                h_values = h_values / sum_h * (n_points_new / 2)
            c5 = evaluate_erdos_solution(h_values)
            return c5
        
        result_new = opt.minimize(
            objective_new,
            h_new,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points_new)],
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        
        best_h = result_new.x
        best_c5 = objective_new(best_h)
        n_points = n_points_new
    
    # Final check to ensure the constraint is met
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    if not np.isclose(sum_h, n_points / 2):
        best_h = best_h / sum_h * (n_points / 2)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Return the best solution
    return (best_h.tolist(), final_c5, n_points)
