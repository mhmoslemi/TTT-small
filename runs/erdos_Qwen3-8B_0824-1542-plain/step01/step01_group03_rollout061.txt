import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.75, 0.75, 0.75, 0.75, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Define function to evaluate the objective and constraint
    def evaluate(h_values):
        h_values = np.clip(h_values, 0.0, 1.0)
        sum_h = np.sum(h_values)
        if not np.isclose(sum_h, len(h_values) / 2):
            h_values = h_values / sum_h * (len(h_values) / 2)
        c5 = evaluate_erdos_solution(h_values)
        return c5
    
    # Define function to ensure the constraint is satisfied
    def constraint(h_values):
        h_values = np.clip(h_values, 0.0, 1.0)
        return np.sum(h_values) - len(h_values) / 2
    
    # Start with the initial guess
    n_points = 128
    h = initial_h_values.copy()
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    if not np.isclose(sum_h, n_points / 2):
        h = h / sum_h * (n_points / 2)
    
    # Define optimization problem with L-BFGS-B
    result = opt.minimize(
        evaluate,
        h,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    
    best_h = result.x
    best_c5 = evaluate(best_h)
    
    # If time remains, try optimizing with different starting points and larger n_points
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 3:
        n_points = 256
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)  # Ensure constraint is satisfied
        
        def objective_new(h_values):
            h_values = np.clip(h_values, 0.0, 1.0)
            sum_h = np.sum(h_values)
            if not np.isclose(sum_h, n_points / 2):
                h_values = h_values / sum_h * (n_points / 2)
            c5 = evaluate_erdos_solution(h_values)
            return c5
        
        result = opt.minimize(
            objective_new,
            h,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        
        best_h = result.x
        best_c5 = objective_new(best_h)
    
    # Final check to ensure constraint is met
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    if not np.isclose(sum_h, len(best_h) / 2):
        best_h = best_h / sum_h * (len(best_h) / 2)
    
    # Final evaluation
    final_c5 = evaluate(best_h)
    
    # Return the best solution
    return (best_h.tolist(), final_c5, len(best_h))
