import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.75, 0.75, 0.75, 0.75, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial search with 128 points
    n_points = 128
    h_initial = initial_h_values.copy()
    h_initial = np.clip(h_initial, 0.0, 1.0)
    sum_h = np.sum(h_initial)
    if not np.isclose(sum_h, n_points / 2):
        h_initial = h_initial / sum_h * (n_points / 2)
    
    def objective(h_values):
        h_values = np.clip(h_values, 0.0, 1.0)
        sum_h = np.sum(h_values)
        if not np.isclose(sum_h, n_points / 2):
            h_values = h_values / sum_h * (n_points / 2)
        c5 = evaluate_erdos_solution(h_values)
        return c5
    
    def constraint(h_values):
        h_values = np.clip(h_values, 0.0, 1.0)
        return np.sum(h_values) - n_points / 2
    
    # Use L-BFGS-B optimization
    result = opt.minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # If time remains, try increasing the number of points to 256
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1:
        n_points = 256
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        
        def objective_new(h_values):
            h_values = np.clip(h_values, 0.0, 1.0)
            sum_h = np.sum(h_values)
            if not np.isclose(sum_h, n_points / 2):
                h_values = h_values / sum_h * (n_points / 2)
            c5 = evaluate_erdos_solution(h_values)
            return c5
        
        result = opt.minimize(
            objective_new,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        
        best_h = result.x
        best_c5 = objective_new(best_h)
    
    # Final check to ensure constraint is met
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    if not np.isclose(sum_h, n_points / 2):
        best_h = best_h / sum_h * (n_points / 2)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
