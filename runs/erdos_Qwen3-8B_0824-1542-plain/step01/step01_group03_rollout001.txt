import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.75, 0.75, 0.75, 0.75, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Start with a fine grid of 128 points based on the initial construction
    n_points = 128
    h = initial_h_values.copy()
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    if not np.isclose(sum_h, n_points / 2):
        h = h / sum_h * (n_points / 2)
    
    # Evaluate the initial solution
    current_c5 = evaluate_erdos_solution(h)
    best_h = h
    best_c5 = current_c5
    
    # Optimization settings
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    max_iterations = 100
    
    # Step 1: Local optimization with L-BFGS-B
    result = opt.minimize(
        lambda x: evaluate_erdos_solution(x),
        h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': max_iterations, 'ftol': 1e-6}
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Check if we can try increasing n_points
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1:
        n_points = 160
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        
        # Step 2: Local optimization with L-BFGS-B on larger grid
        result = opt.minimize(
            lambda x: evaluate_erdos_solution(x),
            h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iterations, 'ftol': 1e-6}
        )
        
        best_h = result.x
        best_c5 = evaluate_erdos_solution(best_h)
    
    # Step 3: Final constraint check and clipping
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    if not np.isclose(sum_h, n_points / 2):
        best_h = best_h / sum_h * (n_points / 2)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
