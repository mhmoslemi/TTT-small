import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.75, 0.75, 0.75, 0.75, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = np.array([])
    n_points = 0
    
    # Try different n_points values to find the best solution within budget
    for n_points in [64, 128, 256]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate initial guess based on initial_h_values or random
        if n_points == 128:
            h = initial_h_values.copy()
        else:
            h = np.random.rand(n_points)
        
        # Ensure the initial guess satisfies the constraint
        h = h / np.sum(h) * (n_points / 2)
        
        # Define objective function with constraint enforcement
        def objective(x):
            x = np.clip(x, 0.0, 1.0)
            sum_x = np.sum(x)
            if not np.isclose(sum_x, n_points / 2):
                x = x / sum_x * (n_points / 2)
            return evaluate_erdos_solution(x)
        
        # Use L-BFGS-B optimizer
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        
        current_h = result.x
        current_c5 = evaluate_erdos_solution(current_h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        
        # Optional: Try refining with a smaller step size if time permits
        if time.time() - start_time < budget_s - 1:
            result = opt.minimize(
                objective,
                current_h,
                method='L-BFGS-B',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                options={'maxiter': 50, 'ftol': 1e-7, 'gtol': 1e-7}
            )
            current_h = result.x
            current_c5 = evaluate_erdos_solution(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    # Ensure the final solution meets the constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    if not np.isclose(sum_h, n_points / 2):
        best_h = best_h / sum_h * (n_points / 2)
    
    return (best_h.tolist(), best_c5, n_points)
