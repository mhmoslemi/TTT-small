import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

# Pre-imported functions
# evaluate_erdos_solution() and initial_h_values are available

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    
    # Use a finer grid for better accuracy
    n_points = 200  # More points for better resolution
    
    # Initial guess using the provided initial_h_values
    if hasattr(initial_h_values, '__iter__') and len(initial_h_values) > 0:
        initial_x = np.linspace(0, 2, len(initial_h_values))
        initial_h_values_interp = np.interp(np.linspace(0, 2, n_points), initial_x, initial_h_values)
    else:
        # Default initial guess: alternating pattern
        initial_h_values_interp = np.zeros(n_points)
        initial_h_values_interp[::2] = 1.0
    
    # Normalize to meet the integral constraint
    initial_h = initial_h_values_interp * (n_points / 2) / np.sum(initial_h_values_interp)
    
    def objective_function(h):
        # Ensure h is within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integrals for all k
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        # Return the maximum overlap as the objective
        return np.max(overlap)
    
    def constraint(h):
        # Ensure the integral constraint is satisfied
        return np.sum(h) * (2.0 / n_points) - 1.0
    
    # Bounds for h[i] in [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    method = 'L-BFGS-B'
    maxiter = 150
    tol = 1e-5
    
    # Run the optimization
    result = opt.minimize(
        fun=objective_function,
        x0=initial_h,
        bounds=bounds,
        method=method,
        tol=tol,
        options={'maxiter': maxiter}
    )
    
    # Extract the best solution found
    best_h = result.x
    best_c5 = objective_function(best_h)
    
    # Ensure the constraints are met
    final_area = np.sum(best_h) * (2.0 / n_points)
    if abs(final_area - 1.0) > 1e-6:
        best_h = best_h * (1.0 / final_area)
    
    # Final evaluation of the best solution
    final_c5 = objective_function(best_h)
    
    # Check time budget and return
    if time.time() - start_time > budget_s - 1:
        return (best_h, final_c5, n_points)
    
    return (best_h, final_c5, n_points)
