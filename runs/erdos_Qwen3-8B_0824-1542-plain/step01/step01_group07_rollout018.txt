import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial construction with n_points = 50
    initial_n_points = 50
    initial_h = initial_h_values.copy()
    initial_h = initial_h[:initial_n_points]
    
    # Function to evaluate C₅ for a given h
    def evaluate_c5(h, n_points):
        dx = 2.0 / n_points
        h = np.concatenate([h, h[:2]])  # Extend to handle full convolution
        corr = np.correlate(h, 1 - h[:n_points], mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Function to optimize h to minimize C₅
    def optimize_h(initial_h, n_points, max_iter=50):
        h = cp.Variable(n_points)
        # Objective: minimize the maximum overlap integral
        objective = cp.Minimize(0)
        
        # Constraints
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        
        # Use a convex optimization solver
        prob = cp.Problem(objective, constraints)
        start_time = time.time()
        
        # Try different solvers if the initial one fails
        for solver in ['SCIPY', 'ECOS', 'MOSEK', 'GUROBI']:
            try:
                prob.solve(solver=solver, max_iter=max_iter, verbose=False)
                if prob.status == 'optimal':
                    break
            except:
                continue
        
        if prob.status != 'optimal':
            return initial_h
        return h.value
    
    # Start with initial_h_values
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h, initial_n_points)
    
    # Try reducing n_points to optimize faster
    for n_points in [50, 40, 30, 20, 10]:
        start_time = time.time()
        h = optimize_h(best_h, n_points)
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        c5 = evaluate_c5(h, n_points)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    # Try a local search with simulated annealing
    def simulated_annealing(h, n_points, budget_s):
        T = 1.0
        cooling_rate = 0.995
        iter_max = 100
        h_best = h.copy()
        c5_best = evaluate_c5(h_best, n_points)
        
        for _ in range(iter_max):
            if T < 0.01:
                break
            for i in range(n_points):
                # Perturb the current h
                new_h = h.copy()
                new_h[i] += np.random.uniform(-0.1, 0.1)
                new_h = np.clip(new_h, 0, 1)
                new_h[0] = new_h[1]
                new_h[-1] = new_h[-2]
                
                # Check constraint
                if np.sum(new_h) < n_points / 2 - 1e-6 or np.sum(new_h) > n_points / 2 + 1e-6:
                    continue
                
                c5_new = evaluate_c5(new_h, n_points)
                
                # Accept better solution or with some probability
                if c5_new < c5_best or np.random.rand() < np.exp((c5_best - c5_new) / T):
                    h_best = new_h
                    c5_best = c5_new
            
            T *= cooling_rate
        return h_best, c5_best
    
    start_time = time.time()
    best_h, best_c5 = simulated_annealing(best_h, len(best_h), budget_s - (time.time() - start_time))
    
    return best_h, best_c5, len(best_h)
