    best_h = initial_h
    best_c5 = objective(initial_h)

        # Perturb the initial guess
        perturbation = np.random.normal(0, 0.05, size=n_points)
        h_initial = initial_h + perturbation
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Optimize
        start_time = time.time()
        h_opt, c5 = optimize_function(h_initial)
        elapsed_time = time.time() - start_time

        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5

        if elapsed_time > budget_s:
            break

    # Try differential evolution for global search
    if budget_s > 10:
        result_de = differential_evolution(objective, bounds, strategy='best1bin',
