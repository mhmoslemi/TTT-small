    def evaluate_c5(h, n_points):
        dx = 2.0 / n_points
        h = np.concatenate([h, h[:2]])  # Extend to handle full convolution
        corr = np.correlate(h, 1 - h[:n_points], mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Optimized function to minimize C₅ using gradient descent with constraints
    def optimize_h(initial_h, n_points, max_iter=200, learning_rate=0.01, tolerance=1e-5):
        h = initial_h.copy()
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)  # Normalize to satisfy the integral constraint
        
        for _ in range(max_iter):
            # Compute the current C₅ value
            c5 = evaluate_c5(h, n_points)
            
            # Compute the gradient of the max integral with respect to h
            # We approximate the gradient using finite differences
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus / np.sum(h_plus) * (n_points / 2)
                c5_plus = evaluate_c5(h_plus, n_points)
                
                h_minus = h.copy()
                h_minus[i] -= 1e-5
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus / np.sum(h_minus) * (n_points / 2)
                c5_minus = evaluate_c5(h_minus, n_points)
                
                grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)
            
            # Update the weights using gradient descent
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            
            # Check for convergence
            if np.abs(grad).max() < tolerance:
                break
        
        return h
    
    # Start with initial_h_values
    best_h = initial_h.copy()

    for n_points in [50, 40, 30, 20, 10]:

            best_c5 = c5
