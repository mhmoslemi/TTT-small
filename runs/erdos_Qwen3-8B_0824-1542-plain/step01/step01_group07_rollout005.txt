import random
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial construction with n_points = 50
    initial_n_points = 50
    initial_h = initial_h_values.copy()

    # Function to evaluate C₅ for a given h
    def evaluate_c5(h, n_points):
        dx = 2.0 / n_points
        h = np.concatenate([h, h[:2]])  # Extend to handle full convolution
        corr = np.correlate(h, 1 - h[:n_points], mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Genetic Algorithm Setup
    def setup_ga(pop_size, n_points):
        # Define fitness function
        def eval_func(individual):
            h = np.array(individual)
            c5 = evaluate_c5(h, n_points)
            return (c5,)

        # Create fitness and individual classes
        creator.create("FitnessMin", base.Fitness, weights=(1.0,))
        creator.create("Individual", list, fitness=creator.FitnessMin)

        # Setup toolboxes
        toolbox = base.Toolbox()
        toolbox.register("attr_float", random.uniform, 0.0, 1.0)
        toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
        toolbox.register("population", tools.initRepeat, list, toolbox.individual)

        toolbox.register("evaluate", eval_func)
        toolbox.register("mate", tools.cxTwoPoint)
        toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
        toolbox.register("select", tools.selTournament, tournsize=3)

        return toolbox, eval_func
    
    # Function to optimize h to minimize C₅
    def optimize_h(initial_h, n_points, max_gen=100, population_size=50, cxpb=0.5, mutpb=0.1, n_elitism=5):
        # Convert initial_h to a list for DEAP
        initial_h_list = initial_h.tolist()
        toolbox, eval_func = setup_ga(population_size, n_points)
        
        # Initialize population
        pop = toolbox.population(n=population_size)
        
        # Evaluate the entire population
        fits = list(map(eval_func, pop))
        for ind, fit in zip(pop, fits):
            ind.fitness.values = fit
        
        # Apply elitism
        elites = sorted(pop, key=lambda ind: ind.fitness.values[0])[:n_elitism]
        
        # Set up the algorithm
        logbook = tools.Logbook()
        logbook.header = ['gen', 'evals', 'best', 'worst', 'avg']
        
        for gen in range(max_gen):
            offspring = algorithms.varAnd(pop, toolbox, cxpb=cxpb, mutpb=mutpb)
            
            # Evaluate the offspring
            offspring_fit = map(eval_func, offspring)
            for ind, fit in zip(offspring, offspring_fit):
                ind.fitness.values = fit
            
            # Select the next generation
            pop = toolbox.select(pop + offspring, len(pop))
            
            # Apply elitism
            new_pop = elites + sorted(pop, key=lambda ind: ind.fitness.values[0])[:population_size - n_elitism]
            
            # Evaluate the new population
            fits = list(map(eval_func, new_pop))
            for ind, fit in zip(new_pop, fits):
                ind.fitness.values = fit
            
            # Log progress
            record = {
                'gen': gen,
                'evals': len(pop),
                'best': min(fits),
                'worst': max(fits),
                'avg': np.mean(fits)
            }
            logbook.record(**record)
        
        # Select the best individual
        best_ind = min(pop, key=lambda ind: ind.fitness.values[0])
        return np.array(best_ind)
