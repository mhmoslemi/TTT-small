import numpy as np
import random
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial solution
    initial_n_points = len(initial_h_values)
    best_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h, initial_n_points)
    
    # Parameters for genetic algorithm
    population_size = 20
    generations = 50
    max_evaluations = 2000
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Function to evaluate fitness of a solution
    def evaluate_fitness(h, n_points):
        dx = 2.0 / n_points
        h = np.concatenate([h, h[:2]])  # Extend to handle full convolution
        corr = np.correlate(h, 1 - h[:n_points], mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Function to create a new solution through mutation
    def mutate(solution, n_points):
        solution = solution.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                solution[i] = np.random.uniform(0, 1)
        return solution
    
    # Function to crossover two solutions
    def crossover(parent1, parent2, n_points):
        child = np.zeros(n_points)
        for i in range(n_points):
            if random.random() < crossover_rate:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child
    
    # Function to check if a solution meets the sum constraint
    def check_sum_constraint(solution, n_points):
        return np.isclose(np.sum(solution), n_points / 2)
    
    # Main genetic algorithm loop
    start_time = time.time()
    evaluations_used = 0
    population = [np.random.uniform(0, 1, initial_n_points) for _ in range(population_size)]
    
    while evaluations_used < max_evaluations and time.time() - start_time < budget_s:
        # Evaluate the population
        fitness_scores = []
        for solution in population:
            if check_sum_constraint(solution, initial_n_points):
                fitness = evaluate_fitness(solution, initial_n_points)
                fitness_scores.append((fitness, solution))
            else:
                fitness_scores.append((float('inf'), solution))
        
        # Sort by fitness
        fitness_scores.sort()
        best_fitness, best_solution = fitness_scores[0]
        
        # Update the best solution

            best_c5 = best_fitness
            best_h = best_solution.copy()

        parents = [solution for _, solution in fitness_scores[:population_size // 2]]

        while len(new_population) < population_size:
            parent1 = random.choice(parents)
            parent2 = random.choice(parents)
            child = crossover(parent1, parent2, initial_n_points)
            child = mutate(child, initial_n_points)
            if check_sum_constraint(child, initial_n_points):
                new_population.append(child)
            evaluations_used += 1
        
        population = new_population
    
    return best_h, best_c5, len(best_h)
