import scipy.signal as signal
import time
from deap import base, creator, tools, algorithms
import random

def run(seed=42, budget_s=60, **kwargs):

    random.seed(seed)
    n_points = 200  # Number of samples, manageable within time constraints
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    
    # Start with a uniform distribution as an initial guess
    initial_h_values[:] = 0.5
    # Adjust to meet the integral constraint
    initial_h_values[0::2] = 0.5 + 0.5 / 2
    initial_h_values[1::2] = 0.5 - 0.5 / 2
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    
    # Define the objective function for the genetic algorithm
    def evaluate(individual):
        h = np.array(individual)
        # Ensure the integral constraint is satisfied
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        corr = signal.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(corr)
        return c5,
    
    # Set up the genetic algorithm
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    # Initialize the toolbox
    toolbox = base.Toolbox()
    toolbox.register("attr_float", lambda: np.random.uniform(0, 1))
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("evaluate", evaluate)
    toolbox.register("mate", tools.cxTwoPoint)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Initialize population with initial guess
    pop = [initial_h_values.tolist()]
    
    # Run the genetic algorithm
    start_time = time.time()
    for gen in range(100):
        offspring = algorithms.varAnd(pop, toolbox, cxpb=0.5, mutpb=0.1)
        fits = toolbox.map(toolbox.evaluate, offspring)
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        pop = tools.selBest(pop + offspring, len(pop))
        if time.time() - start_time > budget_s:
            break
    
    # Evaluate the best individual
    best_ind = min(pop, key=lambda x: x.fitness.values[0])
    best_h = np.array(best_ind) * (n_points / 2) / np.sum(np.array(best_ind))
    best_h = np.clip(best_h, 0, 1)
    corr = signal.correlate(best_h, 1 - best_h, mode="full") * dx
    c5_bound = np.max(corr)
    
    return (best_h.tolist(), c5_bound, n_points)
