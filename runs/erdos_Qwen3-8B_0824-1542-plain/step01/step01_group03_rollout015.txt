import numpy as np
import scipy.optimize as opt
import scipy.signal as signal

import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.75, 0.75, 0.75, 0.75, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    elapsed_time = 0
    
    # Initial guess with 128 samples
    n_points = 128
    h = initial_h_values.copy()
    h = np.clip(h, 0.0, 1.0)
    h = h / np.sum(h) * (n_points / 2)
    
    best_h = h
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization function
    def objective(h_values):
        h_values = np.clip(h_values, 0.0, 1.0)
        sum_h = np.sum(h_values)
        if not np.isclose(sum_h, n_points / 2):
            h_values = h_values / sum_h * (n_points / 2)
        c5 = evaluate_erdos_solution(h_values)
        return c5
    
    # Constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Perform optimization
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Check if we can try a larger number of points
    if (time.time() - start_time) < budget_s - 5:
        n_points = 256
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        
        def objective_large(h_values):
            h_values = np.clip(h_values, 0.0, 1.0)
            sum_h = np.sum(h_values)
            if not np.isclose(sum_h, n_points / 2):
                h_values = h_values / sum_h * (n_points / 2)
            c5 = evaluate_erdos_solution(h_values)
            return c5
        
        result = opt.minimize(
            objective_large,
            h,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        best_h = result.x
        best_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure constraint is satisfied
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    if not np.isclose(sum_h, n_points / 2):
        best_h = best_h / sum_h * (n_points / 2)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Return the best solution
    return (best_h.tolist(), final_c5, n_points)
