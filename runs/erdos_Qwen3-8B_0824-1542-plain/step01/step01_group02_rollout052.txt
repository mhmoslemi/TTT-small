def run(seed=42, budget_s=60, **kwargs):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    def evaluate_candidate(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        # Check the constraint
        if np.abs(np.sum(h) - (n_points / 2)) > 1e-6:
            return np.inf  # Penalize constraint violation
        return compute_c5(h)

    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 100
    time_budget = budget_s

    population = np.clip(population, 0, 1)
    population = population * (n_points / 2) / np.sum(population, axis=1, keepdims=True)  # Normalize each row

    best_solution = population[0].copy()
    best_c5 = evaluate_candidate(best_solution)

    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > time_budget:
            break

        # Evaluate all candidates
        scores = np.array([evaluate_candidate(ind) for ind in population])
        best_idx = np.argmin(scores)
        best_c5 = scores[best_idx]

        # Early stopping if we already reached a good bound
        if best_c5 <= 0.380876:

        parent_indices = np.argsort(scores)[:5]
        parents = population[parent_indices]

        new_population = []
