import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 400  # Use a slightly higher number of points for better precision
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    
    # Start with a better initial guess based on known constructions
    # A piecewise constant function with 5 equal segments
    # This is a heuristic starting point from known constructions
    initial_h_values[::4] = 1.0
    initial_h_values[1::4] = 0.0
    initial_h_values[2::4] = 1.0
    initial_h_values[3::4] = 0.0
    initial_h_values = np.clip(initial_h_values, 0, 1)
    # Scale to meet the integral constraint
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    initial_h_values = np.clip(initial_h_values, 0, 1)

    def objective_function(h_values):
        h = h_values.reshape(-1)
        # Compute the overlap integral for all k in the range
        corr = correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(corr)
        return c5

    # Optimization using a local optimizer with constraint
    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(
        objective_function,
        initial_h_values,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 200, 'ftol': 1e-6}
    )

    h_values = result.x
    c5_bound = objective_function(h_values)
    return (h_values, c5_bound, n_points)
