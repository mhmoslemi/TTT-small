    def evaluate_c5(h, n_points):

            current_c5 = evaluate_c5(h, n_points)
            
            # Compute the gradient of C₅ with respect to h
            # Approximate gradient using finite differences
            grad = np.zeros(n_points)

                h_plus[i] += 1e-4

                h_plus = h_plus / np.sum(h_plus) * (n_points / 2)

            h -= learning_rate * grad
