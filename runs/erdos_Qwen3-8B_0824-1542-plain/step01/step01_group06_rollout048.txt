import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds
from scipy.optimize import minimize

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points_initial = len(initial_h_values)
    dx_initial = 2.0 / n_points_initial
    initial_c5 = evaluate_erdos_solution(initial_h_values)
    
    # Start with the initial construction
    best_h = initial_h_values.copy()
    best_c5 = initial_c5
    best_n_points = n_points_initial
    
    # Define the function to evaluate C5
    def evaluate_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Define the objective function for optimization
    def objective(h_values):
        return evaluate_c5(h_values)
    
    # Define the constraint for sum(h) = n_points / 2
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Try multiple numbers of points in a more efficient way
    for n_points in [70, 100, 120, 150, 200, 250, 300]:
        dx = 2.0 / n_points
        # Initialize with a more intelligent starting guess
        h_initial = np.zeros(n_points)
        # Create a uniform distribution with some randomness
        h_initial[:n_points // 2] = 1.0
        h_initial[n_points // 2:] = 0.0
        np.random.shuffle(h_initial)
        h_initial = np.sort(h_initial)
        h_initial = np.round(h_initial, 5)
        # Normalize to satisfy the integral constraint
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Optimization settings
        bounds = Bounds(0, 1)
        start_time = time.time()
        result = minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            jac=None,
            options={'maxiter': 50, 'disp': False}
        )
        elapsed_time = time.time() - start_time
        
        if elapsed_time < budget_s:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
                best_n_points = n_points
    
    return (best_h, best_c5, best_n_points)
