import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

# Pre-imported functions
# evaluate_erdos_solution() and initial_h_values are available

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    
    # Use the initial_h_values for a better starting point
    n_initial = len(initial_h_values)
    n_points = 256  # Use 256 points for better resolution
    
    # Interpolate initial_h_values to n_points
    x_initial = np.linspace(0, 2, n_initial)
    x_new = np.linspace(0, 2, n_points)
    h_initial = np.interp(x_new, x_initial, initial_h_values)
    
    # Scale the initial guess to satisfy the integral constraint
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    def objective_function(h):
        # Clip values to [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integrals for all k
        overlap = np.correlate(h, 1 - h, mode="full")
        # Multiply by dx
        overlap *= (2.0 / n_points)
        # Return the maximum overlap
        return np.max(overlap)
    
    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    method = 'SLSQP'
    maxiter = 1000
    tol = 1e-5
    
    # Define constraints: integral must be 1
    def constraint(h):
        return np.sum(h) * (2.0 / n_points) - 1.0
    
    # Run optimization
    result = opt.minimize(
        fun=objective_function,
        x0=h_initial,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        method=method,
        tol=tol,
        options={'maxiter': maxiter}
    )
    
    # Extract best solution
    best_h = result.x
    best_c5 = objective_function(best_h)
    
    # Ensure the constraint is met
    final_area = np.sum(best_h) * (2.0 / n_points)
    if abs(final_area - 1.0) > 1e-6:
        best_h = best_h * (1.0 / final_area)
    
    # Final evaluation
    final_c5 = objective_function(best_h)
    
    # Check time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, final_c5, n_points)
    
    # Return best solution
    return (best_h, final_c5, n_points)
