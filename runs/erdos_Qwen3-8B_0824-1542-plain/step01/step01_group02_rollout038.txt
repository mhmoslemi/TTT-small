import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reduced for faster computation
    dx = 2.0 / n_points

    # Initial guess: Use the provided initial construction
    initial_h = initial_h_values.copy()
    initial_h = np.resize(initial_h, n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint(h):
        h = h.reshape(n_points)
        return np.sum(h) - (n_points / 2)

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)}

    # Use a hybrid approach: start with SLSQP and switch to genetic algorithm for exploration
    start_time = time.time()

    # First phase: SLSQP optimization
    result_slsqp = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-9, 'maxiter': 50})
    elapsed_time = time.time() - start_time

    # If time remains, perform a genetic algorithm phase for exploration
    if elapsed_time < budget_s:
        # Convert to a vector for genetic algorithm
        def ga_objective(h_vec):
            h = h_vec.reshape(n_points)
            h = np.clip(h, 0, 1)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5

        # Genetic algorithm parameters
        population_size = 20
        generations = 10
        mutation_rate = 0.1
        elite_ratio = 0.2

        # Initialize population
        population = np.random.uniform(0, 1, (population_size, n_points))
        population = population * (n_points / 2) / np.sum(population, axis=1, keepdims=True)

        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([ga_objective(ind) for ind in population])
            # Select elites
            elite_indices = np.argsort(fitness)[:int(elite_ratio * population_size)]
            elites = population[elite_indices]
            # Crossover and mutation
            new_population = np.copy(elites)
            for i in range(population_size - len(elites)):
                parent1 = np.random.choice(elite_indices)
                parent2 = np.random.choice(elite_indices)
                child = (population[parent1] + population[parent2]) / 2
                # Mutation
                mutation_mask = np.random.rand(n_points) < mutation_rate
                child[mutation_mask] = np.random.uniform(0, 1, np.sum(mutation_mask))
                child = child * (n_points / 2) / np.sum(child)
                new_population = np.vstack((new_population, child))
            population = new_population[:population_size]

        best_h = population[np.argmin(fitness)]
        result_slsqp = minimize(ga_objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-9, 'maxiter': 30})

    h_values = result_slsqp.x
    c5_bound = objective(h_values)
    return (h_values, c5_bound, n_points)
