import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.75, 0.75, 0.75, 0.75, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Try different starting points and increase n_points if time allows
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    # Initial small n_points (start from 64)
    n_points_list = [64, 96, 128, 160, 192]
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate initial h values
        if n_points == 64:
            h = initial_h_values[:64].copy()
        else:
            h = np.random.rand(n_points)
            h = h / np.sum(h) * (n_points / 2)  # Ensure constraint is satisfied
        
        # Objective function
        def objective(h_values):
            h_values = np.clip(h_values, 0.0, 1.0)
            sum_h = np.sum(h_values)
            # Adjust to meet the constraint
            if not np.isclose(sum_h, n_points / 2):
                h_values = h_values / sum_h * (n_points / 2)
            c5 = evaluate_erdos_solution(h_values)
            return c5
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use L-BFGS-B with more iterations for better optimization
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 200, 'ftol': 1e-7}
        )
        
        current_h = result.x
        current_c5 = objective(current_h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
            best_n_points = n_points
    
    # Final check to ensure constraint is met
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    if not np.isclose(sum_h, best_n_points / 2):
        best_h = best_h / sum_h * (best_n_points / 2)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, best_n_points)
