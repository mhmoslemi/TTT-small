    def evaluate_c5(h, n_points):
        dx = 2.0 / n_points
        h = h.reshape(-1)
        conv = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(conv) * dx
        return max_overlap

    # Function to compute the overlap and return it as the fitness
    def evaluate(individual, n_points):
        return evaluate_c5(individual, n_points),

    def optimize_h(n_points, h_initial):
