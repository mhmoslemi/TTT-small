import scipy.optimize as opt
import time
import scipy.signal as signal

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 500  # Increased number of points for better resolution
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    # Initial guess
    if hasattr(initial_h_values, '__iter__') and len(initial_h_values) > 0:
        initial_h = np.interp(x, np.linspace(0, 2, len(initial_h_values)), initial_h_values)

    initial_h = initial_h * scaling_factor

        h = np.clip(h, 0, 1)
        # Compute the full cross-correlation
        corr = signal.correlate(h, 1 - h, mode='full')
        # Multiply by dx to scale appropriately
        overlap = corr * dx
        return np.max(overlap)
    
    def constraints(h):
        return np.sum(h) * dx - 1.0
