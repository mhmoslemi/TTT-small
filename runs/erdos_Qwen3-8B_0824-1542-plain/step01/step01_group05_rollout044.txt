import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.signal import correlate
from sklearn.gaussian_process.kernels import RBF, WhiteKernel
from deap import base, creator, tools, algorithms
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 200  # A manageable number of points, less than 1k
    dx = 2.0 / n_points

    # Initial construction from the provided initial_h_values
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values.npy' in globals() else np.zeros(n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    initial_h_values = np.clip(initial_h_values, 0, 1)

    # Define the fitness function
    def evaluate(individual):
        h_values = np.array(individual, dtype=np.float64)
        h = h_values.reshape(-1)
        # Ensure the integral constraint is met
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        corr = correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(corr)
        return c5,

    # Genetic algorithm setup
    creator.create("FitnessMin", base.Fitness, weights=(1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)

    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0, 1)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)

    toolbox.register("mate", tools.cxBlend, alpha=0.1)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.05)
    toolbox.register("select", tools.selTournament, tournsize=3)
    toolbox.register("evaluate", evaluate)

    # Start with the provided initial_h_values
    population = [initial_h_values.tolist()]

    # Run the genetic algorithm
    start_time = time.time()
    for gen in range(100):
        offspring = algorithms.varAnd(population, toolbox, cxpb=0.5, mutpb=0.1)
        fits = toolbox.map(toolbox.evaluate, offspring)
        offspring = list(map(lambda ind, fit: ind.fitness.values, offspring, fits))
        population = tools.selBest(population + offspring, len(population))

        # Early termination if time is up
        if time.time() - start_time > budget_s - 5:
            break

    # Find the best solution
    best_individual = min(population, key=lambda ind: ind.fitness.values[0])
    best_h_values = np.array(best_individual)
    best_h_values = best_h_values * (n_points / 2) / np.sum(best_h_values)
    best_h_values = np.clip(best_h_values, 0, 1)
    c5_bound = evaluate(best_h_values)[0]

    return (best_h_values, c5_bound, n_points)
