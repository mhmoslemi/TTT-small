    def evaluate_C5(h, dx):

        # Get the maximum value
        max_corr = np.max(corr)
        # Multiply by dx
        c5 = max_corr * dx
        return c5

    def constraint_sum(h, dx):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1 => sum(h) = n_points / 2

    dx = 2.0 / n_points

    def objective(h):
        return evaluate_C5(h, dx)
    
    # Define bounds: h[i] must be in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use scipy.optimize.minimize to find the h_values that minimize C5
    result = optimize.minimize(
        objective,
        h_initial,
        bounds=bounds,
        constraints=[{"type": "eq", "fun": lambda h: constraint_sum(h, dx)}],
        method="SLSQP",
        options={"ftol": 1e-9, "maxiter": 100}
    )
    
    # Update the best solution found so far
    best_h = result.x
    best_c5 = evaluate_C5(best_h, dx)
