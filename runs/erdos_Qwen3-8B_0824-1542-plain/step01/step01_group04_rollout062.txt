import numpy as np
import scipy.optimize as opt
import time
import scipy.signal as signal

# Pre-imported functions
# evaluate_erdos_solution() and initial_h_values are available

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 100
    # Initial guess from provided initial_h_values
    if hasattr(initial_h_values, '__iter__') and len(initial_h_values) > 0:
        x_coords = np.linspace(0, 2, len(initial_h_values))
        h_values = np.array(initial_h_values)
        x_new = np.linspace(0, 2, n_points)
        initial_h = np.interp(x_new, x_coords, h_values)
    else:
        initial_h = np.zeros(n_points)
        initial_h[::2] = 1.0
        initial_h = np.roll(initial_h, 10)
    
    # Normalize to meet the integral constraint
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    def objective_function(h):
        h = np.clip(h, 0, 1)
        h_pad = np.concatenate([h, np.zeros_like(h)])
        corr = signal.correlate(h, 1 - h, mode='full')
        overlap = corr * (2.0 / n_points)
        return np.max(overlap)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = opt.minimize(
        fun=objective_function,
        x0=initial_h,
        bounds=bounds,
        method='L-BFGS-B',
        tol=1e-5,
        options={'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = objective_function(best_h)
    
    # Final constraint check
    final_area = np.sum(best_h) * (2.0 / n_points)
    if abs(final_area - 1.0) > 1e-6:
        best_h = best_h * (1.0 / final_area)
    
    final_c5 = objective_function(best_h)
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, final_c5, n_points)
    
    return (best_h, final_c5, n_points)
