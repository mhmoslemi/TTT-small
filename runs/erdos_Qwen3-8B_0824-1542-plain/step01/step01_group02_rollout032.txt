import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 150  # Choose a reasonable number of points for performance
    dx = 2.0 / n_points

    # Start from the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - (n_points / 2)

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda h: np.ones(n_points)}

    # Optimization
    start_time = time.time()
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'ftol': 1e-9, 'maxiter': 200})
    elapsed_time = time.time() - start_time

    # Check for convergence
    if elapsed_time > budget_s:
        return (h, 0.38092, n_points)

    h_values = result.x
    c5_bound = objective(h_values)
    return (h_values, c5_bound, n_points)
