import numpy as np

from scipy.signal import correlate
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use a smaller number of points for faster optimization
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    
    # Use the provided initial construction
    initial_h_values = np.array(initial_h_values)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    
    def objective_function(h_values):
        h = h_values.reshape(-1)
        # Compute the overlap integral for all k in the range
        corr = correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(corr)
        return c5

    def bounds_function(x):
        # Bounds for the optimization
        return [(0, 1) for _ in range(n_points)]

    # Use differential evolution for global optimization
    result = differential_evolution(
        objective_function,
        bounds_function,
        strategy='best1bin',
        popsize=15,
        maxiter=50,
        tol=1e-4,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        disp=False
    )

    h_values = result.x
    c5_bound = objective_function(h_values)
    return (h_values, c5_bound, n_points)
