import numpy as np
import scipy.optimize as opt
import time

from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Start with manageable number of points
    dx = 2.0 / n_points
    initial_h_values = np.load('initial_h_values.npy')  # Assume initial construction is loaded from file
    
    # Function to compute the overlap integral
    def compute_c5(h_values):
        h = h_values.reshape(-1)
        corr = correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    # Optimization using SLSQP with constraints
    bounds = [(0, 1) for _ in range(n_points)]
    initial_guess = initial_h_values.copy()
    
    # Define the objective function
    def objective(h_values):
        return compute_c5(h_values)

    # Define the constraint: sum(h) * dx == 1
    def constraint_sum(h_values):
        return np.sum(h_values) * dx - 1

    # Perform optimization
    result = opt.minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'ftol': 1e-6, 'disp': False}
    )

    h_values = result.x
    c5_bound = compute_c5(h_values)
    return (h_values, c5_bound, n_points)
