import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = sum(initial_h)
    
    def normalize(h):
        total = sum(h)
        if total == 0:
            return h
        scale = (n_points / 2) / total
        return [x * scale for x in h]
    
    def objective(h):
        h = np.array(h)
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def evaluate_population(population):
        return [objective(individual) for individual in population]
    
    def crossover(parent1, parent2):
        n = len(parent1)
        child = [0] * n
        for i in range(n):
            if random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child
    
    def mutate(individual, mutation_rate=0.15):
        n = len(individual)
        for i in range(n):
            if random.random() < mutation_rate:
                individual[i] = np.clip(np.random.rand(), 0, 1)
        return individual
    
    def create_initial_population(pop_size, initial_h):
        population = [normalize(initial_h.copy())]
        for _ in range(pop_size - 1):
            h = normalize([np.random.rand() for _ in range(n_points)])
            population.append(h)
        return population
    
    pop_size = 50
    mutation_rate = 0.2
    population = create_initial_population(pop_size, initial_h)
    
    start_time = time.time()
    best_solution = initial_h
    best_c5 = objective(initial_h)
    
    while time.time() - start_time < budget_s:
        fitness = evaluate_population(population)
        parents = [population[i] for i in np.argsort(fitness)[:int(pop_size/2)]]
        
        # Local search on top solutions
        for i in range(len(parents)):
            h = parents[i].copy()
            for j in range(n_points):
                h[j] += np.random.uniform(-0.15, 0.15)
                h = normalize(h)
                if objective(h) < objective(parents[i]):
                    parents[i] = h
        
        new_population = parents[:]
        
        while len(new_population) < pop_size:
            parent1, parent2 = random.choices(parents, k=2)
            child = crossover(parent1, parent2)
            child = mutate(child, mutation_rate)
            new_population.append(child)
        
        population = new_population
        population = [normalize(ind) for ind in population]
        
        current_best = min(population, key=objective)
        current_c5 = objective(current_best)
        
        if current_c5 < best_c5:
            best_solution = current_best
            best_c5 = current_c5
    
    return (best_solution, best_c5, n_points)
