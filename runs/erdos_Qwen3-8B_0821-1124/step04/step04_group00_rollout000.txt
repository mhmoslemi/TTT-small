def run(seed=42, budget_s=10, **kwargs):

    dx = 2.0 / n_points
    sum_h = sum(initial_h)
    
    def normalize(h):
        total = sum(h)
        if total == 0:
            return h
        scale = (n_points / 2) / total
        return [x * scale for x in h]
    
    def objective(h):
        h = np.array(h)
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def evaluate_population(population):
        return [objective(individual) for individual in population]
    
    def differential_evolution(population, fitness, bounds, mutation_rate=0.5, crossover_rate=0.7, generations=100):
        for _ in range(generations):
            for i in range(len(population)):
                a, b, c = np.random.choice(len(population), 3, replace=False)
                mutant = population[a] + mutation_rate * (population[b] - population[c])
                mutant = np.clip(mutant, bounds[0], bounds[1])
                mutant = normalize(mutant)
                if np.random.random() < crossover_rate:
                    population[i] = mutant
                else:
                    population[i] = population[i]
            fitness = evaluate_population(population)
        
        return population
    
    pop_size = 30
    bounds = [0, 1]
    population = [normalize(initial_h.copy()) for _ in range(pop_size)]
    
    start_time = time.time()
