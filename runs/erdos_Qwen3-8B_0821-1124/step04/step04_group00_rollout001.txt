            return h

        h = np.array(h)
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def evaluate_population(population):
        return [objective(individual) for individual in population]
    
    def crossover(parent1, parent2):
        n = len(parent1)
        child = [0] * n
        for i in range(n):
            if random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child
    
    def mutate(individual, mutation_rate=0.1):
        n = len(individual)
        for i in range(n):
            if random.random() < mutation_rate:
                individual[i] = np.clip(np.random.normal(individual[i], 0.15), 0, 1)
        return individual
    
    def create_initial_population(pop_size):
        return [normalize([np.random.rand() for _ in range(n_points)]) for _ in range(pop_size)]

        parents = []
        for _ in range(num_parents):
            min_idx = np.argmin(fitness)
            parents.append(population[min_idx])
            fitness[min_idx] = float('inf')
        return parents
    
    def local_search(h, steps=10):
        h = np.array(h)
        for _ in range(steps):
            idx = np.random.randint(len(h))
