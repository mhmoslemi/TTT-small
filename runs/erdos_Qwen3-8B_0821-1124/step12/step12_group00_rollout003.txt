It starts from the given initial construction and evolves the solution by applying crossover and mutation while maintaining the integral constraint. After generating a diverse population, a local optimization with Sequential Least Squares Programming (SLSQP) is applied to refine the best solution. The algorithm prioritizes maintaining the integral constraint during each step and applies targeted perturbations to regions of high overlap to improve the solution.

import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(x):
        h = np.array(x)
        # Normalize to enforce integral constraint
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = [initial_h.copy() for _ in range(population_size)]
    population = [h / np.sum(h) * (n_points / 2) for h in population]
    
    start_time = time.time()
    
    for gen in range(generations):
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Check time budget
        if time.time() - start_time > budget_s:
            return (best_h, best_c5, n_points)
        
        # Create new population
        new_population = []
        for _ in range(population_size):
            # Select parents
            parent1 = random.choice(population)
            parent2 = random.choice(population)
            
            # Crossover
            if random.random() < crossover_rate:
                child = np.zeros(n_points)
                for i in range(n_points):
                    if random.random() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
            else:
                child = parent1.copy()
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] += np.random.normal(0, 0.05)
                    child[i] = np.clip(child[i], 0, 1)
            
            # Normalize
            child = child / np.sum(child) * (n_points / 2)
            new_population.append(child)
        
        population = new_population
    
    # Final optimization with local search
    best_idx = np.argmin([objective(h) for h in population])
    best_h = population[best_idx]
    result = optimize.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
