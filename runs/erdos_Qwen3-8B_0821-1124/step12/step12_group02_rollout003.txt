import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)

    def objective(x):
        h = np.array(x)
        # Enforce integral constraint directly
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization
    x0 = initial_h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    
    # Initial optimization
    result = optimize.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Genetic algorithm for exploration
    def genetic_algorithm(pop_size=20, generations=50):
        population = [np.random.rand(n_points) for _ in range(pop_size)]
        population = [h / np.sum(h) * (n_points / 2) for h in population]
        
        for gen in range(generations):
            if time.time() - start_time > budget_s:
                break
            # Evaluate fitness
            fitness = [objective(h) for h in population]
            best_idx = np.argmin(fitness)
            best_fitness = fitness[best_idx]
            
            # Apply crossover and mutation
            new_population = []
            for _ in range(pop_size // 2):
                parent1 = population[np.random.choice(pop_size)]
                parent2 = population[np.random.choice(pop_size)]
                child1 = np.minimum(parent1, parent2) * 0.5 + np.maximum(parent1, parent2) * 0.5

                # Mutation
                child1 += np.random.normal(0, 0.05, size=n_points)

                if current_fitness < best_fitness:

                    best_h = h
        
        return best_h, best_fitness
    
    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm()
    
    return (best_h, best_c5, n_points)
