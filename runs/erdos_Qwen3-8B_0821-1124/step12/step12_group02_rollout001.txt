        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.7
    generations = 50
    
    def create_individual():
        return np.random.uniform(0, 1, n_points)
    
    def mutate(individual):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(n_points):
            if np.random.rand() < crossover_rate:
                child[i] = parent2[i]
        return child
    
    population = [create_individual() for _ in range(population_size)]
    
    start_time = time.time()
    
    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness_scores = [fitness(ind) for ind in population]
        parents = np.argsort(fitness_scores)[:int(population_size/2)]
        new_population = []
        
        for i in range(int(population_size/2)):
            parent1 = population[parents[i % len(parents)]]
            parent2 = population[parents[(i+1) % len(parents)]]
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
    
    best_h = min(population, key=fitness)
    best_c5 = fitness(best_h)
