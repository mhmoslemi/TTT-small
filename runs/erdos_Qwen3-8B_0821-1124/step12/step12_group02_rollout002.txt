import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    
    def objective(x):
        h = np.array(x)
        # Enforce integral constraint directly
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Generate multiple initial guesses
    initial_guesses = []
    for _ in range(5):

    start_time = time.time()
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        # Optimization
        result = optimize.minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
    
    # Additional refinement
    if best_h is not None:

            if time.time() - start_time > budget_s:
                break
            # Targeted perturbation
            perturb = np.random.uniform(-0.1, 0.1, size=n_points)

            # Optimization
            result = optimize.minimize(
                objective,
                new_h,

                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],

            )
            
            if result.success and result.fun < best_c5:
