import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)

    target_sum = n_points / 2
    
    def objective(h):
        h = np.array(h)
        if abs(np.sum(h) - target_sum) > 1e-6:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    def constraint_bounds(h):
        return np.all((0 <= h) & (h <= 1))
    
    def evaluate_solution(h):
        return objective(h)
    
    def optimize_solution(initial_guess):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [
            {'type': 'eq', 'fun': constraint_sum},
            {'type': 'ineq', 'fun': constraint_bounds}
        ]
        result = optimize.minimize(
            evaluate_solution,
            initial_guess,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        return result.x
    
    best_solution = np.array(initial_h)
    best_c5 = objective(best_solution)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        optimized = optimize_solution(best_solution)
        current_c5 = objective(optimized)
        if current_c5 < best_c5:
            best_solution = optimized
            best_c5 = current_c5
