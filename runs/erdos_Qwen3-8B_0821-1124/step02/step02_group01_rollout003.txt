    n_points = len(initial_h)

    h = np.array(initial_h, dtype=float)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h) * dx
        if abs(sum_h - 1) > 1e-8:
            return np.inf
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    def normalize_h(h):
        h = h.copy()
        total = np.sum(h) * dx
        if abs(total - 1) > 1e-8:
            h = h / total * 1
        return h
    
    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child
    
    def mutate(individual):
        individual = individual.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        return normalize_h(individual)
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    population = [normalize_h(np.random.rand(n_points)) for _ in range(20)]
