import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h, dtype=float)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h) * dx
        if abs(sum_h - 1) > 1e-8:
            return np.inf
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    def normalize(h):
        total = np.sum(h) * dx
        if total == 0:
            return h
        scale = 1.0 / total
        return h * scale / dx
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initial population
    population = [np.copy(h) for _ in range(population_size)]
    population = [normalize(ind) for ind in population]
    
    best_h = np.copy(h)
    best_c5 = objective(best_h)
    time_start = time.time()
    
    for gen in range(generations):
        if time.time() - time_start > budget_s:
            break
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        # Select parents
        parents = np.random.choice(population_size, size=population_size, p=np.array(fitness)/np.sum(fitness))
        parents = [population[i] for i in parents]
        # Crossover
        offspring = []
        for i in range(population_size//2):
            if np.random.rand() < crossover_rate:
                p1, p2 = parents[2*i], parents[2*i+1]
                child = np.concatenate((p1[:n_points//2], p2[n_points//2:]))
            else:
                child = np.copy(parents[i])
            offspring.append(child)
        # Mutation
        for i in range(population_size):
            if np.random.rand() < mutation_rate:
                idx = np.random.choice(n_points, size=2, replace=False)
                for j in idx:
                    delta = np.random.uniform(-0.1, 0.1)
                    offspring[i][j] = np.clip(offspring[i][j] + delta, 0, 1)
        # Replace population
        population = [normalize(ind) for ind in offspring]
        # Update best solution
        for ind in population:
            c5 = objective(ind)
            if c5 < best_c5:
                best_h = np.copy(ind)
                best_c5 = c5
    
    return (best_h.tolist(), best_c5, n_points)
