import time

def run(seed=42, budget_s=10, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h, dtype=float)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h) * dx
        if abs(sum_h - 1) > 1e-8:
            return np.inf
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    def normalize(h):
        total = np.sum(h) * dx
        if total == 0:
            return h
        scale = 1.0 / total
        return h * scale / dx
    
    def mutate(h, mutation_rate):
        h_new = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                delta = np.random.uniform(-0.2, 0.2)
                h_new[i] = np.clip(h_new[i] + delta, 0, 1)
        return h_new
    
    def crossover(parent1, parent2):
        h_new = np.zeros_like(parent1)
        for i in range(len(h_new)):
            if np.random.rand() < 0.5:
                h_new[i] = parent1[i]
            else:
                h_new[i] = parent2[i]
        return h_new
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    population = [h.copy() for _ in range(50)]
    time_start = time.time()
    
    for step in range(200):
        if time.time() - time_start > budget_s:
            break
        new_population = []
        for _ in range(50):
            parent1, parent2 = np.random.choice(population, size=2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child, 0.1)
            child = normalize(child)
            new_population.append(child)
        population = new_population
        for candidate in population:
            c5 = objective(candidate)
            if c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = c5
    
    return (best_h.tolist(), best_c5, n_points)
