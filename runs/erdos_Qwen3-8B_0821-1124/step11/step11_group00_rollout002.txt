import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(x):
        h = np.array(x)
        # Normalize to enforce integral constraint
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization
    x0 = initial_h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    
    # First optimization with SLSQP
    result = optimize.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    # Perturbation phase
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Genetic algorithm for global exploration
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    
    for _ in range(generations):
        population = [best_h + np.random.normal(0, 0.05, n_points) for _ in range(population_size)]
        population = [np.clip(p, 0, 1) for p in population]
        population = [p / np.sum(p) * (n_points / 2) for p in population]
        
        # Evaluate population
        fitness = [objective(p) for p in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Apply mutation
        for i in range(population_size):
            if np.random.rand() < mutation_rate:
                population[i] += np.random.normal(0, 0.05, n_points)
                population[i] = np.clip(population[i], 0, 1)
                population[i] = population[i] / np.sum(population[i]) * (n_points / 2)
        
        # Replace worst with best_h
        worst_idx = np.argmax(fitness)
        population[worst_idx] = best_h
    
    # Final optimization
    result = optimize.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    # Check if we have exceeded the time budget
    if time.time() - start_time > budget_s:
        return (initial_h, objective(initial_h), n_points)
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
