import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(x):
        h = np.array(x)
        # Normalize to enforce integral constraint
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.15
    elite_size = 2
    
    # Initialize population
    population = [initial_h.copy() for _ in range(population_size)]
    population = [h / np.sum(h) * (n_points / 2) for h in population]
    
    start_time = time.time()
    
    for generation in range(generations):
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        # Select elite
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = [population[i] for i in elite_indices]
        # Create new population
        new_population = elite.copy()
        while len(new_population) < population_size:
            # Select parents
            parent1 = np.random.choice(elite)
            parent2 = np.random.choice(elite)
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0, 1)
            # Normalize
            child = child / np.sum(child) * (n_points / 2)
            new_population.append(child)
        population = new_population
        
        # Check if time is up
        if time.time() - start_time > budget_s:
            break
    
    # Find best solution
    best_h = min(population, key=objective)
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
