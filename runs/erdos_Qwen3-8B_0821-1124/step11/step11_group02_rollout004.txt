import numpy as np
import random
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def fitness(h):
        c5 = evaluate(h)
        # Penalize deviation from the integral constraint
        integral_error = np.abs(np.sum(h) - n_points / 2)
        return c5 + 100 * integral_error
    
    def crossover(parent1, parent2):
        split = random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:split], parent2[split:]])
        return child
    
    def mutate(h):
        for i in range(n_points):
            if random.random() < 0.1:
                h[i] = np.random.uniform(0, 1)
        return h
    
    population_size = 50
    population = [np.copy(initial_h) for _ in range(population_size)]
    
    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness_scores = [fitness(ind) for ind in population]
        best_idx = np.argmin(fitness_scores)
        best_h = population[best_idx]
        best_c5 = fitness_scores[best_idx]
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Select parents
            parent1, parent2 = random.choices(population, k=2)
            # Crossover
            child = crossover(parent1, parent2)
            # Mutate
            child = mutate(child)
            # Ensure integral constraint
            child_sum = np.sum(child)
            if child_sum != n_points / 2:
                child = child / child_sum * (n_points / 2)
            new_population.append(child)
        
        population = new_population
    
    # Find the best solution
    best_idx = np.argmin([fitness(ind) for ind in population])
    best_h = population[best_idx]
    best_c5 = evaluate(best_h)
    
    return (best_h, best_c5, n_points)
