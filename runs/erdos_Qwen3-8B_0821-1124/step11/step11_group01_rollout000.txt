        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def normalize(h):
        return h / np.sum(h) * (n_points / 2)

    population_size = 50
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    
    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        
        # Select top 20% as parents
        parents = np.array(population)[np.argsort(fitness)[:int(population_size * 0.2)]]
        
        # Create new population through crossover and mutation
        new_population = []
        for _ in range(population_size):
            # Crossover
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            child = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    child[i] = np.random.uniform(0, 1)
            
            # Normalize
            child = normalize(child)
            new_population.append(child)
        
        population = new_population
    
    # Find the best solution
    best_index = np.argmin([objective(ind) for ind in population])
    best_h = population[best_index]
