import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        # Enforce integral constraint directly
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Initial population
    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    
    def evaluate(individual):
        return objective(individual)
    
    # Genetic algorithm parameters
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 100
    
    start_time = time.time()
    
    for generation in range(generations):
        # Evaluate fitness
        fitness = [evaluate(ind) for ind in population]
        # Find best solution
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
        
        # Check time budget
        if time.time() - start_time > budget_s:
            return (best_h, best_c5, n_points)
        
        # Create new population
        new_population = []
        for i in range(population_size):
            # Select parents
            parent1 = random.choices(population, weights=fitness)[0]
            parent2 = random.choices(population, weights=fitness)[0]
            
            # Crossover
            if random.random() < crossover_rate:
                child = np.zeros(n_points)
                for j in range(n_points):
                    if random.random() < 0.5:
                        child[j] = parent1[j]
                    else:
                        child[j] = parent2[j]
            else:
                child = parent1.copy()
            
            # Mutation
            for j in range(n_points):
                if random.random() < mutation_rate:
                    child[j] = np.random.uniform(0, 1)
            
            # Ensure integral constraint
            child = child / np.sum(child) * (n_points / 2)
            new_population.append(child)
        
        population = new_population
    
    return (best_h, best_c5, n_points)
