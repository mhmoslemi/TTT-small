import random

        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    population_size = 20
    generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.8

    population = [initial_h.copy() for _ in range(population_size)]
    population = [h / np.sum(h) * (n_points / 2) for h in population]
    
    start_time = time.time()

        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)

        best_c5 = fitness[best_idx]
        
        # Check time budget
        if time.time() - start_time > budget_s:
            return (best_h, best_c5, n_points)
        
        # Create new population
        new_population = []
        for _ in range(population_size):
            # Select parents
            parent1 = random.choice(population)
            parent2 = random.choice(population)
            
            # Crossover
            if random.random() < crossover_rate:
                child = np.zeros(n_points)

                    if random.random() < 0.5:
                        child[i] = parent1[i]

            # Mutation
