    start_time = time.time()
    
    for _ in range(generations):
        # Evaluate fitness
        fitness_vals = [fitness(ind) for ind in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_vals)
        population = [population[i] for i in sorted_indices]
        
        # Select elites
        elites = population[:elite_size]
        
        # Create new population
        new_population = elites.copy()
        
        while len(new_population) < pop_size:
            # Select parents
            parent1 = random.choice(population[:pop_size//2])
            parent2 = random.choice(population[:pop_size//2])
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            
            # Mutation
            for i in range(n_points):

                    child[i] = np.random.uniform(0, 1)
