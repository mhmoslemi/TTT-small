import random
import time

        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def mutate(h, mutation_rate=0.05):
        h = h.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                h[i] += np.random.normal(0, 0.05)
                h[i] = np.clip(h[i], 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    def crossover(parent1, parent2):
        child = parent1.copy()
        for i in range(n_points):
            if random.random() < 0.5:
                child[i] = parent2[i]
        child = child / np.sum(child) * (n_points / 2)
        return child
    
    best_h = initial_h.copy()
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    population = [mutate(best_h) for _ in range(20)]
    for _ in range(100):
        if time.time() - start_time > budget_s:

        best_h = population[0]
