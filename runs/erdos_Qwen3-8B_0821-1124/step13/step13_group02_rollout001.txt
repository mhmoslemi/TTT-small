import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(x):
        h = np.array(x)
        # Normalize to enforce integral constraint
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization
    x0 = initial_h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    
    # First optimization with SLSQP
    result = optimize.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Genetic algorithm for global exploration
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    population = [best_h.copy() for _ in range(population_size)]
    
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        # Select parents
        parents = [population[i] for i in np.argsort(fitness)[:population_size//2]]
        # Generate offspring
        offspring = []
        for _ in range(population_size//2):
            if random.random() < crossover_rate:
                parent1, parent2 = random.sample(parents, 2)
                child = np.zeros_like(parent1)
                for i in range(n_points):
                    if random.random() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
            else:
                child = np.copy(random.choice(parents))
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] += np.random.normal(0, 0.05)
                    child[i] = np.clip(child[i], 0, 1)
            # Enforce integral constraint
            child = child / np.sum(child) * (n_points / 2)
            offspring.append(child)
        # Replace population
        population = parents + offspring
        # Update best solution
        for ind in population:
            current_c5 = objective(ind)
            if current_c5 < best_c5:
                best_h = ind.copy()
                best_c5 = current_c5
    
    # Final optimization
    result = optimize.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    # Check if we have exceeded the time budget
    if time.time() - start_time > budget_s:
        return (initial_h, objective(initial_h), n_points)
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
