import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(x):
        h = np.array(x)
        # Normalize to enforce integral constraint
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization
    x0 = initial_h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    
    # First optimization with L-BFGS-B
    result = optimize.minimize(
        objective,
        x0,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Adaptive mutation genetic algorithm
    population_size = 20
    generations = 15
    mutation_rate = 0.15
    elite_size = 2
    
    population = [best_h.copy() for _ in range(population_size)]
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        # Create new population
        new_population = elites.copy()
        for _ in range(population_size - elite_size):
            # Select parents
            parent1 = random.choice(elites)
            parent2 = random.choice(elites)
            # Crossover
            child = np.copy(parent1)

                    child[i] = parent2[i]
            # Mutate

                    child[i] = np.clip(child[i] + np.random.normal(0, 0.05), 0, 1)
            # Normalize
            child = child / np.sum(child) * (n_points / 2)
            new_population.append(child)
        population = new_population
    
    # Final optimization
    result = optimize.minimize(
        objective,
        best_h,
        method='L-BFGS-B',

        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    # Check if we have exceeded the time budget
    if time.time() - start_time > budget_s:
        return (initial_h, objective(initial_h), n_points)

    best_c5 = objective(best_h)
