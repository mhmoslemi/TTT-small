        return c5

        fitness = [objective(ind) for ind in population]
        
        # Find the best individual
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]

        # Create new population through crossover and mutation
        new_population = []
        for _ in range(population_size):
