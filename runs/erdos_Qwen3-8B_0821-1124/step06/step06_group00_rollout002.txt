import random
import time

            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def perturb(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] = np.clip(h[i] + random.uniform(-0.2, 0.2), 0, 1)
        return h
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:

        new_c5 = evaluate(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
    
    return (best_h, best_c5, n_points)
