import numpy as np
import scipy.optimize as optimize
import time
import copy

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.array(h)
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def normalize(h):
        h = np.array(h)
        total = np.sum(h)
        if abs(total) < 1e-10:
            return np.zeros_like(h)
        return h * (n_points / 2) / total
    
    def mutate(h, mutation_rate=0.1):
        h = np.array(h)
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        return h
    
    def crossover(parent1, parent2):
        h = np.array(parent1)
        for i in range(len(h)):
            if np.random.rand() < 0.5:
                h[i] = parent2[i]
        return h
    
    population_size = 20
    population = [normalize(mutate(copy.copy(initial_h))) for _ in range(population_size)]
    best_solution = population[0]
    best_c5 = evaluate(best_solution)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        
        # Select parents
        parents = np.random.choice(population, size=population_size, p=np.array(fitness)/np.sum(fitness))
        
        # Create new population
        new_population = []
        for _ in range(population_size):
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            child = crossover(parent1, parent2)
            child = normalize(mutate(child))
            new_population.append(child)
        
        population = new_population
        
        # Update best solution
        for ind in population:
            c5 = evaluate(ind)
            if c5 < best_c5:
                best_c5 = c5
                best_solution = ind
    
    return (best_solution, best_c5, n_points)
