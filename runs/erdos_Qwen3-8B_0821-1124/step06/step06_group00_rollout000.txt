import random

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = np.array(h)
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h):
        h = np.array(h)
        return np.sum(h) - n_points / 2
    
    # Initialize population
    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    best_solution = initial_h.copy()
    best_c5 = objective(best_solution)
    
    start_time = time.time()
    
    for _ in range(100):  # Number of generations
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        fitness = np.array(fitness)
        
        # Select parents
        parents_indices = np.argsort(fitness)[:population_size//2]
        parents = [population[i] for i in parents_indices]
        
        # Crossover
        offspring = []
        for _ in range(population_size//2):
            parent1 = random.choice(parents)
            parent2 = random.choice(parents)
            child = np.zeros_like(parent1)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            child = offspring[i]
            for j in range(n_points):
                if random.random() < 0.1:
                    child[j] = np.random.uniform(0, 1)
            offspring[i] = child
        
        # Replace population
        population = parents + offspring
        
        # Update best solution
        for ind in population:
            current_c5 = objective(ind)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_solution = ind.copy()
    
    return (best_solution, best_c5, n_points)
