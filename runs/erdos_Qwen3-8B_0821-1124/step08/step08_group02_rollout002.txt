import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = np.array(h)
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Initialize population
    population_size = 50
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = (population - np.mean(population, axis=1, keepdims=True)) / np.std(population, axis=1, keepdims=True)
    population = np.clip(population, 0, 1)
    for i in range(population_size):
        population[i] = initial_h.copy()
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    # Genetic algorithm parameters
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        
        # Select parents
        parents_indices = np.argsort(fitness)[:int(population_size/2)]
        parents = population[parents_indices]
        
        # Create offspring
        offspring = []
        for _ in range(population_size - len(parents)):
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            else:
                child = parent1.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0, 1)
            
            # Maintain integral constraint
            child = child / np.sum(child) * (n_points / 2)
            
            offspring.append(child)
        
        # Replace population
        population = np.vstack((parents, offspring))
        population = np.random.permutation(population)
        
        # Update best solution
        for ind in population:
            current_c5 = objective(ind)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = ind.copy()
    
    return (best_h, best_c5, n_points)
