import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Increased number of samples for better accuracy
    dx = 2.0 / n_points
    h = np.random.rand(n_points)
    h = h / np.sum(h) * (n_points / 2)  # Initialize with random values satisfying the integral constraint
    
    def objective(x):

        # Normalize to maintain integral constraint

        return c5

    start_time = time.time()

        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    
    # Check if we have exceeded the time budget
    if time.time() - start_time > budget_s:
        return (h, objective(h), n_points)
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
