import numpy as np
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def normalize(h):
        total = np.sum(h)
        if abs(total) < 1e-10:
            return np.zeros_like(h)
        return h * (n_points / 2) / total
    
    def crossover(parent1, parent2):
        n = len(parent1)
        child = np.zeros(n)
        for i in range(n):
            if np.random.rand() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if np.random.random() < mutation_rate:
                h[i] = np.clip(h[i] + np.random.uniform(-0.2, 0.2), 0, 1)
        return h
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    population_size = 20
    population = [normalize(mutate(initial_h.copy(), 0.1)) for _ in range(population_size)]
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        # Find the best individual
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_best_c5 = fitness[best_idx]
        if current_best_c5 < best_c5:
            best_h = current_best
            best_c5 = current_best_c5
        # Select parents
        parents = np.random.choice(population_size, size=population_size, replace=True, p=np.array(fitness)/np.sum(fitness))
        # Create new population
        new_population = []
        for i in range(population_size):
            parent1 = population[parents[i]]
            parent2 = population[np.random.choice(population_size, size=1, replace=False, p=np.array(fitness)/np.sum(fitness))[0]]
            child = crossover(parent1, parent2)
            child = normalize(mutate(child, 0.1))
            new_population.append(child)
        population = new_population
    
    return (best_h, best_c5, n_points)
