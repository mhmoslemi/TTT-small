import random
import time

def run(seed=42, budget_s=10, **kwargs):

    dx = 2.0 / n_points
    
    def objective(h):
        h = np.array(h)
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Initialize population
    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    
    start_time = time.time()
    
    for _ in range(100):  # Max generations
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Select parents
        parents = np.random.choice(population_size, size=population_size, p=np.array(fitness)/np.sum(fitness))
        parents = [population[i] for i in parents]
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if i+1 < population_size:
                parent1 = parents[i]
                parent2 = parents[i+1]
                child1 = parent1.copy()
                child2 = parent2.copy()
                
                # Crossover point
                crossover_point = np.random.randint(1, n_points - 1)
                child1[crossover_point:] = parent2[crossover_point:]
                child2[crossover_point:] = parent1[crossover_point:]
                
                offspring.append(child1)
                offspring.append(child2)
        
        # Mutation
        for i in range(len(offspring)):

            for j in range(n_points):
                if random.random() < 0.1:  # Mutation rate
                    ind[j] += np.random.uniform(-0.1, 0.1)

        # Replace population
        population = offspring
        
        # Ensure integral constraint
        for i in range(len(population)):
            population[i] = population[i] / np.sum(population[i]) * (n_points / 2)
    
    return (best_h, best_c5, n_points)
