import numpy as np
import random
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = np.array(h)
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Initialize population
    population_size = 30
    population = [initial_h.copy() for _ in range(population_size)]
    
    start_time = time.time()
    iteration = 0
    
    while iteration < 100 and time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        
        # Select parents
        parents = np.argsort(fitness)[:population_size//2]
        parents = [population[i] for i in parents]
        
        # Crossover
        new_population = []
        for _ in range(population_size):
            parent1 = random.choice(parents)
            parent2 = random.choice(parents)
            child = np.zeros(n_points)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            new_population.append(child)
        
        # Mutate
        for i in range(len(new_population)):
            for j in range(n_points):
                if random.random() < 0.1:
                    new_population[i][j] = np.clip(new_population[i][j] + np.random.normal(0, 0.1), 0, 1)
        
        # Normalize to maintain integral constraint
        for i in range(len(new_population)):
            new_population[i] = new_population[i] / np.sum(new_population[i]) * (n_points / 2)
        
        # Replace population
        population = new_population
        
        # Track best solution
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        iteration += 1
    
    return (best_h, best_c5, n_points)
