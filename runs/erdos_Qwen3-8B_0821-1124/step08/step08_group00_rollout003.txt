import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(x):
        h = np.array(x)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def evaluate_population(population):
        return [objective(ind) for ind in population]
    
    def mutate(individual, mutation_rate=0.1):
        mutated = individual.copy()
        for i in range(len(mutated)):
            if random.random() < mutation_rate:
                mutated[i] = np.random.uniform(0, 1)
        return mutated
    
    def crossover(parent1, parent2):
        child = np.random.rand(n_points)
        for i in range(n_points):
            if random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child
    
    # Ensure the initial population satisfies the integral constraint
    def normalize(individual):
        total = np.sum(individual)
        factor = (n_points / 2) / total if total != 0 else 1.0
        return individual * factor
    
    # Create initial population
    population_size = 20
    population = [normalize(np.random.rand(n_points)) for _ in range(population_size)]
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = evaluate_population(population)
        
        # Find the best individual
        best_index = np.argmin(fitness)
        best_c5_current = fitness[best_index]
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = population[best_index].copy()
        
        # Select parents
        parents = np.random.choice(population, size=10, replace=False)
        
        # Create new population
        new_population = []
        for _ in range(population_size):
            parent1, parent2 = np.random.choice(parents, size=2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            child = normalize(child)
            new_population.append(child)
        
        population = new_population
    
    return (best_h, best_c5, n_points)
