import random

    dx = 2.0 / n_points
    
    def evaluate(h):

            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    def low_pass_filter(h, window_size=5):
        window = np.hanning(window_size)
        return np.convolve(h, window, mode='same') / np.sum(window)
    
    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        for i in range(len(parent1)):
            if np.random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if np.random.random() < mutation_rate:
                h[i] = np.clip(h[i] + np.random.uniform(-0.15, 0.15), 0, 1)
        return h
    
    def create_individual():
        h = np.random.rand(n_points)
        return normalize(h)
    
    population_size = 50
    population = [create_individual() for _ in range(population_size)]
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        # Find the best individual
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]

        worst_idx = np.argmax(fitness)
        population[worst_idx] = create_individual()

        for i in range(population_size // 2):
            parent1 = random.choice(population)
            parent2 = random.choice(population)
            child = crossover(parent1, parent2)
            child = mutate(child)
            child = normalize(child)
            population[i] = child
