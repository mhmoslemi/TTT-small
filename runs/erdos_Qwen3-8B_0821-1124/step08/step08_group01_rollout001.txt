import numpy as np
import time

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    population_size = 20
    crossover_rate = 0.8
    mutation_rate = 0.1
    generations = 100
    
    def mutate(h):
        h = h.copy()
        for i in range(len(h)):
            if np.random.random() < mutation_rate:
                h[i] = np.clip(h[i] + np.random.uniform(-0.1, 0.1), 0, 1)
        return h
    
    def crossover(parent1, parent2):
        child = parent1.copy()
        for i in range(len(child)):
            if np.random.random() < crossover_rate:
                child[i] = parent2[i]
        return child
    
    def normalize(h):
        return h / np.sum(h) * (n_points / 2)
    
    population = [normalize(mutate(initial_h.copy())) for _ in range(population_size)]
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate all
        fitness = [evaluate(ind) for ind in population]
        # Find the best
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        # Create new population
        new_population = []

            parent1 = population[np.random.randint(population_size)]
            parent2 = population[np.random.randint(population_size)]
            # Crossover
            child = crossover(parent1, parent2)
            # Mutate
            child = mutate(child)
            # Normalize
            child = normalize(child)
            new_population.append(child)
        population = new_population
