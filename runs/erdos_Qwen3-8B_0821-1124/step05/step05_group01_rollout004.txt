import scipy.optimize as optimize
import time
import random

        if total == 0:
            return h
        scale = (n_points / 2) / total
        return [x * scale for x in h]
    
    def objective(h):
        h = np.array(h)
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def evaluate_population(population):
        return [objective(individual) for individual in population]
    
    def crossover(parent1, parent2):
        n = len(parent1)
        child = [0] * n
        for i in range(n):
            if random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child
    
    def mutate(individual, mutation_rate=0.1):
        n = len(individual)
        for i in range(n):
            if random.random() < mutation_rate:
                individual[i] = np.clip(np.random.normal(individual[i], 0.1), 0, 1)
        return individual
    
    def local_optimize(h):
        def local_objective(x):
            h_local = np.copy(h)
            for i in range(len(h_local)):
                h_local[i] = x[i]
            return objective(h_local)
        
        bounds = [(0, 1) for _ in range(len(h))]
        result = optimize.minimize(local_objective, h, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}])
        return result.x
    
    def tournament_selection(population, fitness, tournament_size=3):
        selected = []
        for _ in range(len(population)):
            contestants = random.sample(list(zip(population, fitness)), tournament_size)
            winner = min(contestants, key=lambda x: x[1])[0]
            selected.append(winner)
        return selected
    
    def create_initial_population(pop_size):

    start_time = time.time()
    best_solution = initial_h.copy()
    best_c5 = objective(initial_h)
    
    while time.time() - start_time < budget_s:

        population = tournament_selection(population, fitness)
