import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = np.array(h)
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h):
        h = np.array(h)
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h0 = initial_h.copy()
    
    # Optimization
    start_time = time.time()
    result = optimize.minimize(
        fun=objective,
        x0=h0,
        method='SLSQP',
        constraints=[{'type': 'eq', 'fun': constraint}],
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    # Check if optimization completed within budget
    if time.time() - start_time > budget_s:
        return (h0, objective(h0), n_points)
    
    h_opt = result.x
    c5_bound = objective(h_opt)
    
    return (h_opt, c5_bound, n_points)
