import scipy.optimize as optimize
import time
import random

            return h

        return [x * scale for x in h]

        h = np.array(h)

            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def evaluate_population(population):
        return [objective(individual) for individual in population]
    
    def crossover(parent1, parent2):
        n = len(parent1)
        child = [0] * n
        for i in range(n):
            if random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child
    
    def mutate(individual, mutation_rate=0.1):
        n = len(individual)
        for i in range(n):
            if random.random() < mutation_rate:
                individual[i] = np.clip(random.random(), 0, 1)
        return individual
    
    def local_optimize(individual):
        def objective_with_bounds(x):
            h = np.array(x)
            total = np.sum(h)
            if abs(total - n_points / 2) > 1e-6:
                return float('inf')
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            objective_with_bounds,
            individual,
            method='L-BFGS-B',
            bounds=bounds,
            tol=1e-6
        )
        return result.x
    
    pop_size = 30
    num_parents = 10
    mutation_rate = 0.1
    population = [normalize(initial_h.copy())] + [normalize([np.random.rand() for _ in range(n_points)]) for _ in range(pop_size-1)]
