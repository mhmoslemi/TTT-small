import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h) * dx
        if abs(sum_h - 1) > 1e-8:
            return np.inf
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    def evaluate(solution):
        return objective(solution)
    
    # Initial population
    population_size = 50
    population = [np.array(initial_h, dtype=float) + np.random.uniform(-0.1, 0.1, size=n_points) for _ in range(population_size)]
    population = [np.clip(p, 0, 1) for p in population]
    population = [p / (np.sum(p) * dx) * 1 for p in population]
    
    # Genetic algorithm parameters
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 100
    best_solution = None
    best_c5 = np.inf
    
    time_start = time.time()
    
    for gen in range(generations):
        if time.time() - time_start > budget_s:
            break
        
        # Evaluate fitness
        fitness = [evaluate(individual) for individual in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_solution = population[best_idx].copy()
        
        # Selection
        indices = np.argsort(fitness)
        selected = [population[i] for i in indices[:population_size//2]]
        
        # Crossover
        new_population = []
        for _ in range(population_size//2):
            parent1, parent2 = np.random.choice(selected, size=2, replace=False)
            if np.random.rand() < crossover_rate:
                # Single-point crossover
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate((parent1[:split], parent2[split:]))
            else:
                child = parent1.copy()
            new_population.append(child)
        
        # Mutation
        for i in range(len(new_population)):
            child = new_population[i]
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[j] = np.clip(child[j] + np.random.uniform(-0.1, 0.1), 0, 1)
            # Normalize
            child = child / (np.sum(child) * dx) * 1
            new_population[i] = child
        
        # Replace population
        population = new_population
    
    return (best_solution.tolist(), best_c5, n_points)
