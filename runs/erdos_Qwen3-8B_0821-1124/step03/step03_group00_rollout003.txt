def run(seed=42, budget_s=10, **kwargs):

    sum_h = sum(initial_h)
    
    def normalize(h):
        total = sum(h)
        if total == 0:
            return h
        scale = (n_points / 2) / total
        return [x * scale for x in h]
    
    def objective(h):
        h = np.array(h)
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def evaluate_population(population):
        return [objective(individual) for individual in population]
    
    def crossover(parent1, parent2):
        n = len(parent1)
        child = [0] * n
        for i in range(n):
            if random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child
    
    def mutate(individual, mutation_rate=0.1):
        n = len(individual)
        for i in range(n):
            if random.random() < mutation_rate:
                if random.random() < 0.5:
                    individual[i] = np.clip(random.random(), 0, 1)
                else:
                    individual[i] = np.clip(1 - random.random(), 0, 1)
        return individual
    
    def create_initial_population(pop_size):
        return [normalize(initial_h.copy())] + [normalize([np.random.rand() for _ in range(n_points)]) for _ in range(pop_size-1)]
    
    pop_size = 30
    num_parents = 10
    mutation_rate = 0.2
    population = create_initial_population(pop_size)
    
    start_time = time.time()
    best_solution = initial_h.copy()
    best_c5 = objective(initial_h)
    
    while time.time() - start_time < budget_s:
        fitness = evaluate_population(population)
        parents = []
        for _ in range(num_parents):
            min_idx = np.argmin(fitness)
            parents.append(population[min_idx])
            fitness[min_idx] = float('inf')
        new_population = parents[:]
        
        while len(new_population) < pop_size:
            parent1, parent2 = random.choices(parents, k=2)
            child = crossover(parent1, parent2)
            child = mutate(child, mutation_rate)
            new_population.append(child)
        
        population = [normalize(ind) for ind in new_population]
        
        current_best = min(population, key=objective)
