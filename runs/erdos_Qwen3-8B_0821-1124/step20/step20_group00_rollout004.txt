import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):

        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    best_c5 = objective(h)

    result = optimize.minimize(

        method='SLSQP',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=constraints,
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Genetic algorithm with adaptive mutation and crossover
    population_size = 50
    generations = 25
    mutation_rate = 0.2
    mutation_strength = 0.1
    elite_fraction = 0.3
    crossover_rate = 0.8
    
    # Initial population
    population = [np.random.rand(n_points) for _ in range(population_size)]
    population = [h / np.sum(h) * (n_points / 2) for h in population]
    
    start_time = time.time()
    for gen in range(generations):
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        # Elitism: keep best individuals
        elite_indices = np.argsort(fitness)[:int(population_size * elite_fraction)]
        selected = [population[i] for i in elite_indices]
        # Generate offspring
        offspring = []
        for _ in range(population_size - len(selected)):
            parent1 = selected[np.random.randint(len(selected))]
            parent2 = selected[np.random.randint(len(selected))]
            if np.random.rand() < crossover_rate:
                # Two-point crossover with uniform sampling
                crossover_points = np.sort(np.random.choice(n_points, 2, replace=False))
                child = np.concatenate([parent1[:crossover_points[0]], parent2[crossover_points[0]:crossover_points[1]], parent1[crossover_points[1]:]])
            else:
                # Blend crossover
                child = (parent1 + parent2) / 2
            # Apply mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.normal(0, mutation_strength)
                    child[i] = np.clip(child[i], 0, 1)
            # Normalize
            child = child / np.sum(child) * (n_points / 2)
            offspring.append(child)
        # Replace population
        population = selected + offspring
        # Update best solution
        for candidate in population:
            current_c5 = objective(candidate)
            if current_c5 < best_c5:
                best_h = candidate
                best_c5 = current_c5
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_h, best_c5, n_points)
