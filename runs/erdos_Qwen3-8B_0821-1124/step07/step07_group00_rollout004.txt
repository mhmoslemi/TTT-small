import numpy as np
import random
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def normalize(h):
        total = np.sum(h)
        if abs(total) < 1e-6:
            return np.zeros_like(h)
        return h * (n_points / 2) / total
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] = np.clip(h[i] + random.uniform(-0.2, 0.2), 0, 1)
        return normalize(h)
    
    def crossover(parent1, parent2):
        h = parent1.copy()
        for i in range(len(h)):
            if random.random() < 0.5:
                h[i] = parent2[i]
        return normalize(h)
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    population_size = 20
    population = [normalize(initial_h.copy()) for _ in range(population_size)]
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        # Select best individuals
        selected = np.argsort(fitness)[:population_size//2]
        # Create new population
        new_population = []
        for i in range(population_size//2):
            parent1 = population[selected[i]]
            parent2 = population[selected[np.random.randint(len(selected))]]
            child = crossover(parent1, parent2)
            new_population.append(mutate(child))
        population = new_population
    
    best_h = min(population, key=evaluate)
    best_c5 = evaluate(best_h)
    
    return (best_h, best_c5, n_points)
