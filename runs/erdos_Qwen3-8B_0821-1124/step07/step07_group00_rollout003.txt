            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        return evaluate(h)
    
    def constraint(h_vec):
        h = h_vec.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint}
    
    # Initial guess
    x0 = initial_h.copy()

    max_iter = int(budget_s * 10)  # Allow for some overhead
    results = []
    
    # Run optimization
    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, x0, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=cons)
        results.append(result)
        x0 = result.x
    
    # Find best solution
    best_result = min(results, key=lambda r: r.fun)
    best_h = best_result.x
    best_c5 = best_result.fun
