import numpy as np
import random
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    def perturb(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] = np.clip(h[i] + random.uniform(-0.2, 0.2), 0, 1)
        return normalize(h)
    
    def crossover(parent1, parent2):
        n = len(parent1)
        child = np.zeros(n)
        for i in range(n):
            if random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return normalize(child)
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(20)]
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        population = [perturb(individual) for individual in population]
        population.sort(key=evaluate)
        top_half = population[:len(population)//2]
        population = [crossover(p1, p2) for p1 in top_half for p2 in top_half]
        population += [perturb(individual) for individual in top_half]
        current_c5 = evaluate(population[0])
        if current_c5 < best_c5:
            best_h = population[0].copy()
            best_c5 = current_c5
    
    return (best_h, best_c5, n_points)
