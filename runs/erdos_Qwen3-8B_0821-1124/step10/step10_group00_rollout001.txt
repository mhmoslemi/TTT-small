import numpy as np
import scipy.optimize as optimize
import time
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Define the fitness function
    def evaluate(individual):
        h = np.array(individual)
        # Normalize to enforce integral constraint
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return (c5,)
    
    # Create the genetic algorithm
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    # Setup the toolbox
    toolbox = base.Toolbox()
    toolbox.register("attr_float", lambda: np.random.uniform(0, 1))
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("evaluate", evaluate)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=1/n_points)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Initialize population
    pop = toolbox.population(n=50)
    
    # Start the genetic algorithm
    start_time = time.time()
    for gen in range(100):
        # Evaluate the entire population
        offspring = algorithms.varAnd(pop, toolbox, cxpb=0.5, mutpb=0.1)
        fits = map(toolbox.evaluate, offspring)
        # Replace the population
        pop = list(map(lambda x: x[0], zip(offspring, fits)))
        # Check time budget
        if time.time() - start_time > budget_s:
            break
    
    # Find the best solution
    best_ind = min(pop, key=lambda x: x.fitness.values[0])
    best_h = best_ind
    best_c5 = best_ind.fitness.values[0]
    
    return (best_h, best_c5, n_points)
