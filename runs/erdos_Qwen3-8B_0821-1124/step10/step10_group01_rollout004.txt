import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(x):
        h = np.array(x)
        # Enforce integral constraint by normalizing
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = [initial_h.copy() for _ in range(population_size)]
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Local optimization on best solution
        x0 = best_h.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            objective,
            x0,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        best_h = result.x
        best_c5 = objective(best_h)
        
        # Create new population
        new_population = []
        for ind in population:
            # Apply mutation
            mutated = ind.copy()
            for i in range(n_points):
                if random.random() < mutation_rate:
                    mutated[i] = np.random.uniform(0, 1)
            # Apply crossover
            if random.random() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([mutated[:crossover_point], population[best_idx][crossover_point:]])
                new_population.append(child)
            else:
                new_population.append(mutated)
        
        population = new_population
    
    # Return the best solution found
    return (best_h, best_c5, n_points)
