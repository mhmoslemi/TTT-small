import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)

    def objective(x):
        h = np.array(x)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization with constraint handling
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    result = optimize.minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Check time budget
    if time.time() - start_time > budget_s:
        return (initial_h, objective(initial_h), n_points)
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
