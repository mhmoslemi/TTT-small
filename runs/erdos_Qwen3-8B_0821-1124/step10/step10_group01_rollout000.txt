import random

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(x):
        h = np.array(x)
        # Enforce integral constraint by normalizing
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Initialize population
    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    
    def evaluate(individual):
        return objective(individual)
    
    def mutate(individual):
        mutant = individual.copy()
        idx = np.random.randint(len(mutant))
        mutant[idx] = np.random.uniform(0, 1)
        return mutant
    
    def crossover(parent1, parent2):
        child = parent1.copy()
        idx = np.random.randint(len(child))
        child[idx] = parent2[idx]
        return child
    
    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [evaluate(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_fitness = fitness[best_idx]
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Select parents
            parent1, parent2 = random.choices(population, k=2)
            # Crossover
            child = crossover(parent1, parent2)
            # Mutate
            mutant = mutate(child)
            new_population.append(mutant)
        
        population = new_population
    
    # Find best solution
    fitness = [evaluate(ind) for ind in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
