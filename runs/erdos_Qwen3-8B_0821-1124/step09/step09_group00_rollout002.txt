import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Create initial population
    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    
    def evaluate(h):
        # Normalize to enforce integral constraint
        h_normalized = h / np.sum(h) * n_points / 2
        corr = np.correlate(h_normalized, 1 - h_normalized, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def mutate(h, mutation_rate=0.1):
        h_mutated = h.copy()
        for i in range(len(h_mutated)):
            if random.random() < mutation_rate:
                h_mutated[i] = np.random.uniform(0, 1)
        return h_mutated
    
    def crossover(h1, h2):
        crossover_point = random.randint(1, len(h1)-1)
        child = np.concatenate((h1[:crossover_point], h2[crossover_point:]))
        return child
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        fitness = [evaluate(h) for h in population]
        best_index = np.argmin(fitness)
        best_c5 = fitness[best_index]
        best_h = population[best_index].copy()
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Select parents
            parent1, parent2 = random.choices(population, k=2)
            child1 = crossover(parent1, parent2)
            child1 = mutate(child1)
            new_population.append(child1)
            
            parent1, parent2 = random.choices(population, k=2)
            child2 = crossover(parent1, parent2)
            child2 = mutate(child2)
            new_population.append(child2)
        
        population = new_population
    
    return (best_h, best_c5, n_points)
