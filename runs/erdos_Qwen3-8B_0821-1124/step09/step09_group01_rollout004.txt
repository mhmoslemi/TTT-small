import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)

        penalty = 1000 * (np.sum(h) - n_points / 2) ** 2
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5 + penalty
    
    # Initial population with variation
    population = [initial_h.copy() for _ in range(30)]
    for p in population:
        p += np.random.normal(0, 0.05, size=n_points)
        p = np.clip(p, 0, 1)
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(p) for p in population]
        
        # Select elites
        elite_indices = np.argsort(fitness)[:5]
        elites = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elites.copy()
        while len(new_population) < len(population):
            # Select parents
            parent1 = random.choice(elite_indices)
            parent2 = random.choice(elite_indices)
            
            # Crossover
            crossover_point = random.randint(1, n_points - 1)
            child = np.concatenate((population[parent1][:crossover_point], population[parent2][crossover_point:]))
            
            # Mutate
            for i in range(n_points):

                    child[i] += np.random.normal(0, 0.05)
                    child[i] = np.clip(child[i], 0, 1)
            
            new_population.append(child)
        
        population = new_population
    
    # Local optimization on elites
    for i in elite_indices:
