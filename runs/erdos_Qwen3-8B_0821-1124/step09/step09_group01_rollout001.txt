def run(seed=42, budget_s=10, **kwargs):

    def objective(x):
        h = np.array(x)
        # Enforce integral constraint with penalty
        penalty = 1000 * (np.sum(h) - n_points / 2) ** 2
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5 + penalty

    population_size = 50
    generations = 100
    mutation_rate = 0.1
    elite_size = 5

    population = [initial_h.copy() for _ in range(population_size)]
    for p in population:
        p += np.random.normal(0, 0.05, size=n_points)
        p = np.clip(p, 0, 1)
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    
    start_time = time.time()
    
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(p) for p in population]
        
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elites.copy()
        while len(new_population) < population_size:
            # Select parents
            parent1 = random.choice(elite_indices)
            parent2 = random.choice(elite_indices)
            
            # Crossover
            crossover_point = random.randint(1, n_points - 1)

                    child[i] += np.random.normal(0, 0.05)
                    child[i] = np.clip(child[i], 0, 1)
            
            new_population.append(child)
        
        population = new_population
    
    # Find best solution
    for p in population:
        c5 = objective(p)
        if c5 < best_c5:
