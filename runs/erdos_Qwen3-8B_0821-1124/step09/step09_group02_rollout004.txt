import random

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = np.array(h)
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Initialize population
    population_size = 30
    population = [initial_h.copy() for _ in range(population_size)]
    for h in population:
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    
    start_time = time.time()
    
    # Parameters
    generations = 200
    mutation_rate = 0.15
    crossover_rate = 0.8
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Selection
        selected = np.random.choice(len(population), size=population_size, p=np.array(fitness)/np.sum(fitness))
        selected = np.random.choice(selected, size=population_size)
        
        # Crossover
        for i in range(0, population_size, 2):
            if np.random.random() < crossover_rate:
                p1, p2 = population[selected[i]], population[selected[i+1]]
                child1 = np.concatenate([p1[:n_points//2], p2[n_points//2:]])
                child2 = np.concatenate([p2[:n_points//2], p1[n_points//2:]])
                population[i], population[i+1] = child1, child2
        
        # Mutation

                population[i] = h
        
        # Update best solution
        if best_c5 < objective(population[best_idx]):
            best_h = population[best_idx].copy()
