import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(x):
        h = np.array(x)
        # Normalize to enforce integral constraint
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization
    x0 = initial_h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    result = optimize.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    
    # Check if we have exceeded the time budget
    if time.time() - start_time > budget_s:
        return (initial_h, objective(initial_h), n_points)
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
