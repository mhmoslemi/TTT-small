import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = np.array(h)
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Initialize with initial construction
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Genetic algorithm parameters
    population_size = 10
    generations = 100
    mutation_rate = 0.1
    
    # Initialize population
    population = [h.copy() for _ in range(population_size)]
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Select parents
            parent1, parent2 = np.random.choice(population, size=2, replace=False)
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child1 = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            child2 = np.concatenate((parent2[:crossover_point], parent1[crossover_point:]))
            
            # Mutation
            for i in range(n_points):
                if np.random.random() < mutation_rate:
                    child1[i] = np.random.uniform(0, 1)
                    child2[i] = np.random.uniform(0, 1)
            
            # Normalize to maintain integral constraint
            child1 = child1 / np.sum(child1) * (n_points / 2)
            child2 = child2 / np.sum(child2) * (n_points / 2)
            
            new_population.extend([child1, child2])
        
        # Replace population
        population = new_population[:population_size]
    
    return (best_h, best_c5, n_points)
