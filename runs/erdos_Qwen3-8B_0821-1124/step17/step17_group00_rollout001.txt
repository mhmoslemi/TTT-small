        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Genetic algorithm with adaptive mutation and tournament selection
    population_size = 50
    generations = 30
    mutation_rate = 0.15
    mutation_strength = 0.05
    
    # Initial population
    population = [np.random.rand(n_points) for _ in range(population_size)]
    population = [h / np.sum(h) * (n_points / 2) for h in population]
    
    start_time = time.time()
    for gen in range(generations):
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        # Elitism: keep best individuals
        elite_indices = np.argsort(fitness)[:population_size//2]
        selected = [population[i] for i in elite_indices]
        # Generate offspring
        offspring = []
        for _ in range(population_size//2):
            parent1 = selected[np.random.randint(len(selected))]
            parent2 = selected[np.random.randint(len(selected))]
            child = np.copy(parent1)

            for i in range(n_points):
