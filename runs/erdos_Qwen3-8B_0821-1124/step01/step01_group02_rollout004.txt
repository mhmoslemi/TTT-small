        adjustment = (n_points / 2 - sum_h) / n_points

        h = np.array(h)
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return float('inf')
        # Compute overlap integral
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = [initial_h + np.random.uniform(-0.1, 0.1, size=n_points) for _ in range(population_size)]
    population = [np.clip(p, 0, 1) for p in population]
    population = [p for p in population if abs(np.sum(p) - n_points / 2) < 1e-6]
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        
        # Select parents
        parents = np.random.choice(population, size=population_size, p=np.array(fitness)/np.sum(fitness))
        
        # Crossover
        offspring = []
        for _ in range(population_size):
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points-1)
                child = np.concatenate([parent1[:split], parent2[split:]])
            else:
                child = np.random.choice(parents, size=n_points)
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    offspring[i][j] += np.random.uniform(-0.1, 0.1)

            offspring[i] = [val + adjustment for val in offspring[i]]
        
        # Replace population
        population = offspring
    
    # Evaluate best solution
    best_h = min(population, key=objective)
    c5_bound = objective(best_h)
    return (best_h, c5_bound, n_points)
