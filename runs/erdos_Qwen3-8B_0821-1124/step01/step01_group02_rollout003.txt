import numpy as np
import random
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = sum(initial_h)
    # Adjust initial_h to meet the integral constraint
    if abs(sum_h - n_points / 2) > 1e-6:
        adjustment = (n_points / 2 - sum_h) / n_points
        initial_h = [h + adjustment for h in initial_h]
    
    def evaluate(h):
        h = np.array(h)
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def mutate(h, mutation_rate=0.05):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] = np.clip(h[i] + np.random.normal(0, 0.1), 0, 1)
        return h
    
    def crossover(parent1, parent2):
        child = parent1.copy()
        for i in range(len(child)):
            if random.random() < 0.5:
                child[i] = parent2[i]
        return child
    
    population_size = 50
    population = [mutate(initial_h) for _ in range(population_size)]
    best_solution = initial_h
    best_c5 = evaluate(initial_h)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate population
        scores = [evaluate(ind) for ind in population]
        # Select parents
        parents = np.array(population)[np.argsort(scores)[:population_size//2]]
        # Create new population
        new_population = []
        for _ in range(population_size):
            parent1, parent2 = random.choices(parents, k=2)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        population = new_population
        # Update best solution
        for ind in population:
            current_c5 = evaluate(ind)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_solution = ind
    
    return (np.array(best_solution), best_c5, n_points)
