        sum_h = np.sum(h) * dx
        if abs(sum_h - 1) > 1e-8:
            return np.inf
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5

        total = np.sum(h) * dx

            return h
        factor = 1.0 / total
        return np.clip(h * factor, 0, 1)
    
    def crossover(parent1, parent2):
        point = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:point], parent2[point:]])
        return child
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] = np.clip(h[i] + np.random.uniform(-0.1, 0.1), 0, 1)
        return h
    
    population_size = 50
    population = [normalize(h + np.random.uniform(-0.1, 0.1, n_points)) for _ in range(population_size)]
    population = [normalize(h) for h in population]
