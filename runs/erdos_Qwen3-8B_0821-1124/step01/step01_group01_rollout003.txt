import numpy as np

import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h_pop = [np.array(initial_h, dtype=float) for _ in range(32)]
    
    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h) * dx
        if abs(sum_h - 1) > 1e-8:
            return np.inf
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    def normalize(h):
        total = np.sum(h) * dx
        if abs(total - 1) < 1e-8:
            return h
        scaling = 1.0 / total
        return h * scaling
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        idx = np.random.choice(len(child), size=1)
        child[idx] = parent2[idx]
        return child
    
    def mutate(h, mutation_rate=0.1):
        h = np.copy(h)
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] = np.clip(h[i] + np.random.uniform(-0.1, 0.1), 0, 1)
        return h
    
    best_h = np.array(initial_h, dtype=float)
    best_c5 = objective(best_h)
    time_start = time.time()
    
    for gen in range(100):
        if time.time() - time_start > budget_s:
            break
        # Evaluate population
        objectives = [objective(h) for h in h_pop]
        # Select top 16
        indices = np.argsort(objectives)[:16]
        selected = [h_pop[i] for i in indices]
        # Create new population
        new_pop = []
        for _ in range(16):
            p1, p2 = np.random.choice(selected, size=2, replace=False)
            child = crossover(p1, p2)
            child = mutate(child)
            child = normalize(child)
            new_pop.append(child)
        h_pop = new_pop
        # Update best
        for h in h_pop:
            c5 = objective(h)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
    
    return (best_h.tolist(), best_c5, n_points)
