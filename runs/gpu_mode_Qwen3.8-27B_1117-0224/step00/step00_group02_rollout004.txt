@triton.jit
def epilogue_ln_gate_proj_kernel(
    out_h_ptr, gate_ptr, wo_ptr, gamma_ptr, beta_ptr, output_ptr,
    N2, NUM_M_BLOCKS,
    C: tl.constexpr, H: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_C: tl.constexpr, EPS: tl.constexpr):
    pid=tl.program_id(0)
    b=pid//NUM_M_BLOCKS
    mb=pid%NUM_M_BLOCKS
    offs_m=mb*BLOCK_M+tl.arange(0,BLOCK_M)
    m_mask=offs_m<N2
    offs_h=tl.arange(0,H)
    b64=b.to(tl.int64)
    base=b64*H*N2
    ptr = base + offs_h[:,None]*N2 + offs_m[None,:]
    out = tl.load(out_h_ptr + ptr, mask=m_mask[None,:], other=0.0).to(tl.float32)
    gate = tl.load(gate_ptr + ptr, mask=m_mask[None,:], other=0.0).to(tl.float32)
    mean=tl.sum(out, axis=0)/H
    diff=out-mean[None,:]
    var=tl.sum(diff*diff, axis=0)/H
    var=tl.maximum(var,0.0)
    inv=1.0/tl.sqrt(var+EPS)
    gamma=tl.load(gamma_ptr+offs_h)
    beta=tl.load(beta_ptr+offs_h)
    res=(diff*inv[None,:])*gamma[:,None]+beta[:,None]
    res=res*gate
    resb=tl.trans(res.to(tl.bfloat16))
    out_base=b64*N2*C + offs_m[:,None]*C
    for c0 in range(0, C, BLOCK_C):
       offs_c=c0+tl.arange(0,BLOCK_C)
       c_mask=offs_c<C
       w=tl.load(wo_ptr+offs_h[:,None]*C+offs_c[None,:], mask=c_mask[None,:], other=0.0)
       y=tl.dot(resb, w)
       tl.store(output_ptr + out_base + offs_c[None,:], y, mask=m_mask[:,None]&c_mask[None,:])

import torch
import triton
import triton.language as tl


@triton.jit
def _sigmoid(x):
    return 1.0 / (1.0 + tl.exp(-x))


@triton.jit
def fused_norm_proj_kernel(
    x_ptr,
    wl_ptr, wr_ptr, wlg_ptr, wrg_ptr, wog_ptr,
    bl_ptr, br_ptr, blg_ptr, brg_ptr, bog_ptr,
    mask_ptr,
    left_ptr, right_ptr, gate_ptr,
    N2,
    NUM_M_BLOCKS,
    C: tl.constexpr,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
    HAS_MASK: tl.constexpr,
    EPS: tl.constexpr,
):
    pid = tl.program_id(0)
    b = pid // NUM_M_BLOCKS
    mb = pid % NUM_M_BLOCKS

    offs_m = mb * BLOCK_M + tl.arange(0, BLOCK_M)
    m_mask = offs_m < N2
    offs_h = tl.arange(0, H)

    b64 = b.to(tl.int64)
    x_base = b64 * N2 * C

    sum_x = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_x2 = tl.zeros((BLOCK_M,), dtype=tl.float32)

    acc_l = tl.zeros((H, BLOCK_M), dtype=tl.float32)
    acc_r = tl.zeros((H, BLOCK_M), dtype=tl.float32)
    acc_lg = tl.zeros((H, BLOCK_M), dtype=tl.float32)
    acc_rg = tl.zeros((H, BLOCK_M), dtype=tl.float32)
    acc_og = tl.zeros((H, BLOCK_M), dtype=tl.float32)

    if C <= BLOCK_C:
        offs_c = tl.arange(0, BLOCK_C)
        c_mask = offs_c < C

        x = tl.load(
            x_ptr + x_base + offs_m[:, None] * C + offs_c[None, :],
            mask=m_mask[:, None] & c_mask[None, :],
            other=0.0,
        )
        sum_x = tl.sum(x, axis=1)
        sum_x2 = tl.sum(x * x, axis=1)

        c_inv = 1.0 / C
        mean = sum_x * c_inv
        var = sum_x2 * c_inv - mean * mean
        var = tl.maximum(var, 0.0)
        inv_std = 1.0 / tl.sqrt(var + EPS)

        xc = ((x - mean[:, None]) * inv_std[:, None]).to(tl.bfloat16)
        xt = tl.trans(xc)

        w_mask = c_mask[None, :]
        wl = tl.load(wl_ptr + offs_h[:, None] * C + offs_c[None, :], mask=w_mask, other=0.0)
        wr = tl.load(wr_ptr + offs_h[:, None] * C + offs_c[None, :], mask=w_mask, other=0.0)
        wlg = tl.load(wlg_ptr + offs_h[:, None] * C + offs_c[None, :], mask=w_mask, other=0.0)
        wrg = tl.load(wrg_ptr + offs_h[:, None] * C + offs_c[None, :], mask=w_mask, other=0.0)
        wog = tl.load(wog_ptr + offs_h[:, None] * C + offs_c[None, :], mask=w_mask, other=0.0)

        acc_l = tl.dot(wl, xt, acc_l)
        acc_r = tl.dot(wr, xt, acc_r)
        acc_lg = tl.dot(wlg, xt, acc_lg)
        acc_rg = tl.dot(wrg, xt, acc_rg)
        acc_og = tl.dot(wog, xt, acc_og)
    else:
        for c0 in range(0, C, BLOCK_C):
            offs_c = c0 + tl.arange(0, BLOCK_C)
            c_mask = offs_c < C
            x = tl.load(
                x_ptr + x_base + offs_m[:, None] * C + offs_c[None, :],
                mask=m_mask[:, None] & c_mask[None, :],
                other=0.0,
            )
            sum_x += tl.sum(x, axis=1)
            sum_x2 += tl.sum(x * x, axis=1)

        c_inv = 1.0 / C
        mean = sum_x * c_inv
        var = sum_x2 * c_inv - mean * mean
        var = tl.maximum(var, 0.0)
        inv_std = 1.0 / tl.sqrt(var + EPS)

        for c0 in range(0, C, BLOCK_C):
            offs_c = c0 + tl.arange(0, BLOCK_C)
            c_mask = offs_c < C
            w_mask = c_mask[None, :]

            x = tl.load(
                x_ptr + x_base + offs_m[:, None] * C + offs_c[None, :],
                mask=m_mask[:, None] & c_mask[None, :],
                other=0.0,
            )
            xc = ((x - mean[:, None]) * inv_std[:, None]).to(tl.bfloat16)
            xt = tl.trans(xc)

            wl = tl.load(wl_ptr + offs_h[:, None] * C + offs_c[None, :], mask=w_mask, other=0.0)
            wr = tl.load(wr_ptr + offs_h[:, None] * C + offs_c[None, :], mask=w_mask, other=0.0)
            wlg = tl.load(wlg_ptr + offs_h[:, None] * C + offs_c[None, :], mask=w_mask, other=0.0)
            wrg = tl.load(wrg_ptr + offs_h[:, None] * C + offs_c[None, :], mask=w_mask, other=0.0)
            wog = tl.load(wog_ptr + offs_h[:, None] * C + offs_c[None, :], mask=w_mask, other=0.0)

            acc_l = tl.dot(wl, xt, acc_l)
            acc_r = tl.dot(wr, xt, acc_r)
            acc_lg = tl.dot(wlg, xt, acc_lg)
            acc_rg = tl.dot(wrg, xt, acc_rg)
            acc_og = tl.dot(wog, xt, acc_og)

    bl = tl.load(bl_ptr + offs_h)
    br = tl.load(br_ptr + offs_h)
    blg = tl.load(blg_ptr + offs_h)
    brg = tl.load(brg_ptr + offs_h)
    bog = tl.load(bog_ptr + offs_h)

    l_val = acc_l + bl[:, None]
    r_val = acc_r + br[:, None]

    l_gate = _sigmoid(acc_lg + blg[:, None])
    r_gate = _sigmoid(acc_rg + brg[:, None])
    o_gate = _sigmoid(acc_og + bog[:, None])

    if HAS_MASK:
        msk = tl.load(mask_ptr + b64 * N2 + offs_m, mask=m_mask, other=0.0)
        l_gate = l_gate * msk[None, :]
        r_gate = r_gate * msk[None, :]

    left = l_val * l_gate
    right = r_val * r_gate

    out_base = b64 * H * N2
    store_offs_b = out_base + offs_h[:, None] * N2 + offs_m[None, :]
    store_mask = m_mask[None, :]

    tl.store(left_ptr + store_offs_b, left.to(tl.bfloat16), mask=store_mask)
    tl.store(right_ptr + store_offs_b, right.to(tl.bfloat16), mask=store_mask)
    tl.store(gate_ptr + store_offs_b, o_gate.to(tl.bfloat16), mask=store_mask)


@triton.jit
def epilogue_ln_gate_proj_kernel(
    out_h_ptr,
    gate_ptr,
    wo_ptr,
    gamma_ptr,
    beta_ptr,
    output_ptr,
    N2,
    NUM_M_BLOCKS,
    C: tl.constexpr,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
    EPS: tl.constexpr,
):
    pid = tl.program_id(0)
    b = pid // NUM_M_BLOCKS
    mb = pid % NUM_M_BLOCKS

    offs_m = mb * BLOCK_M + tl.arange(0, BLOCK_M)
    m_mask = offs_m < N2
    offs_h = tl.arange(0, H)

    b64 = b.to(tl.int64)
    base = b64 * H * N2
    ptr_offs = offs_h[:, None] * N2 + offs_m[None, :]

    out = tl.load(out_h_ptr + base + ptr_offs, mask=m_mask[None, :], other=0.0).to(tl.float32)
    gate = tl.load(gate_ptr + base + ptr_offs, mask=m_mask[None, :], other=0.0).to(tl.float32)

    mean = tl.sum(out, axis=0) / H
    diff = out - mean[None, :]
    var = tl.sum(diff * diff, axis=0) / H
    var = tl.maximum(var, 0.0)
    inv_std = 1.0 / tl.sqrt(var + EPS)

    gamma = tl.load(gamma_ptr + offs_h)
    beta = tl.load(beta_ptr + offs_h)

    res = (diff * inv_std[None, :]) * gamma[:, None] + beta[:, None]
    res = res * gate

    resb = tl.trans(res.to(tl.bfloat16))
    final_base = b64 * N2 * C + offs_m[:, None] * C

    for c0 in range(0, C, BLOCK_C):
        offs_c = c0 + tl.arange(0, BLOCK_C)
        c_mask = offs_c < C
        w = tl.load(
            wo_ptr + offs_h[:, None] * C + offs_c[None, :],
            mask=c_mask[None, :],
            other=0.0,
        )
        y = tl.dot(resb, w)
        tl.store(
            output_ptr + final_base + offs_c[None, :],
            y,
            mask=m_mask[:, None] & c_mask[None, :],
        )


def _next_power_of_two(n: int) -> int:
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def custom_kernel(data):
    """
    Fused Triton implementation of outgoing TriMul forward pass.

    Stage 1 (Triton): input LayerNorm statistics + five projections/gates fused,
    with LayerNorm gamma/bias folded into effective projection weights. Outputs
    are stored in [B, hidden, N, N] layout to feed cuBLAS batched GEMM.

    Stage 2 (cuBLAS batched GEMM): computes all hidden-channel independent
    N x N matrix products L @ R^T.

    Stage 3 (Triton): final LayerNorm over hidden dim, out-gate, and output
    projection fused, producing float32 output.
    """
    x, mask, weights, config = data

    B, N, _, C = x.shape
    H = int(config.get("hidden_dim", 128))
    N2 = N * N
    device = x.device
    eps = float(config.get("eps", 1e-5))

    x = x.to(torch.float32, copy=False).contiguous()

    norm_w = weights["norm.weight"].float().contiguous()
    norm_b = weights["norm.bias"].float().contiguous()

    def _prepare_projection(W):
        Wf = W.float().contiguous()
        weff = (Wf * norm_w.unsqueeze(0)).to(torch.bfloat16).contiguous()
        bias_term = (Wf * norm_b.unsqueeze(0)).sum(dim=1).float().contiguous()
        return weff, bias_term

    WL, BL = _prepare_projection(weights["left_proj.weight"])
    WR, BR = _prepare_projection(weights["right_proj.weight"])
    WLG, BLG = _prepare_projection(weights["left_gate.weight"])
    WRG, BRG = _prepare_projection(weights["right_gate.weight"])
    WOG, BOG = _prepare_projection(weights["out_gate.weight"])

    to_out_weight_T = weights["to_out_weight"].float().t().to(torch.bfloat16).contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].float().contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].float().contiguous()

    has_mask = mask is not None and not bool(config.get("nomask", False))
    if has_mask:
        mt = mask
        if mt.dim() == 4 and mt.shape[-1] == 1:
            mt = mt.squeeze(-1)
        if mt.dim() == 2:
            mt = mt.unsqueeze(0)
        if mt.shape[0] != B:
            mt = mt.expand(B, -1, -1)
        mask_arg = mt.to(torch.float32).contiguous()
    else:
        mask_arg = x

    left_h = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    right_h = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    gate_h = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)

    BLOCK_M_PRE = 32
    BLOCK_C_PRE = 128 if C >= 128 else _next_power_of_two(max(1, C))
    num_m_blocks_pre = triton.cdiv(N2, BLOCK_M_PRE)

    fused_norm_proj_kernel[(B * num_m_blocks_pre,)](
        x,
        WL, WR, WLG, WRG, WOG,
        BL, BR, BLG, BRG, BOG,
        mask_arg,
        left_h, right_h, gate_h,
        N2,
        num_m_blocks_pre,
        C=C,
        H=H,
        BLOCK_M=BLOCK_M_PRE,
        BLOCK_C=BLOCK_C_PRE,
        HAS_MASK=has_mask,
        EPS=eps,
        num_warps=8,
        num_stages=2,
    )

    BH = B * H
    left_3d = left_h.as_strided((BH, N, N), (N2, N, 1))
    right_3d_t = right_h.as_strided((BH, N, N), (N2, 1, N))

    out_h = torch.empty((BH, N, N), dtype=torch.bfloat16, device=device)
    torch.bmm(left_3d, right_3d_t, out=out_h)
    out_h = out_h.view(B, H, N, N)

    BLOCK_M_EPI = 64
    BLOCK_C_EPI = 128 if C >= 128 else _next_power_of_two(max(1, C))
    num_m_blocks_epi = triton.cdiv(N2, BLOCK_M_EPI)

    output = torch.empty((B, N, N, C), dtype=torch.float32, device=device)

    epilogue_ln_gate_proj_kernel[(B * num_m_blocks_epi,)](
        out_h,
        gate_h,
        to_out_weight_T,
        to_out_norm_w,
        to_out_norm_b,
        output,
        N2,
        num_m_blocks_epi,
        C=C,
        H=H,
        BLOCK_M=BLOCK_M_EPI,
        BLOCK_C=BLOCK_C_EPI,
        EPS=eps,
        num_warps=8,
        num_stages=2,
    )

    return output
