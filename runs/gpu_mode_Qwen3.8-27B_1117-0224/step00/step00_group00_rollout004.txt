   - applies masks and sigmoid gates,
   - writes directly into [B, H, N, N] views for left, right, and out-gate.

2) Batched matrix multiply (cuBLAS via torch.matmul):
   - for each b, h: out[b,h] = left[b,h] @ right[b,h]^T

3) Triton final-kernel:
   - reads out and out-gate in the [B, H, P] layout (coalesced along P for
     each H plane),
   - computes LayerNorm over H,
   - applies out-gate,
   - performs the final H -> C projection and writes float32 output.
"""

@triton.jit
def _trimul_pre_kernel(
    x_ptr,
    mask_ptr,
    l_ptr,
    r_ptr,
    og_ptr,
    m_comb_ptr,
    g_comb_ptr,
    b_comb_ptr,
    P,
    C,
    H: tl.constexpr,
    C_BLOCK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    HAS_MASK: tl.constexpr,
    EPS: tl.float32,
):
    pid = tl.program_id(0)
    b = tl.program_id(1)

    offs_r = pid * BLOCK_R + tl.arange(0, BLOCK_R)
    r_ok = offs_r < P
    row = tl.minimum(offs_r, P - 1)

    offs_h = tl.arange(0, H)

    acc_lp = tl.zeros((BLOCK_R, H), dtype=tl.float32)
    acc_rp = tl.zeros((BLOCK_R, H), dtype=tl.float32)
    acc_lg = tl.zeros((BLOCK_R, H), dtype=tl.float32)
    acc_rg = tl.zeros((BLOCK_R, H), dtype=tl.float32)
    acc_og = tl.zeros((BLOCK_R, H), dtype=tl.float32)

    x_sum = tl.zeros((BLOCK_R,), dtype=tl.float32)
    x_sq = tl.zeros((BLOCK_R,), dtype=tl.float32)

    x_base = b * P * C

    for c0 in range(0, C, C_BLOCK):
        offs_c = c0 + tl.arange(0, C_BLOCK)
        c_ok = offs_c < C

        x_ptrs = x_ptr + (x_base + row[:, None] * C) + offs_c[None, :]
        x_chunk = tl.load(
            x_ptrs,
            mask=r_ok[:, None] & c_ok[None, :],
            other=0.0
        ).to(tl.float32)

        x_sum += tl.sum(x_chunk, axis=1)
        x_sq += tl.sum(x_chunk * x_chunk, axis=1)

        x16 = x_chunk.to(tl.bfloat16)

        m_base = offs_c[:, None] * H + offs_h[None, :]

        m0 = tl.load(m_comb_ptr + m_base, mask=c_ok[:, None], other=0.0)
        acc_lp = tl.dot(x16, m0, acc_lp)

        m1 = tl.load(m_comb_ptr + C * H + m_base, mask=c_ok[:, None], other=0.0)
        acc_rp = tl.dot(x16, m1, acc_rp)

        m2 = tl.load(m_comb_ptr + 2 * C * H + m_base, mask=c_ok[:, None], other=0.0)
        acc_lg = tl.dot(x16, m2, acc_lg)

        m3 = tl.load(m_comb_ptr + 3 * C * H + m_base, mask=c_ok[:, None], other=0.0)
        acc_rg = tl.dot(x16, m3, acc_rg)

        m4 = tl.load(m_comb_ptr + 4 * C * H + m_base, mask=c_ok[:, None], other=0.0)
        acc_og = tl.dot(x16, m4, acc_og)

    mu = x_sum / C
    var = x_sq / C - mu * mu
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + EPS)
    shift = mu * rstd

    g0 = tl.load(g_comb_ptr + offs_h)
    b0 = tl.load(b_comb_ptr + offs_h)
    left = acc_lp * rstd[:, None] - shift[:, None] * g0[None, :] + b0[None, :]

    g2 = tl.load(g_comb_ptr + 2 * H + offs_h)
    b2 = tl.load(b_comb_ptr + 2 * H + offs_h)
    left_gate = acc_lg * rstd[:, None] - shift[:, None] * g2[None, :] + b2[None, :]
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate))

    if HAS_MASK:
        m_row = tl.load(
            mask_ptr + b * P + row,
            mask=r_ok,
            other=0.0
        ).to(tl.float32)
    else:
        m_row = 1.0 + tl.zeros((BLOCK_R,), dtype=tl.float32)

    left = left * left_gate * m_row[:, None]

    g1 = tl.load(g_comb_ptr + 1 * H + offs_h)
    b1 = tl.load(b_comb_ptr + 1 * H + offs_h)
    right = acc_rp * rstd[:, None] - shift[:, None] * g1[None, :] + b1[None, :]

    g3 = tl.load(g_comb_ptr + 3 * H + offs_h)
    b3 = tl.load(b_comb_ptr + 3 * H + offs_h)
    right_gate = acc_rg * rstd[:, None] - shift[:, None] * g3[None, :] + b3[None, :]
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate))

    right = right * right_gate * m_row[:, None]

    g4 = tl.load(g_comb_ptr + 4 * H + offs_h)
    b4 = tl.load(b_comb_ptr + 4 * H + offs_h)
    out_gate = acc_og * rstd[:, None] - shift[:, None] * g4[None, :] + b4[None, :]
    out_gate = 1.0 / (1.0 + tl.exp(-out_gate))

    store_offs = b * H * P + offs_h[:, None] * P + row[None, :]

    tl.store(
        l_ptr + store_offs,
        tl.trans(left.to(tl.bfloat16)),
        mask=r_ok[None, :]
    )
    tl.store(
        r_ptr + store_offs,
        tl.trans(right.to(tl.bfloat16)),
        mask=r_ok[None, :]
    )
    tl.store(
        og_ptr + store_offs,
        tl.trans(out_gate.to(tl.bfloat16)),
        mask=r_ok[None, :]
    )


@triton.jit
def _trimul_final_kernel(
    out_t_ptr,
    gate_t_ptr,
    w_ptr,
    norm_w_ptr,
    norm_b_ptr,
    out_ptr,
    P,
    C,
    H: tl.constexpr,
    C_BLOCK: tl.constexpr,
    BLOCK_P: tl.constexpr,
    EPS: tl.float32,
):
    pid = tl.program_id(0)
    b = tl.program_id(1)

    offs_p_raw = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    p_ok = offs_p_raw < P
    p = tl.minimum(offs_p_raw, P - 1)

    offs_h = tl.arange(0, H)
    base = b * H * P

    out_ptrs = out_t_ptr + base + offs_h[:, None] * P + p[None, :]
    gate_ptrs = gate_t_ptr + base + offs_h[:, None] * P + p[None, :]

    out_hp = tl.load(out_ptrs, mask=p_ok[None, :], other=0.0)
    gate_hp = tl.load(gate_ptrs, mask=p_ok[None, :], other=0.0)

    out_f = out_hp.to(tl.float32)
    gate_f = gate_hp.to(tl.float32)

    sum_h = tl.sum(out_f, axis=0)
    sq_h = tl.sum(out_f * out_f, axis=0)

    inv_h = 1.0 / H
    mean = sum_h * inv_h
    var = sq_h * inv_h - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + EPS)

    tw = tl.load(norm_w_ptr + offs_h)
    tb = tl.load(norm_b_ptr + offs_h)

    norm_f = (out_f - mean[None, :]) * rstd[None, :] * tw[:, None] + tb[:, None]
    gated_f = norm_f * gate_f

    gated = tl.trans(gated_f.to(tl.bfloat16))  # [BLOCK_P, H]

    for c0 in range(0, C, C_BLOCK):
        offs_c = c0 + tl.arange(0, C_BLOCK)
        c_ok = offs_c < C

        w_ptrs = w_ptr + offs_h[:, None] * C + offs_c[None, :]
        w_chunk = tl.load(w_ptrs, mask=c_ok[None, :], other=0.0)

        acc = tl.dot(gated, w_chunk)

        out_ptrs = out_ptr + (b * P + p[:, None]) * C + offs_c[None, :]
        tl.store(out_ptrs, acc, mask=p_ok[:, None] & c_ok[None, :])


_WEIGHT_CACHE = {}
_WEIGHT_NAMES = (
    "norm.weight",
    "norm.bias",
    "left_proj.weight",
    "right_proj.weight",
    "left_gate.weight",
    "right_gate.weight",
    "out_gate.weight",
    "to_out_norm.weight",
    "to_out_norm.bias",
    "to_out.weight",
)


def _pow2_leq(x: int) -> int:
    if x <= 0:
        return 1
    return 1 << (x - 1).bit_length()


def _get_prepared_weights(weights: dict, C: int, H: int, device: torch.device):
    dev_idx = device.index if device.type == "cuda" else 0
    key_parts = [C, H, dev_idx]
    for name in _WEIGHT_NAMES:
        t = weights[name]
        key_parts.append(t.data_ptr())
        key_parts.append(tuple(t.shape))
        key_parts.append(tuple(t.stride()))
        key_parts.append(t.dtype)
        key_parts.append(t.device.index if t.device.type == "cuda" else 0)
    key = tuple(key_parts)

    cached = _WEIGHT_CACHE.get(key)
    if cached is not None:
        return cached

    gamma = weights["norm.weight"].to(torch.float32).contiguous()
    beta = weights["norm.bias"].to(torch.float32).contiguous()
    gamma_row = gamma[None, :]

    w_left = weights["left_proj.weight"].to(torch.float32).contiguous()
    w_right = weights["right_proj.weight"].to(torch.float32).contiguous()
    w_left_g = weights["left_gate.weight"].to(torch.float32).contiguous()
    w_right_g = weights["right_gate.weight"].to(torch.float32).contiguous()
    w_out_g = weights["out_gate.weight"].to(torch.float32).contiguous()

    ws = [w_left, w_right, w_left_g, w_right_g, w_out_g]

    m_list = [(w * gamma_row).t().contiguous() for w in ws]
    m_comb = torch.stack(m_list, dim=0).contiguous().to(torch.bfloat16)

    g_list = [(w * gamma_row).sum(dim=1).contiguous() for w in ws]
    g_comb = torch.stack(g_list, dim=0).contiguous()

    b_list = [w @ beta for w in ws]
    b_comb = torch.stack(b_list, dim=0).contiguous()

    wto_t = weights["to_out.weight"].to(torch.float32).t().contiguous().to(torch.bfloat16)
    out_norm_w = weights["to_out_norm.weight"].to(torch.float32).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(torch.float32).contiguous()

    ret = (m_comb, g_comb, b_comb, wto_t, out_norm_w, out_norm_b)
    _WEIGHT_CACHE[key] = ret
    return ret


def custom_kernel(data):
    if isinstance(data, (tuple, list)) and len(data) == 3:
        x, weights, config = data
        mask = None
    else:
        x, mask, weights, config = data

    C = x.shape[-1]
    H = int(config["hidden_dim"])
    B, N, _, _ = x.shape
    P = N * N
    device = x.device

    if not x.is_contiguous():
        x = x.contiguous()

    nomask = bool(config.get("nomask", False))
    has_mask = (mask is not None) and (not nomask)
    if has_mask:
        mask = mask.contiguous()
        if mask.dim() == 4 and mask.shape[-1] == 1:
            mask = mask.squeeze(-1)
        if mask.dim() == 2:
            mask = mask.unsqueeze(0)
        if mask.shape[0] == 1 and B > 1:
            mask = mask.expand(B, N, N)
        mask = mask.to(torch.float32).contiguous()
    else:
        mask = x

    m_comb, g_comb, b_comb, wto_t, out_norm_w, out_norm_b = _get_prepared_weights(
        weights, C, H, device
    )

    left_t = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    right_t = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    ogate_t = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)

    BLOCK_R = 64 if C <= 384 else 32
    C_BLOCK_PRE = max(16, min(32, _pow2_leq(C)))

    pre_grid = (triton.cdiv(P, BLOCK_R), B)
    _trimul_pre_kernel[pre_grid](
        x,
        mask,
        left_t,
        right_t,
        ogate_t,
        m_comb,
        g_comb,
        b_comb,
        P,
        C,
        H=H,
        C_BLOCK=C_BLOCK_PRE,
        BLOCK_R=BLOCK_R,
        HAS_MASK=has_mask,
        EPS=1e-5,
        num_warps=8,
        num_stages=2,
    )

    out_t = torch.empty_like(left_t)
    torch.matmul(left_t, right_t.transpose(-2, -1), out=out_t)

    del left_t, right_t

    out = torch.empty((B, N, N, C), dtype=torch.float32, device=device)

    BLOCK_P = 64
    C_BLOCK_FINAL = max(16, min(128, _pow2_leq(C)))

    final_grid = (triton.cdiv(P, BLOCK_P), B)
    _trimul_final_kernel[final_grid](
        out_t,
        ogate_t,
        wto_t,
        out_norm_w,
        out_norm_b,
        out,
        P,
        C,
        H=H,
        C_BLOCK=C_BLOCK_FINAL,
        BLOCK_P=BLOCK_P,
        EPS=1e-5,
        num_warps=8,
        num_stages=2,
    )

    return out
