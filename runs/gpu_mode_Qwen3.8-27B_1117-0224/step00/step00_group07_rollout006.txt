```
                        stride_gb, stride_gr,
                        stride_ob, stride_or,
                        stride_wh, stride_wd,
                        H: tl.constexpr,
                        BLOCK_R: tl.constexpr,
                        BLOCK_D: tl.constexpr,
                        EPS: tl.constexpr):
    row_pid = tl.program_id(0)
    b = tl.program_id(1)
    rows = row_pid * BLOCK_R + tl.arange(0, BLOCK_R)
    rmask = rows < N2
    h = tl.arange(0, H)
    c_ptrs = C_ptr + b*stride_cb + h[:, None]*stride_ch + rows[None, :]
    c = tl.load(c_ptrs, mask=rmask[None, :], other=0.0).to(tl.float32)
    g_ptrs = G_ptr + b*stride_gb + rows[:, None]*stride_gr + h[None, :]
    g = tl.load(g_ptrs, mask=rmask[:, None], other=0.0).to(tl.float32)

    sum1 = tl.sum(c, axis=0)

    nw = tl.load(NW_ptr + h).to(tl.float32)
```

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# Fused LayerNorm + 5 projections + sigmoid/gate for small channel dim (C <= 128).
# This path computes LayerNorm explicitly, which is numerically safer for heavy-tailed input.
@triton.jit
def prepro_explicit_kernel(
    X_ptr, L_ptr, R_ptr, G_ptr, W_ptr, NW_ptr, NB_ptr, M_ptr,
    total_points, C, H, N2,
    stride_Lb, stride_Lh, stride_G,
    stride_Ww, stride_Wc,
    BLOCK_P: tl.constexpr,
    BLOCK_C: tl.constexpr,
    EPS: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    pmask = p < total_points

    h = tl.arange(0, H)
    c = tl.arange(0, BLOCK_C)
    cmask = c < C
    load_mask = pmask[:, None] & cmask[None, :]

    x = tl.load(X_ptr + p[:, None] * C + c[None, :], mask=load_mask, other=0.0).to(tl.float32)

    sumx = tl.sum(x, axis=1)
    sumsq = tl.sum(x * x, axis=1)
    mean = sumx / C
    var = sumsq / C - mean * mean
    var = tl.maximum(var, 0.0)
    inv = tl.rsqrt(var + EPS)

    nw = tl.load(NW_ptr + c, mask=cmask, other=0.0).to(tl.float32)
    nb = tl.load(NB_ptr + c, mask=cmask, other=0.0).to(tl.float32)

    xn = (x - mean[:, None]) * inv[:, None]
    xn = xn * nw[None, :] + nb[None, :]
    xn_bf = xn.to(tl.bfloat16)

    wmask = cmask[:, None]

    w0 = tl.load(W_ptr + 0 * stride_Ww + c[:, None] * stride_Wc + h[None, :], mask=wmask, other=0.0)
    w1 = tl.load(W_ptr + 1 * stride_Ww + c[:, None] * stride_Wc + h[None, :], mask=wmask, other=0.0)
    w2 = tl.load(W_ptr + 2 * stride_Ww + c[:, None] * stride_Wc + h[None, :], mask=wmask, other=0.0)
    w3 = tl.load(W_ptr + 3 * stride_Ww + c[:, None] * stride_Wc + h[None, :], mask=wmask, other=0.0)
    w4 = tl.load(W_ptr + 4 * stride_Ww + c[:, None] * stride_Wc + h[None, :], mask=wmask, other=0.0)

    accL = tl.zeros((BLOCK_P, H), dtype=tl.float32)
    accR = tl.zeros((BLOCK_P, H), dtype=tl.float32)
    accLG = tl.zeros((BLOCK_P, H), dtype=tl.float32)
    accRG = tl.zeros((BLOCK_P, H), dtype=tl.float32)
    accOG = tl.zeros((BLOCK_P, H), dtype=tl.float32)

    accL += tl.dot(xn_bf, w0)
    accR += tl.dot(xn_bf, w1)
    accLG += tl.dot(xn_bf, w2)
    accRG += tl.dot(xn_bf, w3)
    accOG += tl.dot(xn_bf, w4)

    lg = tl.sigmoid(accLG)
    rg = tl.sigmoid(accRG)
    og = tl.sigmoid(accOG)

    left = accL * lg
    right = accR * rg

        right = right * m[:, None]

    left = left.to(tl.bfloat16)
    right = right.to(tl.bfloat16)
    og = og.to(tl.bfloat16)

    b = p // N2

    G_ptrs = G_ptr + p[:, None] * stride_G + h[None, :]

    tl.store(L_ptrs, left, mask=pmask[:, None])
    tl.store(R_ptrs, right, mask=pmask[:, None])
    tl.store(G_ptrs, og, mask=pmask[:, None])

# Uses the algebraic identity:

# This avoids materializing normalized activations and is good for normal-distribution inputs.
@triton.jit
def prepro_u_kernel(

    EPS: tl.constexpr,

    accOG = tl.zeros((BLOCK_P, H), dtype=tl.float32)

        x = tl.load(X_ptr + p[:, None] * C + c[None, :], mask=load_mask, other=0.0).to(tl.float32)
        sumx += tl.sum(x, axis=1)
        sumsq += tl.sum(x * x, axis=1)

        x_bf = x.to(tl.bfloat16)
        umask = cmask[:, None]

        u0 = tl.load(U_ptr + 0 * stride_Uw + c[:, None] * H + h[None, :], mask=umask, other=0.0)
        u1 = tl.load(U_ptr + 1 * stride_Uw + c[:, None] * H + h[None, :], mask=umask, other=0.0)
        u2 = tl.load(U_ptr + 2 * stride_Uw + c[:, None] * H + h[None, :], mask=umask, other=0.0)
        u3 = tl.load(U_ptr + 3 * stride_Uw + c[:, None] * H + h[None, :], mask=umask, other=0.0)
        u4 = tl.load(U_ptr + 4 * stride_Uw + c[:, None] * H + h[None, :], mask=umask, other=0.0)

        accL += tl.dot(x_bf, u0)
        accR += tl.dot(x_bf, u1)
        accLG += tl.dot(x_bf, u2)
        accRG += tl.dot(x_bf, u3)
        accOG += tl.dot(x_bf, u4)

    mean = sumx / C
    var = sumsq / C - mean * mean
    var = tl.maximum(var, 0.0)
    inv = tl.rsqrt(var + EPS)

    sL = tl.load(S_ptr + 0 * H + h).to(tl.float32)
    sR = tl.load(S_ptr + 1 * H + h).to(tl.float32)
    sLG = tl.load(S_ptr + 2 * H + h).to(tl.float32)
    sRG = tl.load(S_ptr + 3 * H + h).to(tl.float32)
    sOG = tl.load(S_ptr + 4 * H + h).to(tl.float32)

    cL = tl.load(CONS_ptr + 0 * H + h).to(tl.float32)
    cR = tl.load(CONS_ptr + 1 * H + h).to(tl.float32)
    cLG = tl.load(CONS_ptr + 2 * H + h).to(tl.float32)
    cRG = tl.load(CONS_ptr + 3 * H + h).to(tl.float32)
    cOG = tl.load(CONS_ptr + 4 * H + h).to(tl.float32)

    left = inv[:, None] * (accL - mean[:, None] * sL[None, :]) + cL[None, :]
    right = inv[:, None] * (accR - mean[:, None] * sR[None, :]) + cR[None, :]
    lg = inv[:, None] * (accLG - mean[:, None] * sLG[None, :]) + cLG[None, :]
    rg = inv[:, None] * (accRG - mean[:, None] * sRG[None, :]) + cRG[None, :]
    og = inv[:, None] * (accOG - mean[:, None] * sOG[None, :]) + cOG[None, :]

    lg = tl.sigmoid(lg)
    rg = tl.sigmoid(rg)
    og = tl.sigmoid(og)

    left = left * lg
    right = right * rg

    if HAS_MASK:
        m = tl.load(M_ptr + p, mask=pmask, other=0.0).to(tl.float32)
        left = left * m[:, None]
        right = right * m[:, None]

    left = left.to(tl.bfloat16)
    right = right.to(tl.bfloat16)
    og = og.to(tl.bfloat16)

    b = p // N2
    idx = p - b * N2

    L_ptrs = L_ptr + b[:, None] * stride_Lb + h[None, :] * stride_Lh + idx[:, None]
    R_ptrs = R_ptr + b[:, None] * stride_Lb + h[None, :] * stride_Lh + idx[:, None]
    G_ptrs = G_ptr + p[:, None] * stride_G + h[None, :]

    tl.store(L_ptrs, left, mask=pmask[:, None])
    tl.store(R_ptrs, right, mask=pmask[:, None])
    tl.store(G_ptrs, og, mask=pmask[:, None])


# Batched GEMM over heads:
# For each b,h:
#   C[b,h,i,j] = sum_k L[b,h,i,k] * R[b,h,j,k]
# L/R/C are head-major [B, H, N, N].
@triton.jit
def core_gemm_kernel(
    L_ptr, R_ptr, C_ptr,
    N, H,
    stride_Lb, stride_Lh, stride_Lm, stride_Lk,
    stride_Rb, stride_Rh, stride_Rn, stride_Rk,
    stride_Cb, stride_Ch, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr,
    EVEN: tl.constexpr,
):
    pid = tl.program_id(0)

    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    num_tiles = num_m * num_n

    bh = pid // num_tiles
    tile = pid % num_tiles

    num_pid_in_group = GROUP_M * num_n
    group_id = tile // num_pid_in_group
    first_pid_m = group_id * GROUP_M
    group_size_m = tl.minimum(num_m - first_pid_m, GROUP_M)

    pid_m = first_pid_m + (tile % group_size_m)
    pid_n = (tile % num_pid_in_group) // group_size_m

    b = bh // H
    h = bh % H

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    L_base = L_ptr + b * stride_Lb + h * stride_Lh
    R_base = R_ptr + b * stride_Rb + h * stride_Rh
    C_base = C_ptr + b * stride_Cb + h * stride_Ch

    A_ptrs = L_base + offs_m[:, None] * stride_Lm + offs_k[None, :] * stride_Lk
    B_ptrs = R_base + offs_k[:, None] * stride_Rk + offs_n[None, :] * stride_Rn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    if EVEN:
        for _ in range(0, N, BLOCK_K):
            a = tl.load(A_ptrs)
            b_t = tl.load(B_ptrs)
            acc += tl.dot(a, b_t)
            A_ptrs += BLOCK_K * stride_Lk
            B_ptrs += BLOCK_K * stride_Rk
    else:
        mask_m = (offs_m < N)[:, None]
        mask_n = (offs_n < N)[None, :]
        for k in range(0, N, BLOCK_K):
            kmask = k + offs_k < N
            a = tl.load(A_ptrs, mask=mask_m & kmask[None, :], other=0.0)
            b_t = tl.load(B_ptrs, mask=kmask[:, None] & mask_n, other=0.0)
            acc += tl.dot(a, b_t)
            A_ptrs += BLOCK_K * stride_Lk
            B_ptrs += BLOCK_K * stride_Rk

    c_ptrs = C_base + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    if EVEN:
        tl.store(c_ptrs, acc)
    else:
        tl.store(c_ptrs, acc, mask=(offs_m < N)[:, None] & (offs_n < N)[None, :])


# Second LayerNorm + out_gate + final projection.
# C is head-major [B, H, N, N]; gate is row-major [B, N*N, H]; out is row-major [B, N*N, dim].
@triton.jit
def final_ln_proj_kernel(
    C_ptr, G_ptr, O_ptr, W_t_ptr, NW_ptr, NB_ptr,
    N2, dim,
    stride_Cb, stride_Ch,
    stride_Gb, stride_Gp,
    stride_Ob, stride_Op,
    stride_Wh, stride_Wd,
    H: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_D: tl.constexpr,
    EPS: tl.constexpr,
):
    row_pid = tl.program_id(0)
    b = tl.program_id(1)

    rows = row_pid * BLOCK_R + tl.arange(0, BLOCK_R)
    rmask = rows < N2
    h = tl.arange(0, H)

    # Load C as [H, R] for efficient row-contiguous loads within each head.
    c_ptrs = C_ptr + b * stride_Cb + h[:, None] * stride_Ch + rows[None, :]
    c = tl.load(c_ptrs, mask=rmask[None, :], other=0.0).to(tl.float32)

    cr = tl.trans(c)  # [R, H]

    mean = tl.sum(cr, axis=1) / H
    cc = cr - mean[:, None]
    var = tl.sum(cc * cc, axis=1) / H
    var = tl.maximum(var, 0.0)
    inv = tl.rsqrt(var + EPS)

    nw = tl.load(NW_ptr + h).to(tl.float32)
    nb = tl.load(NB_ptr + h).to(tl.float32)

    y = cc * inv[:, None]
    y = y * nw[None, :] + nb[None, :]

    g_ptrs = G_ptr + b * stride_Gb + rows[:, None] * stride_Gp + h[None, :]
    g = tl.load(g_ptrs, mask=rmask[:, None], other=0.0).to(tl.float32)

    z = (y * g).to(tl.bfloat16)

    for d0 in range(0, dim, BLOCK_D):
        d = d0 + tl.arange(0, BLOCK_D)
        dmask = d < dim

        w_ptrs = W_t_ptr + h[:, None] * stride_Wh + d[None, :] * stride_Wd
        w = tl.load(w_ptrs, mask=dmask[None, :], other=0.0)

        out = tl.dot(z, w)

        o_ptrs = O_ptr + b * stride_Ob + rows[:, None] * stride_Op + d[None, :]
        tl.store(o_ptrs, out, mask=rmask[:, None] & dmask[None, :])


def _next_pow2(n: int) -> int:
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _pytorch_prepro(x, mask, weights, config, B, N, C, H, device):
    x = x.contiguous().float()
    eps = 1e-5
    dt = torch.bfloat16

    norm_w = weights["norm.weight"].float()
    norm_b = weights["norm.bias"].float()
    xn = F.layer_norm(x, (C,), norm_w, norm_b, eps=eps)
    xn_bf = xn.to(dt)

    w_left = weights["left_proj.weight"].to(dt)
    w_right = weights["right_proj.weight"].to(dt)
    w_lg = weights["left_gate.weight"].to(dt)
    w_rg = weights["right_gate.weight"].to(dt)
    w_og = weights["out_gate.weight"].to(dt)

    has_mask = mask is not None and not config.get("nomask", False)
    if has_mask:
        m = mask.contiguous().to(dt).unsqueeze(-1)

    left = xn_bf @ w_left.t()
    lg = torch.sigmoid(xn_bf @ w_lg.t())
    left = left * lg
    if has_mask:
        left = left * m

    right = xn_bf @ w_right.t()
    rg = torch.sigmoid(xn_bf @ w_rg.t())
    right = right * rg
    if has_mask:
        right = right * m

    og = torch.sigmoid(xn_bf @ w_og.t())

    left_h = left.permute(0, 3, 1, 2).contiguous()
    right_h = right.permute(0, 3, 1, 2).contiguous()
    g = og.contiguous().view(B, N * N, H)

    return left_h, right_h, g


def _pytorch_final(C_head, gate, weights, B, N, C_dim, H, device):
    C = C_head.permute(0, 2, 3, 1).contiguous().float()  # [B, N, N, H]
    g = gate.view(B, N * N, H).float()
    y = F.layer_norm(
        C, (H,),
        weights["to_out_norm.weight"].float(),
        weights["to_out_norm.bias"].float(),
        eps=1e-5,
    )
    z = y * g
    out = F.linear(z, weights["to_out.weight"].float())
    return out.view(B, N, N, C_dim).float()


def custom_kernel(data):
    """
    Triton-optimized forward TriMul.

    Algorithm:
      1) Fused first LayerNorm + left/right/gate projections + gating/mask:
         - For C <= 128, explicit robust LayerNorm + bf16 GEMMs.
         - For C  > 128, one-pass algebraic projection after LayerNorm stats.
         Outputs left/right in head-major [B, H, N, N] and out_gate in [B, N*N, H].
      2) Triton batched GEMM over heads:
         C[b,h,i,j] = sum_k L[b,h,i,k] * R[b,h,j,k].
      3) Triton fused second LayerNorm + out_gate + final projection.
    """
    if len(data) == 4:
        input_tensor, mask, weights, config = data
    else:
        input_tensor, weights, config = data
        mask = None

    B, N, _, C_dim = input_tensor.shape
    H = int(config.get("hidden_dim", 128))
    device = input_tensor.device
    N2 = N * N
    total_points = B * N2
    has_mask = mask is not None and not config.get("nomask", False)

    x = input_tensor.contiguous().float()

    w_names = [
        "left_proj.weight",
        "right_proj.weight",
        "left_gate.weight",
        "right_gate.weight",
        "out_gate.weight",
    ]

    # Choose prepro path.
    if H == 128:
        L = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
        R = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
        G = torch.empty((B, N2, H), dtype=torch.bfloat16, device=device)

        m_flat = x  # dummy pointer when no mask
        if has_mask:
            m_flat = mask.contiguous().float().view(-1)

        if C_dim <= 128:
            Wt_list = []
            for name in w_names:
                Wt_list.append(weights[name].float().T.contiguous())
            W_all = torch.stack(Wt_list, dim=0).to(torch.bfloat16).contiguous()  # [5, C, H]

            norm_w = weights["norm.weight"].float().contiguous()
            norm_b = weights["norm.bias"].float().contiguous()

            BLOCK_P = 32
            BLOCK_C = _next_pow2(C_dim)
            if BLOCK_C < 16:
                BLOCK_C = 16

            grid = (triton.cdiv(total_points, BLOCK_P),)
            prepro_explicit_kernel[grid](
                x, L, R, G,
                W_all, norm_w, norm_b, m_flat,
                total_points, C_dim, H, N2,
                H * N2, N2, H,
                C_dim * H, H,
                BLOCK_P, BLOCK_C,
                1e-5,
                has_mask,
                num_warps=8,
                num_stages=3,
            )
        else:
            norm_w = weights["norm.weight"].float()
            norm_b = weights["norm.bias"].float()

            U_list = []
            S_list = []
            CONS_list = []

            for name in w_names:
                W = weights[name].float()  # [H, C]
                u = (W * norm_w[None, :]).T.contiguous()  # [C, H]
                U_list.append(u)
                S_list.append(u.sum(dim=0, dtype=torch.float32))  # [H]
                CONS_list.append(norm_b @ W.T)  # [H]

            U_all = torch.stack(U_list, dim=0).to(torch.bfloat16).contiguous()  # [5, C, H]
            S_all = torch.stack(S_list, dim=0).float().contiguous()             # [5, H]
            CONS_all = torch.stack(CONS_list, dim=0).float().contiguous()       # [5, H]

            BLOCK_P = 32
            BLOCK_C = 128

            grid = (triton.cdiv(total_points, BLOCK_P),)
            prepro_u_kernel[grid](
                x, L, R, G,
                U_all, S_all, CONS_all, m_flat,
                total_points, C_dim, H, N2,
                H * N2, N2, H,
                C_dim * H,
                BLOCK_P, BLOCK_C,
                1e-5,
                has_mask,
                num_warps=8,
                num_stages=3,
            )
        del x
    else:
        L, R, G = _pytorch_prepro(x, mask, weights, config, B, N, C_dim, H, device)
        del x

    # Core batched GEMM over heads.
    C_head = torch.empty((B, H, N, N), dtype=torch.float32, device=device)

    if N >= 512:
        BLOCK_M = 128
        BLOCK_N = 128
        num_warps_core = 8
    else:
        BLOCK_M = 64
        BLOCK_N = 64
        num_warps_core = 4
    BLOCK_K = 64
    GROUP_M = 8

    num_m = triton.cdiv(N, BLOCK_M)
    num_n = triton.cdiv(N, BLOCK_N)
    grid_core = (B * H * num_m * num_n,)

    even = (N % BLOCK_M == 0) and (N % BLOCK_N == 0) and (N % BLOCK_K == 0)

    l_str_b, l_str_h, l_str_m, l_str_k = L.stride()
    r_str_b, r_str_h, r_str_n, r_str_k = R.stride()
    c_str_b, c_str_h, c_str_m, c_str_n = C_head.stride()

    core_gemm_kernel[grid_core](
        L, R, C_head,
        N, H,
        l_str_b, l_str_h, l_str_m, l_str_k,
        r_str_b, r_str_h, r_str_n, r_str_k,
        c_str_b, c_str_h, c_str_m, c_str_n,
        BLOCK_M, BLOCK_N, BLOCK_K, GROUP_M, even,
        num_warps=num_warps_core,
        num_stages=3,
    )

    # Final fused LN + gate + projection.
    out_dim = C_dim
    final_supported = H in (128, 256, 512, 1024)

    if final_supported:
        out = torch.empty((B, N, N, out_dim), dtype=torch.float32, device=device)
        W_t = weights["to_out.weight"].float().T.contiguous().to(torch.bfloat16)  # [H, dim]
        norm_w2 = weights["to_out_norm.weight"].float().contiguous()
        norm_b2 = weights["to_out_norm.bias"].float().contiguous()

        if H == 128:
            BLOCK_R = 64
        elif H == 256:
            BLOCK_R = 32
        elif H == 512:
            BLOCK_R = 16
        else:
            BLOCK_R = 8

        BLOCK_D = 128
        grid_final = (triton.cdiv(N2, BLOCK_R), B)

        final_ln_proj_kernel[grid_final](
            C_head, G, out, W_t, norm_w2, norm_b2,
            N2, out_dim,
            H * N2, N2,
            N2 * H, H,
            N2 * out_dim, out_dim,
            out_dim, 1,
            H, BLOCK_R, BLOCK_D,
            1e-5,
            num_warps=4,
            num_stages=3,
        )
    else:
        out = _pytorch_final(C_head, G, weights, B, N, out_dim, H, device)

    return out.float()
```
