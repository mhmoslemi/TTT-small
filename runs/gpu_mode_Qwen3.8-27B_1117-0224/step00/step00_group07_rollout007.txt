```
@triton.jit
def _pre_trimul_kernel(
    x_ptr, mask_ptr, out_l_ptr, out_r_ptr, out_g_ptr,
    wcat_ptr, wbeta_ptr, gamma_ptr,
    M, C, H, SS, HSS, HC, eps,
    HAS_MASK: tl.constexpr,
    BM: tl.constexpr,
    BH: tl.constexpr,
    BC: tl.constexpr,
):
    pid = tl.program_id(0)
    tok = pid * BM + tl.arange(0, BM)
    tok_mask = tok < M
    b = tok // SS
    seq = tok % SS
    b_off = b * HSS

    s1 = tl.zeros((BM,), dtype=tl.float32)
    s2 = tl.zeros((BM,), dtype=tl.float32)
    for c0 in range(0, C, BC):
        c_off = c0 + tl.arange(0, BC)
        c_mask = c_off < C
        x = tl.load(x_ptr + tok[:, None] * C + c_off[None, :],
                    mask=tok_mask[:, None] & c_mask[None, :], other=0.0).to(tl.float32)
        s1 += tl.sum(x, axis=1)
        s2 += tl.sum(x * x, axis=1)
    mean = s1 / C
    var = s2 / C - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + eps)

    if HAS_MASK:
        m = tl.load(mask_ptr + tok, mask=tok_mask, other=0.0).to(tl.float32)
    else:
        m = tl.zeros((BM,), dtype=tl.float32) + 1.0

    w0_ptr = wcat_ptr
    w1_ptr = wcat_ptr + HC
    w2_ptr = wcat_ptr + 2 * HC
    w3_ptr = wcat_ptr + 3 * HC
    w4_ptr = wcat_ptr + 4 * HC
    wb0_ptr = wbeta_ptr
    wb1_ptr = wbeta_ptr + H
    wb2_ptr = wbeta_ptr + 2 * H
    wb3_ptr = wbeta_ptr + 3 * H
    wb4_ptr = wbeta_ptr + 4 * H

    for h0 in range(0, H, BH):
        h_off = h0 + tl.arange(0, BH)
        h_mask = h_off < H

        acc0 = tl.zeros((BM, BH), dtype=tl.float32)
        acc1 = tl.zeros((BM, BH), dtype=tl.float32)
        acc2 = tl.zeros((BM, BH), dtype=tl.float32)
        acc3 = tl.zeros((BM, BH), dtype=tl.float32)
        acc4 = tl.zeros((BM, BH), dtype=tl.float32)

        for c0 in range(0, C, BC):
            c_off = c0 + tl.arange(0, BC)
            c_mask = c_off < C
            x = tl.load(x_ptr + tok[:, None] * C + c_off[None, :],
                        mask=tok_mask[:, None] & c_mask[None, :], other=0.0).to(tl.float32)
            g = tl.load(gamma_ptr + c_off, mask=c_mask, other=0.0).to(tl.float32)
            xn = ((x - mean[:, None]) * rstd[:, None]) * g[None, :]
            xnb = xn.to(tl.bfloat16)

            w0 = tl.load(w0_ptr + h_off[:, None] * C + c_off[None, :],
                         mask=h_mask[:, None] & c_mask[None, :], other=0.0)
            acc0 = tl.dot(xnb, tl.trans(w0), acc=acc0)
            w1 = tl.load(w1_ptr + h_off[:, None] * C + c_off[None, :],
                         mask=h_mask[:, None] & c_mask[None, :], other=0.0)
            acc1 = tl.dot(xnb, tl.trans(w1), acc=acc1)
            ...
        wb0 = tl.load(wb0_ptr + h_off, mask=h_mask, other=0.0)
        ...
        y0 = acc0 + wb0[None, :]
        ...
        sg1 = 1.0 / (1.0 + tl.exp(-y1))
        ...
        left = y0 * sg1 * m[:, None]
        right = y2 * sg3 * m[:, None]
        gate = sg4
        sm = h_mask[:, None] & tok_mask[None, :]
        ptr = out_l_ptr + b_off[None, :] + h_off[:, None] * SS + seq[None, :]
        tl.store(ptr, tl.trans(left.to(tl.bfloat16)), mask=sm)
        ptr = out_r_ptr + b_off[None, :] + h_off[:, None] * SS + seq[None, :]
        tl.store(ptr, tl.trans(right.to(tl.bfloat16)), mask=sm)
        ptr = out_g_ptr + b_off[None, :] + h_off[:, None] * SS + seq[None, :]
        tl.store(ptr, tl.trans(gate.to(tl.bfloat16)), mask=sm)
```

```
@triton.jit
def _final_trimul_kernel(
    out_h_ptr, gate_h_ptr, wout_ptr, lnw_ptr, lnb_ptr, out_ptr,
    M, C, H, SS, HSS, eps,
    BM: tl.constexpr, BH: tl.constexpr, BC: tl.constexpr,
):
    pid = tl.program_id(0)
    tok = pid * BM + tl.arange(0, BM)
    tok_mask = tok < M
    b = tok // SS
    seq = tok % SS
    b_off = b * HSS
    seq_2d = seq[None, :]
    b_2d = b_off[None, :]

    s1 = tl.zeros((BM,), dtype=tl.float32)
    s2 = tl.zeros((BM,), dtype=tl.float32)
    for h0 in range(0, H, BH):
        h_off = h0 + tl.arange(0, BH)
        h_mask = h_off < H
        ptr = out_h_ptr + b_2d + h_off[:, None] * SS + seq_2d
        o = tl.load(ptr, mask=tok_mask[None, :] & h_mask[:, None], other=0.0).to(tl.float32)
        s1 += tl.sum(o, axis=0)
        s2 += tl.sum(o * o, axis=0)
    mean = s1 / H
    var = s2 / H - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + eps)

    for c0 in range(0, C, BC):
        c_off = c0 + tl.arange(0, BC)
        c_mask = c_off < C
        acc = tl.zeros((BM, BC), dtype=tl.float32)
        for h0 in range(0, H, BH):
            h_off = h0 + tl.arange(0, BH)
            h_mask = h_off < H
            o_ptr = out_h_ptr + b_2d + h_off[:, None] * SS + seq_2d
            g_ptr = gate_h_ptr + b_2d + h_off[:, None] * SS + seq_2d
            o = tl.load(o_ptr, mask=tok_mask[None, :] & h_mask[:, None], other=0.0).to(tl.float32)
            g = tl.load(g_ptr, mask=tok_mask[None, :] & h_mask[:, None], other=0.0).to(tl.float32)
            lnw = tl.load(lnw_ptr + h_off, mask=h_mask, other=0.0).to(tl.float32)
            lnb = tl.load(lnb_ptr + h_off, mask=h_mask, other=0.0).to(tl.float32)
            val_hb = (o - mean[None, :]) * rstd[None, :]
            val_hb = val_hb * lnw[:, None] + lnb[:, None]
            val_hb = val_hb * g
            val = tl.trans(val_hb.to(tl.bfloat16))
            w_ch = tl.load(wout_ptr + c_off[:, None] * H + h_off[None, :],
                           mask=c_mask[:, None] & h_mask[None, :], other=0.0)
            w = tl.trans(w_ch)
            acc = tl.dot(val, w, acc=acc)
        optr = out_ptr + tok[:, None] * C + c_off[None, :]
        tl.store(optr, acc, mask=tok_mask[:, None] & c_mask[None, :])
```

```python
"""
Fused AlphaFold3-style outgoing TriMul.

Pipeline:
1) Triton kernel: fused LayerNorm + left/right projections + input gates + mask.
   Left/right/gate tensors are written in [B, H, N, N] layout to feed a batched
   matrix multiply efficiently.
2) PyTorch cuBLAS batched matmul:
       out[b, h, i, j] = sum_k left[b, h, i, k] * right[b, h, j, k]
3) Triton kernel: fused LayerNorm over H, out_gate, and final to_out projection.

The final result is returned in float32.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _pre_trimul_c128_kernel(
    x_ptr,
    mask_ptr,
    out_l_ptr,
    out_r_ptr,
    out_g_ptr,
    wcat_ptr,
    wbeta_ptr,
    gamma_ptr,
    M,
    H,
    SS,
    HSS,
    HC,
    eps,
    HAS_MASK: tl.constexpr,
    BM: tl.constexpr,
    BH: tl.constexpr,
):
    # Specialized fast path for C == 128.
    pid = tl.program_id(0)
    tok_raw = pid * BM + tl.arange(0, BM)
    tok_valid = tok_raw < M
    tok = tl.where(tok_valid, tok_raw, 0)

    b = tok // SS
    seq = tok % SS
    b_off = b * HSS

    c_off = tl.arange(0, 128)
    c_mask = c_off < 128
    x_mask = tok_valid[:, None] & c_mask[None, :]

    x = tl.load(x_ptr + tok[:, None] * 128 + c_off[None, :], mask=x_mask, other=0.0).to(tl.float32)

    s1 = tl.sum(x, axis=1)
    s2 = tl.sum(x * x, axis=1)
    mean = s1 / 128.0
    var = s2 / 128.0 - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + eps)

    gamma = tl.load(gamma_ptr + c_off, mask=c_mask, other=0.0).to(tl.float32)
    xn = ((x - mean[:, None]) * rstd[:, None]) * gamma[None, :]
    xnb = xn.to(tl.bfloat16)

    if HAS_MASK:
        m = tl.load(mask_ptr + tok, mask=tok_valid, other=0.0).to(tl.float32)
    else:
        m = tl.zeros((BM,), dtype=tl.float32) + 1.0

    # 0: left_proj, 1: left_gate, 2: right_proj, 3: right_gate, 4: out_gate
    w0_ptr = wcat_ptr
    w1_ptr = wcat_ptr + HC
    w2_ptr = wcat_ptr + 2 * HC
    w3_ptr = wcat_ptr + 3 * HC
    w4_ptr = wcat_ptr + 4 * HC

    wb0_ptr = wbeta_ptr
    wb1_ptr = wbeta_ptr + H
    wb2_ptr = wbeta_ptr + 2 * H
    wb3_ptr = wbeta_ptr + 3 * H
    wb4_ptr = wbeta_ptr + 4 * H

    for h0 in range(0, H, BH):
        h_off = h0 + tl.arange(0, BH)
        h_mask = h_off < H
        h_safe = tl.where(h_mask, h_off, 0)
        wm = h_mask[:, None] & c_mask[None, :]

        # left
        acc0 = tl.zeros((BM, BH), dtype=tl.float32)
        acc1 = tl.zeros((BM, BH), dtype=tl.float32)

        w0 = tl.load(w0_ptr + h_safe[:, None] * 128 + c_off[None, :], mask=wm, other=0.0)
        acc0 = tl.dot(xnb, tl.trans(w0), acc0)

        w1 = tl.load(w1_ptr + h_safe[:, None] * 128 + c_off[None, :], mask=wm, other=0.0)
        acc1 = tl.dot(xnb, tl.trans(w1), acc1)

        wb0 = tl.load(wb0_ptr + h_safe, mask=h_mask, other=0.0)
        wb1 = tl.load(wb1_ptr + h_safe, mask=h_mask, other=0.0)

        y0 = acc0 + wb0[None, :]
        y1 = acc1 + wb1[None, :]
        sg1 = 1.0 / (1.0 + tl.exp(-y1))
        left = y0 * sg1 * m[:, None]

        store_mask = h_mask[:, None] & tok_valid[None, :]
        ptr_l = out_l_ptr + b_off[None, :] + h_safe[:, None] * SS + seq[None, :]
        tl.store(ptr_l, tl.trans(left.to(tl.bfloat16)), mask=store_mask)

        # right
        acc2 = tl.zeros((BM, BH), dtype=tl.float32)
        acc3 = tl.zeros((BM, BH), dtype=tl.float32)

        w2 = tl.load(w2_ptr + h_safe[:, None] * 128 + c_off[None, :], mask=wm, other=0.0)
        acc2 = tl.dot(xnb, tl.trans(w2), acc=acc2)

        w3 = tl.load(w3_ptr + h_safe[:, None] * 128 + c_off[None, :], mask=wm, other=0.0)
        acc3 = tl.dot(xnb, tl.trans(w3), acc=acc3)

        wb2 = tl.load(wb2_ptr + h_safe, mask=h_mask, other=0.0)
        wb3 = tl.load(wb3_ptr + h_safe, mask=h_mask, other=0.0)

        y2 = acc2 + wb2[None, :]
        y3 = acc3 + wb3[None, :]
        sg3 = 1.0 / (1.0 + tl.exp(-y3))
        right = y2 * sg3 * m[:, None]

        ptr_r = out_r_ptr + b_off[None, :] + h_safe[:, None] * SS + seq[None, :]
        tl.store(ptr_r, tl.trans(right.to(tl.bfloat16)), mask=store_mask)

        # out gate
        acc4 = tl.zeros((BM, BH), dtype=tl.float32)
        w4 = tl.load(w4_ptr + h_safe[:, None] * 128 + c_off[None, :], mask=wm, other=0.0)
        acc4 = tl.dot(xnb, tl.trans(w4), acc=acc4)

        wb4 = tl.load(wb4_ptr + h_safe, mask=h_mask, other=0.0)
        y4 = acc4 + wb4[None, :]
        gate = 1.0 / (1.0 + tl.exp(-y4))

        ptr_g = out_g_ptr + b_off[None, :] + h_safe[:, None] * SS + seq[None, :]
        tl.store(ptr_g, tl.trans(gate.to(tl.bfloat16)), mask=store_mask)


@triton.jit
def _pre_trimul_generic_kernel(
    x_ptr,
    mask_ptr,
    out_l_ptr,
    out_r_ptr,
    out_g_ptr,
    wcat_ptr,
    wbeta_ptr,
    gamma_ptr,
    M,
    C,
    H,
    SS,
    HSS,
    HC,
    eps,
    HAS_MASK: tl.constexpr,
    BM: tl.constexpr,
    BH: tl.constexpr,
    BC: tl.constexpr,
):
    # Generic path for C != 128 (e.g. 384/768).
    pid = tl.program_id(0)
    tok_raw = pid * BM + tl.arange(0, BM)
    tok_valid = tok_raw < M
    tok = tl.where(tok_valid, tok_raw, 0)

    b = tok // SS
    seq = tok % SS
    b_off = b * HSS

    s1 = tl.zeros((BM,), dtype=tl.float32)
    s2 = tl.zeros((BM,), dtype=tl.float32)

    for c0 in range(0, C, BC):
        c_off = c0 + tl.arange(0, BC)
        c_mask = c_off < C
        c_safe = tl.where(c_mask, c_off, 0)
        x = tl.load(
            x_ptr + tok[:, None] * C + c_safe[None, :],
            mask=tok_valid[:, None] & c_mask[None, :],
            other=0.0,
        ).to(tl.float32)
        s1 += tl.sum(x, axis=1)
        s2 += tl.sum(x * x, axis=1)

    mean = s1 / C
    var = s2 / C - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + eps)

    if HAS_MASK:
        m = tl.load(mask_ptr + tok, mask=tok_valid, other=0.0).to(tl.float32)
    else:
        m = tl.zeros((BM,), dtype=tl.float32) + 1.0

    w0_ptr = wcat_ptr
    w1_ptr = wcat_ptr + HC
    w2_ptr = wcat_ptr + 2 * HC
    w3_ptr = wcat_ptr + 3 * HC
    w4_ptr = wcat_ptr + 4 * HC

    wb0_ptr = wbeta_ptr
    wb1_ptr = wbeta_ptr + H
    wb2_ptr = wbeta_ptr + 2 * H
    wb3_ptr = wbeta_ptr + 3 * H
    wb4_ptr = wbeta_ptr + 4 * H

    for h0 in range(0, H, BH):
        h_off = h0 + tl.arange(0, BH)
        h_mask = h_off < H
        h_safe = tl.where(h_mask, h_off, 0)

        acc0 = tl.zeros((BM, BH), dtype=tl.float32)
        acc1 = tl.zeros((BM, BH), dtype=tl.float32)
        acc2 = tl.zeros((BM, BH), dtype=tl.float32)
        acc3 = tl.zeros((BM, BH), dtype=tl.float32)
        acc4 = tl.zeros((BM, BH), dtype=tl.float32)

        for c0 in range(0, C, BC):
            c_off = c0 + tl.arange(0, BC)
            c_mask = c_off < C
            c_safe = tl.where(c_mask, c_off, 0)

            x = tl.load(
                x_ptr + tok[:, None] * C + c_safe[None, :],
                mask=tok_valid[:, None] & c_mask[None, :],
                other=0.0,
            ).to(tl.float32)

            g = tl.load(gamma_ptr + c_safe, mask=c_mask, other=0.0).to(tl.float32)
            xn = ((x - mean[:, None]) * rstd[:, None]) * g[None, :]
            xnb = xn.to(tl.bfloat16)

            wm = h_mask[:, None] & c_mask[None, :]

            w0 = tl.load(w0_ptr + h_safe[:, None] * C + c_safe[None, :], mask=wm, other=0.0)
            acc0 = tl.dot(xnb, tl.trans(w0), acc=acc0)

            w1 = tl.load(w1_ptr + h_safe[:, None] * C + c_safe[None, :], mask=wm, other=0.0)
            acc1 = tl.dot(xnb, tl.trans(w1), acc=acc1)

            w2 = tl.load(w2_ptr + h_safe[:, None] * C + c_safe[None, :], mask=wm, other=0.0)
            acc2 = tl.dot(xnb, tl.trans(w2), acc=acc2)

            w3 = tl.load(w3_ptr + h_safe[:, None] * C + c_safe[None, :], mask=wm, other=0.0)
            acc3 = tl.dot(xnb, tl.trans(w3), acc=acc3)

            w4 = tl.load(w4_ptr + h_safe[:, None] * C + c_safe[None, :], mask=wm, other=0.0)
            acc4 = tl.dot(xnb, tl.trans(w4), acc=acc4)

        wb0 = tl.load(wb0_ptr + h_safe, mask=h_mask, other=0.0)
        wb1 = tl.load(wb1_ptr + h_safe, mask=h_mask, other=0.0)
        wb2 = tl.load(wb2_ptr + h_safe, mask=h_mask, other=0.0)
        wb3 = tl.load(wb3_ptr + h_safe, mask=h_mask, other=0.0)
        wb4 = tl.load(wb4_ptr + h_safe, mask=h_mask, other=0.0)

        y0 = acc0 + wb0[None, :]
        y1 = acc1 + wb1[None, :]
        y2 = acc2 + wb2[None, :]
        y3 = acc3 + wb3[None, :]
        y4 = acc4 + wb4[None, :]

        left = y0 * (1.0 / (1.0 + tl.exp(-y1))) * m[:, None]
        right = y2 * (1.0 / (1.0 + tl.exp(-y3))) * m[:, None]
        gate = 1.0 / (1.0 + tl.exp(-y4))

        store_mask = h_mask[:, None] & tok_valid[None, :]

        ptr_l = out_l_ptr + b_off[None, :] + h_safe[:, None] * SS + seq[None, :]
        tl.store(ptr_l, tl.trans(left.to(tl.bfloat16)), mask=store_mask)

        ptr_r = out_r_ptr + b_off[None, :] + h_safe[:, None] * SS + seq[None, :]
        tl.store(ptr_r, tl.trans(right.to(tl.bfloat16)), mask=store_mask)

        ptr_g = out_g_ptr + b_off[None, :] + h_safe[:, None] * SS + seq[None, :]
        tl.store(ptr_g, tl.trans(gate.to(tl.bfloat16)), mask=store_mask)


@triton.jit
def _final_trimul_kernel(
    out_h_ptr,
    gate_h_ptr,
    wout_ptr,
    lnw_ptr,
    lnb_ptr,
    out_ptr,
    M,
    C,
    H,
    SS,
    HSS,
    eps,
    BM: tl.constexpr,
    BH: tl.constexpr,
    BC: tl.constexpr,
):
    # Fused LayerNorm over H + out_gate + to_out projection.
    pid = tl.program_id(0)
    tok_raw = pid * BM + tl.arange(0, BM)
    tok_valid = tok_raw < M
    tok = tl.where(tok_valid, tok_raw, 0)

    b = tok // SS
    seq = tok % SS
    b_off = b * HSS

    b_2d = b_off[None, :]
    seq_2d = seq[None, :]

    s1 = tl.zeros((BM,), dtype=tl.float32)
    s2 = tl.zeros((BM,), dtype=tl.float32)

    for h0 in range(0, H, BH):
        h_off = h0 + tl.arange(0, BH)
        h_mask = h_off < H
        h_safe = tl.where(h_mask, h_off, 0)

        ptr = out_h_ptr + b_2d + h_safe[:, None] * SS + seq_2d
        o = tl.load(ptr, mask=tok_valid[None, :] & h_mask[:, None], other=0.0).to(tl.float32)
        s1 += tl.sum(o, axis=0)
        s2 += tl.sum(o * o, axis=0)

    mean = s1 / H
    var = s2 / H - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + eps)

    for c0 in range(0, C, BC):
        c_off = c0 + tl.arange(0, BC)
        c_mask = c_off < C
        c_safe = tl.where(c_mask, c_off, 0)

        acc = tl.zeros((BM, BC), dtype=tl.float32)

        for h0 in range(0, H, BH):
            h_off = h0 + tl.arange(0, BH)
            h_mask = h_off < H
            h_safe = tl.where(h_mask, h_off, 0)

            ptr_o = out_h_ptr + b_2d + h_safe[:, None] * SS + seq_2d
            ptr_g = gate_h_ptr + b_2d + h_safe[:, None] * SS + seq_2d

            o = tl.load(ptr_o, mask=tok_valid[None, :] & h_mask[:, None], other=0.0).to(tl.float32)
            g = tl.load(ptr_g, mask=tok_valid[None, :] & h_mask[:, None], other=0.0).to(tl.float32)

            lnw = tl.load(lnw_ptr + h_safe, mask=h_mask, other=0.0).to(tl.float32)
            lnb = tl.load(lnb_ptr + h_safe, mask=h_mask, other=0.0).to(tl.float32)

            val_h = (o - mean[None, :]) * rstd[None, :]
            val_h = val_h * lnw[:, None] + lnb[:, None]
            val_h = val_h * g

            val = tl.trans(val_h.to(tl.bfloat16))  # [BM, BH]

            w_ch = tl.load(
                wout_ptr + c_safe[:, None] * H + h_safe[None, :],
                mask=c_mask[:, None] & h_mask[None, :],
                other=0.0,
            )
            w = tl.trans(w_ch)  # [BH, BC]

            acc = tl.dot(val, w, acc=acc)

        ptr_out = out_ptr + tok[:, None] * C + c_safe[None, :]
        tl.store(ptr_out, acc, mask=tok_valid[:, None] & c_mask[None, :])


def custom_kernel(data):
    """
    data can be:
      (input, mask, weights, config)
    or:
      (input, weights, config)
    """
    if len(data) == 3:
        x, weights, config = data
        mask = None
    else:
        x, mask, weights, config = data

    x = x.contiguous()
    B, S, _, C = x.shape

    try:
        H = int(config.get("hidden_dim", config.get("c_z", weights["left_proj.weight"].shape[0])))
    except Exception:
        H = int(weights["left_proj.weight"].shape[0])

    eps = float(config.get("norm_eps", config.get("eps", 1e-5)))
    device = x.device

    has_mask = (mask is not None) and (not bool(config.get("nomask", False)))
    if has_mask:
        mask_t = mask.to(torch.float32).contiguous()
    else:
        mask_t = x  # dummy pointer; not read when HAS_MASK=False

    gamma = weights["norm.weight"].detach().float().contiguous()
    beta = weights["norm.bias"].detach().float().contiguous()

    Wl = weights["left_proj.weight"].detach().float().contiguous()
    Wlg = weights["left_gate.weight"].detach().float().contiguous()
    Wr = weights["right_proj.weight"].detach().float().contiguous()
    Wrg = weights["right_gate.weight"].detach().float().contiguous()
    Wgo = weights["out_gate.weight"].detach().float().contiguous()

    # Weight concatenation: [left_proj, left_gate, right_proj, right_gate, out_gate]
    W_cat = torch.stack([Wl, Wlg, Wr, Wrg, Wgo], dim=0).to(torch.bfloat16).contiguous()

    # Constants needed to recover beta @ W^T after using (x * gamma - mean) @ (W^T gamma).
    wbeta_cat = torch.stack(
        [
            Wl @ beta,
            Wlg @ beta,
            Wr @ beta,
            Wrg @ beta,
            Wgo @ beta,
        ],
        dim=0,
    ).float().contiguous()

    W_out = weights["to_out.weight"].detach().float().to(torch.bfloat16).contiguous()
    lnw = weights["to_out_norm.weight"].detach().float().contiguous()
    lnb = weights["to_out_norm.bias"].detach().float().contiguous()

    SS = S * S
    M = B * SS
    HSS = H * SS
    HC = H * C

    left_h = torch.empty((B, H, S, S), dtype=torch.bfloat16, device=device)
    right_h = torch.empty_like(left_h)
    gate_h = torch.empty_like(left_h)

    BH = 128 if H >= 128 else max(16, triton.next_power_of_2(H))

    if C == 128:
        BM = 64
        grid = (triton.cdiv(M, BM),)
        _pre_trimul_c128_kernel[grid](
            x,
            mask_t,
            left_h,
            right_h,
            gate_h,
            W_cat,
            wbeta_cat,
            gamma,
            M,
            H,
            SS,
            HSS,
            HC,
            eps,
            HAS_MASK=has_mask,
            BM=BM,
            BH=BH,
        )
    else:
        BM = 16
        BC = 64 if C >= 64 else max(16, triton.next_power_of_2(C))
        grid = (triton.cdiv(M, BM),)
        _pre_trimul_generic_kernel[grid](
            x,
            mask_t,
            left_h,
            right_h,
            gate_h,
            W_cat,
            wbeta_cat,
            gamma,
            M,
            C,
            H,
            SS,
            HSS,
            HC,
            eps,
            HAS_MASK=has_mask,
            BM=BM,
            BH=BH,
            BC=BC,
        )

    # Huge N^3*H contraction as a batched GEMM over H.
    if hasattr(torch.backends.cuda.matmul, "allow_bf16_reduced_precision_reduction"):
        torch.backends.cuda.matmul.allow_bf16_reduced_precision_reduction = True

    out_h = torch.matmul(left_h, right_h.transpose(-1, -2)).contiguous()
    del left_h, right_h

    out = torch.empty((B, S, S, C), dtype=torch.float32, device=device)

    BMF = 32
    BHF = BH
    BCF = 128 if C >= 128 else max(16, triton.next_power_of_2(C))

    grid_f = (triton.cdiv(M, BMF),)
    _final_trimul_kernel[grid_f](
        out_h,
        gate_h,
        W_out,
        lnw,
        lnb,
        out,
        M,
        C,
        H,
        SS,
        HSS,
        eps,
        BM=BMF,
        BH=BHF,
        BC=BCF,
    )

    return out
```
