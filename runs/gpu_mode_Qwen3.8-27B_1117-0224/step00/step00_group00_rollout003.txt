def _ln_transpose_kernel(O, G, W, Bias, Y, NN, H, B, MB: tl.constexpr, HB: tl.constexpr, EPS: tl.constexpr):

  mb = pid % num_mb
  b = pid // num_mb
  offs_m = mb*MB + tl.arange(0,MB)
  m_mask = offs_m < NN
  offs_h = tl.arange(0,HB)
  h_mask = offs_h < H
  o_ptrs = O + (b*H + offs_h[:,None]) * NN + offs_m[None,:]
  x = tl.load(o_ptrs, mask=h_mask[:,None] & m_mask[None,:], other=0.0)
  sum_x = tl.sum(x, axis=0)
  mean = sum_x / H
  diff = x - mean[None,:]
  diff = tl.where(h_mask[None,:], diff, 0.0)
  var = tl.sum(diff*diff, axis=0) / H
  rstd = 1.0 / tl.sqrt(var + EPS)
  w = tl.load(W + offs_h, mask=h_mask, other=0.0).to(tl.float32)
  bias = tl.load(Bias + offs_h, mask=h_mask, other=0.0).to(tl.float32)
  y = diff * rstd[None,:]

  y_t = tl.trans(y)
  g_ptrs = G + (b*NN + offs_m[:,None]) * H + offs_h[None,:]
  g = tl.load(g_ptrs, mask=m_mask[:,None]&h_mask[None,:], other=0.0).to(tl.float32)
  res = (y_t * g).to(tl.bfloat16)

def _gate_transpose_kernel(P_ptr, L_ptr, R_ptr, OG_ptr, MASK_ptr,
                           NN, H, FIVE_H,
                           MB: tl.constexpr, HT: tl.constexpr,
                           MASKED: tl.constexpr):
    pid = tl.program_id(0)
    num_hb = tl.cdiv(H, HT)
    num_mb = tl.cdiv(NN, MB)
    hb = pid % num_hb
    tmp = pid // num_hb
    mb = tmp % num_mb
    b = tmp // num_mb

    offs_m = mb * MB + tl.arange(0, MB)
    m_mask = offs_m < NN
    flat_m = b * NN + offs_m
    h_off = hb * HT + tl.arange(0, HT)
    h_mask = h_off < H

    if MASKED:
        msk = tl.load(MASK_ptr + flat_m, mask=m_mask, other=0.0).to(tl.float32)
    else:
        msk = tl.zeros((MB,), dtype=tl.float32) + 1.0

    full_mask = m_mask[:, None] & h_mask[None, :]

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


def _next_power_of_2(x: int) -> int:
    x = max(1, int(x))
    p = 1
    while p < x:
        p <<= 1
    return p


@triton.jit
def _layer_norm_rows_kernel(
    X, Y, W, Bias,
    M, C,
    MB: tl.constexpr,
    CB: tl.constexpr,
    EPS: tl.constexpr,
):
    """LayerNorm over the last row, written as bf16."""
    offs_m = tl.program_id(0) * MB + tl.arange(0, MB)
    offs_c = tl.arange(0, CB)

    m_mask = offs_m < M
    c_mask = offs_c < C
    full_mask = m_mask[:, None] & c_mask[None, :]

    x_ptrs = X + offs_m[:, None] * C + offs_c[None, :]
    x = tl.load(x_ptrs, mask=full_mask, other=0.0).to(tl.float32)

    mean = tl.sum(x, axis=1) / C
    diff = x - mean[:, None]
    diff = tl.where(c_mask[None, :], diff, 0.0)
    var = tl.sum(diff * diff, axis=1) / C
    rstd = 1.0 / tl.sqrt(var + EPS)

    w = tl.load(W + offs_c, mask=c_mask, other=0.0).to(tl.float32)
    b = tl.load(Bias + offs_c, mask=c_mask, other=0.0).to(tl.float32)

    y = diff * rstd[:, None] * w[None, :] + b[None, :]
    tl.store(Y + offs_m[:, None] * C + offs_c[None, :], y.to(tl.bfloat16), mask=full_mask)


@triton.jit
def _gate_transpose_kernel(
    P, L, R, OG, MASK,
    NN, H, FIVE_H,
    MB: tl.constexpr,
    HT: tl.constexpr,
    MASKED: tl.constexpr,
):
    """
    Input P: [B, NN, 5H], layout:
        0:H       left projection
        H:2H      right projection
        2H:3H     left gate projection
        3H:4H     right gate projection
        4H:5H     output gate projection

    Outputs:
        L, R: [B, H, NN] bf16
        OG:   [B*NN, H] bf16
    """
    pid = tl.program_id(0)

    num_hb = tl.cdiv(H, HT)
    num_mb = tl.cdiv(NN, MB)

    hb = pid % num_hb
    tmp = pid // num_hb
    mb = tmp % num_mb
    b = tmp // num_mb

    offs_m = mb * MB + tl.arange(0, MB)
    m_mask = offs_m < NN
    flat_m = b * NN + offs_m

    h_off = hb * HT + tl.arange(0, HT)
    h_mask = h_off < H
    full_mask = m_mask[:, None] & h_mask[None, :]

    if MASKED:
        msk = tl.load(MASK + flat_m, mask=m_mask, other=0.0).to(tl.float32)
    else:
        msk = tl.zeros((MB,), dtype=tl.float32) + 1.0

    base = P + flat_m[:, None] * FIVE_H + h_off[None, :]

    # left
    left = tl.load(base, mask=full_mask, other=0.0).to(tl.float32)
    left_gate = tl.load(base + 2 * H, mask=full_mask, other=0.0).to(tl.float32)
    l = (left * tl.sigmoid(left_gate) * msk[None, :]).to(tl.bfloat16)
    lt = tl.trans(l)
    l_ptrs = L + (b * H + h_off[:, None]) * NN + offs_m[None, :]
    tl.store(l_ptrs, lt, mask=h_mask[:, None] & m_mask[None, :])

    # right
    right = tl.load(base + H, mask=full_mask, other=0.0).to(tl.float32)
    right_gate = tl.load(base + 3 * H, mask=full_mask, other=0.0).to(tl.float32)
    r = (right * tl.sigmoid(right_gate) * msk[None, :]).to(tl.bfloat16)
    rt = tl.trans(r)
    r_ptrs = R + (b * H + h_off[:, None]) * NN + offs_m[None, :]
    tl.store(r_ptrs, rt, mask=h_mask[:, None] & m_mask[None, :])

    # output gate
    og = tl.load(base + 4 * H, mask=full_mask, other=0.0).to(tl.float32)
    ogate = tl.sigmoid(og).to(tl.bfloat16)
    og_ptrs = OG + flat_m[:, None] * H + h_off[None, :]
    tl.store(og_ptrs, ogate, mask=full_mask)


@triton.jit
def _tri_bmm_kernel(
    L, R, O,
    B, N, H, NN,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    EVEN: tl.constexpr,
):
    """
    For every (b, h):
        O[b, h, i, j] = sum_k L[b, h, i, k] * R[b, h, j, k]
    """
    pid = tl.program_id(0)

    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    h = pid % H
    tmp = pid // H
    j = tmp % num_n
    tmp = tmp // num_n
    i = tmp % num_m
    b = tmp // num_m

    p = b * H + h
    base = p * NN

    offs_i = i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    a_ptrs = L + base + offs_i[:, None] * N + offs_k[None, :]
    b_ptrs = R + base + offs_j[:, None] * N + offs_k[None, :]

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    if EVEN:
        for _ in range(0, N, BLOCK_K):
            a = tl.load(a_ptrs)
            b = tl.load(b_ptrs)
            acc += tl.dot(a, tl.trans(b))
            a_ptrs += BLOCK_K
            b_ptrs += BLOCK_K
    else:
        i_mask = offs_i < N
        j_mask = offs_j < N
        for k in range(0, N, BLOCK_K):
            k_mask = (k + offs_k) < N
            a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
            b = tl.load(b_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0)
            acc += tl.dot(a, tl.trans(b))
            a_ptrs += BLOCK_K
            b_ptrs += BLOCK_K

    o_ptrs = O + base + offs_i[:, None] * N + offs_j[None, :]
    if EVEN:
        tl.store(o_ptrs, acc.to(O.dtype.element_ty))
    else:
        tl.store(o_ptrs, acc.to(O.dtype.element_ty), mask=i_mask[:, None] & j_mask[None, :])


@triton.jit
def _ln_transpose_kernel(
    O, G, W, Bias, Y,
    NN, H,
    MB: tl.constexpr,
    HB: tl.constexpr,
    EPS: tl.constexpr,
):
    """
    O: [B, H, NN] bf16
    G: [B*NN, H] bf16
    Y: [B*NN, H] bf16

    For each batch and position m=i*NN+j:
        Y[m, h] = LN_h(O[:, m]) * G[m, h]
    """
    pid = tl.program_id(0)
    num_mb = tl.cdiv(NN, MB)

    mb = pid % num_mb
    b = pid // num_mb

    offs_m = mb * MB + tl.arange(0, MB)
    m_mask = offs_m < NN

    offs_h = tl.arange(0, HB)
    h_mask = offs_h < H

    # Load [H, MB] tile; memory is coalesced along m.
    o_ptrs = O + (b * H + offs_h[:, None]) * NN + offs_m[None, :]
    x = tl.load(o_ptrs, mask=h_mask[:, None] & m_mask[None, :], other=0.0).to(tl.float32)

    mean = tl.sum(x, axis=0) / H
    diff = x - mean[None, :]
    diff = tl.where(h_mask[None, :], diff, 0.0)
    var = tl.sum(diff * diff, axis=0) / H
    rstd = 1.0 / tl.sqrt(var + EPS)

    w = tl.load(W + offs_h, mask=h_mask, other=0.0).to(tl.float32)
    bias = tl.load(Bias + offs_h, mask=h_mask, other=0.0).to(tl.float32)

    y = diff * rstd[None, :] * w[:, None] + bias[:, None]
    yt = tl.trans(y)  # [MB, HB]

    g_ptrs = G + (b * NN + offs_m[:, None]) * H + offs_h[None, :]
    g = tl.load(g_ptrs, mask=m_mask[:, None] & h_mask[None, :], other=0.0).to(tl.float32)

    y_out = (yt * g).to(tl.bfloat16)
    y_ptrs = Y + (b * NN + offs_m[:, None]) * H + offs_h[None, :]
    tl.store(y_ptrs, y_out, mask=m_mask[:, None] & h_mask[None, :])


def custom_kernel(data):
    """
    Fused Triton implementation of AlphaFold3 outgoing TriMul forward pass.
    """
    input_tensor, mask, weights, config = data

    B, N, _, C = input_tensor.shape
    H = int(config.get("hidden_dim", weights["left_proj.weight"].shape[0]))
    NN = N * N
    M = B * NN
    device = input_tensor.device

    input_tensor = input_tensor.contiguous()
    x = input_tensor.view(M, C)

    norm_w = weights["norm.weight"].contiguous()
    norm_b = weights["norm.bias"].contiguous()

    # First LayerNorm: read input in its dtype, compute in fp32, write bf16.
    x_bf16 = torch.empty((M, C), dtype=torch.bfloat16, device=device)
    cb0 = _next_power_of_2(C)
    if cb0 <= 128:
        mb0, warps0 = 64, 8
    elif cb0 <= 512:
        mb0, warps0 = 16, 8
    else:
        mb0, warps0 = 8, 8
    _layer_norm_rows_kernel[(triton.cdiv(M, mb0),)](
        x, x_bf16, norm_w, norm_b, M, C,
        MB=mb0, CB=cb0, EPS=1e-5,
        num_warps=warps0,
    )
    del x, input_tensor

    # Concatenated projection: [5H, C]
    W_all = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.bfloat16, copy=False).contiguous()

    P = F.linear(x_bf16, W_all)  # [M, 5H]
    del x_bf16

    Lb = torch.empty((B, H, NN), dtype=torch.bfloat16, device=device)
    Rb = torch.empty((B, H, NN), dtype=torch.bfloat16, device=device)
    OG = torch.empty((M, H), dtype=torch.bfloat16, device=device)

    use_mask = mask is not None and not config.get("nomask", False)
    if use_mask:
        mask_flat = mask.reshape(-1).contiguous()
    else:
        mask_flat = P  # dummy, not loaded when MASKED=False

    ht = 128 if (H >= 128 and H % 128 == 0) else _next_power_of_2(H)
    if ht <= 128:
        mb_g = 32
    else:
        mb_g = 16

    _gate_transpose_kernel[(B * triton.cdiv(NN, mb_g) * triton.cdiv(H, ht),)](
        P, Lb, Rb, OG, mask_flat, NN, H, 5 * H,
        MB=mb_g, HT=ht, MASKED=use_mask,
        num_warps=8,
        num_stages=2,
    )
    del P

    # Batched triangle GEMM. Output is bf16 to save bandwidth; LN below upcasts.
    if N <= 512:
        bm = bn = 64
        bk = 64
        gemm_warps = 4
    else:
        bm = bn = 128
        bk = 64
        gemm_warps = 8

    even_gemm = (N % bm == 0) and (N % bn == 0) and (N % bk == 0)
    Ob = torch.empty((B, H, NN), dtype=torch.bfloat16, device=device)

    grid_gemm = (B * triton.cdiv(N, bm) * triton.cdiv(N, bn) * H,)
    _tri_bmm_kernel[grid_gemm](
        Lb, Rb, Ob, B, N, H, NN,
        BLOCK_M=bm, BLOCK_N=bn, BLOCK_K=bk, EVEN=even_gemm,
        num_warps=gemm_warps,
        num_stages=4,
    )
    del Lb, Rb

    # Transpose + LayerNorm over H + output gate.
    hb_ln = _next_power_of_2(H)
    if hb_ln <= 128:
        mb_ln = 64
    elif hb_ln <= 256:
        mb_ln = 32
    elif hb_ln <= 512:
        mb_ln = 16
    else:
        mb_ln = 8

    Y = torch.empty((M, H), dtype=torch.bfloat16, device=device)
    _ln_transpose_kernel[(B * triton.cdiv(NN, mb_ln),)](
        Ob, OG,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        Y, NN, H,
        MB=mb_ln, HB=hb_ln, EPS=1e-5,
        num_warps=8,
    )
    del Ob, OG

    W_out = weights["to_out.weight"].to(torch.bfloat16, copy=False).contiguous()
    out_flat = F.linear(Y, W_out)
    del Y

    return out_flat.view(B, N, N, C).to(torch.float32)
