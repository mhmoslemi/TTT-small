import torch
import triton
import triton.language as tl


@triton.jit
def _stable_sigmoid(x):
    z = tl.exp(-tl.abs(x))
    return tl.where(x >= 0.0, 1.0 / (1.0 + z), z / (1.0 + z))


@triton.jit
def trimul_init_kernel(
    x_ptr,
    mask_ptr,
    norm_w_ptr,
    norm_b_ptr,
    w_left_ptr,
    w_left_gate_ptr,
    w_right_ptr,
    w_right_gate_ptr,
    w_out_gate_ptr,
    left_ptr,
    right_ptr,
    out_gate_ptr,
    N2,
    C,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
    HAS_MASK: tl.constexpr,
    EPS: tl.constexpr,
):
    pid = tl.program_id(0)
    num_m = tl.cdiv(N2, BLOCK_M)
    b = pid // num_m
    m0 = (pid % num_m) * BLOCK_M

    ms = m0 + tl.arange(0, BLOCK_M)
    m_mask = ms < N2

    base_x = x_ptr + b * N2 * C

    # LayerNorm statistics over the channel dimension.
    sum = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sumsq = tl.zeros((BLOCK_M,), dtype=tl.float32)
    for c0 in range(0, C, BLOCK_C):
        cs = c0 + tl.arange(0, BLOCK_C)
        c_mask = cs < C
        ptrs = base_x + ms[:, None] * C + cs[None, :]
        x = tl.load(ptrs, mask=m_mask[:, None] & c_mask[None, :], other=0.0).to(tl.float32)
        sum += tl.sum(x, axis=1)
        sumsq += tl.sum(x * x, axis=1)

    inv_C = 1.0 / C
    mean = sum * inv_C
    var = sumsq * inv_C - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + EPS)

    if HAS_MASK:
        maskv = tl.load(mask_ptr + b * N2 + ms, mask=m_mask, other=0.0).to(tl.float32)
    else:
        maskv = tl.full((BLOCK_M,), 1.0, dtype=tl.float32)

    hs = tl.arange(0, H)
    store_mask = (hs[:, None] < H) & m_mask[None, :]
    out_batch_off = b * H * N2

    # Left projection + left gate + mask.
    acc_l = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    acc_lg = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    for c0 in range(0, C, BLOCK_C):
        cs = c0 + tl.arange(0, BLOCK_C)
        c_mask = cs < C

        ptrs = base_x + ms[:, None] * C + cs[None, :]
        x = tl.load(ptrs, mask=m_mask[:, None] & c_mask[None, :], other=0.0).to(tl.float32)

        nw = tl.load(norm_w_ptr + cs, mask=c_mask, other=0.0).to(tl.float32)
        nb = tl.load(norm_b_ptr + cs, mask=c_mask, other=0.0).to(tl.float32)
        xn = (x - mean[:, None]) * (rstd[:, None] * nw[None, :]) + nb[None, :]
        xn_bf = xn.to(tl.bfloat16)

        wl = tl.load(
            w_left_ptr + hs[:, None] * C + cs[None, :],
            mask=c_mask[None, :],
            other=0.0,
        )
        acc_l = tl.dot(xn_bf, tl.trans(wl), acc_l)

        wlg = tl.load(
            w_left_gate_ptr + hs[:, None] * C + cs[None, :],
            mask=c_mask[None, :],
            other=0.0,
        )
        acc_lg = tl.dot(xn_bf, tl.trans(wlg), acc_lg)

    left = acc_l * _stable_sigmoid(acc_lg) * maskv[:, None]
    tl.store(
        left_ptr + out_batch_off + hs[:, None] * N2 + ms[None, :],
        tl.trans(left).to(tl.bfloat16),
        mask=store_mask,
    )

    # Right projection + right gate + mask.
    acc_r = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    acc_rg = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    for c0 in range(0, C, BLOCK_C):
        cs = c0 + tl.arange(0, BLOCK_C)
        c_mask = cs < C

        ptrs = base_x + ms[:, None] * C + cs[None, :]
        x = tl.load(ptrs, mask=m_mask[:, None] & c_mask[None, :], other=0.0).to(tl.float32)

        nw = tl.load(norm_w_ptr + cs, mask=c_mask, other=0.0).to(tl.float32)
        nb = tl.load(norm_b_ptr + cs, mask=c_mask, other=0.0).to(tl.float32)
        xn = (x - mean[:, None]) * (rstd[:, None] * nw[None, :]) + nb[None, :]
        xn_bf = xn.to(tl.bfloat16)

        wr = tl.load(
            w_right_ptr + hs[:, None] * C + cs[None, :],
            mask=c_mask[None, :],
            other=0.0,
        )
        acc_r = tl.dot(xn_bf, tl.trans(wr), acc_r)

        wrg = tl.load(
            w_right_gate_ptr + hs[:, None] * C + cs[None, :],
            mask=c_mask[None, :],
            other=0.0,
        )
        acc_rg = tl.dot(xn_bf, tl.trans(wrg), acc_rg)

    right = acc_r * _stable_sigmoid(acc_rg) * maskv[:, None]
    tl.store(
        right_ptr + out_batch_off + hs[:, None] * N2 + ms[None, :],
        tl.trans(right).to(tl.bfloat16),
        mask=store_mask,
    )

    # Output gate projection.
    acc_og = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    for c0 in range(0, C, BLOCK_C):
        cs = c0 + tl.arange(0, BLOCK_C)
        c_mask = cs < C

        ptrs = base_x + ms[:, None] * C + cs[None, :]
        x = tl.load(ptrs, mask=m_mask[:, None] & c_mask[None, :], other=0.0).to(tl.float32)

        nw = tl.load(norm_w_ptr + cs, mask=c_mask, other=0.0).to(tl.float32)
        nb = tl.load(norm_b_ptr + cs, mask=c_mask, other=0.0).to(tl.float32)
        xn = (x - mean[:, None]) * (rstd[:, None] * nw[None, :]) + nb[None, :]
        xn_bf = xn.to(tl.bfloat16)

        wo = tl.load(
            w_out_gate_ptr + hs[:, None] * C + cs[None, :],
            mask=c_mask[None, :],
            other=0.0,
        )
        acc_og = tl.dot(xn_bf, tl.trans(wo), acc_og)

    og = _stable_sigmoid(acc_og)
    tl.store(
        out_gate_ptr + out_batch_off + hs[:, None] * N2 + ms[None, :],
        tl.trans(og).to(tl.bfloat16),
        mask=store_mask,
    )


@triton.jit
def trimul_final_kernel(
    einsum_ptr,
    gate_ptr,
    norm_w_ptr,
    norm_b_ptr,
    w_out_ptr,
    out_ptr,
    N2,
    C,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
    EPS: tl.constexpr,
):
    pid = tl.program_id(0)
    num_m = tl.cdiv(N2, BLOCK_M)
    b = pid // num_m
    m0 = (pid % num_m) * BLOCK_M

    ms = m0 + tl.arange(0, BLOCK_M)
    m_mask = ms < N2

    hs = tl.arange(0, H)
    load_mask = (hs[:, None] < H) & m_mask[None, :]

    # Load einsum results in hidden-first layout: [H, M].
    e = tl.load(
        einsum_ptr + b * H * N2 + hs[:, None] * N2 + ms[None, :],
        mask=load_mask,
        other=0.0,
    ).to(tl.float32)

    # LayerNorm over hidden dimension H.
    inv_H = 1.0 / H
    mean = tl.sum(e, axis=0) * inv_H
    var = tl.sum(e * e, axis=0) * inv_H - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + EPS)

    nw = tl.load(norm_w_ptr + hs).to(tl.float32)
    nb = tl.load(norm_b_ptr + hs).to(tl.float32)

    g = tl.load(
        gate_ptr + b * H * N2 + hs[:, None] * N2 + ms[None, :],
        mask=load_mask,
        other=0.0,
    ).to(tl.float32)

    normed = (e - mean[None, :]) * (rstd[None, :] * nw[:, None]) + nb[:, None]
    normed = normed * g
    normed_bf = normed.to(tl.bfloat16)

    # Project normalized hidden features back to C.
    for c0 in range(0, C, BLOCK_C):
        cs = c0 + tl.arange(0, BLOCK_C)
        c_mask = cs < C

        w = tl.load(
            w_out_ptr + cs[:, None] * H + hs[None, :],
            mask=c_mask[:, None],
            other=0.0,
        )

        acc = tl.zeros((BLOCK_C, BLOCK_M), dtype=tl.float32)
        acc = tl.dot(w, normed_bf, acc)

        out_ptrs = out_ptr + b * N2 * C + ms[:, None] * C + cs[None, :]
        out_mask = m_mask[:, None] & c_mask[None, :]
        tl.store(out_ptrs, tl.trans(acc).to(tl.float32), mask=out_mask)


def custom_kernel(data):
    """
    Fused TriMul outgoing forward.

    - A Triton kernel fuses LayerNorm, projection, gates, and masking into
      hidden-first bf16 left/right/out-gate tensors.
    - The N^3 triangular multiplication is evaluated with cuBLAS bmm.
    - A Triton kernel fuses the second LayerNorm, out-gating, and final linear.
    """
    input_tensor, mask_tensor, weights, config = data

    B, N, _, C = input_tensor.shape
    H = int(config["hidden_dim"])
    N2 = N * N
    device = input_tensor.device

    x = input_tensor.contiguous()

    has_mask = mask_tensor is not None
    if has_mask:
        mask_arg = mask_tensor.contiguous().to(torch.float32)
    else:
        mask_arg = torch.empty(1, device=device, dtype=torch.float32)

    norm_w = weights["norm.weight"].to(torch.float32).contiguous()
    norm_b = weights["norm.bias"].to(torch.float32).contiguous()

    w_left = weights["left_proj.weight"].to(torch.bfloat16).contiguous()
    w_right = weights["right_proj.weight"].to(torch.bfloat16).contiguous()
    w_left_gate = weights["left_gate.weight"].to(torch.bfloat16).contiguous()
    w_right_gate = weights["right_gate.weight"].to(torch.bfloat16).contiguous()
    w_out_gate = weights["out_gate.weight"].to(torch.bfloat16).contiguous()

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float32).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float32).contiguous()
    w_to_out = weights["to_out.weight"].to(torch.bfloat16).contiguous()

    left = torch.empty((B, H, N, N), device=device, dtype=torch.bfloat16)
    right = torch.empty((B, H, N, N), device=device, dtype=torch.bfloat16)
    out_gate = torch.empty((B, H, N, N), device=device, dtype=torch.bfloat16)

    BLOCK_M_INIT = 128
    BLOCK_C_INIT = 64
    grid_init = (B * triton.cdiv(N2, BLOCK_M_INIT),)

    trimul_init_kernel[grid_init](
        x,
        mask_arg,
        norm_w,
        norm_b,
        w_left,
        w_left_gate,
        w_right,
        w_right_gate,
        w_out_gate,
        left,
        right,
        out_gate,
        N2,
        C,
        H=H,
        BLOCK_M=BLOCK_M_INIT,
        BLOCK_C=BLOCK_C_INIT,
        HAS_MASK=has_mask,
        EPS=1e-5,
        num_warps=8,
        num_stages=2,
    )

    BH = B * H
    left_3d = left.view(BH, N, N)
    right_3d = right.view(BH, N, N)
    einsum_out = torch.empty((BH, N, N), device=device, dtype=torch.bfloat16)
    torch.bmm(left_3d, right_3d.transpose(1, 2), out=einsum_out)

    del left, right, left_3d, right_3d

    out = torch.empty((B, N, N, C), device=device, dtype=torch.float32)

    BLOCK_M_FINAL = 64
    BLOCK_C_FINAL = 128
    grid_final = (B * triton.cdiv(N2, BLOCK_M_FINAL),)

    trimul_final_kernel[grid_final](
        einsum_out,
        out_gate,
        to_out_norm_w,
        to_out_norm_b,
        w_to_out,
        out,
        N2,
        C,
        H=H,
        BLOCK_M=BLOCK_M_FINAL,
        BLOCK_C=BLOCK_C_FINAL,
        EPS=1e-5,
        num_warps=8,
        num_stages=2,
    )

    return out
