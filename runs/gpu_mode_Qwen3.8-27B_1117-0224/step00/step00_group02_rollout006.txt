```python
"""
Fused TriMul (outgoing) forward, AlphaFold2-style triple multiplicative update.

Pipeline:
  K1: LayerNorm(C) + 5 linears (left/right + 3 gates) + mask*gate fusion.
      Emits L,R directly in the GEMM-friendly [B, N, H, N] layout plus the
      out-gates [B, N, N, H] (fp16 data, fp32 stats).
  K2: The H=... batch of N x N GEMMs:  out[b,i,j,d] = sum_k L[b,i,d,k]*R[b,j,d,k]
      (one program per (b, d-channel, i-tile, j-tile), fp16 tensor-core dots,
      fp32 accumulate).
  K3: Final LayerNorm(H) * out_gate, then the to_out linear (fp16 matmul),
      writing the fp32 result.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _k_norm_proj(
    x_p, m_p, l_p, r_p, og_p,
    wl_p, wlg_p, wr_p, wrg_p, wo_p,
    N, C, H,
    BIR: tl.constexpr, NK: tl.constexpr, BT: tl.constexpr,
    KCB: tl.constexpr, BH: tl.constexpr,
    HAS_MASK: tl.constexpr, EPS: tl.constexpr,
):
    pr = tl.program_id(0)
    n_k = tl.cdiv(N, NK)
    n_i = tl.cdiv(N, BIR)
    pk = pr % n_k
    pi = (pr // n_k) % n_i
    b = pr // (n_k * n_i)

    i0 = pi * BIR
    k0 = pk * NK
    i = i0 + tl.arange(0, BIR)
    k = k0 + tl.arange(0, NK)
    im = i < N
    km = k < N
    im2 = im[:, None] & km[None, :]

    base = b * N * N
    xi = (i[:, None] * N + k[None, :, None])  # (BIR, NK, 1)
    cc = tl.arange(0, KCB)

    s1 = tl.zeros((BIR, NK), dtype=tl.float32)
    s2 = tl.zeros((BIR, NK), dtype=tl.float32)
    for cb in range(0, C, KCB):
        cm = (cb + cc) < C
        lm = im2[:, :, None] & cm[None, None, :]
        xc = tl.load(x_p + base * C + xi * C + (cb + cc)[None, None, :],
                     mask=lm, other=0.0)
        s1 += tl.sum(xc, 2)
        s2 += tl.sum(xc * xc, 2)

    mean = s1 / C
    var = s2 / C - mean * mean
    rstd = 1.0 / tl.sqrt(var + EPS)

    aL = tl.zeros((BT, BH), dtype=tl.float32)
    aLG = tl.zeros((BT, BH), dtype=tl.float32)
    aR = tl.zeros((BT, BH), dtype=tl.float32)
    aRG = tl.zeros((BT, BH), dtype=tl.float32)
    aOG = tl.zeros((BT, BH), dtype=tl.float32)

    for cb in range(0, C, KCB):
        cm = (cb + cc) < C
        lm = im2[:, :, None] & cm[None, None, :]
        xc = tl.load(x_p + base * C + xi * C + (cb + cc)[None, None, :],
                     mask=lm, other=0.0)
        xh = tl.reshape((xc - mean[:, :, None]) * rstd[:, :, None], (BT, KCB)).to(tl.float16)
        wm = cm[:, None]
        wl = tl.load(wl_p + (cb + cc)[:, None] * C + tl.arange(0, BH)[None, :], mask=wm, other=0.0)
        wlg = tl.load(wlg_p + (cb + cc)[:, None] * C + tl.arange(0, BH)[None, :], mask=wm, other=0.0)
        wr = tl.load(wr_p + (cb + cc)[:, None] * C + tl.arange(0, BH)[None, :], mask=wm, other=0.0)
        wrg = tl.load(wrg_p + (cb + cc)[:, None] * C + tl.arange(0, BH)[None, :], mask=wm, other=0.0)
        wo = tl.load(wo_p + (cb + cc)[:, None] * C + tl.arange(0, BH)[None, :], mask=wm, other=0.0)
        aL += tl.dot(xh, wl)
        aLG += tl.dot(xh, wlg)
        aR += tl.dot(xh, wr)
        aRG += tl.dot(xh, wrg)
        aOG += tl.dot(xh, wo)

    lg = tl.sigmoid(aLG)
    rg = tl.sigmoid(aRG)
    og = tl.sigmoid(aOG)

    if HAS_MASK:
        mt = tl.load(m_p + base + i[:, None] * N + k[None, :], mask=im2, other=0.0).to(tl.float32)
        mtb = tl.reshape(mt, (BT,))
        lf = (aL * lg) * mtb[:, None]
        rf = (aR * rg) * mtb[:, None]
    else:
        lf = aL * lg
        rf = aR * rg

    oh = tl.arange(0, BH)
    hm = oh < H
    kb = k0 + tl.arange(0, NK)
    kbm = kb < N

    l3 = tl.trans(tl.reshape(lf.to(tl.float16), (BIR, NK, BH)), 0, 2, 1)  # (BIR, BH, NK)
    lptr = l_p + b * N * N * H + i[:, None, None] * (H * N) + oh[None, :, None] * N + kb[None, None, :]
    tl.store(lptr, l3, mask=im[:, None, None] & hm[None, :, None] & kbm[None, None, :])

    r3 = tl.trans(tl.reshape(rf.to(tl.float16), (BIR, NK, BH)), 0, 2, 1)
    rptr = r_p + b * N * N * H + i[:, None, None] * (H * N) + oh[None, :, None] * N + kb[None, None, :]
    tl.store(rptr, r3, mask=im[:, None, None] & hm[None, :, None] & kbm[None, None, :])

    og3 = tl.reshape(og.to(tl.float16), (BIR, NK, BH))
    oport = og_p + b * N * N * H + i[:, None, None] * (N * H) + kb[None, :, None] * H + oh[None, None, :]
    tl.store(oport, og3, mask=im[:, None, None] & kbm[None, :, None] & hm[None, None, :])


@triton.jit
def _k_einsum(
    l_p, r_p, o_p,
    N, H, NNH,
    BM: tl.constexpr, BN: tl.constexpr, BK: tl.constexpr, BH: tl.constexpr,
):
    pr = tl.program_id(0)
    ni = tl.cdiv(N, BM)
    nj = tl.cdiv(N, BN)
    r = pr % (ni * nj)
    jj = r % nj
    ii = r // nj
    t = pr // (ni * nj)
    d = t % H
    b = t // H

    i0 = ii * BM
    j0 = jj * BN
    oi = i0 + tl.arange(0, BM)
    oj = j0 + tl.arange(0, BN)
    im = oi < N
    jm = oj < N

    base = b * NNH
    ok = tl.arange(0, BK)
    baseL = l_p + base + oi[:, None] * (H * N) + d * N
    baseR = r_p + base + oj[None, :] * (H * N) + d * N

    acc = tl.zeros((BM, BN), dtype=tl.float32)
    for kb in range(0, N, BK):
        kt = kb + ok
        km = kt < N
        a = tl.load(baseL + kt[None, :], mask=im[:, None] & km[None, :], other=0.0)
        bt = tl.load(baseR + kt[:, None], mask=km[:, None] & jm[None, :], other=0.0)
        acc += tl.dot(a, bt)

    od = tl.arange(0, BH)
    optr = o_p + base + oi[:, None, None] * (N * H) + oj[None, :, None] * H + od[None, None, :]
    omask = im[:, None, None] & jm[None, :, None] & (od < H)[None, None, :]
    tl.store(optr, acc[:, :, None], mask=omask)


@triton.jit
def _k_final(
    out_p, og_p, w_p, res_p,
    T, H, C,
    BT: tl.constexpr, BH: tl.constexpr, CB: tl.constexpr,
    EPS: tl.constexpr,
):
    pr = tl.program_id(0)
    t0 = pr * BT
    ot = t0 + tl.arange(0, BT)
    om = ot < T
    oh = tl.arange(0, BH)
    hm = oh < H

    x = tl.load(out_p + ot[:, None] * H + oh[None, :], mask=om[:, None] & hm[None, :], other=0.0)
    og = tl.load(og_p + ot[:, None] * H + oh[None, :], mask=om[:, None] & hm[None, :], other=0.0).to(tl.float32)

    s1 = tl.sum(x, 1)
    s2 = tl.sum(x * x, 1)
    mean = s1 / H
    var = s2 / H - mean * mean
    rstd = 1.0 / tl.sqrt(var + EPS)

    on = ((x - mean[:, None]) * rstd[:, None] * og).to(tl.float16)

    cc = tl.arange(0, CB)
    for cb in range(0, C, CB):
        co = cb + cc
        cm = co < C
        w = tl.load(w_p + co[:, None] * H + oh[None, :], mask=cm[:, None] & hm[None, :], other=0.0)
        a = tl.dot(on, w.to(tl.float16))
        rptr = res_p + ot[:, None] * C + co[None, :]
        tl.store(rptr, a, mask=om[:, None] & cm[None, :])


def custom_kernel(data):
    x, mask, weights, config = data
    B, N, N2, C = x.shape
    H = int(config["hidden_dim"])

    has_mask = isinstance(mask, torch.Tensor) and mask.numel() > 0
    mp = mask if has_mask else x

    # fp16 copies of the linear weights (applied only in Triton dots)
    wl16 = weights["left_proj.weight"].to(torch.float16)
    wr16 = weights["right_proj.weight"].to(torch.float16)
    wlg16 = weights["left_gate.weight"].to(torch.float16)
    wrg16 = weights["right_gate.weight"].to(torch.float16)
    wo16 = weights["out_gate.weight"].to(torch.float16)
    wout16 = weights["to_out.weight"].to(torch.float16)

    dev = x.device
    Lt = torch.empty((B, N, H, N), device=dev, dtype=torch.float16)
    Rt = torch.empty((B, N, H, N), device=dev, dtype=torch.float16)
    Og = torch.empty((B, N, N, H), device=dev, dtype=torch.float16)
    Out = torch.empty((B, N, N, H), device=dev, dtype=torch.float32)
    Res = torch.empty((B, N, N, C), device=dev, dtype=torch.float32)

    BH = triton.next_power_of_2(H)

    # ---- Kernel 1: LayerNorm + projections + gating, writes transposed L/R
    BIR, NK, KCB = 8, 16, 128
    BT1 = BIR * NK
    n_i = triton.cdiv(N, BIR)
    n_k = triton.cdiv(N, NK)
    _k_norm_proj[(B * n_i * n_k,)](
        x, mp, Lt, Rt, Og, wl16, wlg16, wr16, wrg16, wo16,
        N, C, H,
        BIR=BIR, NK=NK, BT=BT1, KCB=KCB, BH=BH,
        HAS_MASK=has_mask, EPS=1e-5,
        num_warps=16, num_stages=2,
    )

    # ---- Kernel 2: batched N x N GEMMs over H channels
    BM = 64 if N <= 512 else 128
    BN = BM
    BK = 64
    ni = triton.cdiv(N, BM)
    nj = triton.cdiv(N, BN)
    _k_einsum[(B * H * ni * nj,)](
        Lt, Rt, Out, N, H, N * N * H,
        BM=BM, BN=BN, BK=BK, BH=BH,
        num_warps=8 if BM >= 128 else 4, num_stages=3,
    )

    # ---- Kernel 3: final LayerNorm * out_gate + to_out linear
    BT3 = 64
    T = B * N * N
    _k_final[(triton.cdiv(T, BT3),)](
        Out, Og, wout16, Res, T, H, C,
        BT=BT3, BH=BH, CB=128, EPS=1e-5,
        num_warps=4, num_stages=2,
    )

    return Res
```
