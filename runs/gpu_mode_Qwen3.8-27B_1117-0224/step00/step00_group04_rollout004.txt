```
import torch, triton, triton.language as tl

@triton.jit
def layer_norm_kernel(x_ptr, w_ptr, b_ptr, out_ptr, n_rows, C, eps, stride_row, BLOCK_C: tl.constexpr):
   row = tl.program_id(0)
   offs = tl.arange(0, BLOCK_C)
   mask = offs < C
   x = tl.load(x_ptr + row*stride_row + offs, mask=mask, other=0.0).to(tl.float32)
   mean = tl.sum(x, axis=0) / C
   xmean = tl.where(mask, x - mean, 0.0)
   var = tl.sum(xmean*xmean, axis=0) / C
   rstd = tl.rsqrt(var + eps)
   scale = tl.load(w_ptr+offs, mask=mask, other=1.0).to(tl.float32)*rstd
   bias = tl.load(b_ptr+offs,...)
   y = xmean*scale + bias
   tl.store(out_ptr+row*stride_row+offs, y, mask=mask)
```

```
def fused_ln_proj_kernel(

    out_l_ptr, out_r_ptr, out_g_ptr,

    BLOCK_H: tl.constexpr,
    BLOCK_M: tl.constexpr,
```

```python
import torch
import triton
import triton.language as tl


def _pow2(n: int) -> int:
    """Smallest power of two >= n (minimum 1)."""
    n = int(n)
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


@triton.jit
def _fused_ln_proj_kernel(
    x_ptr,
    norm_w_ptr,
    norm_b_ptr,
    wl_ptr,
    wlg_ptr,
    wr_ptr,
    wrg_ptr,
    wo_ptr,
    mask_ptr,
    ol_ptr,
    or_ptr,
    og_ptr,
    M,
    Nseq,
    H,
    C,
    eps,
    BLOCK_C: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    NUM_H_BLOCKS: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """
    Fused:
      - LayerNorm over C
      - left/right projections
      - left/right/out gates + sigmoid
      - mask multiplication for left/right
    Writes hidden-first tensors:
      L[b, h, i, k], R[b, h, j, k], G[b, h, i, j]
    where (i,k) / (j,k) are flattened as pos = (i,k).
    """
    pid = tl.program_id(0)
    pos = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    pos_mask = pos < M

    b = pos // Nseq
    s = pos - b * Nseq

    c = tl.arange(0, BLOCK_C)
    c_mask = c < C

    x_ptrs = x_ptr + pos[:, None] * C + c[None, :]
    x = tl.load(x_ptrs, mask=pos_mask[:, None] & c_mask[None, :], other=0.0).to(tl.float32)

    # LayerNorm statistics
    sum_x = tl.sum(tl.where(c_mask[None, :], x, 0.0), axis=1)
    mean = sum_x / C
    xm = tl.where(c_mask[None, :], x - mean[:, None], 0.0)
    var = tl.sum(xm * xm, axis=1) / C
    rstd = 1.0 / tl.sqrt(var + eps)

    gw = tl.load(norm_w_ptr + c, mask=c_mask, other=0.0).to(tl.float32)
    gb = tl.load(norm_b_ptr + c, mask=c_mask, other=0.0).to(tl.float32)

    yf = (xm * rstd[:, None]) * gw[None, :] + gb[None, :]
    yf = tl.where(pos_mask[:, None] & c_mask[None, :], yf, 0.0)
    y = yf.to(tl.bfloat16)

    if HAS_MASK:
        mv = tl.load(mask_ptr + pos, mask=pos_mask, other=1.0).to(tl.float32)
        mv = mv[:, None]

    for i in range(NUM_H_BLOCKS):
        h = i * BLOCK_H + tl.arange(0, BLOCK_H)
        h_mask = h < H
        w_mask = h_mask[:, None] & c_mask[None, :]

        # left projection + left gate
        w = tl.load(wl_ptr + h[:, None] * C + c[None, :], mask=w_mask, other=0.0)
        acc_l = tl.dot(y, tl.trans(w))

        w = tl.load(wlg_ptr + h[:, None] * C + c[None, :], mask=w_mask, other=0.0)
        acc_lg = tl.dot(y, tl.trans(w))
        gate_l = 1.0 / (1.0 + tl.exp(-acc_lg))

        val_l = acc_l * gate_l
        if HAS_MASK:
            val_l = val_l * mv
        v_l = tl.trans(val_l.to(tl.bfloat16))
        ptr_l = ol_ptr + (b[None, :] * H + h[:, None]) * Nseq + s[None, :]
        tl.store(ptr_l, v_l, mask=h_mask[:, None] & pos_mask[None, :])

        # right projection + right gate
        w = tl.load(wr_ptr + h[:, None] * C + c[None, :], mask=w_mask, other=0.0)
        acc_r = tl.dot(y, tl.trans(w))

        w = tl.load(wrg_ptr + h[:, None] * C + c[None, :], mask=w_mask, other=0.0)
        acc_rg = tl.dot(y, tl.trans(w))
        gate_r = 1.0 / (1.0 + tl.exp(-acc_rg))

        val_r = acc_r * gate_r
        if HAS_MASK:
            val_r = val_r * mv
        v_r = tl.trans(val_r.to(tl.bfloat16))
        ptr_r = or_ptr + (b[None, :] * H + h[:, None]) * Nseq + s[None, :]
        tl.store(ptr_r, v_r, mask=h_mask[:, None] & pos_mask[None, :])

        # output gate only
        w = tl.load(wo_ptr + h[:, None] * C + c[None, :], mask=w_mask, other=0.0)
        acc_o = tl.dot(y, tl.trans(w))
        val_o = 1.0 / (1.0 + tl.exp(-acc_o))
        v_o = tl.trans(val_o.to(tl.bfloat16))
        ptr_o = og_ptr + (b[None, :] * H + h[:, None]) * Nseq + s[None, :]
        tl.store(ptr_o, v_o, mask=h_mask[:, None] & pos_mask[None, :])


@triton.jit
def _final_ln_gate_linear_kernel(
    ot_ptr,
    og_ptr,
    gamma_ptr,
    beta_ptr,
    w_ptr,
    out_ptr,
    Nseq,
    H,
    C,
    eps,
    BLOCK_H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_OC: tl.constexpr,
    NUM_C_BLOCKS: tl.constexpr,
):
    """
    Fused final step:
      - LayerNorm over H, reading hidden-first O_t[b, h, pos]
      - multiply by out_gate
      - final linear projection to C
    """
    pid_s = tl.program_id(0)
    pid_b = tl.program_id(1)

    s = pid_s * BLOCK_M + tl.arange(0, BLOCK_M)
    s_mask = s < Nseq

    h = tl.arange(0, BLOCK_H)
    h_mask = h < H

    base = pid_b * H * Nseq
    ptr = base + h[:, None] * Nseq + s[None, :]
    load_mask = h_mask[:, None] & s_mask[None, :]

    o = tl.load(ot_ptr + ptr, mask=load_mask, other=0.0).to(tl.float32)

    # LayerNorm over H (axis 0 in this [H, M] tile)
    sum_o = tl.sum(o, axis=0)
    mean = sum_o / H
    xm = tl.where(load_mask, o - mean[None, :], 0.0)
    var = tl.sum(xm * xm, axis=0) / H
    rstd = 1.0 / tl.sqrt(var + eps)

    gamma = tl.load(gamma_ptr + h, mask=h_mask, other=0.0).to(tl.float32)
    beta = tl.load(beta_ptr + h, mask=h_mask, other=0.0).to(tl.float32)

    norm = (xm * rstd[None, :]) * gamma[:, None] + beta[:, None]
    norm = tl.where(load_mask, norm, 0.0)

    g = tl.load(og_ptr + ptr, mask=load_mask, other=0.0).to(tl.float32)
    y = (norm * g).to(tl.bfloat16)
    y = tl.trans(y)  # [BLOCK_M, BLOCK_H]

    pos = pid_b * Nseq + s

    for i in range(NUM_C_BLOCKS):
        c = i * BLOCK_OC + tl.arange(0, BLOCK_OC)
        c_mask = c < C

        # Weight is [C, H]; load as [C, H] then transpose to [H, C].
        wm = c_mask[:, None] & h_mask[None, :]
        w = tl.load(w_ptr + c[:, None] * H + h[None, :], mask=wm, other=0.0)
        wt = tl.trans(w)

        acc = tl.dot(y, wt)

        out_ptrs = out_ptr + pos[:, None] * C + c[None, :]
        tl.store(out_ptrs, acc.to(tl.float32), mask=s_mask[:, None] & c_mask[None, :])


def custom_kernel(data):
    """
    TriMul forward (AlphaFold3 outgoing style).

    Pipeline:
      1) Triton: LayerNorm + all initial projections/gates, writing hidden-first L/R/G.
      2) cuBLAS: batched GEMM for the triangle contraction.
      3) Triton: final LayerNorm + output gate + final linear.
    """
    if len(data) == 3:
        x_full, weights, config = data
        mask = None
    else:
        x_full, mask, weights, config = data

    Bsz = x_full.shape[0]
    N = x_full.shape[1]
    C = x_full.shape[-1]
    Nseq = N * N
    M = Bsz * Nseq

    H = weights["left_proj.weight"].shape[0]

    torch.backends.cuda.matmul.allow_tf32 = True

    x = x_full.contiguous()
    x_flat = x.view(M, C)

    has_mask = mask is not None
    if has_mask:
        mask_flat = mask.contiguous().view(-1).float()
    else:
        mask_flat = x_flat

    dev = x.device
    eps = float(config.get("eps", 1e-5)) if isinstance(config, dict) else 1e-5

    norm_w = weights["norm.weight"].float().contiguous()
    norm_b = weights["norm.bias"].float().contiguous()

    wl = weights["left_proj.weight"].to(torch.bfloat16).contiguous()
    wlg = weights["left_gate.weight"].to(torch.bfloat16).contiguous()
    wr = weights["right_proj.weight"].to(torch.bfloat16).contiguous()
    wrg = weights["right_gate.weight"].to(torch.bfloat16).contiguous()
    wo = weights["out_gate.weight"].to(torch.bfloat16).contiguous()

    out_norm_w = weights["to_out_norm.weight"].float().contiguous()
    out_norm_b = weights["to_out_norm.bias"].float().contiguous()
    out_w = weights["to_out.weight"].to(torch.bfloat16).contiguous()

    # ---------------- Stage 1: LN + projections + gates ----------------
    BLOCK_C = max(16, _pow2(C))

    if BLOCK_C <= 128:
        BLOCK_M1 = 64
        BLOCK_H1 = 128
        NUM_WARPS1 = 4
    elif BLOCK_C <= 512:
        BLOCK_M1 = 64
        BLOCK_H1 = 32
        NUM_WARPS1 = 8
    else:
        BLOCK_M1 = 32
        BLOCK_H1 = 32
        NUM_WARPS1 = 8

    if H < BLOCK_H1:
        BLOCK_H1 = max(16, _pow2(H))

    NUM_H1 = (H + BLOCK_H1 - 1) // BLOCK_H1

    L_t = torch.empty((Bsz * H, Nseq), dtype=torch.bfloat16, device=dev)
    R_t = torch.empty((Bsz * H, Nseq), dtype=torch.bfloat16, device=dev)
    OG_t = torch.empty((Bsz * H, Nseq), dtype=torch.bfloat16, device=dev)

    grid1 = (triton.cdiv(M, BLOCK_M1),)
    _fused_ln_proj_kernel[grid1](
        x_flat,
        norm_w,
        norm_b,
        wl,
        wlg,
        wr,
        wrg,
        wo,
        mask_flat,
        L_t,
        R_t,
        OG_t,
        M,
        Nseq,
        H,
        C,
        eps,
        BLOCK_C,
        BLOCK_H1,
        BLOCK_M1,
        NUM_H1,
        has_mask,
        num_warps=NUM_WARPS1,
    )

    del x_flat
    del mask_flat

    # ---------------- Stage 2: triangle contraction (batched GEMM) ----------------
    batch_h = Bsz * H
    L_batch = L_t.view(batch_h, N, N)
    R_batch = R_t.view(batch_h, N, N)
    O_t = torch.bmm(L_batch, R_batch.transpose(1, 2))

    del L_t
    del R_t

    # ---------------- Stage 3: final LN + gate + linear ----------------
    BLOCK_HF = max(16, _pow2(H))

    # Keep the [H, M] tile around 16k elements for register stability.
    raw_mf = 8192 // BLOCK_HF
    BLOCK_MF = 1
    while (BLOCK_MF * 2 <= raw_mf) and (BLOCK_MF < 128):
        BLOCK_MF *= 2
    BLOCK_MF = max(16, min(128, BLOCK_MF))

    if BLOCK_HF >= 512 or C > 256:
        BLOCK_OC = 64
    else:
        BLOCK_OC = 128
    BLOCK_OC = max(16, min(128, BLOCK_OC))

    NUM_OC = (C + BLOCK_OC - 1) // BLOCK_OC
    NUM_WARPSF = 8 if BLOCK_HF >= 512 else 4

    out = torch.empty((M, C), dtype=torch.float32, device=dev)

    gridF = (triton.cdiv(Nseq, BLOCK_MF), Bsz)
    _final_ln_gate_linear_kernel[gridF](
        O_t,
        OG_t,
        out_norm_w,
        out_norm_b,
        out_w,
        out,
        Nseq,
        H,
        C,
        eps,
        BLOCK_HF,
        BLOCK_MF,
        BLOCK_OC,
        NUM_OC,
        num_warps=NUM_WARPSF,
    )

    return out.view(Bsz, N, N, C)
```
