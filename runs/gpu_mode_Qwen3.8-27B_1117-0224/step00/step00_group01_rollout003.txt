```python
"""
Triton-accelerated TriMul forward.

Pipeline:
1) Fused Triton kernel:
   - LayerNorm over last input dimension.
   - All five projections (left/right, left/right gates, out gate).
   - Sigmoid gating + input mask.
   - Writes left/right/out-gate in head-major layout [B, H, N, N] for cuBLAS.
2) cuBLAS batched GEMM:
   - out_h[b, h, i, j] = sum_k left[b, h, i, k] * right[b, h, j, k]
3) Fused Triton final kernel:
   - LayerNorm over H.
   - Multiply by out gate.
   - Project back to C.
   - Store float32 output.

For non-fused input dimensionalities, the first stage falls back to PyTorch while
the final fused Triton kernel is still used when H == 128.
"""

import torch
import triton
import triton.language as tl

try:
    torch.backends.cuda.matmul.allow_bf16_reduced_precision_reduction = True
    torch.backends.cuda.matmul.allow_tf32 = True
except Exception:
    pass


F = torch.nn.functional


@triton.jit
def _store_h_major(
    out_ptr,
    res,
    b,
    ij,
    NN,
    H: tl.constexpr,
    BM: tl.constexpr,
):
    local = tl.arange(0, BM)
    m_mask = local < NN
    r = tl.trans(res).to(tl.bfloat16)
    offs_h = tl.arange(0, H)
    ptrs = out_ptr + b * (H * NN) + ij[None, :] + offs_h[:, None] * NN
    tl.store(ptrs, r, mask=m_mask[None, :])


@triton.jit
def _dot_proj_chunks(
    x0,
    x1,
    x2,
    w_ptr,
    C: tl.constexpr,
    H: tl.constexpr,
    BM: tl.constexpr,
):
    # x chunks are already bfloat16, shape [BM, 128].
    # w pointer points to a weight matrix [H, C] in row-major order.
    offs_k = tl.arange(0, 128)
    offs_h = tl.arange(0, H)

    if C <= 128:
        w0 = tl.load(w_ptr + offs_k[:, None] + offs_h[None, :] * C)
        return tl.dot(x0, w0)

    if C <= 256:
        w0 = tl.load(w_ptr + offs_k[:, None] + offs_h[None, :] * C)
        w1 = tl.load(w_ptr + (128 + offs_k)[:, None] + offs_h[None, :] * C)
        acc = tl.dot(x0, w0)
        acc += tl.dot(x1, w1)
        return acc

    if C <= 384:
        w0 = tl.load(w_ptr + offs_k[:, None] + offs_h[None, :] * C)
        w1 = tl.load(w_ptr + (128 + offs_k)[:, None] + offs_h[None, :] * C)
        w2 = tl.load(w_ptr + (256 + offs_k)[:, None] + offs_h[None, :] * C)
        acc = tl.dot(x0, w0)
        acc += tl.dot(x1, w1)
        acc += tl.dot(x2, w2)
        return acc

    # Should not be reached for supported fused paths.
    acc = tl.zeros((BM, H), dtype=tl.float32)
    return acc


@triton.jit
def trimul_first_kernel(
    x_ptr,
    mask_ptr,
    left_h_ptr,
    right_h_ptr,
    gate_h_ptr,

    norm_w_ptr,
    norm_b_ptr,
    w_left_ptr,
    w_right_ptr,
    w_lgate_ptr,
    w_rgate_ptr,
    w_ogate_ptr,

    NN,
    eps,
    C: tl.constexpr,
    H: tl.constexpr,
    BM: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_m = tl.program_id(0)
    b = tl.program_id(1)

    local = tl.arange(0, BM)
    ij = pid_m * BM + local
    m_mask = local < NN
    row = b * NN + ij

    offs_k = tl.arange(0, 128)
    base = x_ptr + row[:, None] * C

    x0 = tl.load(base + offs_k[None, :], mask=m_mask[:, None], other=0.0).to(tl.float32)

    if C <= 128:
        x1 = x0
        x2 = x0
    else:
        x1 = tl.load(base + 128 + offs_k[None, :], mask=m_mask[:, None], other=0.0).to(tl.float32)
        if C <= 256:
            x2 = x1
        else:
            x2 = tl.load(base + 256 + offs_k[None, :], mask=m_mask[:, None], other=0.0).to(tl.float32)

    s = tl.sum(x0, axis=1)
    if C > 128:
        s += tl.sum(x1, axis=1)
    if C > 256:
        s += tl.sum(x2, axis=1)
    mean = s / C

    x0c = x0 - mean[:, None]
    ss = tl.sum(x0c * x0c, axis=1)

    if C > 128:
        x1c = x1 - mean[:, None]
        ss += tl.sum(x1c * x1c, axis=1)
    else:
        x1c = x0c

    if C > 256:
        x2c = x2 - mean[:, None]
        ss += tl.sum(x2c * x2c, axis=1)
    else:
        x2c = x1c

    var = ss / C
    inv_std = 1.0 / tl.sqrt(var + eps)

    wn0 = tl.load(norm_w_ptr + offs_k).to(tl.float32)
    bn0 = tl.load(norm_b_ptr + offs_k).to(tl.float32)

    if C > 128:
        wn1 = tl.load(norm_w_ptr + 128 + offs_k).to(tl.float32)
        bn1 = tl.load(norm_b_ptr + 128 + offs_k).to(tl.float32)
    else:
        wn1 = wn0
        bn1 = bn0

    if C > 256:
        wn2 = tl.load(norm_w_ptr + 256 + offs_k).to(tl.float32)
        bn2 = tl.load(norm_b_ptr + 256 + offs_k).to(tl.float32)
    else:
        wn2 = wn1
        bn2 = bn1

    xn0 = (x0c * inv_std[:, None] * wn0[None, :] + bn0[None, :]).to(tl.bfloat16)
    xn1 = xn0
    xn2 = xn0

    if C > 128:
        xn1 = (x1c * inv_std[:, None] * wn1[None, :] + bn1[None, :]).to(tl.bfloat16)
        xn2 = xn1
    if C > 256:
        xn2 = (x2c * inv_std[:, None] * wn2[None, :] + bn2[None, :]).to(tl.bfloat16)

    if HAS_MASK:
        mk = tl.load(mask_ptr + row, mask=m_mask, other=0.0).to(tl.float32)
        mask_2d = mk[:, None]
    else:
        mask_2d = 1.0

    # left gate -> left
    lg = _dot_proj_chunks(xn0, xn1, xn2, w_lgate_ptr, C, H, BM)
    lv = _dot_proj_chunks(xn0, xn1, xn2, w_left_ptr, C, H, BM)
    left = lv * tl.sigmoid(lg) * mask_2d
    _store_h_major(left_h_ptr, left, b, ij, NN, H, BM)

    # right gate -> right
    rg = _dot_proj_chunks(xn0, xn1, xn2, w_rgate_ptr, C, H, BM)
    rv = _dot_proj_chunks(xn0, xn1, xn2, w_right_ptr, C, H, BM)
    right = rv * tl.sigmoid(rg) * mask_2d
    _store_h_major(right_h_ptr, right, b, ij, NN, H, BM)

    # out gate
    og = _dot_proj_chunks(xn0, xn1, xn2, w_ogate_ptr, C, H, BM)
    gate = tl.sigmoid(og)
    _store_h_major(gate_h_ptr, gate, b, ij, NN, H, BM)


@triton.jit
def trimul_final_kernel(
    out_ptr,
    gate_ptr,
    w_ptr,
    nw_ptr,
    nb_ptr,
    res_ptr,

    NN,
    eps,
    C: tl.constexpr,
    H: tl.constexpr,
    BM: tl.constexpr,
    BC: tl.constexpr,
):
    pid_m = tl.program_id(0)
    b = tl.program_id(1)

    local = tl.arange(0, BM)
    ij = pid_m * BM + local
    m_mask = local < NN
    row = b * NN + ij

    offs_h = tl.arange(0, H)
    in_ptrs = b * (H * NN) + ij[None, :] + offs_h[:, None] * NN

    out = tl.load(out_ptr + in_ptrs, mask=m_mask[None, :], other=0.0).to(tl.float32)
    gate = tl.load(gate_ptr + in_ptrs, mask=m_mask[None, :], other=0.0).to(tl.float32)

    mean = tl.sum(out, axis=0) / H
    out_c = out - mean[None, :]
    var = tl.sum(out_c * out_c, axis=0) / H
    inv_std = 1.0 / tl.sqrt(var + eps)

    nw = tl.load(nw_ptr + offs_h).to(tl.float32)
    nb = tl.load(nb_ptr + offs_h).to(tl.float32)

    xh = out_c * inv_std[None, :] * nw[None, :] + nb[None, :]
    xh = xh * gate
    x = tl.trans(xh).to(tl.bfloat16)

    for c0 in range(0, C, BC):
        c = c0 + tl.arange(0, BC)
        c_mask = c < C

        # w_ptr is [C, H] row-major; we need B[h, c] = W[c, h].
        w = tl.load(
            w_ptr + c[None, :] * H + offs_h[:, None],
            mask=c_mask[None, :],
            other=0.0,
        ).to(tl.bfloat16)

        part = tl.dot(x, w)
        out_ptrs = res_ptr + row[:, None] * C + c[None, :]
        tl.store(out_ptrs, part.to(tl.float32), mask=m_mask[:, None] & c_mask[None, :])


def _first_pytorch(
    x,
    mask,
    weights,
    B,
    N,
    C,
    H,
    has_mask,
    eps,
):
    M = B * N * N
    x_flat = x.view(M, C).to(torch.float32, non_blocking=True)

    norm_w = weights["norm.weight"].to(torch.float32, non_blocking=True)
    norm_b = weights["norm.bias"].to(torch.float32, non_blocking=True)
    xn = F.layer_norm(x_flat, (C,), norm_w, norm_b, eps)

    proj_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.bfloat16, non_blocking=True)

    y = xn.to(torch.bfloat16, non_binding=True if False else non_blocking=True) @ proj_w.t()

    left = y[:, :H].view(B, N, N, H)
    right = y[:, H:2*H].view(B, N, N, H)
    lg = torch.sigmoid(y[:, 2*H:3*H]).view(B, N, N, H)
    rg = torch.sigmoid(y[:, 3*H:4*H]).view(B, N, N, H)
    og = torch.sigmoid(y[:, 4*H:5*H]).view(B, N, N, H)

    if has_mask:
        m = mask.to(torch.bfloat16).reshape(B, N, N, 1)
        left = left * m
        right = right * m

    left = (left * lg).to(torch.bfloat16)
    right = (right * rg).to(torch.bfloat16)
    gate = og.to(torch.bfloat16)

    left_h = left.permute(0, 3, 1, 2).contiguous()
    right_h = right.permute(0, 3, 1, 2).contiguous()
    gate_h = gate.permute(0, 3, 1, 2).contiguous()

    return left_h, right_h, gate_h


def _final_pytorch(
    out_h,
    gate_h,
    weights,
    B,
    N,
    H,
    C,
    eps,
    to_out_w_bf16,
):
    out = out_h.view(B, H, N, N).permute(0, 2, 3, 1)
    gate = gate_h.view(B, H, N, N).permute(0, 2, 3, 1)

    out = F.layer_norm(
        out,
        (H,),
        weights["to_out_norm.weight"].to(torch.float32),
        weights["to_out_norm.bias"].to(torch.float32),
        eps,
    )
    out = out * gate
    out = out.to(torch.bfloat16) @ to_out_w_bf16.t()
    return out.to(torch.float32)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    config = config if isinstance(config, dict) else {}

    x = input_tensor
    if x.dtype not in (torch.float16, torch.bfloat16, torch.float32):
        x = x.to(torch.float32)
    x = x.contiguous()

    B, N, _, C = x.shape
    H = int(config.get("hidden_dim", 128))
    eps = float(config.get("eps", config.get("norm_eps", 1e-5)))

    device = x.device
    NN = N * N

    no_mask_flag = bool(config.get("nomask", False)) if isinstance(config, dict) else False
    has_mask = (mask is not None) and (not no_mask_flag) and (mask.numel() > 0)

    mask_flat = None
    if has_mask:
        mask_flat = mask.to(torch.float32, non_blocking=True).contiguous().reshape(-1)

    norm_w = weights["norm.weight"].to(torch.bfloat16, non_blocking=True).contiguous()
    norm_b = weights["norm.bias"].to(torch.bfloat16, non_blocking=True).contiguous()

    w_left = weights["left_proj.weight"].to(torch.bfloat16, non_blocking=True).contiguous()
    w_right = weights["right_proj.weight"].to(torch.bfloat16, non_blocking=True).contiguous()
    w_lgate = weights["left_gate.weight"].to(torch.bfloat16, non_blocking=True).contiguous()
    w_rgate = weights["right_gate.weight"].to(torch.bfloat16, non_blocking=True).contiguous()
    w_ogate = weights["out_gate.weight"].to(torch.bfloat16, non_blocking=True).contiguous()

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.bfloat16, non_blocking=True).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.bfloat16, non_blocking=True).contiguous()
    to_out_w = weights["to_out.weight"].to(torch.bfloat16, non_blocking=True).contiguous()

    use_fused_first = (H == 128 and C in (128, 256, 384))

    if use_fused_first:
        left_h = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
        right_h = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
        gate_h = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)

        bm = 64 if C <= 128 else 32
        grid = (triton.cdiv(NN, bm), B)
        trimul_first_kernel[grid](
            x,
            mask_flat if has_mask else x,
            left_h,
            right_h,
            gate_h,
            norm_w,
            norm_b,
            w_left,
            w_right,
            w_lgate,
            w_rgate,
            w_ogate,
            NN,
            eps,
            C=C,
            H=H,
            BM=bm,
            HAS_MASK=has_mask,
            num_warps=4,
        )
    else:
        left_h, right_h, gate_h = _first_pytorch(
            x,
            mask,
            weights,
            B,
            N,
            C,
            H,
            has_mask,
            eps,
        )

    # Heavy part: 128 (or H) independent N x N GEMMs.
    L = left_h.reshape(B * H, N, N)
    R = right_h.reshape(B * H, N, N)
    out_h = torch.bmm(L, R.transpose(1, 2))

    del left_h
    del right_h

    final_tri = (H == 128)
    if final_tri:
        out = torch.empty((B, N, N, C), dtype=torch.float32, device=device)
        bm_final = 64
        grid_final = (triton.cdiv(NN, bm_final), B)
        trimul_final_kernel[grid_final](
            out_h,
            gate_h,
            to_out_w,
            to_out_norm_w,
            to_out_norm_b,
            out,
            NN,
            eps,
            C=C,
            H=H,
            BM=bm_final,
            BC=128,
            num_warps=4,
        )
    else:
        out = _final_pytorch(
            out_h,
            gate_h,
            weights,
            B,
            N,
            H,
            C,
            eps,
            to_out_w,
        )

    return out
```
