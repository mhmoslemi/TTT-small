```python
import torch
import triton
import triton.language as tl


"""
TriMul outgoing forward pass with fused Triton kernels.

The operator is:
  xn = LayerNorm(x)
  L  = xn @ Wl^T * mask * sigmoid(xn @ Wlg^T)
  R  = xn @ Wr^T * mask * sigmoid(xn @ Wrg^T)
  Y  = sum_k L[i,k,:] * R[j,k,:]   (for every b,i,j, elementwise over hidden)
  z  = LayerNorm(Y) * sigmoid(xn @ Wog^T)
  out = z @ Wout^T

Kernels:
  1. Triton LayerNorm: x -> normalized bf16 token tensor.
  2. Triton fused projection/gating: one pass over normalized tokens to produce
     h-major L, R and h-major out-gate.
  3. cuBLAS batched GEMM for the O(N^3) triangular product.
  4. Triton transpose + LayerNorm moments for Y.
  5. Triton final LayerNorm + gating + linear projection to float32 output.
"""


def _npow2(n: int) -> int:
    return 1 << (n - 1).bit_length()


@triton.jit
def _norm_kernel(
    x_ptr, xn_ptr,
    norm_w_ptr, norm_b_ptr,
    E,
    C: tl.constexpr,
    CPAD: tl.constexpr,
    T: tl.constexpr,
    EPS: tl.constexpr,
):
    pid = tl.program_id(0)
    t = pid * T + tl.arange(0, T)
    t_ok = t < E

    c = tl.arange(0, CPAD)
    c_ok = c < C
    x_mask = t_ok[:, None] & c_ok[None, :]

    x = tl.load(
        x_ptr + t[:, None] * C + c[None, :],
        mask=x_mask,
        other=0.0,
    ).to(tl.float32)

    x = tl.where(c_ok[None, :], x, 0.0)

    s = tl.sum(x, axis=1)
    mean = s / C

    centered = tl.where(c_ok[None, :], x - mean[:, None], 0.0)
    sq = tl.sum(centered * centered, axis=1)
    var = sq / C
    var = tl.maximum(var, 0.0)
    rstd = tl.rsqrt(var + EPS)

    w = tl.load(norm_w_ptr + c, mask=c_ok, other=0.0).to(tl.float32)
    b = tl.load(norm_b_ptr + c, mask=c_ok, other=0.0).to(tl.float32)

    y = centered * rstd[:, None] * w[None, :] + b[None, :]

    tl.store(
        xn_ptr + t[:, None] * C + c[None, :],
        y.to(tl.bfloat16),
        mask=x_mask,
    )


@triton.jit
def _proj_kernel(
    xn_ptr, mask_ptr,
    L_ptr, R_ptr, OG_ptr,
    WLP, WRP,
    WLG, WRG,
    WOG,
    N,
    C: tl.constexpr,
    H: tl.constexpr,
    CHUNK: tl.constexpr,
    T: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_t = tl.program_id(1)

    N2 = N * N
    t = pid_t * T + tl.arange(0, T)
    t_ok = t < N2

    h = tl.arange(0, H)
    h_ok = h >= 0

    acc_lp = tl.zeros((H, T), dtype=tl.float32)
    acc_rp = tl.zeros((H, T), dtype=tl.float32)
    acc_lg = tl.zeros((H, T), dtype=tl.float32)
    acc_rg = tl.zeros((H, T), dtype=tl.float32)
    acc_og = tl.zeros((H, T), dtype=tl.float32)

    base_xn = (pid_b * N2 + t)[:, None] * C
    w_mask = h_ok[:, None] & (tl.arange(0, CHUNK) < 64)[None, :]

    for c0 in range(0, C, CHUNK):
        c = c0 + tl.arange(0, CHUNK)
        c_ok = c < C

        x_mask = t_ok[:, None] & c_ok[None, :]
        a = tl.load(
            xn_ptr + base_xn + c[None, :],
            mask=x_mask,
            other=0.0,
        )
        at = tl.trans(a)

        wm = h_ok[:, None] & c_ok[None, :]

        w = tl.load(WLP + h[:, None] * C + c[None, :], mask=wm, other=0.0)
        acc_lp = tl.dot(w, at, acc=acc_lp)

        w = tl.load(WLG + h[:, None] * C + c[None, :], mask=wm, other=0.0)
        acc_lg = tl.dot(w, at, acc=acc_lg)

        w = tl.load(WRP + h[:, None] * C + c[None, :], mask=wm, other=0.0)
        acc_rp = tl.dot(w, at, acc=acc_rp)

        w = tl.load(WRG + h[:, None] * C + c[None, :], mask=wm, other=0.0)
        acc_rg = tl.dot(w, at, acc=acc_rg)

        w = tl.load(WOG + h[:, None] * C + c[None, :], mask=wm, other=0.0)
        acc_og = tl.dot(w, at, acc=acc_og)

    if HAS_MASK:
        m = tl.load(
            mask_ptr + pid_b * N2 + t,
            mask=t_ok,
            other=0.0,
        ).to(tl.float32)
        m1 = m[None, :]
    else:
        m1 = 1.0

    lg = tl.sigmoid(acc_lg)
    rg = tl.sigmoid(acc_rg)

    l_out = acc_lp * lg * m1
    r_out = acc_rp * rg * m1
    og = tl.sigmoid(acc_og)

    store_mask = h_ok[:, None] & t_ok[None, :]
    base = pid_b * H * N2
    offs = base + h[:, None] * N2 + t[None, :]

    tl.store(L_ptr + offs, l_out.to(tl.bfloat16), mask=store_mask)
    tl.store(R_ptr + offs, r_out.to(tl.bfloat16), mask=store_mask)
    tl.store(OG_ptr + offs, og.to(tl.bfloat16), mask=store_mask)


@triton.jit
def _transpose_moments_kernel(
    Yh_ptr, Ycnt_ptr,
    mean_ptr, rstd_ptr,
    N,
    H: tl.constexpr,
    T: tl.constexpr,
    EPS: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_t = tl.program_id(1)

    N2 = N * N
    t = pid_t * T + tl.arange(0, T)
    t_ok = t < N2

    h = tl.arange(0, H)
    h_ok = h >= 0

    load_mask = h_ok[:, None] & t_ok[None, :]
    y_ht = tl.load(
        Yh_ptr + pid_b * H * N2 + h[:, None] * N2 + t[None, :],
        mask=load_mask,
        other=0.0,
    ).to(tl.float32)

    y_t = tl.trans(y_ht)  # [T, H]

    s = tl.sum(y_t, axis=1)
    mean = s / H

    yc = y_t - mean[:, None]
    sq = tl.sum(yc * yc, axis=1)
    var = sq / H
    var = tl.maximum(var, 0.0)
    rstd = tl.rsqrt(var + EPS)

    base = pid_b * N2 + t
    store_mask = t_ok[:, None] & h_ok[None, :]

    tl.store(
        Ycnt_ptr + base[:, None] * H + h[None, :],
        y_t.to(tl.bfloat16),
        mask=store_mask,
    )
    tl.store(mean_ptr + base, mean, mask=t_ok)
    tl.store(rstd_ptr + base, rstd, mask=t_ok)


@triton.jit
def _final_kernel(
    Y_ptr, OGh_ptr,
    mean_ptr, rstd_ptr,
    lnw_ptr, lnb_ptr,
    Wout_ptr,
    out_ptr,
    N,
    C: tl.constexpr,
    H: tl.constexpr,
    T: tl.constexpr,
    CB: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_t = tl.program_id(1)

    N2 = N * N
    t = pid_t * T + tl.arange(0, T)
    t_ok = t < N2
    t_base = pid_b * N2 + t

    h = tl.arange(0, H)
    h_ok = h >= 0

    y_mask = t_ok[:, None] & h_ok[None, :]
    y = tl.load(
        Y_ptr + t_base[:, None] * H + h[None, :],
        mask=y_mask,
        other=0.0,
    ).to(tl.float32)

    og_mask = h_ok[:, None] & t_ok[None, :]
    og_ht = tl.load(
        OGh_ptr + pid_b * H * N2 + h[:, None] * N2 + t[None, :],
        mask=og_mask,
        other=0.0,
    ).to(tl.float32)
    og = tl.trans(og_ht)

    mean = tl.load(mean_ptr + t_base, mask=t_ok, other=0.0).to(tl.float32)
    rstd = tl.load(rstd_ptr + t_base, mask=t_ok, other=0.0).to(tl.float32)

    lnw = tl.load(lnw_ptr + h, mask=h_ok, other=0.0).to(tl.float32)
    lnb = tl.load(lnb_ptr + h, mask=h_ok, other=0.0).to(tl.float32)

    z = (y - mean[:, None]) * rstd[:, None] * lnw[None, :] + lnb[None, :]
    z = z * og
    zb = z.to(tl.bfloat16)

    for c0 in range(0, C, CB):
        c = c0 + tl.arange(0, CB)
        c_ok = c < C
        w_mask = h_ok[:, None] & c_ok[None, :]

        w = tl.load(
            Wout_ptr + h[:, None] * C + c[None, :],
            mask=w_mask,
            other=0.0,
        )

        ob = tl.dot(zb, w).to(tl.float32)

        store_mask = t_ok[:, None] & c_ok[None, :]
        tl.store(
            out_ptr + t_base[:, None] * C + c[None, :],
            ob,
            mask=store_mask,
        )


def custom_kernel(data):
    """Execute fused TriMul outgoing forward pass and return float32 output."""
    if len(data) == 4:
        x, mask_in, weights, config = data
    elif len(data) == 3:
        x, weights, config = data
        mask_in = None
    else:
        raise ValueError("custom_kernel expects (x, mask, weights, config) or (x, weights, config).")

    with torch.no_grad():
        if not x.is_contiguous() or x.dtype != torch.float32:
            x = x.to(torch.float32, copy=False).contiguous()

        B, N, N2, C = x.shape
        device = x.device
        E = B * N2

        H = int(weights["left_proj.weight"].shape[0])

        norm_w = weights["norm.weight"].to(torch.float32, copy=False).contiguous()
        norm_b = weights["norm.bias"].to(torch.float32, copy=False).contiguous()

        WLP = weights["left_proj.weight"].to(torch.bfloat16, copy=False).contiguous()
        WRP = weights["right_proj.weight"].to(torch.bfloat16, copy=False).contiguous()
        WLG = weights["left_gate.weight"].to(torch.bfloat16, copy=False).contiguous()
        WRG = weights["right_gate.weight"].to(torch.bfloat16, copy=False).contiguous()
        WOG = weights["out_gate.weight"].to(torch.bfloat16, copy=False).contiguous()

        Wout = weights["to_out.weight"].to(torch.bfloat16, copy=False).t().contiguous()

        to_out_ln_w = weights["to_out_norm.weight"].to(torch.float32, copy=False).contiguous()
        to_out_ln_b = weights["to_out_norm.bias"].to(torch.float32, copy=False).contiguous()

        has_mask = mask_in is not None and torch.is_tensor(mask_in)
        if has_mask:
            mask_t = mask_in.to(torch.float32, copy=False).contiguous()
        else:
            mask_t = torch.empty(1, dtype=torch.float32, device=device)

        # 1) LayerNorm -> bf16 normalized tokens.
        CPAD = _npow2(C)
        if CPAD <= 128:
            T0 = 64
        elif CPAD <= 256:
            T0 = 32
        else:
            T0 = 16

        xn = torch.empty((B, N, N2, C), dtype=torch.bfloat16, device=device)
        grid0 = (triton.cdiv(E, T0),)
        _norm_kernel[grid0](
            x, xn,
            norm_w, norm_b,
            E,
            C=C,
            CPAD=CPAD,
            T=T0,
            EPS=1e-5,
            num_warps=4,
        )
        x = None

        # 2) Fused projections, masks, gates -> h-major L/R and out-gate.
        Lh = torch.empty((B, H, N, N2), dtype=torch.bfloat16, device=device)
        Rh = torch.empty((B, H, N, N2), dtype=torch.bfloat16, device=device)
        Ogh = torch.empty((B, H, N, N2), dtype=torch.bfloat16, device=device)

        T1 = 16
        CHUNK1 = 64 if C >= 64 else max(16, _npow2(C))
        grid1 = (B, triton.cdiv(N2, T1))
        _proj_kernel[grid1](
            xn, mask_t,
            Lh, Rh, Ogh,
            WLP, WRP,
            WLG, WRG,
            WOG,
            N,
            C=C,
            H=H,
            CHUNK=CHUNK1,
            T=T1,
            HAS_MASK=has_mask,
            num_warps=4,
        )
        xn = None

        # 3) Core TriMul product: for each (b,h), L @ R^T over sequence dimension.
        A = Lh.view(B * H, N, N2)
        Bm = Rh.view(B * H, N, N2)
        Yh = torch.bmm(A, Bm.transpose(-1, -2)).contiguous()
        Lh = None
        Rh = None

        # 4) Transpose hidden-last layout and compute LayerNorm moments for Y.
        Ycnt = torch.empty((B, N, N2, H), dtype=torch.bfloat16, device=device)
        mean = torch.empty((B, N, N2), dtype=torch.float32, device=device)
        rstd = torch.empty((B, N, N2), dtype=torch.float32, device=device)

        T3 = 128
        grid3 = (B, triton.cdiv(N2, T3))
        _transpose_moments_kernel[grid3](
            Yh, Ycnt,
            mean, rstd,
            N,
            H=H,
            T=T3,
            EPS=1e-5,
            num_warps=8,
        )
        Yh = None

        # 5) Final LayerNorm + out-gate + linear projection.
        out = torch.empty((B, N, N2, C), dtype=torch.float32, device=device)

        T4 = 64
        CB = 128 if C >= 128 else max(16, _npow2(C))
        grid4 = (B, triton.cdiv(N2, T4))
        _final_kernel[grid4](
            Ycnt, Ogh,
            mean, rstd,
            to_out_ln_w, to_out_ln_b,
            Wout,
            out,
            N,
            C=C,
            H=H,
            T=T4,
            CB=CB,
            num_warps=8,
        )

        return out
```
