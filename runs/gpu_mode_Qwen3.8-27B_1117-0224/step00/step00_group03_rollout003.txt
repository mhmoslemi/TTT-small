```python
import torch
import triton
import triton.language as tl


"""
TriMul forward pass:
  1) fused LayerNorm + 5 linear projections + gating/masking in Triton
  2) per-channel triangle multiplication as batched GEMMs in Triton
       out[b,h,i,j] = sum_k left[b,h,i,k] * right[b,h,j,k]
  3) fused LayerNorm + out-gate + final linear in Triton
"""


@triton.jit
def _pre_load_norm_xt(
    x_ptr,
    x_off,
    rows,
    row_mask,
    c0,
    C,
    mean,
    rstd,
    normw_ptr,
    normb_ptr,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    c_offs = tl.arange(0, BLOCK_C)
    cs = c0 + c_offs
    cm = cs < C

    x_ptrs = x_ptr + x_off + rows[None, :] * C + cs[None, :]
    x = tl.load(x_ptrs, mask=row_mask[:, None] & cm[None, :], other=0.0).to(tl.float32)

    nw = tl.load(normw_ptr + cs, mask=cm, other=0.0).to(tl.float32)
    nb = tl.load(normb_ptr + cs, mask=cm, other=0.0).to(tl.float32)

    xn = (x - mean[None, :]) * rstd[None, :] * nw[None, :] + nb[None, :]
    return tl.trans(xn)


@triton.jit
def _pre_load_w(
    w_ptr,
    H,
    h0,
    c0,
    C,
    BLOCK_H: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    h_offs = tl.arange(0, BLOCK_H)
    c_offs = tl.arange(0, BLOCK_C)

    hs = h0 + h_offs
    cs = c0 + c_offs

    mask = (hs < H)[:, None] & (cs < C)[None, :]
    ptrs = w_ptr + hs[:, None] * C + cs[None, :]
    return tl.load(ptrs, mask=mask, other=0.0).to(tl.float32)


@triton.jit
def _pre_kernel(
    x_ptr,
    mask_ptr,
    left_ptr,
    right_ptr,
    outgate_ptr,
    wl_ptr,
    wr_ptr,
    wlg_ptr,
    wrg_ptr,
    wog_ptr,
    normw_ptr,
    normb_ptr,
    plane,
    C,
    H,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
    BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    EPS: tl.constexpr,
):
    b = tl.program_id(0)
    mb = tl.program_id(1)

    rows = mb * BLOCK_M + tl.arange(0, BLOCK_M)
    row_mask = rows < plane
    x_off = b * plane * C

    # LayerNorm statistics over the input channel dimension.
    sum1 = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum2 = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_c = tl.cdiv(C, BLOCK_C)
    for cb in range(num_c):
        c0 = cb * BLOCK_C
        c_offs = tl.arange(0, BLOCK_C)
        cs = c0 + c_offs
        cm = cs < C

        x_ptrs = x_ptr + x_off + rows[None, :] * C + cs[None, :]
        x = tl.load(x_ptrs, mask=row_mask[:, None] & cm[None, :], other=0.0).to(tl.float32)

        sum1 += tl.sum(x, axis=1)
        sum2 += tl.sum(x * x, axis=1)

    mean = sum1 / C
    var = sum2 / C - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + EPS)

    if HAS_MASK:
        m = tl.load(mask_ptr + b * plane + rows, mask=row_mask, other=0.0).to(tl.float32)
    else:
        m = tl.full((BLOCK_M,), 1.0, tl.float32)

    h_offs = tl.arange(0, BLOCK_H)
    store_base = b * H * plane

    num_h = tl.cdiv(H, BLOCK_H)
    for hb in range(num_h):
        h0 = hb * BLOCK_H
        hm = (h0 + h_offs) < H
        store_mask = hm[:, None] & row_mask[None, :]

        acc_l = tl.zeros((BLOCK_H, BLOCK_M), dtype=tl.float32)
        acc_r = tl.zeros((BLOCK_H, BLOCK_M), dtype=tl.float32)
        acc_lg = tl.zeros((BLOCK_H, BLOCK_M), dtype=tl.float32)
        acc_rg = tl.zeros((BLOCK_H, BLOCK_M), dtype=tl.float32)
        acc_og = tl.zeros((BLOCK_H, BLOCK_M), dtype=tl.float32)

        for cb in range(num_c):
            c0 = cb * BLOCK_C

            xT = _pre_load_norm_xt(
                x_ptr, x_off, rows, row_mask,
                c0, C, mean, rstd, normw_ptr, normb_ptr,
                BLOCK_M, BLOCK_C,
            )

            wl = _pre_load_w(wl_ptr, H, h0, c0, C, BLOCK_H, BLOCK_C)
            wr = _pre_load_w(wr_ptr, H, h0, c0, C, BLOCK_H, BLOCK_C)
            wlg = _pre_load_w(wlg_ptr, H, h0, c0, C, BLOCK_H, BLOCK_C)
            wrg = _pre_load_w(wrg_ptr, H, h0, c0, C, BLOCK_H, BLOCK_C)
            wog = _pre_load_w(wog_ptr, H, h0, c0, C, BLOCK_H, BLOCK_C)

            acc_l += tl.dot(wl, xT, input_precision="tf32")
            acc_r += tl.dot(wr, xT, input_precision="tf32")
            acc_lg += tl.dot(wlg, xT, input_precision="tf32")
            acc_rg += tl.dot(wrg, xT, input_precision="tf32")
            acc_og += tl.dot(wog, xT, input_precision="tf32")

        left_val = acc_l * tl.sigmoid(acc_lg) * m[None, :]
        right_val = acc_r * tl.sigmoid(acc_rg) * m[None, :]
        outgate_val = tl.sigmoid(acc_og)

        out_offs = (h0 + h_offs)[:, None] * plane + rows[None, :]
        tl.store(
            left_ptr + store_base + out_offs,
            left_val,
            mask=store_mask,
        )
        tl.store(
            right_ptr + store_base + out_offs,
            right_val,
            mask=store_mask,
        )
        tl.store(
            outgate_ptr + store_base + out_offs,
            outgate_val,
            mask=store_mask,
        )


@triton.jit
def _core_kernel(
    l_ptr,
    r_ptr,
    out_ptr,

    N,

    # left strides: batch, h, i, k
    stride_lb, stride_lh, stride_li, stride_lk,

    # right strides: batch, h, j, k
    stride_rb, stride_rh, stride_rj, stride_rk,

    # out strides: batch, h, i, j
    stride_ob, stride_oh, stride_oi, stride_oj,

    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_ij = tl.program_id(0)
    h = tl.program_id(1)
    b = tl.program_id(2)

    num_j = tl.cdiv(N, BLOCK_J)
    pid_i = pid_ij // num_j
    pid_j = pid_ij % num_j

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)

    im = offs_i < N
    jm = offs_j < N

    l_base = l_ptr + b * stride_lb + h * stride_lh
    r_base = r_ptr + b * stride_rb + h * stride_rh

    a_ptrs = l_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
    b_ptrs = r_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k):
        k_mask = (kb * BLOCK_K + offs_k) < N

        a = tl.load(a_ptrs, mask=im[:, None] & k_mask[None, :], other=0.0)
        bmat = tl.load(b_ptrs, mask=k_mask[:, None] & jm[None, :], other=0.0)

        acc += tl.dot(a, bmat, input_precision="tf32")

        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    o_base = out_ptr + b * stride_ob + h * stride_oh
    o_ptrs = o_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(o_ptrs, acc, mask=im[:, None] & jm[None, :])


@triton.jit
def _post_kernel(
    out_ptr,
    gate_ptr,
    w_ptr,
    normw_ptr,
    normb_ptr,
    y_ptr,

    plane,
    C,
    H,

    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_C: tl.constexpr,
    FULL_H: tl.constexpr,
    EPS: tl.constexpr,
):
    b = tl.program_id(0)
    mb = tl.program_id(1)

    rows = mb * BLOCK_M + tl.arange(0, BLOCK_M)
    row_mask = rows < plane
    out_base = b * H * plane

    c_offs = tl.arange(0, BLOCK_C)
    num_c = tl.cdiv(C, BLOCK_C)

    if FULL_H:
        h_offs = tl.arange(0, BLOCK_H)
        hm = h_offs < H
        load_mask = hm[:, None] & row_mask[None, :]

        x = tl.load(
            out_ptr + out_base + h_offs[:, None] * plane + rows[None, :],
            mask=load_mask,
            other=0.0,
        ).to(tl.float32)

        g = tl.load(
            gate_ptr + out_base + h_offs[:, None] * plane + rows[None, :],
            mask=load_mask,
            other=0.0,
        ).to(tl.float32)

        s1 = tl.sum(x, axis=0)
        s2 = tl.sum(x * x, axis=0)

        mean = s1 / H
        var = tl.maximum(s2 / H - mean * mean, 0.0)
        rstd = 1.0 / tl.sqrt(var + EPS)

        hw = tl.load(normw_ptr + h_offs, mask=hm, other=0.0).to(tl.float32)
        hb = tl.load(normb_ptr + h_offs, mask=hm, other=0.0).to(tl.float32)

        val = ((x - mean[None, :]) * rstd[None, :]) * hw[:, None] + hb[:, None]
        val = val * g

        vals = tl.trans(val)  # [BLOCK_M, BLOCK_H]

        for cb in range(num_c):
            c0 = cb * BLOCK_C
            cs = c0 + c_offs
            cm = cs < C

            w = tl.load(
                w_ptr + cs[:, None] * H + h_offs[None, :],
                mask=cm[:, None] & hm[None, :],
                other=0.0,
            ).to(tl.float32)

            wT = tl.trans(w)  # [BLOCK_H, BLOCK_C]
            y = tl.dot(vals, wT, input_precision="tf32")

            tl.store(
                y_ptr + b * plane * C + rows[:, None] * C + cs[None, :],
                y,
                mask=row_mask[:, None] & cm[None, :],
            )
    else:
        h_offs = tl.arange(0, BLOCK_H)
        num_h = tl.cdiv(H, BLOCK_H)

        s1 = tl.zeros((BLOCK_M,), dtype=tl.float32)
        s2 = tl.zeros((BLOCK_M,), dtype=tl.float32)

        for hb in range(num_h):
            h0 = hb * BLOCK_H
            hh = h0 + h_offs
            hm = hh < H
            load_mask = hm[:, None] & row_mask[None, :]

            x = tl.load(
                out_ptr + out_base + hh[:, None] * plane + rows[None, :],
                mask=load_mask,
                other=0.0,
            ).to(tl.float32)

            s1 += tl.sum(x, axis=0)
            s2 += tl.sum(x * x, axis=0)

        mean = s1 / H
        var = tl.maximum(s2 / H - mean * mean, 0.0)
        rstd = 1.0 / tl.sqrt(var + EPS)

        for cb in range(num_c):
            c0 = cb * BLOCK_C
            cs = c0 + c_offs
            cm = cs < C

            y_acc = tl.zeros((BLOCK_M, BLOCK_C), dtype=tl.float32)

            for hb in range(num_h):
                h0 = hb * BLOCK_H
                hh = h0 + h_offs
                hm = hh < H
                load_mask = hm[:, None] & row_mask[None, :]

                x = tl.load(
                    out_ptr + out_base + hh[:, None] * plane + rows[None, :],
                    mask=load_mask,
                    other=0.0,
                ).to(tl.float32)

                g = tl.load(
                    gate_ptr + out_base + hh[:, None] * plane + rows[None, :],
                    mask=load_mask,
                    other=0.0,
                ).to(tl.float32)

                hw = tl.load(normw_ptr + hh, mask=hm, other=0.0).to(tl.float32)
                hbias = tl.load(normb_ptr + hh, mask=hm, other=0.0).to(tl.float32)

                val = ((x - mean[None, :]) * rstd[None, :]) * hw[:, None] + hbias[:, None]
                val = val * g

                vals = tl.trans(val)  # [BLOCK_M, BLOCK_H]

                w = tl.load(
                    w_ptr + cs[:, None] * H + hh[None, :],
                    mask=cm[:, None] & hm[None, :],
                    other=0.0,
                ).to(tl.float32)

                wT = tl.trans(w)  # [BLOCK_H, BLOCK_C]
                y_acc += tl.dot(vals, wT, input_precision="tf32")

            tl.store(
                y_ptr + b * plane * C + rows[:, None] * C + cs[None, :],
                y_acc,
                mask=row_mask[:, None] & cm[None, :],
            )


def _next_pow2(n: int) -> int:
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _as_float32_contiguous(t: torch.Tensor) -> torch.Tensor:
    return t.detach().float().contiguous()


def custom_kernel(data):
    """Forward TriMul module implemented with fused Triton kernels."""
    if len(data) == 3:
        x, weights, config = data
        mask = None
    else:
        x, mask, weights, config = data

    x = x.contiguous()
    B, N1, N2, C = x.shape
    N = N1
    plane = N * N

    # We assume the square interface used by TriMul; tests provide N1 == N2.
    if N2 != N1:
        plane = N1 * N2
        N = N1

    if "hidden_dim" in config:
        H = int(config["hidden_dim"])
    else:
        H = int(weights["left_proj.weight"].shape[0])

    if mask is None:
        has_mask = False
        mask_arg = x
    else:
        if mask.dim() == 4:
            mask = mask.squeeze(-1)
        if mask.dim() == 2:
            mask = mask.unsqueeze(0)
        if mask.shape[0] == 1 and B != 1:
            mask = mask.expand(B, -1, -1)
        mask = mask.contiguous().to(torch.float32, copy=False)
        mask = mask.view(B, plane)
        has_mask = True
        mask_arg = mask

    norm_w = _as_float32_contiguous(weights["norm.weight"])
    norm_b = _as_float32_contiguous(weights["norm.bias"])

    wl = _as_float32_contiguous(weights["left_proj.weight"])
    wr = _as_float32_contiguous(weights["right_proj.weight"])
    wlg = _as_float32_contiguous(weights["left_gate.weight"])
    wrg = _as_float32_contiguous(weights["right_gate.weight"])
    wog = _as_float32_contiguous(weights["out_gate.weight"])

    to_norm_w = _as_float32_contiguous(weights["to_out_norm.weight"])
    to_norm_b = _as_float32_contiguous(weights["to_out_norm.bias"])
    to_out_w = _as_float32_contiguous(weights["to_out.weight"])

    device = x.device
    dtype = torch.float32

    # Planar [B, H, N, N] layout makes the core a standard batched GEMM:
    # out_h = left_h @ right_h^T for each h.
    left = torch.empty((B, H, plane), dtype=dtype, device=device)
    right = torch.empty((B, H, plane), dtype=dtype, device=device)
    outgate = torch.empty((B, H, plane), dtype=dtype, device=device)

    # --- Fused LayerNorm + projections + gating/mask -----------------------
    PRE_BLOCK_M = 32
    PRE_BLOCK_C = 64
    PRE_BLOCK_H = 64

    grid_pre = (B, triton.cdiv(plane, PRE_BLOCK_M))
    _pre_kernel[grid_pre](
        x,
        mask_arg,
        left,
        right,
        outgate,
        wl,
        wr,
        wlg,
        wrg,
        wog,
        norm_w,
        norm_b,
        plane,
        C,
        H,
        PRE_BLOCK_M,
        PRE_BLOCK_C,
        PRE_BLOCK_H,
        HAS_MASK=has_mask,
        EPS=1e-5,
        num_warps=4,
        num_stages=2,
    )

    # --- Core triangle multiplication --------------------------------------
    out = torch.empty((B, H, plane), dtype=dtype, device=device)

    if N <= 256:
        BLOCK_I = 64
        BLOCK_J = 64
        BLOCK_K = 64
        core_warps = 4
    else:
        BLOCK_I = 128
        BLOCK_J = 128
        BLOCK_K = 64
        core_warps = 8

    num_i = triton.cdiv(N, BLOCK_I)
    num_j = triton.cdiv(N, BLOCK_J)

    # Left/right are [B,H,N,N] contiguous:
    # stride: b=H*plane, h=plane, row=N, col=1.
    s_batch = H * plane
    s_h = plane
    s_row = N
    s_col = 1

    grid_core = (num_i * num_j, H, B)
    _core_kernel[grid_core](
        left,
        right,
        out,
        N,
        s_batch, s_h, s_row, s_col,
        s_batch, s_h, s_row, s_col,
        s_batch, s_h, s_row, s_col,
        BLOCK_I,
        BLOCK_J,
        BLOCK_K,
        num_warps=core_warps,
        num_stages=3,
    )

    # --- Fused final LayerNorm + gate + linear ------------------------------
    y = torch.empty((B, N, N, C), dtype=dtype, device=device)

    POST_BLOCK_M = 32
    POST_BLOCK_C = 128

    if H <= 128:
        FULL_H = True
        POST_BLOCK_H = _next_pow2(H)
    else:
        FULL_H = False
        POST_BLOCK_H = 128

    grid_post = (B, triton.cdiv(plane, POST_BLOCK_M))
    _post_kernel[grid_post](
        out,
        outgate,
        to_out_w,
        to_norm_w,
        to_norm_b,
        y,
        plane,
        C,
        H,
        POST_BLOCK_M,
        POST_BLOCK_H,
        POST_BLOCK_C,
        FULL_H=FULL_H,
        EPS=1e-5,
        num_warps=4,
        num_stages=2,
    )

    return y
```
