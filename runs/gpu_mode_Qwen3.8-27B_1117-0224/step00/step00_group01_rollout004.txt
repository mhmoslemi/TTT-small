"""
Fused TriMul forward for AlphaFold3-style outgoing triangle multiplicative update.

Pipeline:
1) Triton kernel: fused LayerNorm + left/right/out-gate projections.
   Left and right projections are written in [B, H, N, N] layout, and
   out-gate is also stored in [B, H, N, N]; this makes the heavy
   batched matmul and final normalization coalesced.
2) Triton batched GEMM:
       S[b, h, i, j] = sum_k L[b, h, i, k] * R[b, h, j, k]
   which is the TriMul outgoing einsum.
3) Triton fused final kernel:
   - reads S and out-gate in [B, H, N, N]
   - computes LayerNorm over H
   - multiplies by sigmoid(out-gate)
   - does the final projection to [B, N, N, dim] with tl.dot
   - writes float32 output.

For compatibility, a PyTorch reference path is used only if hidden_dim is not
a multiple of the 128-wide Triton block used by the kernels.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _trimul_proj_kernel(
    x_ptr,
    norm_w_ptr,
    norm_b_ptr,
    w_l_ptr,
    w_lg_ptr,
    w_r_ptr,
    w_rg_ptr,
    w_og_ptr,
    l_ptr,
    r_ptr,
    og_ptr,
    mask_ptr,

    M,
    N2,
    C,
    H,
    EPS,

    HAS_MASK: tl.constexpr,
    BLOCK_P: tl.constexpr,
    BLOCK_C: tl.constexpr,
    H_BLOCK: tl.constexpr,
    NUM_HB: tl.constexpr,
):
    pid = tl.program_id(0)
    mode = pid % 3
    pid_p = pid // 3

    offs_p = pid_p * BLOCK_P + tl.arange(0, BLOCK_P)
    p_ok = offs_p < M
    offs_b = offs_p // N2
    local_p = offs_p - offs_b * N2

    # LayerNorm statistics over C.
    sums = tl.zeros((BLOCK_P,), dtype=tl.float32)
    sumsq = tl.zeros((BLOCK_P,), dtype=tl.float32)
    for c0 in range(0, C, BLOCK_C):
        c_offs = c0 + tl.arange(0, BLOCK_C)
        c_ok = c_offs < C
        x = tl.load(
            x_ptr + offs_p[:, None] * C + c_offs[None, :],
            mask=p_ok[:, None] & c_ok[None, :],
            other=0.0,
        ).to(tl.float32)
        sums += tl.sum(x, axis=1)
        sumsq += tl.sum(x * x, axis=1)

    mean = sums / C
    var = sumsq / C - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = tl.rsqrt(var + EPS)

    if mode == 0:
        if HAS_MASK:
            mask_vals = tl.load(mask_ptr + offs_p, mask=p_ok, other=0.0).to(tl.float32)
        else:
            mask_vals = tl.full((BLOCK_P,), 1.0, dtype=tl.float32)

        for hb in tl.static_range(NUM_HB):
            h_offs = hb * H_BLOCK + tl.arange(0, H_BLOCK)
            acc = tl.zeros((BLOCK_P, H_BLOCK), dtype=tl.float32)
            accg = tl.zeros((BLOCK_P, H_BLOCK), dtype=tl.float32)

            for c0 in range(0, C, BLOCK_C):
                c_offs = c0 + tl.arange(0, BLOCK_C)
                c_ok = c_offs < C

                x = tl.load(
                    x_ptr + offs_p[:, None] * C + c_offs[None, :],
                    mask=p_ok[:, None] & c_ok[None, :],
                    other=0.0,
                ).to(tl.float32)

                nw = tl.load(norm_w_ptr + c_offs, mask=c_ok, other=0.0).to(tl.float32)
                nb = tl.load(norm_b_ptr + c_offs, mask=c_ok, other=0.0).to(tl.float32)
                a = ((x - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]).to(tl.bfloat16)

                w = tl.load(
                    w_l_ptr + c_offs[:, None] * H + h_offs[None, :],
                    mask=c_ok[:, None],
                    other=0.0,
                )
                acc = tl.dot(a, w, acc)

                wg = tl.load(
                    w_lg_ptr + c_offs[:, None] * H + h_offs[None, :],
                    mask=c_ok[:, None],
                    other=0.0,
                )
                accg = tl.dot(a, wg, accg)

            val = acc * mask_vals[:, None] * tl.sigmoid(accg)
            val_t = tl.trans(val.to(tl.bfloat16))
            st_ptr = l_ptr + (offs_b[None, :] * H + h_offs[:, None]) * N2 + local_p[None, :]
            tl.store(st_ptr, val_t, mask=p_ok[None, :])

    elif mode == 1:
        if HAS_MASK:
            mask_vals = tl.load(mask_ptr + offs_p, mask=p_ok, other=0.0).to(tl.float32)
        else:
            mask_vals = tl.full((BLOCK_P,), 1.0, dtype=tl.float32)

        for hb in tl.static_range(NUM_HB):
            h_offs = hb * H_BLOCK + tl.arange(0, H_BLOCK)
            acc = tl.zeros((BLOCK_P, H_BLOCK), dtype=tl.float32)
            accg = tl.zeros((BLOCK_P, H_BLOCK), dtype=tl.float32)

            for c0 in range(0, C, BLOCK_C):
                c_offs = c0 + tl.arange(0, BLOCK_C)
                c_ok = c_offs < C

                x = tl.load(
                    x_ptr + offs_p[:, None] * C + c_offs[None, :],
                    mask=p_ok[:, None] & c_ok[None, :],
                    other=0.0,
                ).to(tl.float32)

                nw = tl.load(norm_w_ptr + c_offs, mask=c_ok, other=0.0).to(tl.float32)
                nb = tl.load(norm_b_ptr + c_offs, mask=c_ok, other=0.0).to(tl.float32)
                a = ((x - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]).to(tl.bfloat16)

                w = tl.load(
                    w_r_ptr + c_offs[:, None] * H + h_offs[None, :],
                    mask=c_ok[:, None],
                    other=0.0,
                )
                acc = tl.dot(a, w, acc)

                wg = tl.load(
                    w_rg_ptr + c_offs[:, None] * H + h_offs[None, :],
                    mask=c_ok[:, None],
                    other=0.0,
                )
                accg = tl.dot(a, wg, accg)

            val = acc * mask_vals[:, None] * tl.sigmoid(accg)
            val_t = tl.trans(val.to(tl.bfloat16))
            st_ptr = r_ptr + (offs_b[None, :] * H + h_offs[:, None]) * N2 + local_p[None, :]
            tl.store(st_ptr, val_t, mask=p_ok[None, :])

    else:
        for hb in tl.static_range(NUM_HB):
            h_offs = hb * H_BLOCK + tl.arange(0, H_BLOCK)
            acc = tl.zeros((BLOCK_P, H_BLOCK), dtype=tl.float32)

            for c0 in range(0, C, BLOCK_C):
                c_offs = c0 + tl.arange(0, BLOCK_C)
                c_ok = c_offs < C

                x = tl.load(
                    x_ptr + offs_p[:, None] * C + c_offs[None, :],
                    mask=p_ok[:, None] & c_ok[None, :],
                    other=0.0,
                ).to(tl.float32)

                nw = tl.load(norm_w_ptr + c_offs, mask=c_ok, other=0.0).to(tl.float32)
                nb = tl.load(norm_b_ptr + c_offs, mask=c_ok, other=0.0).to(tl.float32)
                a = ((x - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]).to(tl.bfloat16)

                w = tl.load(
                    w_og_ptr + c_offs[:, None] * H + h_offs[None, :],
                    mask=c_ok[:, None],
                    other=0.0,
                )
                acc = tl.dot(a, w, acc)

            val = tl.sigmoid(acc)
            val_t = tl.trans(val.to(tl.bfloat16))
            st_ptr = og_ptr + (offs_b[None, :] * H + h_offs[:, None]) * N2 + local_p[None, :]
            tl.store(st_ptr, val_t, mask=p_ok[None, :])


@triton.jit
def _trimul_bmm_kernel(
    l_ptr,
    r_ptr,
    s_ptr,

    N,
    N2,
    batch,

    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    EVEN: tl.constexpr,
):
    pid = tl.program_id(0)

    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)

    bid = pid // (num_i * num_j)
    rem = pid % (num_i * num_j)
    pid_i = rem // num_j
    pid_j = rem % num_j

    base = bid.to(tl.int64) * N2

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    if EVEN:
        for k0 in range(0, N, BLOCK_K):
            offs_k = k0 + tl.arange(0, BLOCK_K)

            a = tl.load(l_ptr + base + offs_i[:, None] * N + offs_k[None, :])
            b = tl.load(r_ptr + base + offs_j[:, None] * N + offs_k[None, :])

            acc = tl.dot(a, tl.trans(b), acc)

        tl.store(s_ptr + base + offs_i[:, None] * N + offs_j[None, :], acc.to(tl.bfloat16))
    else:
        i_ok = offs_i < N
        j_ok = offs_j < N

        for k0 in range(0, N, BLOCK_K):
            offs_k = k0 + tl.arange(0, BLOCK_K)
            k_ok = offs_k < N

            a = tl.load(
                l_ptr + base + offs_i[:, None] * N + offs_k[None, :],
                mask=i_ok[:, None] & k_ok[None, :],
                other=0.0,
            )
            b = tl.load(
                r_ptr + base + offs_j[:, None] * N + offs_k[None, :],
                mask=j_ok[:, None] & k_ok[None, :],
                other=0.0,
            )

            acc = tl.dot(a, tl.trans(b), acc)

        tl.store(
            s_ptr + base + offs_i[:, None] * N + offs_j[None, :],
            acc.to(tl.bfloat16),
            mask=i_ok[:, None] & j_ok[None, :],
        )


@triton.jit
def _trimul_final_kernel(
    s_ptr,
    og_ptr,
    ln_w_ptr,
    ln_b_ptr,
    w_ptr,
    out_ptr,

    N2,
    M,
    DIM,
    H,
    EPS,

    H_BLOCK: tl.constexpr,
    NUM_HB: tl.constexpr,
    BLOCK_P: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    p_ok = offs_p < M
    offs_b = offs_p // N2
    local_p = offs_p - offs_b * N2

    if NUM_HB == 1:
        h_offs = tl.arange(0, H_BLOCK)
        base_s = (offs_b[None, :] * H + h_offs[:, None]) * N2 + local_p[None, :]

        s = tl.load(s_ptr + base_s, mask=p_ok[None, :], other=0.0).to(tl.float32)

        mean = tl.sum(s, axis=0) / H
        sd = s - mean[None, :]
        var = tl.sum(sd * sd, axis=0) / H
        var = tl.maximum(var, 0.0)
        rstd = tl.rsqrt(var + EPS)

        gamma = tl.load(ln_w_ptr + h_offs)
        beta = tl.load(ln_b_ptr + h_offs)

        v = sd * rstd[None, :] * gamma[:, None] + beta[None, :]

        og = tl.load(og_ptr + base_s, mask=p_ok[None, :], other=0.0).to(tl.float32)
        v = v * tl.sigmoid(og)

        a = tl.trans(v).to(tl.bfloat16)

        for c0 in range(0, DIM, BLOCK_C):
            c_offs = c0 + tl.arange(0, BLOCK_C)
            c_ok = c_offs < DIM

            w = tl.load(
                w_ptr + h_offs[:, None] * DIM + c_offs[None, :],
                mask=c_ok[None, :],
                other=0.0,
            )
            acc = tl.dot(a, w)

            tl.store(
                out_ptr + offs_p[:, None] * DIM + c_offs[None, :],
                acc.to(tl.float32),
                mask=p_ok[:, None] & c_ok[None, :],
            )
    else:
        sums = tl.zeros((BLOCK_P,), dtype=tl.float32)
        sumsq = tl.zeros((BLOCK_P,), dtype=tl.float32)

        for hb in tl.static_range(NUM_HB):
            h_offs = hb * H_BLOCK + tl.arange(0, H_BLOCK)
            base_s = (offs_b[None, :] * H + h_offs[:, None]) * N2 + local_p[None, :]
            s = tl.load(s_ptr + base_s, mask=p_ok[None, :], other=0.0).to(tl.float32)
            sums += tl.sum(s, axis=0)
            sumsq += tl.sum(s * s, axis=0)

        mean = sums / H
        var = sumsq / H - mean * mean
        var = tl.maximum(var, 0.0)
        rstd = tl.rsqrt(var + EPS)

        for c0 in range(0, DIM, BLOCK_C):
            c_offs = c0 + tl.arange(0, BLOCK_C)
            c_ok = c_offs < DIM

            acc = tl.zeros((BLOCK_P, BLOCK_C), dtype=tl.float32)

            for hb in tl.static_range(NUM_HB):
                h_offs = hb * H_BLOCK + tl.arange(0, H_BLOCK)
                base_s = (offs_b[None, :] * H + h_offs[:, None]) * N2 + local_p[None, :]

                s = tl.load(s_ptr + base_s, mask=p_ok[None, :], other=0.0).to(tl.float32)

                gamma = tl.load(ln_w_ptr + h_offs)
                beta = tl.load(ln_b_ptr + h_offs)

                sd = s - mean[None, :]
                v = sd * rstd[None, :] * gamma[:, None] + beta[None, :]

                og = tl.load(og_ptr + base_s, mask=p_ok[None, :], other=0.0).to(tl.float32)
                v = v * tl.sigmoid(og)

                a = tl.trans(v).to(tl.bfloat16)

                w = tl.load(
                    w_ptr + h_offs[:, None] * DIM + c_offs[None, :],
                    mask=c_ok[None, :],
                    other=0.0,
                )
                acc = tl.dot(a, w, acc)

            tl.store(
                out_ptr + offs_p[:, None] * DIM + c_offs[None, :],
                acc.to(tl.float32),
                mask=p_ok[:, None] & c_ok[None, :],
            )


def _reference_tri_mul(x: torch.Tensor, mask, weights, config) -> torch.Tensor:
    import torch.nn.functional as F

    dim = int(config.get("dim", x.shape[-1]))
    hidden = int(config.get("hidden_dim", 128))

    x = x.to(torch.float32)

    def lin(t, w):
        return F.linear(t, w.to(t))

    x = F.layer_norm(x, (dim,), weights["norm.weight"].to(x), weights["norm.bias"].to(x))

    left = lin(x, weights["left_proj.weight"])
    right = lin(x, weights["right_proj.weight"])

    if mask is not None:
        m = mask.to(left)
        if m.dim() == 4 and m.shape[-1] == 1:
            m = m.squeeze(-1)
        left = left * m.unsqueeze(-1)
        right = right * m.unsqueeze(-1)

    left = left * torch.sigmoid(lin(x, weights["left_gate.weight"]))
    right = right * torch.sigmoid(lin(x, weights["right_gate.weight"]))

    out = torch.einsum("bikd,bjkd->bij d", left, right)

    out = F.layer_norm(out, (hidden,), weights["to_out_norm.weight"].to(out), weights["to_out_norm.bias"].to(out))
    out = out * torch.sigmoid(lin(x, weights["out_gate.weight"]))
    out = lin(out, weights["to_out.weight"])

    return out.to(torch.float32)


def custom_kernel(data):
    if len(data) == 4:
        x, mask, weights, config = data
    elif len(data) == 3:
        x, mask_or_weights, config_or_none = data
        if isinstance(mask_or_weights, dict):
            x, weights, config = data
            mask = None
        else:
            x, mask, weights, config = data, mask_or_weights, config_or_none, None
    else:
        x = data[0]
        if len(data) >= 4 and isinstance(data[1], torch.Tensor):
            mask = data[1]
        else:
            mask = None
        weights = data[-2]
        config = data[-1]

    x = x.contiguous()

    if mask is not None:
        mask = mask.contiguous()
        if mask.dim() == 4 and mask.shape[-1] == 1:
            mask = mask.squeeze(-1)

    B, N, _, C = x.shape
    H = int(config.get("hidden_dim", 128))
    DIM = int(config.get("dim", C))
    device = x.device

    # The optimized path is tuned around 128-wide hidden blocks.
    # If a future shape does not match, use the correct PyTorch path.
    if H <= 0 or H % 128 != 0:
        return _reference_tri_mul(x, mask, weights, config)

    dt = torch.bfloat16

    norm_w = weights["norm.weight"].to(device, torch.float32).contiguous()
    norm_b = weights["norm.bias"].to(device, torch.float32).contiguous()

    w_l = weights["left_proj.weight"].to(device, dt).t().contiguous()
    w_lg = weights["left_gate.weight"].to(device, dt).t().contiguous()
    w_r = weights["right_proj.weight"].to(device, dt).t().contiguous()
    w_rg = weights["right_gate.weight"].to(device, dt).t().contiguous()
    w_og = weights["out_gate.weight"].to(device, dt).t().contiguous()

    out_ln_w = weights["to_out_norm.weight"].to(device, torch.float32).contiguous()
    out_ln_b = weights["to_out_norm.bias"].to(device, torch.float32).contiguous()
    w_out_t = weights["to_out.weight"].to(device, dt).t().contiguous()

    N2 = N * N
    M = B * N2
    NUM_HB = H // 128

    L = torch.empty((B, H, N, N), dtype=dt, device=device)
    R = torch.empty((B, H, N, N), dtype=dt, device=device)
    OG = torch.empty((B, H, N, N), dtype=dt, device=device)
    S = torch.empty((B, H, N, N), dtype=dt, device=device)

    # 1) Fused LN + projections + gates/masks.
    PROJ_P = 64
    grid_proj = (triton.cdiv(M, PROJ_P) * 3,)
    _trimul_proj_kernel[grid_proj](
        x,
        norm_w,
        norm_b,
        w_l,
        w_lg,
        w_r,
        w_rg,
        w_og,
        L,
        R,
        OG,
        mask if mask is not None else x,
        M,
        N2,
        C,
        H,
        1e-5,
        HAS_MASK=(mask is not None),
        BLOCK_P=PROJ_P,
        BLOCK_C=128,
        H_BLOCK=128,
        NUM_HB=NUM_HB,
        num_warps=4,
        num_stages=2,
    )

    # 2) Batched TriMul core.
    TRI_I = 128
    TRI_J = 128
    TRI_K = 64
    even = (N % TRI_I == 0) and (N % TRI_J == 0) and (N % TRI_K == 0)
    batch = B * H
    grid_bmm = (batch * triton.cdiv(N, TRI_I) * triton.cdiv(N, TRI_J),)

    _trimul_bmm_kernel[grid_bmm](
        L,
        R,
        S,
        N,
        N2,
        batch,
        BLOCK_I=TRI_I,
        BLOCK_J=TRI_J,
        BLOCK_K=TRI_K,
        EVEN=even,
        num_warps=8,
        num_stages=3,
    )

    # Allow the allocator to reuse L/R memory before allocating the final output.
    del L
    del R

    # 3) Fused final LayerNorm + out-gate + projection.
    FINAL_P = 64
    FINAL_C = 128
    out = torch.empty((B, N, N, DIM), dtype=torch.float32, device=device)
    grid_final = (triton.cdiv(M, FINAL_P),)

    _trimul_final_kernel[grid_final](
        S,
        OG,
        out_ln_w,
        out_ln_b,
        w_out_t,
        out,
        N2,
        M,
        DIM,
        H,
        1e-5,
        H_BLOCK=128,
        NUM_HB=NUM_HB,
        BLOCK_P=FINAL_P,
        BLOCK_C=FINAL_C,
        num_warps=4,
        num_stages=2,
    )

    return out
