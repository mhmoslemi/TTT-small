Let's implement LN kernel:

@triton.jit
def layernorm_bf16_kernel(x_ptr, out_ptr, n_tokens, C, w_ptr, b_ptr, eps, BLOCK_M: tl.constexpr, BLOCK_C: tl.constexpr, C_PAD: tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid*BLOCK_M + tl.arange(0,BLOCK_M)
    mask_m = offs_m < n_tokens
    offs_c = tl.arange(0,BLOCK_C)
    mask_c = offs_c < C
    x_ptrs = x_ptr + offs_m[:,None]*C + offs_c[None,:]
    x = tl.load(x_ptrs, mask=mask_m[:,None] & mask_c[None,:], other=0.0).to(tl.float32)
    cvalid = mask_c[None,:].to(tl.float32)
    # sum over valid C: since padding zeros, sum(x*valid)
    sum = tl.sum(x * cvalid, axis=1)
    mean = sum / C
    diff = x - mean[:,None]
    diff2 = tl.where(mask_c[None,:], diff*diff, 0.0)
    var = tl.sum(diff2, axis=1) / C
    rstd = 1.0 / tl.sqrt(var + eps)
    w = tl.load(w_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
    bb = tl.load(b_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
    y = diff * rstd[:,None] * w[None,:] + bb[None,:]
    y = tl.where(mask_c[None,:], y, 0.0)
    out_ptrs = out_ptr + offs_m[:,None]*C + offs_c[None,:]
    tl.store(out_ptrs, y.to(tl.bfloat16), mask=mask_m[:,None] & mask_c[None,:])

import torch

@triton.jit
def layernorm_kernel(x_ptr, out_ptr, n_tokens, C, w_ptr, b_ptr, eps,
                     BLOCK_M: tl.constexpr, BLOCK_C: tl.constexpr):
    pid=tl.program_id(0)
    offs_m=pid*BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m=offs_m < n_tokens
    offs_c=tl.arange(0, BLOCK_C)
    mask_c=offs_c < C
    c_mask_2d = mask_m[:,None] & mask_c[None,:]
    x_ptrs = x_ptr + offs_m[:,None]*C + offs_c[None,:]
    x = tl.load(x_ptrs, mask=c_mask_2d, other=0.0)
    # ensure fp32
    x = x.to(tl.float32)
    valid = mask_c[None,:].to(tl.float32)
    # if mask_m false, rows zeros; mean zero; output not stored.
    s = tl.sum(x * valid, axis=1)
    mean = s / C
    diff = x - mean[:,None]
    diff_sq = tl.where(mask_c[None,:], diff*diff, 0.0)
    var = tl.sum(diff_sq, axis=1) / C
    rstd = 1.0 / tl.sqrt(var + eps)
    w = tl.load(w_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
    bb = tl.load(b_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
    y = diff * rstd[:,None] * w[None,:] + bb[None,:]
    y = tl.where(mask_c[None,:], y, 0.0)
    out_ptrs = out_ptr + offs_m[:,None]*C + offs_c[None,:]
    tl.store(out_ptrs, y.to(tl.bfloat16), mask=c_mask_2d)

def _layernorm_bf16_kernel(
    x_ptr,
    out_ptr,
    n_tokens,
    C,
    w_ptr,
    b_ptr,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_c = tl.arange(0, BLOCK_C)

    mask_m = offs_m < n_tokens
    mask_c = offs_c < C
    mask_2d = mask_m[:, None] & mask_c[None, :]

    x_ptrs = x_ptr + offs_m[:, None] * C + offs_c[None, :]
    x = tl.load(x_ptrs, mask=mask_2d, other=0.0).to(tl.float32)

    mean = tl.sum(x, axis=1) / C
    diff = x - mean[:, None]
    diff_sq = tl.where(mask_c[None, :], diff * diff, 0.0)
    var = tl.sum(diff_sq, axis=1) / C
    rstd = 1.0 / tl.sqrt(var + eps)

    w = tl.load(w_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
    b = tl.load(b_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)

    y = diff * rstd[:, None] * w[None, :] + b[None, :]
    y = tl.where(mask_c[None, :], y, 0.0)

    out_ptrs = out_ptr + offs_m[:, None] * C + offs_c[None, :]
    tl.store(out_ptrs, y.to(tl.bfloat16), mask=mask_2d)


@triton.jit
def _act_permute_kernel(
    y_ptr,
    mask_ptr,
    out_ptr,
    S,
    KB,
    H: tl.constexpr,
    P: tl.constexpr,
    PROJ_COL: tl.constexpr,
    GATE_COL: tl.constexpr,
    BM: tl.constexpr,
    NO_MASK: tl.constexpr,
    MASK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    pid_i = pid // KB
    pid_k = pid % KB

    i = pid_i % S
    b = pid_i // S

    k0 = pid_k * BM
    offs_k = k0 + tl.arange(0, BM)
    offs_h = tl.arange(0, H)

    base_y = ((b * S + i) * S + k0) * P
    proj_ptrs = y_ptr + base_y + PROJ_COL * H + offs_k[:, None] * P + offs_h[None, :]
    gate_ptrs = y_ptr + base_y + GATE_COL * H + offs_k[:, None] * P + offs_h[None, :]

    if MASK_K:
        kmask = offs_k < S
        proj = tl.load(proj_ptrs, mask=kmask[:, None], other=0.0)
        gate = tl.load(gate_ptrs, mask=kmask[:, None], other=0.0)
    else:
        proj = tl.load(proj_ptrs)
        gate = tl.load(gate_ptrs)

    proj_f = proj.to(tl.float32)
    gate_f = gate.to(tl.float32)
    sig = 1.0 / (1.0 + tl.exp(-gate_f))

    if NO_MASK:
        val = proj_f * sig
    else:
        if MASK_K:
            mval = tl.load(mask_ptr + (b * S + i) * S + offs_k, mask=(offs_k < S), other=0.0)
        else:
            mval = tl.load(mask_ptr + (b * S + i) * S + offs_k)
        mval = mval.to(tl.float32)
        val = proj_f * sig * mval

    val_t = tl.trans(val.to(tl.bfloat16))

    out_ptrs = out_ptr + (b * S + i) * H * S + offs_h[:, None] * S + offs_k[None, :]
    if MASK_K:
        tl.store(out_ptrs, val_t, mask=kmask[None, :])
    else:
        tl.store(out_ptrs, val_t)


@triton.jit
def _post_kernel(
    c_ptr,
    gate_ptr,
    norm_w_ptr,
    norm_b_ptr,
    wout_ptr,
    out_ptr,
    S,
    C,
    NB_J,
    NB_C,
    H: tl.constexpr,
    P: tl.constexpr,
    GATE_COL: tl.constexpr,
    BJ: tl.constexpr,
    BC: tl.constexpr,
    eps,
    MASK_J: tl.constexpr,
    MASK_C: tl.constexpr,
):
    pid = tl.program_id(0)

    pid_ij = pid // NB_J
    jb = pid % NB_J

    i = pid_ij % S
    b = pid_ij // S

    offs_j = jb * BJ + tl.arange(0, BJ)
    offs_h = tl.arange(0, H)

    # Load C as [H, BJ], then transpose to [BJ, H].  This is coalesced over j.
    c_ptrs = (
        c_ptr
        + b * H * S * S
        + offs_h[:, None] * (S * S)
        + i * S
        + offs_j[None, :]
    )
    if MASK_J:
        c = tl.load(c_ptrs, mask=(offs_j[None, :] < S), other=0.0)
    else:
        c = tl.load(c_ptrs)

    c = tl.trans(c).to(tl.float32)

    # Load out_gate from packed projection output.
    gate_base = ((b * S + i) * S + offs_j[:, None]) * P + GATE_COL * H
    gate_ptrs = gate_base + gate_ptr + offs_h[None, :]
    if MASK_J:
        gate = tl.load(gate_ptrs, mask=(offs_j[:, None] < S), other=0.0)
    else:
        gate = tl.load(gate_ptrs)
    gate = gate.to(tl.float32)
    gate_sig = 1.0 / (1.0 + tl.exp(-gate))

    # Final LayerNorm over hidden dimension H.
    norm_w = tl.load(norm_w_ptr + offs_h).to(tl.float32)
    norm_b = tl.load(norm_b_ptr + offs_h).to(tl.float32)

    mean = tl.sum(c, axis=1) / H
    x = c - mean[:, None]
    var = tl.sum(x * x, axis=1) / H
    rstd = 1.0 / tl.sqrt(var + eps)

    y = x * rstd[:, None] * norm_w[None, :] + norm_b[None, :]
    y = y * gate_sig
    y_bf = y.to(tl.bfloat16)

    # Final linear projection, [BJ, H] @ [H, C], stored as float32.
    for cb in range(0, NB_C):
        offs_c = cb * BC + tl.arange(0, BC)

        w_ptrs = wout_ptr + offs_h[:, None] * C + offs_c[None, :]
        if MASK_C:
            w = tl.load(w_ptrs, mask=(offs_c[None, :] < C), other=0.0)
        else:
            w = tl.load(w_ptrs)

        o = tl.dot(y_bf, w)

        out_ptrs = (
            out_ptr
            + ((b * S + i) * S + offs_j[:, None]) * C
            + cb * BC
            + offs_c[None, :]
        )

        if MASK_J:
            if MASK_C:
                store_mask = (offs_j[:, None] < S) & (offs_c[None, :] < C)
            else:
                store_mask = offs_j[:, None] < S
            tl.store(out_ptrs, o, mask=store_mask)
        else:
            if MASK_C:
                tl.store(out_ptrs, o, mask=(offs_c[None, :] < C))
            else:
                tl.store(out_ptrs, o)


def custom_kernel(data):
    x, mask, weights, config = data

    B, S, _, C = x.shape
    H = int(config["hidden_dim"])
    device = x.device
    eps = 1e-5

    x_src = x.contiguous()
    n_tokens = B * S * S

    # 1) LayerNorm(input) -> bf16.
    x_norm = torch.empty(x_src.shape, dtype=torch.bfloat16, device=device)

    if C <= 128:
        BLOCK_C_LN = 128
        BLOCK_M_LN = 64
    elif C <= 512:
        BLOCK_C_LN = 512
        BLOCK_M_LN = 32
    else:
        BLOCK_C_LN = 1024
        BLOCK_M_LN = 16

    grid_ln = (triton.cdiv(n_tokens, BLOCK_M_LN),)
    _layernorm_bf16_kernel[grid_ln](
        x_src,
        x_norm,
        n_tokens,
        C,
        weights["norm.weight"],
        weights["norm.bias"],
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_C=BLOCK_C_LN,
        num_warps=8,
    )

    # 2) Packed projection: [tokens, C] x [C, 5H] -> [tokens, 5H].
    packed_proj_weights = torch.cat(
        [
            weights["left_proj.weight"].t(),
            weights["right_proj.weight"].t(),
            weights["left_gate.weight"].t(),
            weights["right_gate.weight"].t(),
            weights["out_gate.weight"].t(),
        ],
        dim=1,
    ).to(torch.bfloat16).contiguous()

    y = torch.mm(x_norm.reshape(n_tokens, C), packed_proj_weights)
    del x_norm

    # 3) Apply gates / mask and permute left/right to [B, H, S, S].
    ACT_BM = 64
    ACT_KB = triton.cdiv(S, ACT_BM)
    ACT_MASK_K = (S % ACT_BM != 0)
    P = 5 * H

    L = torch.empty((B, H, S, S), dtype=torch.bfloat16, device=device)
    R = torch.empty((B, H, S, S), dtype=torch.bfloat16, device=device)

    no_mask = mask is None
    if no_mask:
        mask_arg = weights["norm.bias"].to(torch.float32).contiguous()
    else:
        m = mask
        if m.dim() == 4:
            m = m.squeeze(-1)
        mask_arg = m.to(torch.float32).contiguous()

    grid_act = (B * S * ACT_KB,)
    _act_permute_kernel[grid_act](
        y,
        mask_arg,
        L,
        S,
        ACT_KB,
        H=H,
        P=P,
        PROJ_COL=0,
        GATE_COL=2,
        BM=ACT_BM,
        NO_MASK=True if no_mask else False,
        MASK_K=ACT_MASK_K,
        num_warps=8,
    )

    _act_permute_kernel[grid_act](
        y,
        mask_arg,
        R,
        S,
        ACT_KB,
        H=H,
        P=P,
        PROJ_COL=1,
        GATE_COL=3,
        BM=ACT_BM,
        NO_MASK=True if no_mask else False,
        MASK_K=ACT_MASK_K,
        num_warps=8,
    )

    # 4) TriMul einsum as batched matmuls:
    #    C[b,h,i,j] = sum_k L[b,h,i,k] * R[b,h,j,k]
    batch_count = B * H
    L3 = L.reshape(batch_count, S, S)
    R3 = R.reshape(batch_count, S, S)
    C_t3 = torch.bmm(L3, R3.transpose(1, 2))
    C_t = C_t3.view(B, H, S, S)
    del L, R, L3, R3

    # 5) Final fused LayerNorm + out_gate + to_out.
    w_out_t = weights["to_out.weight"].t().contiguous().to(torch.bfloat16)
    out = torch.empty((B, S, S, C), dtype=torch.float32, device=device)

    POST_BJ = 128
    POST_BC = 128
    NB_J = triton.cdiv(S, POST_BJ)
    NB_C = triton.cdiv(C, POST_BC)

    grid_post = (B * S * NB_J,)
    _post_kernel[grid_post](
        C_t,
        y,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        w_out_t,
        out,
        S,
        C,
        NB_J,
        NB_C,
        H=H,
        P=P,
        GATE_COL=4,
        BJ=POST_BJ,
        BC=POST_BC,
        eps=eps,
        MASK_J=(S % POST_BJ != 0),
        MASK_C=(C % POST_BC != 0),
        num_warps=8,
    )

    return out
