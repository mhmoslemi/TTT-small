```python
"""
Fused Triton forward pass for the outgoing TriMul operation.

Pipeline:
1) One fused kernel:
   - LayerNorm over input dim C.
   - five projections: left, right, left_gate, right_gate, out_gate.
   - apply sigmoid gates and the spatial mask to left/right.
   - write left/right in a transposed channel-first layout for efficient GEMM,
     and write out_gate in a layout convenient for the final layer-norm/projection.

2) A Triton batched/GEMM-style kernel computes, per hidden channel h,
     O[b, i, h, j] = sum_k L[b, h, i, k] * R[b, h, j, k].
   The output is stored as [B, N, H, N] to keep the last dimension contiguous.

3) A final fused kernel:
   - LayerNorm over hidden dim H,
   - multiply by out_gate,
   - project back to model dim C.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}


def _prepare_weights(weights, device, dt):
    """Convert/cache weight tensors to the dtypes needed by the kernels."""
    # A conservative key.  It is meant to save repeated .to() calls in
    # benchmark loops, not to act as a semantic cache.
    key_parts = []
    for k in sorted(weights.keys()):
        v = weights[k]
        key_parts.append((k, v.data_ptr(), tuple(v.shape), v.dtype, v.is_contiguous()))
    key = (str(device), tuple(key_parts))

    cached = _WEIGHT_CACHE.get(key)
    if cached is not None:
        return cached

    fp32 = torch.float32
    prep = {
        "norm_w": weights["norm.weight"].to(fp32).contiguous(),
        "norm_b": weights["norm.bias"].to(fp32).contiguous(),
        "left_w": weights["left_proj.weight"].to(dt).contiguous(),
        "right_w": weights["right_proj.weight"].to(dt).contiguous(),
        "left_gate_w": weights["left_gate.weight"].to(dt).contiguous(),
        "right_gate_w": weights["right_gate.weight"].to(dt).contiguous(),
        "out_gate_w": weights["out_gate.weight"].to(dt).contiguous(),
        "to_out_norm_w": weights["to_out_norm.weight"].to(fp32).contiguous(),
        "to_out_norm_b": weights["to_out_norm.bias"].to(fp32).contiguous(),
        "to_out_w": weights["to_out.weight"].to(dt).contiguous(),
    }
    _WEIGHT_CACHE[key] = prep
    return prep


@triton.jit
def _projection_kernel(
    x_ptr,
    mask_ptr,
    norm_w_ptr,
    norm_b_ptr,
    wl_ptr,
    wr_ptr,
    wlg_ptr,
    wrg_ptr,
    wog_ptr,
    L_ptr,
    R_ptr,
    G_ptr,
    S,
    N,
    H,
    C,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    H_BLOCKS: tl.constexpr,
    CHUNK_A: tl.constexpr,
    CHUNK_B: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """
    Fused LayerNorm + left/right/gates projection + mask/gate application.

    Layouts:
      x: [B, N, N, C], viewed as [S, C]
      L: [B, H, N, N]  (channel-first; last dim is k)
      R: [B, H, N, N]
      G: [B, N, H, N]  (out_gate, row (i,j), hidden h, last dim is j)
    """
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    row_ok = offs_m < S
    r = offs_m.to(tl.int64)

    N64 = N.to(tl.int64)
    H64 = H.to(tl.int64)
    C64 = C.to(tl.int64)
    Cf = C.to(tl.float32)

    NN = N64 * N64
    HNN = H64 * NN
    NHN = N64 * H64 * N64

    # Load normalized input in one or two K chunks.  Padding is zeroed and the
    # corresponding normalized values become zero because norm_w/bias are zero.
    x_row = r * C64

    c0 = tl.arange(0, CHUNK_A)
    c0_ok = c0 < C
    x0 = tl.load(
        x_ptr + x_row[:, None] + c0[None, :],
        mask=row_ok[:, None] & c0_ok[None, :],
        other=0.0,
    ).to(tl.float32)

    sum0 = tl.sum(x0, axis=1)
    if CHUNK_B > 0:
        c1 = tl.arange(0, CHUNK_B) + CHUNK_A
        c1_ok = c1 < C
        x1 = tl.load(
            x_ptr + x_row[:, None] + c1[None, :],
            mask=row_ok[:, None] & c1_ok[None, :],
            other=0.0,
        ).to(tl.float32)
        sumc = sum0 + tl.sum(x1, axis=1)
    else:
        sumc = sum0

    mean = sumc / Cf
    diff0 = x0 - mean[:, None]

    if CHUNK_B > 0:
        diff1 = x1 - mean[:, None]
        sq = tl.sum(diff0 * diff0, axis=1) + tl.sum(diff1 * diff1, axis=1)
    else:
        sq = tl.sum(diff0 * diff0, axis=1)

    inv_std = tl.rsqrt(sq / Cf + 1e-5)

    nw0 = tl.load(norm_w_ptr + c0, mask=c0_ok, other=0.0).to(tl.float32)
    nb0 = tl.load(norm_b_ptr + c0, mask=c0_ok, other=0.0).to(tl.float32)
    norm0 = (diff0 * inv_std[:, None] * nw0[None, :] + nb0[None, :]).to(tl.bfloat16)

    if CHUNK_B > 0:
        nw1 = tl.load(norm_b_ptr + c1, mask=c1_ok, other=0.0).to(tl.float32)
        nb1 = tl.load(norm_b_ptr + c1, mask=c1_ok, other=0.0).to(tl.float32)
        norm1 = (diff1 * inv_std[:, None] * nw1[None, :] + nb1[None, :]).to(tl.bfloat16)

    if HAS_MASK:
        mvals = tl.load(mask_ptr + r, mask=row_ok, other=0.0).to(tl.float32)
    else:
        mvals = tl.full((BLOCK_M,), 1.0, tl.float32)

    # Row index decomposition for the flattened [B, N, N] rows.
    b = (r // (N64 * N64)).to(tl.int64)
    i = ((r // N64) % N64).to(tl.int64)
    v = (r % N64).to(tl.int64)

    h0 = tl.arange(0, BLOCK_H)

    for ob in tl.static_range(H_BLOCKS):
        h = ob * BLOCK_H + h0
        h_ok = h < H
        h_i64 = h.to(tl.int64)

        # Left projection and left gate.
        wl0 = tl.load(
            wl_ptr + h_i64[:, None] * C64 + c0[None, :],
            mask=h_ok[:, None] & c0_ok[None, :],
            other=0.0,
        ).to(tl.bfloat16)
        wlg0 = tl.load(
            wlg_ptr + h_i64[:, None] * C64 + c0[None, :],
            mask=h_ok[:, None] & c0_ok[None, :],
            other=0.0,
        ).to(tl.bfloat16)

        y_left = tl.dot(norm0, tl.trans(wl0))
        y_lgate = tl.dot(norm0, tl.trans(wlg0))

        if CHUNK_B > 0:
            wl1 = tl.load(
                wl_ptr + h_i64[:, None] * C64 + c1[None, :],
                mask=h_ok[:, None] & c1_ok[None, :],
                other=0.0,
            ).to(tl.bfloat16)
            wlg1 = tl.load(
                wlg_ptr + h_i64[:, None] * C64 + c1[None, :],
                mask=h_ok[:, None] & c1_ok[None, :],
                other=0.0,
            ).to(tl.bfloat16)

            y_left += tl.dot(norm1, tl.trans(wl1))
            y_lgate += tl.dot(norm1, tl.trans(wlg1))

        val_left = (y_left * tl.sigmoid(y_lgate)) * mvals[:, None]

        L_ptrs = (
            L_ptr
            + b[None, :] * HNN
            + i[None, :] * N64
            + v[None, :]
            + h_i64[:, None] * NN
        )
        tl.store(
            L_ptrs,
            tl.trans(val_left).to(tl.bfloat16),
            mask=h_ok[:, None] & row_ok[None, :],
        )

        # Right projection and right gate.
        wr0 = tl.load(
            wr_ptr + h_i64[:, None] * C64 + c0[None, :],
            mask=h_ok[:, None] & c0_ok[None, :],
            other=0.0,
        ).to(tl.bfloat16)
        wrg0 = tl.load(
            wrg_ptr + h_i64[:, None] * C64 + c0[None, :],
            mask=h_ok[:, None] & c0_ok[None, :],
            other=0.0,
        ).to(tl.bfloat16)

        y_right = tl.dot(norm0, tl.trans(wr0))
        y_rgate = tl.dot(norm0, tl.trans(wrg0))

        if CHUNK_B > 0:
            wr1 = tl.load(
                wr_ptr + h_i64[:, None] * C64 + c1[None, :],
                mask=h_ok[:, None] & c1_ok[None, :],
                other=0.0,
            ).to(tl.bfloat16)
            wrg1 = tl.load(
                wrg_ptr + h_i64[:, None] * C64 + c1[None, :],
                mask=h_ok[:, None] & c1_ok[None, :],
                other=0.0,
            ).to(tl.bfloat16)

            y_right += tl.dot(norm1, tl.trans(wr1))
            y_rgate += tl.dot(norm1, tl.trans(wrg1))

        val_right = (y_right * tl.sigmoid(y_rgate)) * mvals[:, None]

        R_ptrs = (
            R_ptr
            + b[None, :] * HNN
            + i[None, :] * N64
            + v[None, :]
            + h_i64[:, None] * NN
        )
        tl.store(
            R_ptrs,
            tl.trans(val_right).to(tl.bfloat16),
            mask=h_ok[:, None] & row_ok[None, :],
        )

        # Out gate, written to [B, N_i, H, N_j] where (i, v) is (i, j).
        wog0 = tl.load(
            wog_ptr + h_i64[:, None] * C64 + c0[None, :],
            mask=h_ok[:, None] & c0_ok[None, :],
            other=0.0,
        ).to(tl.bfloat16)
        y_og = tl.dot(norm0, tl.trans(wog0))

        if CHUNK_B > 0:
            wog1 = tl.load(
                wog_ptr + h_i64[:, None] * C64 + c1[None, :],
                mask=h_ok[:, None] & c1_ok[None, :],
                other=0.0,
            ).to(tl.bfloat16)
            y_og += tl.dot(norm1, tl.trans(wog1))

        val_og = tl.sigmoid(y_og)

        G_ptrs = (
            G_ptr
            + b[None, :] * NHN
            + i[None, :] * (H64 * N64)
            + v[None, :]
            + h_i64[:, None] * N64
        )
        tl.store(
            G_ptrs,
            tl.trans(val_og).to(tl.bfloat16),
            mask=h_ok[:, None] & row_ok[None, :],
        )


@triton.jit
def _core_kernel(
    L_ptr,
    R_ptr,
    O_ptr,
    N,
    H,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    N_ITERS,
    EVEN_N: tl.constexpr,
):
    """
    For each h:
      O[b, i, h, j] = sum_k L[b, h, i, k] * R[b, h, j, k]

    L/R are [B, H, N, N].  O is [B, N, H, N].
    """
    pid_j = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_bh = tl.program_id(2)

    N64 = N.to(tl.int64)
    H64 = H.to(tl.int64)
    NN = N64 * N64
    NHN = N64 * H64 * N64

    bh = pid_bh.to(tl.int64)
    b = bh // H64
    h = bh % H64

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)

    base = (b * H64 + h) * NN

    A_ptrs = L_ptr + base + offs_i[:, None] * N64 + offs_k[None, :]
    B_ptrs = R_ptr + base + offs_j[:, None] * N64 + offs_k[None, :]

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    if EVEN_N:
        for _ in range(0, N_ITERS):
            a = tl.load(A_ptrs)
            bmat = tl.load(B_ptrs)
            acc += tl.dot(a, tl.trans(bmat))
            A_ptrs += BLOCK_K
            B_ptrs += BLOCK_K
    else:
        i_ok = offs_i < N64
        j_ok = offs_j < N64
        for ki in range(0, N_ITERS):
            k0 = ki * BLOCK_K
            k_ok = (k0 + offs_k) < N64

            a = tl.load(
                A_ptrs,
                mask=i_ok[:, None] & k_ok[None, :],
                other=0.0,
            )
            bmat = tl.load(
                B_ptrs,
                mask=j_ok[:, None] & k_ok[None, :],
                other=0.0,
            )
            acc += tl.dot(a, tl.trans(bmat))

            A_ptrs += BLOCK_K
            B_ptrs += BLOCK_K

    O_ptrs = (
        O_ptr
        + b * NHN
        + h * N64
        + offs_i[:, None] * (H64 * N64)
        + offs_j[None, :]
    )

    if EVEN_N:
        tl.store(O_ptrs, acc.to(tl.bfloat16))
    else:
        i_ok = offs_i < N64
        j_ok = offs_j < N64
        tl.store(O_ptrs, acc.to(tl.bfloat16), mask=i_ok[:, None] & j_ok[None, :])


@triton.jit
def _final_kernel(
    O_ptr,
    G_ptr,
    W_ptr,
    nw_ptr,
    nb_ptr,
    out_ptr,
    N,
    H,
    C,
    BLOCK_J: tl.constexpr,
    BLOCK_C: tl.constexpr,
    BLOCK_H: tl.constexpr,
    H_BLOCKS: tl.constexpr,
    C_BLOCKS: tl.constexpr,
    SINGLE_H: tl.constexpr,
):
    """
    Final fused operation:
      out_norm = LayerNorm(O) over H
      gated = out_norm * out_gate
      output = gated @ to_out_weight^T

    O/G are [B, N, H, N].  Output is [B, N, N, C].
    """
    pid_j = tl.program_id(0)
    i = tl.program_id(1)
    b = tl.program_id(2)
    cb = tl.program_id(3)

    N64 = N.to(tl.int64)
    H64 = H.to(tl.int64)
    C64 = C.to(tl.int64)
    Hf = H.to(tl.float32)

    NHN = N64 * H64 * N64

    j_off = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    j_ok = j_off < N64

    base = b.to(tl.int64) * NHN + i.to(tl.int64) * (H64 * N64)
    out_row = (
        b.to(tl.int64) * N64 * N64
        + i.to(tl.int64) * N64
        + j_off
    ) * C64

    if SINGLE_H:
        h = tl.arange(0, BLOCK_H)
        h_ok = h < H
        h_i64 = h.to(tl.int64)

        valid = h_ok[:, None] & j_ok[None, :]

        o = tl.load(
            O_ptr + base + h_i64[:, None] * N64 + j_off[None, :],
            mask=valid,
            other=0.0,
        ).to(tl.float32)
        g = tl.load(
            G_ptr + base + h_i64[:, None] * N64 + j_off[None, :],
            mask=valid,
            other=0.0,
        ).to(tl.float32)

        o_valid = tl.where(valid, o, 0.0)
        mean = tl.sum(o_valid, axis=0) / Hf

        diff_valid = o_valid - mean[None, :]
        diff_valid = tl.where(valid, diff_valid, 0.0)
        var = tl.sum(diff_valid * diff_valid, axis=0) / Hf
        inv_std = tl.rsqrt(var + 1e-5)

        nw = tl.load(nw_ptr + h, mask=h_ok, other=0.0).to(tl.float32)
        nb = tl.load(nb_ptr + h, mask=h_ok, other=0.0).to(tl.float32)

        norm = ((diff_valid * inv_std[None, :]) * nw[None, :] + nb[None, :]) * g
        norm_t = tl.trans(norm.to(tl.bfloat16))

        for cbl in tl.static_range(C_BLOCKS):
            c = cbl * BLOCK_C + tl.arange(0, BLOCK_C)
            c_ok = c < C
            c_i64 = c.to(tl.int64)

            w = tl.load(
                W_ptr + c_i64[:, None] * H64 + h_i64[None, :],
                mask=c_ok[:, None] & h_ok[None, :],
                other=0.0,
            ).to(tl.bfloat16)

            acc = tl.dot(norm_t, tl.trans(w))

            out_ptrs = out_ptr + out_row[:, None] + c[None, :]
            tl.store(
                out_ptrs,
                acc.to(tl.float32),
                mask=j_ok[:, None] & c_ok[None, :],
            )
    else:
        # Multiple H blocks.  For simplicity and to keep register pressure low,
        # each program handles one C block.
        c = cb * BLOCK_C + tl.arange(0, BLOCK_C)
        c_ok = c < C
        c_i64 = c.to(tl.int64)

        sum_h = tl.zeros((BLOCK_J,), dtype=tl.float32)
        sq_h = tl.zeros((BLOCK_J,), dtype=tl.float32)

        for ob in tl.static_range(H_BLOCKS):
            h = ob * BLOCK_H + tl.arange(0, BLOCK_H)
            h_ok = h < H
            h_i64 = h.to(tl.int64)
            valid = h_ok[:, None] & j_ok[None, :]

            o = tl.load(
                O_ptr + base + h_i64[:, None] * N64 + j_off[None, :],
                mask=valid,
                other=0.0,
            ).to(tl.float32)
            o_valid = tl.where(valid, o, 0.0)

            sum_h += tl.sum(o_valid, axis=0)
            sq_h += tl.sum(o_valid * o_valid, axis=0)

        mean = sum_h / Hf
        var = sq_h / Hf - mean * mean
        var = tl.maximum(var, 0.0)
        inv_std = tl.rsqrt(var + 1e-5)

        acc = tl.zeros((BLOCK_J, BLOCK_C), dtype=tl.float32)

        for ob in tl.static_range(H_BLOCKS):
            h = ob * BLOCK_H + tl.arange(0, BLOCK_H)
            h_ok = h < H
            h_i64 = h.to(tl.int64)
            valid = h_ok[:, None] & j_ok[None, :]

            o = tl.load(
                O_ptr + base + h_i64[:, None] * N64 + j_off[None, :],
                mask=valid,
                other=0.0,
            ).to(tl.float32)
            g = tl.load(
                G_ptr + base + h_i64[:, None] * N64 + j_off[None, :],
                mask=valid,
                other=0.0,
            ).to(tl.float32)

            diff = o - mean[None, :]
            diff = tl.where(valid, diff, 0.0)

            nw = tl.load(nw_ptr + h, mask=h_ok, other=0.0).to(tl.float32)
            nb = tl.load(nb_ptr + h, mask=h_ok, other=0.0).to(tl.float32)

            norm = ((diff * inv_std[None, :]) * nw[None, :] + nb[None, :]) * g
            norm = tl.where(valid, norm, 0.0)
            norm_t = tl.trans(norm.to(tl.bfloat16))

            w = tl.load(
                W_ptr + c_i64[:, None] * H64 + h_i64[None, :],
                mask=c_ok[:, None] & h_ok[None, :],
                other=0.0,
            ).to(tl.bfloat16)

            acc += tl.dot(norm_t, tl.trans(w))

        out_ptrs = out_ptr + out_row[:, None] + c[None, :]
        tl.store(
            out_ptrs,
            acc.to(tl.float32),
            mask=j_ok[:, None] & c_ok[None, :],
        )


def _projection_chunks(C):
    """
    Choose K chunks for the fused projection kernel.

    Returns:
      CHUNK_A, CHUNK_B, BLOCK_M, num_warps
    """
    if C <= 128:
        return 128, 0, 128, 8
    if C == 256:
        return 256, 0, 64, 8
    if C < 256:
        return 128, 128, 64, 8
    if C <= 384:
        return 256, 128, 64, 8
    if C <= 512:
        return 256, 256, 32, 8
    if C <= 768:
        return 512, 256, 16, 4
    return 512, 512, 16, 4


def custom_kernel(data):
    """
    TriMul outgoing forward pass.

    Input:
      data = (input_tensor, mask, weights, config)
             or (input_tensor, weights, config) if mask is None.

    Input tensor shape:  [B, N, N, C]
    Mask shape:          [B, N, N] or None
    Output shape:        [B, N, N, C]
    """
    if len(data) == 3:
        input_tensor, weights, config = data
        mask = None
    else:
        input_tensor, mask, weights, config = data

    B, N, _, C = input_tensor.shape
    H = int(config["hidden_dim"])
    S = B * N * N
    device = input_tensor.device

    dt = torch.bfloat16

    if input_tensor.is_contiguous():
        x = input_tensor.view(S, C)
    else:
        x = input_tensor.contiguous().view(S, C)

    if mask is None:
        mask_tensor = x
        has_mask = False
    else:
        mask_tensor = mask.contiguous().to(torch.float32).view(S)
        has_mask = True

    weights = _prepare_weights(weights, device, dt)

    CHUNK_A, CHUNK_B, BLOCK_M, PROJ_WARPS = _projection_chunks(C)
    BLOCK_H = 128
    H_BLOCKS = triton.cdiv(H, BLOCK_H)

    L_t = torch.empty((B, H, N, N), dtype=dt, device=device)
    R_t = torch.empty((B, H, N, N), dtype=dt, device=device)
    G_t = torch.empty((B, N, H, N), dtype=dt, device=device)

    grid_proj = (triton.cdiv(S, BLOCK_M),)
    _projection_kernel[grid_proj](
        x,
        mask_tensor,
        weights["norm_w"],
        weights["norm_b"],
        weights["left_w"],
        weights["right_w"],
        weights["left_gate_w"],
        weights["right_gate_w"],
        weights["out_gate_w"],
        L_t,
        R_t,
        G_t,
        S,
        N,
        H,
        C,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        H_BLOCKS=H_BLOCKS,
        CHUNK_A=CHUNK_A,
        CHUNK_B=CHUNK_B,
        HAS_MASK=has_mask,
        num_warps=PROJ_WARPS,
    )

    # Main TriMul contraction.
    def _pow2_min(x, bound):
        p = max(16, triton.next_power_of_2(x))
        return min(bound, p)

    BLOCK_I = _pow2_min(N, 128)
    BLOCK_J = _pow2_min(N, 128)
    BLOCK_K = _pow2_min(N, 64)

    EVEN_N = (N % BLOCK_I == 0) and (N % BLOCK_J == 0) and (N % BLOCK_K == 0)
    N_ITERS = triton.cdiv(N, BLOCK_K)

    O_t = torch.empty((B, N, H, N), dtype=dt, device=device)

    grid_core = (
        triton.cdiv(N, BLOCK_J),
        triton.cdiv(N, BLOCK_I),
        B * H,
    )
    _core_kernel[grid_core](
        L_t,
        R_t,
        O_t,
        N,
        H,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        N_ITERS=N_ITERS,
        EVEN_N=EVEN_N,
        num_warps=8,
        num_stages=3,
    )

    # Final layer norm + gate + projection.
    out = torch.empty((B, N, N, C), dtype=torch.float32, device=device)

    FBJ = 128 if C <= 128 else 64
    FBJ = min(FBJ, _pow2_min(N, 128))
    FBC = 128
    FHB = 128
    FINAL_H_BLOCKS = triton.cdiv(H, FHB)
    FINAL_SINGLE_H = H <= FHB
    FINAL_C_BLOCKS = triton.cdiv(C, FBC)

    grid_final = (
        triton.cdiv(N, FBJ),
        N,
        B,
        1 if FINAL_SINGLE_H else FINAL_C_BLOCKS,
    )

    _final_kernel[grid_final](
        O_t,
        G_t,
        weights["to_out_w"],
        weights["to_out_norm_w"],
        weights["to_out_norm_b"],
        out,
        N,
        H,
        C,
        BLOCK_J=FBJ,
        BLOCK_C=FBC,
        BLOCK_H=FHB,
        H_BLOCKS=FINAL_H_BLOCKS,
        C_BLOCKS=FINAL_C_BLOCKS,
        SINGLE_H=FINAL_SINGLE_H,
        num_warps=8,
    )

    return out
```
