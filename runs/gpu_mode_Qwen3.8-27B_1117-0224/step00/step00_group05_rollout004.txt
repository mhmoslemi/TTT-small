@triton.jit
def tri_mul_core_kernel(L_ptr, R_ptr, O_ptr, B, N, H,
                        stride_lb, stride_lh, stride_ri, stride_rk?, ...
                        BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, GROUP: tl.constexpr):
    pid = tl.program_id(0)
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    batch = pid // (num_i*num_j)
    local = pid % (num_i*num_j)
    # group
    num_pid_in_group = GROUP * num_j
    group_id = local // num_pid_in_group
    first_pid_m = group_id * GROUP
    group_size_m = tl.minimum(num_i - first_pid_m, GROUP)
    pid_m = first_pid_m + (local % group_size_m)
    pid_n = (local % num_pid_in_group) // group_size_m
    b = batch // H
    h = batch % H
    offs_i = pid_m*BLOCK_I + tl.arange(0,BLOCK_I)
    offs_j = pid_n*BLOCK_J + tl.arange(0,BLOCK_J)
    offs_k = tl.arange(0,BLOCK_K)
    # base offsets
    base = L_ptr + b*stride_lb + h*N*N # if permuted strides lb=N2*H, lh=N2
    a_ptrs = base + offs_i[:,None]*N + offs_k[None,:]
    r_base = R_ptr + b*stride_rb + h*N*N
    b_ptrs = r_base + offs_j[None,:]*N + offs_k[:,None] # [BK,BJ]
    acc = zeros(BI,BJ, fp32)
    n_k = tl.cdiv(N, BK)
    for _ in range(n_k):
       a = tl.load(a_ptrs, mask=(offs_i[:,None] < N)&(offs_k[None,:]<N), other=0)
       bmat = tl.load(b_ptrs, mask=(offs_j[None,:] < N)&(offs_k[:,None]<N), other=0)
       acc = tl.dot(a, bmat, acc)
       a_ptrs += BLOCK_K
       b_ptrs += BLOCK_K
    o_base = O_ptr + b*stride_ob + h*N*N + offs_i[:,None]*N + offs_j[None,:]
    acc = acc.to(tl.bfloat16)
    tl.store(o_base, acc, mask=...)

def core_kernel(l_ptr, r_ptr, o_ptr,

import math
import torch
import triton
import triton.language as tl


def _next_pow2(n: int) -> int:
    """Smallest power of two >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


@triton.jit
def fused_pre_permute_kernel(
    x_ptr,
    mask_ptr,
    norm_w_ptr,
    norm_b_ptr,
    wl_ptr,
    wr_ptr,
    wlg_ptr,
    wrg_ptr,
    wog_ptr,
    ol_ptr,
    or_ptr,
    og_ptr,
    N,
    N2,
    C: tl.constexpr,
    C_PAD: tl.constexpr,
    H,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    H_BLOCKS: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """
    Fused forward stage:
    - LayerNorm over C
    - left/right projections
    - left/right/out gates
    - mask + gate multiplications
    - write left/right in permuted [B, H, N, N] layout for fast GEMM
    - write out_gate in original [B, N, N, H] layout
    """
    pid_m = tl.program_id(0)
    b = tl.program_id(1)

    m0 = pid_m * BLOCK_M
    offs_m = m0 + tl.arange(0, BLOCK_M)
    m_valid = offs_m < N2

    offs_c = tl.arange(0, C_PAD)
    c_valid = offs_c < C

    # Load input tile [M, C_PAD]
    x_ptrs = x_ptr + b * N2 * C + offs_m[:, None] * C + offs_c[None, :]
    x_mask = m_valid[:, None] & c_valid[None, :]
    x = tl.load(x_ptrs, mask=x_mask, other=0.0).to(tl.float32)

    # LayerNorm statistics
    s = tl.sum(x, axis=1)
    ss = tl.sum(x * x, axis=1)
    mean = s / C
    var = ss / C - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    nw = tl.load(norm_w_ptr + offs_c, mask=c_valid, other=0.0).to(tl.float32)
    nb = tl.load(norm_b_ptr + offs_c, mask=c_valid, other=0.0).to(tl.float32)

    a = (x - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]
    a = tl.where(c_valid[None, :], a, 0.0)
    a_bf = a.to(tl.bfloat16)

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + b * N2 + offs_m, mask=m_valid, other=0.0).to(tl.float32)
    else:
        mask_val = tl.zeros((BLOCK_M,), dtype=tl.float32) + 1.0

    for hb in tl.static_range(H_BLOCKS):
        h0 = hb * BLOCK_H
        offs_h = h0 + tl.arange(0, BLOCK_H)
        h_valid = offs_h < H

        # ---- left ----
        wl_tile = tl.load(
            wl_ptr + offs_h[None, :] * C + offs_c[:, None],
            mask=c_valid[:, None] & h_valid[None, :],
            other=0.0,
        )
        wlg_tile = tl.load(
            wlg_ptr + offs_h[None, :] * C + offs_c[:, None],
            mask=c_valid[:, None] & h_valid[None, :],
            other=0.0,
        )

        acc_l = tl.dot(a_bf, wl_tile)
        acc_lg = tl.dot(a_bf, wlg_tile)

        gate_l = 1.0 / (1.0 + tl.exp(-acc_lg))
        out_l = acc_l * mask_val[:, None] * gate_l

        ol_off = b * H * N2 + offs_h[:, None] * N2 + offs_m[None, :]
        tl.store(
            ol_ptr + ol_off,
            tl.trans(out_l.to(tl.bfloat16)),
            mask=h_valid[:, None] & m_valid[None, :],
        )

        # ---- right ----
        wr_tile = tl.load(
            wr_ptr + offs_h[None, :] * C + offs_c[:, None],
            mask=c_valid[:, None] & h_valid[None, :],
            other=0.0,
        )
        wrg_tile = tl.load(
            wrg_ptr + offs_h[None, :] * C + offs_c[:, None],
            mask=c_valid[:, None] & h_valid[None, :],
            other=0.0,
        )

        acc_r = tl.dot(a_bf, wr_tile)
        acc_rg = tl.dot(a_bf, wrg_tile)

        gate_r = 1.0 / (1.0 + tl.exp(-acc_rg))
        out_r = acc_r * mask_val[:, None] * gate_r

        or_off = b * H * N2 + offs_h[:, None] * N2 + offs_m[None, :]
        tl.store(
            or_ptr + or_off,
            tl.trans(out_r.to(tl.bfloat16)),
            mask=h_valid[:, None] & m_valid[None, :],
        )

        # ---- out gate ----
        wog_tile = tl.load(
            wog_ptr + offs_h[None, :] * C + offs_c[:, None],
            mask=c_valid[:, None] & h_valid[None, :],
            other=0.0,
        )
        acc_og = tl.dot(a_bf, wog_tile)
        gate_og = 1.0 / (1.0 + tl.exp(-acc_og))

        og_off = b * N2 * H + offs_m[:, None] * H + offs_h[None, :]
        tl.store(
            og_ptr + og_off,
            gate_og.to(tl.bfloat16),
            mask=m_valid[:, None] & h_valid[None, :],
        )


@triton.jit
def core_kernel(
    l_ptr,
    r_ptr,
    o_ptr,
    B,
    N,
    H,
    N2,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP: tl.constexpr,
):
    """
    Core incoming/outgoing TriMul einsum:
        out[b, i, j, h] = sum_k left[b, h, i, k] * right[b, h, j, k]
    with left/right stored as [B, H, N, N].
    """
    pid = tl.program_id(0)

    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    num_local = num_i * num_j

    batch = pid // num_local
    local = pid % num_local

    # Grouped tile ordering for better locality
    num_pid_in_group = GROUP * num_j
    group_id = local // num_pid_in_group
    first_pid_m = group_id * GROUP
    group_size_m = tl.minimum(num_i - first_pid_m, GROUP)
    pid_m = first_pid_m + (local % group_size_m)
    pid_n = (local % num_pid_in_group) // group_size_m

    b = batch // H
    h = batch % H

    offs_i = pid_m * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_n * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)

    i_valid = offs_i < N
    j_valid = offs_j < N
    k_valid = offs_k < N

    l_base = l_ptr + b * H * N2 + h * N2
    r_base = r_ptr + b * H * N2 + h * N2

    a_ptrs = l_base + offs_i[:, None] * N + offs_k[None, :]
    b_ptrs = r_base + offs_j[None, :] * N + offs_k[:, None]

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for _ in range(0, num_k):
        a = tl.load(a_ptrs, mask=i_valid[:, None] & k_valid[None, :], other=0.0)
        bmat = tl.load(b_ptrs, mask=k_valid[:, None] & j_valid[None, :], other=0.0)
        acc = tl.dot(a, bmat, acc)
        a_ptrs += BLOCK_K
        b_ptrs += BLOCK_K

    o_ptrs = o_ptr + b * H * N2 + h * N2 + offs_i[:, None] * N + offs_j[None, :]
    tl.store(o_ptrs, acc.to(tl.bfloat16), mask=i_valid[:, None] & j_valid[None, :])


@triton.jit
def final_single_h_kernel(
    op_ptr,
    og_ptr,
    out_ptr,
    wout_ptr,
    ln_w_ptr,
    ln_b_ptr,
    N,
    H,
    N2,
    C_OUT,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_C: tl.constexpr,
    C_BLOCKS: tl.constexpr,
):
    """
    Final fused LayerNorm + sigmoid(out_gate) + linear projection.
    Optimized for H <= BLOCK_H (common case H=128).
    """
    pid_m = tl.program_id(0)
    b = tl.program_id(1)

    m0 = pid_m * BLOCK_M
    offs_m = m0 + tl.arange(0, BLOCK_M)
    m_valid = offs_m < N2

    offs_h = tl.arange(0, BLOCK_H)
    h_valid = offs_h < H

    # Load OP tile as [H, M], since OP is [B, H, N, N] and M is contiguous within a batch/h
    op_ptrs = op_ptr + b * H * N2 + offs_h[:, None] * N2 + offs_m[None, :]
    op_t = tl.load(op_ptrs, mask=h_valid[:, None] & m_valid[None, :], other=0.0).to(tl.float32)

    # LayerNorm statistics over H
    s = tl.sum(op_t, axis=0)
    ss = tl.sum(op_t * op_t, axis=0)
    mean = s / H
    var = ss / H - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    nw = tl.load(ln_w_ptr + offs_h, mask=h_valid, other=0.0).to(tl.float32)
    nb = tl.load(ln_b_ptr + offs_h, mask=h_valid, other=0.0).to(tl.float32)

    # Load gate as [M, H], then transpose to [H, M]
    og_ptrs = og_ptr + b * N2 * H + offs_m[:, None] * H + offs_h[None, :]
    gate_mh = tl.load(og_ptrs, mask=m_valid[:, None] & h_valid[None, :], other=0.0).to(tl.float32)
    gate_t = tl.trans(gate_mh)

    y_t = (op_t - mean[None, :]) * rstd[None, :] * nw[:, None] + nb[:, None]
    gate_sig = 1.0 / (1.0 + tl.exp(-gate_t))
    y_t = y_t * gate_sig
    y_t = tl.where(h_valid[:, None], y_t, 0.0)

    y = tl.trans(y_t.to(tl.bfloat16))  # [M, H]

    for cb in tl.static_range(C_BLOCKS):
        c0 = cb * BLOCK_C
        offs_c = c0 + tl.arange(0, BLOCK_C)
        c_valid = offs_c < C_OUT

        # Wout is [C_OUT, H]. Load as [H, C] for dot.
        w_ptrs = wout_ptr + offs_c[None, :] * H + offs_h[:, None]
        w = tl.load(w_ptrs, mask=h_valid[:, None] & c_valid[None, :], other=0.0)

        acc = tl.dot(y, w)  # [M, C]

        out_ptrs = out_ptr + b * N2 * C_OUT + offs_m[:, None] * C_OUT + offs_c[None, :]
        tl.store(out_ptrs, acc, mask=m_valid[:, None] & c_valid[None, :])


@triton.jit
def final_multi_h_kernel(
    op_ptr,
    og_ptr,
    out_ptr,
    wout_ptr,
    ln_w_ptr,
    ln_b_ptr,
    N,
    H,
    N2,
    C_OUT,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_C: tl.constexpr,
    H_BLOCKS: tl.constexpr,
    C_BLOCKS: tl.constexpr,
):
    """
    General final fused LayerNorm + gate + linear path for H > BLOCK_H.
    """
    pid_m = tl.program_id(0)
    b = tl.program_id(1)
    pid_c = tl.program_id(2)

    m0 = pid_m * BLOCK_M
    c0 = pid_c * BLOCK_C

    offs_m = m0 + tl.arange(0, BLOCK_M)
    offs_c = c0 + tl.arange(0, BLOCK_C)
    m_valid = offs_m < N2
    c_valid = offs_c < C_OUT

    # First pass: LayerNorm statistics
    s = tl.zeros((BLOCK_M,), dtype=tl.float32)
    ss = tl.zeros((BLOCK_M,), dtype=tl.float32)

    for hb in tl.static_range(H_BLOCKS):
        h0 = hb * BLOCK_H
        offs_h = h0 + tl.arange(0, BLOCK_H)
        h_valid = offs_h < H

        op_ptrs = op_ptr + b * H * N2 + offs_h[:, None] * N2 + offs_m[None, :]
        op_t = tl.load(op_ptrs, mask=h_valid[:, None] & m_valid[None, :], other=0.0).to(tl.float32)

        s += tl.sum(op_t, axis=0)
        ss += tl.sum(op_t * op_t, axis=0)

    mean = s / H
    var = ss / H - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    acc = tl.zeros((BLOCK_M, BLOCK_C), dtype=tl.float32)

    # Second pass: normalize, gate, and multiply with Wout
    for hb in tl.static_range(H_BLOCKS):
        h0 = hb * BLOCK_H
        offs_h = h0 + tl.arange(0, BLOCK_H)
        h_valid = offs_h < H

        op_ptrs = op_ptr + b * H * N2 + offs_h[:, None] * N2 + offs_m[None, :]
        op_t = tl.load(op_ptrs, mask=h_valid[:, None] & m_valid[None, :], other=0.0).to(tl.float32)

        nw = tl.load(ln_w_ptr + offs_h, mask=h_valid, other=0.0).to(tl.float32)
        nb = tl.load(ln_b_ptr + offs_h, mask=h_valid, other=0.0).to(tl.float32)

        og_ptrs = og_ptr + b * N2 * H + offs_m[:, None] * H + offs_h[None, :]
        gate_mh = tl.load(og_ptrs, mask=m_valid[:, None] & h_valid[None, :], other=0.0).to(tl.float32)
        gate_t = tl.trans(gate_mh)

        y_t = (op_t - mean[None, :]) * rstd[None, :] * nw[:, None] + nb[:, None]
        gate_sig = 1.0 / (1.0 + tl.exp(-gate_t))
        y_t = y_t * gate_sig
        y_t = tl.where(h_valid[:, None], y_t, 0.0)

        y = tl.trans(y_t.to(tl.bfloat16))  # [M, BLOCK_H]

        w_ptrs = wout_ptr + offs_c[None, :] * H + offs_h[:, None]
        w = tl.load(w_ptrs, mask=h_valid[:, None] & c_valid[None, :], other=0.0)

        acc = tl.dot(y, w, acc)

    out_ptrs = out_ptr + b * N2 * C_OUT + offs_m[:, None] * C_OUT + offs_c[None, :]
    tl.store(out_ptrs, acc, mask=m_valid[:, None] & c_valid[None, :])


def custom_kernel(data):
    """
    Optimized Triton implementation of AlphaFold3 TriMul (outgoing) forward pass.

    Strategy:
    1) Triton kernel fuses LayerNorm + projections + gates + mask and writes
       left/right in a GEMM-friendly permuted layout [B, H, N, N].
    2) Triton batched GEMM computes the O(N^3) core:
           out[b, i, j, h] = sum_k left[b, h, i, k] * right[b, h, j, k]
       using tl.dot with bf16 inputs and fp32 accumulation.
    3) Triton kernel fuses final LayerNorm + sigmoid(out_gate) + final linear.
    """
    # Parse input tuple flexibly: (x, mask, weights, config) or (x, weights, config)
    if len(data) == 4:
        input_tensor, mask, weights, config = data
    else:
        input_tensor, weights, config = data
        mask = None

    x = input_tensor
    if not x.is_contiguous():
        x = x.contiguous()

    device = x.device
    B, N, _, C_from_shape = x.shape
    N2 = N * N

    C = int(config.get("dim", C_from_shape))
    if C != C_from_shape:
        C = C_from_shape
    H = int(config.get("hidden_dim", 128))

    # Mask handling
    if mask is not None and hasattr(mask, "numel") and mask.numel() > 0:
        if mask.dim() == 2:
            mask = mask.unsqueeze(0)
        if mask.shape[0] == 1 and B > 1:
            mask = mask.expand(B, N, N)
        mask_t = mask.to(torch.float32).contiguous()
        has_mask = True
    else:
        mask_t = torch.empty(1, dtype=torch.float32, device=device)
        has_mask = False

    # Output and intermediate buffers
    out = torch.empty((B, N, N, C), dtype=torch.float32, device=device)
    Lp = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    Rp = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    OG = torch.empty((B, N, N, H), dtype=torch.bfloat16, device=device)
    OP = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)

    # Prepare weights
    bf16 = torch.bfloat16
    fp32 = torch.float32

    norm_w = weights["norm.weight"].to(fp32).contiguous()
    norm_b = weights["norm.bias"].to(fp32).contiguous()

    wl = weights["left_proj.weight"].to(bf16).contiguous()
    wr = weights["right_proj.weight"].to(bf16).contiguous()
    wlg = weights["left_gate.weight"].to(bf16).contiguous()
    wrg = weights["right_gate.weight"].to(bf16).contiguous()
    wog = weights["out_gate.weight"].to(bf16).contiguous()

    to_out_norm_w = weights["to_out_norm.weight"].to(fp32).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(fp32).contiguous()
    wout = weights["to_out.weight"].to(bf16).contiguous()

    # ---- Stage 1: fused LN + projections + gates + mask + permute ----
    C_PAD = _next_pow2(C)
    if C_PAD <= 128:
        fused_block_m = 64
        fused_block_h = 128
    else:
        fused_block_m = 32
        fused_block_h = 64

    fused_h_blocks = triton.cdiv(H, fused_block_h)
    grid_fused = (triton.cdiv(N2, fused_block_m), B)

    fused_pre_permute_kernel[grid_fused](
        x,
        mask_t,
        norm_w,
        norm_b,
        wl,
        wr,
        wlg,
        wrg,
        wog,
        Lp,
        Rp,
        OG,
        N,
        N2,
        C=C,
        C_PAD=C_PAD,
        H=H,
        BLOCK_M=fused_block_m,
        BLOCK_H=fused_block_h,
        H_BLOCKS=fused_h_blocks,
        HAS_MASK=has_mask,
        num_warps=8,
        num_stages=2,
    )

    # ---- Stage 2: core O(N^3) batched GEMM ----
    BLOCK_I = 128
    BLOCK_J = 128
    BLOCK_K = 64
    GROUP = 8

    grid_core = (
        triton.cdiv(N, BLOCK_I)
        * triton.cdiv(N, BLOCK_J)
        * (B * H),
    )

    core_kernel[grid_core](
        Lp,
        Rp,
        OP,
        B,
        N,
        H,
        N2,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        GROUP=GROUP,
        num_warps=8,
        num_stages=3,
    )

    # Free large intermediates before final stage
    del Lp
    del Rp

    # ---- Stage 3: fused final LN + gate + linear ----
    FINAL_BLOCK_M = 32
    FINAL_BLOCK_H = 128
    FINAL_BLOCK_C = 128
    C_BLOCKS = triton.cdiv(C, FINAL_BLOCK_C)

    if H <= FINAL_BLOCK_H:
        grid_final = (triton.cdiv(N2, FINAL_BLOCK_M), B)
        final_single_h_kernel[grid_final](
            OP,
            OG,
            out,
            wout,
            to_out_norm_w,
            to_out_norm_b,
            N,
            H,
            N2,
            C,
            BLOCK_M=FINAL_BLOCK_M,
            BLOCK_H=FINAL_BLOCK_H,
            BLOCK_C=FINAL_BLOCK_C,
            C_BLOCKS=C_BLOCKS,
            num_warps=4,
            num_stages=2,
        )
    else:
        H_BLOCKS = triton.cdiv(H, FINAL_BLOCK_H)
        grid_final = (triton.cdiv(N2, FINAL_BLOCK_M), B, C_BLOCKS)
        final_multi_h_kernel[grid_final](
            OP,
            OG,
            out,
            wout,
            to_out_norm_w,
            to_out_norm_b,
            N,
            H,
            N2,
            C,
            BLOCK_M=FINAL_BLOCK_M,
            BLOCK_H=FINAL_BLOCK_H,
            BLOCK_C=FINAL_BLOCK_C,
            H_BLOCKS=H_BLOCKS,
            C_BLOCKS=C_BLOCKS,
            num_warps=4,
            num_stages=2,
        )

    return out
