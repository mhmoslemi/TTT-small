@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program ID
    pid = tl.program_id(0)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_mn = num_pid_m * num_pid_n
    
    batch_id = pid // num_pid_mn
    pid_mn = pid % num_pid_mn
    pid_m = pid_mn // num_pid_n
    pid_n = pid_mn % num_pid_n
    
    b = batch_id // H
    d = batch_id % H
    
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    
    # Pointers for left[b, i, k, d]
    # left offset: b * stride_lb + i * stride_li + k * stride_lk + d * stride_ld
    left_ptrs = left_ptr + b * stride_lb + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk + d * stride_ld
    
    # Pointers for right[b, j, k, d]
    # We need right^T, so right[b, k, j, d] -> we access right[b, j, k, d] but in transposed fashion
    # For the GEMM: C[i,j] = sum_k A[i,k] * B[k,j]
    # B[k,j] = right[b, j, k, d]
    # So B is accessed with k as the first index and j as the second
    # right offset: b * stride_rb + j * stride_rj + k * stride_rk + d * stride_rd
    # B[k, j] -> offset = b * stride_rb + j * stride_rj + k * stride_rk + d * stride_rd
    right_ptrs = right_ptr + b * stride_rb + offs_n[None, :] * stride_rj + offs_k[:, None] * stride_rk + d * stride_rd
    
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    for k in range(0, N, BLOCK_K):
        a = tl.load(left_ptrs, mask=(offs_m[:, None] < N) & ((offs_k[None, :] + k) < N), other=0.0)
        b_mat = tl.load(right_ptrs, mask=(offs_n[None, :] < N) & ((offs_k[:, None] + k) < N), other=0.0)
        acc += tl.dot(a, b_mat)
        left_ptrs += BLOCK_K * stride_lk
        right_ptrs += BLOCK_K * stride_rk
    
    # Store out[b, i, j, d]
    # out offset: b * stride_ob + i * stride_oi + j * stride_oj + d * stride_od
    out_ptrs = out_ptr + b * stride_ob + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj + d * stride_od
    tl.store(out_ptrs, acc, mask=(offs_m[:, None] < N) & (offs_n[None, :] < N))
```

import torch
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(
    x_ptr, weight_ptr, bias_ptr, out_ptr,
    n_rows, dim,
    eps,
    BLOCK_DIM: tl.constexpr,
):
    row = tl.program_id(0)
    if row >= n_rows:
        return
    
    cols = tl.arange(0, BLOCK_DIM)
    mask = cols < dim
    
    x = tl.load(x_ptr + row * dim + cols, mask=mask, other=0.0).to(tl.float32)
    
    mean = tl.sum(x) / dim
    x_centered = x - mean
    var = tl.sum(x_centered * x_centered) / dim
    x_norm = x_centered / tl.sqrt(var + eps)
    
    w = tl.load(weight_ptr + cols, mask=mask, other=0.0).to(tl.float32)
    b = tl.load(bias_ptr + cols, mask=mask, other=0.0).to(tl.float32)
    
    out = x_norm * w + b
    tl.store(out_ptr + row * dim + cols, out, mask=mask)
```

import triton

@triton.jit
def layernorm_fwd_kernel(

    dim, eps,

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    
    # Reshape input to 2D for linear ops: [B*N*N, dim]
    B, N, N2, D = input_tensor.shape
    assert N == N2
    
    x = input_tensor.reshape(-1, D)
    
    # LayerNorm (Triton kernel)
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x_normed = torch.empty_like(x)

    BLOCK = triton.next_power_of_2(D)
    layernorm_fwd_kernel[grid](x, norm_w, norm_b, x_normed, D, 1e-5, BLOCK=BLOCK)
    
    # Projections (PyTorch)
    left = x_normed @ weights["left_proj.weight"].T
    right = x_normed @ weights["right_proj.weight"].T
    left_gate = torch.sigmoid(x_normed @ weights["left_gate_weight"].T)
    right_gate = torch.sigmoid(x_normed @ weights["right_gate_weight"].T)
    out_gate = torch.sigmoid(x_normed @ weights["out_gate_weight"].T)
    
    # Reshape back
    left = left.reshape(B, N, N, hidden_dim)
    right = right.reshape(B, N, N, hidden_dim)
    left_gate = left_gate.reshape(B, N, N, hidden_dim)
    right_gate = right_gate.reshape(B, N, N, hidden_dim)
    out_gate = out_gate.reshape(B, N, N, hidden_dim)
    
    # Apply mask
    if mask is not None:

        left = left * mask_4d
        right = right * mask_4d

    left = left * left_gate
    right = right * right_gate

    out_final = out_final.reshape(B, N, N, dim)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

"""
TriMul (Triangle Multiplicative Update) - Outgoing version.
Fuses LayerNorm, linear projections, gating, and the core O(N^3) tensor
contraction out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d].
Uses a Triton kernel for LayerNorm and delegates the batched GEMM to cuBLAS.
"""

@triton.jit
def _layernorm_kernel(
    x_ptr, w_ptr, b_ptr, out_ptr,
    dim, eps,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    cols = tl.arange(0, BLOCK)
    mask = cols < dim
    
    x = tl.load(x_ptr + pid * dim + cols, mask=mask, other=0.0).to(tl.float32)
    mean = tl.sum(x, axis=0) / dim
    xc = x - mean
    var = tl.sum(xc * xc, axis=0) / dim
    rstd = 1.0 / tl.sqrt(var + eps)
    
    w = tl.load(w_ptr + cols, mask=mask, other=0.0).to(tl.float32)
    b = tl.load(b_ptr + cols, mask=mask, other=0.0).to(tl.float32)
    
    out = xc * rstd * w + b
    tl.store(out_ptr + pid * dim + cols, out, mask=mask)


def _layernorm(x, weight, bias, dim, eps=1e-5):
    """Apply LayerNorm using the Triton kernel."""
    shape = x.shape
    x_2d = x.reshape(-1, dim)
    out = torch.empty_like(x_2d)
    BLOCK = triton.next_power_of_2(dim)
    grid = (x_2d.shape[0],)
    _layernorm_kernel[grid](x_2d, weight, bias, out, dim, eps, BLOCK=BLOCK)
    return out.reshape(shape)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    
    B, N, N2, D = input_tensor.shape
    x = input_tensor.reshape(-1, D)
    
    # 1. LayerNorm
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x_normed = _layernorm(x, norm_w, norm_b, D)
    
    # 2. Projections and gates
    left = x_normed @ weights["left_proj.weight"].T
    right = x_normed @ weights["right_proj.weight"].T
    lg = torch.sigmoid(x_normed @ weights["left_gate.weight"].T)
    rg = torch.sigmoid(x_normed @ weights["right_gate.weight"].T)
    og = torch.sigmoid(x_normed @ weights["out_gate.weight"].T)
    
    # Reshape to 4D
    left = left.reshape(B, N, N, hidden_dim)
    right = right.reshape(B, N, N, hidden_dim)
    lg = lg.reshape(B, N, N, hidden_dim)
    rg = rg.reshape(B, N, N, hidden_dim)
    og = og.reshape(B, N, N, hidden_dim)
    
    # 3. Apply mask
    if mask is not None:
        m4 = mask.unsqueeze(-1)
        left = left * m4
        right = right * m4
    
    # 4. Apply gates
    left = left * lg
    right = right * rg
    
    # 5. Core TriMul (batched GEMM)
    left_perm = left.permute(0, 3, 1, 2)   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2)  # [B, H, N, N]
    
    # Use bf16 for the GEMM (accumulated in fp32 by cuBLAS)
    out_perm = torch.bmm(
        left_perm.reshape(B * hidden_dim, N, N).to(torch.bfloat16),
        right_perm.reshape(B * hidden_dim, N, N).to(torch.bfloat16).transpose(1, 2)
    )
    out = out_perm.float().reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    
    # 6. Final LayerNorm
    out_2d = out.reshape(-1, hidden_dim)
    on_w = weights["to_out_norm.weight"]
    on_b = weights["to_out_norm.bias"]
    out_normed = _layernorm(out_2d, on_w, on_b, hidden_dim)
    
    # 7. Apply output gate and final projection
    og_2d = og.reshape(-1, hidden_dim)
    out_normed = out_normed * og_2d
    out_final = out_normed @ weights["to_out.weight"].T
    
    return out_final.reshape(B, N, N, dim).float()

import triton
import triton.language as tl

"""
TriMul (Triangle Multiplicative Update) - Outgoing version for protein structure prediction.
Computes out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d] with fused LayerNorm and gating.
Uses Triton kernel for LayerNorm; delegates batched GEMM to cuBLAS via torch.bmm.
"""

@triton.jit
def _layernorm_kernel(
    x_ptr, w_ptr, b_ptr, out_ptr,
    dim, eps,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    cols = tl.arange(0, BLOCK)
    m = cols < dim
    x = tl.load(x_ptr + pid * dim + cols, mask=m, other=0.0).to(tl.float32)
    mean = tl.sum(x) / dim
    xc = x - mean
    var = tl.sum(xc * xc) / dim
    rstd = 1.0 / tl.sqrt(var + eps)
    w = tl.load(w_ptr + cols, mask=m, other=0.0).to(tl.float32)
    b = tl.load(b_ptr + cols, mask=m, other=0.0).to(tl.float32)
    out = xc * rstd * w + b
    tl.store(out_ptr + pid * dim + cols, out, mask=m)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, D = input_tensor.shape
    
    # FlattEen to 2D
    x = input_tensor.view(-1, D)
    M = x.shape[0]
    
    # LayerNorm (Triton)
    nw = weights["norm.weight"]
    nb = weights["norm.bias"]
    x_norm = torch.empty_like(x)
    BLK = triton.next_power_of_2(D)
    _layernorm_kernel[(M,)](x, nw, nb, x_norm, D, 1e-5, BLOCK=BLK)

        weights["right_proj.weight"],     # [H, D]
        weights["left_gate.weight"],      # [H, D]
        weights["right_gate.weight"],     # [H, D]
        weights["out_gate.weight"],       # [H, D]
    ], dim=0)  # [5H, D]

    left = proj[:, :hidden_dim]                    # [M, H]
    right = proj[:, hidden_dim:2*hidden_dim]       # [M, H]
    lg = torch.sigmoid(proj[:, 2*hidden_dim:3*hidden_dim])
    rg = torch.sigmoid(proj[:, 3*hidden_dim:4*hidden_dim])
    og = torch.sigmoid(proj[:, 4*hidden_dim:5*hidden_dim])

    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    lg = lg.view(B, N, N, hidden_dim)
    rg = rg.view(B, N, N, hidden_dim)
    og = og.view(B, N, N, hidden_dim)

        left = left * m4
        right = right * m4

    right = right * rg

    left_4d = left.permute(0, 3, 1, 2)  # [B, H, N, N]
    right_4d = right.permute(0, 3, 1, 2)  # [B, H, N, N]
    
    out_raw = torch.bmm(

    onb = weights["to_out_norm.bias"]

    # Apply output gate
    og_2d = og.view(-1, hidden_dim)
    out_norm = out_norm * og_2d
    
    # Final projection
    out_final = out_norm @ weights["to_out.weight"].T  # [M, dim]
    
    return out_final.view(B, N, N, dim).float()

import torch
import triton
import triton.language as tl

"""
TriMul (Triangle Multiplicative Update) - Outgoing version for protein structure prediction.
Computes out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d] with fused LayerNorm and gating.
Uses a Triton kernel for LayerNorm; delegates the batched GEMM to cuBLAS via torch.bmm.
"""

@triton.jit
def _layernorm_kernel(
    x_ptr, w_ptr, b_ptr, out_ptr,
    dim, eps,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    cols = tl.arange(0, BLOCK)
    m = cols < dim
    x = tl.load(x_ptr + pid * dim + cols, mask=m, other=0.0).to(tl.float32)
    mean = tl.sum(x) / dim
    xc = x - mean
    var = tl.sum(xc * xc) / dim
    rstd = 1.0 / tl.sqrt(var + eps)
    w = tl.load(w_ptr + cols, mask=m, other=0.0).to(tl.float32)
    b = tl.load(b_ptr + cols, mask=m, other=0.0).to(tl.float32)
    out = xc * rstd * w + b
    tl.store(out_ptr + pid * dim + cols, out, mask=m)


@triton.jit
def _permute_to_batched_kernel(
    x_ptr, out_ptr,
    N, H,
    BLOCK_N: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Permute x from [B, N, N, H] layout to [B*H, N, N] (contiguous).
    Each program handles a BLOCK_N x BLOCK_H tile.
    """
    pid = tl.program_id(0)
    total_tiles = (N // BLOCK_N) * (N // BLOCK_N) * H
    # pid encodes (b, d, i_tile, j_tile)
    tiles_ij = (N // BLOCK_N) * (N // BLOCK_N)
    d = pid // tiles_ij
    pid_ij = pid % tiles_ij
    i_tile = pid_ij // (N // BLOCK_N)
    j_tile = pid_ij % (N // BLOCK_N)

    b = pid // (H * tiles_ij)
    
    offs_i = i_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, BLOCK_H)

    # Load from x[b, i, j, h]: offset = b*N*N*H + i*N*H + j*H + h
    in_ptrs = x_ptr + b * N * N * H + offs_i[:, None] * N * H + offs_j[None, :] * H + offs_h[None, None]
    # We need a 3D load: [BLOCK_N, BLOCK_N, BLOCK_H] but Triton loads are 2D
    # Let's do it differently: load a 2D tile [BLOCK_N, BLOCK_H] for each j

    # Actually, let's just do a simple element-wise copy with the right index mapping.
    # For out[(b*H + d)*N + i, j] = x[b*N*N*H + i*N*H + j*H + d]
    # We process a BLOCK_N x BLOCK_N tile for fixed (b, d)

    # Load: for each (i, j) in the tile, read x[b, i, j, d]
    # This is a 2D load: [BLOCK_N, BLOCK_N] from x with strides (N*H, H)
    h = tl.arange(0, BLOCK_H)
    # We only need h = 0 (single element per (i,j) for this d)
    # Actually d is fixed, so we just read one element per (i,j) pair
    
    in_ptrs_2d = x_ptr + b * N * N * H + offs_i[:, None] * N * H + offs_j[None, :] * H + d
    vals = tl.load(in_ptrs_2d)
    
    # Store to out[(b*H + d) * N + i, j]
    out_base = (b * H + d) * N
    out_ptrs_2d = out_ptr + out_base * N + offs_i[:, None] * N + offs_j[None, :]
    tl.store(out_ptrs_2d, vals)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, D = input_tensor.shape
    M = B * N * N
    
    x = input_tensor.reshape(M, D)
    
    # LayerNorm (Triton kernel)
    nw = weights["norm.weight"]
    nb = weights["norm.bias"]
    x_norm = torch.empty(M, D, device=x.device, dtype=torch.float32)
    BLK = triton.next_power_of_2(D)
    _layernorm_kernel[(M,)](x, nw, nb, x_norm, D, 1e-5, BLOCK=BLK)
    
    # Combined projection: all 5 linears in one GEMM
    W_cat = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0)  # [5H, D]
    
    proj = (x_norm @ W_cat.T)  # [M, 5H]
    
    H = hidden_dim
    left = proj[:, 0:H]
    right = proj[:, H:2*H]
    lg = torch.sigmoid(proj[:, 2*H:3*H])
    rg = torch.sigmoid(proj[:, 3*H:4*H])
    og = torch.sigmoid(proj[:, 4*H:5*H])
    
    # Reshape to 4D and apply mask + gates
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    lg = lg.view(B, N, N, H)
    rg = rg.view(B, N, N, H)
    og = og.view(B, N, N, H)
    
    if mask is not None:
        m4 = mask.unsqueeze(-1).float()
        left = left * m4
        right = right * m4
    
    left = left * lg
    right = right * rg
    
    # Core TriMul: permute to [B*H, N, N] then batched GEMM
    left_perm = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_perm = right.permute(0, 3, 1, 2).reshape(B * H, N, N)
    
    out_bmm = torch.bmm(left_perm.bfloat16(), right_perm.bfloat16().transpose(1, 2))
    out = out_bmm.float().view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    
    # Final LayerNorm (Triton kernel)
    out_2d = out.view(M, H)
    onw = weights["to_out_norm.weight"]
    onb = weights["to_out_norm.bias"]
    out_norm = torch.empty(M, H, device=x.device, dtype=torch.float32)
    BLK2 = triton.next_power_of_2(H)
    _layernorm_kernel[(M,)](out_2d, onw, onb, out_norm, H, 1e-5, BLOCK=BLK2)
    
    # Apply output gate and final projection
    og_2d = og.view(M, H)
    out_norm = out_norm * og_2d
    out_final = out_norm @ weights["to_out.weight"].T
    
    return out_final.view(B, N, N, dim).float()
