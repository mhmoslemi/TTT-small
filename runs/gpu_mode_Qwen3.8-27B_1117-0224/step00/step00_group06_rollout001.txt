def batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,  # batch, seq_len, hidden_dim
    # strides for left: [b, i, k, d]
    stride_lb, stride_li, stride_lk, stride_ld,
    # strides for right: [b, j, k, d]
    stride_rb, stride_rj, stride_rk, stride_rd,
    # strides for out: [b, i, j, d]
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,  # not needed if D fits in one block
):
    # Grid: (B * D * cdiv(N, BLOCK_I) * cdiv(N, BLOCK_J),)
    # Or better: (cdiv(N, BLOCK_I) * cdiv(N, BLOCK_J) * D * B,)
    
    pid = tl.program_id(0)
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    
    # Decode pid
    pid_ij = pid % (num_i * num_j)
    pid_bd = pid // (num_i * num_j)
    pid_d = pid_bd % D
    pid_b = pid_bd // D
    
    pid_i = pid_ij % num_i
    pid_j = pid_ij // num_i
    
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)
    
    # Pointers for left: [b, i, k, d]
    # We want left[b, offs_i, offs_k, pid_d]
    # Shape: [BLOCK_I, BLOCK_K]
    left_ptrs = left_ptr + pid_b * stride_lb + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk + pid_d * stride_ld
    
    # Pointers for right: [b, j, k, d]
    # We want right[b, offs_j, offs_k, pid_d]
    # Shape: [BLOCK_J, BLOCK_K]
    right_ptrs = right_ptr + pid_b * stride_rb + offs_j[:, None] * stride_rj + offs_k[None, :] * stride_rk + pid_d * stride_rd
    
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)
    
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(left_ptrs)  # [BLOCK_I, BLOCK_K]
        b = tl.load(right_ptrs)  # [BLOCK_J, BLOCK_K]
        acc += tl.dot(a, tl.trans(b))  # [BLOCK_I, BLOCK_J]
        left_ptrs += BLOCK_K * stride_lk
        right_ptrs += BLOCK_K * stride_rk
    
    # Store: out[b, offs_i, offs_j, pid_d]
    out_ptrs = out_ptr + pid_b * stride_ob + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj + pid_d * stride_od
    tl.store(out_ptrs, acc)

def tri_mul_contract_kernel(

TriMul (Triangle Multiplicative Update) forward pass optimized with Triton.
Core optimization: the expensive N^3 contraction (einsum over k) is implemented
as a batched GEMM in Triton using bfloat16 tensor cores. All other operations
(LayerNorm, linear projections, gating) use PyTorch/cuBLAS.

import torch
import triton
import triton.language as tl


@triton.jit
def _tri_mul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    N, D, B,
    slb, sli, slk, sld,
    srb, srj, srk, srd,
    sob, soi, soj, sod,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    n_ij = num_i * num_j

    pid_ij = pid % n_ij
    pid_bd = pid // n_ij
    pid_d = pid_bd % D
    pid_b = pid_bd // D
    pid_i = pid_ij % num_i
    pid_j = pid_ij // num_i

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)

    mi = offs_i < N
    mj = offs_j < N

    lb = left_ptr + pid_b * slb + pid_d * sld
    rb = right_ptr + pid_b * srb + pid_d * srd

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    for k_idx in range(0, tl.cdiv(N, BLOCK_K)):
        ok = k_idx * BLOCK_K + offs_k
        mk = ok < N

                    mask=mi[:, None] & mk[None, :], other=0.0)

                    mask=mj[:, None] & mk[None, :], other=0.0)
