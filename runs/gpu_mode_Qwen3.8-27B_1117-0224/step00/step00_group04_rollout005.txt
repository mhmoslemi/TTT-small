  pid=tl.program_id(0)
  num_m=tl.cdiv(M,BM)
  b=pid//num_m; mb=pid%num_m
  offs_m = mb*BM + tl.arange(0,BM)
  offs_h = tl.arange(0,FH)
  m_mask = offs_m < M
  ptrs = out_ptr + b*M*FH + offs_h[:,None]*M + offs_m[None,:]
  net = tl.load(ptrs, mask=m_mask[None,:], other=0.0).to(tl.float32) # shape [FH,BM]
  net_t = tl.trans(net) # [BM,FH]
  gate = tl.load(gate_ptr + ..., mask=m_mask[None,:], other=0.0).to(tl.float32)
  gate_t = tl.trans(gate)
  mean = tl.sum(net_t, axis=1)/FH

  gated = (normed * gate_t).to(tl.bfloat16)

      offs_c = c0 + tl.arange(0,C_BLOCK)
      c_mask = offs_c < C

@triton.jit
def _core_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    N,
    M,
    H,
    BI: tl.constexpr,
    BJ: tl.constexpr,
    BK: tl.constexpr,
):
    pid = tl.program_id(0)

    num_i = tl.cdiv(N, BI)
    num_j = tl.cdiv(N, BJ)
    ij = num_i * num_j

    pid_ij = pid % ij
    pid_i = pid_ij // num_j
    pid_j = pid_ij % num_j

    pid_bh = pid // ij
    b = pid_bh // H
    h = pid_bh % H

    offs_i = pid_i * BI + tl.arange(0, BI)
    offs_j = pid_j * BJ + tl.arange(0, BJ)
    offs_k = tl.arange(0, BK)

    i_mask = offs_i < N
    j_mask = offs_j < N

    base = b * H * M + h * M

    acc = tl.zeros((BI, BJ), dtype=tl.float32)

    for k0 in range(0, N, BK):
        k_off = k0 + offs_k
        k_mask = k_off < N

        a_ptrs = left_ptr + base + offs_i[:, None] * N + k_off[None, :]
        a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # right tensor is stored as [b, h, k, j]
        b_ptrs = right_ptr + base + k_off[:, None] * N + offs_j[None, :]
        bmat = tl.load(b_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        acc = tl.dot(a, bmat, acc)

    o_ptrs = out_ptr + base + offs_i[:, None] * N + offs_j[None, :]
    tl.store(o_ptrs, acc.to(tl.bfloat16), mask=i_mask[:, None] & j_mask[None, :])


@triton.jit
def _final_kernel(
    out_ptr,
    gate_ptr,
    ln_w_ptr,
    ln_b_ptr,
    w_ptr,
    final_ptr,
    M,
    C,
    BM: tl.constexpr,
    FH: tl.constexpr,
    CB: tl.constexpr,
):
    pid = tl.program_id(0)
    num_m = tl.cdiv(M, BM)
    b = pid // num_m
    mb = pid % num_m

    offs_m = mb * BM + tl.arange(0, BM)
    offs_h = tl.arange(0, FH)

    m_mask = offs_m < M
    base = b * FH * M

    # Load as [FH, BM] so that BM (spatial index) is contiguous, then transpose.
    net_ptrs = out_ptr + base + offs_h[:, None] * M + offs_m[None, :]
    net = tl.load(net_ptrs, mask=m_mask[None, :], other=0.0).to(tl.float32)
    net_t = tl.trans(net)

    gate_ptrs = gate_ptr + base + offs_h[:, None] * M + offs_m[None, :]
    gate = tl.load(gate_ptrs, mask=m_mask[None, :], other=0.0).to(tl.float32)
    gate_t = tl.trans(gate)

    mean = tl.sum(net_t, axis=1) / FH
    xc = net_t - mean[:, None]
    var = tl.sum(xc * xc, axis=1) / FH
    rstd = tl.rsqrt(var + 1e-5)

    ln_w = tl.load(ln_w_ptr + offs_h).to(tl.float32)
    ln_b = tl.load(ln_b_ptr + offs_h).to(tl.float32)

    normed = xc * rstd[:, None] * ln_w[None, :] + ln_b[None, :]
    gated = (normed * gate_t).to(tl.bfloat16)

    for c0 in range(0, C, CB):
        offs_c = c0 + tl.arange(0, CB)
        c_mask = offs_c < C

        w = tl.load(
            w_ptr + offs_h[:, None] * C + offs_c[None, :],
            mask=c_mask[None, :],
            other=0.0,
        )
        acc = tl.dot(gated, w)

        o_ptrs = final_ptr + b * M * C + offs_m[:, None] * C + offs_c[None, :]
        tl.store(o_ptrs, acc, mask=m_mask[:, None] & c_mask[None, :])


def custom_kernel(data):
    """
    Triton-forward TriMul:
    - fuses LayerNorm/projections/gates/mask for left/right operands,
    - stores operands in channel-major forms for efficient per-channel batched GEMMs,
    - computes out[i,j,c] = sum_k left[i,k,c] * right[j,k,c],
    - fuses output LayerNorm/gate/projection.
    """
    if len(data) == 3:
        x, weights, config = data
        mask = None
    else:
        x, mask, weights, config = data

    x = x.float().contiguous()
    B, N, _, C = x.shape
    H = int(config.get("hidden_dim", config.get("hidden", 128)))
    M = N * N
    device = x.device

    nomask = bool(config.get("nomask", False))
    if mask is None or nomask:
        has_mask = False
        mask_t = x
    else:
        has_mask = True
        mask_t = mask.reshape(B * M).float().contiguous()

    bf16 = torch.bfloat16

    def _pack_proj_weight(key: str) -> torch.Tensor:
        w = weights[key].float()
        # w is [H_in, C_in] for torch.nn.Linear
        cc = min(C, w.shape[1])
        hh = min(H, w.shape[0])
        out = torch.zeros((cc, hh), dtype=bf16, device=device)
        out[:cc, :hh] = w[:hh, :cc].t().to(bf16)
        return out

    left_w = _pack_proj_weight("left_proj.weight")
    left_g_w = _pack_proj_weight("left_gate.weight")
    right_w = _pack_proj_weight("right_proj.weight")
    right_g_w = _pack_proj_weight("right_gate.weight")
    out_g_w = _pack_proj_weight("out_gate.weight")

    left_t = torch.empty((B, H, M), dtype=bf16, device=device)
    right_t = torch.empty((B, H, M), dtype=bf16, device=device)
    outgate_t = torch.empty((B, H, M), dtype=bf16, device=device)

    prep_bh = max(16, _next_pow2(min(H, 128)))
    prep_bm = 64

    # For small power-of-two channel dimensions, fuse LayerNorm + projections into
    # one kernel and avoid an explicit normalized intermediate.
    use_fused = (C <= 128 and _is_pow2(C))

    if use_fused:
        grid = (B * triton.cdiv(M, prep_bm),)
        bc = C

        _prepare_left_fused_kernel[grid](
            x,
            mask_t,
            weights["norm.weight"],
            weights["norm.bias"],
            left_w,
            left_g_w,
            out_g_w,
            left_t,
            outgate_t,
            M,
            H,
            C,
            BC=bc,
            BM=prep_bm,
            BH=prep_bh,
            HAS_MASK=has_mask,
            num_warps=4,
            num_stages=3,
        )

        _prepare_right_fused_kernel[grid](
            x,
            mask_t,
            weights["norm.weight"],
            weights["norm.bias"],
            right_w,
            right_g_w,
            right_t,
            N,
            M,
            H,
            C,
            BC=bc,
            BM=prep_bm,
            BH=prep_bh,
            HAS_MASK=has_mask,
            num_warps=4,
            num_stages=3,
        )

        x = None
    else:
        # General path: explicit LayerNorm into bf16, then projection/gate kernels
        # with 128-chunk GEMMs so non-power-of-two C values are handled cleanly.
        bc = max(128, _next_pow2(C))
        xn = torch.empty((B, M, C), dtype=bf16, device=device)

        ln_bm = 16 if bc > 128 else 64
        ln_grid = (B * triton.cdiv(M, ln_bm),)
        _layer_norm_kernel[ln_grid](
            x,
            xn,
            weights["norm.weight"],
            weights["norm.bias"],
            M,
            C,
            BC=bc,
            BM=ln_bm,
            num_warps=4,
            num_stages=2,
        )

        x = None

        cchunk = 128
        grid = (B * triton.cdiv(M, prep_bm),)

        _prepare_left_sep_kernel[grid](
            xn,
            mask_t,
            left_w,
            left_g_w,
            out_g_w,
            left_t,
            outgate_t,
            M,
            H,
            C,
            BM=prep_bm,
            BH=prep_bh,
            CCHUNK=cchunk,
            HAS_MASK=has_mask,
            num_warps=4,
            num_stages=3,
        )

        _prepare_right_sep_kernel[grid](
            xn,
            mask_t,
            right_w,
            right_g_w,
            right_t,
            N,
            M,
            H,
            C,
            BM=prep_bm,
            BH=prep_bh,
            CCHUNK=cchunk,
            HAS_MASK=has_mask,
            num_warps=4,
            num_stages=3,
        )

        xn = None

    # Core O(N^3) contraction: for each hidden channel c,
    # out_c = left_c @ right_c^T.
    out_t = torch.empty((B, H, M), dtype=bf16, device=device)

    if N <= 256:
        core_bi = 64
        core_bj = 64
    else:
        core_bi = 128
        core_bj = 128

    core_bi = min(core_bi, _next_pow2(max(16, N)))
    core_bj = min(core_bj, _next_pow2(max(16, N)))
    core_bk = min(64, _next_pow2(max(16, N)))

    num_i = triton.cdiv(N, core_bi)
    num_j = triton.cdiv(N, core_bj)
    core_grid = (B * H * num_i * num_j,)
    core_warps = 8 if core_bi >= 128 else 4

    _core_kernel[core_grid](
        left_t,
        right_t,
        out_t,
        N,
        M,
        H,
        BI=core_bi,
        BJ=core_bj,
        BK=core_bk,
        num_warps=core_warps,
        num_stages=3,
    )

    final_out = torch.empty((B, N, N, C), dtype=torch.float32, device=device)

    # Fast fused final pass for common hidden_dim=128 (and smaller pow2 cases).
    if H in (16, 32, 64, 128):
        w_to_out_t = weights["to_out.weight"].float().t().contiguous().to(bf16)
        final_bm = 64
        final_grid = (B * triton.cdiv(M, final_bm),)

        _final_kernel[final_grid](
            out_t,
            outgate_t,
            weights["to_out_norm.weight"],
            weights["to_out_norm.bias"],
            w_to_out_t,
            final_out,
            M,
            C,
            BM=final_bm,
            FH=H,
            CB=128,
            num_warps=8,
            num_stages=3,
        )
    else:
        # Generic fallback for non-pow2 / large hidden dimensions.
        out = out_t.view(B, H, N, N).permute(0, 2, 3, 1).reshape(B, N, N, H)
        gate = outgate_t.view(B, H, N, N).permute(0, 2, 3, 1).reshape(B, N, N, H)

        out = out.to(torch.float32, copy=True)
        gate = gate.to(torch.float32, copy=True)

        out = F.layer_norm(
            out,
            (H,),
            weights["to_out_norm.weight"].float(),
            weights["to_out_norm.bias"].float(),
            eps=1e-5,
        )
        out = out * gate
        w = weights["to_out.weight"].float()
        out = torch.matmul(out.reshape(B * M, H), w.t()).reshape(B, N, N, C)

        final_out.copy_(out)

    return final_out
