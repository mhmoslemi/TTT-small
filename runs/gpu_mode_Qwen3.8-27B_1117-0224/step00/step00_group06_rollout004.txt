```python
"""
TriMul outgoing forward:
- A fused Triton front-end computes LayerNorm, all linear projections, gating,
  mask application, and directly writes left/right/out_gate in [B, C_z, N, N].
- The N^3 contraction is then executed as a batched GEMM over C_z using cuBLAS.
- A fused Triton epilogue computes the final LayerNorm over C_z, applies
  out_gate, and projects back to [B, N, N, C].
"""

import torch
import triton
import triton.language as tl

# Use bf16 for large tensor-core intermediate tensors. Final output is fp32.
WORK_DTYPE = torch.bfloat16

if hasattr(torch.backends.cuda.matmul, "allow_tf32"):
    torch.backends.cuda.matmul.allow_tf32 = True
if hasattr(torch.backends.cuda.matmul, "allow_bf16_reduced_precision_reduction"):
    torch.backends.cuda.matmul.allow_bf16_reduced_precision_reduction = True


@triton.jit
def _sigmoid(x):
    return 1.0 / (1.0 + tl.exp(-x))


# Helper for input channel width C=384: three 128-wide K chunks.
@triton.jit
def _dot_c384(x0, x1, x2, w_ptr, H, h_start,
              BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    c = tl.arange(0, 128)
    h = h_start + tl.arange(0, BLOCK_H)
    valid_h = h < H
    mask_h = valid_h[None, :]

    w0 = tl.load(w_ptr + c[:, None] * H + h[None, :],
                 mask=mask_h, other=0.0).to(tl.bfloat16)
    acc = tl.dot(x0, w0)

    w1 = tl.load(w_ptr + (128 + c[:, None]) * H + h[None, :],
                 mask=mask_h, other=0.0).to(tl.bfloat16)
    acc += tl.dot(x1, w1)

    w2 = tl.load(w_ptr + (256 + c[:, None]) * H + h[None, :],
                 mask=mask_h, other=0.0).to(tl.bfloat16)
    acc += tl.dot(x2, w2)

    return acc

    nw_ptr, nb_ptr,
    lw_ptr, rw_ptr, lg_ptr, rg_ptr, og_ptr,
    M, N,
    H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    valid_m = m < M

    c = tl.arange(0, 128)
    m64 = m.to(tl.int64)

    x = tl.load(
        x_ptr + m64[:, None] * 128 + c[None, :],
        mask=valid_m[:, None],
        other=0.0
    ).to(tl.float32)

    # Stable LayerNorm over C=128.
    mean = tl.sum(x, axis=1) * (1.0 / 128.0)
    dx = x - mean[:, None]
    var = tl.sum(dx * dx, axis=1) * (1.0 / 128.0)
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    nw = tl.load(nw_ptr + c).to(tl.float32)
    nb = tl.load(nb_ptr + c).to(tl.float32)

    xn = (x - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]
    x16 = xn.to(tl.bfloat16)

    N2 = N * N
    rem = m % N2
    ii = rem // N
    jj = rem - ii * N
    bb = m // N2

    b64 = bb.to(tl.int64)
    i64 = ii.to(tl.int64)
    j64 = jj.to(tl.int64)
    N64 = N.to(tl.int64)
    NN64 = N64 * N64

    base64 = b64 * (H * NN64) + i64 * N64 + j64

    if HAS_MASK:
        mask_off = b64 * NN64 + i64 * N64 + j64
        mask_val = tl.load(mask_ptr + mask_off, mask=valid_m, other=0.0).to(tl.float32)
        use_mask = True
    else:
        use_mask = False

    for hb in range(0, H, BLOCK_H):
        h = hb + tl.arange(0, BLOCK_H)
        valid_h = h < H
        h64 = h.to(tl.int64)
        w_mask = valid_h[None, :]
        store_mask = valid_h[:, None] & valid_m[None, :]

        # left
        lw = tl.load(lw_ptr + c[:, None] * H + h[None, :],
                     mask=w_mask, other=0.0).to(tl.bfloat16)
        left_raw = tl.dot(x16, lw)
        lg = tl.load(lg_ptr + c[:, None] * H + h[None, :],
                     mask=w_mask, other=0.0).to(tl.bfloat16)
        left_gate = _sigmoid(tl.dot(x16, lg))
        if HAS_MASK:
            left_val = left_raw * left_gate * mask_val[:, None]
        else:
            left_val = left_raw * left_gate
        left16 = left_val.to(tl.bfloat16)
        leftT = tl.trans(left16)
        l_ptrs = l_ptr + base64[None, :] + h64[:, None] * NN64 + \
            i64[None, :] * N64 + j64[None, :]
        tl.store(l_ptrs, leftT, mask=store_mask)

        # right
        rw = tl.load(rw_ptr + c[:, None] * H + h[None, :],
                     mask=w_mask, other=0.0).to(tl.bfloat16)
        right_raw = tl.dot(x16, rw)
        rg = tl.load(rg_ptr + c[:, None] * H + h[None, :],
                     mask=w_mask, other=0.0).to(tl.bfloat16)
        right_gate = _sigmoid(tl.dot(x16, rg))
        if HAS_MASK:
            right_val = right_raw * right_gate * mask_val[:, None]
        else:
            right_val = right_raw * right_gate
        right16 = right_val.to(tl.bfloat16)
        rightT = tl.trans(right16)
        r_ptrs = r_ptr + base64[None, :] + h64[:, None] * NN64 + \
            i64[None, :] * N64 + j64[None, :]
        tl.store(r_ptrs, rightT, mask=store_mask)

        # out_gate
        ow = tl.load(og_ptr + c[:, None] * H + h[None, :],
                     mask=w_mask, other=0.0).to(tl.bfloat16)
        out_gate = _sigmoid(tl.dot(x16, ow))
        out16 = out_gate.to(tl.bfloat16)
        outT = tl.trans(out16)
        g_ptrs = g_ptr + base64[None, :] + h64[:, None] * NN64 + \
            i64[None, :] * N64 + j64[None, :]
        tl.store(g_ptrs, outT, mask=store_mask)


@triton.jit
def front_c384_kernel(
    x_ptr, mask_ptr,
    l_ptr, r_ptr, g_ptr,
    nw_ptr, nb_ptr,
    lw_ptr, rw_ptr, lg_ptr, rg_ptr, og_ptr,
    M, N,
    H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    valid_m = m < M

    k = tl.arange(0, 128)
    m64 = m.to(tl.int64)

    x0 = tl.load(
        x_ptr + m64[:, None] * 384 + k[None, :],
        mask=valid_m[:, None], other=0.0
    ).to(tl.float32)
    x1 = tl.load(
        x_ptr + m64[:, None] * 384 + (128 + k[None, :]),
        mask=valid_m[:, None], other=0.0
    ).to(tl.float32)
    x2 = tl.load(
        x_ptr + m64[:, None] * 384 + (256 + k[None, :]),
        mask=valid_m[:, None], other=0.0
    ).to(tl.float32)

    # Stable LayerNorm over C=384.
    s = tl.sum(x0, axis=1) + tl.sum(x1, axis=1) + tl.sum(x2, axis=1)
    mean = s * (1.0 / 384.0)

    d0 = x0 - mean[:, None]
    d1 = x1 - mean[:, None]
    d2 = x2 - mean[:, None]
    var = (tl.sum(d0 * d0, axis=1) + tl.sum(d1 * d1, axis=1) + tl.sum(d2 * d2, axis=1)) * (1.0 / 384.0)
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    nw0 = tl.load(nw_ptr + k).to(tl.float32)
    nw1 = tl.load(nw_ptr + 128 + k).to(tl.float32)
    nw2 = tl.load(nw_ptr + 256 + k).to(tl.float32)
    nb0 = tl.load(nb_ptr + k).to(tl.float32)
    nb1 = tl.load(nb_ptr + 128 + k).to(tl.float32)
    nb2 = tl.load(nb_ptr + 256 + k).to(tl.float32)

    xn0 = (x0 - mean[:, None]) * rstd[:, None] * nw0[None, :] + nb0[None, :]
    xn1 = (x1 - mean[:, None]) * rstd[:, None] * nw1[None, :] + nb1[None, :]
    xn2 = (x2 - mean[:, None]) * rstd[:, None] * nw2[None, :] + nb2[None, :]

    x016 = xn0.to(tl.bfloat16)
    x116 = xn1.to(tl.bfloat16)
    x216 = xn2.to(tl.bfloat16)

    N2 = N * N
    rem = m % N2
    ii = rem // N
    jj = rem - ii * N
    bb = m // N2

    b64 = bb.to(tl.int64)
    i64 = ii.to(tl.int64)
    j64 = jj.to(tl.int64)
    N64 = N.to(tl.int64)
    NN64 = N64 * N64

    base64 = b64 * (H * NN64) + i64 * N64 + j64

    if HAS_MASK:
        mask_off = b64 * NN64 + i64 * N64 + j64
        mask_val = tl.load(mask_ptr + mask_off, mask=valid_m, other=0.0).to(tl.float32)

    for hb in range(0, H, BLOCK_H):
        h = hb + tl.arange(0, BLOCK_H)
        valid_h = h < H
        h64 = h.to(tl.int64)
        store_mask = valid_h[:, None] & valid_m[None, :]

        # left
        left_raw = _dot_c384(x016, x116, x216, lw_ptr, H, hb, BLOCK_M, BLOCK_H)
        left_gate = _sigmoid(_dot_c384(x016, x116, x216, lg_ptr, H, hb, BLOCK_M, BLOCK_H))
        if HAS_MASK:
            left_val = left_raw * left_gate * mask_val[:, None]
        else:
            left_val = left_raw * left_gate
        left16 = left_val.to(tl.bfloat16)
        leftT = tl.trans(left16)
        l_ptrs = l_ptr + base64[None, :] + h64[:, None] * NN64 + \
            i64[None, :] * N64 + j64[None, :]
        tl.store(l_ptrs, leftT, mask=store_mask)

        # right
        right_raw = _dot_c384(x016, x116, x216, rw_ptr, H, hb, BLOCK_M, BLOCK_H)
        right_gate = _sigmoid(_dot_c384(x016, x116, x216, rg_ptr, H, hb, BLOCK_M, BLOCK_H))
        if HAS_MASK:
            right_val = right_raw * right_gate * mask_val[:, None]
        else:
            right_val = right_raw * right_gate
        right16 = right_val.to(tl.bfloat16)
        rightT = tl.trans(right16)
        r_ptrs = r_ptr + base64[None, :] + h64[:, None] * NN64 + \
            i64[None, :] * N64 + j64[None, :]
        tl.store(r_ptrs, rightT, mask=store_mask)

        # out_gate
        out_gate = _sigmoid(_dot_c384(x016, x116, x216, og_ptr, H, hb, BLOCK_M, BLOCK_H))
        out16 = out_gate.to(tl.bfloat16)
        outT = tl.trans(out16)
        g_ptrs = g_ptr + base64[None, :] + h64[:, None] * NN64 + \
            i64[None, :] * N64 + j64[None, :]
        tl.store(g_ptrs, outT, mask=store_mask)


@triton.jit
def fuse_transpose_kernel(
    p_ptr, mask_ptr,
    l_ptr, r_ptr, g_ptr,
    M, N,
    H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """Fallback: reads fused projection output P [B,N,N,5H] and writes
    L/R/G [B,H,N,N], applying mask and sigmoid gates."""
    pid = tl.program_id(0)
    m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    valid_m = m < M

    m64 = m.to(tl.int64)

    N2 = N * N
    rem = m % N2
    ii = rem // N
    jj = rem - ii * N
    bb = m // N2

    b64 = bb.to(tl.int64)
    i64 = ii.to(tl.int64)
    j64 = jj.to(tl.int64)
    N64 = N.to(tl.int64)
    NN64 = N64 * N64

    base64 = b64 * (H * NN64) + i64 * N64 + j64

    if HAS_MASK:
        mask_off = b64 * NN64 + i64 * N64 + j64
        mask_val = tl.load(mask_ptr + mask_off, mask=valid_m, other=0.0).to(tl.float32)

    for hb in range(0, H, BLOCK_H):
        h = hb + tl.arange(0, BLOCK_H)
        valid_h = h < H
        h64 = h.to(tl.int64)
        load_mask = valid_m[:, None] & valid_h[None, :]
        store_mask = valid_h[:, None] & valid_m[None, :]

        p_base = m64[:, None] * (5 * H) + h[None, :]

        l_raw = tl.load(p_ptr + p_base, mask=load_mask, other=0.0).to(tl.float32)
        l_gate = tl.load(p_ptr + p_base + (2 * H), mask=load_mask, other=0.0).to(tl.float32)
        if HAS_MASK:
            l_val = l_raw * _sigmoid(l_gate) * mask_val[:, None]
        else:
            l_val = l_raw * _sigmoid(l_gate)
        l16 = l_val.to(tl.bfloat16)
        lT = tl.trans(l16)
        l_ptrs = l_ptr + base64[None, :] + h64[:, None] * NN64 + \
            i64[None, :] * N64 + j64[None, :]
        tl.store(l_ptrs, lT, mask=store_mask)

        r_raw = tl.load(p_ptr + p_base + H, mask=load_mask, other=0.0).to(tl.float32)
        r_gate = tl.load(p_ptr + p_ptr + p_base + (3 * H), mask=load_mask, other=0.0).to(tl.float32)
        if HAS_MASK:
            r_val = r_raw * _sigmoid(r_gate) * mask_val[:, None]
        else:
            r_val = r_raw * _sigmoid(r_gate)
        r16 = r_val.to(tl.bfloat16)
        rT = tl.trans(r16)
        r_ptrs = r_ptr + base64[None, :] + h64[:, None] * NN64 + \
            i64[None, :] * N64 + j64[None, :]
        tl.store(r_ptrs, rT, mask=store_mask)

        o_gate = tl.load(p_ptr + p_base + (4 * H), mask=load_mask, other=0.0).to(tl.float32)
        o16 = _sigmoid(o_gate).to(tl.bfloat16)
        oT = tl.trans(o16)
        g_ptrs = g_ptr + base64[None, :] + h64[:, None] * NN64 + \
            i64[None, :] * N64 + j64[None, :]
        tl.store(g_ptrs, oT, mask=store_mask)


@triton.jit
def final_kernel(
    out_ptr, gate_ptr, y_ptr,
    wn_ptr, wb_ptr, wto_ptr,
    N,
    H: tl.constexpr,
    C: tl.constexpr,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    """Fused final LayerNorm over C_z, gating by out_gate, and to_out projection.
    out/gate are [B, H, N, N], output is [B, N, N, C]."""
    pid = tl.program_id(0)

    num_j = tl.cdiv(N, BLOCK_J)
    num_i = tl.cdiv(N, BLOCK_I)
    tiles_per_b = num_i * num_j

    b = pid // tiles_per_b
    local = pid % tiles_per_b
    pi = local // num_j
    pj = local % num_j

    ROWS = BLOCK_I * BLOCK_J
    r = tl.arange(0, ROWS)
    ili = r // BLOCK_J
    ilj = r - ili * BLOCK_J

    i = pi * BLOCK_I + ili
    j = pj * BLOCK_J + ilj
    valid = (i < N) & (j < N)

    b64 = b.to(tl.int64)
    i64 = i.to(tl.int64)
    j64 = j.to(tl.int64)
    N64 = N.to(tl.int64)
    NN64 = N64 * N64

    row = i64 * N64 + j64
    out_base = b64 * (H * NN64)
    y_base = b64 * (C * NN64) + row * C

    # Fast path for the common H=128 case: load out/gate once into registers.
    if H == BLOCK_H:
        h = tl.arange(0, BLOCK_H)
        valid_h = h < H
        load_mask = valid_h[:, None] & valid[None, :]
        out_ptrs = out_ptr + out_base + h[:, None] * NN64 + row[None, :]
        gate_ptrs = gate_ptr + out_base + h[:, None] * NN64 + row[None, :]

        a = tl.load(out_ptrs, mask=load_mask, other=0.0).to(tl.float32)
        g = tl.load(gate_ptrs, mask=load_mask, other=0.0).to(tl.float32)

        mean = tl.sum(a, axis=0) * (1.0 / H)
        d = a - mean[None, :]
        var = tl.sum(d * d, axis=0) * (1.0 / H)
        var = tl.maximum(var, 0.0)
        rstd = 1.0 / tl.sqrt(var + 1e-5)

        hw = tl.load(wn_ptr + h, mask=valid_h, other=0.0).to(tl.float32)
        hb = tl.load(wb_ptr + h, mask=valid_h, other=0.0).to(tl.float32)

        a = (a - mean[None, :]) * rstd[None, :] * hw[:, None] + hb[:, None]
        a = a * g
        a16 = tl.trans(a).to(tl.bfloat16)

        for cb in range(0, C, BLOCK_C):
            c = cb + tl.arange(0, BLOCK_C)
            valid_c = c < C
            w = tl.load(
                wto_ptr + h[:, None] * C + c[None, :],
                mask=valid_h[:, None] & valid_c[None, :],
                other=0.0
            ).to(tl.bfloat16)

            acc = tl.dot(a16, w)
            y_ptrs = y_ptr + y_base[:, None] + c.to(tl.int64)[None, :]
            tl.store(
                y_ptrs,
                acc.to(tl.float32),
                mask=valid[:, None] & valid_c[None, :]
            )
    else:
        # General path for H > 128: first pass computes LN stats,
        # second pass normalizes, gates, and accumulates the to_out projection.
        s = tl.zeros((ROWS,), tl.float32)
        ss = tl.zeros((ROWS,), tl.float32)

        for hb in range(0, H, BLOCK_H):
            h = hb + tl.arange(0, BLOCK_H)
            valid_h = h < H
            load_mask = valid_h[:, None] & valid[None, :]
            out_ptrs = out_ptr + out_base + h[:, None] * NN64 + row[None, :]
            a = tl.load(out_ptrs, mask=load_mask, other=0.0).to(tl.float32)
            s += tl.sum(a, axis=0)
            ss += tl.sum(a * a, axis=0)

        mean = s * (1.0 / H)
        var = ss * (1.0 / H) - mean * mean
        var = tl.maximum(var, 0.0)
        rstd = 1.0 / tl.sqrt(var + 1e-5)

        for cb in range(0, C, BLOCK_C):
            c = cb + tl.arange(0, BLOCK_C)
            valid_c = c < C
            acc = tl.zeros((ROWS, BLOCK_C), tl.float32)

            for hb in range(0, H, BLOCK_H):
                h = hb + tl.arange(0, BLOCK_H)
                valid_h = h < H
                load_mask = valid_h[:, None] & valid[None, :]

                out_ptrs = out_ptr + out_base + h[:, None] * NN64 + row[None, :]
                gate_ptrs = gate_ptr + out_base + h[:, None] * NN64 + row[None, :]

                a = tl.load(out_ptrs, mask=load_mask, other=0.0).to(tl.float32)
                g = tl.load(gate_ptrs, mask=load_mask, other=0.0).to(tl.float32)

                hw = tl.load(wn_ptr + h, mask=valid_h, other=0.0).to(tl.float32)
                hb = tl.load(wb_ptr + h, mask=valid_h, other=0.0).to(tl.float32)

                a = (a - mean[None, :]) * rstd[None, :] * hw[:, None] + hb[:, None]
                a = a * g
                a16 = tl.trans(a).to(tl.bfloat16)

                w = tl.load(
                    wto_ptr + h[:, None] * C + c[None, :],
                    mask=valid_h[:, None] & valid_c[None, :],
                    other=0.0
                ).to(tl.bfloat16)

                acc += tl.dot(a16, w)

            y_ptrs = y_ptr + y_base[:, None] + c.to(tl.int64)[None, :]
            tl.store(
                y_ptrs,
                acc.to(tl.float32),
                mask=valid[:, None] & valid_c[None, :]
            )


def _torch_final(out, gate, wn, wb, wto, B, N, H, C):
    # Avoided in normal test shapes; kept only as a dimensional fallback.
    out_n = out.permute(0, 2, 3, 1).reshape(B, N, N, H).contiguous()
    gate_n = gate.permute(0, 2, 3, 1).reshape(B, N, N, H).contiguous()
    out_n = torch.nn.functional.layer_norm(out_n, (H,), wn, wb)
    out_n = out_n * gate_n
    y = torch.matmul(out_n.reshape(-1, H), wto.t()).view(B, N, N, C)
    return y.float()


def custom_kernel(data):
    if len(data) == 4:
        x, mask, weights, config = data
    elif len(data) == 3:
        a, b, c = data
        if isinstance(b, dict):
            x, weights, config = a, b, c
            mask = None
        else:
            x, mask, config = a, b, c
            weights = config.get("weights", {})
    else:
        raise ValueError("Invalid data tuple length")

    config = {} if config is None else config
    x = x.contiguous()
    B, N1, N2, C = x.shape
    N = N1
    H = int(config.get("hidden_dim", weights.get("to_out.weight", torch.empty(1, 0)).shape[1]))

    M = B * N * N
    device = x.device

    # Parse / normalize mask.
    if mask is not None:
        mask = mask.contiguous()
        if mask.dim() == 2:
            mask = mask.unsqueeze(0).expand(B, N, N).contiguous()
        elif mask.dim() == 3 and mask.shape[0] == 1 and B != 1:
            mask = mask.expand(B, N, N).contiguous()
        elif mask.numel() == N * N:
            mask = mask.reshape(N, N).unsqueeze(0).expand(B, N, N).contiguous()
        elif mask.numel() == M:
            mask = mask.reshape(B, N, N).contiguous()
        else:
            mask = mask.reshape(B, N, N)
        has_mask = not bool(config.get("nomask", False))
        if mask.shape != (B, N, N):
            has_mask = False
    else:
        has_mask = False

    # Weights / parameters.
    nw = weights["norm.weight"].float().contiguous()
    nb = weights["norm.bias"].float().contiguous()
    wn = weights["to_out_norm.weight"].float().contiguous()
    wb = weights["to_out_norm.bias"].float().contiguous()
    wto_t = weights["to_out.weight"].t().contiguous().to(WORK_DTYPE)

    L = torch.empty((B, H, N, N), device=device, dtype=WORK_DTYPE)
    R = torch.empty((B, H, N, N), device=device, dtype=WORK_DTYPE)
    G = torch.empty((B, H, N, N), device=device, dtype=WORK_DTYPE)

    dummy = x  # Pointer argument for unused mask when HAS_MASK=False.

    use_custom_front = (C == 128) or (C == 384)

    if use_custom_front:
        BLOCK_H = 128
        if C == 128:
            BLOCK_M = 128
            lw = weights["left_proj.weight"].t().contiguous().to(WORK_DTYPE)
            rw = weights["right_proj.weight"].t().contiguous().to(WORK_DTYPE)
            lg = weights["left_gate.weight"].t().contiguous().to(WORK_DTYPE)
            rg = weights["right_gate.weight"].t().contiguous().to(WORK_DTYPE)
            og = weights["out_gate.weight"].t().contiguous().to(WORK_DTYPE)

            grid = (triton.cdiv(M, BLOCK_M),)
            front_c128_kernel[grid](
                x, dummy,
                L, R, G,
                nw, nb,
                lw, rw, lg, rg, og,
                M, N,
                H=H,
                HAS_MASK=has_mask,
                BLOCK_M=BLOCK_M,
                BLOCK_H=BLOCK_H,
                num_warps=8,
            )
        else:
            BLOCK_M = 64
            lw = weights["left_proj.weight"].t().contiguous().to(WORK_DTYPE)
            rw = weights["right_proj.weight"].t().contiguous().to(WORK_DTYPE)
            lg = weights["left_gate.weight"].t().contiguous().to(WORK_DTYPE)
            rg = weights["right_gate.weight"].t().contiguous().to(WORK_DTYPE)
            og = weights["out_gate.weight"].t().contiguous().to(WORK_DTYPE)

            grid = (triton.cdiv(M, BLOCK_M),)
            front_c384_kernel[grid](
                x, dummy,
                L, R, G,
                nw, nb,
                lw, rw, lg, rg, og,
                M, N,
                H=H,
                HAS_MASK=has_mask,
                BLOCK_M=BLOCK_M,
                BLOCK_H=BLOCK_H,
                num_warps=8,
            )
    else:
        # Fallback front-end: torch for LayerNorm/projections + Triton fused transpose.
        x_norm = torch.nn.functional.layer_norm(x, (C,), nw, nb)
        xw = x_norm.reshape(M, C).to(WORK_DTYPE)

        W_cat = torch.cat(
            [
                weights["left_proj.weight"],
                weights["right_proj.weight"],
                weights["left_gate.weight"],
                weights["right_gate.weight"],
                weights["out_gate.weight"],
            ],
            dim=0,
        ).to(WORK_DTYPE).t().contiguous()

        P = torch.mm(xw, W_cat).view(B, N, N, 5 * H)

        BLOCK_M = 128
        BLOCK_H = 128
        grid = (triton.cdiv(M, BLOCK_M),)
        fuse_transpose_kernel[grid](
            P, dummy,
            L, R, G,
            M, N,
            H=H,
            HAS_MASK=has_mask,
            BLOCK_M=BLOCK_M,
            BLOCK_H=BLOCK_H,
            num_warps=4,
        )
        P = None

    # Core triangular multiplication:
    # For each h: L_h [N,N] @ R_h^T [N,N] -> out_h [N,N].
    Lb = L.view(B * H, N, N)
    Rb = R.view(B * H, N, N)
    out = torch.bmm(Lb, Rb.transpose(1, 2))
    out = out.view(B, H, N, N)

    L = None
    R = None

    # Final fused epilogue.
    y = torch.empty((B, N, N, C), device=device, dtype=torch.float32)

    # Choose tiles that keep the contraction to_out at M=256 while maximizing
    # large contiguous loads from [B,H,N,N].
    if N % 256 == 0:
        BLOCK_I = 1
        BLOCK_J = 256
    elif N % 128 == 0:
        BLOCK_I = 2
        BLOCK_J = 128
    else:
        BLOCK_I = 16
        BLOCK_J = 16

    grid = (B * triton.cdiv(N, BLOCK_I) * triton.cdiv(N, BLOCK_J),)
    final_kernel[grid](
        out, G, y,
        wn, wb, wto_t,
        N,
        H=H,
        C=C,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=128,
        BLOCK_C=128,
        num_warps=8,
        num_stages=2,
    )

    return y
```
