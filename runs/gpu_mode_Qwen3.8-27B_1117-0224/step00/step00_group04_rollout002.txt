```python
"""
Fused Triton implementation of outgoing TriMul.

Pipeline:
1. Triton LayerNorm on the last dimension, producing bf16 x.
2. Triton fused projection+gate kernels:
   - left  = (x @ W_left^T) * sigmoid(x @ W_left_gate^T) * mask
   - right = (x @ W_right^T) * sigmoid(x @ W_right_gate^T) * mask
   - out_gate = sigmoid(x @ W_out_gate^T)
   left/right are written directly in [B, H, N, N] layout for cuBLAS bmm.
3. cuBLAS bf16 batched GEMM computes the pair term:
   pair[b,h,i,j] = sum_k left[b,h,i,k] * right[b,h,j,k]
4. Fused Triton kernel does LayerNorm over H, multiplication by out_gate,
   and projection to the output dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


def _next_pow2(n: int) -> int:
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


@triton.jit
def _layernorm_rows_kernel(
    x_ptr,
    w_ptr,
    b_ptr,
    out_ptr,
    M,
    C,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid_m = tl.program_id(0)
    b = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M
    offs_c = tl.arange(0, BLOCK_C)
    mask_c = offs_c < C

    x_ptrs = x_ptr + (b * M + offs_m)[:, None] * C + offs_c[None, :]
    x = tl.load(x_ptrs, mask=mask_m[:, None] & mask_c[None, :], other=0.0).to(tl.float32)

    mean = tl.sum(x, axis=1) / C
    xc = x - mean[:, None]
    xc = tl.where(mask_c[None, :], xc, 0.0)
    var = tl.sum(xc * xc, axis=1) / C
    rstd = 1.0 / tl.sqrt(var + eps)

    w = tl.load(w_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
    bias = tl.load(b_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)

    y = xc * rstd[:, None] * w[None, :] + bias[None, :]

    out_ptrs = out_ptr + (b * M + offs_m)[:, None] * C + offs_c[None, :]
    tl.store(out_ptrs, y.to(tl.bfloat16), mask=mask_m[:, None] & mask_c[None, :])


@triton.jit
def _proj_gate_transposed_kernel(
    x_ptr,
    wp_ptr,
    wg_ptr,
    mask_ptr,
    out_ptr,
    M,
    C,
    H,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_m = tl.program_id(0)
    b = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    acc_proj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    for k0 in range(0, tl.cdiv(C, BLOCK_K)):
        offs_k = k0 * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < C

        x_ptrs = x_ptr + (b * M + offs_m)[:, None] * C + offs_k[None, :]
        x = tl.load(x_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        wp_ptrs = wp_ptr + offs_k[:, None] + offs_h[None, :] * C
        wg_ptrs = wg_ptr + offs_k[:, None] + offs_h[None, :] * C

        wp = tl.load(wp_ptrs, mask=mask_k[:, None] & mask_h[None, :], other=0.0)
        wg = tl.load(wg_ptrs, mask=mask_k[:, None] & mask_h[None, :], other=0.0)

        acc_proj = tl.dot(x, wp, acc_proj)
        acc_gate = tl.dot(x, wg, acc_gate)

    gate = 1.0 / (1.0 + tl.exp(-acc_gate))
    val = acc_proj * gate

    if HAS_MASK:
        mval = tl.load(mask_ptr + b * M + offs_m, mask=mask_m, other=0.0).to(tl.float32)
        val = val * mval[:, None]

    out_val = val.to(tl.bfloat16)
    out_ptrs = out_ptr + b * (H * M) + offs_h[:, None] * M + offs_m[None, :]
    tl.store(out_ptrs, tl.trans(out_val), mask=mask_h[:, None] & mask_m[None, :])


@triton.jit
def _gate_kernel(
    x_ptr,
    wg_ptr,
    out_ptr,
    M,
    C,
    H,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)
    b = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    for k0 in range(0, tl.cdiv(C, BLOCK_K)):
        offs_k = k0 * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < C

        x_ptrs = x_ptr + (b * M + offs_m)[:, None] * C + offs_k[None, :]
        x = tl.load(x_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        wg_ptrs = wg_ptr + offs_k[:, None] + offs_h[None, :] * C
        wg = tl.load(wg_ptrs, mask=mask_k[:, None] & mask_h[None, :], other=0.0)

        acc = tl.dot(x, wg, acc)

    y = 1.0 / (1.0 + tl.exp(-acc))

    out_ptrs = out_ptr + (b * M + offs_m)[:, None] * H + offs_h[None, :]
    tl.store(out_ptrs, y.to(tl.bfloat16), mask=mask_m[:, None] & mask_h[None, :])


@triton.jit
def _final_ln_gate_mm_kernel(
    pair_ptr,
    gate_ptr,
    w_ptr,
    lnw_ptr,
    lnb_ptr,
    out_ptr,
    M,
    H,
    C,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid_m = tl.program_id(0)
    b = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # pair is [B, H, M]
    pair_ptrs = pair_ptr + b * (H * M) + offs_h[:, None] * M + offs_m[None, :]
    pair = tl.load(pair_ptrs, mask=mask_h[:, None] & mask_m[None, :], other=0.0).to(tl.float32)

    mean = tl.sum(pair, axis=0) / H
    pc = pair - mean[None, :]
    pc = tl.where(mask_h[:, None], pc, 0.0)
    var = tl.sum(pc * pc, axis=0) / H
    var = tl.where(var < 0.0, 0.0, var)
    rstd = 1.0 / tl.sqrt(var + eps)

    lnw = tl.load(lnw_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)
    lnb = tl.load(lnb_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)

    normed = pc * rstd[None, :] * lnw[:, None] + lnb[:, None]

    # gate is [B, M, H]
    gate_ptrs = gate_ptr + b * (M * H) + offs_m[:, None] * H + offs_h[None, :]
    gate = tl.load(gate_ptrs, mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

    y = tl.trans(normed) * gate
    y_bf = y.to(tl.bfloat16)

    for c0 in range(0, tl.cdiv(C, BLOCK_C)):
        offs_c = c0 * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C

        # to_out.weight is [C, H]; load as [H, C]
        w_ptrs = w_ptr + offs_h[:, None] + offs_c[None, :] * H
        w = tl.load(w_ptrs, mask=mask_h[:, None] & mask_c[None, :], other=0.0)

        acc = tl.dot(y_bf, w)

        out_ptrs = out_ptr + b * (M * C) + offs_m[:, None] * C + offs_c[None, :]
        tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_c[None, :])


def custom_kernel(data):
    if len(data) == 4:
        input_tensor, mask, weights, config = data
    else:
        input_tensor, weights, config = data
        mask = None

    with torch.no_grad():
        B, S, S2, C = input_tensor.shape
        M = S * S
        H = int(config.get("hidden_dim", weights["left_proj.weight"].shape[0]))
        device = input_tensor.device
        eps = 1e-5

        x = input_tensor.contiguous()

        has_mask = (
            mask is not None
            and mask.numel() > 0
            and not config.get("nomask", False)
        )
        if has_mask:
            mask = mask.contiguous()
        else:
            mask = None

        def _prepare(name, dtype):
            t = weights[name]
            if t.device != device:
                t = t.to(device)
            if t.dtype != dtype:
                t = t.to(dtype)
            if not t.is_contiguous():
                t = t.contiguous()
            return t

        norm_w = _prepare("norm.weight", torch.float32)
        norm_b = _prepare("norm.bias", torch.float32)

        left_w = _prepare("left_proj.weight", torch.bfloat16)
        right_w = _prepare("right_proj.weight", torch.bfloat16)
        left_gw = _prepare("left_gate.weight", torch.bfloat16)
        right_gw = _prepare("right_gate.weight", torch.bfloat16)
        out_gw = _prepare("out_gate.weight", torch.bfloat16)

        to_out_w = _prepare("to_out.weight", torch.bfloat16)
        to_out_norm_w = _prepare("to_out_norm.weight", torch.float32)
        to_out_norm_b = _prepare("to_out_norm.bias", torch.float32)

        # Almost all deployed cases in this task use hidden_dim=128.
        # Keep a compact fast Triton path for that, and a correct PyTorch fallback otherwise.
        use_fast = 16 <= H <= 128

        if use_fast:
            # ------------------------------------------------------------
            # 1. LayerNorm x -> xn (bf16)
            # ------------------------------------------------------------
            xn = torch.empty((B, S, S, C), dtype=torch.bfloat16, device=device)
            BLOCK_M_LN = 32
            BLOCK_C_LN = _next_pow2(C)
            _layernorm_rows_kernel[(triton.cdiv(M, BLOCK_M_LN), B)](
                x, norm_w, norm_b, xn, M, C, eps,
                BLOCK_M=BLOCK_M_LN, BLOCK_C=BLOCK_C_LN,
            )
            del x

            # ------------------------------------------------------------
            # 2. Fused projection + gate, writing left/right in [B, H, N, N]
            # ------------------------------------------------------------
            left_t = torch.empty((B, H, S, S), dtype=torch.bfloat16, device=device)
            right_t = torch.empty((B, H, S, S), dtype=torch.bfloat16, device=device)
            gate = torch.empty((B, S, S, H), dtype=torch.bfloat16, device=device)

            BLOCK_M_P = 64
            BLOCK_H_P = max(16, _next_pow2(H))
            BLOCK_K_P = 128 if C >= 128 else max(16, _next_pow2(C))

            grid_p = (triton.cdiv(M, BLOCK_M_P), B)
            mask_arg = mask if has_mask else xn

            _proj_gate_transposed_kernel[grid_p](
                xn, left_w, left_gw, mask_arg, left_t,
                M, C, H,
                BLOCK_M=BLOCK_M_P,
                BLOCK_H=BLOCK_H_P,
                BLOCK_K=BLOCK_K_P,
                HAS_MASK=has_mask,
            )

            _proj_gate_transposed_kernel[grid_p](
                xn, right_w, right_gw, mask_arg, right_t,
                M, C, H,
                BLOCK_M=BLOCK_M_P,
                BLOCK_H=BLOCK_H_P,
                BLOCK_K=BLOCK_K_P,
                HAS_MASK=has_mask,
            )

            _gate_kernel[grid_p](
                xn, out_gw, gate,
                M, C, H,
                BLOCK_M=BLOCK_M_P,
                BLOCK_H=BLOCK_H_P,
                BLOCK_K=BLOCK_K_P,
            )

            del xn

            # ------------------------------------------------------------
            # 3. Pair term via cuBLAS batched GEMM.
            #    left_t/right_t are contiguous [B, H, S, S] and can be seen
            #    as [B*H, S, S].
            # ------------------------------------------------------------
            batch = B * H
            left_3d = left_t.view(batch, S, S)
            right_3d = right_t.view(batch, S, S)
            pair_3d = torch.empty((batch, S, S), dtype=torch.bfloat16, device=device)
            torch.bmm(left_3d, right_3d.transpose(1, 2), out=pair_3d)
            del left_t, right_t

            pair = pair_3d.view(B, H, S, S)

            # ------------------------------------------------------------
            # 4. Fused LayerNorm + gate + to_out projection.
            # ------------------------------------------------------------
            out = torch.empty((B, S, S, C), dtype=torch.float32, device=device)

            BLOCK_M_F = 64
            BLOCK_H_F = max(16, _next_pow2(H))
            BLOCK_C_F = 128 if C >= 128 else max(16, _next_pow2(C))

            _final_ln_gate_mm_kernel[(triton.cdiv(M, BLOCK_M_F), B)](
                pair, gate, to_out_w, to_out_norm_w, to_out_norm_b, out,
                M, H, C, eps,
                BLOCK_M=BLOCK_M_F,
                BLOCK_H=BLOCK_H_F,
                BLOCK_C=BLOCK_C_F,
            )

            return out

        # ----------------------------------------------------------------
        # Fallback for larger hidden dimensions: keep Triton LayerNorm,
        # use PyTorch for the remaining ops.
        # ----------------------------------------------------------------
        xn = torch.empty((B, S, S, C), dtype=torch.bfloat16, device=device)
        BLOCK_M_LN = 32
        BLOCK_C_LN = _next_pow2(C)
        _layernorm_rows_kernel[(triton.cdiv(M, BLOCK_M_LN), B)](
            x, norm_w, norm_b, xn, M, C, eps,
            BLOCK_M=BLOCK_M_LN, BLOCK_C=BLOCK_C_LN,
        )
        del x

        xn_2d = xn.reshape(B * M, C)

        def _lin(w):
            return torch.matmul(xn_2d, w.T)

        mask_bcast = None
        if has_mask:
            mask_bcast = mask.to(torch.bfloat16).view(B, M, 1)

        lp = _lin(left_w).view(B, M, H)
        lg = torch.sigmoid(_lin(left_gw).to(torch.float32)).to(torch.bfloat16)
        if mask_bcast is not None:
            lp = lp * mask_bcast
        left = lp * lg
        del lp, lg

        rp = _lin(right_w).view(B, M, H)
        rg = torch.sigmoid(_lin(right_gw).to(torch.float32)).to(torch.bfloat16)
        if mask_bcast is not None:
            rp = rp * mask_bcast
        right = rp * rg
        del rp, rg

        out_gate = torch.sigmoid(_lin(out_gw).to(torch.float32)).to(torch.bfloat16)
        del xn, xn_2d

        left_t = left.reshape(B, S, S, H).permute(0, 3, 1, 2).contiguous()
        right_t = right.reshape(B, S, S, H).permute(0, 3, 1, 2).contiguous()
        del left, right

        batch = B * H
        left_3d = left_t.view(batch, S, S)
        right_3d = right_t.view(batch, S, S)
        pair_3d = torch.empty((batch, S, S), dtype=torch.bfloat16, device=device)
        torch.bmm(left_3d, right_3d.transpose(1, 2), out=pair_3d)
        del left_t, right_t, left_3d, right_3d

        pair_n = pair_3d.view(B, H, S, S).permute(0, 2, 1).reshape(B, M, H).to(torch.float32)
        del pair_3d

        y = F.layer_norm(pair_n, (H,), to_out_norm_w, to_out_norm_b)
        y = y * out_gate.view(B, M, H)
        del pair_n, out_gate

        out = torch.matmul(
            y.to(torch.bfloat16).reshape(B * M, H),
            to_out_w.T,
        ).to(torch.float32).view(B, S, S, C)

        return out
```
