import torch
import triton
import triton.language as tl


def _next_pow2(n: int) -> int:
    n = int(n)
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


@triton.jit
def _layer_norm_kernel(
    x_ptr,
    y_ptr,
    w_ptr,
    b_ptr,
    M,
    D,
    eps,
    BLOCK_D: tl.constexpr,
    BLOCK_R: tl.constexpr,
):
    rows = (tl.program_id(0) * BLOCK_R + tl.arange(0, BLOCK_R)).to(tl.int64)
    cols = tl.arange(0, BLOCK_D)

    row_mask = rows < M
    col_mask = cols < D
    mask = row_mask[:, None] & col_mask[None, :]

    x = tl.load(x_ptr + rows[:, None] * D + cols[None, :], mask=mask, other=0.0).to(tl.float32)

    mean = tl.sum(x, axis=1) / D
    diff = x - mean[:, None]
    diff = tl.where(col_mask[None, :], diff, 0.0)

    var = tl.sum(diff * diff, axis=1) / D
    rstd = 1.0 / tl.sqrt(var + eps)

    w = tl.load(w_ptr + cols, mask=col_mask, other=0.0).to(tl.float32)
    bias = tl.load(b_ptr + cols, mask=col_mask, other=0.0).to(tl.float32)

    y = (diff * rstd[:, None]) * w[None, :] + bias[None, :]

    tl.store(y_ptr + rows[:, None] * D + cols[None, :], y.to(y_ptr.dtype.element_ty), mask=mask)


@triton.jit
def _elementwise_kernel(
    proj_ptr,
    mask_ptr,
    left_ptr,
    right_ptr,
    gate_ptr,
    M,
    H,
    HAS_MASK: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_R: tl.constexpr,
):
    rows = (tl.program_id(0) * BLOCK_R + tl.arange(0, BLOCK_R)).to(tl.int64)
    h = tl.arange(0, BLOCK_H)

    row_mask = rows < M
    h_mask = h < H
    mask = row_mask[:, None] & h_mask[None, :]

    base = proj_ptr + rows[:, None] * (5 * H)

    lp = tl.load(base + h[None, :], mask=mask, other=0.0).to(tl.float32)
    rp = tl.load(base + H + h[None, :], mask=mask, other=0.0).to(tl.float32)
    lg = tl.load(base + 2 * H + h[None, :], mask=mask, other=0.0).to(tl.float32)
    rg = tl.load(base + 3 * H + h[None, :], mask=mask, other=0.0).to(tl.float32)
    og = tl.load(base + 4 * H + h[None, :], mask=mask, other=0.0).to(tl.float32)

    if HAS_MASK:
        mt = tl.load(mask_ptr + rows, mask=row_mask, other=0.0).to(tl.float32)[:, None]
    else:
        mt = 1.0

    sil = 1.0 / (1.0 + tl.exp(-lg))
    sir = 1.0 / (1.0 + tl.exp(-rg))
    sog = 1.0 / (1.0 + tl.exp(-og))

    left = lp * mt * sil
    right = rp * mt * sir
    gate = sog

    out_ptrs = rows[:, None] * H + h[None, :]
    tl.store(left_ptr + out_ptrs, left.to(tl.bfloat16), mask=mask)
    tl.store(right_ptr + out_ptrs, right.to(tl.bfloat16), mask=mask)
    tl.store(gate_ptr + out_ptrs, gate.to(tl.bfloat16), mask=mask)


@triton.jit
def _final_kernel(
    in_ptr,
    gate_ptr,
    nw_ptr,
    nb_ptr,
    w_ptr,
    o_ptr,
    M,
    H,
    D,
    eps,
    BLOCK_H: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_DO: tl.constexpr,
):
    rows = (tl.program_id(0) * BLOCK_R + tl.arange(0, BLOCK_R)).to(tl.int64)
    h = tl.arange(0, BLOCK_H)

    row_mask = rows < M
    h_mask = h < H
    mask = row_mask[:, None] & h_mask[None, :]

    x = tl.load(in_ptr + rows[:, None] * H + h[None, :], mask=mask, other=0.0).to(tl.float32)

    mean = tl.sum(x, axis=1) / H
    diff = x - mean[:, None]
    diff = tl.where(h_mask[None, :], diff, 0.0)

    var = tl.sum(diff * diff, axis=1) / H
    rstd = 1.0 / tl.sqrt(var + eps)

    nw = tl.load(nw_ptr + h, mask=h_mask, other=0.0).to(tl.float32)
    nb = tl.load(nb_ptr + h, mask=h_mask, other=0.0).to(tl.float32)

    normed = (diff * rstd[:, None]) * nw[None, :] + nb[None, :]
    normed = tl.where(h_mask[None, :], normed, 0.0)

    g = tl.load(gate_ptr + rows[:, None] * H + h[None, :], mask=mask, other=0.0).to(tl.float32)
    g = tl.where(h_mask[None, :], g, 0.0)

    hval = (normed * g).to(tl.bfloat16)

    d = tl.arange(0, BLOCK_DO)
    for d0 in range(0, D, BLOCK_DO):
        do = d0 + d
        do_mask = do < D

        wch = tl.load(
            w_ptr + do[None, :] * H + h[:, None],
            mask=(do_mask[None, :] & h_mask[:, None]),
            other=0.0,
        ).to(tl.bfloat16)

        acc = tl.dot(hval, wch)

        tl.store(
            o_ptr + rows[:, None] * D + do[None, :],
            acc.to(tl.float32),
            mask=row_mask[:, None] & do_mask[None, :],
        )


def custom_kernel(data):
    """
    Fused Triton+PyTorch TriMul forward:
      - Triton LayerNorm on [B, N, N, C]
      - One bf16 GEMM computes all five linear projections/gates
      - Triton elementwise applies sigmoid gates and mask, producing left/right/gate
      - Batched bmm computes the outgoing TriMul contraction sum_k left[i,k,d] * right[j,k,d]
      - Triton fused final LayerNorm * gate * linear projection, output float32
    """
    input_tensor, mask, weights, config = data

    B, N, _, C_in = input_tensor.shape
    C_out = int(config.get("dim", C_in))
    H = int(weights["left_proj.weight"].shape[0])

    device = input_tensor.device
    eps = 1e-5
    M = B * N * N

    try:
        torch.backends.cuda.matmul.allow_tf32 = True
    except Exception:
        pass
    try:
        torch.backends.cuda.matmul.allow_bf16_reduced_precision_reduction = True
    except Exception:
        pass

    with torch.no_grad():
        x = input_tensor.contiguous()

        # 1) LayerNorm(input_dim) -> bf16
        x_norm = torch.empty((M, C_in), dtype=torch.bfloat16, device=device)

        BLOCK_D = max(16, _next_pow2(C_in))
        BLOCK_R_LN = 32 if BLOCK_D <= 128 else (16 if BLOCK_D <= 256 else 8)
        _layer_norm_kernel[(triton.cdiv(M, BLOCK_R_LN),)](
            x.view(M, C_in),
            x_norm,
            weights["norm.weight"],
            weights["norm.bias"],
            M,
            C_in,
            eps,
            BLOCK_D=BLOCK_D,
            BLOCK_R=BLOCK_R_LN,
        )

        # 2) One GEMM for all five input projections/gates.
        w5 = torch.cat(
            [
                weights["left_proj.weight"].to(torch.bfloat16),
                weights["right_proj.weight"].to(torch.bfloat16),
                weights["left_gate.weight"].to(torch.bfloat16),
                weights["right_gate.weight"].to(torch.bfloat16),
                weights["out_gate.weight"].to(torch.bfloat16),
            ],
            dim=0,
        ).contiguous()

        proj = torch.empty((M, 5 * H), dtype=torch.bfloat16, device=device)
        torch.mm(x_norm, w5.t(), out=proj)
        del x_norm, w5

        # 3) Elementwise sigmoid gates and mask, producing natural layout tensors.
        left = torch.empty((M, H), dtype=torch.bfloat16, device=device)
        right = torch.empty((M, H), dtype=torch.bfloat16, device=device)
        gate = torch.empty((M, H), dtype=torch.bfloat16, device=device)

        has_mask = mask is not None and not config.get("nomask", False)
        if has_mask:
            mask_flat = mask.contiguous().to(torch.float32).reshape(-1)
        else:
            mask_flat = left  # dummy pointer, not dereferenced

        BLOCK_H_E = max(16, _next_pow2(H))
        BLOCK_R_E = 32 if BLOCK_H_E <= 128 else (16 if BLOCK_H_E <= 256 else 8)
        _elementwise_kernel[(triton.cdiv(M, BLOCK_R_E),)](
            proj,
            mask_flat,
            left,
            right,
            gate,
            M,
            H,
            has_mask,
            BLOCK_H_E,
            BLOCK_R_E,
        )
        del proj

        # 4) Transpose to [B, H, N, N] for batched matrix multiply.
        left_t = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
        del left
        right_t = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
        del right

        # 5) Core TriMul contraction:
        #    out[b, h, i, j] = sum_k left[b, h, i, k] * right[b, h, j, k]
        BH = B * H
        L = left_t.view(BH, N, N)
        R = right_t.view(BH, N, N)

        out_t = torch.empty((BH, N, N), dtype=torch.bfloat16, device=device)
        torch.bmm(L, R.transpose(1, 2), out=out_t)
        del left_t, right_t, L, R

        # 6) Return to natural [B, N, N, H] layout for fused final projection.
        out = out_t.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
        del out_t

        to_out_w = weights["to_out.weight"].to(torch.bfloat16).contiguous()
        out_final = torch.empty((M, C_out), dtype=torch.float32, device=device)

        BLOCK_H_F = max(16, _next_pow2(H))

        # Keep final dot blocks reasonably sized. For larger hidden dims,
        # use smaller output-channel tiles.
        BLOCK_DO = 128
        if BLOCK_H_F > 512 or C_out < 128:
            BLOCK_DO = 64
        if BLOCK_DO > C_out:
            BLOCK_DO = max(16, _next_pow2(C_out))
        BLOCK_DO = max(16, BLOCK_DO)

        BLOCK_R_F = 32 if BLOCK_H_F <= 256 else 16
        _final_kernel[(triton.cdiv(M, BLOCK_R_F),)](
            out.view(M, H),
            gate.view(M, H),
            weights["to_out_norm.weight"],
            weights["to_out_norm.bias"],
            to_out_w,
            out_final,
            M,
            H,
            C_out,
            eps,
            BLOCK_H_F,
            BLOCK_R_F,
            BLOCK_DO,
        )

        return out_final.view(B, N, N, C_out)
