def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    N, num_batches,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,

    num_pid_m = tl.cdiv(N, BLOCK_M)
    tiles_per_batch = num_pid_m * num_pid_n

    pid_in_batch = pid % tiles_per_batch
    pid_m = pid_in_batch // num_pid_n
    pid_n = pid_in_batch % num_pid_n
    
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # We want: c[batch, m, n] = sum_k a[batch, m, k] * b[batch, n, k]

    b_ptrs = b_ptr + batch_id * N * N + offs_n[:, None] * N + offs_k[None, :]

        b = tl.load(b_ptrs)
        acc = tl.dot(a, tl.trans(b), acc)
        a_ptrs += BLOCK_K
        b_ptrs += BLOCK_K
    
    c_ptrs = c_ptr + batch_id * N * N + offs_m[:, None] * N + offs_n[None, :]

    right_proj_weight = weights["right_proj.weight"]

    left = F.linear(x_norm, left_proj_weight)  # [B, N, N, H]
    right = F.linear(x_norm, right_proj_weight)  # [B, N, N, H]

    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))
    
    left = left * left_gate
    right = right * right_gate

    left_t = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N).to(torch.bfloat16)

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    N, num_batches,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    
    num_pid_n = tl.cdiv(N, BLOCK_N)
    num_pid_m = tl.cdiv(N, BLOCK_M)
    tiles_per_batch = num_pid_m * num_pid_n
    
    batch_id = pid // tiles_per_batch
    pid_in_batch = pid % tiles_per_batch
    pid_m = pid_in_batch // num_pid_n
    pid_n = pid_in_batch % num_pid_n
    
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    
    a_ptrs = a_ptr + batch_id * N * N + offs_m[:, None] * N + offs_k[None, :]
    b_ptrs = b_ptr + batch_id * N * N + offs_n[:, None] * N + offs_k[None, :]
    
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        a = tl.load(a_ptrs)
        b = tl.load(b_ptrs)
        acc = tl.dot(a, tl.trans(b), acc)
        a_ptrs += BLOCK_K
        b_ptrs += BLOCK_K
    
    c_ptrs = c_ptr + batch_id * N * N + offs_m[:, None] * N + offs_n[None, :]
    tl.store(c_ptrs, acc)
