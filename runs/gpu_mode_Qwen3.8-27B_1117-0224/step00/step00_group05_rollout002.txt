```
    left_ptr, right_ptr, gate_ptr,
    B, N,
    C: tl.constexpr,
    H: tl.constexpr,
    BM: tl.constexpr,
    HB: tl.constexpr,
    CP: tl.constexpr,
    HAS_MASK: tl.constexpr,
):

    pid_h = tl.program_id(1)
    offs_m = pid_m * BM + tl.arange(0, BM)
    mmask = offs_m < B * N * N
    NN = N * N
    b_m = offs_m // NN
    ml = offs_m - b_m * NN
    i_m = ml // N
    k_m = ml - i_m * N

    offs_c = tl.arange(0, CP)
    cmask = offs_c < C
    x_ptrs = x_ptr + offs_m[:, None] * C + offs_c[None, :]
    x = tl.load(x_ptrs, mask=mmask[:, None] & cmask[None, :], other=0.0)

    s = tl.sum(x, axis=1)
    ss = tl.sum(x * x, axis=1)
    mean = s / C
    var = ss / C - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = tl.rsqrt(var + 1e-5)
    nw = tl.load(nw_ptr + offs_c, mask=cmask, other=0.0)
    nb = tl.load(nb_ptr + offs_c, mask=cmask, other=0.0)

    else:
```

```python
import torch
import triton
import triton.language as tl


def _next_pow2(n: int) -> int:
    n = int(n)
    if n <= 0:
        return 1
    return 1 << (n - 1).bit_length()


@triton.jit
def _projection_kernel(
    x_ptr,
    mask_ptr,
    norm_w_ptr,
    norm_b_ptr,
    left_w_ptr,
    right_w_ptr,
    left_gate_w_ptr,
    right_gate_w_ptr,
    out_gate_w_ptr,
    left_ptr,
    right_ptr,
    out_gate_ptr,
    M,
    N,
    C: tl.constexpr,
    H: tl.constexpr,
    BM: tl.constexpr,
    HB: tl.constexpr,
    CP: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_m = tl.program_id(axis=0)
    pid_h = tl.program_id(axis=1)

    offs_m = pid_m * BM + tl.arange(0, BM)
    mmask = offs_m < M
    safe_m = tl.where(mmask, offs_m, 0)

    NN = N * N
    b_m = safe_m // NN
    m_local = safe_m - b_m * NN
    i_m = m_local // N
    k_m = m_local - i_m * N

    offs_c = tl.arange(0, CP)
    cmask = offs_c < C
    x_mask = mmask[:, None] & cmask[None, :]

    x_ptrs = x_ptr + safe_m[:, None] * C + offs_c[None, :]
    x = tl.load(x_ptrs, mask=x_mask, other=0.0)

    # LayerNorm over the channel dimension.
    s = tl.sum(x, axis=1)
    ss = tl.sum(x * x, axis=1)
    mean = s / C
    var = ss / C - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = tl.rsqrt(var + 1e-5)

    nw = tl.load(norm_w_ptr + offs_c, mask=cmask, other=0.0)
    nb = tl.load(norm_b_ptr + offs_c, mask=cmask, other=0.0)
    xn = ((x - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :])
    xn = tl.where(x_mask, xn, 0.0).to(tl.bfloat16)

    if HAS_MASK:
        mval = tl.load(mask_ptr + safe_m, mask=mmask, other=0.0).to(tl.float32)
    else:
        mval = tl.full((BM,), 1.0, dtype=tl.float32)

    h0 = pid_h * HB
    h = h0 + tl.arange(0, HB)
    hmask = h < H
    sh = tl.where(hmask, h, 0)

    wmask = cmask[:, None] & hmask[None, :]

    # left = (x @ W_left^T) * mask * sigmoid(x @ W_left_gate^T)
    wl = tl.load(left_w_ptr + sh[None, :] * C + offs_c[:, None], mask=wmask, other=0.0).to(tl.bfloat16)
    lv = tl.dot(xn, wl)

    wgl = tl.load(left_gate_w_ptr + sh[None, :] * C + offs_c[:, None], mask=wmask, other=0.0).to(tl.bfloat16)
    lg = tl.dot(xn, wgl)

    left_tile = (lv * mval[:, None] * tl.sigmoid(lg)).to(tl.bfloat16)
    left_tile = tl.trans(left_tile)

    row_off = b_m * (H * NN) + i_m * N + k_m
    left_ptrs = left_ptr + row_off[None, :] + sh[:, None] * NN
    tl.store(left_ptrs, left_tile, mask=hmask[None, :] & mmask[:, None])

    # right = (x @ W_right^T) * mask * sigmoid(x @ W_right_gate^T)
    wr = tl.load(right_w_ptr + sh[None, :] * C + offs_c[:, None], mask=wmask, other=0.0).to(tl.bfloat16)
    rv = tl.dot(xn, wr)

    wgr = tl.load(right_gate_w_ptr + sh[None, :] * C + offs_c[:, None], mask=wmask, other=0.0).to(tl.bfloat16)
    rg = tl.dot(xn, wgr)

    right_tile = (rv * mval[:, None] * tl.sigmoid(rg)).to(tl.bfloat16)
    right_tile = tl.trans(right_tile)

    right_ptrs = right_ptr + row_off[None, :] + sh[:, None] * NN
    tl.store(right_ptrs, right_tile, mask=hmask[None, :] & mmask[:, None])

    # out_gate = sigmoid(x @ W_out_gate^T), stored in [B, N, N, H]
    wgo = tl.load(out_gate_w_ptr + sh[None, :] * C + offs_c[:, None], mask=wmask, other=0.0).to(tl.bfloat16)
    og = tl.dot(xn, wgo)
    gate_tile = tl.sigmoid(og).to(tl.bfloat16)

    gate_ptrs = out_gate_ptr + safe_m[:, None] * H + sh[None, :]
    tl.store(gate_ptrs, gate_tile, mask=mmask[:, None] & hmask[None, :])


@triton.jit
def _final_kernel(
    c_ptr,
    gate_ptr,
    out_ptr,
    out_norm_w_ptr,
    out_norm_b_ptr,
    out_w_ptr,
    N,
    H: tl.constexpr,
    C: tl.constexpr,
    BI: tl.constexpr,
    BJ: tl.constexpr,
    HP: tl.constexpr,
    CC: tl.constexpr,
):
    pid_i = tl.program_id(axis=0)
    pid_j = tl.program_id(axis=1)
    b = tl.program_id(axis=2)

    offs_i = pid_i * BI + tl.arange(0, BI)
    offs_j = pid_j * BJ + tl.arange(0, BJ)
    valid_i = offs_i < N
    valid_j = offs_j < N
    safe_i = tl.where(valid_i, offs_i, 0)
    safe_j = tl.where(valid_j, offs_j, 0)

    NN = N * N

    h = tl.arange(0, HP)
    hmask = h < H
    sh = tl.where(hmask, h, 0)

    # c is stored as [B, H, N, N]
    c_base = c_ptr + b * (H * NN)
    c_ptrs = c_base + sh[:, None, None] * NN + safe_i[None, :, None] * N + safe_j[None, None, :]
    c_mask = hmask[:, None, None] & valid_i[None, :, None] & valid_j[None, None, :]
    c = tl.load(c_ptrs, mask=c_mask, other=0.0).to(tl.float32)

    # out_gate is stored as [B, N, N, H]
    g_base = gate_ptr + b * (N * N * H)
    g_ptrs = g_base + (safe_i[:, None] * N + safe_j[None, :]) * H + sh[None, None, :]
    g_mask = valid_i[:, None] & valid_j[None, :] & hmask[None, None, :]
    g = tl.load(g_ptrs, mask=g_mask, other=0.0).to(tl.float32)

    # LayerNorm statistics over H for each (i, j).
    sum_c = tl.sum(c, axis=0)
    sumsq_c = tl.sum(c * c, axis=0)
    mean = sum_c / H
    var = sumsq_c / H - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = tl.rsqrt(var + 1e-5)

    E: tl.constexpr = BI * BJ

    # c: [H, BI, BJ] -> [BI * BJ, H]
    c_e = tl.reshape(c, (HP, E))
    cT = tl.trans(c_e)

    # g: [BI, BJ, H] -> [BI * BJ, H]
    gE = tl.reshape(g, (E, HP))

    meanE = tl.reshape(mean, (E,))
    rstdE = tl.reshape(rstd, (E,))

    lnw = tl.load(out_norm_w_ptr + sh, mask=hmask, other=0.0).to(tl.float32)
    lnb = tl.load(out_norm_b_ptr + sh, mask=hmask, other=0.0).to(tl.float32)

    normed = (cT - meanE[:, None]) * rstdE[:, None] * lnw[None, :] + lnb[None, :]
    normed = tl.where(hmask[None, :], normed, 0.0)
    y = (normed * gE).to(tl.bfloat16)

    # y [E, H] @ W^T [H, C] -> output [E, C]
    for c0 in tl.static_range(0, C, CC):
        offs_c = c0 + tl.arange(0, CC)
        cmask_c = offs_c < C

        w = tl.load(
            out_w_ptr + sh[:, None] * C + offs_c[None, :],
            mask=hmask[:, None] & cmask_c[None, :],
            other=0.0,
        ).to(tl.bfloat16)

        acc = tl.dot(y, w)
        acc3 = tl.reshape(acc, (BI, BJ, CC))

        out_ptrs = (
            out_ptr
            + b * (N * N * C)
            + (safe_i[:, None] * N + safe_j[None, :]) * C
            + offs_c[None, None, :]
        )
        out_mask = valid_i[:, None] & valid_j[None, :] & cmask_c[None, None, :]
        tl.store(out_ptrs, acc3.to(tl.float32), mask=out_mask)


def custom_kernel(data):
    """
    Fused TriMul forward:
      1) Triton kernel does LayerNorm + 5 linear projections + gates/masks,
         writing left/right in [B, H, N, N] and out_gate in [B, N, N, H].
      2) cuBLAS batched GEMM computes out[b,i,j,h] = sum_k left[b,h,i,k] * right[b,h,j,k].
      3) Triton kernel does final LayerNorm(out), applies out_gate, and projects back to C.
    """
    input_tensor, mask, weights, config = data

    x = input_tensor
    if x.dtype != torch.float32:
        x = x.float()
    x = x.contiguous()

    B, N, _, C = x.shape
    H = int(weights["left_proj.weight"].shape[0])
    device = x.device
    out_dtype = torch.bfloat16

    has_mask = mask is not None
    if has_mask:
        mask = mask.float().contiguous()

    norm_w = weights["norm.weight"].float().contiguous()
    norm_b = weights["norm.bias"].float().contiguous()

    left_w = weights["left_proj.weight"].to(out_dtype).contiguous()
    right_w = weights["right_proj.weight"].to(out_dtype).contiguous()
    left_gate_w = weights["left_gate.weight"].to(out_dtype).contiguous()
    right_gate_w = weights["right_gate.weight"].to(out_dtype).contiguous()
    out_gate_w = weights["out_gate.weight"].to(out_dtype).contiguous()

    to_out_norm_w = weights["to_out_norm.weight"].float().contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].float().contiguous()
    to_out_w = weights["to_out.weight"].to(out_dtype).contiguous()

    NN = N * N
    M = B * NN

    # Block sizes for projection kernel.
    CP = max(16, _next_pow2(C))
    if CP <= 128:
        BM = 64
    else:
        BM = 32
    BM = max(16, BM)

    if H <= 128:
        HB = max(16, _next_pow2(H))
    elif CP <= 512:
        HB = 128
    else:
        HB = 64
    HB = max(16, min(HB, _next_pow2(H)))

    left = torch.empty((B, H, N, N), dtype=out_dtype, device=device)
    right = torch.empty((B, H, N, N), dtype=out_dtype, device=device)
    out_gate = torch.empty((B, N, N, H), dtype=out_dtype, device=device)

    grid_proj = ((M + BM - 1) // BM, (H + HB - 1) // HB)
    _projection_kernel[grid_proj](
        x,
        mask if has_mask else x,
        norm_w,
        norm_b,
        left_w,
        right_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
        left,
        right,
        out_gate,
        M,
        N,
        C=C,
        H=H,
        BM=BM,
        HB=HB,
        CP=CP,
        HAS_MASK=has_mask,
    )

    # Batched GEMM: for each (b, h): out[b, h, i, j] = sum_k left[b,h,i,k] * right[b,h,j,k]
    BH = B * H
    left_3d = left.reshape(BH, N, N)
    right_3d = right.reshape(BH, N, N)

    c = torch.empty((B, H, N, N), dtype=out_dtype, device=device)
    c_3d = c.reshape(BH, N, N)

    torch.bmm(left_3d, right_3d.transpose(-1, -2), out=c_3d)

    # Free large intermediate references before allocating final output.
    del left, right, left_3d, right_3d

    out = torch.empty((B, N, N, C), dtype=torch.float32, device=device)

    # Final fused LayerNorm + gate + linear.
    HP = max(16, _next_pow2(H))
    if HP <= 256:
        BI, BJ = 8, 8
    else:
        BI, BJ = 4, 4
    CC = max(16, min(128, _next_pow2(C)))

    grid_final = ((N + BI - 1) // BI, (N + BJ - 1) // BJ, B)
```
