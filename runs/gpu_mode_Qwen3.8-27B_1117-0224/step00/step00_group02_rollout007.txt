def prologue_kernel(x_ptr, mask_ptr, left_ptr, right_ptr, og_ptr, norm_w, norm_b, w_l, w_lg, w_r, w_rg, w_og,
                    M, C, H, NN, HAS_MASK: tl.constexpr, BR: tl.constexpr, CB: tl.constexpr, HC: tl.constexpr, NCHUNKS: tl.constexpr, EPS: tl.constexpr):
    pid = tl.program_id(0)
    r = pid*BR + tl.arange(0, BR)
    r_mask = r < M
    c = tl.arange(0, CB)
    c_mask = (c < C)
    x_ptrs = x_ptr + r[:,None]*C + c[None,:]
    x = tl.load(x_ptrs, mask=r_mask[:,None] & c_mask[None,:], other=0.0).to(tl.float32)
    mean = tl.sum(x, axis=1) / C
    centered = tl.where(c_mask[None,:], x - mean[:,None], 0.0)
    var = tl.sum(centered*centered, axis=1) / C
    inv = tl.rsqrt(var + EPS)
    w = tl.load(norm_w + c, mask=c_mask, other=0.0)[None,:]
    b = tl.load(norm_b + c, mask=c_mask, other=0.0)[None,:]
    xnorm = (x - mean[:,None]) * inv[:,None] * w + b
    xnorm = tl.where(c_mask[None,:], xnorm, 0.0)
    xb = xnorm.to(tl.bfloat16)
    if HAS_MASK:
       m = tl.load(mask_ptr + r, mask=r_mask, other=0.0).to(tl.float32)[:,None]
    else:
       m = 1.0
    batch = r // NN
    local = r - batch * NN
    row_base = (batch * (NN * H)) + local
    hm = c_mask[:,None]
    for ch in tl.static_range(NCHUNKS):
        h = ch*HC + tl.arange(0, HC)
        h_mask = h < H
        wmask = hm & h_mask[None,:]
        # W arrays [C,H] contiguous

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


def _cdiv(a: int, b: int) -> int:
    return (int(a) + int(b) - 1) // int(b)


def _pow2_min(n: int, min_val: int = 16) -> int:
    n = int(n)
    p = int(min_val)
    while p < n:
        p <<= 1
    return p


def _bf_t(w: torch.Tensor) -> torch.Tensor:
    return w.t().contiguous().to(torch.bfloat16)


@triton.jit
def _prologue_kernel(
    x_ptr,
    mask_ptr,
    left_ptr,
    right_ptr,
    ogate_ptr,
    norm_w_ptr,
    norm_b_ptr,
    wl_ptr,
    wlg_ptr,
    wr_ptr,
    wrg_ptr,
    wog_ptr,
    M,
    C,
    H,
    NN,
    HAS_MASK: tl.constexpr,
    BR: tl.constexpr,
    CB: tl.constexpr,
    HC: tl.constexpr,
    NCHUNKS: tl.constexpr,
    EPS: tl.constexpr,
):
    pid = tl.program_id(0)
    r = pid * BR + tl.arange(0, BR)
    r_mask = r < M

    c = tl.arange(0, CB)
    c_mask = c < C

    x_ptrs = x_ptr + r[:, None] * C + c[None, :]
    x = tl.load(x_ptrs, mask=r_mask[:, None] & c_mask[None, :], other=0.0).to(tl.float32)

    mean = tl.sum(x, axis=1) / C
    centered = tl.where(c_mask[None, :], x - mean[:, None], 0.0)
    var = tl.sum(centered * centered, axis=1) / C
    inv = 1.0 / tl.sqrt(var + EPS)

    wn = tl.load(norm_w_ptr + c, mask=c_mask, other=0.0)[None, :]
    bn = tl.load(norm_b_ptr + c, mask=c_mask, other=0.0)[None, :]
    xn = (x - mean[:, None]) * inv[:, None] * wn + bn
    xn = tl.where(c_mask[None, :], xn, 0.0)
    xb = xn.to(tl.bfloat16)

    if HAS_MASK:
        m = tl.load(mask_ptr + r, mask=r_mask, other=0).to(tl.float32)[:, None]
    else:
        m = tl.full((BR, 1), 1.0, tl.float32)

    batch = r // NN
    local = r - batch * NN
    row_base = batch * (NN * H) + local

    for ch in tl.static_range(NCHUNKS):
        h = ch * HC + tl.arange(0, HC)
        h_mask = h < H
        wmask = c_mask[:, None] & h_mask[None, :]

        wl = tl.load(wl_ptr + c[:, None] * H + h[None, :], mask=wmask, other=0.0)
        wlg = tl.load(wlg_ptr + c[:, None] * H + h[None, :], mask=wmask, other=0.0)
        yl = tl.dot(xb, wl)
        ylg = tl.dot(xb, wlg)
        left = (yl * tl.sigmoid(ylg)) * m
        tl.store(
            left_ptr + row_base[:, None] + h[None, :] * NN,
            left.to(tl.bfloat16),
            mask=r_mask[:, None] & h_mask[None, :],
        )

        wr = tl.load(wr_ptr + c[:, None] * H + h[None, :], mask=wmask, other=0.0)
        wr_g = tl.load(wrg_ptr + c[:, None] * H + h[None, :], mask=wmask, other=0.0)
        yr = tl.dot(xb, wr)
        yr_g = tl.dot(xb, wr_g)
        right = (yr * tl.sigmoid(yr_g)) * m
        tl.store(
            right_ptr + row_base[:, None] + h[None, :] * NN,
            right.to(tl.bfloat16),
            mask=r_mask[:, None] & h_mask[None, :],
        )

        wog = tl.load(wog_ptr + c[:, None] * H + h[None, :], mask=wmask, other=0.0)
        yog = tl.dot(xb, wog)
        og = tl.sigmoid(yog)
        tl.store(
            ogate_ptr + r[:, None] * H + h[None, :],
            og.to(tl.bfloat16),
            mask=r_mask[:, None] & h_mask[None, :],
        )


@triton.jit
def _core_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    N,
    H,
    NN,
    num_h,
    num_i,
    num_j,
    HB: tl.constexpr,
    BI: tl.constexpr,
    BJ: tl.constexpr,
    BK: tl.constexpr,
):
    pid = tl.program_id(0)

    tiles_per_b = num_h * num_i * num_j
    b = pid // tiles_per_b
    rem = pid - b * tiles_per_b

    ij_tiles = num_i * num_j
    hb = rem // ij_tiles
    rem2 = rem - hb * ij_tiles
    ib = rem2 // num_j
    jb = rem2 - ib * num_j

    h = hb * HB + tl.arange(0, HB)
    i = ib * BI + tl.arange(0, BI)
    j = jb * BJ + tl.arange(0, BJ)

    h_mask = h < H
    i_mask = i < N
    j_mask = j < N

    left_base = left_ptr + b * (NN * H)
    right_base = right_ptr + b * (NN * H)
    out_base = out_ptr + b * (NN * H)

    k0 = tl.arange(0, BK)

    # left: [b, h, i, k]
    l_ptrs = left_base + h[:, None, None] * NN + i[None, :, None] * N + k0[None, None, :]

    # right: [b, h, j, k], viewed as [h, k, j] for dot
    r_ptrs = right_base + h[:, None, None] * NN + k0[None, :, None] + j[None, None, :] * N

    acc = tl.zeros((HB, BI, BJ), dtype=tl.float32)

    num_k = (N + BK - 1) // BK
    for _ in range(num_k):
        k = tl.arange(0, BK)
        k_mask = k < N

        l = tl.load(
            l_ptrs,
            mask=h_mask[:, None, None] & i_mask[None, :, None] & k_mask[None, None, :],
            other=0.0,
        )
        r = tl.load(
            r_ptrs,
            mask=h_mask[:, None, None] & k_mask[None, :, None] & j_mask[None, None, :],
            other=0.0,
        )

        acc += tl.dot(l, r)

        l_ptrs += BK
        r_ptrs += BK

    # out: [b, i, j, h]
    o_ptrs = out_base + h[:, None, None] + i[None, :, None] * (N * H) + j[None, None, :] * H
    tl.store(
        o_ptrs,
        acc.to(tl.bfloat16),
        mask=h_mask[:, None, None] & i_mask[None, :, None] & j_mask[None, None, :],
    )


@triton.jit
def _final_kernel(
    out_ptr,
    ogate_ptr,
    final_ptr,
    gn_w_ptr,
    gn_b_ptr,
    w_to_ptr,
    M,
    H,
    C,
    BR: tl.constexpr,
    HB: tl.constexpr,
    CB: tl.constexpr,
    EPS: tl.constexpr,
):
    pid = tl.program_id(0)
    r = pid * BR + tl.arange(0, BR)
    r_mask = r < M

    h = tl.arange(0, HB)
    c = tl.arange(0, CB)

    h_mask = h < H
    c_mask = c < C

    x_ptrs = out_ptr + r[:, None] * H + h[None, :]
    x = tl.load(x_ptrs, mask=r_mask[:, None] & h_mask[None, :], other=0.0).to(tl.float32)
    g = tl.load(ogate_ptr + r[:, None] * H + h[None, :], mask=r_mask[:, None] & h_mask[None, :], other=0.0).to(tl.float32)

    mean = tl.sum(x, axis=1) / H
    centered = tl.where(h_mask[None, :], x - mean[:, None], 0.0)
    var = tl.sum(centered * centered, axis=1) / H
    inv = 1.0 / tl.sqrt(var + EPS)

    wn = tl.load(gn_w_ptr + h, mask=h_mask, other=0.0)[None, :]
    bn = tl.load(gn_b_ptr + h, mask=h_mask, other=0.0)[None, :]

    y = (x - mean[:, None]) * inv[:, None] * wn + bn
    y = y * g
    y = y.to(tl.bfloat16)

    # w_to is [H, C] contiguous
    w_ptrs = w_to_ptr + h[:, None] * C + c[None, :]
    w = tl.load(w_ptrs, mask=h_mask[:, None] & c_mask[None, :], other=0.0)

    acc = tl.dot(y, w)

    tl.store(
        final_ptr + r[:, None] * C + c[None, :],
        acc.to(tl.float32),
        mask=r_mask[:, None] & c_mask[None, :],
    )


def custom_kernel(data):
    """Fused TriMul forward:
    LayerNorm -> 5 projections/gates -> mask/nonlinearity -> Triton batched
    Hadamard-product contraction -> LayerNorm -> gate -> final linear.
    """
    input_tensor, mask, weights, config = data

    dim_cfg = int(config.get("dim", input_tensor.shape[-1]))
    hidden = int(config["hidden_dim"])
    eps = float(config.get("eps", 1e-5))

    B, N, _, C = input_tensor.shape
    C = int(C)
    if C != dim_cfg:
        C = int(dim_cfg)

    NN = N * N
    M = B * NN
    device = input_tensor.device

    x = input_tensor.reshape(M, C).contiguous()

    has_mask = mask is not None and getattr(mask, "numel", lambda: 0)() == M
    if has_mask:
        mask_flat = mask.reshape(M).contiguous()
    else:
        mask_flat = x

    norm_w = weights["norm.weight"].to(torch.float32).contiguous()
    norm_b = weights["norm.bias"].to(torch.float32).contiguous()

    wl = _bf_t(weights["left_proj.weight"])
    wlg = _bf_t(weights["left_gate.weight"])
    wr = _bf_t(weights["right_proj.weight"])
    wr_g = _bf_t(weights["right_gate.weight"])
    wog = _bf_t(weights["out_gate.weight"])

    left = torch.empty(M * hidden, dtype=torch.bfloat16, device=device)
    right = torch.empty(M * hidden, dtype=torch.bfloat16, device=device)
    ogate = torch.empty(M * hidden, dtype=torch.bfloat16, device=device)

    CB = _pow2_min(C)
    HC = 128 if hidden > 128 else _pow2_min(hidden)
    NCHUNKS = _cdiv(hidden, HC)

    BR = 64
    while BR > 16 and BR * max(CB, 2 * HC) > 16384:
        BR //= 2
    BR = max(16, BR)

    prologue_stage = 2 if CB <= 128 else 1
    _prologue_kernel[(_cdiv(M, BR),)](
        x,
        mask_flat,
        left,
        right,
        ogate,
        norm_w,
        norm_b,
        wl,
        wlg,
        wr,
        wr_g,
        wog,
        M,
        C,
        hidden,
        NN,
        HAS_MASK=has_mask,
        BR=BR,
        CB=CB,
        HC=HC,
        NCHUNKS=NCHUNKS,
        EPS=eps,
        num_warps=4,
        num_stages=prologue_stage,
    )

    out_core = torch.empty(M * hidden, dtype=torch.bfloat16, device=device)

    CORE_H = 8
    if N <= 128:
        BI = 32
        BJ = 32
    else:
        BI = 64
        BJ = 64
    BK = 32

    num_h = _cdiv(hidden, CORE_H)
    num_i = _cdiv(N, BI)
    num_j = _cdiv(N, BJ)

    _core_kernel[(B * num_h * num_i * num_j,)](
        left,
        right,
        out_core,
        N,
        hidden,
        NN,
        num_h,
        num_i,
        num_j,
        HB=CORE_H,
        BI=BI,
        BJ=BJ,
        BK=BK,
        num_warps=8,
        num_stages=2,
    )

    del left
    del right

    if hidden <= 256:
        out = torch.empty(M * C, dtype=torch.float32, device=device)

        to_w = _bf_t(weights["to_out.weight"])  # [C, H] -> [H, C]
        gn_w = weights["to_out_norm.weight"].to(torch.float32).contiguous()
        gn_b = weights["to_out_norm.bias"].to(torch.float32).contiguous()

        HB = _pow2_min(hidden)
        CBf = _pow2_min(C)

        BRf = 64
        while BRf > 16 and BRf * max(CBf, HB) > 16384:
            BRf //= 2
        BRf = max(16, BRf)

        final_stage = 2 if CBf <= 128 else 1
        _final_kernel[(_cdiv(M, BRf),)](
            out_core,
            ogate,
            out,
            gn_w,
            gn_b,
            to_w,
            M,
            hidden,
            C,
            BR=BRf,
            HB=HB,
            CB=CBf,
            EPS=eps,
            num_warps=4,
            num_stages=final_stage,
        )

        return out.view(B, N, N, C)

    # Robust PyTorch epilogue for uncommonly large hidden dims.
    xf = out_core.reshape(M, hidden)
    ogf = ogate.reshape(M, hidden)

    gn_w = weights["to_out_norm.weight"].to(xf.dtype)
    gn_b = weights["to_out_norm.bias"].to(xf.dtype)
    xf = F.layer_norm(xf, (hidden,), gn_w, gn_b, eps=eps)
    xf = xf * ogf

    to_w = weights["to_out.weight"].t().contiguous().to(xf.dtype)
    out = torch.mm(xf, to_w).to(torch.float32).view(B, N, N, C)
    return out
