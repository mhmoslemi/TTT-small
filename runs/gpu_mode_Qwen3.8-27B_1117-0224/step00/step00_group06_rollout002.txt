```python
import torch
import triton
import triton.language as tl


def _np2(n: int) -> int:
    n = int(n)
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _cdiv(a: int, b: int) -> int:
    return (int(a) + int(b) - 1) // int(b)


@triton.jit
def _prepare_dot(
    xd,
    w_ptr,
    offs_c,
    offs_h,
    C,
    c_mask,
    h_mask,
    BLOCK_C: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    w_ptrs = w_ptr + offs_c[:, None] + offs_h[None, :] * C
    w = tl.load(w_ptrs, mask=c_mask[:, None] & h_mask[None, :], other=0.0).to(tl.bfloat16)
    return tl.dot(xd, w)


@triton.jit
def trimul_prepare_kernel(
    x_ptr,
    mask_ptr,
    lw_ptr,
    rw_ptr,
    llw_ptr,
    rlw_ptr,
    olw_ptr,
    nw_ptr,
    nb_ptr,
    l_ptr,
    r_ptr,
    og_ptr,
    N,
    C,
    H,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
    BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_h = tl.program_id(2)

    N2 = N * N

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_c = tl.arange(0, BLOCK_C)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    m_mask = offs_m < N2
    c_mask = offs_c < C
    h_mask = offs_h < H

    x_ptrs = x_ptr + b * (N2 * C) + offs_m[:, None] * C + offs_c[None, :]
    x = tl.load(x_ptrs, mask=m_mask[:, None] & c_mask[None, :], other=0.0)

    mean = tl.sum(x, axis=1) / C
    diff = x - mean[:, None]
    diff = tl.where(c_mask[None, :], diff, 0.0)
    var = tl.sum(diff * diff, axis=1) / C
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    nw = tl.load(nw_ptr + offs_c, mask=c_mask, other=0.0)
    nb = tl.load(nb_ptr + offs_c, mask=c_mask, other=0.0)

    x_norm = (x - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]
    x_norm = tl.where(c_mask[None, :], x_norm, 0.0)
    xd = x_norm.to(tl.bfloat16)

    if HAS_MASK:
        mask_vals = tl.load(mask_ptr + b * N2 + offs_m, mask=m_mask, other=0.0).to(tl.float32)
    else:
        mask_vals = tl.full((BLOCK_M,), 1.0, tl.float32)

    y_l = _prepare_dot(xd, lw_ptr, offs_c, offs_h, C, c_mask, h_mask, BLOCK_C, BLOCK_H)
    y_lg = _prepare_dot(xd, llw_ptr, offs_c, offs_h, C, c_mask, h_mask, BLOCK_C, BLOCK_H)
    sig_l = 1.0 / (1.0 + tl.exp(-y_lg))
    left = y_l * sig_l * mask_vals[:, None]
    l_ptrs = l_ptr + b * (N2 * H) + offs_m[:, None] * H + offs_h[None, :]
    tl.store(l_ptrs, left.to(tl.bfloat16), mask=m_mask[:, None] & h_mask[None, :])

    y_r = _prepare_dot(xd, rw_ptr, offs_c, offs_h, C, c_mask, h_mask, BLOCK_C, BLOCK_H)
    y_rg = _prepare_dot(xd, rlw_ptr, offs_c, offs_h, C, c_mask, h_mask, BLOCK_C, BLOCK_H)
    sig_r = 1.0 / (1.0 + tl.exp(-y_rg))
    right = y_r * sig_r * mask_vals[:, None]
    r_ptrs = r_ptr + b * (N2 * H) + offs_m[:, None] * H + offs_h[None, :]
    tl.store(r_ptrs, right.to(tl.bfloat16), mask=m_mask[:, None] & h_mask[None, :])

    y_og = _prepare_dot(xd, olw_ptr, offs_c, offs_h, C, c_mask, h_mask, BLOCK_C, BLOCK_H)
    og = 1.0 / (1.0 + tl.exp(-y_og))
    og_ptrs = og_ptr + b * (N2 * H) + offs_m[:, None] * H + offs_h[None, :]
    tl.store(og_ptrs, og.to(tl.bfloat16), mask=m_mask[:, None] & h_mask[None, :])


@triton.jit
def trimul_core_kernel(
    l_ptr,
    r_ptr,
    o_ptr,
    N,
    H,
    N2,
    BM: tl.constexpr,
    BN: tl.constexpr,
    BK: tl.constexpr,
    GROUP_M: tl.constexpr,
    OUT_FP32: tl.constexpr,
):
    pid = tl.program_id(0)

    num_pid_m = tl.cdiv(N, BM)
    num_pid_n = tl.cdiv(N, BN)
    num_pid_mn = num_pid_m * num_pid_n

    hb = pid // num_pid_mn
    pid_mn = pid % num_pid_mn

    b = hb // H
    h = hb % H

    rem = pid_mn
    num_pid_in_group = GROUP_M * num_pid_n
    group_id = rem // num_pid_in_group
    first_pid_m = group_id * GROUP_M
    group_size_m = tl.minimum(num_pid_m - first_pid_m, GROUP_M)
    pid_m = first_pid_m + (rem % group_size_m)
    pid_n = (rem % num_pid_in_group) // group_size_m

    offs_i = pid_m * BM + tl.arange(0, BM)
    offs_j = pid_n * BN + tl.arange(0, BN)
    offs_k = tl.arange(0, BK)

    i_mask = offs_i < N
    j_mask = offs_j < N

    a_ptrs = l_ptr + (b * N2 + offs_i[:, None]) * (N * H) + offs_k[None, :] * H + h
    b_ptrs = r_ptr + (b * N2 + offs_j[:, None]) * (N * H) + offs_k[None, :] * H + h

    acc = tl.zeros((BM, BN), dtype=tl.float32)
    num_k = tl.cdiv(N, BK)

    for k_iter in range(0, num_k):
        k0 = k_iter * BK
        k_mask = (k0 + offs_k) < N

        a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        b_tile = tl.load(b_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0)
        b_mat = tl.trans(b_tile)

        acc = tl.dot(a, b_mat, acc)

        a_ptrs += BK * H
        b_ptrs += BK * H

    o_ptrs = o_ptr + b * (H * N2) + h * N2 + offs_i[:, None] * N + offs_j[None, :]
    o_mask = i_mask[:, None] & j_mask[None, :]

    if OUT_FP32:
        tl.store(o_ptrs, acc, mask=o_mask)
    else:
        tl.store(o_ptrs, acc.to(tl.bfloat16), mask=o_mask)


@triton.jit
def trimul_final_kernel(
    o_ptr,
    og_ptr,
    w_ptr,
    tow_ptr,
    tnb_ptr,
    out_ptr,
    H,
    C,
    N2,
    BM: tl.constexpr,
    BH: tl.constexpr,
    BC: tl.constexpr,
    SINGLE_H: tl.constexpr,
):
    b = tl.program_id(0)
    pid_m = tl.program_id(1)

    offs_m = pid_m * BM + tl.arange(0, BM)
    m_mask = offs_m < N2

    offs_h = tl.arange(0, BH)
    offs_c = tl.arange(0, BC)
    c_mask = offs_c < C

    o_base = o_ptr + b * (H * N2)
    og_base = og_ptr + b * (N2 * H)
    out_base = out_ptr + b * (N2 * C)

    if SINGLE_H:
        h_cur = offs_h
        h_mask = h_cur < H

        o_ptrs = o_base + h_cur[:, None] * N2 + offs_m[None, :]
        o = tl.load(o_ptrs, mask=h_mask[:, None] & m_mask[None, :], other=0.0).to(tl.float32)

        mean = tl.sum(o, axis=0) / H
        diff = o - mean[None, :]
        diff = tl.where(h_mask[:, None], diff, 0.0)
        var = tl.sum(diff * diff, axis=0) / H
        var = tl.maximum(var, 0.0)
        rstd = 1.0 / tl.sqrt(var + 1e-5)

        acc = tl.zeros((BM, BC), dtype=tl.float32)

        norm_t = tl.trans(diff * rstd[None, :])

        tnw = tl.load(tow_ptr + h_cur, mask=h_mask, other=0.0)
        tnb = tl.load(tnb_ptr + h_cur, mask=h_mask, other=0.0)
        affine = norm_t * tnw[None, :] + tnb[None, :]

        og_ptrs = og_base + offs_m[:, None] * H + h_cur[None, :]
        og = tl.load(og_ptrs, mask=m_mask[:, None] & h_mask[None, :], other=0.0).to(tl.float32)
        y = affine * og

        w_ptrs = w_ptr + h_cur[:, None] * C + offs_c[None, :]
        w = tl.load(w_ptrs, mask=h_mask[:, None] & c_mask[None, :], other=0.0)

        acc = tl.dot(y.to(tl.bfloat16), w, acc)

        out_ptrs = out_base + offs_m[:, None] * C + offs_c[None, :]
        tl.store(out_ptrs, acc, mask=m_mask[:, None] & c_mask[None, :])
        return

    sum1 = tl.zeros((BM,), dtype=tl.float32)
    sum2 = tl.zeros((BM,), dtype=tl.float32)
    num_h = tl.cdiv(H, BH)

    for h_iter in range(0, num_h):
        h_cur = h_iter * BH + offs_h
        h_mask = h_cur < H
        o_ptrs = o_base + h_cur[:, None] * N2 + offs_m[None, :]
        o = tl.load(o_ptrs, mask=h_mask[:, None] & m_mask[None, :], other=0.0).to(tl.float32)
        sum1 += tl.sum(o, axis=0)
        sum2 += tl.sum(o * o, axis=0)

    mean = sum1 / H
    var = sum2 / H - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    acc = tl.zeros((BM, BC), dtype=tl.float32)

    for h_iter in range(0, num_h):
        h_cur = h_iter * BH + offs_h
        h_mask = h_cur < H

        o_ptrs = o_base + h_cur[:, None] * N2 + offs_m[None, :]
        o = tl.load(o_ptrs, mask=h_mask[:, None] & m_mask[None, :], other=0.0).to(tl.float32)
        norm = (o - mean[None, :]) * rstd[None, :]
        norm_t = tl.trans(norm)

        tnw = tl.load(tow_ptr + h_cur, mask=h_mask, other=0.0)
        tnb = tl.load(tnb_ptr + h_cur, mask=h_mask, other=0.0)
        affine = norm_t * tnw[None, :] + tnb[None, :]

        og_ptrs = og_base + offs_m[:, None] * H + h_cur[None, :]
        og = tl.load(og_ptrs, mask=m_mask[:, None] & h_mask[None, :], other=0.0).to(tl.float32)
        y = affine * og

        w_ptrs = w_ptr + h_cur[:, None] * C + offs_c[None, :]
        w = tl.load(w_ptrs, mask=h_mask[:, None] & c_mask[None, :], other=0.0)

        acc = tl.dot(y.to(tl.bfloat16), w, acc)

    out_ptrs = out_base + offs_m[:, None] * C + offs_c[None, :]
    tl.store(out_ptrs, acc, mask=m_mask[:, None] & c_mask[None, :])


def custom_kernel(data):
    """Fused Triton TriMul forward: prepare projections/gates, batched contraction, final LayerNorm/projection."""
    if len(data) == 4:
        input_tensor, mask, weights, config = data
    elif len(data) == 3:
        input_tensor, weights, config = data
        mask = None
    else:
        raise ValueError("data must contain (input, mask, weights, config) or (input, weights, config)")

    if config is None:
        config = {}

    x = input_tensor.detach()
    device = x.device
    if x.dtype != torch.float32:
        x = x.to(device=device, dtype=torch.float32)
    if not x.is_contiguous():
        x = x.contiguous()

    B, N, _, C = x.shape
    N2 = N * N
    H = int(weights["to_out.weight"].shape[1])

    nomask = bool(config.get("nomask", False))
    has_mask = (mask is not None) and (not nomask)

    if has_mask:
        if mask.ndim == 4 and mask.shape[-1] == 1:
            mask = mask.squeeze(-1)
        if mask.ndim == 2:
            mask = mask.unsqueeze(0)
        if mask.shape[0] == 1 and B > 1:
            mask = mask.expand(B, N, N)
        mask_arg = mask.to(device=device, dtype=torch.float32).contiguous()
    else:
        mask_arg = x

    bf16 = torch.bfloat16
    fp32 = torch.float32

    lw = weights["left_proj.weight"].to(device=device, dtype=bf16).contiguous()
    rw = weights["right_proj.weight"].to(device=device, dtype=bf16).contiguous()
    llw = weights["left_gate.weight"].to(device=device, dtype=bf16).contiguous()
    rlw = weights["right_gate.weight"].to(device=device, dtype=bf16).contiguous()
    olw = weights["out_gate.weight"].to(device=device, dtype=bf16).contiguous()
    nw = weights["norm.weight"].to(device=device, dtype=fp32).contiguous()
    nb = weights["norm.bias"].to(device=device, dtype=fp32).contiguous()
    to_w = weights["to_out_norm.weight"].to(device=device, dtype=fp32).contiguous()
    to_b = weights["to_out_norm.bias"].to(device=device, dtype=fp32).contiguous()
    out_w = weights["to_out.weight"].to(device=device, dtype=bf16).contiguous()

    L = torch.empty((B, N, N, H), device=device, dtype=bf16)
    R = torch.empty((B, N, N, H), device=device, dtype=bf16)
    OG = torch.empty((B, N, N, H), device=device, dtype=bf16)

    PC = max(16, _np2(C))
    if PC <= 128:
        PM = 128
    elif PC <= 512:
        PM = 32
    else:
        PM = 16
    PH = 128 if H >= 128 else max(16, _np2(H))

    grid_prepare = (B, _cdiv(N2, PM), _cdiv(H, PH))
    trimul_prepare_kernel[grid_prepare](
        x,
        mask_arg,
        lw,
        rw,
        llw,
        rlw,
        olw,
        nw,
        nb,
        L,
        R,
        OG,
        N,
        C,
        H,
        BLOCK_M=PM,
        BLOCK_C=PC,
        BLOCK_H=PH,
        HAS_MASK=has_mask,
        num_warps=8,
        num_stages=2,
    )

    o_elems = B * H * N2
    O_dtype = torch.bfloat16 if o_elems > 50_000_000 else torch.float32
    O = torch.empty((B, H, N, N), device=device, dtype=O_dtype)

    if N <= 128:
        CM, CN, CK = 64, 64, 64
        core_warps = 4
        core_stages = 4
    else:
        CM, CN, CK = 128, 128, 64
        core_warps = 8
        core_stages = 3

    num_m = _cdiv(N, CM)
    num_n = _cdiv(N, CN)
    grid_core = (B * H * num_m * num_n,)

    trimul_core_kernel[grid_core](
        L,
        R,
        O,
        N,
        H,
        N2,
        BM=CM,
        BN=CN,
        BK=CK,
        GROUP_M=8,
        OUT_FP32=(O.dtype == torch.float32),
        num_warps=core_warps,
        num_stages=core_stages,
    )

    out = torch.empty((B, N, N, C), device=device, dtype=torch.float32)
    BCF = max(16, _np2(C))
    if BCF <= 128:
        BMF = 64
    elif BCF <= 512:
        BMF = 32
    else:
        BMF = 16
    BHF = 128 if H >= 128 else max(16, _np2(H))

    grid_final = (B, _cdiv(N2, BMF))
    trimul_final_kernel[grid_final](
        O,
        OG,
        out_w,
        to_w,
        to_b,
        out,
        H,
        C,
        N2,
        BLOCK_M=BMF,
        BLOCK_H=BHF,
        BLOCK_C=BCF,
        SINGLE_H=(H <= BHF),
        num_warps=8,
        num_stages=2,
    )

    return out
```
