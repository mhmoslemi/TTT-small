import torch
import triton
import triton.language as tl


"""
TriMul forward:
  x -> LayerNorm -> 5 linear projections -> mask/gates ->
  out[i,j,d] = sum_k left[i,k,d] * right[j,k,d] -> LayerNorm -> out_gate -> to_out.

Implementation:
  - One fused Triton kernel performs input LayerNorm + the five linear projections +
    mask/gate combinations and writes left/right directly in d-major layout for cuBLAS.
  - The O(N^3 * hidden) triangle product is a cuBLAS batched matrix multiply.
  - A second fused Triton kernel performs the output LayerNorm, multiplies by out_gate,
    and writes the final to_out GEMM directly in float32.
"""


@triton.jit
def _sigmoid(x):
    return 1.0 / (1.0 + tl.exp(-x))


@triton.jit
def fused_ln_proj_chunked_kernel(
    x_ptr,
    lnw_ptr,
    lnb_ptr,
    wl_ptr,
    wr_ptr,
    wgl_ptr,
    wgr_ptr,
    wgo_ptr,
    mask_ptr,
    left_d_ptr,
    right_d_ptr,
    outgate_ptr,
    M,
    N,
    H,
    N2,
    C,
    HAS_MASK: tl.constexpr,
    BM: tl.constexpr,
    BH: tl.constexpr,
    BK: tl.constexpr,
    CK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BM + tl.arange(0, BM)
    m_ok = offs_m < M

    b = offs_m // N2
    loc = offs_m - b * N2

    c0 = tl.arange(0, BK)

    x0 = tl.load(x_ptr + offs_m[:, None] * C + c0[None, :], mask=m_ok[:, None], other=0.0).to(tl.float32)

    x1 = x0
    if CK >= 2:
        x1 = tl.load(x_ptr + offs_m[:, None] * C + (c0 + BK)[None, :], mask=m_ok[:, None], other=0.0).to(tl.float32)
    x2 = x0
    if CK >= 3:
        x2 = tl.load(x_ptr + offs_m[:, None] * C + (c0 + 2 * BK)[None, :], mask=m_ok[:, None], other=0.0).to(tl.float32)
    x3 = x0
    if CK >= 4:
        x3 = tl.load(x_ptr + offs_m[:, None] * C + (c0 + 3 * BK)[None, :], mask=m_ok[:, None], other=0.0).to(tl.float32)
    x4 = x0
    if CK >= 5:
        x4 = tl.load(x_ptr + offs_m[:, None] * C + (c0 + 4 * BK)[None, :], mask=m_ok[:, None], other=0.0).to(tl.float32)
    x5 = x0
    if CK >= 6:
        x5 = tl.load(x_ptr + offs_m[:, None] * C + (c0 + 5 * BK)[None, :], mask=m_ok[:, None], other=0.0).to(tl.float32)
    x6 = x0
    if CK >= 7:
        x6 = tl.load(x_ptr + offs_m[:, None] * C + (c0 + 6 * BK)[None, :], mask=m_ok[:, None], other=0.0).to(tl.float32)
    x7 = x0
    if CK >= 8:
        x7 = tl.load(x_ptr + offs_m[:, None] * C + (c0 + 7 * BK)[None, :], mask=m_ok[:, None], other=0.0).to(tl.float32)

    sum_x = tl.sum(x0, axis=1)
    sum_x2 = tl.sum(x0 * x0, axis=1)
    if CK >= 2:
        sum_x += tl.sum(x1, axis=1)
        sum_x2 += tl.sum(x1 * x1, axis=1)
    if CK >= 3:
        sum_x += tl.sum(x2, axis=1)
        sum_x2 += tl.sum(x2 * x2, axis=1)
    if CK >= 4:
        sum_x += tl.sum(x3, axis=1)
        sum_x2 += tl.sum(x3 * x3, axis=1)
    if CK >= 5:
        sum_x += tl.sum(x4, axis=1)
        sum_x2 += tl.sum(x4 * x4, axis=1)
    if CK >= 6:
        sum_x += tl.sum(x5, axis=1)
        sum_x2 += tl.sum(x5 * x5, axis=1)
    if CK >= 7:
        sum_x += tl.sum(x6, axis=1)
        sum_x2 += tl.sum(x6 * x6, axis=1)
    if CK >= 8:
        sum_x += tl.sum(x7, axis=1)
        sum_x2 += tl.sum(x7 * x7, axis=1)

    mean = sum_x / C
    var = sum_x2 / C - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    lnw0 = tl.load(lnw_ptr + c0).to(tl.float32)
    lnb0 = tl.load(lnb_ptr + c0).to(tl.float32)
    xn0 = ((x0 - mean[:, None]) * rstd[:, None] * lnw0[None, :] + lnb0[None, :]).to(tl.bfloat16)

    xn1 = xn0
    if CK >= 2:
        lnw1 = tl.load(lnw_ptr + c0 + BK).to(tl.float32)
        lnb1 = tl.load(lnb_ptr + c0 + BK).to(tl.float32)
        xn1 = ((x1 - mean[:, None]) * rstd[:, None] * lnw1[None, :] + lnb1[None, :]).to(tl.bfloat16)
    xn2 = xn0
    if CK >= 3:
        lnw2 = tl.load(lnw_ptr + c0 + 2 * BK).to(tl.float32)
        lnb2 = tl.load(lnb_ptr + c0 + 2 * BK).to(tl.float32)
        xn2 = ((x2 - mean[:, None]) * rstd[:, None] * lnw2[None, :] + lnb2[None, :]).to(tl.bfloat16)
    xn3 = xn0
    if CK >= 4:
        lnw3 = tl.load(lnw_ptr + c0 + 3 * BK).to(tl.float32)
        lnb3 = tl.load(lnb_ptr + c0 + 3 * BK).to(tl.float32)
        xn3 = ((x3 - mean[:, None]) * rstd[:, None] * lnw3[None, :] + lnb3[None, :]).to(tl.bfloat16)
    xn4 = xn0
    if CK >= 5:
        lnw4 = tl.load(lnw_ptr + c0 + 4 * BK).to(tl.float32)
        lnb4 = tl.load(lnb_ptr + c0 + 4 * BK).to(tl.float32)
        xn4 = ((x4 - mean[:, None]) * rstd[:, None] * lnw4[None, :] + lnb4[None, :]).to(tl.bfloat16)
    xn5 = xn0
    if CK >= 6:
        lnw5 = tl.load(lnw_ptr + c0 + 5 * BK).to(tl.float32)
        lnb5 = tl.load(lnb_ptr + c0 + 5 * BK).to(tl.float32)
        xn5 = ((x5 - mean[:, None]) * rstd[:, None] * lnw5[None, :] + lnb5[None, :]).to(tl.bfloat16)
    xn6 = xn0
    if CK >= 7:
        lnw6 = tl.load(lnw_ptr + c0 + 6 * BK).to(tl.float32)
        lnb6 = tl.load(lnb_ptr + c0 + 6 * BK).to(tl.float32)
        xn6 = ((x6 - mean[:, None]) * rstd[:, None] * lnw6[None, :] + lnb6[None, :]).to(tl.bfloat16)
    xn7 = xn0
    if CK >= 8:
        lnw7 = tl.load(lnw_ptr + c0 + 7 * BK).to(tl.float32)
        lnb7 = tl.load(lnb_ptr + c0 + 7 * BK).to(tl.float32)
        xn7 = ((x7 - mean[:, None]) * rstd[:, None] * lnw7[None, :] + lnb7[None, :]).to(tl.bfloat16)

    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_m, mask=m_ok, other=0.0).to(tl.float32)
    else:
        mval = tl.full((BM,), 1.0, dtype=tl.float32)
    mval_b = mval[:, None]

    stride_batch = H * N2

    for hb in range(0, H, BH):
        offs_h = hb + tl.arange(0, BH)
        h_ok = offs_h < H
        h_mask_row = h_ok[None, :]
        store_mask_hm = h_ok[:, None] & m_ok[None, :]

        base_d = b * stride_batch + loc
        left_store_offs = offs_h[:, None] * N2 + base_d[None, :]
        og_offs = offs_m[:, None] * H + offs_h[None, :]

        # left projection and left gate
        acc_l = tl.zeros((BM, BH), dtype=tl.float32)
        acc_gl = tl.zeros((BM, BH), dtype=tl.float32)
        for kc in tl.static_range(CK):
            xk = xn0
            if kc == 1:
                xk = xn1
            elif kc == 2:
                xk = xn2
            elif kc == 3:
                xk = xn3
            elif kc == 4:
                xk = xn4
            elif kc == 5:
                xk = xn5
            elif kc == 6:
                xk = xn6
            elif kc == 7:
                xk = xn7

            coff = c0 + kc * BK
            w_off = coff[:, None] * H + offs_h[None, :]

            wl_t = tl.load(wl_ptr + w_off, mask=h_mask_row, other=0.0)
            wgl_t = tl.load(wgl_ptr + w_off, mask=h_mask_row, other=0.0)

            acc_l += tl.dot(xk, wl_t)
            acc_gl += tl.dot(xk, wgl_t)

        left = acc_l * mval_b * _sigmoid(acc_gl)
        tl.store(left_d_ptr + left_store_offs, tl.trans(left.to(tl.bfloat16)), mask=store_mask_hm)

        # right projection and right gate
        acc_r = tl.zeros((BM, BH), dtype=tl.float32)
        acc_gr = tl.zeros((BM, BH), dtype=tl.float32)
        for kc in tl.static_range(CK):
            xk = xn0
            if kc == 1:
                xk = xn1
            elif kc == 2:
                xk = xn2
            elif kc == 3:
                xk = xn3
            elif kc == 4:
                xk = xn4
            elif kc == 5:
                xk = xn5
            elif kc == 6:
                xk = xn6
            elif kc == 7:
                xk = xn7

            coff = c0 + kc * BK
            w_off = coff[:, None] * H + offs_h[None, :]

            wr_t = tl.load(wr_ptr + w_off, mask=h_mask_row, other=0.0)
            wgr_t = tl.load(wgr_ptr + w_off, mask=h_mask_row, other=0.0)

            acc_r += tl.dot(xk, wr_t)
            acc_gr += tl.dot(xk, wgr_t)

        right = acc_r * mval_b * _sigmoid(acc_gr)
        tl.store(right_d_ptr + left_store_offs, tl.trans(right.to(tl.bfloat16)), mask=store_mask_hm)

        # out gate
        acc_go = tl.zeros((BM, BH), dtype=tl.float32)
        for kc in tl.static_range(CK):
            xk = xn0
            if kc == 1:
                xk = xn1
            elif kc == 2:
                xk = xn2
            elif kc == 3:
                xk = xn3
            elif kc == 4:
                xk = xn4
            elif kc == 5:
                xk = xn5
            elif kc == 6:
                xk = xn6
            elif kc == 7:
                xk = xn7

            coff = c0 + kc * BK
            w_off = coff[:, None] * H + offs_h[None, :]

            wgo_t = tl.load(wgo_ptr + w_off, mask=h_mask_row, other=0.0)
            acc_go += tl.dot(xk, wgo_t)

        out_gate = _sigmoid(acc_go)
        tl.store(outgate_ptr + og_offs, out_gate.to(tl.bfloat16), mask=m_ok[:, None] & h_mask_row)


@triton.jit
def fused_ln_proj_generic_kernel(
    x_ptr,
    lnw_ptr,
    lnb_ptr,
    wl_ptr,
    wr_ptr,
    wgl_ptr,
    wgr_ptr,
    wgo_ptr,
    mask_ptr,
    left_d_ptr,
    right_d_ptr,
    outgate_ptr,
    M,
    N,
    H,
    N2,
    C,
    HAS_MASK: tl.constexpr,
    BM: tl.constexpr,
    BH: tl.constexpr,
    CU: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BM + tl.arange(0, BM)
    m_ok = offs_m < M

    b = offs_m // N2
    loc = offs_m - b * N2

    c = tl.arange(0, CU)
    c_ok = c < C
    c_mask_row = c_ok[None, :]

    x = tl.load(
        x_ptr + offs_m[:, None] * C + c[None, :],
        mask=m_ok[:, None] & c_mask_row,
        other=0.0,
    ).to(tl.float32)
    x = tl.where(c_mask_row, x, 0.0)

    sum_x = tl.sum(x, axis=1)
    sum_x2 = tl.sum(x * x, axis=1)

    mean = sum_x / C
    var = sum_x2 / C - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    lnw = tl.load(lnw_ptr + c, mask=c_ok, other=0.0).to(tl.float32)
    lnb = tl.load(lnb_ptr + c, mask=c_ok, other=0.0).to(tl.float32)

    xn = ((x - mean[:, None]) * rstd[:, None] * lnw[None, :] + lnb[None, :]).to(tl.bfloat16)
    xn = tl.where(c_mask_row, xn, 0.0)

    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_m, mask=m_ok, other=0.0).to(tl.float32)
    else:
        mval = tl.full((BM,), 1.0, dtype=tl.float32)
    mval_b = mval[:, None]

    stride_batch = H * N2

    for hb in range(0, H, BH):
        offs_h = hb + tl.arange(0, BH)
        h_ok = offs_h < H
        h_mask_row = h_ok[None, :]
        store_mask_hm = h_ok[:, None] & m_ok[None, :]

        base_d = b * stride_batch + loc
        left_store_offs = offs_h[:, None] * N2 + base_d[None, :]
        og_offs = offs_m[:, None] * H + offs_h[None, :]

        cmask_by_c = c_ok[:, None] & h_mask_row

        # left
        wl_t = tl.load(wl_ptr + c[:, None] * H + offs_h[None, :], mask=cmask_by_c, other=0.0)
        wgl_t = tl.load(wgl_ptr + c[:, None] * H + offs_h[None, :], mask=cmask_by_c, other=0.0)
        acc_l = tl.dot(xn, wl_t)
        acc_gl = tl.dot(xn, wgl_t)
        left = acc_l * mval_b * _sigmoid(acc_gl)
        tl.store(left_d_ptr + left_store_offs, tl.trans(left.to(tl.bfloat16)), mask=store_mask_hm)

        # right
        wr_t = tl.load(wr_ptr + c[:, None] * H + offs_h[None, :], mask=cmask_by_c, other=0.0)
        wgr_t = tl.load(wgr_ptr + c[:, None] * H + offs_h[None, :], mask=cmask_by_c, other=0.0)
        acc_r = tl.dot(xn, wr_t)
        acc_gr = tl.dot(xn, wgr_t)
        right = acc_r * mval_b * _sigmoid(acc_gr)
        tl.store(right_d_ptr + left_store_offs, tl.trans(right.to(tl.bfloat16)), mask=store_mask_hm)

        # out gate
        wgo_t = tl.load(wgo_ptr + c[:, None] * H + offs_h[None, :], mask=cmask_by_c, other=0.0)
        acc_go = tl.dot(xn, wgo_t)
        out_gate = _sigmoid(acc_go)
        tl.store(outgate_ptr + og_offs, out_gate.to(tl.bfloat16), mask=m_ok[:, None] & h_mask_row)


@triton.jit
def post_final_kernel(
    out_d_ptr,
    gate_ptr,
    nw_ptr,
    nb_ptr,
    wout_ptr,
    out_ptr,
    M,
    N,
    H,
    N2,
    C,
    eps,
    BM: tl.constexpr,
    BH: tl.constexpr,
    BC: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BM + tl.arange(0, BM)
    m_ok = offs_m < M

    b = offs_m // N2
    loc = offs_m - b * N2

    d = tl.arange(0, BH)
    d_ok = d < H
    d_mask_row = d_ok[:, None]
    m_mask_col = m_ok[None, :]

    row_base = b * (H * N2) + loc
    out_d_offs = d[:, None] * N2 + row_base[None, :]

    tile = tl.load(out_d_ptr + out_d_offs, mask=d_mask_row & m_mask_col, other=0.0).to(tl.float32)

    sum_x = tl.sum(tile, axis=0)
    sum_x2 = tl.sum(tile * tile, axis=0)

    mean = sum_x / H
    var = sum_x2 / H - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + eps)

    nw = tl.load(nw_ptr + d, mask=d_ok, other=0.0).to(tl.float32)
    nb = tl.load(nb_ptr + d, mask=d_ok, other=0.0).to(tl.float32)

    normed = (tile - mean[None, :]) * rstd[None, :] * nw[:, None] + nb[:, None]

    gate_offs = d[:, None] + offs_m[None, :] * H
    gate = tl.load(gate_ptr + gate_offs, mask=d_mask_row & m_mask_col, other=0.0).to(tl.float32)

    gated = normed * gate
    gated = tl.where(d_mask_row, gated, 0.0)

    a = tl.trans(gated.to(tl.bfloat16))  # [BM, BH]

    for cb in range(0, C, BC):
        c = cb + tl.arange(0, BC)
        c_ok = c < C
        c_mask_row = c_ok[None, :]

        w_offs = d[:, None] * C + c[None, :]
        w = tl.load(wout_ptr + w_offs, mask=d_mask_row & c_mask_row, other=0.0)

        acc = tl.dot(a, w)

        out_offs = offs_m[:, None] * C + c[None, :]
        tl.store(out_ptr + out_offs, acc.to(tl.float32), mask=m_ok[:, None] & c_mask_row)


def custom_kernel(data):
    if len(data) == 4:
        input_tensor, mask, weights, config = data
    else:
        input_tensor, weights, config = data
        mask = None

    B, N, _, C = input_tensor.shape
    H = int(config["hidden_dim"])
    N2 = N * N
    M = B * N2
    device = input_tensor.device

    x = input_tensor.contiguous().view(M, C)

    if mask is None:
        mask_flat = x
        has_mask = False
    else:
        mask_flat = mask.reshape(-1).contiguous().to(torch.float32)
        has_mask = True

    ln_w = weights["norm.weight"].to(torch.float32).contiguous()
    ln_b = weights["norm.bias"].to(torch.float32).contiguous()

    def prepare_weight(w):
        return w.to(torch.bfloat16).t().contiguous()

    w_left = prepare_weight(weights["left_proj.weight"])
    w_right = prepare_weight(weights["right_proj.weight"])
    w_left_gate = prepare_weight(weights["left_gate.weight"])
    w_right_gate = prepare_weight(weights["right_gate.weight"])
    w_out_gate = prepare_weight(weights["out_gate.weight"])
    w_to_out = prepare_weight(weights["to_out.weight"])

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float32).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float32).contiguous()

    left_d = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    right_d = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    out_gate = torch.empty((M, H), dtype=torch.bfloat16, device=device)

    # Fused LN + projections + mask/gates + d-major writes.
    if C % 128 == 0 and (C // 128) <= 8:
        CK = C // 128
        BK = 128
        BH = 128
        if CK <= 3:
            BM = 32
        else:
            BM = 16
        grid = (triton.cdiv(M, BM),)
        fused_ln_proj_chunked_kernel[grid](
            x, ln_w, ln_b,
            w_left, w_right, w_left_gate, w_right_gate, w_out_gate,
            mask_flat, left_d, right_d, out_gate,
            M, N, H, N2, C,
            HAS_MASK=has_mask,
            BM=BM,
            BH=BH,
            BK=BK,
            CK=CK,
        )
    else:
        CU = max(16, triton.next_power_of_2(C))
        BH = 128
        if CU <= 512:
            BM = 32
        else:
            BM = 16
        grid = (triton.cdiv(M, BM),)
        fused_ln_proj_generic_kernel[grid](
            x, ln_w, ln_b,
            w_left, w_right, w_left_gate, w_right_gate, w_out_gate,
            mask_flat, left_d, right_d, out_gate,
            M, N, H, N2, C,
            HAS_MASK=has_mask,
            BM=BM,
            BH=BH,
            CU=CU,
        )

    # Triangle product: for each hidden dimension d, out_d = L_d @ R_d^T.
    torch.backends.cuda.matmul.allow_bf16_reduced_precision_reduction = True
    batch = B * H
    left_b = left_d.view(batch, N, N)
    right_b = right_d.view(batch, N, N)
    out_d = torch.bmm(left_b, right_b.transpose(1, 2))

    # Fused output LN + out gate + final to_out GEMM, writing float32.
    out = torch.empty((M, C), dtype=torch.float32, device=device)

    BH = max(16, triton.next_power_of_2(H))
    if H <= 128:
        BM = 64
    elif H <= 256:
        BM = 32
    else:
        BM = 16

    if C <= 128:
        BC = max(16, triton.next_power_of_2(C))
    else:
        BC = 128

    grid_post = (triton.cdiv(M, BM),)
    post_final_kernel[grid_post](
        out_d, out_gate, to_out_norm_w, to_out_norm_b, w_to_out, out,
        M, N, H, N2, C,
        1e-5,
        BM=BM,
        BH=BH,
        BC=BC,
    )

    return out.view(B, N, N, C)
