import torch
import triton
import triton.language as tl


def _prep_weight(weights: dict, name: str, transpose: bool = False) -> torch.Tensor:
    w = weights[name].to(torch.float32, copy=False)
    if not w.is_contiguous():
        w = w.contiguous()
    if transpose:
        w = w.T.contiguous()
    return w


@triton.jit
def _fused_preop_kernel(
    x_ptr, mask_ptr,
    norm_w_ptr, norm_b_ptr,
    wl_ptr, wr_ptr, wlg_ptr, wrg_ptr, wog_ptr,
    l_ptr, r_ptr, og_ptr,
    S, C, H,
    BLOCK_P: tl.constexpr,
    BLOCK_C_ALL: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
    SMALL_C: tl.constexpr,
    HAS_MASK: tl.constexpr,
    PREC: tl.constexpr,
):
    pid_b = tl.program_id(0).to(tl.int64)
    pid_p = tl.program_id(1)

    offs_p = pid_p * BLOCK_P + tl.arange(0, BLOCK_P)
    p_mask = offs_p < S

    if HAS_MASK:
        m = tl.load(mask_ptr + pid_b * S + offs_p, mask=p_mask, other=0.0).to(tl.float32)
    else:
        m = tl.zeros((BLOCK_P,), dtype=tl.float32)

    base_x = pid_b * C * S

    if SMALL_C:
        offs_c = tl.arange(0, BLOCK_C_ALL)
        c_mask = offs_c < C

        x = tl.load(
            base_x + offs_p[:, None] * C + offs_c[None, :],
            mask=p_mask[:, None] & c_mask[None, :],
            other=0.0,
        )

        sum_x = tl.sum(x, axis=1)
        sum_sq = tl.sum(x * x, axis=1)
        mean = sum_x / C
        var = sum_sq / C - mean * mean
        var = tl.where(var > 0.0, var, 0.0)
        rstd = 1.0 / tl.sqrt(var + 1e-5)

        nw = tl.load(norm_w_ptr + offs_c, mask=c_mask, other=0.0)
        nb = tl.load(norm_b_ptr + offs_c, mask=c_mask, other=0.0)

        xw = (x * rstd[:, None] - mean[:, None] * rstd[:, None]) * nw[None, :] + nb[None, :]
        xw = tl.where(p_mask[:, None] & c_mask[None, :], xw, 0.0)

        for h0 in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h0 * BLOCK_H + tl.arange(0, BLOCK_H)
            h_mask = offs_h < H
            wmask = c_mask[:, None] & h_mask[None, :]
            store_mask = h_mask[:, None] & p_mask[None, :]

            base_lh = pid_b * H * S

            # left
            wl = tl.load(wl_ptr + offs_c[:, None] * H + offs_h[None, :], mask=wmask, other=0.0)
            left = tl.dot(xw, wl, input_precision=PREC)

            wlg = tl.load(wlg_ptr + offs_c[:, None] * H + offs_h[None, :], mask=wmask, other=0.0)
            left_gate = tl.dot(xw, wlg, input_precision=PREC)

            val = left * (1.0 / (1.0 + tl.exp(-left_gate)))
            val = val * m[:, None]
            tl.store(base_lh + offs_h[:, None] * S + offs_p[None, :], tl.trans(val), mask=store_mask)

            # right
            wr = tl.load(wr_ptr + offs_c[:, None] * H + offs_h[None, :], mask=wmask, other=0.0)
            right = tl.dot(xw, wr, input_precision=PREC)

            wrg = tl.load(wrg_ptr + offs_c[:, None] * H + offs_h[None, :], mask=wmask, other=0.0)
            right_gate = tl.dot(xw, wrg, input_precision=PREC)

            val = right * (1.0 / (1.0 + tl.exp(-right_gate)))
            val = val * m[:, None]
            tl.store(base_lh + offs_h[:, None] * S + offs_p[None, :], tl.trans(val), mask=store_mask)

            # out gate
            wog = tl.load(wog_ptr + offs_c[:, None] * H + offs_h[None, :], mask=wmask, other=0.0)
            out_gate = tl.dot(xw, wog, input_precision=PREC)

            tl.store(
                base_lh + offs_h[:, None] * S + offs_p[None, :],
                tl.trans(1.0 / (1.0 + tl.exp(-out_gate))),
                mask=store_mask,
            )
    else:
        sum_x = tl.zeros((BLOCK_P,), dtype=tl.float32)
        sum_sq = tl.zeros((BLOCK_P,), dtype=tl.float32)

        # Pass 1: layer norm statistics.
        for c0 in range(0, tl.cdiv(C, BLOCK_K)):
            offs_c = c0 * BLOCK_K + tl.arange(0, BLOCK_K)
            c_mask = offs_c < C
            x = tl.load(
                base_x + offs_p[:, None] * C + offs_c[None, :],
                mask=p_mask[:, None] & c_mask[None, :],
                other=0.0,
            )
            sum_x += tl.sum(x, axis=1)
            sum_sq += tl.sum(x * x, axis=1)

        mean = sum_x / C
        var = sum_sq / C - mean * mean
        var = tl.where(var > 0.0, var, 0.0)
        rstd = 1.0 / tl.sqrt(var + 1e-5)

        for h0 in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h0 * BLOCK_H + tl.arange(0, BLOCK_H)
            h_mask = offs_h < H
            store_mask = h_mask[:, None] & p_mask[None, :]
            base_lh = pid_b * H * S

            # left + left gate
            left_acc = tl.zeros((BLOCK_P, BLOCK_H), dtype=tl.float32)
            left_gate_acc = tl.zeros((BLOCK_P, BLOCK_H), dtype=tl.float32)
            for c0 in range(0, tl.cdiv(C, BLOCK_K)):
                offs_c = c0 * BLOCK_K + tl.arange(0, BLOCK_K)
                c_mask = offs_c < C
                wmask = c_mask[:, None] & h_mask[None, :]

                x = tl.load(
                    base_x + offs_p[:, None] * C + offs_c[None, :],
                    mask=p_mask[:, None] & c_mask[None, :],
                    other=0.0,
                )
                nw = tl.load(norm_w_ptr + offs_c, mask=c_mask, other=0.0)
                nb = tl.load(norm_b_ptr + offs_c, mask=c_mask, other=0.0)
                xw = (x * rstd[:, None] - mean[:, None] * rstd[:, None]) * nw[None, :] + nb[None, :]
                xw = tl.where(p_mask[:, None] & c_mask[None, :], xw, 0.0)

                wlc = tl.load(wl_ptr + offs_c[:, None] * H + offs_h[None, :], mask=wmask, other=0.0)
                left_acc = tl.dot(xw, wlc, left_acc, input_precision=PREC)

                wlgc = tl.load(wlg_ptr + offs_c[:, None] * H + offs_h[None, :], mask=wmask, other=0.0)
                left_gate_acc = tl.dot(xw, wlgc, left_gate_acc, input_precision=PREC)

            val = left_acc * (1.0 / (1.0 + tl.exp(-left_gate_acc)))
            val = val * m[:, None]
            tl.store(base_lh + offs_h[:, None] * S + offs_p[None, :], tl.trans(val), mask=store_mask)

            # right + right gate
            right_acc = tl.zeros((BLOCK_P, BLOCK_H), dtype=tl.float32)
            right_gate_acc = tl.zeros((BLOCK_P, BLOCK_H), dtype=tl.float32)
            for c0 in range(0, tl.cdiv(C, BLOCK_K)):
                offs_c = c0 * BLOCK_K + tl.arange(0, BLOCK_K)
                c_mask = offs_c < C
                wmask = c_mask[:, None] & h_mask[None, :]

                x = tl.load(
                    base_x + offs_p[:, None] * C + offs_c[None, :],
                    mask=p_mask[:, None] & c_mask[None, :],
                    other=0.0,
                )
                nw = tl.load(norm_w_ptr + offs_c, mask=c_mask, other=0.0)
                nb = tl.load(norm_b_ptr + offs_c, mask=c_mask, other=0.0)
                xw = (x * rstd[:, None] - mean[:, None] * rstd[:, None]) * nw[None, :] + nb[None, :]
                xw = tl.where(p_mask[:, None] & c_mask[None, :], xw, 0.0)

                wrc = tl.load(wr_ptr + offs_c[:, None] * H + offs_h[None, :], mask=wmask, other=0.0)
                right_acc = tl.dot(xw, wrc, right_acc, input_precision=PREC)

                wrgc = tl.load(wrg_ptr + offs_c[:, None] * H + offs_h[None, :], mask=wmask, other=0.0)
                right_gate_acc = tl.dot(xw, wrgc, right_gate_acc, input_precision=PREC)

            val = right_acc * (1.0 / (1.0 + tl.exp(-right_gate_acc)))
            val = val * m[:, None]
            tl.store(base_lh + offs_h[:, None] * S + offs_p[None, :], tl.trans(val), mask=store_mask)

            # out gate
            out_gate_acc = tl.zeros((BLOCK_P, BLOCK_H), dtype=tl.float32)
            for c0 in range(0, tl.cdiv(C, BLOCK_K)):
                offs_c = c0 * BLOCK_K + tl.arange(0, BLOCK_K)
                c_mask = offs_c < C
                wmask = c_mask[:, None] & h_mask[None, :]

                x = tl.load(
                    base_x + offs_p[:, None] * C + offs_c[None, :],
                    mask=p_mask[:, None] & c_mask[None, :],
                    other=0.0,
                )
                nw = tl.load(norm_w_ptr + offs_c, mask=c_mask, other=0.0)
                nb = tl.load(norm_b_ptr + offs_c, mask=c_mask, other=0.0)
                xw = (x * rstd[:, None] - mean[:, None] * rstd[:, None]) * nw[None, :] + nb[None, :]
                xw = tl.where(p_mask[:, None] & c_mask[None, :], xw, 0.0)

                wogc = tl.load(wog_ptr + offs_c[:, None] * H + offs_h[None, :], mask=wmask, other=0.0)
                out_gate_acc = tl.dot(xw, wogc, out_gate_acc, input_precision=PREC)

            tl.store(
                base_lh + offs_h[:, None] * S + offs_p[None, :],
                tl.trans(1.0 / (1.0 + tl.exp(-out_gate_acc))),
                mask=store_mask,
            )


@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    S, N,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    EVEN: tl.constexpr,
    PREC: tl.constexpr,
):
    pid_bh = tl.program_id(0).to(tl.int64)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    base = pid_bh * S

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mi = offs_i < N
    mj = offs_j < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    for k0 in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k0 * BLOCK_K + tl.arange(0, BLOCK_K)

        a_ptrs = base + offs_i[:, None] * N + offs_k[None, :]
        b_ptrs = base + offs_j[:, None] * N + offs_k[None, :]

        if EVEN:
            a = tl.load(a_ptr + a_ptrs)
            b = tl.load(b_ptr + b_ptrs)
        else:
            mk = offs_k < N
            a = tl.load(a_ptr + a_ptrs, mask=mi[:, None] & mk[None, :], other=0.0)
            b = tl.load(b_ptr + b_ptrs, mask=mj[:, None] & mk[None, :], other=0.0)

        acc = tl.dot(a, tl.trans(b), acc, input_precision=PREC)

    c_ptrs = base + offs_i[:, None] * N + offs_j[None, :]
    if EVEN:
        tl.store(c_ptr + c_ptrs, acc)
    else:
        tl.store(c_ptr + c_ptrs, acc, mask=mi[:, None] & mj[None, :])


@triton.jit
def _final_norm_gate_proj_kernel(
    out_h_ptr, og_h_ptr, out_ptr,
    norm_w_ptr, norm_b_ptr, w_ptr,
    S, C, H,
    BLOCK_P: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_C: tl.constexpr,
    SINGLE_H: tl.constexpr,
    PREC: tl.constexpr,
):
    if SINGLE_H:
        pid_p = tl.program_id(0)
        b = tl.program_id(1).to(tl.int64)

        offs_p = pid_p * BLOCK_P + tl.arange(0, BLOCK_P)
        p_mask = offs_p < S

        offs_h = tl.arange(0, BLOCK_H)
        h_mask = offs_h < H

        base_h = b * H * S
        load_mask = h_mask[:, None] & p_mask[None, :]

        x = tl.load(
            out_h_ptr + base_h + offs_h[:, None] * S + offs_p[None, :],
            mask=load_mask,
            other=0.0,
        )

        mean = tl.sum(x, axis=0) / H
        diff = x - mean[None, :]
        var = tl.sum(diff * diff, axis=0) / H
        var = tl.where(var > 0.0, var, 0.0)
        rstd = 1.0 / tl.sqrt(var + 1e-5)

        nw = tl.load(norm_w_ptr + offs_h, mask=h_mask, other=0.0)
        nb = tl.load(norm_b_ptr + offs_h, mask=h_mask, other=0.0)

        scale = nw[:, None] * rstd[None, :]
        shift = nb[:, None] - mean[None, :] * scale
        y = x * scale + shift

        gate = tl.load(
            og_h_ptr + base_h + offs_h[:, None] * S + offs_p[None, :],
            mask=load_mask,
            other=0.0,
        )
        y = y * gate

        base_out = b * C * S
        for c0 in range(0, tl.cdiv(C, BLOCK_C)):
            offs_c = c0 * BLOCK_C + tl.arange(0, BLOCK_C)
            c_mask = offs_c < C

            w = tl.load(
                w_ptr + offs_c[:, None] * H + offs_h[None, :],
                mask=c_mask[:, None] & h_mask[None, :],
                other=0.0,
            )
            acc = tl.dot(w, y, input_precision=PREC)

            out_ptrs = base_out + offs_p[:, None] * C + offs_c[None, :]
            tl.store(out_ptr + out_ptrs, tl.trans(acc), mask=p_mask[:, None] & c_mask[None, :])
    else:
        pid_p = tl.program_id(0)
        pid_c = tl.program_id(1)
        b = tl.program_id(2).to(tl.int64)

        offs_p = pid_p * BLOCK_P + tl.arange(0, BLOCK_P)
        p_mask = offs_p < S

        offs_c = pid_c * BLOCK_C + tl.arange(0, BLOCK_C)
        c_mask = offs_c < C

        base_h = b * H * S

        sum_x = tl.zeros((BLOCK_P,), dtype=tl.float32)
        sum_sq = tl.zeros((BLOCK_P,), dtype=tl.float32)

        for h0 in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h0 * BLOCK_H + tl.arange(0, BLOCK_H)
            h_mask = offs_h < H
            load_mask = h_mask[:, None] & p_mask[None, :]

            x = tl.load(
                out_h_ptr + base_h + offs_h[:, None] * S + offs_p[None, :],
                mask=load_mask,
                other=0.0,
            )
            sum_x += tl.sum(x, axis=0)
            sum_sq += tl.sum(x * x, axis=0)

        mean = sum_x / H
        var = sum_sq / H - mean * mean
        var = tl.where(var > 0.0, var, 0.0)
        rstd = 1.0 / tl.sqrt(var + 1e-5)

        acc = tl.zeros((BLOCK_C, BLOCK_P), dtype=tl.float32)

        for h0 in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h0 * BLOCK_H + tl.arange(0, BLOCK_H)
            h_mask = offs_h < H
            load_mask = h_mask[:, None] & p_mask[None, :]

            x = tl.load(
                out_h_ptr + base_h + offs_h[:, None] * S + offs_p[None, :],
                mask=load_mask,
                other=0.0,
            )
            gate = tl.load(
                og_h_ptr + base_h + offs_h[:, None] * S + offs_p[None, :],
                mask=load_mask,
                other=0.0,
            )

            nw = tl.load(norm_w_ptr + offs_h, mask=h_mask, other=0.0)
            nb = tl.load(norm_b_ptr + offs_h, mask=h_mask, other=0.0)

            scale = nw[:, None] * rstd[None, :]
            shift = nb[:, None] - mean[None, :] * scale
            y = x * scale + shift
            y = y * gate

            w = tl.load(
                w_ptr + offs_c[:, None] * H + offs_h[None, :],
                mask=c_mask[:, None] & h_mask[None, :],
                other=0.0,
            )
            acc = tl.dot(w, y, acc, input_precision=PREC)

        base_out = b * C * S
        out_ptrs = base_out + offs_p[:, None] * C + offs_c[None, :]
        tl.store(out_ptr + out_ptrs, tl.trans(acc), mask=p_mask[:, None] & c_mask[None, :])


def custom_kernel(data):
    """
    TriMul outgoing pass:
      - fuses LayerNorm, left/right projections, gates, masks, and sigmoid
        into H-major tensors for Tensor Core-friendly batched GEMMs;
      - computes out[b,h,i,j] = sum_k left[b,h,i,k] * right[b,h,j,k]
        with a Triton batched GEMM (A @ B^T per h);
      - fuses the final LayerNorm, out-gate, and to_out projection into the
        final [B, N, N, C] output.
    """
    if len(data) == 4:
        x_in, mask, weights, config = data
    else:
        x_in, weights, config = data
        mask = None

    B, N, _, C = x_in.shape
    H = int(config["hidden_dim"])
    S = N * N
    device = x_in.device

    x = x_in.to(torch.float32, copy=False)
    if not x.is_contiguous():
        x = x.contiguous()

    has_mask = mask is not None and not bool(config.get("nomask", False))
    if has_mask:
        m = mask
        if m.dim() == 2:
            m = m.unsqueeze(0)
        if m.shape[0] == 1 and B != 1:
            m = m.expand(B, N, N)
        mask_flat = m.to(torch.float32).contiguous().view(B, S)
    else:
        mask_flat = x

    norm_w = _prep_weight(weights, "norm.weight")
    norm_b = _prep_weight(weights, "norm.bias")

    wl_T = _prep_weight(weights, "left_proj.weight", transpose=True)
    wr_T = _prep_weight(weights, "right_proj.weight", transpose=True)
    wlg_T = _prep_weight(weights, "left_gate.weight", transpose=True)
    wrg_T = _prep_weight(weights, "right_gate.weight", transpose=True)
    wog_T = _prep_weight(weights, "out_gate.weight", transpose=True)

    to_norm_w = _prep_weight(weights, "to_out_norm.weight")
    to_norm_b = _prep_weight(weights, "to_out_norm.bias")
    to_out_w = _prep_weight(weights, "to_out.weight")

    left_h = torch.empty((B, H, S), device=device, dtype=torch.float32)
    right_h = torch.empty((B, H, S), device=device, dtype=torch.float32)
    og_h = torch.empty((B, H, S), device=device, dtype=torch.float32)

    PREC = "tf32"
    small_c = C <= 128

    block_p_pre = 64
    block_c_all = max(16, triton.next_power_of_2(C))
    block_k_pre = 64
    if H <= 128:
        block_h_pre = max(16, triton.next_power_of_2(H))
    else:
        block_h_pre = 128

    pre_grid = (B, triton.cdiv(S, block_p_pre))
    _fused_preop_kernel[pre_grid](
        x,
        mask_flat,
        norm_w,
        norm_b,
        wl_T,
        wr_T,
        wlg_T,
        wrg_T,
        wog_T,
        left_h,
        right_h,
        og_h,
        S, C, H,
        BLOCK_P=block_p_pre,
        BLOCK_C_ALL=block_c_all,
        BLOCK_K=block_k_pre,
        BLOCK_H=block_h_pre,
        SMALL_C=small_c,
        HAS_MASK=has_mask,
        PREC=PREC,
        num_warps=8,
        num_stages=3,
    )

    # No longer need raw input for fused kernels.
    del x

    out_h = torch.empty((B, H, S), device=device, dtype=torch.float32)

    if N <= 256:
        block_i = 64
        block_j = 64
        block_k = 32
        main_warps = 4
    else:
        block_i = 128
        block_j = 128
        block_k = 32
        main_warps = 8

    even = (N % block_i == 0) and (N % block_j == 0) and (N % block_k == 0)
    gemm_grid = (B * H, triton.cdiv(N, block_i), triton.cdiv(N, block_j))

    _trimul_gemm_kernel[gemm_grid](
        left_h,
        right_h,
        out_h,
        S, N,
        BLOCK_I=block_i,
        BLOCK_J=block_j,
        BLOCK_K=block_k,
        EVEN=even,
        PREC=PREC,
        num_warps=main_warps,
        num_stages=3,
    )

    del left_h, right_h

    out = torch.empty((B, N, N, C), device=device, dtype=torch.float32)

    if H <= 128:
        bh_small = max(16, triton.next_power_of_2(H))
        single_h = (bh_small == H)
    else:
        single_h = False
        bh_small = 128

    if single_h:
        final_block_h = bh_small
    else:
        final_block_h = 128

    final_block_p = 64
    if C <= 128:
        final_block_c = max(16, triton.next_power_of_2(C))
    else:
        final_block_c = 128

    if single_h:
        final_grid = (triton.cdiv(S, final_block_p), B)
    else:
        final_grid = (triton.cdiv(S, final_block_p), triton.cdiv(C, final_block_c), B)

    _final_norm_gate_proj_kernel[final_grid](
        out_h,
        og_h,
        out,
        to_norm_w,
        to_norm_b,
        to_out_w,
        S, C, H,
        BLOCK_P=final_block_p,
        BLOCK_H=final_block_h,
        BLOCK_C=final_block_c,
        SINGLE_H=single_h,
        PREC=PREC,
        num_warps=4,
        num_stages=2,
    )

    return out
