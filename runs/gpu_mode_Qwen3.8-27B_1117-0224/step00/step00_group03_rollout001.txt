    B, N, H,

    stride_rb, stride_rj, streamline_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,

    left_ptr, right_ptr, out_ptr,
    B, N, H: tl.constexpr,
    stride_lb, stride_li, stride_lk,
    stride_rb, stride_rj, stride_rk,
    stride_ob, stride_oi, stride_oj,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
    GROUP_I: tl.constexpr,
):
    # Grid: (B*H, cdiv(N, BLOCK_I), cdiv(N, BLOCK_J))
    pid_bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    
    b = pid_bh // H
    h = pid_bh % H
    
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)
    
    # Left: left[b, i, k, h]
    # stride for b: stride_lb, for i: stride_li, for k: stride_lk, for h: 1
