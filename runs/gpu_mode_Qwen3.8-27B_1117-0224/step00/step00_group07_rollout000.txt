import torch
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak,  # A: [batch, M, K]
    stride_bb, stride_bk, stride_bn,  # B: [batch, K, N] (already transposed)
    stride_cb, stride_cm, stride_cn,  # C: [batch, M, N]
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr,
):
    # This computes C[batch] = A[batch] @ B[batch] for all batches
    # where A is [batch, M, K], B is [batch, K, N], C is [batch, M, N]
    
    pid = tl.program_id(0)
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    num_pid_in_group = GROUP_M * num_pid_n
    
    pid_m = pid // num_pid_in_group
    first_pid_m = pid_m * GROUP_M
    group_size_m = min(num_pid_m - first_pid_m, GROUP_M)
    pid_m = first_pid_m + ((pid % num_pid_in_group) % group_size_m)
    pid_n = (pid % num_pid_in_group) // group_size_m
    
    batch_id = tl.program_id(1)
    
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    
    a_ptrs = a_ptr + batch_id * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
    
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)
        acc = tl.dot(a, b, acc)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
    
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_ptr + batch_id * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(c_ptr.dtype.element_ty), mask=c_mask)

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    H = hidden_dim
    
    device = input_tensor.device

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # Step 1: LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (C,), norm_weight, norm_bias, eps=1e-5)
    
    # Step 2: Projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B, N, N, H]
    
    # Step 3: Apply mask
    if mask is not None:
        mask_expanded = mask.unsqueeze(-1)
        left = left * mask_expanded
        right = right * mask_expanded
    
    # Step 4: Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))
    
    left = left * left_gate
    right = right * right_gate
    
    # Step 5: The einsum (batched matmul)
    # out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]
    # Reshape to [B*H, N, N] for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]
    
    batch_size = B * H
    left_2d = left_perm.reshape(batch_size, N, N)
    right_2d = right_perm.reshape(batch_size, N, N)

    left_2d = left_2d.to(torch.bfloat16)
    right_2d = right_2d.to(torch.bfloat16)
    
    out_2d = torch.empty(batch_size, N, N, device=device, dtype=torch.bfloat16)
    
    # Launch Triton kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    GROUP_M = 8
    
    grid = (triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N), batch_size)
    batched_matmul_kernel[grid](
        left_2d, right_2d, out_2d,
        N, N, N,

        right_2d.stride(0), right_2d.stride(1), right_2d.stride(2),
        out_2d.stride(0), out_2d.stride(1), out_2d.stride(2),

    # Reshape back

import triton
import triton.language as tl

@triton.jit
def trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    N: tl.constexpr, H: tl.constexpr,
    stride_am: tl.constexpr, stride_ak: tl.constexpr, stride_ab: tl.constexpr,
    stride_bn: tl.constexpr, stride_bk: tl.constexpr, stride_bb: tl.constexpr,
    stride_cm: tl.constexpr, stride_cn: tl.constexpr, stride_cb: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr,
):

    batch_id = tl.program_id(1)
    
    b_idx = batch_id // H
    d_idx = batch_id % H
    
    num_pid_m = N // BLOCK_M

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    
    a_base = b_idx * stride_ab + d_idx
    b_base = b_idx * stride_bb + d_idx
    
    a_ptrs = a_ptr + a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + b_base + offs_n[:, None] * stride_bn + offs_k[None, :] * stride_bk

    hidden_dim = config["hidden_dim"]

    H = hidden_dim
    device = input_tensor.device

    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # Step 1: LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (C,), norm_weight, norm_bias, eps=1e-5)

    left = torch.nn.functional.linear(x, left_proj_weight)
    right = torch.nn.functional.linear(x, right_proj_weight)

        left = left * mask_expanded
        right = right * mask_expanded

    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))
    
    left = left * left_gate
    right = right * right_gate

    left_bf16 = left.to(torch.bfloat16)

    stride_am = N * H  # stride for i

    stride_bk = H      # stride for k
    stride_bb = N * N * H  # stride for b
    
    stride_cm = N * H  # stride for i
    stride_cn = H      # stride for j
    stride_cb = N * N * H  # stride for b
    
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    GROUP_M = 8
    
    grid = ((N // BLOCK_M) * (N // BLOCK_N), B * H)

"""
Fused TriMul (Triangle Multiplicative Update) implementation.
Uses PyTorch for projections, LayerNorms, and element-wise ops,
and a custom Triton kernel for the O(N^3) batched matmul (einsum) core.
The einsum out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
is computed as B*H independent N×N matrix multiplications using a
tiled Triton kernel with bf16 inputs and fp32 accumulation.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    N: tl.constexpr, H: tl.constexpr,
    stride_am, stride_ak, stride_ab,
    stride_bn, stride_bk, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr,
):
    """
    Computes for each (b, d) batch:
        C[b, :, :, d] = A[b, :, :, d] @ A[b, :, :, d]^T  (conceptually)
    where A = left, B = right, and:
        C[b,i,j,d] = sum_k A[b,i,k,d] * B[b,j,k,d]
    
    A and B have shape [B, N, N, H] in row-major layout.
    C has shape [B, N, N, H] in row-major layout.
    The kernel handles the batch index (b, d) via program_id(1).
    """
    pid = tl.program_id(0)
    batch_id = tl.program_id(1)

    b_idx = batch_id // H
    d_idx = batch_id % H

    # Grouped scheduling for L2 locality
    num_pid_m = N // BLOCK_M
    num_pid_n = N // BLOCK_N
    num_pid_in_group = GROUP_M * num_pid_n

    group_id = pid // num_pid_in_group
    first_pid_m = group_id * GROUP_M
    group_size_m = min(num_pid_m - first_pid_m, GROUP_M)
    pid_m = first_pid_m + (pid % num_pid_in_group) % group_size_m
    pid_n = (pid % num_pid_in_group) // group_size_m

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Base offsets for this (b, d) slice
    a_base = b_idx * stride_ab + d_idx
    b_base = b_idx * stride_bb + d_idx

    # A tile: [BLOCK_M, BLOCK_K] -> A[i, k] = left[b, i, k, d]
    a_ptrs = a_ptr + a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    # B tile: [BLOCK_K, BLOCK_N] -> B[k, j] = right[b, j, k, d] (loaded transposed)
    b_ptrs = b_ptr + b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for _k in range(0, N // BLOCK_K):
        a = tl.load(a_ptrs)
        b_tile = tl.load(b_ptrs)
        acc = tl.dot(a, b_tile, acc)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store C[i, j] = out[b, i, j, d]
    c_base = b_idx * stride_cb + d_idx
    c_ptrs = c_ptr + c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(c_ptr.dtype.element_ty))


def custom_kernel(data):
    """
    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor|None, 
                       weights: Dict[str, torch.Tensor], config: Dict)
            - input: [bs, seq_len, seq_len, dim]
            - mask:  [bs, seq_len, seq_len] or None
            - weights: model weight dictionary
            - config: {"dim": int, "hidden_dim": int}
    Returns:
        output: [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    B, N, _, C = input_tensor.shape
    H = hidden_dim
    device = input_tensor.device

    # Access weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Step 1: Input LayerNorm over last dim (C)
    x = torch.nn.functional.layer_norm(
        input_tensor, (C,), norm_weight, norm_bias, eps=1e-5
    )

    # Step 2: Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B, N, N, H]

    # Step 3: Apply mask
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # Step 4: Sigmoid gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Step 5: Core einsum via Triton batched matmul
    # out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    left_bf16 = left.to(torch.bfloat16)
    right_bf16 = right.to(torch.bfloat16)

    out = torch.empty(B, N, N, H, device=device, dtype=torch.float32)

    # Strides for [B, N, N, H] row-major tensor
    stride_am = N * H       # stride along i (dim 1)
    stride_ak = H           # stride along k (dim 2)
    stride_ab = N * N * H   # stride along b (dim 0)

    stride_bn = N * H       # stride along j (dim 1)
    stride_bk = H           # stride along k (dim 2)
    stride_bb = N * N * H   # stride along b (dim 0)

    stride_cm = N * H       # stride along i (dim 1)
    stride_cn = H           # stride along j (dim 2)
    stride_cb = N * N * H   # stride along b (dim 0)

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    GROUP_M = 8

    grid = ((N // BLOCK_M) * (N // BLOCK_N), B * H)

    _trimul_matmul_kernel[grid](
        left_bf16, right_bf16, out,
        N, H,
        stride_am, stride_ak, stride_ab,
        stride_bn, stride_bk, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, GROUP_M=GROUP_M,
        num_warps=8, num_stages=3,
    )

    # Step 6: Output LayerNorm over last dim (H)
    out = torch.nn.functional.layer_norm(
        out, (H,), to_out_norm_weight, to_out_norm_bias, eps=1e-5
    )

    # Step 7: Apply output gate
    out = out * out_gate

    # Step 8: Final linear projection
    out = torch.nn.functional.linear(out, to_out_weight)

    return out.to(torch.float32)
