@triton.jit
def _layer_norm_kernel(
    x_ptr, y_ptr, g_ptr, b_ptr,
    M, C, eps,
    BLOCK_M: tl.constexpr, BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = (pid * BLOCK_M + tl.arange(0, BLOCK_M)).to(tl.int64)
    cols = tl.arange(0, BLOCK_C)
    rmask = rows < M
    cmask = cols < C
    mask = rmask[:, None] & cmask[None, :]
    x = tl.load(x_ptr + rows[:, None] * C + cols[None, :], mask=mask, other=0.0).to(tl.float32)
    x = tl.where(mask, x, 0.0)
    mean = tl.sum(x, axis=1) / C
    diff = tl.where(mask, x - mean[:, None], 0.0)
    var = tl.sum(diff * diff, axis=1) / C
    rstd = 1.0 / tl.sqrt(var + eps)
    g = tl.load(g_ptr + cols, mask=cmask, other=0.0).to(tl.float32)
    b = tl.load(b_ptr + cols, mask=cmask, other=0.0).to(tl.float32)
    y = diff * rstd[:, None] * g[None, :] + b[None, :]
    tl.store(y_ptr + rows[:, None] * C + cols[None, :], y.to(tl.bfloat16), mask=mask)

2. PyTorch cuBLAS batch of linear projections: [B,N,N,C] @ packed_weights^T
   producing left/right and the three sigmoid gates in one GEMM.
3. Triton pointwise kernel: applies mask and sigmoid gates, then stores
   left/right/out_gate in H-first layout [B, H, N*N] for batched matmul.
4. PyTorch cuBLAS batched matmul:
       G[b,h,i,j] = sum_k left[b,h,i,k] * right[b,h,j,k]
   stored in H-first layout [B*H, N, N].
5. Triton fused final kernel: applies LayerNorm over H, multiplies by the
   stored out_gate, and performs the final linear projection in one pass,
   writing the float32 output.
"""


@triton.jit
def _layer_norm_kernel(
    x_ptr,
    y_ptr,
    gamma_ptr,
    beta_ptr,
    M,
    C,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = (pid * BLOCK_M + tl.arange(0, BLOCK_M)).to(tl.int64)
    cols = tl.arange(0, BLOCK_C)

    row_mask = rows < M
    col_mask = cols < C
    mask = row_mask[:, None] & col_mask[None, :]

    x = tl.load(x_ptr + rows[:, None] * C + cols[None, :], mask=mask, other=0.0).to(tl.float32)
    x = tl.where(mask, x, 0.0)

    mean = tl.sum(x, axis=1) / C
    diff = tl.where(mask, x - mean[:, None], 0.0)
    var = tl.sum(diff * diff, axis=1) / C
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + eps)

    gamma = tl.load(gamma_ptr + cols, mask=col_mask, other=0.0).to(tl.float32)
    beta = tl.load(beta_ptr + cols, mask=col_mask, other=0.0).to(tl.float32)

    y = diff * rstd[:, None] * gamma[None, :] + beta[None, :]
    tl.store(y_ptr + rows[:, None] * C + cols[None, :], y.to(tl.bfloat16), mask=mask)


@triton.jit
def _pointwise_gate_kernel(
    proj_ptr,
    left_ptr,
    right_ptr,
    outgate_ptr,
    mask_ptr,
    M,
    NN,
    H,
    HNN,
    BLOCK_M: tl.constexpr,
    H_BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """
    proj layout: [B*N*N, 5*H]
      0:H      => left raw
      H:2H     => right raw
      2H:3H    => left gate raw
      3H:4H    => right gate raw
      4H:5H    => out gate raw

    Stores:
      left, right, outgate in H-first layout [B, H, N*N].
    """
    pid = tl.program_id(0)
    rows = (pid * BLOCK_M + tl.arange(0, BLOCK_M)).to(tl.int64)
    row_mask = rows < M

    hcols = tl.arange(0, H_BLOCK).to(tl.int64)
    batch = rows // NN
    rem = rows - batch * NN

    if HAS_MASK:
        m = tl.load(mask_ptr + rows, mask=row_mask, other=0.0).to(tl.float32)[:, None]
    else:
        m = 1.0

    five_H = 5 * H
    for h_off in range(0, H, H_BLOCK):
        hi = h_off + hcols
        h_valid = hi < H
        load_mask = row_mask[:, None] & h_valid[None, :]

        base = rows[:, None] * five_H

        l_raw = tl.load(proj_ptr + base + hi[None, :], mask=load_mask, other=0.0).to(tl.float32)
        r_raw = tl.load(proj_ptr + base + (H + hi)[None, :], mask=load_mask, other=0.0).to(tl.float32)
        l_gate = tl.load(proj_ptr + base + (2 * H + hi)[None, :], mask=load_mask, other=0.0).to(tl.float32)
        r_gate = tl.load(proj_ptr + base + (3 * H + hi)[None, :], mask=load_mask, other=0.0).to(tl.float32)
        og_raw = tl.load(proj_ptr + base + (4 * H + hi)[None, :], mask=load_mask, other=0.0).to(tl.float32)

        left = l_raw * tl.sigmoid(l_gate) * m
        right = r_raw * tl.sigmoid(r_gate) * m
        outgate = tl.sigmoid(og_raw)

        # H-first store:
        # offset = b * H * NN + h * NN + (i*N + k)
        left_off = left_ptr + batch[:, None] * HNN + hi[None, :] * NN + rem[:, None]
        right_off = right_ptr + batch[:, None] * HNN + hi[None, :] * NN + rem[:, None]
        og_off = outgate_ptr + batch[:, None] * HNN + hi[None, :] * NN + rem[:, None]

        store_mask = row_mask[:, None] & h_valid[None, :]
        tl.store(left_off, left.to(tl.bfloat16), mask=store_mask)
        tl.store(right_off, right.to(tl.bfloat16), mask=store_mask)
        tl.store(og_off, outgate.to(tl.bfloat16), mask=store_mask)


@triton.jit
def _final_fused_kernel(
    g_ptr,
    gate_ptr,
    out_ptr,
    w_ptr,
    gamma_ptr,
    beta_ptr,
    M,
    NN,
    C,
    H,
    HNN,
    eps,
    BLOCK_P: tl.constexpr,
    BLOCK_C: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    g_ptr / gate_ptr: H-first layout [B, H, N*N]
    out_ptr:          [B, N, N, C] float32

    For each flat pair token (b, i, j), loads the H-vector,
    applies LayerNorm, multiplies by out_gate, and raises to C.
    """
    pid_p = tl.program_id(0)
    pid_c = tl.program_id(1)

    rows = (pid_p * BLOCK_P + tl.arange(0, BLOCK_P)).to(tl.int64)
    row_mask = rows < M

    batch = rows // NN
    rem = rows - batch * NN

    hs = tl.arange(0, BLOCK_H).to(tl.int64)
    h_valid = hs < H

    cs = pid_c * BLOCK_C + tl.arange(0, BLOCK_C)
    c_valid = cs < C

    load_mask = row_mask[:, None] & h_valid[None, :]

    g_off = batch[:, None] * HNN + rem[:, None] + hs[None, :] * NN
    g = tl.load(g_ptr + g_off, mask=load_mask, other=0.0).to(tl.float32)
    g = tl.where(load_mask, g, 0.0)

    mean = tl.sum(g, axis=1) / H
    diff = tl.where(load_mask, g - mean[:, None], 0.0)
    var = tl.sum(diff * diff, axis=1) / H
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + eps)

    gamma = tl.load(gamma_ptr + hs, mask=h_valid, other=0.0).to(tl.float32)
    beta = tl.load(beta_ptr + hs, mask=h_valid, other=0.0).to(tl.float32)

    y = diff * rstd[:, None] * gamma[None, :] + beta[None, :]

    gate = tl.load(gate_ptr + g_off, mask=load_mask, other=0.0).to(tl.float32)
    gated = (y * gate).to(tl.bfloat16)

    w = tl.load(
        w_ptr + hs[:, None] * C + cs[None, :],
        mask=h_valid[:, None] & c_valid[None, :],
        other=0.0,
    )

    acc = tl.dot(gated, w)

    out_mask = row_mask[:, None] & c_valid[None, :]
    tl.store(out_ptr + rows[:, None] * C + cs[None, :], acc.to(tl.float32), mask=out_mask)


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    with torch.no_grad():
        B, N, _, C = input_tensor.shape
        try:
            H = int(config.get("hidden_dim", weights["left_proj.weight"].shape[0]))
        except Exception:
            H = int(weights["left_proj.weight"].shape[0])

        NN = N * N
        M = B * NN
        HNN = H * NN
        device = input_tensor.device
        eps = float(config.get("eps", 1e-5))

        x = input_tensor.contiguous()

        norm_w = weights["norm.weight"].to(device=device, dtype=torch.float32).contiguous()
        norm_b = weights["norm.bias"].to(device=device, dtype=torch.float32).contiguous()

        # -----------------------
        # 1) LayerNorm -> bf16
        # -----------------------
        block_c_ln = triton.next_power_of_2(C)
        if block_c_ln <= 128:
            block_m_ln = 64
        elif block_c_ln <= 512:
            block_m_ln = 32
        else:
            block_m_ln = 16

        x_norm = torch.empty((M, C), dtype=torch.bfloat16, device=device)
        _layer_norm_kernel[(triton.cdiv(M, block_m_ln),)](
            x,
            x_norm,
            norm_w,
            norm_b,
            M,
            C,
            eps,
            BLOCK_M=block_m_ln,
            BLOCK_C=block_c_ln,
            num_warps=4,
        )
        del x

        # -----------------------
        # 2) Five projections in one packed GEMM
        # -----------------------
        w5 = torch.empty((5 * H, C), dtype=torch.bfloat16, device=device)
        proj_names = [
            "left_proj.weight",
            "right_proj.weight",
            "left_gate.weight",
            "right_gate.weight",
            "out_gate.weight",
        ]
        for i, name in enumerate(proj_names):
            w5[i * H:(i + 1) * H].copy_(weights[name], non_blocking=True)

        w5t = w5.t().contiguous()
        proj = x_norm @ w5t  # [M, 5*H], bf16
        del x_norm, w5, w5t

        # -----------------------
        # 3) Mask + gates + H-first store
        # -----------------------
        left = torch.empty((B, H, NN), dtype=torch.bfloat16, device=device)
        right = torch.empty((B, H, NN), dtype=torch.bfloat16, device=device)
        outgate = torch.empty((B, H, NN), dtype=torch.bfloat16, device=device)

        has_mask = mask is not None
        mask_flat = None
        if has_mask:
            if mask.dim() == 4 and mask.shape[-1] == 1:
                mask = mask.squeeze(-1)
            mask_flat = mask.reshape(-1).to(torch.float32).contiguous()

        mask_arg = mask_flat if has_mask else proj
        block_m_pw = 32
        h_block_pw = 128

        _pointwise_gate_kernel[(triton.cdiv(M, block_m_pw),)](
            proj,
            left,
            right,
            outgate,
            mask_arg,
            M,
            NN,
            H,
            HNN,
            BLOCK_M=block_m_pw,
            H_BLOCK=h_block_pw,
            HAS_MASK=has_mask,
            num_warps=4,
        )

        del proj
        if has_mask:
            del mask_flat

        # -----------------------
        # 4) Pair multiplication: G[b,h,i,j] = sum_k L[b,h,i,k] * R[b,h,j,k]
        # -----------------------
        left_flat = left.view(B * H, N, N)
        right_flat = right.view(B * H, N, N)
        g = torch.bmm(left_flat, right_flat.transpose(1, 2))  # [B*H, N, N], bf16

        del left, right, left_flat, right_flat

        # -----------------------
        # 5) Final LayerNorm + out_gate + linear projection
        # -----------------------
        norm2_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float32).contiguous()
        norm2_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float32).contiguous()

        to_out_w = (
            weights["to_out.weight"]
            .t()
            .contiguous()
            .to(device=device, dtype=torch.bfloat16)
        )  # [H, C]

        out = torch.empty((B, N, N, C), dtype=torch.float32, device=device)

        block_h = triton.next_power_of_2(H)

        if C <= 128:
            block_c = 128
        else:
            block_c = 256

        if block_h >= 512:
            block_c = 64
        elif block_h >= 256:
            block_c = 128

        if block_h <= 128:
            block_p = 64
        elif block_h <= 256:
            block_p = 32
        else:
            block_p = 16

        grid_final = (triton.cdiv(M, block_p), triton.cdiv(C, block_c))

        _final_fused_kernel[grid_final](
            g,
            outgate,
            out,
            to_out_w,
            norm2_w,
            norm2_b,
            M,
            NN,
            C,
            H,
            HNN,
            eps,
            BLOCK_P=block_p,
            BLOCK_C=block_c,
            BLOCK_H=block_h,
            num_warps=8,
        )

        return out
