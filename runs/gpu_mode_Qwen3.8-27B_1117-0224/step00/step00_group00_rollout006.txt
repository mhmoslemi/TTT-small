```
def _final_hfirst_kernel(o_ptr, g_ptr, w_ptr, lnw_ptr, lnb_ptr, out_ptr,

    pid_p = tl.program_id(1)
    pid_c = tl.program_id(2)

    offs_c = pid_c*BC + tl.arange(0,BC)
    offs_h = tl.arange(0,H)
    p_mask = offs_p < ns
    c_mask = offs_c < c
```

```
@triton.jit
def _ln_one_kernel(x_ptr, w_ptr, b_ptr, out_ptr, M, eps: tl.constexpr, C_CON: tl.constexpr, BM: tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid*BM + tl.arange(0, BM)
    m_mask = offs_m < M
    offs_c = tl.arange(0, C_CON)
    x_ptrs = x_ptr + offs_m[:, None].to(tl.int64) * C_CON + offs_c[None, :]
    x = tl.load(x_ptrs, mask=m_mask[:, None], other=0.0).to(tl.float32)
    mean = tl.sum(x, axis=1) / C_CON
    diff = x - mean[:, None]
    var = tl.sum(diff * diff, axis=1) / C_CON
    rstd = 1.0 / tl.sqrt(var + eps)
    w = tl.load(w_ptr + offs_c).to(tl.float32)
    bias = tl.load(b_ptr + offs_c).to(tl.float32)
    y = diff * rstd[:, None] * w[None, :] + bias[None, :]
    out_ptrs = out_ptr + offs_m[:, None].to(tl.int64) * C_CON + offs_c[None, :]
    tl.store(out_ptrs, y.to(out_ptr.dtype.element_ty), mask=m_mask[:, None])
```

```python
"""
Triton TriMul forward implementation.

Algorithm:
- Fused LayerNorm over the final channel dimension.
- One cuBLAS GEMM computes all five linear projections (left/right and three gates).
- A Triton kernel applies sigmoids, optional masking, and writes left/right/out_gate
  in a layout [B, H, N, N] that is row-major for each channel h.
- A Triton batched-GEMM kernel computes the N^3 pairwise operation:
      O[b, h, i, j] = sum_k L[b, h, i, k] * R[b, h, j, k]
- A final Triton kernel applies LayerNorm over H, multiplies by out_gate,
  and performs the final projection back to C in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _ln_one_kernel(x_ptr, w_ptr, b_ptr, out_ptr, M,
                   EPS: tl.constexpr, C_CON: tl.constexpr,
                   BM: tl.constexpr):
    pid = tl.program_id(0)
    idx = pid * BM + tl.arange(0, BM)
    m_mask = idx < M

    row = idx.to(tl.int64)
    cols = tl.arange(0, C_CON)

    x_ptrs = x_ptr + row[:, None] * C_CON + cols[None, :]
    x = tl.load(x_ptrs, mask=m_mask[:, None], other=0.0).to(tl.float32)

    mean = tl.sum(x, axis=1) / C_CON
    diff = x - mean[:, None]
    var = tl.sum(diff * diff, axis=1) / C_CON
    rstd = 1.0 / tl.sqrt(var + EPS)

    w = tl.load(w_ptr + cols).to(tl.float32)
    bias = tl.load(b_ptr + cols).to(tl.float32)

    y = diff * rstd[:, None] * w[None, :] + bias[None, :]

    out_ptrs = out_ptr + row[:, None] * C_CON + cols[None, :]
    tl.store(out_ptrs, y.to(out_ptr.dtype.element_ty), mask=m_mask[:, None])


@triton.jit
def _ln_two_kernel(x_ptr, w_ptr, b_ptr, out_ptr, M, C,
                   EPS: tl.constexpr, BM: tl.constexpr, BC: tl.constexpr):
    pid = tl.program_id(0)
    idx = pid * BM + tl.arange(0, BM)
    m_mask = idx < M

    row = idx.to(tl.int64)
    cols = tl.arange(0, BC)

    sum1 = tl.zeros((BM,), dtype=tl.float32)
    sum2 = tl.zeros((BM,), dtype=tl.float32)

    num_chunk = tl.cdiv(C, BC)

    # First pass: mean and E[x^2].
    for ci in range(0, num_chunk):
        c0 = ci * BC
        c_off = c0 + cols
        cmask = c_off < C
        mask2d = m_mask[:, None] & cmask[None, :]

        x_ptrs = x_ptr + row[:, None] * C + c_off[None, :]
        x = tl.load(x_ptrs, mask=mask2d, other=0.0).to(tl.float32)

        sum1 += tl.sum(x, axis=1)
        sum2 += tl.sum(x * x, axis=1)

    mean = sum1 / C
    var = sum2 / C - mean * mean
    var = tl.where(var > 0.0, var, 0.0)
    rstd = 1.0 / tl.sqrt(var + EPS)

    # Second pass: normalize and store.
    for ci in range(0, num_chunk):
        c0 = ci * BC
        c_off = c0 + cols
        cmask = c_off < C
        mask2d = m_mask[:, None] & cmask[None, :]

        x_ptrs = x_ptr + row[:, None] * C + c_off[None, :]
        x = tl.load(x_ptrs, mask=mask2d, other=0.0).to(tl.float32)

        w = tl.load(w_ptr + c_off, mask=cmask, other=0.0).to(tl.float32)
        bias = tl.load(b_ptr + c_off, mask=cmask, other=0.0).to(tl.float32)

        diff = x - mean[:, None]
        y = diff * rstd[:, None] * w[None, :] + bias[None, :]

        out_ptrs = out_ptr + row[:, None] * C + c_off[None, :]
        tl.store(out_ptrs, y.to(out_ptr.dtype.element_ty), mask=mask2d)


@triton.jit
def _prep_hfirst_kernel(proj_ptr, out_ptr, mask_ptr, ns, H, stride_m,
                        seg_start, gate_start,
                        APPLY_GATE: tl.constexpr, HAS_MASK: tl.constexpr,
                        BM: tl.constexpr, BH: tl.constexpr):
    b = tl.program_id(0)
    pid_p = tl.program_id(1)
    pid_h = tl.program_id(2)

    p_idx = pid_p * BM + tl.arange(0, BM)
    h_idx = pid_h * BH + tl.arange(0, BH)

    p_mask = p_idx < ns
    h_mask = h_idx < H

    p64 = p_idx.to(tl.int64)
    row = p64 + (b * ns).to(tl.int64)

    src_off = seg_start + h_idx[None, :]
    src_ptrs = proj_ptr + row[:, None] * stride_m + src_off
    vals = tl.load(src_ptrs, mask=p_mask[:, None] & h_mask[None, :], other=0.0).to(tl.float32)

    if APPLY_GATE:
        gate_off = gate_start + h_idx[None, :]
        gate_ptrs = proj_ptr + row[:, None] * stride_m + gate_off
        g = tl.load(gate_ptrs, mask=p_mask[:, None] & h_mask[None, :], other=0.0).to(tl.float32)
        gate = 1.0 / (1.0 + tl.exp(-g))

        if HAS_MASK:
            mv = tl.load(mask_ptr + row, mask=p_mask, other=0.0).to(tl.float32)
            vals = vals * gate * mv[:, None]
        else:
            vals = vals * gate
    else:
        vals = 1.0 / (1.0 + tl.exp(-vals))

    vals_t = tl.trans(vals)

    out_base = ((b * H + h_idx[:, None]).to(tl.int64) * ns) + p64[None, :]
    out_ptrs = out_ptr + out_base
    tl.store(out_ptrs, vals_t.to(out_ptr.dtype.element_ty),
             mask=h_mask[:, None] & p_mask[None, :])


@triton.jit
def _core_hfirst_kernel(L, R, O, seq, H,
                        BI: tl.constexpr, BJ: tl.constexpr, BK: tl.constexpr):
    pi = tl.program_id(0)
    pj = tl.program_id(1)
    pb = tl.program_id(2)

    h = pb % H
    b = pb // H

    ns = seq * seq
    sb = H * ns
    sh = ns
    sr = seq

    L_base = L + b * sb + h * sh
    R_base = R + b * sb + h * sh
    O_base = O + b * sb + h * sh

    offs_i = pi * BI + tl.arange(0, BI)
    offs_j = pj * BJ + tl.arange(0, BJ)
    offs_k = tl.arange(0, BK)

    i_mask = offs_i < seq
    j_mask = offs_j < seq

    l_ptrs = L_base + offs_i[:, None].to(tl.int64) * sr + offs_k[None, :]
    r_ptrs = R_base + offs_j[:, None].to(tl.int64) * sr + offs_k[None, :]

    acc = tl.zeros((BI, BJ), dtype=tl.float32)

    for k in range(0, tl.cdiv(seq, BK)):
        k_off = k * BK + offs_k
        k_mask = k_off < seq

        l = tl.load(l_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        r = tl.load(r_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0)

        acc = tl.dot(l, tl.trans(r), acc)

        l_ptrs += BK
        r_ptrs += BK

    o_ptrs = O_base + offs_i[:, None].to(tl.int64) * sr + offs_j[None, :]
    tl.store(o_ptrs, acc.to(O.dtype.element_ty), mask=i_mask[:, None] & j_mask[None, :])


@triton.jit
def _final_full_hfirst_kernel(o_ptr, g_ptr, w_ptr, lnw_ptr, lnb_ptr, out_ptr,
                              ns, c,
                              H_CON: tl.constexpr,
                              BM: tl.constexpr, BC: tl.constexpr,
                              EPS: tl.constexpr):
    b = tl.program_id(0)
    pid_p = tl.program_id(1)
    pid_c = tl.program_id(2)

    p_idx = pid_p * BM + tl.arange(0, BM)
    c_idx = pid_c * BC + tl.arange(0, BC)
    h_idx = tl.arange(0, H_CON)

    p_mask = p_idx < ns
    c_mask = c_idx < c

    p64 = p_idx.to(tl.int64)
    c64 = c_idx.to(tl.int64)

    base = ((b * H_CON + h_idx[:, None]).to(tl.int64) * ns) + p64[None, :]

    o = tl.load(o_ptr + base, mask=p_mask[None, :], other=0.0).to(tl.float32)
    g = tl.load(g_ptr + base, mask=p_mask[None, :], other=0.0).to(tl.float32)

    mean = tl.sum(o, axis=0) / H_CON
    diff = o - mean[None, :]
    var = tl.sum(diff * diff, axis=0) / H_CON
    rstd = 1.0 / tl.sqrt(var + EPS)

    lnw = tl.load(lnw_ptr + h_idx).to(tl.float32)
    lnb = tl.load(lnb_ptr + h_idx).to(tl.float32)

    a = diff * lnw[:, None] * rstd[None, :] + lnb[:, None]
    a = (a * g).to(tl.bfloat16)
    a = tl.trans(a)

    w_ptrs = w_ptr + h_idx[:, None].to(tl.int64) * c + c64[None, :]
    w = tl.load(w_ptrs, mask=c_mask[None, :], other=0.0)

    acc = tl.dot(a, w)

    out_ptrs = out_ptr + (b * ns + p64[:, None]).to(tl.int64) * c + c64[None, :]
    tl.store(out_ptrs, acc.to(out_ptr.dtype.element_ty),
             mask=p_mask[:, None] & c_mask[None, :])


@triton.jit
def _final_loop_hfirst_kernel(o_ptr, g_ptr, w_ptr, lnw_ptr, lnb_ptr, out_ptr,
                              ns, H, c,
                              HCH: tl.constexpr,
                              BM: tl.constexpr, BC: tl.constexpr,
                              EPS: tl.constexpr):
    b = tl.program_id(0)
    pid_p = tl.program_id(1)
    pid_c = tl.program_id(2)

    p_idx = pid_p * BM + tl.arange(0, BM)
    c_idx = pid_c * BC + tl.arange(0, BC)
    h_base = tl.arange(0, HCH)

    p_mask = p_idx < ns
    c_mask = c_idx < c

    p64 = p_idx.to(tl.int64)
    c64 = c_idx.to(tl.int64)

    sum1 = tl.zeros((BM,), dtype=tl.float32)
    sum2 = tl.zeros((BM,), dtype=tl.float32)

    num_h = tl.cdiv(H, HCH)

    # First pass: statistics over H.
    for hi in range(0, num_h):
        h0 = hi * HCH
        h_off = h0 + h_base
        h_mask = h_off < H

        base = ((b * H + h_off[:, None]).to(tl.int64) * ns) + p64[None, :]
        mask2d = h_mask[:, None] & p_mask[None, :]

        o = tl.load(o_ptr + base, mask=mask2d, other=0.0).to(tl.float32)
        sum1 += tl.sum(o, axis=0)
        sum2 += tl.sum(o * o, axis=0)

    mean = sum1 / H
    var = sum2 / H - mean * mean
    var = tl.where(var > 0.0, var, 0.0)
    rstd = 1.0 / tl.sqrt(var + EPS)

    acc = tl.zeros((BM, BC), dtype=tl.float32)

    # Second pass: normalize, gate, and accumulate the final projection.
    for hi in range(0, num_h):
        h0 = hi * HCH
        h_off = h0 + h_base
        h_mask = h_off < H

        base = ((b * H + h_off[:, None]).to(tl.int64) * ns) + p64[None, :]
        mask2d = h_mask[:, None] & p_mask[None, :]

        o = tl.load(o_ptr + base, mask=mask2d, other=0.0).to(tl.float32)
        g = tl.load(g_ptr + base, mask=mask2d, other=0.0).to(tl.float32)

        diff = o - mean[None, :]

        lnw = tl.load(lnw_ptr + h_off, mask=h_mask, other=0.0).to(tl.float32)
        lnb = tl.load(lnb_ptr + h_off, mask=h_mask, other=0.0).to(tl.float32)

        a = diff * lnw[:, None] * rstd[None, :] + lnb[:, None]
        a = (a * g).to(tl.bfloat16)
        a = tl.trans(a)

        w_ptrs = w_ptr + h_off[:, None].to(tl.int64) * c + c64[None, :]
        w = tl.load(w_ptrs, mask=h_mask[:, None] & c_mask[None, :], other=0.0)

        acc = tl.dot(a, w, acc)

    out_ptrs = out_ptr + (b * ns + p64[:, None]).to(tl.int64) * c + c64[None, :]
    tl.store(out_ptrs, acc.to(out_ptr.dtype.element_ty),
             mask=p_mask[:, None] & c_mask[None, :])


def _is_power_of_two(x: int) -> bool:
    return x > 0 and (x & (x - 1)) == 0


def custom_kernel(data):
    """Run the TriMul forward pass and return a float32 tensor."""
    with torch.inference_mode():
        if len(data) == 4:
            x, mask, weights, config = data
        elif len(data) == 3:
            x, weights, config = data
            mask = None
        else:
            raise ValueError("Expected data tuple of length 3 (x, weights, config) "
                             "or 4 (x, mask, weights, config).")

        x = x.contiguous()
        B, N, _, C = x.shape
        ns = N * N
        M = B * ns

        if "hidden_dim" in config:
            H = int(config["hidden_dim"])
        else:
            H = int(weights["left_proj.weight"].shape[0])

        device = x.device
        dtype_lp = torch.bfloat16

        # ------------------------------------------------------------------
        # 1) Input LayerNorm -> bf16.
        # ------------------------------------------------------------------
        norm_w = weights["norm.weight"].to(device, dtype=torch.float32).contiguous()
        norm_b = weights["norm.bias"].to(device, dtype=torch.float32).contiguous()
        x_flat = x.view(M, C)
        x_norm = torch.empty((M, C), dtype=dtype_lp, device=device)

        eps = 1e-5

        if _is_power_of_two(C) and C <= 512:
            if C <= 128:
                BM_LN = 128
            elif C <= 256:
                BM_LN = 64
            else:
                BM_LN = 32
            grid_ln = (triton.cdiv(M, BM_LN),)
            _ln_one_kernel[grid_ln](
                x_flat, norm_w, norm_b, x_norm, M,
                EPS=eps, C_CON=C, BM=BM_LN,
                num_warps=4, num_stages=2,
            )
        else:
            BC_LN = min(triton.next_power_of_two(max(C, 16)), 128)
            BM_LN = 64
            grid_ln = (triton.cdiv(M, BM_LN),)
            _ln_two_kernel[grid_ln](
                x_flat, norm_w, norm_b, x_norm, M, C,
                EPS=eps, BM=BM_LN, BC=BC_LN,
                num_warps=4, num_stages=2,
            )

        # ------------------------------------------------------------------
        # 2) All linear projections in one cuBLAS call.
        # ------------------------------------------------------------------
        left_proj_w = weights["left_proj.weight"].to(device, dtype=dtype_lp).contiguous()
        right_proj_w = weights["right_proj.weight"].to(device, dtype=dtype_lp).contiguous()
        left_gate_w = weights["left_gate.weight"].to(device, dtype=dtype_lp).contiguous()
        right_gate_w = weights["right_gate.weight"].to(device, dtype=dtype_lp).contiguous()
        out_gate_w = weights["out_gate.weight"].to(device, dtype=dtype_lp).contiguous()

        W_all = torch.cat((left_proj_w, right_proj_w,
                           left_gate_w, right_gate_w, out_gate_w), dim=0)
        proj = F.linear(x_norm, W_all)

        del x_norm, W_all

        # ------------------------------------------------------------------
        # 3) Fused sigmoid/mask/write to H-first layout:
        #    L/R/OG: [B, H, N, N]
        # ------------------------------------------------------------------
        L = torch.empty((B, H, ns), dtype=dtype_lp, device=device)
        R = torch.empty((B, H, ns), dtype=dtype_lp, device=device)
        OG = torch.empty((B, H, ns), dtype=dtype_lp, device=device)

        has_mask = mask is not None
        if has_mask:
            mask_flat = mask.contiguous().view(-1).to(device, dtype=torch.float32)
            mask_arg = mask_flat
        else:
            mask_arg = L

        BM_PREP = 128
        BH_PREP = min(triton.next_power_of_two(max(H, 16)), 128)
        stride_m = 5 * H

        grid_prep = (B, triton.cdiv(ns, BM_PREP), triton.cdiv(H, BH_PREP))

        # Left.
        _prep_hfirst_kernel[grid_prep](
            proj, L, mask_arg, ns, H, stride_m,
            seg_start=0, gate_start=2 * H,
            APPLY_GATE=True, HAS_MASK=has_mask,
            BM=BM_PREP, BH=BH_PREP,
            num_warps=4, num_stages=2,
        )

        # Right.
        _prep_hfirst_kernel[grid_prep](
            proj, R, mask_arg, ns, H, stride_m,
            seg_start=H, gate_start=3 * H,
            APPLY_GATE=True, HAS_MASK=has_mask,
            BM=BM_PREP, BH=BH_PREP,
            num_warps=4, num_stages=2,
        )

        # Output gate (no mask).
        _prep_hfirst_kernel[grid_prep](
            proj, OG, mask_arg, ns, H, stride_m,
            seg_start=4 * H, gate_start=0,
            APPLY_GATE=False, HAS_MASK=False,
            BM=BM_PREP, BH=BH_PREP,
            num_warps=4, num_stages=2,
        )

        del proj

        # ------------------------------------------------------------------
        # 4) Core N^3 operation:
        #    O[b,h,i,j] = sum_k L[b,h,i,k] * R[b,h,j,k]
        # ------------------------------------------------------------------
        O = torch.empty((B, H, ns), dtype=dtype_lp, device=device)

        if N <= 512:
            BI = BJ = 64
        else:
            BI = BJ = 128
        BK = 64

        grid_core = (triton.cdiv(N, BI), triton.cdiv(N, BJ), B * H)
        core_warps = 8 if BI * BJ >= 16384 else 4

        _core_hfirst_kernel[grid_core](
            L, R, O, N, H,
            BI=BI, BJ=BJ, BK=BK,
            num_warps=core_warps, num_stages=3,
        )

        del L, R

        # ------------------------------------------------------------------
        # 5) Final LayerNorm + gate + projection, fused.
        # ------------------------------------------------------------------
        w_out = weights["to_out.weight"].to(device, dtype=dtype_lp)
        w_out_t = w_out.t().contiguous()

        to_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float32).contiguous()
        to_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float32).contiguous()

        out = torch.empty((M, C), dtype=torch.float32, device=device)

        if C <= 16:
            BC_FIN = 16
        elif C <= 128:
            BC_FIN = triton.next_power_of_two(C)
        else:
            BC_FIN = 256

        BM_FIN = 64 if BC_FIN <= 128 else 32

        if H == 128:
            grid_fin = (B, triton.cdiv(ns, BM_FIN), triton.cdiv(C, BC_FIN))
            _final_full_hfirst_kernel[grid_fin](
                O, OG, w_out_t, to_norm_w, to_norm_b, out,
                ns, C,
                H_CON=H, BM=BM_FIN, BC=BC_FIN, EPS=eps,
                num_warps=4, num_stages=3,
            )
        else:
            HCH = min(triton.next_power_of_two(max(H, 16)), 128)

            # Keep register pressure sane for longer channels.
            if HCH > 128:
                BM_FIN = 32

            grid_fin = (B, triton.cdiv(ns, BM_FIN), triton.cdiv(C, BC_FIN))
            _final_loop_hfirst_kernel[grid_fin](
                O, OG, w_out_t, to_norm_w, to_norm_b, out,
                ns, C, C,
                HCH=HCH, BM=BM_FIN, BC=BC_FIN, EPS=eps,
                num_warps=4, num_stages=3,
            )

        return out.view(B, N, N, C)
```
