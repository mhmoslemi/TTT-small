def _tri_core_kernel(

    N, H, B,

    GROUP_M: tl.constexpr,

):

    h = tl.program_id(1)
    b = tl.program_id(2)
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    group_size = GROUP_M * num_pid_n
    group_id = pid // group_size
    first_pid_m = group_id * GROUP_M
    group_size_m = tl.minimum(num_pid_m - first_pid_m, GROUP_M)
    pid_m = first_pid_m + (pid % group_size_m)

def _layernorm_row_kernel(x_ptr, xn_ptr, w_ptr, b_ptr, M, C, stride_x, stride_xn, BLOCK_M: tl.constexpr, C_BLOCK: tl.constexpr, EPS):
  pid=tl.program_id(0)
  offs_m=pid*BLOCK_M+...
  mask_m=offs_m<M
  offs_c=tl.arange(0,C_BLOCK)
  mask_c=offs_c<C
  mask=mask_m[:,None]&mask_c[None,:]
  x=tl.load(x_ptr + offs_m[:,None]*stride_x + offs_c[None,:], mask=mask, other=0.0)
  mean=tl.sum(x, axis=1)/C
  diff = tl.where(mask_c[None,:], x - mean[:,None], 0.0)
  var=tl.sum(diff*diff, axis=1)/C
  rstd=tl.math.rsqrt(var+EPS)
  w=tl.load(w_ptr+offs_c, mask=mask_c, other=0.0)

Kernel code:
```
@triton.jit
def _proj_gate_kernel(p_ptr, l_ptr, r_ptr, og_ptr, mask_ptr, M, H, PH,
                      HAS_MASK: tl.constexpr, BLOCK_M: tl.constexpr, H_CHUNK: tl.constexpr):
    pid = tl.program_id(0)
    hpid = tl.program_id(1)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M
    offs_h = hpid * H_CHUNK + tl.arange(0, H_CHUNK)
    mask_h = offs_h < H
    mask2 = mask_m[:, None] & mask_h[None, :]
    row_off = offs_m[:, None] * PH
    h_off = offs_h[None, :]
    left = tl.load(p_ptr + row_off + h_off, mask=mask2, other=0.0)
    lg = tl.load(p_ptr + row_off + H + h_off, mask=mask2, other=0.0)
    right = tl.load(p_ptr + row_off + 2*H + h_off, mask=mask2, other=0.0)
    rg = tl.load(p_ptr + row_off + 3*H + h_off, mask=mask2, other=0.0)
    og = tl.load(p_ptr + row_off + 4*H + h_off, mask=mask2, other=0.0)
    lxf = left.to(tl.float32)
    rgf = right.to(tl.float32)
    ogf = og.to(tl.float32)
    lgs = 1.0 / (1.0 + tl.exp(-lg.to(tl.float32)))
    rgs = 1.0 / (1.0 + tl.exp(-rg.to(tl.float32)))
    ogs = 1.0 / (1.0 + tl.exp(-ogf))
    if HAS_MASK:
        mv = tl.load(mask_ptr + offs_m, mask=mask_m, other=0.0).to(tl.float32)
        mv = mv[:, None]
        l_val = (lxf * lgs * mv).to(tl.bfloat16)
        r_val = (rgf * rgs * mv).to(tl.bfloat16)
    else:
        l_val = (lxf * lgs).to(tl.bfloat16)
        r_val = (rgf * rgs).to(tl.bfloat16)
    og_val = ogs.to(tl.bfloat16)
    out_off = offs_m[:, None] * H + offs_h[None, :]
    tl.store(l_ptr + out_off, l_val, mask=mask2)
    tl.store(r_ptr + out_off, r_val, mask=mask2)

import torch
import triton
import triton.language as tl


"""
TriMul forward (outgoing form) with fused Triton kernels.

Pipeline:
1) Triton row-wise LayerNorm -> normalized bf16 activations.
2) Batched/sequential bf16 projections for left/right projections, gates, and out_gate.
   - For small/medium hidden_dim, a single concatenated projection is used and a
     fused Triton kernel applies sigmoid/masks and writes L, R, out_gate.
3) Triton batched GEMM kernel computes:
       out[b, i, j, h] = sum_k L[b, i, k, h] * R[b, j, k, h]
   transposed storage is used so the inner loop is a standard tensor-core GEMM.
4) Fused Triton final kernel: LayerNorm over hidden_dim, out_gate multiplication,
   and the final linear projection, returning float32.
"""


def _pow2(n: int) -> int:
    return 1 << (max(1, int(n)) - 1).bit_length()


@triton.jit
def _row_layernorm_kernel(
    x_ptr,
    y_ptr,
    w_ptr,
    b_ptr,
    M,
    C,
    INV_C,
    EPS,
    BLOCK_M: tl.constexpr,
    C_BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_c = tl.arange(0, C_BLOCK)

    mask_m = offs_m < M
    mask_c = offs_c < C
    mask = mask_m[:, None] & mask_c[None, :]

    x = tl.load(x_ptr + offs_m[:, None] * C + offs_c[None, :], mask=mask, other=0.0)

    mean = tl.sum(x, axis=1) * INV_C
    diff = tl.where(mask_c[None, :], x - mean[:, None], 0.0)
    var = tl.sum(diff * diff, axis=1) * INV_C
    rstd = 1.0 / tl.sqrt(var + EPS)

    w = tl.load(w_ptr + offs_c, mask=mask_c, other=0.0)
    b = tl.load(b_ptr + offs_c, mask=mask_c, other=0.0)

    y = diff * rstd[:, None] * w[None, :] + b[None, :]
    tl.store(y_ptr + offs_m[:, None] * C + offs_c[None, :], y.to(tl.bfloat16), mask=mask)


@triton.jit
def _proj_gate_kernel(
    p_ptr,
    l_ptr,
    r_ptr,
    og_ptr,
    mask_ptr,
    M,
    H,
    PH,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
    H_CHUNK: tl.constexpr,
):
    pid = tl.program_id(0)
    hpid = tl.program_id(1)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = hpid * H_CHUNK + tl.arange(0, H_CHUNK)

    mask_m = offs_m < M
    mask_h = offs_h < H
    valid = mask_m[:, None] & mask_h[None, :]

    row_off = offs_m[:, None] * PH
    h_off = offs_h[None, :]

    left = tl.load(p_ptr + row_off + h_off, mask=valid, other=0.0)
    lg = tl.load(p_ptr + row_off + H + h_off, mask=valid, other=0.0)
    right = tl.load(p_ptr + row_off + 2 * H + h_off, mask=valid, other=0.0)
    rg = tl.load(p_ptr + row_off + 3 * H + h_off, mask=valid, other=0.0)
    og = tl.load(p_ptr + row_off + 4 * H + h_off, mask=valid, other=0.0)

    left_f = left.to(tl.float32)
    right_f = right.to(tl.float32)
    og_f = og.to(tl.float32)

    lg_s = 1.0 / (1.0 + tl.exp(-lg.to(tl.float32)))
    rg_s = 1.0 / (1.0 + tl.exp(-rg.to(tl.float32)))
    og_s = 1.0 / (1.0 + tl.exp(-og_f))

    if HAS_MASK:
        m = tl.load(mask_ptr + offs_m, mask=mask_m, other=0.0).to(tl.float32)[:, None]
        l_val = (left_f * lg_s * m).to(tl.bfloat16)
        r_val = (right_f * rg_s * m).to(tl.bfloat16)
    else:
        l_val = (left_f * lg_s).to(tl.bfloat16)
        r_val = (right_f * rg_s).to(tl.bfloat16)

    og_val = og_s.to(tl.bfloat16)

    out_off = offs_m[:, None] * H + offs_h[None, :]
    tl.store(l_ptr + out_off, l_val, mask=valid)
    tl.store(r_ptr + out_off, r_val, mask=valid)
    tl.store(og_ptr + out_off, og_val, mask=valid)


@triton.jit
def _tri_core_kernel(
    a_ptr,
    bb_ptr,
    c_ptr,
    N,
    sa_b,
    sa_h,
    sa_i,
    sa_k,
    sb_b,
    sb_h,
    sb_k,
    sb_j,
    sc_b,
    sc_h,
    sc_i,
    sc_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr,
    EVEN: tl.constexpr,
):
    pid = tl.program_id(0)
    h = tl.program_id(1)
    bid = tl.program_id(2)

    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    group_size = GROUP_M * num_pid_n
    group_id = pid // group_size
    first_pid_m = group_id * GROUP_M
    group_size_m = tl.minimum(num_pid_m - first_pid_m, GROUP_M)

    pid_m = first_pid_m + (pid % group_size_m)
    pid_n = (pid % group_size) // group_size_m

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    a_ptrs = a_ptr + bid * sa_b + h * sa_h + offs_m[:, None] * sa_i + offs_k[None, :] * sa_k
    b_ptrs = bb_ptr + bid * sb_b + h * sb_h + offs_k[:, None] * sb_k + offs_n[None, :] * sb_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    k_iters = tl.cdiv(N, BLOCK_K)

    if EVEN:
        for _ in range(0, k_iters):
            a = tl.load(a_ptrs)
            b = tl.load(b_ptrs)
            acc = tl.dot(a, b, acc)
            a_ptrs += BLOCK_K * sa_k
            b_ptrs += BLOCK_K * sb_k
    else:
        a_mask = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        b_mask = (offs_k[:, None] < N) & (offs_n[None, :] < N)
        for _ in range(0, k_iters):
            a = tl.load(a_ptrs, mask=a_mask, other=0.0)
            b = tl.load(b_ptrs, mask=b_mask, other=0.0)
            acc = tl.dot(a, b, acc)
            a_ptrs += BLOCK_K * sa_k
            b_ptrs += BLOCK_K * sb_k

    c_ptrs = c_ptr + bid * sc_b + h * sc_h + offs_m[:, None] * sc_i + offs_n[None, :] * sc_j
    if EVEN:
        tl.store(c_ptrs, acc)
    else:
        c_mask = (offs_m[:, None] < N) & (offs_n[None, :] < N)
        tl.store(c_ptrs, acc, mask=c_mask)


@triton.jit
def _final_layernorm_gate_linear_kernel(
    x_ptr,
    og_ptr,
    y_ptr,
    M,
    H,
    C,
    INV_H,
    EPS,
    w_ptr,
    lw_ptr,
    lb_ptr,
    BLOCK_M: tl.constexpr,
    H_BLOCK: tl.constexpr,
    C_BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = tl.arange(0, H_BLOCK)
    offs_c = tl.arange(0, C_BLOCK)

    mask_m = offs_m < M
    mask_h = offs_h < H
    mask_c = offs_c < C

    x_mask = mask_m[:, None] & mask_h[None, :]
    x = tl.load(x_ptr + offs_m[:, None] * H + offs_h[None, :], mask=x_mask, other=0.0)

    mean = tl.sum(x, axis=1) * INV_H
    diff = tl.where(mask_h[None, :], x - mean[:, None], 0.0)
    var = tl.sum(diff * diff, axis=1) * INV_H
    rstd = 1.0 / tl.sqrt(var + EPS)

    lw = tl.load(lw_ptr + offs_h, mask=mask_h, other=0.0)
    lb = tl.load(lb_ptr + offs_h, mask=mask_h, other=0.0)

    xn = diff * rstd[:, None] * lw[None, :] + lb[None, :]

    og = tl.load(og_ptr + offs_m[:, None] * H + offs_h[None, :], mask=x_mask, other=0.0)
    g = xn * og.to(tl.float32)

    w_mask = mask_h[:, None] & mask_c[None, :]
    w = tl.load(w_ptr + offs_h[:, None] * C + offs_c[None, :], mask=w_mask, other=0.0)

    acc = tl.dot(g.to(tl.bfloat16), w.to(tl.bfloat16))

    y_mask = mask_m[:, None] & mask_c[None, :]
    tl.store(y_ptr + offs_m[:, None] * C + offs_c[None, :], acc, mask=y_mask)

        W5[:, 3 * H:4 * H] = Wrg
        W5[:, 4 * H:5 * H] = Wo

        proj = (xn2d @ W5).view(B, N, N, 5 * H)

        left = torch.empty((B, N, N, H), dtype=bf16, device=device)
        right = torch.empty((B, N, N, H), dtype=bf16, device=device)
        og = torch.empty((B, N, N, H), dtype=bf16, device=device)

        H_CHUNK = max(16, min(128, _pow2(H)))
        num_h_chunks = triton.cdiv(H, H_CHUNK)
        if H_CHUNK <= 64:
            PG_M = 64
        elif H_CHUNK <= 128:
            PG_M = 32
        else:
            PG_M = 16

        if has_mask:
            pg_mask = mask_flat
        else:
            pg_mask = left  # never read when HAS_MASK is False

        _proj_gate_kernel[(triton.cdiv(M, PG_M), num_h_chunks)](
            proj, left, right, og, pg_mask,
            M, H, 5 * H,
            has_mask, PG_M, H_CHUNK,
            num_warps=4,
            num_stages=1,
        )

        del proj, W5
    else:
        # Sequential path avoids a huge 5*H intermediate tensor.
        left = (xn2d @ Wl).view(B, N, N, H)
        lg = xn2d @ Wlg
        torch.sigmoid_(lg)
        if has_mask:
            left.mul_(mask_4)
        left.mul_(lg.view(B, N, N, H))
        del lg

        right = (xn2d @ Wr).view(B, N, N, H)
        rg = xn2d @ Wrg
        torch.sigmoid_(rg)
        if has_mask:
            right.mul_(mask_4)
        right.mul_(rg.view(B, N, N, H))
        del rg

        og = (xn2d @ Wo).view(B, N, N, H)
        torch.sigmoid_(og)

    del xn, xn2d, Wl, Wlg, Wr, Wrg, Wo

    # Store in GEMM-friendly layouts:
    #   a[b, h, i, k] = left[b, i, k, h]
    #   bb[b, h, k, j] = right[b, j, k, h]
    a = left.permute(0, 3, 1, 2).contiguous()
    bmat = right.permute(0, 3, 2, 1).contiguous()
    del left, right

    # ---- 3) Core TriMul: batched GEMM over h ----
    core_out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    if N <= 128:
        BM = BN = 64
        BK = 64
        nwarps = 4
        nstages = 3
    elif N <= 256:
        BM = BN = 128
        BK = 64
        nwarps = 8
        nstages = 3
    else:
        BM = BN = 128
        BK = 64
        nwarps = 8
        nstages = 4

    num_m = triton.cdiv(N, BM)
    num_n = triton.cdiv(N, BN)
    EVEN = (N % BM == 0) and (N % BN == 0) and (N % BK == 0)

    _tri_core_kernel[(num_m * num_n, H, B)](
        a, bmat, core_out,
        N,
        a.stride(0), a.stride(1), a.stride(2), a.stride(3),
        bmat.stride(0), bmat.stride(1), bmat.stride(2), bmat.stride(3),
        core_out.stride(0), core_out.stride(3), core_out.stride(1), core_out.stride(2),
        BM, BN, BK, 8, EVEN,
        num_warps=nwarps,
        num_stages=nstages,
    )

    del a, bmat

    # ---- 4) Final LayerNorm + gate + linear, fused in Triton ----
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float32).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float32).contiguous()
    Wout_t = weights["to_out.weight"].t().to(device=device, dtype=bf16).contiguous()

    out = torch.empty((B, N, N, C), dtype=torch.float32, device=device)

    H_BLOCK = _pow2(H)
    C_BLOCK = _pow2(C)
    H_BLOCK = max(16, H_BLOCK)
    C_BLOCK = max(16, C_BLOCK)

    max_bm = 16384 // max(1, H_BLOCK + C_BLOCK)
    FM = 16
    while (FM * 2 <= max_bm) and (FM * 2 <= 128):
        FM *= 2
    FM = max(16, FM)

    _final_layernorm_gate_linear_kernel[(triton.cdiv(M, FM),)](
        core_out, og, out,
        M, H, C, 1.0 / float(H), EPS,
        Wout_t, to_out_norm_w, to_out_norm_b,
        FM, H_BLOCK, C_BLOCK,
        num_warps=4,
        num_stages=1,
    )

    del core_out, og, Wout_t

    return out
```
