```python
"""
Hybrid Triton/PyTorch forward pass for TriMul.

The operator is:
    x -> LayerNorm
    left/right/gates -> linear projections
    elementwise mask + sigmoid gating
    out[b,i,j,:] = sum_k left[b,i,k,:] * right[b,j,k,:]
    out -> LayerNorm -> out_gate gating -> linear projection

Triton is used for:
  - fused LayerNorm (and fused LayerNorm + first GEMM when dim is power-of-two and small)
  - fused mask/gate materialization
  - final LayerNorm + gating + projection (for hidden_dim <= 128), or
    fallback final LayerNorm + generic GEMM for larger hidden dims.

The expensive H-batched N x N matmuls are executed by cuBLAS via torch.bmm on
transposed contiguous layouts.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _layer_norm_kernel(
    x_ptr,
    y_ptr,
    w_ptr,
    b_ptr,
    M,
    C,
    eps,
    BM: tl.constexpr,
    BC: tl.constexpr,
    IS_FP16: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = (pid * BM + tl.arange(0, BM)).to(tl.int64)
    offs_c = tl.arange(0, BC)

    mask_m = offs_m < M
    mask_c = offs_c < C
    mask2d = mask_m[:, None] & mask_c[None, :]

    x = tl.load(
        x_ptr + offs_m[:, None] * C + offs_c[None, :],
        mask=mask2d,
        other=0.0,
    ).to(tl.float32)

    mean = tl.sum(x, axis=1) / C
    dev = x - mean[:, None]
    var = tl.sum(tl.where(mask2d, dev * dev, 0.0), axis=1) / C
    inv_std = 1.0 / tl.sqrt(tl.maximum(var, 0.0) + eps)

    w = tl.load(w_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
    bias = tl.load(b_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)

    y = dev * inv_std[:, None] * w[None, :] + bias[None, :]

    if IS_FP16:
        y = y.to(tl.float16)
    else:
        y = y.to(tl.bfloat16)

    tl.store(y_ptr + offs_m[:, None] * C + offs_c[None, :], y, mask=mask2d)


@triton.jit
def _ln_proj_kernel(
    x_ptr,
    y_ptr,
    w_ptr,
    b_ptr,
    wproj_ptr,
    M,
    NOUT,
    eps,
    C: tl.constexpr,
    BM: tl.constexpr,
    BN: tl.constexpr,
    IS_FP16: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = (pid * BM + tl.arange(0, BM)).to(tl.int64)
    mask_m = offs_m < M

    offs_c = tl.arange(0, C)
    x = tl.load(
        x_ptr + offs_m[:, None] * C + offs_c[None, :],
        mask=mask_m[:, None],
        other=0.0,
    ).to(tl.float32)

    mean = tl.sum(x, axis=1) / C
    dev = x - mean[:, None]
    var = tl.sum(dev * dev, axis=1) / C
    inv_std = 1.0 / tl.sqrt(tl.maximum(var, 0.0) + eps)

    w = tl.load(w_ptr + offs_c).to(tl.float32)
    bias = tl.load(b_ptr + offs_c).to(tl.float32)

    a = dev * inv_std[:, None] * w[None, :] + bias[None, :]
    if IS_FP16:
        a_dot = a.to(tl.float16)
    else:
        a_dot = a.to(tl.bfloat16)

    for n in range(0, NOUT, BN):
        offs_n = n + tl.arange(0, BN)
        mask_n = offs_n < NOUT

        b = tl.load(
            wproj_ptr + offs_c[:, None] * NOUT + offs_n[None, :],
            mask=mask_n[None, :],
            other=0.0,
        )

        acc = tl.dot(a_dot, b, tl.zeros((BM, BN), dtype=tl.float32))

        if IS_FP16:
            y = acc.to(tl.float16)
        else:
            y = acc.to(tl.bfloat16)

        tl.store(
            y_ptr + offs_m[:, None] * NOUT + offs_n[None, :],
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


@triton.jit
def _fuse_gate_kernel(
    proj_ptr,
    mask_ptr,
    l_ptr,
    r_ptr,
    og_ptr,
    M,
    H,
    HAS_MASK: tl.constexpr,
    IS_FP16: tl.constexpr,
    TB: tl.constexpr,
    HB: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_t = (pid * TB + tl.arange(0, TB)).to(tl.int64)
    tmask = offs_t < M

    if HAS_MASK:
        m = tl.load(mask_ptr + offs_t, mask=tmask, other=0.0).to(tl.float32)
    else:
        m = tl.full((TB,), 1.0, tl.float32)

    h_offs = tl.arange(0, HB)

    for h_off in range(0, H, HB):
        h_mask = (h_off + h_offs) < H
        mask2d = tmask[:, None] & h_mask[None, :]

        p = proj_ptr + offs_t[:, None] * (5 * H) + h_off + h_offs[None, :]

        left = tl.load(p, mask=mask2d, other=0.0).to(tl.float32)
        right = tl.load(p + H, mask=mask2d, other=0.0).to(tl.float32)
        left_gate = tl.load(p + 2 * H, mask=mask2d, other=0.0).to(tl.float32)
        right_gate = tl.load(p + 3 * H, mask=mask2d, other=0.0).to(tl.float32)
        out_gate = tl.load(p + 4 * H, mask=mask2d, other=0.0).to(tl.float32)

        left_gate = 1.0 / (1.0 + tl.exp(-left_gate))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate))
        out_gate = 1.0 / (1.0 + tl.exp(-out_gate))

        left = left * m[:, None] * left_gate
        right = right * m[:, None] * right_gate

        if IS_FP16:
            left = left.to(tl.float16)
            right = right.to(tl.float16)
            out_gate = out_gate.to(tl.float16)
        else:
            left = left.to(tl.bfloat16)
            right = right.to(tl.bfloat16)
            out_gate = out_gate.to(tl.bfloat16)

        op = offs_t[:, None] * H + h_off + h_offs[None, :]
        tl.store(l_ptr + op, left, mask=mask2d)
        tl.store(r_ptr + op, right, mask=mask2d)
        tl.store(og_ptr + op, out_gate, mask=mask2d)


@triton.jit
def _final_fused_kernel(
    a_ptr,
    og_ptr,
    b_ptr,
    c_ptr,
    w_ptr,
    bias_ptr,
    M,
    H,
    C,
    eps,
    BM: tl.constexpr,
    BN: tl.constexpr,
    BH: tl.constexpr,
    IS_FP16: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = (pid * BM + tl.arange(0, BM)).to(tl.int64)
    mask_m = offs_m < M

    offs_h = tl.arange(0, BH)
    mask_h = offs_h < H
    mask_ah = mask_m[:, None] & mask_h[None, :]

    a = tl.load(
        a_ptr + offs_m[:, None] * H + offs_h[None, :],
        mask=mask_ah,
        other=0.0,
    ).to(tl.float32)

    og = tl.load(
        og_ptr + offs_m[:, None] * H + offs_h[None, :],
        mask=mask_ah,
        other=0.0,
    ).to(tl.float32)

    mean = tl.sum(a, axis=1) / H
    dev = a - mean[:, None]
    var = tl.sum(tl.where(mask_ah, dev * dev, 0.0), axis=1) / H
    inv_std = 1.0 / tl.sqrt(tl.maximum(var, 0.0) + eps)

    w = tl.load(w_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)
    bias = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)

    a_norm = dev * inv_std[:, None] * w[None, :] + bias[None, :]
    a_norm = a_norm * og

    if IS_FP16:
        a_dot = a_norm.to(tl.float16)
    else:
        a_dot = a_norm.to(tl.bfloat16)

    for n in range(0, C, BN):
        offs_n = n + tl.arange(0, BN)
        mask_n = offs_n < C

        b = tl.load(
            b_ptr + offs_h[:, None] * C + offs_n[None, :],
            mask=mask_h[:, None] & mask_n[None, :],
            other=0.0,
        )

        acc = tl.dot(a_dot, b, tl.zeros((BM, BN), dtype=tl.float32))

        tl.store(
            c_ptr + offs_m[:, None] * C + offs_n[None, :],
            acc,
            mask=mask_m[:, None] & mask_n[None, :],
        )


@triton.jit
def _final_ln_gate_kernel(
    a_ptr,
    og_ptr,
    y_ptr,
    w_ptr,
    bias_ptr,
    M,
    H,
    eps,
    BM: tl.constexpr,
    BH: tl.constexpr,
    IS_FP16: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = (pid * BM + tl.arange(0, BM)).to(tl.int64)
    offs_h = tl.arange(0, BH)

    mask_m = offs_m < M
    mask_h = offs_h < H
    mask2d = mask_m[:, None] & mask_h[None, :]

    a = tl.load(
        a_ptr + offs_m[:, None] * H + offs_h[None, :],
        mask=mask2d,
        other=0.0,
    ).to(tl.float32)

    og = tl.load(
        og_ptr + offs_m[:, None] * H + offs_h[None, :],
        mask=mask2d,
        other=0.0,
    ).to(tl.float32)

    mean = tl.sum(a, axis=1) / H
    dev = a - mean[:, None]
    var = tl.sum(tl.where(mask2d, dev * dev, 0.0), axis=1) / H
    inv_std = 1.0 / tl.sqrt(tl.maximum(var, 0.0) + eps)

    w = tl.load(w_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)
    bias = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)

    y = dev * inv_std[:, None] * w[None, :] + bias[None, :]
    y = y * og

    if IS_FP16:
        y = y.to(tl.float16)
    else:
        y = y.to(tl.bfloat16)

    tl.store(y_ptr + offs_m[:, None] * H + offs_h[None, :], y, mask=mask2d)


@triton.jit
def _final_mm_kernel(
    a_ptr,
    b_ptr,
    c_ptr,
    M,
    H,
    C,
    BM: tl.constexpr,
    BN: tl.constexpr,
    BK: tl.constexpr,
):
    pid = tl.program_id(0)
    num_n = (C + BN - 1) // BN
    pid_m = pid // num_n
    pid_n = pid - pid_m * num_n

    offs_m = (pid_m * BM + tl.arange(0, BM)).to(tl.int64)
    offs_n = pid_n * BN + tl.arange(0, BN)
    offs_k = tl.arange(0, BK)

    mask_m = offs_m < M
    mask_n = offs_n < C

    a_ptrs = a_ptr + offs_m[:, None] * H + offs_k[None, :]
    b_ptrs = b_ptr + offs_k[:, None] * C + offs_n[None, :]

    acc = tl.zeros((BM, BN), dtype=tl.float32)

    for k in range(0, H, BK):
        k_mask = offs_k < (H - k)

        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & k_mask[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=k_mask[:, None] & mask_n[None, :],
            other=0.0,
        )

        acc = tl.dot(a, b, acc)

        a_ptrs += BK
        b_ptrs += BK * C

    tl.store(
        c_ptr + offs_m[:, None] * C + offs_n[None, :],
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    if len(data) == 4:
        x, mask, weights, config = data
    else:
        x, weights, config = data
        mask = None

    if not x.is_contiguous():
        x = x.contiguous()

    B, S, _, C = x.shape
    C = int(config.get("dim", C))
    H = int(config.get("hidden_dim", weights["left_proj.weight"].shape[0]))
    M = B * S * S
    device = x.device
    eps = 1e-5

    dist = str(config.get("distribution", "normal")).lower()
    use_bf16 = dist.startswith("cauchy") or C > 128
    is_fp16 = not use_bf16
    dtype = torch.float16 if is_fp16 else torch.bfloat16

    norm_w = weights["norm.weight"].contiguous()
    norm_b = weights["norm.bias"].contiguous()

    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    w_cat = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )
    w_proj_t = w_cat.t().to(dtype).contiguous()  # [C, 5H]
    NOUT = 5 * H

    # First Stage: fused LayerNorm + first projection when cheap and blockable.
    if C <= 128 and C == triton.next_power_of_two(C):
        proj = torch.empty((M, NOUT), dtype=dtype, device=device)
        BM0 = 128
        BN0 = 128
        grid0 = (triton.cdiv(M, BM0),)
        _ln_proj_kernel[grid0](
            x,
            proj,
            norm_w,
            norm_b,
            w_proj_t,
            M,
            NOUT,
            eps,
            C=C,
            BM=BM0,
            BN=BN0,
            IS_FP16=is_fp16,
            num_warps=8,
            num_stages=2,
        )
    else:
        xn = torch.empty((M, C), dtype=dtype, device=device)
        BC = triton.next_power_of_two(C)
        BM0 = 128 if BC <= 128 else 32
        grid0 = (triton.cdiv(M, BM0),)
        _layer_norm_kernel[grid0](
            x,
            xn,
            norm_w,
            norm_b,
            M,
            C,
            eps,
            BM=BM0,
            BC=BC,
            IS_FP16=is_fp16,
            num_warps=4,
        )
        del x
        proj = torch.mm(xn, w_proj_t)
        del xn

    # Mask handling.
    if mask is None:
        has_mask = False
        mask_arg = proj  # Not loaded when HAS_MASK is False.
    else:
        mask_flat = mask.to(torch.float32).reshape(-1)
        has_mask = True
        mask_arg = mask_flat

    # Materialize gated left/right and final out-gate.
    L = torch.empty((M, H), dtype=dtype, device=device)
    R = torch.empty((M, H), dtype=dtype, device=device)
    OG = torch.empty((M, H), dtype=dtype, device=device)

    TB = 64
    HB = min(triton.next_power_of_two(H), 128)
    grid_gate = (triton.cdiv(M, TB),)
    _fuse_gate_kernel[grid_gate](
        proj,
        mask_arg,
        L,
        R,
        OG,
        M,
        H,
        HAS_MASK=has_mask,
        IS_FP16=is_fp16,
        TB=TB,
        HB=HB,
        num_warps=4,
    )

    del proj
    if mask is not None:
        del mask_arg

    # Central TriMul:
    # For every hidden channel h:
    #   out_h = L_h @ R_h^T
    # where L_h[i,k] = left[b,i,k,h] and R_h[j,k] = right[b,j,k,h].
    BT = B * H
    Lp = L.view(B, S, S, H).permute(0, 3, 1, 2).contiguous().reshape(BT, S, S)
    del L

    Rp_t = R.view(B, S, S, H).permute(0, 3, 2, 1).contiguous().reshape(BT, S, S)
    del R

    out_h = torch.bmm(Lp, Rp_t)
    del Lp, Rp_t

    out_tok = (
        out_h.reshape(B, H, S, S)
        .permute(0, 2, 3, 1)
        .reshape(M, H)
        .contiguous()
    )
    del out_h

    # Final output.
    out = torch.empty((B, S, S, C), dtype=torch.float32, device=device)
    to_out_norm_w = weights["to_out_norm.weight"].contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].contiguous()
    w_out_t = weights["to_out.weight"].t().to(dtype).contiguous()  # [H, C]

    if H <= 128:
        BMF = 128
        BNF = min(128, triton.next_power_of_two(C))
        BHF = triton.next_power_of_two(H)
        grid_final = (triton.cdiv(M, BMF),)
        _final_fused_kernel[grid_final](
            out_tok,
            OG,
            w_out_t,
            out,
            to_out_norm_w,
            to_out_norm_b,
            M,
            H,
            C,
            eps,
            BM=BMF,
            BN=BNF,
            BH=BHF,
            IS_FP16=is_fp16,
            num_warps=8,
            num_stages=2,
        )
    else:
        act = torch.empty((M, H), dtype=dtype, device=device)
        BMG = 32
        BHG = triton.next_power_of_two(H)
        grid_lg = (triton.cdiv(M, BMG),)
        _final_ln_gate_kernel[grid_lg](
            out_tok,
            OG,
            act,
            to_out_norm_w,
            to_out_norm_b,
            M,
            H,
            eps,
            BM=BMG,
            BH=BHG,
            IS_FP16=is_fp16,
            num_warps=4,
        )
        del out_tok, OG

        BMm = 128
        BNm = min(128, triton.next_power_of_two(C))
        BKm = max(16, min(triton.next_power_of_two(H), 128))
        grid_mm = (triton.cdiv(M, BMm) * triton.cdiv(C, BNm),)
        _final_mm_kernel[grid_mm](
            act,
            w_out_t,
            out,
            M,
            H,
            C,
            BM=BMm,
            BN=BNm,
            BK=BKm,
            num_warps=8,
            num_stages=3,
        )

    return out
```
