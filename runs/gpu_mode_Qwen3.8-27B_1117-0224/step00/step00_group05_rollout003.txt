```python
import torch
import triton
import triton.language as tl


"""
TriMul outgoing forward:
- Fused Triton kernel 1: LayerNorm + all five linear projections + sigmoid gates +
  pairwise mask -> left/right/out-gate tensors stored in channel-major [B,H,S,S]
  layout for the next stage.
- Triton batched GEMM kernel 2: for each b,h compute
    out[b,h,i,j] = sum_k left[b,h,i,k] * right[b,h,j,k]
  which is the TriMul operator.
- Fused Triton kernel 3: final LayerNorm over H, multiplication by out_gate,
  and projection back to [B,S,S,dim] using a tiled GEMM.
"""


@triton.jit
def _trmul_preprocess_kernel(
    x_ptr,
    mask_ptr,
    nw_ptr,
    nb_ptr,

    wl_ptr,
    wlg_ptr,
    wr_ptr,
    wrg_ptr,
    wg_ptr,

    l_ptr,
    r_ptr,
    g_ptr,

    C,
    MP,
    H: tl.constexpr,
    BLOCK_P: tl.constexpr,
    BLOCK_K: tl.constexpr,
    USE_MASK: tl.constexpr,
    EPS: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_p = tl.program_id(1)

    offs_p = pid_p * BLOCK_P + tl.arange(0, BLOCK_P)
    p_mask = offs_p < MP
    x_base = x_ptr + pid_b * (MP * C)
    offs_h = tl.arange(0, H)

    num_k = (C + BLOCK_K - 1) // BLOCK_K

    # LayerNorm statistics: mean and variance over the trailing feature dimension C.
    sum_x = tl.zeros((BLOCK_P,), dtype=tl.float32)
    sum_x2 = tl.zeros((BLOCK_P,), dtype=tl.float32)
    for kc in range(num_k):
        offs_k = kc * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = offs_k < C
        xk = tl.load(
            x_base + offs_p[:, None] * C + offs_k[None, :],
            mask=p_mask[:, None] & k_mask[None, :],
            other=0.0,
        ).to(tl.float32)
        sum_x += tl.sum(xk, axis=1)
        sum_x2 += tl.sum(xk * xk, axis=1)

    mean = sum_x / C
    var = sum_x2 / C - mean * mean
    var = tl.where(var > 0.0, var, 0.0)
    rstd = 1.0 / tl.sqrt(var + EPS)

    # left projection + left gate
    acc_l = tl.zeros((BLOCK_P, H), dtype=tl.float32)
    acc_lg = tl.zeros((BLOCK_P, H), dtype=tl.float32)
    for kc in range(num_k):
        offs_k = kc * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = offs_k < C

        xk = tl.load(
            x_base + offs_p[:, None] * C + offs_k[None, :],
            mask=p_mask[:, None] & k_mask[None, :],
            other=0.0,
        ).to(tl.float32)

        nw = tl.load(nw_ptr + offs_k, mask=k_mask, other=0.0).to(tl.float32)
        nb = tl.load(nb_ptr + offs_k, mask=k_mask, other=0.0).to(tl.float32)
        x16 = ((xk - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]).to(
            tl.bfloat16
        )

        w_mask = k_mask[:, None] & (offs_h[None, :] < H)
        w_off = offs_k[:, None] * H + offs_h[None, :]

        wl = tl.load(wl_ptr + w_off, mask=w_mask, other=0.0).to(tl.bfloat16)
        acc_l = tl.dot(x16, wl, acc_l)

        wlg = tl.load(wlg_ptr + w_off, mask=w_mask, other=0.0).to(tl.bfloat16)
        acc_lg = tl.dot(x16, wlg, acc_lg)

    if USE_MASK:
        mask_vals = tl.load(
            mask_ptr + pid_b * MP + offs_p, mask=p_mask, other=0.0
        ).to(tl.float32)

    sig_lg = 1.0 / (1.0 + tl.exp(-acc_lg))
    left = acc_l * sig_lg
    if USE_MASK:
        left = left * mask_vals[:, None]
    left16 = left.to(tl.bfloat16)

    l_ptrs = l_ptr + pid_b * (H * MP) + offs_h[:, None] * MP + offs_p[None, :]
    tl.store(l_ptrs, tl.trans(left16), mask=p_mask[None, :])

    # right projection + right gate
    acc_r = tl.zeros((BLOCK_P, H), dtype=tl.float32)
    acc_rg = tl.zeros((BLOCK_P, H), dtype=tl.float32)
    for kc in range(num_k):
        offs_k = kc * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = offs_k < C

        xk = tl.load(
            x_base + offs_p[:, None] * C + offs_k[None, :],
            mask=p_mask[:, None] & k_mask[None, :],
            other=0.0,
        ).to(tl.float32)

        nw = tl.load(nw_ptr + offs_k, mask=k_mask, other=0.0).to(tl.float32)
        nb = tl.load(nb_ptr + offs_k, mask=k_mask, other=0.0).to(tl.float32)
        x16 = ((xk - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]).to(
            tl.bfloat16
        )

        w_mask = k_mask[:, None] & (offs_h[None, :] < H)
        w_off = offs_k[:, None] * H + offs_h[None, :]

        wr = tl.load(wr_ptr + w_off, mask=w_mask, other=0.0).to(tl.bfloat16)
        acc_r = tl.dot(x16, wr, acc_r)

        wrg = tl.load(wrg_ptr + w_off, mask=w_mask, other=0.0).to(tl.bfloat16)
        acc_rg = tl.dot(x16, wrg, acc_rg)

    sig_rg = 1.0 / (1.0 + tl.exp(-acc_rg))
    right = acc_r * sig_rg
    if USE_MASK:
        right = right * mask_vals[:, None]
    right16 = right.to(tl.bfloat16)

    r_ptrs = r_ptr + pid_b * (H * MP) + offs_h[:, None] * MP + offs_p[None, :]
    tl.store(r_ptrs, tl.trans(right16), mask=p_mask[None, :])

    # out gate
    acc_og = tl.zeros((BLOCK_P, H), dtype=tl.float32)
    for kc in range(num_k):
        offs_k = kc * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = offs_k < C

        xk = tl.load(
            x_base + offs_p[:, None] * C + offs_k[None, :],
            mask=p_mask[:, None] & k_mask[None, :],
            other=0.0,
        ).to(tl.float32)

        nw = tl.load(nw_ptr + offs_k, mask=k_mask, other=0.0).to(tl.float32)
        nb = tl.load(nb_ptr + offs_k, mask=k_mask, other=0.0).to(tl.float32)
        x16 = ((xk - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]).to(
            tl.bfloat16
        )

        w_mask = k_mask[:, None] & (offs_h[None, :] < H)
        wg = tl.load(wg_ptr + w_off, mask=w_mask, other=0.0).to(tl.bfloat16)
        acc_og = tl.dot(x16, wg, acc_og)

    sig_og = 1.0 / (1.0 + tl.exp(-acc_og))
    gate16 = sig_og.to(tl.bfloat16)

    g_ptrs = g_ptr + pid_b * (H * MP) + offs_h[:, None] * MP + offs_p[None, :]
    tl.store(g_ptrs, tl.trans(gate16), mask=p_mask[None, :])


@triton.jit
def _trmul_mul_kernel(
    l_ptr,
    r_ptr,
    o_ptr,
    S,
    MP,
    BI: tl.constexpr,
    BJ: tl.constexpr,
    BK: tl.constexpr,
    EVEN: tl.constexpr,
):
    j = tl.program_id(0)
    i = tl.program_id(1)
    hb = tl.program_id(2)

    base = hb * MP

    offs_i = i * BI + tl.arange(0, BI)
    offs_j = j * BJ + tl.arange(0, BJ)
    offs_k = tl.arange(0, BK)

    # left: [I,K], right: [J,K]. We need left @ right^T.
    l_ptrs = l_ptr + base + offs_i[:, None] * S + offs_k[None, :]
    r_ptrs = r_ptr + base + offs_j[:, None] * S + offs_k[None, :]

    acc = tl.zeros((BI, BJ), dtype=tl.float32)

    for k0 in range(0, S, BK):
        if EVEN:
            a = tl.load(l_ptrs)
            b = tl.load(r_ptrs)
        else:
            k_valid = (k0 + offs_k) < S
            i_valid = offs_i < S
            j_valid = offs_j < S
            a = tl.load(l_ptrs, mask=i_valid[:, None] & k_valid[None, :], other=0.0)
            b = tl.load(r_ptrs, mask=j_valid[:, None] & k_valid[None, :], other=0.0)

        acc = tl.dot(a, tl.trans(b), acc)

        l_ptrs += BK
        r_ptrs += BK

    o_ptrs = o_ptr + base + offs_i[:, None] * S + offs_j[None, :]
    if EVEN:
        tl.store(o_ptrs, acc.to(tl.bfloat16))
    else:
        tl.store(
            o_ptrs,
            acc.to(tl.bfloat16),
            mask=(offs_i < S)[:, None] & (offs_j < S)[None, :],
        )


@triton.jit
def _trmul_finalize_kernel(
    o_ptr,
    gate_ptr,
    out_ptr,
    wn_ptr,
    wb_ptr,
    wt_ptr,
    C,
    MP,
    H: tl.constexpr,
    PM: tl.constexpr,
    CB: tl.constexpr,
    EVEN_C: tl.constexpr,
    EPS: tl.constexpr,
):
    pid_p = tl.program_id(0)
    b = tl.program_id(1)

    offs_p = pid_p * PM + tl.arange(0, PM)
    p_mask = offs_p < MP
    offs_h = tl.arange(0, H)

    h_off = b * H * MP + offs_h[:, None] * MP + offs_p[None, :]

    x = tl.load(o_ptr + h_off, mask=p_mask[None, :], other=0.0).to(tl.float32)
    gate = tl.load(gate_ptr + h_off, mask=p_mask[None, :], other=0.0).to(tl.float32)

    # LayerNorm over H; in [H, P] layout this is a reduction over axis 0.
    mean = tl.sum(x, axis=0) / H
    diff = x - mean[None, :]
    var = tl.sum(diff * diff, axis=0) / H
    var = tl.where(var > 0.0, var, 0.0)
    rstd = 1.0 / tl.sqrt(var + EPS)

    wn = tl.load(wn_ptr + offs_h).to(tl.float32)
    wb = tl.load(wb_ptr + offs_h).to(tl.float32)

    y = diff * rstd[None, :] * wn[:, None] + wb[:, None]
    a = tl.trans((y * gate).to(tl.bfloat16))  # [P, H]

    num_c = (C + CB - 1) // CB
    for cc in range(num_c):
        offs_c = cc * CB + tl.arange(0, CB)
        if EVEN_C:
            w = tl.load(wt_ptr + offs_h[:, None] * C + offs_c[None, :]).to(
                tl.bfloat16
            )
        else:
            c_mask = offs_c < C
            w_mask = (offs_h[:, None] < H) & c_mask[None, :]
            w = tl.load(
                wt_ptr + offs_h[:, None] * C + offs_c[None, :],
                mask=w_mask,
                other=0.0,
            ).to(tl.bfloat16)

        acc = tl.dot(a, w)  # [P, CB]

        out_off = out_ptr + (b * MP + offs_p)[:, None] * C + offs_c[None, :]
        if EVEN_C:
            tl.store(out_off, acc.to(tl.float32), mask=p_mask[:, None])
        else:
            c_mask = offs_c < C
            tl.store(
                out_off,
                acc.to(tl.float32),
                mask=p_mask[:, None] & c_mask[None, :],
            )


def custom_kernel(data):
    """
    Forward pass for outgoing TriMul.

    data = (
        input:  [B, S, S, dim] tensor,
        mask:   [B, S, S] tensor or None,
        weights: dict of weights,
        config: dict with dim/hidden_dim,
    )
    """
    x, mask, weights, config = data

    B, S, _, C = x.shape
    H = int(config.get("hidden_dim", 128))
    device = x.device
    MP = S * S
    eps = float(config.get("eps", 1e-5))

    # Flatten input to [B*S*S, C].
    x_flat = x.contiguous().reshape(B * MP, C)

    # Prepare weights in the layouts used by the kernels.
    bf16 = torch.bfloat16

    wl = weights["left_proj.weight"].to(bf16).t().contiguous()       # [C, H]
    wlg = weights["left_gate.weight"].to(bf16).t().contiguous()      # [C, H]
    wr = weights["right_proj.weight"].to(bf16).t().contiguous()      # [C, H]
    wrg = weights["right_gate.weight"].to(bf16).t().contiguous()     # [C, H]
    wg = weights["out_gate.weight"].to(bf16).t().contiguous()        # [C, H]

    nw = weights["norm.weight"].to(torch.float32).contiguous()       # [C]
    nb = weights["norm.bias"].to(torch.float32).contiguous()          # [C]

    wn = weights["to_out_norm.weight"].to(torch.float32).contiguous()# [H]
    wb = weights["to_out_norm.bias"].to(torch.float32).contiguous()   # [H]
    wt = weights["to_out.weight"].to(bf16).t().contiguous()           # [H, C]

    use_mask = (
        mask is not None
        and mask.numel() > 0
        and not config.get("nomask", False)
    )

    if use_mask:
        mask_flat = mask.contiguous().to(torch.float32).reshape(B, MP)
        mask_ptr = mask_flat
    else:
        # Dummy pointer; not dereferenced when USE_MASK=False.
        mask_flat = torch.empty(1, device=device, dtype=torch.float32)
        mask_ptr = mask_flat

    # Allocations for the intermediate channel-major [B,H,MP] tensors.
    left_t = torch.empty((B, H, MP), device=device, dtype=bf16)
    right_t = torch.empty((B, H, MP), device=device, dtype=bf16)
    gate_t = torch.empty((B, H, MP), device=device, dtype=bf16)

    # ---- Kernel 1: LayerNorm + projections + gating/masking + reorder ----------
    BLOCK_P = 64
    BLOCK_K = 128
    grid_pre = (B, triton.cdiv(MP, BLOCK_P))

    _trmul_preprocess_kernel[grid_pre](
        x_flat,
        mask_ptr,
        nw,
        nb,
        wl,
        wlg,
        wr,
        wrg,
        wg,
        left_t,
        right_t,
        gate_t,
        C,
        MP,
        H=H,
        BLOCK_P=BLOCK_P,
        BLOCK_K=BLOCK_K,
        USE_MASK=use_mask,
        EPS=eps,
        num_warps=8,
        num_stages=2,
    )

    # ---- Kernel 2: batched GEMM / TriMul core ---------------------------------
    # left_t[b,h,i,k], right_t[b,h,j,k]
    # out[b,h,i,j] = sum_k left_t[b,h,i,k] * right_t[b,h,j,k]
    out_mid = torch.empty((B, H, MP), device=device, dtype=bf16)

    BI = 128
    BJ = 128
    BK = 128
    even_core = (S % BI == 0) and (S % BJ == 0) and (S % BK == 0)

    num_bi = triton.cdiv(S, BI)
    num_bj = triton.cdiv(S, BJ)

    # x fastest = j, so the same left tile is reused by adjacent programs.
    grid_mul = (num_bj, num_bi, B * H)

    _trmul_mul_kernel[grid_mul](
        left_t,
        right_t,
        out_mid,
        S,
        MP,
        BI=BI,
        BJ=BJ,
        BK=BK,
        EVEN=even_core,
        num_warps=8,
        num_stages=3,
    )

    # ---- Kernel 3: final LayerNorm + gate + projection to output dim ----------
    out = torch.empty((B, S, S, C), device=device, dtype=torch.float32)

    PM = 64
    CB = 128
    even_c = (C % CB == 0)

    grid_final = (triton.cdiv(MP, PM), B)

    _trmul_finalize_kernel[grid_final](
        out_mid,
        gate_t,
        out,
        wn,
        wb,
        wt,
        C,
        MP,
        H=H,
        PM=PM,
        CB=CB,
        EVEN_C=even_c,
        EPS=eps,
        num_warps=8,
        num_stages=2,
    )

    return out
```
