@triton.jit
def _trimul_preprocess_kernel(
    x_ptr, mask_ptr, L_ptr, R_ptr, G_ptr,
    wl_ptr, wgl_ptr, wr_ptr, wgr_ptr, wog_ptr,
    norm_w_ptr, norm_b_ptr,
    B, N, C, H,
    BK: tl.constexpr, CB: tl.constexpr, HB: tl.constexpr, HAS_MASK: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_k = tl.program_id(1)
    b = tl.program_id(2)
    i = pid_i
    k = pid_k * BK + tl.arange(0, BK)
    c = tl.arange(0, CB)
    h = tl.arange(0, HB)
    row_mask = k < N
    c_mask = c < C
    # x load
    x_ptrs = x_ptr + ((b * N + i) * N + k)[:, None] * C + c[None, :]
    x = tl.load(x_ptrs, mask=row_mask[:, None] & c_mask[None, :], other=0.0).to(tl.float32)
    x_sum = tl.sum(x, axis=1)
    x_sumsq = tl.sum(x * x, axis=1)
    C_f = C.to(tl.float32)
    mean = x_sum / C_f

    nb = tl.load(norm_b_ptr + c, mask=c_mask, other=0.0).to(tl.float32)

@triton.jit
def _trimul_finalize_kernel(
  C_ptr, G_ptr, gamma_ptr, beta_ptr, W_ptr, out_ptr,
  B, N, H, D,
  BN: tl.constexpr, BH: tl.constexpr, DB: tl.constexpr,
):
  pid_i = tl.program_id(0)
  pid_j = tl.program_id(1)
  b = tl.program_id(2)
  i = pid_i
  j = pid_j * BN + tl.arange(0, BN)
  h = tl.arange(0, BH)
  d = tl.arange(0, DB)
  j_mask = j < N
  d_mask = d < D
  NN = N * N
  base_c = C_ptr + b * H * NN + i * N
  base_g = G_ptr + b * H * NN + i * N
  sumv = tl.zeros((BN,), dtype=tl.float32)
  sums = tl.zeros((BN,), dtype=tl.float32)
  for h0 in range(0, H, BH):
     h_off = h0 + h
     h_mask = h_off < H
     ptrs = base_c + h_off[:, None] * NN + j[None, :]
     c = tl.load(ptrs, mask=h_mask[:, None] & j_mask[None, :], other=0.0).to(tl.float32)
     sumv += tl.sum(c, axis=0)
     sums += tl.sum(c * c, axis=0)
  Hf = H.to(tl.float32)
  mean = sumv / Hf
  var = sums / Hf - mean * mean
  var = tl.maximum(var, 0.0)
  rstd = 1.0 / tl.sqrt(var + 1e-5)
  acc = tl.zeros((BN, DB), dtype=tl.float32)
  for h0 in range(0, H, BH):
     h_off = h0 + h
     h_mask = h_off < H
     h_full = h_mask & j_mask
     c = tl.load(base_c + h_off[:,None]*NN + j[None,:], mask=h_full, other=0.0).to(tl.float32)
     g = tl.load(base_g + h_off[:,None]*NN + j[None,:], mask=h_full, other=0.0).to(tl.float32)
     gamma = tl.load(gamma_ptr + h_off, mask=h_mask, other=0.0).to(tl.float32)
     beta = tl.load(beta_ptr + h_off, mask=h_mask, other=0.0).to(tl.float32)
     y = (c - mean[None, :]) * rstd[None, :] * gamma[:, None] + beta[:, None]
     y = y * g
     # dot
     y_t = tl.trans(y)  # [BN,BH]
     # w load: [DB,BH]
     w_ptrs = W_ptr + d[:, None] * H + h_off[None, :]
     w = tl.load(w_ptrs, mask=d_mask[:, None] & h_mask[None, :], other=0.0).to(tl.float32)
     w_t = tl.trans(w).to(tl.float32)  # [BH,DB]
     acc = tl.dot(y_t, w_t, acc, input_precision="tf32")
  out_ptrs = out_ptr + (b * N + i) * N * D + j[:, None] * D + d[None, :]
  tl.store(out_ptrs, acc, mask=j_mask[:, None] & d_mask[None, :])

import torch
import triton
import triton.language as tl

def _next_pow2(n):
    n = int(n)
    p = 1
    while p < n:
        p <<= 1
    return p

def _choose_hb(h):
    if h >= 64:
        return 64
    if h >= 16:
        return 32
    return 16

@triton.jit
def _sigmoid_clamp(x):
    x = tl.minimum(x, 50.0)
    x = tl.maximum(x, -50.0)
    return 1.0 / (1.0 + tl.exp(-x))

@triton.jit
def _preprocess_kernel(
    x_ptr, mask_ptr, L_ptr, R_ptr,
    wl_ptr, wgl_ptr, wr_ptr, wgr_ptr,
    norm_w_ptr, norm_b_ptr,
    B, N, C, H,
    BK: tl.constexpr,
    CB: tl.constexpr,
    HB: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_k = tl.program_id(1)
    b = tl.program_id(2)

    i = pid_i
    k = pid_k * BK + tl.arange(0, BK)
    c = tl.arange(0, CB)
    h = tl.arange(0, HB)

    row_mask = k < N
    c_mask = c < C
    x_mask = row_mask[:, None] & c_mask[None, :]

    x_ptrs = x_ptr + ((b * N + i) * N + k)[:, None] * C + c[None, :]
    x = tl.load(x_ptrs, mask=x_mask, other=0.0).to(tl.float32)

    x_sum = tl.sum(x, axis=1)
    x_sumsq = tl.sum(x * x, axis=1)
    mean = x_sum / C
    var = x_sumsq / C - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    nw = tl.load(norm_w_ptr + c, mask=c_mask, other=0.0).to(tl.float32)
    nb = tl.load(norm_b_ptr + c, mask=c_mask, other=0.0).to(tl.float32)
    xn = (x - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]
    xn_bf = xn.to(tl.bfloat16)

    if HAS_MASK:
        mask_vals = tl.load(mask_ptr + (b * N + i) * N + k, mask=row_mask, other=1.0).to(tl.float32)
    else:
        mask_vals = tl.full((BK,), 1.0, dtype=tl.float32)

    NN = N * N
    for h0 in range(0, H, HB):
        h_off = h0 + h
        h_mask = h_off < H
        wmask = c_mask[:, None] & h_mask[None, :]

        wl = tl.load(wl_ptr + c[:, None] * H + h_off[None, :], mask=wmask, other=0.0).to(tl.bfloat16)
        acc = tl.zeros((BK, HB), dtype=tl.float32)
        pl = tl.dot(xn_bf, wl, acc)

        wgl = tl.load(wgl_ptr + c[:, None] * H + h_off[None, :], mask=wmask, other=0.0).to(tl.bfloat16)
        acc = tl.zeros((BK, HB), dtype=tl.float32)
        gll = tl.dot(xn_bf, wgl, acc)
        gll = _sigmoid_clamp(gll)

        val = pl * gll * mask_vals[:, None]
        ptrs = L_ptr + (b * H + h_off)[:, None] * NN + i * N + k[None, :]

```python
"""
Fused forward pass for TriMul outgoing update.

Pipeline:
1) Triton kernel computes LayerNorm + left/right projections + left/right gates,
   applies the pair mask, and writes bf16 L/R in [B, H, N, N] layout for fast
   batched GEMM.
2) torch.bmm computes C_h = L_h @ R_h^T for every hidden channel h.
3) Triton kernel recomputes the output gate from the original input, performs
   LayerNorm over the hidden dimension of C, multiplies by the output gate,
   and applies the final linear projection.

The expensive O(N^3) contraction is done by cuBLAS batched GEMM in bf16,
while the surrounding elementwise/gate/norm work is fused in Triton.
"""

import torch
import triton
import triton.language as tl


def _next_pow2(n: int) -> int:
    n = int(n)
    p = 1
    while p < n:
        p <<= 1
    return p


def _choose_hb(h: int) -> int:
    if h >= 64:
        return 64
    if h >= 16:
        return 32
    return 16


def _get_weight_float(weights, name, device):
    return weights[name].detach().to(device=device, dtype=torch.float32).contiguous()


def _get_weight_bf16_trans(weights, name, device):
    # Linear weight is [out, in]. We load it as [in, out] for tl.dot.
    t = weights[name].detach().to(device=device, dtype=torch.float32)
    return t.t().contiguous().to(dtype=torch.bfloat16).contiguous()


@triton.jit
def _sigmoid_clamp(x):
    x = tl.minimum(x, 50.0)
    x = tl.maximum(x, -50.0)
    return 1.0 / (1.0 + tl.exp(-x))


@triton.jit
def _trimul_preprocess_kernel(
    x_ptr,
    mask_ptr,
    L_ptr,
    R_ptr,
    wl_ptr,
    wgl_ptr,
    wr_ptr,
    wgr_ptr,
    norm_w_ptr,
    norm_b_ptr,
    B,
    N,
    C,
    H,
    BK: tl.constexpr,
    CB: tl.constexpr,
    HB: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_k = tl.program_id(1)
    b = tl.program_id(2)

    i = pid_i
    k = pid_k * BK + tl.arange(0, BK)
    c = tl.arange(0, CB)
    h = tl.arange(0, HB)

    row_mask = k < N
    c_mask = c < C
    x_mask = row_mask[:, None] & c_mask[None, :]

    x_ptrs = x_ptr + ((b * N + i) * N + k)[:, None] * C + c[None, :]
    x = tl.load(x_ptrs, mask=x_mask, other=0.0).to(tl.float32)

    x_sum = tl.sum(x, axis=1)
    x_sumsq = tl.sum(x * x, axis=1)
    mean = x_sum / C
    var = x_sumsq / C - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    nw = tl.load(norm_w_ptr + c, mask=c_mask, other=0.0).to(tl.float32)
    nb = tl.load(norm_b_ptr + c, mask=c_mask, other=0.0).to(tl.float32)

    xn = (x - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]
    xn_bf = xn.to(tl.bfloat16)

    if HAS_MASK:
        mask_vals = tl.load(
            mask_ptr + (b * N + i) * N + k,
            mask=row_mask,
            other=1.0,
        ).to(tl.float32)
    else:
        mask_vals = tl.full((BK,), 1.0, dtype=tl.float32)

    NN = N * N

    for h0 in range(0, H, HB):
        h_off = h0 + h
        h_mask = h_off < H
        wmask = c_mask[:, None] & h_mask[None, :]

        # left = left_proj(xn), gate = sigmoid(left_gate(xn))
        wl = tl.load(
            wl_ptr + c[:, None] * H + h_off[None, :],
            mask=wmask,
            other=0.0,
        ).to(tl.bfloat16)
        acc = tl.zeros((BK, HB), dtype=tl.float32)
        pl = tl.dot(xn_bf, wl, acc)

        wgl = tl.load(
            wgl_ptr + c[:, None] * H + h_off[None, :],
            mask=wmask,
            other=0.0,
        ).to(tl.bfloat16)
        acc = tl.zeros((BK, HB), dtype=tl.float32)
        gll = tl.dot(xn_bf, wgl, acc)
        gll = _sigmoid_clamp(gll)

        val = pl * gll * mask_vals[:, None]
        ptrs = L_ptr + (b * H + h_off)[:, None] * NN + i * N + k[None, :]
        tl.store(ptrs, tl.trans(val), mask=h_mask[:, None] & row_mask[None, :])

        # right = right_proj(xn), gate = sigmoid(right_gate(xn))
        wr = tl.load(
            wr_ptr + c[:, None] * H + h_off[None, :],
            mask=wmask,
            other=0.0,
        ).to(tl.bfloat16)
        acc = tl.zeros((BK, HB), dtype=tl.float32)
        pr = tl.dot(xn_bf, wr, acc)

        wgr = tl.load(
            wgr_ptr + c[:, None] * H + h_off[None, :],
            mask=wmask,
            other=0.0,
        ).to(tl.bfloat16)
        acc = tl.zeros((BK, HB), dtype=tl.float32)
        gr = tl.dot(xn_bf, wgr, acc)
        gr = _sigmoid_clamp(gr)

        val = pr * gr * mask_vals[:, None]
        ptrs = R_ptr + (b * H + h_off)[:, None] * NN + i * N + k[None, :]
        tl.store(ptrs, tl.trans(val), mask=h_mask[:, None] & row_mask[None, :])


@triton.jit
def _trimul_finalize_kernel(
    bmm_ptr,
    x_ptr,
    wog_ptr,
    norm_w_ptr,
    norm_b_ptr,
    gamma_ptr,
    beta_ptr,
    wout_ptr,
    out_ptr,
    B,
    N,
    C,
    H,
    D,
    BN: tl.constexpr,
    BH: tl.constexpr,
    CB: tl.constexpr,
    DB: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    b = tl.program_id(2)

    i = pid_i
    j = pid_j * BN + tl.arange(0, BN)
    h = tl.arange(0, BH)
    c = tl.arange(0, CB)
    d = tl.arange(0, DB)

    j_mask = j < N
    c_mask = c < C
    d_mask = d < D

    NN = N * N
    base = bmm_ptr + b * H * NN + i * N

    # First pass: statistics over h for each (i, j).
    sumv = tl.zeros((BN,), dtype=tl.float32)
    sumsq = tl.zeros((BN,), dtype=tl.float32)
    for h0 in range(0, H, BH):
        h_off = h0 + h
        h_mask = h_off < H
        ptr = base + h_off[:, None] * NN + j[None, :]
        m = h_mask[:, None] & j_mask[None, :]
        y0 = tl.load(ptr, mask=m, other=0.0).to(tl.float32)
        sumv += tl.sum(y0, axis=0)
        sumsq += tl.sum(y0 * y0, axis=0)

    mean = sumv / H
    var = sumsq / H - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    # Recompute output gate from original input: gate = sigmoid(out_gate(norm(x))).
    xg_ptrs = x_ptr + ((b * N + i) * N + j)[:, None] * C + c[None, :]
    xgm = j_mask[:, None] & c_mask[None, :]
    xg = tl.load(xg_ptrs, mask=xgm, other=0.0).to(tl.float32)

    xg_sum = tl.sum(xg, axis=1)
    xg_sumsq = tl.sum(xg * xg, axis=1)
    mean_x = xg_sum / C
    var_x = xg_sumsq / C - mean_x * mean_x
    var_x = tl.maximum(var_x, 0.0)
    rstd_x = 1.0 / tl.sqrt(var_x + 1e-5)

    nw = tl.load(norm_w_ptr + c, mask=c_mask, other=0.0).to(tl.float32)
    nb = tl.load(norm_b_ptr + c, mask=c_mask, other=0.0).to(tl.float32)
    xgn = (xg - mean_x[:, None]) * rstd_x[:, None] * nw[None, :] + nb[None, :]
    xgn_bf = xgn.to(tl.bfloat16)

    # Second pass: normalize C, multiply by gate, project to output dimension.
    acc = tl.zeros((BN, DB), dtype=tl.float32)
    for h0 in range(0, H, BH):
        h_off = h0 + h
        h_mask = h_off < H

        ptr = base + h_off[:, None] * NN + j[None, :]
        m = h_mask[:, None] & j_mask[None, :]
        y = tl.load(ptr, mask=m, other=0.0).to(tl.float32)

        wog_m = c_mask[:, None] & h_mask[None, :]
        wog = tl.load(
            wog_ptr + c[:, None] * H + h_off[None, :],
            mask=wog_m,
            other=0.0,
        ).to(tl.bfloat16)

        gacc = tl.zeros((BN, BH), dtype=tl.float32)
        pog = tl.dot(xgn_bf, wog, gacc)
        gate = _sigmoid_clamp(pog)

        gamma = tl.load(gamma_ptr + h_off, mask=h_mask, other=0.0).to(tl.float32)
        beta = tl.load(beta_ptr + h_off, mask=h_mask, other=0.0).to(tl.float32)

        y = (y - mean[None, :]) * rstd[None, :] * gamma[:, None] + beta[:, None]
        y = y * tl.trans(gate)
        y_t = tl.trans(y)  # [BN, BH]

        wout_m = d_mask[:, None] & h_mask[None, :]
        w = tl.load(
            wout_ptr + d[:, None] * H + h_off[None, :],
            mask=wout_m,
            other=0.0,
        ).to(tl.float32)
        w_t = tl.trans(w)  # [BH, DB]

        acc = tl.dot(y_t, w_t, acc, input_precision="tf32")

    out_ptrs = out_ptr + (b * N + i) * N * D + j[:, None] * D + d[None, :]
    tl.store(out_ptrs, acc.to(tl.float32), mask=j_mask[:, None] & d_mask[None, :])


def custom_kernel(data):
    # Accept either (x, mask, weights, config) or (x, weights, config).
    if isinstance(data, (tuple, list)):
        if len(data) == 4:
            x, mask, weights, config = data
        elif len(data) == 3:
            x, weights, config = data
            mask = None
        else:
            x, mask, weights, config = data[0], data[1], data[2], data[3]
    else:  # Defensive fallback, should not be needed.
        x, mask, weights, config = data

    x = x.detach()
    if not x.is_contiguous():
        x = x.contiguous()

    device = x.device
    B, N, _, C_in = x.shape

    H = int(weights["left_proj.weight"].shape[0])
    D_out = int(weights["to_out.weight"].shape[0])

    if B == 0 or N == 0:
        return torch.empty((B, N, N, D_out), dtype=torch.float32, device=device)

    has_mask = mask is not None
    if has_mask:
        mask = mask.detach()
        if not mask.is_contiguous():
            mask = mask.contiguous()
        if mask.dim() == 4 and mask.shape[-1] == 1:
            mask = mask.squeeze(-1).contiguous()
        mask_arg = mask
    else:
        mask_arg = x

    norm_w = _get_weight_float(weights, "norm.weight", device)
    norm_b = _get_weight_float(weights, "norm.bias", device)

    wl = _get_weight_bf16_trans(weights, "left_proj.weight", device)
    wgl = _get_weight_bf16_trans(weights, "left_gate.weight", device)
    wr = _get_weight_bf16_trans(weights, "right_proj.weight", device)
    wgr = _get_weight_bf16_trans(weights, "right_gate.weight", device)

    gamma = _get_weight_float(weights, "to_out_norm.weight", device)
    beta = _get_weight_float(weights, "to_out_norm.bias", device)
    wout = _get_weight_float(weights, "to_out.weight", device)
    wog = _get_weight_bf16_trans(weights, "out_gate.weight", device)

    # Stage 1: fused LayerNorm + projections/gates + mask.
    CB = _next_pow2(C_in)
    if CB <= 128:
        BK = 128
    elif CB <= 256:
        BK = 64
    elif CB <= 512:
        BK = 32
    else:
        BK = 16
    HB = _choose_hb(H)

    L = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    R = torch.empty_like(L)

    grid = (N, triton.cdiv(N, BK), B)
    _trimul_preprocess_kernel[grid](
        x,
        mask_arg,
        L,
        R,
        wl,
        wgl,
        wr,
        wgr,
        norm_w,
        norm_b,
        B,
        N,
        C_in,
        H,
        BK=BK,
        CB=CB,
        HB=HB,
        HAS_MASK=has_mask,
    )

    # Batched GEMM over all hidden channels: C[b, h, i, j] = sum_k L[b,h,i,k]*R[b,h,j,k]
    L2 = L.view(B * H, N, N)
    R2 = R.view(B * H, N, N)
    bmm_out = torch.bmm(L2, R2.transpose(-1, -2))
    bmm_out = bmm_out.contiguous()

    del L, R, L2, R2

    # Stage 2: LayerNorm over hidden dim + output gate + final projection.
    DB = _next_pow2(D_out)
    work = max(CB, DB)
    BN = 16
    for cand in (128, 64, 32, 16):
        if cand * work <= 8192:
            BN = cand
            break
    if BN > N:
        BN = max(16, _next_pow2(N))
        if BN > 128:
            BN = 128
    FBH = _choose_hb(H)

    out = torch.empty((B, N, N, D_out), dtype=torch.float32, device=device)
    grid = (N, triton.cdiv(N, BN), B)
    _trimul_finalize_kernel[grid](
        bmm_out,
        x,
        wog,
        norm_w,
        norm_b,
        gamma,
        beta,
        wout,
        out,
        B,
        N,
        C_in,
        H,
        D_out,
        BN=BN,
        BH=FBH,
        CB=CB,
        DB=DB,
    )

    return out
```
