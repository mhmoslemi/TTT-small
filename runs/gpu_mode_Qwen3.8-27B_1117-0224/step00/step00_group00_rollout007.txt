```python
"""
TriMul outgoing forward fused with Triton + cuBLAS.

Pipeline:
1. Fused LayerNorm + five projection/gate linear layers in one Triton kernel for the
   supported C sizes (128/256/384). It directly writes:
       L        [B, H, N, N]   (left values, h-major)
       RT       [B, H, N, N]   (right values transposed to [H, K, J] for batched GEMM)
       out_gate [B*N*N, H]     (sigmoid of out projection)
   For other C, it falls back to PyTorch GEMM + a Triton scatter kernel.
2. Central O(N^3 H) update as batched cuBLAS GEMM:
       out_h[b,h,i,j] = sum_k L[b,h,i,k] * RT[b,h,k,j]
3. Triton kernel fuses final LayerNorm over H, out_gate, and the final linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _sigmoid(x):
    pos = x > 0
    # Stable sigmoid: use exp(-x) for x >= 0 and exp(x)/(1+exp(x)) for x < 0.
    exp_pos = tl.exp(-tl.where(pos, x, 0.0))
    exp_neg = tl.exp(tl.where(pos, 0.0, x))
    out_pos = 1.0 / (1.0 + exp_pos)
    out_neg = exp_neg / (1.0 + exp_neg)
    return tl.where(pos, out_pos, out_neg)


@triton.jit
def _dot_fused(
    x0,
    x1,
    x2,
    w_ptr,
    base,
    h,
    C: tl.constexpr,
    BM: tl.constexpr,
    BH: tl.constexpr,
    KCH: tl.constexpr,
):
    """Compute x[BM,C] @ W[group_base+H, C]^T for one H-chunk using kept k-chunks."""
    acc = tl.zeros((BM, BH), dtype=tl.float32)
    k = tl.arange(0, KCH)

    if C == 128:
        w0 = tl.load(w_ptr + k[:, None] + (base + h[None, :]) * C)
        acc = tl.dot(x0, w0, acc)
    elif C == 256:
        w0 = tl.load(w_ptr + k[:, None] + (base + h[None, :]) * C)
        acc = tl.dot(x0, w0, acc)

        k1 = k + KCH
        w1 = tl.load(w_ptr + k1[:, None] + (base + h[None, :]) * C)
        acc = tl.dot(x1, w1, acc)
    else:
        # C == 384 in the fused path.
        w0 = tl.load(w_ptr + k[:, None] + (base + h[None, :]) * C)
        acc = tl.dot(x0, w0, acc)

        k1 = k + KCH
        w1 = tl.load(w_ptr + k1[:, None] + (base + h[None, :]) * C)
        acc = tl.dot(x1, w1, acc)

        k2 = k + 2 * KCH
        w2 = tl.load(w_ptr + k2[:, None] + (base + h[None, :]) * C)
        acc = tl.dot(x2, w2, acc)

    return acc


@triton.jit
def _preproc_fused_kernel(
    x_ptr,
    mask_ptr,
    L_ptr,
    RT_ptr,
    G_ptr,
    w_ptr,
    wsum_ptr,
    bias_ptr,
    N,
    B,
    C: tl.constexpr,
    H: tl.constexpr,
    BM: tl.constexpr,
    P: tl.constexpr,
    Q: tl.constexpr,
    KCH: tl.constexpr,
    BH: tl.constexpr,
    NUM_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)

    num_qb = (N + Q - 1) // Q
    num_pb = (N + P - 1) // P
    num_pq = num_pb * num_qb

    pid_b = pid // num_pq
    pid_pq = pid % num_pq
    pid_p = pid_pq // num_qb
    pid_q = pid_pq % num_qb

    p0 = pid_p * P
    q0 = pid_q * Q

    r = tl.arange(0, BM)
    # Token order: p varies faster than q, so both L and RT stores get P-long contiguous runs.
    pi = r % P
    qi = r // P

    p = p0 + pi
    q = q0 + qi
    valid = (p < N) & (q < N)

    NN = N * N
    m = pid_b * NN + p * N + q

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + m, mask=valid, other=0.0).to(tl.float32)
    else:
        mask_val = tl.full((BM,), 1.0, dtype=tl.float32)

    k = tl.arange(0, KCH)
    x_ptrs = x_ptr + m[:, None] * C + k[None, :]

    if C == 128:
        x0 = tl.load(x_ptrs, mask=valid[:, None], other=0.0)

        total_sum = tl.sum(x0, axis=1)
        mean = total_sum / C

        xm0 = x0 - mean[:, None]
        total_sq = tl.sum(xm0 * xm0, axis=1)
        var = total_sq / C
        var = tl.maximum(var, 0.0)
        rstd = 1.0 / tl.sqrt(var + 1e-5)

        xb0 = x0.to(tl.bfloat16)
        xb1 = xb0
        xb2 = xb0

    elif C == 256:
        x0 = tl.load(x_ptrs, mask=valid[:, None], other=0.0)
        x1 = tl.load(x_ptrs + KCH, mask=valid[:, None], other=0.0)

        s0 = tl.sum(x0, axis=1)
        s1 = tl.sum(x1, axis=1)
        mean = (s0 + s1) / C

        xm0 = x0 - mean[:, None]
        xm1 = x1 - mean[:, None]
        total_sq = tl.sum(xm0 * xm0, axis=1) + tl.sum(xm1 * xm1, axis=1)
        var = total_sq / C
        var = tl.maximum(var, 0.0)
        rstd = 1.0 / tl.sqrt(var + 1e-5)

        xb0 = x0.to(tl.bfloat16)
        xb1 = x1.to(tl.bfloat16)
        xb2 = xb0

    else:
        # C == 384
        x0 = tl.load(x_ptrs, mask=valid[:, None], other=0.0)
        x1 = tl.load(x_ptrs + KCH, mask=valid[:, None], other=0.0)
        x2 = tl.load(x_ptrs + 2 * KCH, mask=valid[:, None], other=0.0)

        s0 = tl.sum(x0, axis=1)
        s1 = tl.sum(x1, axis=1)
        s2 = tl.sum(x2, axis=1)
        mean = (s0 + s1 + s2) / C

        xm0 = x0 - mean[:, None]
        xm1 = x1 - mean[:, None]
        xm2 = x2 - mean[:, None]
        total_sq = (
            tl.sum(xm0 * xm0, axis=1)
            + tl.sum(xm1 * xm1, axis=1)
            + tl.sum(xm2 * xm2, axis=1)
        )
        var = total_sq / C
        var = tl.maximum(var, 0.0)
        rstd = 1.0 / tl.sqrt(var + 1e-5)

        xb0 = x0.to(tl.bfloat16)
        xb1 = x1.to(tl.bfloat16)
        xb2 = x2.to(tl.bfloat16)

    col_l = p * N + q
    col_rt = q * N + p

    for h_idx in range(0, NUM_H):
        h = h_idx * BH + tl.arange(0, BH)

        # left
        accL = _dot_fused(xb0, xb1, xb2, w_ptr, 0, h, C, BM, BH, KCH)
        bL = tl.load(bias_ptr + h)
        sL = tl.load(wsum_ptr + h)
        valL = (accL - mean[:, None] * sL[None, :]) * rstd[:, None] + bL[None, :]

        # left gate
        accLG = _dot_fused(xb0, xb1, xb2, w_ptr, 2 * H, h, C, BM, BH, KCH)
        bLG = tl.load(bias_ptr + 2 * H + h)
        sLG = tl.load(wsum_ptr + 2 * H + h)
        valLG = (accLG - mean[:, None] * sLG[None, :]) * rstd[:, None] + bLG[None, :]

        sigL = _sigmoid(valLG)
        valL = valL * sigL * mask_val[:, None]

        L_ptrs = L_ptr + (pid_b * H + h)[:, None] * NN + col_l[None, :]
        tl.store(L_ptrs, tl.trans(valL.to(tl.bfloat16)), mask=valid[None, :])

        # right
        accR = _dot_fused(xb0, xb1, xb2, w_ptr, H, h, C, BM, BH, KCH)
        bR = tl.load(bias_ptr + H + h)
        sR = tl.load(wsum_ptr + H + h)
        valR = (accR - mean[:, None] * sR[None, :]) * rstd[:, None] + bR[None, :]

        # right gate
        accRG = _dot_fused(xb0, xb1, xb2, w_ptr, 3 * H, h, C, BM, BH, KCH)
        bRG = tl.load(bias_ptr + 3 * H + h)
        sRG = tl.load(wsum_ptr + 3 * H + h)
        valRG = (accRG - mean[:, None] * sRG[None, :]) * rstd[:, None] + bRG[None, :]

        sigR = _sigmoid(valRG)
        valR = valR * sigR * mask_val[:, None]

        RT_ptrs = RT_ptr + (pid_b * H + h)[:, None] * NN + col_rt[None, :]
        tl.store(RT_ptrs, tl.trans(valR.to(tl.bfloat16)), mask=valid[None, :])

        # out gate
        accOG = _dot_fused(xb0, xb1, xb2, w_ptr, 4 * H, h, C, BM, BH, KCH)
        bOG = tl.load(bias_ptr + 4 * H + h)
        sOG = tl.load(wsum_ptr + 4 * H + h)
        valOG = (accOG - mean[:, None] * sOG[None, :]) * rstd[:, None] + bOG[None, :]
        og = _sigmoid(valOG)

        G_ptrs = G_ptr + m[:, None] * H + h[None, :]
        tl.store(G_ptrs, og.to(tl.bfloat16), mask=valid[:, None])


@triton.jit
def _scatter_kernel(
    proj_ptr,
    mask_ptr,
    L_ptr,
    RT_ptr,
    G_ptr,
    N,
    H: tl.constexpr,
    BM: tl.constexpr,
    P: tl.constexpr,
    Q: tl.constexpr,
    BH: tl.constexpr,
    NUM_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """Scatter/proj fusion for the generic path.

    proj layout: [B*N*N, 5H], channel order:
    0:H left, H:2H right, 2H:3H left_gate, 3H:4H right_gate, 4H:5H out_gate.
    """
    pid = tl.program_id(0)

    num_qb = (N + Q - 1) // Q
    num_pb = (N + P - 1) // P
    num_pq = num_pb * num_qb

    pid_b = pid // num_pq
    pid_pq = pid % num_pq
    pid_p = pid_pq // num_qb
    pid_q = pid_pq % num_qb

    p0 = pid_p * P
    q0 = pid_q * Q

    r = tl.arange(0, BM)
    pi = r % P
    qi = r // P

    p = p0 + pi
    q = q0 + qi
    valid = (p < N) & (q < N)

    NN = N * N
    m = pid_b * NN + p * N + q

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + m, mask=valid, other=0.0).to(tl.float32)
    else:
        mask_val = tl.full((BM,), 1.0, dtype=tl.float32)

    col_l = p * N + q
    col_rt = q * N + p
    M5H = 5 * H

    for h_idx in range(0, NUM_H):
        h = h_idx * BH + tl.arange(0, BH)
        row = m[:, None] * M5H + h[None, :]

        left = tl.load(proj_ptr + row, mask=valid[:, None], other=0.0).to(tl.float32)
        right = tl.load(proj_ptr + row + H, mask=valid[:, None], other=0.0).to(tl.float32)
        lg = tl.load(proj_ptr + row + 2 * H, mask=valid[:, None], other=0.0).to(tl.float32)
        rg = tl.load(proj_ptr + row + 3 * H, mask=valid[:, None], other=0.0).to(tl.float32)
        og = tl.load(proj_ptr + row + 4 * H, mask=valid[:, None], other=0.0).to(tl.float32)

        valL = left * _sigmoid(lg) * mask_val[:, None]
        valR = right * _sigmoid(rg) * mask_val[:, None]

        L_ptrs = L_ptr + (pid_b * H + h)[:, None] * NN + col_l[None, :]
        tl.store(L_ptrs, tl.trans(valL.to(tl.bfloat16)), mask=valid[None, :])

        RT_ptrs = RT_ptr + (pid_b * H + h)[:, None] * NN + col_rt[None, :]
        tl.store(RT_ptrs, tl.trans(valR.to(tl.bfloat16)), mask=valid[None, :])

        G_ptrs = G_ptr + m[:, None] * H + h[None, :]
        tl.store(G_ptrs, _sigmoid(og).to(tl.bfloat16), mask=valid[:, None])


@triton.jit
def _final_kernel(
    out_ptr,
    gate_ptr,
    nw_ptr,
    nb_ptr,
    wto_ptr,
    outf_ptr,
    N,
    C,
    H,
    BM: tl.constexpr,
    BH: tl.constexpr,
    BC: tl.constexpr,
    NUM_H: tl.constexpr,
    NUM_C: tl.constexpr,
):
    pid = tl.program_id(0)
    NN = N * N
    m_blocks = (NN + BM - 1) // BM

    pid_b = pid // m_blocks
    pid_m = pid % m_blocks

    m = pid_m * BM + tl.arange(0, BM)
    m_valid = m < NN

    sum_v = tl.zeros((BM,), dtype=tl.float32)
    sum_sq = tl.zeros((BM,), dtype=tl.float32)

    # Pass 1: LayerNorm statistics over H.
    for h_idx in range(0, NUM_H):
        h = h_idx * BH + tl.arange(0, BH)
        raw_ptrs = out_ptr + (pid_b * H + h)[:, None] * NN + m[None, :]
        raw = tl.load(raw_ptrs, mask=m_valid[None, :], other=0.0).to(tl.float32)
        sum_v += tl.sum(raw, axis=0)
        sum_sq += tl.sum(raw * raw, axis=0)

    mean = sum_v / H
    var = sum_sq / H - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    # Pass 2: normalize + gate + final linear projection.
    for c_idx in range(0, NUM_C):
        c = c_idx * BC + tl.arange(0, BC)
        c_valid = c < C
        acc = tl.zeros((BM, BC), dtype=tl.float32)

        for h_idx in range(0, NUM_H):
            h = h_idx * BH + tl.arange(0, BH)

            raw_ptrs = out_ptr + (pid_b * H + h)[:, None] * NN + m[None, :]
            raw = tl.load(raw_ptrs, mask=m_valid[None, :], other=0.0).to(tl.float32)

            gate_ptrs = gate_ptr + m[:, None] * H + h[None, :]
            gate = tl.load(gate_ptrs, mask=m_valid[:, None], other=0.0).to(tl.float32)

            nw = tl.load(nw_ptr + h)
            nb = tl.load(nb_ptr + h)

            norm = (raw - mean[None, :]) * rstd[None, :]
            norm = norm * nw[:, None] + nb[:, None]

            ny = norm * tl.trans(gate)
            a = tl.trans(ny.to(tl.bfloat16))

            # to_out weight is [C, H], so [H, C] tile is h + c*H.
            wto_ptrs = wto_ptr + h[:, None] + c[None, :] * H
            wtile = tl.load(wto_ptrs, mask=c_valid[None, :], other=0.0)

            acc = tl.dot(a, wtile, acc)

        out_ptrs = outf_ptr + (pid_b * NN + m)[:, None] * C + c[None, :]
        tl.store(out_ptrs, acc, mask=m_valid[:, None] & c_valid[None, :])


def _pow2_divisor(n, max_block, min_block=16):
    n = int(n)
    max_block = int(max_block)
    if n <= 0:
        return 1
    start = min(n, max_block)
    b = 1 << (start.bit_length() - 1)
    while b > min_block and n % b != 0:
        b //= 2
    return b


def custom_kernel(data):
    """Entry point matching the task interface."""
    if len(data) == 3:
        input_tensor, weights, config = data
        mask = None
    else:
        input_tensor, mask, weights, config = data

    device = input_tensor.device
    B, N, N2, C = input_tensor.shape
    if N2 != N:
        N = min(N, N2)

    # Hidden dimension from weights is the safest source.
    H = int(weights["left_proj.weight"].shape[0])

    x = input_tensor.contiguous()
    if x.dtype != torch.float32:
        x = x.float()

    has_mask = mask is not None and isinstance(mask, torch.Tensor) and mask.numel() > 0
    if has_mask:
        mask_f = mask.to(torch.float32)
        if mask_f.dim() == 4 and mask_f.shape[-1] == 1:
            mask_f = mask_f.squeeze(-1)
        mask_f = mask_f.reshape(B, N, N).contiguous()
        mask_arg = mask_f
    else:
        mask_arg = x  # dummy pointer when HAS_MASK=False

    norm_w = weights["norm.weight"].float().contiguous()
    norm_b = weights["norm.bias"].float().contiguous()

    left_proj_w = weights["left_proj.weight"].float().contiguous()
    right_proj_w = weights["right_proj.weight"].float().contiguous()
    left_gate_w = weights["left_gate.weight"].float().contiguous()
    right_gate_w = weights["right_gate.weight"].float().contiguous()
    out_gate_w = weights["out_gate.weight"].float().contiguous()

    to_out_norm_w = weights["to_out_norm.weight"].float().contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].float().contiguous()
    to_out_w = weights["to_out.weight"].float().contiguous()

    L = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    RT = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    G = torch.empty((B * N * N, H), dtype=torch.bfloat16, device=device)

    # Choose block divisors that are powers of two and split H cleanly.
    BH = _pow2_divisor(H, 64)
    NUM_H = triton.cdiv(H, BH)

    # Fused preproc path supports C=128,256,384 with a power-of-two k-chunk of 128.
    use_fused_preproc = C in (128, 256, 384)

    if use_fused_preproc:
        if C == 128:
            BM = 128
        else:
            BM = 64
        P = 32
        Q = BM // P
        KCH = 128

        W1 = torch.cat(
            [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
            dim=0,
        ).contiguous()

        # Projection of LayerNorm output can be written as:
        #   rstd * (x @ (W * w)) - rstd * mean * sum_c(W_c * w_c) + W @ b
        Wscaled = W1 * norm_w.view(1, -1)
        Wnorm = Wscaled.to(torch.bfloat16).contiguous()
        Wsum = Wscaled.sum(dim=1).float().contiguous()
        biasB = (W1 @ norm_b).float().contiguous()

        num_p = triton.cdiv(N, P)
        num_q = triton.cdiv(N, Q)
        grid = (B * num_p * num_q,)

        _preproc_fused_kernel[grid](
            x,
            mask_arg,
            L,
            RT,
            G,
            Wnorm,
            Wsum,
            biasB,
            N,
            B,
            C,
            H,
            BM,
            P,
            Q,
            KCH,
            BH,
            NUM_H,
            has_mask,
            num_warps=8,
            num_stages=1,
        )

        del W1, Wscaled, Wnorm, Wsum, biasB

    else:
        # Generic path: PyTorch LayerNorm + one large GEMM, then Triton scatter.
        M = B * N * N
        xv = x.reshape(M, C)
        x_norm = F.layer_norm(xv, (C,), norm_w, norm_b, eps=1e-5)
        x_norm = x_norm.to(torch.bfloat16).contiguous()

        W1 = torch.cat(
            [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
            dim=0,
        ).to(torch.bfloat16).contiguous()

        proj = torch.mm(x_norm, W1.t()).contiguous()

        BM = 64
        P = 32
        Q = BM // P

        num_p = triton.cdiv(N, P)
        num_q = triton.cdiv(N, Q)
        grid = (B * num_p * num_q,)

        _scatter_kernel[grid](
            proj,
            mask_arg,
            L,
            RT,
            G,
            N,
            H,
            BM,
            P,
            Q,
            BH,
            NUM_H,
            has_mask,
            num_warps=4,
            num_stages=2,
        )

        del proj, x_norm, W1

    # Central TriMul: batched GEMM over (B, H).
    batch = B * H
    L_flat = L.view(batch, N, N)
    RT_flat = RT.view(batch, N, N)
    out_h = torch.bmm(L_flat, RT_flat)
    out_h = out_h.view(B, H, N, N).contiguous()

    # Final fused LayerNorm + gate + linear projection.
    out_final = torch.empty((B, N, N, C), dtype=torch.float32, device=device)
    Wto_bf = to_out_w.to(torch.bfloat16).contiguous()

    fm_BM = 64
    fm_BH = BH
    fm_BC = _pow2_divisor(C, 128)
    fm_NUM_H = triton.cdiv(H, fm_BH)
    fm_NUM_C = triton.cdiv(C, fm_BC)

    m_blocks = triton.cdiv(N * N, fm_BM)
    grid = (B * m_blocks,)

    _final_kernel[grid](
        out_h,
        G,
        to_out_norm_w,
        to_out_norm_b,
        Wto_bf,
        out_final,
        N,
        C,
        H,
        fm_BM,
        fm_BH,
        fm_BC,
        fm_NUM_H,
        fm_NUM_C,
        num_warps=4,
        num_stages=2,
    )

    return out_final
```
