```python
import torch
import triton
import triton.language as tl


"""
TriMul forward fused hybrid implementation.

Pipeline:
1) Triton LayerNorm over input channels C, producing bf16 activations.
2) PyTorch batched GEMM for all five linear projections into one tensor [B*N*N, 5*H].
3) Triton kernel applies sigmoid gates and optional mask, and scatters left/right/out_gate
   into sequence-major layouts [B, H, N, N] that are optimal for batched GEMM.
4) PyTorch cuBLAS batched GEMM computes the core Triangle Multiplicative Update:
       out[b, h, i, j] = sum_k left[b, h, i, k] * right[b, h, j, k].
5) Triton fused kernel transposes back, applies the hidden-layer LayerNorm,
   multiplies by out_gate, and performs the final projection to [B, N, N, C] fp32.
"""


@triton.jit
def _trimul_layer_norm_kernel(
    x_ptr,
    y_ptr,
    w_ptr,
    b_ptr,
    M,
    C: tl.constexpr,
    EPS: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    """LayerNorm over the last dimension C, written in low precision."""
    pid = tl.program_id(0)

    m_offs = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    c_offs = tl.arange(0, BLOCK_C)

    m_mask = m_offs < M
    c_mask = c_offs < C
    mask = m_mask[:, None] & c_mask[None, :]

    x = tl.load(x_ptr + m_offs[:, None] * C + c_offs[None, :], mask=mask, other=0.0).to(tl.float32)

    mean = tl.sum(x, axis=1) / C
    xc = x - mean[:, None]
    var = tl.sum(xc * xc, axis=1) / C
    var = tl.maximum(var, 0.0)
    inv_std = 1.0 / tl.sqrt(var + EPS)

    w = tl.load(w_ptr + c_offs, mask=c_mask, other=0.0).to(tl.float32)
    b = tl.load(b_ptr + c_offs, mask=c_mask, other=0.0).to(tl.float32)

    y = xc * inv_std[:, None] * w[None, :] + b[None, :]

    tl.store(
        y_ptr + m_offs[:, None] * C + c_offs[None, :],
        y.to(y_ptr.dtype.element_ty),
        mask=mask,
    )


@triton.jit
def _trimul_scatter_gates_kernel(
    z_ptr,
    mask_ptr,
    l_ptr,
    r_ptr,
    og_ptr,
    P,
    H: tl.constexpr,
    H_PAD: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_P: tl.constexpr,
):
    """
    Input  Z: [B, P, 5*H] in one row:
             [0:H] left_proj, [H:2H] right_proj, [2H:3H] left_gate,
             [3H:4H] right_gate, [4H:5H] out_gate.

    Output L/R/OG: [B, H, P], i.e. [B, H, N, N].
    Applies stable sigmoid gates and optional mask.
    """
    pid_p = tl.program_id(0)
    pid_b = tl.program_id(1)

    p_offs = pid_p * BLOCK_P + tl.arange(0, BLOCK_P)
    p_mask = p_offs < P

    h_offs = tl.arange(0, H_PAD)
    h_mask = h_offs < H

    load_mask = p_mask[:, None] & h_mask[None, :]
    store_mask = h_mask[:, None] & p_mask[None, :]

    row_stride = 5 * H
    z_base = z_ptr + pid_b * P * row_stride + p_offs[:, None] * row_stride

    l_base = l_ptr + pid_b * H * P + h_offs[:, None] * P + p_offs[None, :]
    r_base = r_ptr + pid_b * H * P + h_offs[:, None] * P + p_offs[None, :]
    og_base = og_ptr + pid_b * H * P + h_offs[:, None] * P + p_offs[None, :]

    # left = left_proj(x) * sigmoid(left_gate(x)) * mask
    left = tl.load(z_base + h_offs[None, :], mask=load_mask, other=0.0).to(tl.float32)
    gate_l = tl.load(z_base + 2 * H + h_offs[None, :], mask=load_mask, other=0.0).to(tl.float32)
    gate_l = tl.minimum(tl.maximum(gate_l, -50.0), 50.0)
    left_g = left * (1.0 / (1.0 + tl.exp(-gate_l)))
    if HAS_MASK:
        m = tl.load(mask_ptr + pid_b * P + p_offs, mask=p_mask, other=0.0).to(tl.float32)
        tl.store(l_base, tl.trans(left_g * m[:, None]).to(l_ptr.dtype.element_ty), mask=store_mask)
    else:
        tl.store(l_base, tl.trans(left_g).to(l_ptr.dtype.element_ty), mask=store_mask)

    # right = right_proj(x) * sigmoid(right_gate(x)) * mask
    right = tl.load(z_base + H + h_offs[None, :], mask=load_mask, other=0.0).to(tl.float32)
    gate_r = tl.load(z_base + 3 * H + h_offs[None, :], mask=load_mask, other=0.0).to(tl.float32)
    gate_r = tl.minimum(tl.maximum(gate_r, -50.0), 50.0)
    right_g = right * (1.0 / (1.0 + tl.exp(-gate_r)))
    if HAS_MASK:
        m = tl.load(mask_ptr + pid_b * P + p_offs, mask=p_mask, other=0.0).to(tl.float32)
        tl.store(r_base, tl.trans(right_g * m[:, None]).to(r_ptr.dtype.element_ty), mask=store_mask)
    else:
        tl.store(r_base, tl.trans(right_g).to(r_ptr.dtype.element_ty), mask=store_mask)

    # out_gate = sigmoid(out_gate(x))
    gate_o = tl.load(z_base + 4 * H + h_offs[None, :], mask=load_mask, other=0.0).to(tl.float32)
    gate_o = tl.minimum(tl.maximum(gate_o, -50.0), 50.0)
    og = 1.0 / (1.0 + tl.exp(-gate_o))
    tl.store(og_base, tl.trans(og).to(og_ptr.dtype.element_ty), mask=store_mask)


@triton.jit
def _trimul_post_kernel(
    y_ptr,
    og_ptr,
    w_ptr,
    gamma_ptr,
    beta_ptr,
    out_ptr,
    P,
    H: tl.constexpr,
    C: tl.constexpr,
    H_PAD: tl.constexpr,
    EPS: tl.constexpr,
    BLOCK_P: tl.constexpr,
    BLOCK_C: tl.constexpr,
    NUM_C: tl.constexpr,
):
    """
    Fused:
    - transpose contraction output from [B, H, P] to [P, H],
    - LayerNorm over H,
    - multiply by sigmoid out_gate,
    - final linear projection to C channels.
    """
    pid_p = tl.program_id(0)
    pid_b = tl.program_id(1)

    p_offs = pid_p * BLOCK_P + tl.arange(0, BLOCK_P)
    p_mask = p_offs < P

    h_offs = tl.arange(0, H_PAD)
    h_mask = h_offs < H

    y_mask = h_mask[:, None] & p_mask[None, :]

    y = tl.load(
        y_ptr + pid_b * H * P + h_offs[:, None] * P + p_offs[None, :],
        mask=y_mask,
        other=0.0,
    ).to(tl.float32)

    mean = tl.sum(y, axis=0) / H
    yc = y - mean[None, :]
    var = tl.sum(yc * yc, axis=0) / H
    var = tl.maximum(var, 0.0)
    inv_std = 1.0 / tl.sqrt(var + EPS)

    gamma = tl.load(gamma_ptr + h_offs, mask=h_mask, other=0.0).to(tl.float32)
    beta = tl.load(beta_ptr + h_offs, mask=h_mask, other=0.0).to(tl.float32)

    y_norm = yc * inv_std[None, :] * gamma[:, None] + beta[:, None]

    og = tl.load(
        og_ptr + pid_b * H * P + h_offs[:, None] * P + p_offs[None, :],
        mask=y_mask,
        other=0.0,
    ).to(tl.float32)

    z_t = tl.trans(y_norm * og).to(y_ptr.dtype.element_ty)

    for c_idx in tl.static_range(NUM_C):
        c_offs = c_idx * BLOCK_C + tl.arange(0, BLOCK_C)
        c_mask = c_offs < C

        w = tl.load(
            w_ptr + h_offs[:, None] * C + c_offs[None, :],
            mask=c_mask[None, :],
            other=0.0,
        )

        acc = tl.dot(z_t, w.to(z_t.dtype))

        out_ptrs = out_ptr + pid_b * P * C + p_offs[:, None] * C + c_offs[None, :]
        tl.store(out_ptrs, acc, mask=p_mask[:, None] & c_mask[None, :])


def custom_kernel(data):
    """
    Forward pass for TriMul.

    Expected input:
        data = (x, mask, weights, config)
    or:
        data = (x, weights, config) when no mask is provided.
    """
    if len(data) == 4:
        x, mask, weights, config = data
    else:
        x, weights, config = data
        mask = None

    with torch.no_grad():
        B, N, _, C = x.shape
        P = N * N
        M = B * P

        C = int(config.get("dim", C))
        H = int(config.get("hidden_dim", 128))

        eps_in = float(config.get("norm_eps", config.get("eps", 1e-5)))
        eps_out = float(config.get("to_out_norm_eps", eps_in))

        device = x.device
        work_dtype = torch.bfloat16

        # ----- weights -----
        norm_w = weights["norm.weight"].to(torch.float32).contiguous()
        norm_b = weights["norm.bias"].to(torch.float32).contiguous()

        parts = [
            weights["left_proj.weight"].to(work_dtype).t(),
            weights["right_proj.weight"].to(work_dtype).t(),
            weights["left_gate.weight"].to(work_dtype).t(),
            weights["right_gate.weight"].to(work_dtype).t(),
            weights["out_gate.weight"].to(work_dtype).t(),
        ]
        W_stack = torch.cat(parts, dim=1).contiguous()  # [C, 5H]

        w_out = weights["to_out.weight"].to(work_dtype)  # [C, H]
        H_PAD = 1 << (max(1, H) - 1).bit_length()
        if H_PAD < 16:
            H_PAD = 16

        if H_PAD == H:
            W_out_t = w_out.t().contiguous()
        else:
            W_out_t = torch.zeros((H_PAD, C), dtype=work_dtype, device=device)
            W_out_t[:H] = w_out.t()

        to_norm_w = weights["to_out_norm.weight"].to(torch.float32).contiguous()
        to_norm_b = weights["to_out_norm.bias"].to(torch.float32).contiguous()

        # ----- mask -----
        has_mask = mask is not None and not bool(config.get("nomask", False))
        if has_mask:
            if mask.numel() == B * P:
                mask_flat = mask.reshape(B, P)
            elif mask.numel() == P:
                mask_flat = mask.reshape(1, P).expand(B, P)
            else:
                mask_flat = mask.reshape(B, P)
            mask_flat = mask_flat.contiguous()
        else:
            mask_flat = x.reshape(M, C)  # dummy pointer, not loaded when has_mask=False

        # ----- Step 1: input LayerNorm, output in low precision -----
        x_flat = x.reshape(M, C)

        xn = torch.empty((M, C), dtype=work_dtype, device=device)

        BLOCK_LN_C = max(16, 1 << (C - 1).bit_length())
        if BLOCK_LN_C <= 128:
            BLOCK_LN_M = 128
        elif BLOCK_LN_C <= 512:
            BLOCK_LN_M = 64
        else:
            BLOCK_LN_M = 32

        ln_grid = ((M + BLOCK_LN_M - 1) // BLOCK_LN_M,)
        _trimul_layer_norm_kernel[ln_grid](
            x_flat,
            xn,
            norm_w,
            norm_b,
            M,
            C=C,
            EPS=eps_in,
            BLOCK_M=BLOCK_LN_M,
            BLOCK_C=BLOCK_LN_C,
            num_warps=4,
        )

        # ----- Step 2: all five linear projections in one GEMM -----
        Z = torch.mm(xn, W_stack)  # [B*N*N, 5H]
        del xn

        # ----- Step 3: gates, mask, scatter to [B, H, N, N] -----
        L = torch.empty((B, H, N, N), dtype=work_dtype, device=device)
        R = torch.empty_like(L)
        OG = torch.empty_like(L)

        BLOCK_P = 128
        scatter_grid = ((P + BLOCK_P - 1) // BLOCK_P, B)
        _trimul_scatter_gates_kernel[scatter_grid](
            Z,
            mask_flat,
            L,
            R,
            OG,
            P,
            H=H,
            H_PAD=H_PAD,
            HAS_MASK=has_mask,
            BLOCK_P=BLOCK_P,
            num_warps=4,
        )
        del Z
        del W_stack

        # ----- Step 4: core TriMul contraction via batched GEMM -----
        Lf = L.view(B * H, N, N)
        Rf = R.view(B * H, N, N)
        Y = torch.bmm(Lf, Rf.transpose(1, 2)).view(B, H, N, N)
        del L
        del R

        # ----- Step 5: fused LayerNorm + gate + final projection -----
        out_flat = torch.empty((B, P, C), dtype=torch.float32, device=device)

        if C >= 128 and C % 128 == 0:
            BLOCK_C = 128
        elif C >= 64 and C % 64 == 0:
            BLOCK_C = 64
        elif C >= 32 and C % 32 == 0:
            BLOCK_C = 32
        else:
            BLOCK_C = 16

        NUM_C = (C + BLOCK_C - 1) // BLOCK_C

        BLOCK_POST_P = 128
        post_grid = ((P + BLOCK_POST_P - 1) // BLOCK_POST_P, B)
        _trimul_post_kernel[post_grid](
            Y,
            OG,
            W_out_t,
            to_norm_w,
            to_norm_b,
            out_flat,
            P,
            H=H,
            C=C,
            H_PAD=H_PAD,
            EPS=eps_out,
            BLOCK_P=BLOCK_POST_P,
            BLOCK_C=BLOCK_C,
            NUM_C=NUM_C,
            num_warps=4,
        )

        return out_flat.view(B, N, N, C)
```
