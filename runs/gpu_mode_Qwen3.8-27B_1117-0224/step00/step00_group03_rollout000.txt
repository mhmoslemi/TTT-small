import torch.nn.functional as F
import triton

def fused_gate_kernel(

    left_gate_ptr, right_gate_ptr,
    out_ptr,  # same as left_ptr but we write in-place? No, let's use separate output

import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fused_gating_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    base = pid * H
    d = tl.arange(0, H)
    
    m = tl.load(mask_ptr + pid)
    
    left = tl.load(left_ptr + base + d)
    right = tl.load(right_ptr + base + d)
    lg = tl.load(left_gate_ptr + base + d)
    rg = tl.load(right_gate_ptr + base + d)
    
    tl.store(left_ptr + base + d, left * m * lg)
    tl.store(right_ptr + base + d, right * m * rg)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    
    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weights["norm.weight"], weights["norm.bias"])
    
    # Projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))
    
    # Fused gating (Triton)
    total = B * N * N
    fused_gating_kernel[(total,)](left, right, mask, left_gate, right_gate, H=H)
    
    # Core matmul
    left_perm = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_perm = right.permute(0, 3, 1, 2).reshape(B * H, N, N)
    
    # Use BF16 for tensor core acceleration
    bmm_out = torch.bmm(left_perm.to(torch.bfloat16), right_perm.to(torch.bfloat16).transpose(1, 2))
    
    core_out = bmm_out.to(torch.float32).reshape(B, H, N, N).permute(0, 2, 3, 1)
    
    # Final LayerNorm + gate + projection
    core_out = F.layer_norm(core_out, (H,), weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    core_out = core_out * out_gate
    out = F.linear(core_out, weights["to_out.weight"])
