@triton.jit
def fused_preprocess_kernel(
    x_ptr, mask_ptr,
    norm_w_ptr, norm_b_ptr,
    left_proj_w_ptr, right_proj_w_ptr,
    left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    B, N, dim: tl.constexpr, hidden_dim: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    # Each block handles BLOCK_M elements of (b, i, k)
    
    # Load x_tile: [BLOCK_M, dim]
    # x is [B, N, N, dim], element (b, i, k, d) is at offset b*N*N*dim + i*N*dim + k*dim + d
    # Flatten index: idx = b*N*N + i*N + k, so offset = idx * dim + d
    
    offsets_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    offsets_d = tl.arange(0, BLOCK_D)  # [BLOCK_D], BLOCK_D = dim
    
    # Load x: [BLOCK_M, BLOCK_D]
    x_ptrs = x_ptr + offsets_m[:, None] * dim + offsets_d[None, :]
    x_tile = tl.load(x_ptrs)  # [BLOCK_M, dim]
    
    # LayerNorm
    mean = tl.sum(x_tile, axis=1) / dim  # [BLOCK_M]
    diff = x_tile - mean[:, None]
    var = tl.sum(diff * diff, axis=1) / dim  # [BLOCK_M]
    std = tl.sqrt(var + 1e-5)
    x_norm = diff / std[:, None]
    
    # Load norm weight and bias
    norm_w = tl.load(norm_w_ptr + offsets_d)  # [dim]
    norm_b = tl.load(norm_b_ptr + offsets_d)  # [dim]
    x_norm = x_norm * norm_w[None, :] + norm_b[None, :]
    
    # Now compute 5 projections
    # For each: result = x_norm @ W^T where W is [dim, hidden_dim]
    # result[m, h] = sum_d x_norm[m, d] * W[h, d] = sum_d x_norm[m, d] * W^T[d, h]
    
    # We can do this as a series of dot products
    # For hidden_dim = 128, we can process it in blocks of BLOCK_H
    
    for h_block in range(0, hidden_dim, BLOCK_H):
        offsets_h = h_block + tl.arange(0, BLOCK_H)  # [BLOCK_H]
        
        # Load weight tiles: [dim, BLOCK_H] for each projection
        # W[h, d] is at offset h * dim + d
        w_left = tl.load(left_proj_w_ptr + offsets_h[:, None] * dim + offsets_d[None, :])  # [BLOCK_H, dim]
        w_right = tl.load(right_proj_w_ptr + offsets_h[:, None] * dim + offsets_d[None, :])
        w_lg = tl.load(left_gate_w_ptr + offsets_h[:, None] * dim + offsets_d[None, :])
        w_rg = tl.load(right_gate_w_ptr + offsets_h[:, None] * dim + offsets_d[None, :])
        w_og = tl.load(out_gate_w_ptr + offsets_h[:, None] * dim + offsets_d[None, :])
        
        # Compute projections: [BLOCK_M, BLOCK_H] = [BLOCK_M, dim] @ [dim, BLOCK_H]
        # Need to transpose W: W^T is [dim, BLOCK_H]
        left_proj = tl.dot(x_norm, tl.trans(w_left))  # [BLOCK_M, BLOCK_H]
        right_proj = tl.dot(x_norm, tl.trans(w_right))
        left_gate = tl.sigmoid(tl.dot(x_norm, tl.trans(w_lg)))
        right_gate = tl.sigmoid(tl.dot(x_norm, tl.trans(w_rg)))
        out_gate = tl.sigmoid(tl.dot(x_norm, tl.trans(w_og)))
        
        # Load mask: mask[b, i, k] where (b, i, k) corresponds to offsets_m
        # mask is [B, N, N], element (b, i, k) is at offset b*N*N + i*N + k
        mask_val = tl.load(mask_ptr + offsets_m)  # [BLOCK_M]
        
        # Apply mask and gates
        left_proj = left_proj * mask_val[:, None] * left_gate
        right_proj = right_proj * mask_val[:, None] * right_gate
        
        # Store to transposed layout
        # left_out[b, h, i, k] where the flat index is b*hidden_dim*N*N + h*N*N + i*N + k
        # = (b*N + i)*N + k mapped to b*hd*N*N + h*N*N + i*N + k
        # offsets_m = b*N*N + i*N + k
        # We need to extract b, i, k from offsets_m
        b_idx = offsets_m // (N * N)
        ik_idx = offsets_m % (N * N)  # = i*N + k
        
        # left_out offset: b_idx * hd * N * N + h * N * N + ik_idx

        tl.store(left_out_ptrs, left_proj)
        
        right_out_ptrs = right_out_ptr + b_idx[:, None] * (hidden_dim * N * N) + offsets_h[None, :] * (N * N) + ik_idx[:, None]
        tl.store(right_out_ptrs, right_proj)

        tl.store(out_gate_ptrs, out_gate)
