```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


def _next_pow2(x: int) -> int:
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


@triton.jit
def _layer_norm_kernel(
    x_ptr,
    xn_ptr,
    w_ptr,
    b_ptr,
    R,
    C,
    eps,
    BM: tl.constexpr,
    BC: tl.constexpr,
):
    """LayerNorm over the last C dimensions, one block of rows per program."""
    pid_b = tl.program_id(0)
    pid_r = tl.program_id(1)

    rows = pid_r * BM + tl.arange(0, BM)
    cols = tl.arange(0, BC)

    row_m = rows < R
    col_m = cols < C
    load_m = row_m[:, None] & col_m[None, :]

    base = pid_b * R * C
    ptrs = x_ptr + base + rows[:, None] * C + cols[None, :]
    x = tl.load(ptrs, mask=load_m, other=0.0).to(tl.float32)

    col_m_f = col_m.to(tl.float32)
    s = tl.sum(x * col_m_f[None, :], axis=1)
    mean = s / C

    centered = x - mean[:, None]
    centered = tl.where(col_m[None, :], centered, 0.0)
    var = tl.sum(centered * centered, axis=1) / C
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + eps)

    w = tl.load(w_ptr + cols, mask=col_m, other=0.0).to(tl.float32)
    b = tl.load(b_ptr + cols, mask=col_m, other=0.0).to(tl.float32)

    y = (x - mean[:, None]) * rstd[:, None] * w[None, :] + b[None, :]
    y = tl.where(col_m[None, :], y, 0.0)

    tl.store(
        xn_ptr + base + rows[:, None] * C + cols[None, :],
        y.to(tl.bfloat16),
        mask=load_m,
    )


@triton.jit
def _proj_fused_kernel(
    xn_ptr,
    mask_ptr,
    wl_ptr,
    wlg_ptr,
    wr_ptr,
    wrg_ptr,
    wo_ptr,
    left_ptr,
    right_ptr,
    gate_ptr,
    R,
    C,
    H,
    BM: tl.constexpr,
    BK: tl.constexpr,
    HB: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """
    Fused projection + gate + mask for the H=power-of-two case (especially H=128).

    For a block of spatial rows, compute:
      left = mask * sigmoid(xn @ W_left_gate^T) * (xn @ W_left^T)
      right = mask * sigmoid(xn @ W_right_gate^T) * (xn @ W_right^T)
      out_gate = sigmoid(xn @ W_out_gate^T)

    Outputs are written transposed as [B, H, R] (feature-major), which is
    feature-major layout ideal for the later batched contraction.
    """
    pid_r = tl.program_id(0)
    pid_b = tl.program_id(1)

    rows = pid_r * BM + tl.arange(0, BM)
    d = tl.arange(0, HB)

    row_m = rows < R
    d_m = d < H

    base = pid_b * R * C
    out_base = pid_b * H * R

    acc_l = tl.zeros((BM, HB), dtype=tl.float32)
    acc_lg = tl.zeros((BM, HB), dtype=tl.float32)
    acc_r = tl.zeros((BM, HB), dtype=tl.float32)
    acc_rg = tl.zeros((BM, HB), dtype=tl.float32)
    acc_og = tl.zeros((BM, HB), dtype=tl.float32)

    w_base_l = wl_ptr
    w_base_lg = wlg_ptr
    w_base_r = wr_ptr
    w_base_rg = wrg_ptr
    w_base_o = wo_ptr

    for k0 in range(0, C, BK):
        k = k0 + tl.arange(0, BK)
        k_m = k < C

        x_mask = row_m[:, None] & k_m[None, :]
        x = tl.load(xn_ptr + base + rows[:, None] * C + k[None, :], mask=x_mask, other=0.0)

        w_mask = k_m[:, None] & d_m[None, :]

        wl = tl.load(w_base_l + k[:, None] * H + d[None, :], mask=w_mask, other=0.0)
        acc_l = tl.dot(x, wl, acc_l)

        wlg = tl.load(w_base_lg + k[:, None] * H + d[None, :], mask=w_mask, other=0.0)
        acc_lg = tl.dot(x, wlg, acc_lg)

        wr = tl.load(w_base_r + k[:, None] * H + d[None, :], mask=w_mask, other=0.0)
        acc_r = tl.dot(x, wr, acc_r)

        wrng = tl.load(w_base_rg + k[:, None] * H + d[None, :], mask=w_mask, other=0.0)
        acc_rg = tl.dot(x, wrng, acc_rg)

        wo = tl.load(w_base_o + k[:, None] * H + d[None, :], mask=w_mask, other=0.0)
        acc_og = tl.dot(x, wo, acc_og)

    left = acc_l * tl.sigmoid(acc_lg)
    right = acc_r * tl.sigmoid(acc_rg)
    gate = tl.sigmoid(acc_og)

    if HAS_MASK:
        m = tl.load(mask_ptr + pid_b * R + rows, mask=row_m, other=0.0).to(tl.float32)
        left = left * m[:, None]
        right = right * m[:, None]

    store_mask = d_m[:, None] & row_m[None, :]
    out_offsets = d[:, None] * R + rows[None, :]

    tl.store(
        left_ptr + out_base + out_offsets,
        left.to(tl.bfloat16),
        mask=store_mask,
    )
    tl.store(
        right_ptr + out_base + out_offsets,
        right.to(tl.bfloat16),
        mask=store_mask,
    )
    tl.store(
        gate_ptr + out_base + out_offsets,
        gate.to(tl.bfloat16),
        mask=store_mask,
    )


@triton.jit
def _core_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    B,
    H,
    N,
    R,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Core contraction:
      out[b, d, i, j] = sum_k left[b, d, i, k] * right[b, d, j, k]

    For each (b,d), this is left[d] @ right[d]^T, computed with blocked GEMM.
    """
    pid = tl.program_id(0)

    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    tiles = num_i * num_j

    batch_id = pid // tiles
    rem = pid % tiles
    pid_i = rem // num_j
    pid_j = rem % num_j

    b = batch_id // H
    d = batch_id % H
    base = (b * H + d) * R

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_m = offs_i[:, None] < N
    j_m = offs_j[:, None] < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    a_base = left_ptr + base + offs_i[:, None] * N
    b_base = right_ptr + base + offs_j[:, None] * N

    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        k_m = k[None, :] < N

        a = tl.load(a_base + k[None, :], mask=i_m & k_m, other=0.0)
        b = tl.load(b_base + k[None, :], mask=j_m & k_m, other=0.0)

        acc = tl.dot(a, tl.trans(b), acc)

    out_ptrs = out_ptr + base + offs_i[:, None] * N + offs_j[None, :]
    store_mask = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(out_ptrs, acc.to(tl.bfloat16), mask=store_mask)


@triton.jit
def _final_kernel(
    out_ptr,
    gate_ptr,
    lnw_ptr,
    lnb_ptr,
    w_ptr,
    y_ptr,
    R,
    H,
    DIM,
    eps,
    BM: tl.constexpr,
    BK: tl.constexpr,
    NB: tl.constexpr,
):
    """
    Fused LayerNorm(input=core_out) + out_gate + final linear.

    Data layout:
      out_ptr/gate_ptr: [B, H, R], i.e. feature-major.
      w_ptr: [H, DIM], transposed final projection.
      y_ptr: [B, R, DIM].
    """
    pid_m = tl.program_id(0)
    pid_b = tl.program_id(1)

    rows = pid_m * BM + tl.arange(0, BM)
    row_m = rows < R
    base = pid_b * H * R

    sum_ = tl.zeros((BM,), dtype=tl.float32)
    sumsq = tl.zeros((BM,), dtype=tl.float32)

    for k0 in range(0, H, BK):
        k = k0 + tl.arange(0, BK)
        k_m = k < H
        load_m = k_m[:, None] & row_m[None, :]
        v = tl.load(out_ptr + base + k[:, None] * R + rows[None, :], mask=load_m, other=0.0).to(tl.float32)
        sum_ += tl.sum(v, axis=0)
        sumsq += tl.sum(v * v, axis=0)

    mean = sum_ / H
    var = sumsq / H - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + eps)

    for n0 in range(0, DIM, NB):
        n = n0 + tl.arange(0, NB)
        n_m = n < DIM
        acc = tl.zeros((BM, NB), dtype=tl.float32)

        for k0 in range(0, H, BK):
            k = k0 + tl.arange(0, BK)
            k_m = k < H
            load_m = k_m[:, None] & row_m[None, :]

            v = tl.load(out_ptr + base + k[:, None] * R + rows[None, :], mask=load_m, other=0.0).to(tl.float32)
            g = tl.load(gate_ptr + base + k[:, None] * R + rows[None, :], mask=load_m, other=0.0).to(tl.float32)

            a = (v - mean[None, :]) * rstd[None, :]

            lw = tl.load(lnw_ptr + k, mask=k_m, other=0.0).to(tl.float32)
            lb = tl.load(lnb_ptr + k, mask=k_m, other=0.0).to(tl.float32)

            a = a * lw[:, None] + lb[None, :]
            a = a * g

            w = tl.load(w_ptr + k[:, None] * DIM + n[None, :], mask=k_m[:, None] & n_m[None, :], other=0.0)
            acc = tl.dot(tl.trans(a.to(tl.bfloat16)), w.to(tl.bfloat16), acc)

        y_ptrs = y_ptr + (pid_b * R + rows[:, None]) * DIM + n[None, :]
        store_mask = (rows[:, None] < R) & (n[None, :] < DIM)
        tl.store(y_ptrs, acc, mask=store_mask)


def custom_kernel(data):
    """
    TriMul forward (outgoing) with fused Triton kernels:
      LN -> fused projections/gates/masks -> feature-major left/right ->
      batched contraction over k -> fused LN+gate+final projection.

    The core Triton path is optimized for hidden_dim=128; for other hidden
    dimensions, projections are computed with PyTorch while all remaining
    fused stages are still Triton.
    """
    # Robust unpacking: expected (input, mask, weights, config),
    # but tolerate (input, weights, config) where mask is absent/None.
    if isinstance(data, (tuple, list)):
        if len(data) == 4:
            input_tensor, mask, weights, config = data
        elif len(data) == 3:
            if isinstance(data[1], dict):
                input_tensor, weights, config = data
                mask = None
            else:
                input_tensor, mask, weights = data
                config = {}
        elif len(data) == 5:
            input_tensor, mask, weights, config = data[0], data[1], data[2], data[3]
        else:
            input_tensor = data[0]
            mask = data[1] if len(data) > 1 else None
            weights = data[2] if len(data) > 2 else {}
            config = data[3] if len(data) > 3 else {}
    else:
        input_tensor, mask, weights, config = data

    if weights is None:
        weights = {}
    if config is None:
        config = {}

    device = input_tensor.device
    B, N1, N2, C = input_tensor.shape
    # The provided TriMul formulation is for square sequence dims.
    N = N1
    R = N * N

    H = int(weights["left_proj.weight"].shape[0])
    DIM = int(weights["to_out.weight"].shape[0])

    eps = 1e-5
    if hasattr(config, "get"):
        eps = float(config.get("eps", 1e-5))

    # Flatten input to [B, R, C].
    x_flat = input_tensor.contiguous().view(B, R, C)

    # Prepare mask.
    HAS_MASK = mask is not None
    mask_flat = None
    if HAS_MASK:
        mask_flat = mask.contiguous()
        if mask_flat.dim() == 2:
            mask_flat = mask_flat.unsqueeze(0).expand(B, N, N).contiguous()
        mask_flat = mask_flat.reshape(B, R)

    # Norm parameters.
    norm_w = weights["norm.weight"].contiguous().float().to(device)
    if "norm.bias" in weights and weights["norm.bias"] is not None:
        norm_b = weights["norm.bias"].contiguous().float().to(device)
    else:
        norm_b = torch.zeros(C, dtype=torch.float32, device=device)

    # 1) LayerNorm input -> bf16 [B, R, C].
    x_norm = torch.empty((B, R, C), dtype=torch.bfloat16, device=device)

    BC = _next_pow2(C)
    BM_LN = max(16, min(128, 16384 // BC))
    grid_ln = (B, triton.cdiv(R, BM_LN))
    _layer_norm_kernel[grid_ln](
        x_flat,
        x_norm,
        norm_w,
        norm_b,
        R,
        C,
        eps,
        BM_LN,
        BC,
        num_warps=8 if BM_LN * BC >= 16384 else 4,
        num_stages=2,
    )

    # 2) Projections / gates / masks.
    left_t = torch.empty((B, H, R), dtype=torch.bfloat16, device=device)
    right_t = torch.empty((B, H, R), dtype=torch.bfloat16, device=device)
    gate_t = torch.empty((B, H, R), dtype=torch.bfloat16, device=device)

    bf16 = torch.bfloat16

    # Optimized fully fused Triton path for common hidden_dim=128.
    if H <= 128 and H == _next_pow2(H):
        HB = H
        Wl = weights["left_proj.weight"].t().contiguous().to(bf16, device=device)
        Wlg = weights["left_gate.weight"].t().contiguous().to(bf16, device=device)
        Wr = weights["right_proj.weight"].t().contiguous().to(bf16, device=device)
        Wrg = weights["right_gate.weight"].t().contiguous().to(bf16, device=device)
        Wo = weights["out_gate.weight"].t().contiguous().to(bf16, device=device)

        BM_P = 32
        BK_P = 64
        grid_p = (triton.cdiv(R, BM_P), B)
        mask_arg = mask_flat if HAS_MASK else x_norm

        _proj_fused_kernel[grid_p](
            x_norm,
            mask_arg,
            Wl,
            Wlg,
            Wr,
            Wrg,
            Wo,
            left_t,
            right_t,
            gate_t,
            R,
            C,
            H,
            BM_P,
            BK_P,
            HB,
            HAS_MASK,
            num_warps=4,
            num_stages=2,
        )
    else:
        # General path: let cuBLAS handle projections, then place into
        # feature-major layout required by the Triton contraction/final kernels.
        xn2 = x_norm
        mask_f = None
        if HAS_MASK:
            mask_f = mask_flat.to(torch.float32).view(B, R, 1)

        def lin_t(key):
            w = weights[key].to(bf16, device=device)
            return F.linear(xn2, w).float()

        lp = lin_t("left_proj.weight")
        lg = torch.sigmoid(lin_t("left_gate.weight"))
        left = lp * lg
        if mask_f is not None:
            left = left * mask_f
        left_t.copy_(left.permute(0, 2, 1).contiguous().to(bf16))

        rp = lin_t("right_proj.weight")
        rg = torch.sigmoid(lin_t("right_gate.weight"))
        right = rp * rg
        if mask_f is not None:
            right = right * mask_f
        right_t.copy_(right.permute(0, 2, 1).contiguous().to(bf16))

        og = torch.sigmoid(lin_t("out_gate.weight")).to(bf16)
        gate_t.copy_(og.permute(0, 2, 1).contiguous())

    # Release A-normalized input before allocating/computing core outputs.
    del x_flat
    del x_norm

    # 3) Core batched contraction: out_t[b,d,i,j] = sum_k left[b,d,i,k]*right[b,d,j,k].
    out_t = torch.empty((B, H, R), dtype=torch.bfloat16, device=device)

    if N <= 128:
        BM_C = 64
        BN_C = 64
    else:
        BM_C = 128
        BN_C = 128
    BK_C = 64

    ni = triton.cdiv(N, BM_C)
    nj = triton.cdiv(N, BN_C)
    grid_core = (B * H * ni * nj,)

    _core_kernel[grid_core](
        left_t,
        right_t,
        out_t,
        B,
        H,
        N,
        R,
        BM_C,
        BN_C,
        BK_C,
        num_warps=8 if BM_C * BN_C >= 16384 else 4,
        num_stages=3,
    )

    # 4) Fused final LN + gate + projection.
    out = torch.empty((B, R, DIM), dtype=torch.float32, device=device)

    to_norm_w = weights["to_out_norm.weight"].contiguous().float().to(device)
    if "to_out_norm.bias" in weights and weights["to_out_norm.bias"] is not None:
        to_norm_b = weights["to_out_norm.bias"].contiguous().float().to(device)
    else:
        to_norm_b = torch.zeros(H, dtype=torch.float32, device=device)

    Wout_t = weights["to_out.weight"].t().contiguous().to(bf16, device=device)

    NB_OUT = 128
    BM_F = 128 if DIM <= 128 else 64
    BK_F = 128 if H <= 128 else 64
    grid_f = (triton.cdiv(R, BM_F), B)

    _final_kernel[grid_f](
        out_t,
        gate_t,
        to_norm_w,
        to_norm_b,
        Wout_t,
        out,
        R,
        H,
        DIM,
        eps,
        BM_F,
        BK_F,
        NB_OUT,
        num_warps=8 if BM_F * NB_OUT >= 16384 else 4,
        num_stages=2,
    )

    return out.view(B, N, N, DIM)
```
