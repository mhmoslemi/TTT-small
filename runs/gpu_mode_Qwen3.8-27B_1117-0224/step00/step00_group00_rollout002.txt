import torch
import triton
import triton.language as tl


def _next_pow2(n: int) -> int:
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


@triton.jit
def _prefix_kernel(
    x_ptr,
    mask_ptr,
    w_l_ptr,
    w_r_ptr,
    w_lg_ptr,
    w_rg_ptr,
    w_og_ptr,
    norm_w_ptr,
    norm_b_ptr,
    L_ptr,
    R_ptr,
    G_ptr,
    M,
    N,
    H,
    C,
    NN,
    BP: tl.constexpr,
    BC: tl.constexpr,
    HB: tl.constexpr,
    HAS_MASK: tl.constexpr,
    EPS: tl.constexpr,
):
    pid_p = tl.program_id(0)
    pid_h = tl.program_id(1)

    p_raw = pid_p * BP + tl.arange(0, BP)
    p_mask = p_raw < M
    p = tl.where(p_mask, p_raw, M - 1)

    b = p // NN
    p_in_b = p - b * NN

    h_raw = pid_h * HB + tl.arange(0, HB)
    h_mask = h_raw < H
    h = tl.where(h_mask, h_raw, H - 1)

    c_raw = tl.arange(0, BC)
    c_mask = c_raw < C
    c = tl.where(c_mask, c_raw, C - 1)

    nw = tl.load(norm_w_ptr + c_raw)
    nb = tl.load(norm_b_ptr + c_raw)

    x_ptrs = x_ptr + p[:, None] * C + c[None, :]
    x = tl.load(x_ptrs, mask=p_mask[:, None] & c_mask[None, :], other=0.0).to(tl.float32)

    mean = tl.sum(x, axis=1) / C
    xc = tl.where(c_mask[None, :], x - mean[:, None], 0.0)
    var = tl.sum(xc * xc, axis=1) / C
    rstd = 1.0 / tl.sqrt(var + EPS)

    x_norm = (x - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]
    x_norm = tl.where(p_mask[:, None] & c_mask[None, :], x_norm, 0.0)
    x_bf = x_norm.to(tl.bfloat16)

    if HAS_MASK:
        m = tl.load(mask_ptr + p, mask=p_mask, other=0.0).to(tl.float32)
    else:
        m = tl.zeros((BP,), dtype=tl.float32) + 1.0

    # Left gate + left projection.
    w = tl.load(w_lg_ptr + h[:, None] * BC + c_raw[None, :], mask=h_mask[:, None], other=0.0)
    lg_acc = tl.dot(x_bf, tl.trans(w))
    lg_gate = 1.0 / (1.0 + tl.exp(-lg_acc))

    w = tl.load(w_l_ptr + h[:, None] * BC + c_raw[None, :], mask=h_mask[:, None], other=0.0)
    left_acc = tl.dot(x_bf, tl.trans(w))
    left = left_acc * lg_gate * m[:, None]
    left_val = tl.trans(left.to(tl.bfloat16))

    l_ptrs = L_ptr + h[:, None] * NN + b[None, :] * (H * NN) + p_in_b[None, :]
    tl.store(l_ptrs, left_val, mask=h_mask[:, None] & p_mask[None, :])

    # Right gate + right projection.
    w = tl.load(w_rg_ptr + h[:, None] * BC + c_raw[None, :], mask=h_mask[:, None], other=0.0)
    rg_acc = tl.dot(x_bf, tl.trans(w))
    rg_gate = 1.0 / (1.0 + tl.exp(-rg_acc))

    w = tl.load(w_r_ptr + h[:, None] * BC + c_raw[None, :], mask=h_mask[:, None], other=0.0)
    right_acc = tl.dot(x_bf, tl.trans(w))
    right = right_acc * rg_gate * m[:, None]
    right_val = tl.trans(right.to(tl.bfloat16))

    r_ptrs = R_ptr + h[:, None] * NN + b[None, :] * (H * NN) + p_in_b[None, :]
    tl.store(r_ptrs, right_val, mask=h_mask[:, None] & p_mask[None, :])

    # Output gate, stored in [B, N_i, H, N_j] order.
    w = tl.load(w_og_ptr + h[:, None] * BC + c_raw[None, :], mask=h_mask[:, None], other=0.0)
    og_acc = tl.dot(x_bf, tl.trans(w))
    og = 1.0 / (1.0 + tl.exp(-og_acc))
    og_val = tl.trans(og.to(tl.bfloat16))

    i = p_in_b // N
    k = p_in_b - i * N
    g_ptrs = G_ptr + b[None, :] * (N * H * N) + (i[None, :] * H + h[:, None]) * N + k[None, :]
    tl.store(g_ptrs, og_val, mask=h_mask[:, None] & p_mask[None, :])


@triton.jit
def _mid_kernel(
    L_ptr,
    R_ptr,
    O_ptr,
    B,
    H,
    N,
    NN,
    BI: tl.constexpr,
    BJ: tl.constexpr,
    BK: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_gh = tl.program_id(2)

    b = pid_gh // H
    h = pid_gh % H

    i_raw = pid_i * BI + tl.arange(0, BI)
    i_mask = i_raw < N
    i = tl.where(i_mask, i_raw, N - 1)

    j_raw = pid_j * BJ + tl.arange(0, BJ)
    j_mask = j_raw < N
    j = tl.where(j_mask, j_raw, N - 1)

    acc = tl.zeros((BI, BJ), dtype=tl.float32)

    for k_off in range(0, N, BK):
        k_raw = k_off + tl.arange(0, BK)
        k_mask = k_raw < N
        k = tl.where(k_mask, k_raw, N - 1)

        l_ptrs = L_ptr + pid_gh * NN + i[:, None] * N + k[None, :]
        l = tl.load(l_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        r_ptrs = R_ptr + pid_gh * NN + j[:, None] * N + k[None, :]
        r = tl.load(r_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0)

        acc = tl.dot(l, tl.trans(r), acc)

    # Output is stored as [B, N_i, H, N_j] so later kernels can use j-contiguous tiles.
    o_ptrs = O_ptr + b * (N * H * N) + i[:, None] * (H * N) + h * N + j[None, :]
    tl.store(o_ptrs, acc.to(tl.bfloat16), mask=i_mask[:, None] & j_mask[None, :])


@triton.jit
def _stats_kernel(
    O_ptr,
    mean_ptr,
    inv_ptr,
    B,
    N,
    H,
    NN,
    BP: tl.constexpr,
    BH: tl.constexpr,
    EPS: tl.constexpr,
):
    pid_p = tl.program_id(0)
    b = tl.program_id(1)

    p_raw = pid_p * BP + tl.arange(0, BP)
    p_mask = p_raw < NN
    p = tl.where(p_mask, p_raw, NN - 1)

    i = p // N
    j = p - i * N
    base = b * (N * H * N)

    sum_v = tl.zeros((BP,), dtype=tl.float32)
    sum_sq = tl.zeros((BP,), dtype=tl.float32)

    for h0 in range(0, H, BH):
        h_raw = h0 + tl.arange(0, BH)
        h_mask = h_raw < H
        h = tl.where(h_mask, h_raw, H - 1)

        offs = base + (i[None, :] * H + h[:, None]) * N + j[None, :]
        vals = tl.load(O_ptr + offs, mask=h_mask[:, None] & p_mask[None, :], other=0.0).to(tl.float32)

        sum_v += tl.sum(vals, axis=0)
        sum_sq += tl.sum(vals * vals, axis=0)

    mean = sum_v / H
    var = sum_sq / H - mean * mean
    var = tl.maximum(var, 0.0)
    inv = 1.0 / tl.sqrt(var + EPS)

    flat = b * NN + p
    tl.store(mean_ptr + flat, mean, mask=p_mask)
    tl.store(inv_ptr + flat, inv, mask=p_mask)


@triton.jit
def _final_kernel(
    O_ptr,
    G_ptr,
    mean_ptr,
    inv_ptr,
    W_ptr,
    Out_ptr,
    B,
    N,
    H,
    C,
    NN,
    BJ: tl.constexpr,
    BH: tl.constexpr,
    CB: tl.constexpr,
):
    pid_j = tl.program_id(0)
    pid_c = tl.program_id(1)
    pid_bi = tl.program_id(2)

    b = pid_bi // N
    i = pid_bi % N

    j_raw = pid_j * BJ + tl.arange(0, BJ)
    j_mask = j_raw < N
    j = tl.where(j_mask, j_raw, N - 1)

    pos = b * NN + i * N + j
    mean = tl.load(mean_ptr + pos, mask=j_mask, other=0.0)
    inv = tl.load(inv_ptr + pos, mask=j_mask, other=0.0)

    c_start = pid_c * CB
    cs_raw = c_start + tl.arange(0, CB)
    c_mask = cs_raw < C
    cs = tl.where(c_mask, cs_raw, C - 1)

    base = b * (N * H * N) + i * (H * N)
    acc = tl.zeros((BJ, CB), dtype=tl.float32)

    for h0 in range(0, H, BH):
        h_raw = h0 + tl.arange(0, BH)
        h_mask = h_raw < H
        h = tl.where(h_mask, h_raw, H - 1)

        ptr = base + h[:, None] * N + j[None, :]

        o = tl.load(O_ptr + ptr, mask=h_mask[:, None] & j_mask[None, :], other=0.0).to(tl.float32)
        g = tl.load(G_ptr + ptr, mask=h_mask[:, None] & j_mask[None, :], other=0.0).to(tl.float32)

        y = (o - mean[None, :]) * inv[None, :] * g
        y_t = tl.trans(y).to(tl.bfloat16)

        w_ptrs = W_ptr + h[:, None] * C + cs[None, :]
        w = tl.load(w_ptrs, mask=h_mask[:, None] & c_mask[None, :], other=0.0)

        acc = tl.dot(y_t, w, acc)

    out_ptrs = Out_ptr + pos[:, None] * C + cs[None, :]
    tl.store(out_ptrs, acc, mask=j_mask[:, None] & c_mask[None, :])


def custom_kernel(data):
    """
    TriMul forward in Triton + PyTorch.

    The implementation:
    1) fuses LayerNorm + five linear projections + gating/masking into one kernel,
    2) computes the N^3 triangle multiplication as batched bfloat16 Triton GEMMs,
    3) computes LayerNorm statistics over the hidden dimension,
    4) fuses the final LayerNorm + output gate + output projection.
    """
    input_tensor, mask, weights, config = data

    B, N, _, C = input_tensor.shape
    H = int(config.get("hidden_dim", 128))
    M = B * N * N
    NN = N * N

    device = input_tensor.device
    x = input_tensor.contiguous()

    has_mask = mask is not None and isinstance(mask, torch.Tensor)
    if has_mask:
        mask = mask.contiguous()
        if mask.dim() == 4:
            mask = mask[..., 0].contiguous()
        mask_t = mask.to(torch.float32)
    else:
        mask_t = x

    eps = float(config.get("eps", 1e-5)) if isinstance(config, dict) else 1e-5

    BC = _next_pow2(C)

    def _pad_1d(t: torch.Tensor) -> torch.Tensor:
        vt = t.float().contiguous()
        if BC > C:
            vp = torch.zeros(BC, device=device, dtype=torch.float32)
            vp[:C] = vt
            return vp
        return vt

    def _pad_proj_w(key: str) -> torch.Tensor:
        wt = weights[key].to(torch.bfloat16).contiguous()
        if BC > C:
            wp = torch.zeros(H, BC, device=device, dtype=torch.bfloat16)
            wp[:, :C] = wt
            return wp
        return wt

    norm_w = _pad_1d(weights["norm.weight"])
    norm_b = _pad_1d(weights["norm.bias"])

    w_l = _pad_proj_w("left_proj.weight")
    w_r = _pad_proj_w("right_proj.weight")
    w_lg = _pad_proj_w("left_gate.weight")
    w_rg = _pad_proj_w("right_gate.weight")
    w_og = _pad_proj_w("out_gate.weight")

    # to_out weight is [C, H] in PyTorch, but final matmul needs [H, C].
    w_to = weights["to_out.weight"].t().contiguous().to(torch.bfloat16)

    L = torch.empty((B, H, N, N), device=device, dtype=torch.bfloat16)
    R = torch.empty((B, H, N, N), device=device, dtype=torch.bfloat16)
    G = torch.empty((B, N, H, N), device=device, dtype=torch.bfloat16)
    O = torch.empty((B, N, H, N), device=device, dtype=torch.bfloat16)

    mean = torch.empty(M, device=device, dtype=torch.float32)
    inv_rstd = torch.empty(M, device=device, dtype=torch.float32)

    # Prefix: LayerNorm + projections + masks + sigmoid gates.
    if C <= 128:
        BP = 64
        HB = 128 if H >= 128 else 64
        p_warps = 8
    else:
        BP = 64
        HB = 64 if H >= 64 else 32
        p_warps = 8

    grid_prefix = (triton.cdiv(M, BP), triton.cdiv(H, HB))
    _prefix_kernel[grid_prefix](
        x,
        mask_t,
        w_l,
        w_r,
        w_lg,
        w_rg,
        w_og,
        norm_w,
        norm_b,
        L,
        R,
        G,
        M,
        N,
        H,
        C,
        NN,
        BP=BP,
        BC=BC,
        HB=HB,
        HAS_MASK=has_mask,
        EPS=eps,
        num_warps=p_warps,
        num_stages=2,
    )

    # Middle: out[b, i, j, h] = sum_k L[b, h, i, k] * R[b, h, j, k]
    if N <= 128:
        BI = 64
        BJ = 64
        mid_warps = 4
    else:
        BI = 128
        BJ = 128
        mid_warps = 8
    BK = 64

    grid_mid = (triton.cdiv(N, BI), triton.cdiv(N, BJ), B * H)
    _mid_kernel[grid_mid](
        L,
        R,
        O,
        B,
        H,
        N,
        NN,
        BI=BI,
        BJ=BJ,
        BK=BK,
        num_warps=mid_warps,
        num_stages=3,
    )

    del L
    del R

    # Statistics over hidden dimension H for the second LayerNorm.
    BP_S = 128
    for cand in (1024, 512, 256, 128):
        if N >= cand:
            BP_S = cand
            break
    BH_S = 32
    s_warps = 8 if BP_S >= 512 else 4

    grid_stats = (triton.cdiv(NN, BP_S), B)
    _stats_kernel[grid_stats](
        O,
        mean,
        inv_rstd,
        B,
        N,
        H,
        NN,
        BP=BP_S,
        BH=BH_S,
        EPS=eps,
        num_warps=s_warps,
        num_stages=2,
    )

    out = torch.empty((B, N, N, C), device=device, dtype=torch.float32)

    # Final: LayerNorm(O) * output_gate @ W_to^T
    if N <= 128:
        BJ = 64
    else:
        BJ = 128
    BH = 64 if H >= 64 else 32
    CB = 128
    f_warps = 8

    grid_final = (triton.cdiv(N, BJ), triton.cdiv(C, CB), B * N)
    _final_kernel[grid_final](
        O,
        G,
        mean,
        inv_rstd,
        w_to,
        out,
        B,
        N,
        H,
        C,
        NN,
        BJ=BJ,
        BH=BH,
        CB=CB,
        num_warps=f_warps,
        num_stages=2,
    )

    return out
