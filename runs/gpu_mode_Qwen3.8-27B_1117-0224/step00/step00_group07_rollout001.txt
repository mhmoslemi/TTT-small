```python
import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def fused_norm_proj_gate_kernel(
    x_ptr, mask_ptr,
    norm_w_ptr, norm_b_ptr,
    left_proj_w_ptr, right_proj_w_ptr,
    left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    B, N, C, D,
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_mb, stride_mi, stride_mj,
    BLOCK_C: tl.constexpr,
    BLOCK_D: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    # Each program handles one (b, i, j) position
    # Computes layer norm, then projections and gates
    
    pid = tl.program_id(0)
    # pid corresponds to (b, i, j)
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N
    
    # Load x[b, i, j, :] -> C values
    offs_c = tl.arange(0, BLOCK_C)
    c_mask = offs_c < C
    x = tl.load(x_ptr + b * stride_xb + i * stride_xi + j * stride_xj + offs_c, mask=c_mask, other=0.0).to(tl.float32)
    
    # LayerNorm
    mean = tl.sum(x) / C
    x_centered = x - mean
    var = tl.sum(x_centered * x_centered) / C
    std = tl.sqrt(var + 1e-5)
    x_normed = x_centered / std
    
    # Load norm weight and bias
    norm_w = tl.load(norm_w_ptr + offs_c, mask=c_mask, other=0.0).to(tl.float32)
    norm_b = tl.load(norm_b_ptr + offs_c, mask=c_mask, other=0.0).to(tl.float32)
    x_normed = x_normed * norm_w + norm_b
    
    # Now compute projections: left_proj, right_proj, left_gate, right_gate, out_gate
    # Each is a [C, D] matmul with x_normed [C]
    # We can do this as a small GEMM
    
    # left_proj: x_normed @ W_left^T where W_left is [D, C]
    # Result: [D]
    # For each output d: sum_c x_normed[c] * W_left[d, c]
    
    # Since D might be up to 768, we need to loop over D in blocks
    # Actually for the projections, D=hidden_dim=128, so BLOCK_D=128 works
    
    acc_left = tl.zeros((BLOCK_D,), dtype=tl.float32)
    acc_right = tl.zeros((BLOCK_D,), dtype=tl.float32)
    acc_left_gate = tl.zeros((BLOCK_D,), dtype=tl.float32)
    acc_right_gate = tl.zeros((BLOCK_D,), dtype=tl.float32)
    acc_out_gate = tl.zeros((BLOCK_D,), dtype=tl.float32)
    
    offs_d = tl.arange(0, BLOCK_D)
    d_mask = offs_d < D
    
    # Load weight matrices and compute
    # left_proj_w is [D, C], so left_proj_w[d, c]
    # left = sum_c x_normed[c] * left_proj_w[d, c]
    
    for d_block in range(0, D, BLOCK_D):
        d_offs = d_block + tl.arange(0, BLOCK_D)
        d_m = d_offs < D
        
        lw = tl.load(left_proj_w_ptr + d_offs[:, None] * C + offs_c[None, :], mask=d_m[:, None] & c_mask[None, :], other=0.0).to(tl.float32)
        rw = tl.load(right_proj_w_ptr + d_offs[:, None] * C + offs_c[None, :], mask=d_m[:, None] & c_mask[None, :], other=0.0).to(tl.float32)
        lgw = tl.load(left_gate_w_ptr + d_offs[:, None] * C + offs_c[None, :], mask=d_m[:, None] & c_mask[None, :], other=0.0).to(tl.float32)
        rgw = tl.load(right_gate_w_ptr + d_offs[:, None] * C + offs_c[None, :], mask=d_m[:, None] & c_mask[None, :], other=0.0).to(tl.float32)
        ogw = tl.load(out_gate_w_ptr + d_offs[:, None] * C + offs_c[None, :], mask=d_m[:, None] & c_mask[None, :], other=0.0).to(tl.float32)
        
        # [BLOCK_D, BLOCK_C] @ [BLOCK_C] -> [BLOCK_D]
        # Use dot product
        acc_left += tl.sum(lw * x_normed[None, :], axis=1)
        acc_right += tl.sum(rw * x_normed[None, :], axis=1)
        acc_left_gate += tl.sum(lgw * x_normed[None, :], axis=1)
        acc_right_gate += tl.sum(rgw * x_normed[None, :], axis=1)
        acc_out_gate += tl.sum(ogw * x_normed[None, :], axis=1)
    
    # Apply sigmoid to gates
    left_gate = tl.sigmoid(acc_left_gate)
    right_gate = tl.sigmoid(acc_right_gate)
    out_gate = tl.sigmoid(acc_out_gate)
    
    # Apply mask
    if HAS_MASK:
        m_val = tl.load(mask_ptr + b * stride_mb + i * stride_mi + j * stride_mj, mask=d_mask, other=0.0).to(tl.float32)
        # Wait, mask is [B, N, N], so mask[b, i, j] is a scalar
        # But we need to apply it to left and right which are [D]
        # So left[d] *= mask[b, i, j] for all d
        m_val = tl.load(mask_ptr + b * stride_mb + i * stride_mi + j * stride_mj).to(tl.float32)
        acc_left = acc_left * m_val
        acc_right = acc_right * m_val
    
    # Apply gates
    acc_left = acc_left * left_gate
    acc_right = acc_right * right_gate
    
    # Store results
    tl.store(left_out_ptr + pid * D + offs_d, acc_left, mask=d_mask)
    tl.store(right_out_ptr + pid * D + offs_d, acc_right, mask=d_mask)
    tl.store(out_gate_out_ptr + pid * D + offs_d, out_gate, mask=d_mask)
```

```python
import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def layernorm_kernel(
    x_ptr, w_ptr, b_ptr, out_ptr,
    N_total,  # total number of rows (B * N * N)
    C,  # channel dim
    BLOCK_C: tl.constexpr,
):
    """Fused LayerNorm kernel. Each program handles one row."""
    pid = tl.program_id(0)
    offs_c = tl.arange(0, BLOCK_C)
    mask_c = offs_c < C
    
    # Load row
    x = tl.load(x_ptr + pid * C + offs_c, mask=mask_c, other=0.0).to(tl.float32)
    
    # Compute mean and variance
    mean = tl.sum(x, axis=0) / C
    x_centered = x - mean
    var = tl.sum(x_centered * x_centered, axis=0) / C
    std = tl.sqrt(var + 1e-5)
    
    # Load weight and bias
    w = tl.load(w_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
    b = tl.load(b_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
    
    # Normalize
    out = x_centered / std * w + b
    
    tl.store(out_ptr + pid * C + offs_c, out, mask=mask_c)


@triton.jit
def elementwise_ops_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr, out_gate_ptr,
    N_total,  # B * N * N
    N,  # seq_len (for mask indexing)
    D,  # hidden_dim
    BLOCK_D: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """Apply mask, sigmoid gates, and element-wise multiply."""
    pid = tl.program_id(0)
    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < D
    
    # Load projections
    left = tl.load(left_ptr + pid * D + offs_d, mask=mask_d, other=0.0).to(tl.float32)
    right = tl.load(right_ptr + pid * D + offs_d, mask=mask_d, other=0.0).to(tl.float32)
    left_gate = tl.load(left_gate_ptr + pid * D + offs_d, mask=mask_d, other=0.0).to(tl.float32)
    right_gate = tl.load(right_gate_ptr + pid * D + offs_d, mask=mask_d, other=0.0).to(tl.float32)
    out_gate = tl.load(out_gate_ptr + pid * D + offs_d, mask=mask_d, other=0.0).to(tl.float32)
    
    # Sigmoid
    left_gate = tl.sigmoid(left_gate)
    right_gate = tl.sigmoid(right_gate)
    out_gate_val = tl.sigmoid(out_gate)
    
    # Apply mask (scalar per position)
    if HAS_MASK:
        # pid = b * N * N + i * N + j
        b = pid // (N * N)
        ij = pid % (N * N)
        i = ij // N
        j = ij % N
        m_val = tl.load(mask_ptr + b * N * N + i * N + j).to(tl.float32)
        left = left * m_val
        right = right * m_val
    
    # Apply gates
    left = left * left_gate
    right = right * right_gate
    
    # Store
    tl.store(left_out_ptr + pid * D + offs_d, left, mask=mask_d)
    tl.store(right_out_ptr + pid * D + offs_d, right, mask=mask_d)
    tl.store(out_gate_ptr + pid * D + offs_d, out_gate_val, mask=mask_d)


@triton.jit
def final_ops_kernel(
    x_ptr, w_ptr, b_ptr, gate_ptr, out_ptr,
    N_total,  # B * N * N
    D,  # hidden_dim
    BLOCK_D: tl.constexpr,
):
    """Fused LayerNorm + gate multiplication."""
    pid = tl.program_id(0)
    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < D
    
    # Load row
    x = tl.load(x_ptr + pid * D + offs_d, mask=mask_d, other=0.0).to(tl.float32)
    
    # Compute mean and variance
    mean = tl.sum(x, axis=0) / D
    x_centered = x - mean
    var = tl.sum(x_centered * x_centered, axis=0) / D
    std = tl.sqrt(var + 1e-5)
    
    # Load weight and bias
```

```python
import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def layernorm_kernel(
    x_ptr, w_ptr, b_ptr, out_ptr,
    C: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    """Fused LayerNorm. Each program handles one row of C elements."""
    pid = tl.program_id(0)
    offs_c = tl.arange(0, BLOCK_C)
    mask_c = offs_c < C
    
    x = tl.load(x_ptr + pid * C + offs_c, mask=mask_c, other=0.0).to(tl.float32)
    
    mean = tl.sum(x, axis=0) / C
    x_centered = x - mean
    var = tl.sum(x_centered * x_centered, axis=0) / C
    rstd = 1.0 / tl.sqrt(var + 1e-5)
    
    w = tl.load(w_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
    b = tl.load(b_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
    
    out = x_centered * rstd * w + b
    tl.store(out_ptr + pid * C + offs_c, out, mask=mask_c)


@triton.jit
def apply_mask_gates_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, out_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    N, D: tl.constexpr,
    BLOCK_D: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """Apply mask and sigmoid gates element-wise."""
    pid = tl.program_id(0)
    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < D
    
    left = tl.load(left_ptr + pid * D + offs_d, mask=mask_d, other=0.0).to(tl.float32)
    right = tl.load(right_ptr + pid * D + offs_d, mask=mask_d, other=0.0).to(tl.float32)
    lg = tl.load(left_gate_ptr + pid * D + offs_d, mask=mask_d, other=0.0).to(tl.float32)
    rg = tl.load(right_gate_ptr + pid * D + offs_d, mask=mask_d, other=0.0).to(tl.float32)
    og = tl.load(out_gate_ptr + pid * D + offs_d, mask=mask_d, other=0.0).to(tl.float32)
    
    lg = tl.sigmoid(lg)
    rg = tl.sigmoid(rg)
    og = tl.sigmoid(og)
    
    if HAS_MASK:
        # mask shape is [B, N, N], indexed by position pid = b*N*N + i*N + j
        m_val = tl.load(mask_ptr + pid).to(tl.float32)
        left = left * m_val
        right = right * m_val
    
    left = left * lg
    right = right * rg
    
    tl.store(left_out_ptr + pid * D + offs_d, left, mask=mask_d)
    tl.store(right_out_ptr + pid * D + offs_d, right, mask=mask_d)
    tl.store(out_gate_out_ptr + pid * D + offs_d, og, mask=mask_d)
```
