```python
"""
Optimized AlphaFold3-style "outgoing" TriMul forward pass.

Fused Triton kernels:
  1. Input LayerNorm + 5 linear projections + sigmoid gates + mask,
     directly writing left / right tensors in [B, H, N, N] layout and
     out_gate in [B, N, N, H] layout.
  2. Final LayerNorm + gate + output linear, reading the core product in
     [B, N, N, H] layout.

The only heavy O(N^3) core is left[b,h] @ right[b,h]^T for all h, executed
as a batched GEMM by cuBLAS through torch.bmm, then transposed once to the
layout desired by the final fused kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _pre_c128_kernel(
    x_ptr,
    mask_ptr,
    norm_w_ptr,
    norm_b_ptr,
    wl_ptr,
    wr_ptr,
    wlg_ptr,
    wrg_ptr,
    wog_ptr,
    lb_ptr,
    rb_ptr,
    og_ptr,
    N,
    BLOCK_M: tl.constexpr,
    USE_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    b = tl.program_id(1)

    M = N * N
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    valid = offs_m < M
    p = tl.where(valid, offs_m, 0)
    i = p // N
    k = p - i * N
    row_local = i * N + k
    row = b * M + row_local

    offs_c = tl.arange(0, 128)
    x_ptrs = x_ptr + row[:, None] * 128 + offs_c[None, :]
    x = tl.load(x_ptrs, mask=valid[:, None], other=0.0).to(tl.float32)

    mean = tl.sum(x, axis=1) / 128.0
    xc = x - mean[:, None]
    var = tl.sum(xc * xc, axis=1) / 128.0
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    nw = tl.load(norm_w_ptr + offs_c).to(tl.float32)
    nb = tl.load(norm_b_ptr + offs_c).to(tl.float32)
    xa = (xc * rstd[:, None] * nw[None, :] + nb[None, :]).to(tl.bfloat16)

    if USE_MASK:
        m = tl.load(mask_ptr + row, mask=valid, other=0.0).to(tl.float32)
    else:
        m = tl.zeros((BLOCK_M,), dtype=tl.float32) + 1.0

    offs_h = tl.arange(0, 128)
    # Weight tensors are [H, C] = [128, 128].  For tl.dot we need [C, H].
    offw = offs_h[None, :] * 128 + offs_c[:, None]

    # left = mask * left_proj(xn) * sigmoid(left_gate(xn))
    wg = tl.load(wlg_ptr + offw).to(tl.bfloat16)
    y_lg = tl.dot(xa, wg).to(tl.float32)
    y_lg = 1.0 / (1.0 + tl.exp(-y_lg))

    wl = tl.load(wl_ptr + offw).to(tl.bfloat16)
    y_l = tl.dot(xa, wl).to(tl.float32)
    left = y_l * m[:, None] * y_lg
    left_bf = left.to(tl.bfloat16)

    # Store [H, BLOCK_M] so each program writes contiguous p blocks for fixed h.
    lb_off = offs_h[:, None] * M + row_local[None, :]
    tl.store(lb_ptr + b * 128 * M + lb_off, tl.trans(left_bf), mask=valid[None, :])

    # right = mask * right_proj(xn) * sigmoid(right_gate(xn))
    wg = tl.load(wrg_ptr + offw).to(tl.bfloat16)
    y_rg = tl.dot(xa, wg).to(tl.float32)
    y_rg = 1.0 / (1.0 + tl.exp(-y_rg))

    wr = tl.load(wr_ptr + offw).to(tl.bfloat16)
    y_r = tl.dot(xa, wr).to(tl.float32)
    right = y_r * m[:, None] * y_rg
    right_bf = right.to(tl.bfloat16)

    rb_off = offs_h[:, None] * M + row_local[None, :]
    tl.store(rb_ptr + b * 128 * M + rb_off, tl.trans(right_bf), mask=valid[None, :])

    # out_gate = sigmoid(out_gate(xn)), stored as [B, N, N, H]
    wog = tl.load(wog_ptr + offw).to(tl.bfloat16)
    y_og = tl.dot(xa, wog).to(tl.float32)
    y_og = 1.0 / (1.0 + tl.exp(-y_og))

    og_off = row[:, None] * 128 + offs_h[None, :]
    tl.store(og_ptr + og_off, y_og.to(tl.bfloat16), mask=valid[:, None])


@triton.jit
def _pre_cbig_kernel(
    x_ptr,
    mask_ptr,
    norm_w_ptr,
    norm_b_ptr,
    wl_ptr,
    wr_ptr,
    wlg_ptr,
    wrg_ptr,
    wog_ptr,
    lb_ptr,
    rb_ptr,
    og_ptr,
    N,
    C,
    BLOCK_M: tl.constexpr,
    USE_MASK: tl.constexpr,
):
    """
    Fused pre-kernel for C being a multiple of 128 (e.g. 384) and H = 128.
    """
    pid = tl.program_id(0)
    b = tl.program_id(1)

    M = N * N
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    valid = offs_m < M
    p = tl.where(valid, offs_m, 0)
    i = p // N
    k = p - i * N
    row_local = i * N + k
    row = b * M + row_local
    offs_c128 = tl.arange(0, 128)
    offs_h = tl.arange(0, 128)

    # Stable one-pass Welford LayerNorm statistics over C.
    mean = tl.zeros((BLOCK_M,), dtype=tl.float32)
    m2 = tl.zeros((BLOCK_M,), dtype=tl.float32)
    n = 0.0

    for c0 in range(0, C, 128):
        cc = c0 + offs_c128
        x_ptrs = x_ptr + row[:, None] * C + cc[None, :]
        x_chunk = tl.load(x_ptrs, mask=valid[:, None], other=0.0).to(tl.float32)

        n = n + 128.0
        inv_n = 1.0 / n
        delta = x_chunk - mean[:, None]
        mean = mean + delta * inv_n
        delta2 = x_chunk - mean[:, None]
        m2 += delta * delta2

    var = m2 / C
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    acc_l = tl.zeros((BLOCK_M, 128), dtype=tl.float32)
    acc_r = tl.zeros((BLOCK_M, 128), dtype=tl.float32)
    acc_lg = tl.zeros((BLOCK_M, 128), dtype=tl.float32)
    acc_rg = tl.zeros((BLOCK_M, 128), dtype=tl.float32)
    acc_og = tl.zeros((BLOCK_M, 128), dtype=tl.float32)

    for c0 in range(0, C, 128):
        cc = c0 + offs_c128
        x_ptrs = x_ptr + row[:, None] * C + cc[None, :]
        x_chunk = tl.load(x_ptrs, mask=valid[:, None], other=0.0).to(tl.float32)

        nw = tl.load(norm_w_ptr + cc).to(tl.float32)
        nb = tl.load(norm_b_ptr + cc).to(tl.float32)
        xn = (x_chunk - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]
        xa = xn.to(tl.bfloat16)

        # Weights are [H, C]; need [C_chunk, H] chunks for tl.dot.
        offw = offs_h[None, :] * C + cc[:, None]

        wl_c = tl.load(wl_ptr + offw).to(tl.bfloat16)
        acc_l = tl.dot(xa, wl_c, acc_l)

        wr_c = tl.load(wr_ptr + offw).to(tl.bfloat16)
        acc_r = tl.dot(xa, wr_c, acc_r)

        wlg_c = tl.load(wlg_ptr + offw).to(tl.bfloat16)
        acc_lg = tl.dot(xa, wlg_c, acc_lg)

        wrg_c = tl.load(wrg_ptr + offw).to(tl.bfloat16)
        acc_rg = tl.dot(xa, wrg_c, acc_rg)

        wog_c = tl.load(wog_ptr + offw).to(tl.bfloat16)
        acc_og = tl.dot(xa, wog_c, acc_og)

    y_lg = acc_lg
    y_lg = 1.0 / (1.0 + tl.exp(-y_lg))
    if USE_MASK:
        m = tl.load(mask_ptr + row, mask=valid, other=0.0).to(tl.float32)
    else:
        m = tl.zeros((BLOCK_M,), dtype=tl.float32) + 1.0

    left = acc_l * m[:, None] * y_lg
    left_bf = left.to(tl.bfloat16)
    lb_off = offs_h[:, None] * M + row_local[None, :]
    tl.store(lb_ptr + b * 128 * M + lb_off, tl.trans(left_bf), mask=valid[None, :])

    y_rg = acc_rg
    y_rg = 1.0 / (1.0 + tl.exp(-y_rg))
    right = acc_r * m[:, None] * y_rg
    right_bf = right.to(tl.bfloat16)
    rb_off = offs_h[:, None] * M + row_local[None, :]
    tl.store(rb_ptr + b * 128 * M + rb_off, tl.trans(right_bf), mask=valid[None, :])

    y_og = acc_og
    y_og = 1.0 / (1.0 + tl.exp(-y_og))
    og_off = row[:, None] * 128 + offs_h[None, :]
    tl.store(og_ptr + og_off, y_og.to(tl.bfloat16), mask=valid[:, None])


@triton.jit
def _final_kernel(
    k_ptr,
    og_ptr,
    w_ptr,
    norm_w_ptr,
    norm_b_ptr,
    out_ptr,
    N,
    C,
    H: tl.constexpr,
    BLOCK_P: tl.constexpr,
    H_CHUNK: tl.constexpr,
    C_BLOCK: tl.constexpr,
    SINGLE_H: tl.constexpr,
):
    pid_p = tl.program_id(0)
    b = tl.program_id(1)
    pid_c = tl.program_id(2)

    M = N * N
    offs_p = pid_p * BLOCK_P + tl.arange(0, BLOCK_P)
    valid = offs_p < M
    p = tl.where(valid, offs_p, 0)
    base_row = b * M + p

    offs_c = pid_c * C_BLOCK + tl.arange(0, C_BLOCK)
    c_valid = offs_c < C

    if SINGLE_H:
        offs_h = tl.arange(0, H)
        ptr = base_row[:, None] * H + offs_h[None, :]

        k = tl.load(k_ptr + ptr, mask=valid[:, None], other=0.0).to(tl.float32)
        g = tl.load(og_ptr + ptr, mask=valid[:, None], other=0.0).to(tl.float32)

        mean = tl.sum(k, axis=1) / H
        kc = k - mean[:, None]
        var = tl.sum(kc * kc, axis=1) / H
        var = tl.maximum(var, 0.0)
        rstd = 1.0 / tl.sqrt(var + 1e-5)

        nw = tl.load(norm_w_ptr + offs_h).to(tl.float32)
        nb = tl.load(norm_b_ptr + offs_h).to(tl.float32)

        normed = kc * rstd[:, None] * nw[None, :] + nb[None, :]
        feat = (normed * g).to(tl.bfloat16)

        # w is [C, H].  Need [H, C_BLOCK].
        offw = offs_c[None, :] * H + offs_h[:, None]
        w = tl.load(w_ptr + offw, mask=c_valid[None, :], other=0.0).to(tl.bfloat16)

        acc = tl.dot(feat, w)
        out_off = base_row[:, None] * C + offs_c[None, :]
        tl.store(out_ptr + out_off, acc.to(tl.float32), mask=valid[:, None] & c_valid[None, :])
    else:
        sum_k = tl.zeros((BLOCK_P,), dtype=tl.float32)
        sum_sq = tl.zeros((BLOCK_P,), dtype=tl.float32)

        for h0 in range(0, H, H_CHUNK):
            offs_h = h0 + tl.arange(0, H_CHUNK)
            h_ok = (offs_h < H)[None, :]
            ptr = base_row[:, None] * H + offs_h[None, :]
            k = tl.load(k_ptr + ptr, mask=valid[:, None] & h_ok, other=0.0).to(tl.float32)
            sum_k += tl.sum(k, axis=1)
            sum_sq += tl.sum(k * k, axis=1)

        mean = sum_k / H
        var = sum_sq / H - mean * mean
        var = tl.maximum(var, 0.0)
        rstd = 1.0 / tl.sqrt(var + 1e-5)

        acc = tl.zeros((BLOCK_P, C_BLOCK), dtype=tl.float32)

        for h0 in range(0, H, H_CHUNK):
            offs_h = h0 + tl.arange(0, H_CHUNK)
            h_ok = offs_h < H
            h_ok2d = h_ok[None, :]
            ptr = base_row[:, None] * H + offs_h[None, :]

            k = tl.load(k_ptr + ptr, mask=valid[:, None] & h_ok2d, other=0.0).to(tl.float32)
            g = tl.load(og_ptr + ptr, mask=valid[:, None] & h_ok2d, other=0.0).to(tl.float32)

            nw = tl.load(norm_w_ptr + offs_h, mask=h_ok, other=0.0).to(tl.float32)
            nb = tl.load(norm_b_ptr + offs_h, mask=h_ok, other=0.0).to(tl.float32)

            normed = (k - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]
            feat = (normed * g).to(tl.bfloat16)

            offw = offs_c[None, :] * H + offs_h[:, None]
            w_mask = h_ok[:, None] & c_valid[None, :]
            w = tl.load(w_ptr + offw, mask=w_mask, other=0.0).to(tl.bfloat16)

            acc = tl.dot(feat, w, acc)

        out_off = base_row[:, None] * C + offs_c[None, :]
        tl.store(out_ptr + out_off, acc.to(tl.float32), mask=valid[:, None] & c_valid[None, :])


def _pre_pytorch(x, mask, weights, B, N, C, H, has_mask, device):
    """
    Fallback pre-kernel in PyTorch.  Used when the shape is not covered by
    the specialized Triton pre-kernels.
    """
    bf16 = torch.bfloat16

    x32 = x.to(torch.float32)
    norm_w = weights["norm.weight"].to(torch.float32, device=device).contiguous()
    norm_b = weights["norm.bias"].to(torch.float32, device=device).contiguous()
    x_norm = F.layer_norm(x32, (C,), norm_w, norm_b).to(bf16)

    wl = weights["left_proj.weight"].to(device).contiguous()
    wr = weights["right_proj.weight"].to(device).contiguous()
    wlg = weights["left_gate.weight"].to(device).contiguous()
    wrg = weights["right_gate.weight"].to(device).contiguous()
    wog = weights["out_gate.weight"].to(device).contiguous()

    w_cat = torch.cat([wl, wr, wlg, wrg, wog], dim=0).to(bf16, device=device)
    proj = torch.matmul(x_norm, w_cat.t())

    left = proj[..., 0:H]
    right = proj[..., H:2*H]
    lg = torch.sigmoid(proj[..., 2*H:3*H])
    rg = torch.sigmoid(proj[..., 3*H:4*H])
    og = torch.sigmoid(proj[..., 4*H:5*H]).to(bf16).contiguous()

    if has_mask:
        m = mask.to(bf16, device=device).unsqueeze(-1)
        left = left * lg * m
        right = right * rg * m
    else:
        left = left * lg
        right = right * rg

    left_b = left.permute(0, 3, 1, 2).to(bf16, device=device).contiguous()
    right_b = right.permute(0, 3, 1, 2).to(bf16, device=device).contiguous()
    return left_b, right_b, og


def custom_kernel(data):
    """
    Args:
        data: (input, mask, weights, config) or (input, weights, config).
          input: [B, N, N, C]
          mask:  [B, N, N] or None
          weights: dict of TriMul weights
          config: dict containing at least hidden_dim and optional nomask.

    Returns:
        output: [B, N, N, C] float32
    """
    if len(data) == 4:
        x, mask, weights, config = data
    elif len(data) == 3:
        x, weights, config = data
        mask = None
    else:
        raise ValueError("custom_kernel expects data of length 3 or 4")

    config = config or {}
    device = x.device

    if mask is not None:
        mask = mask.to(device).contiguous()
        if mask.dim() == 4:
            mask = mask.squeeze(-1)
        if mask.dim() == 2 and x.dim() >= 3 and x.shape[0] == 1:
            mask = mask.unsqueeze(0)
        mask = mask.to(torch.float32, device=device).contiguous()

    x = x.to(device).contiguous()
    B, N, _, C = x.shape
    hidden = int(config.get("hidden_dim", 128))
    has_mask = (mask is not None) and (not bool(config.get("nomask", False)))

    def _w(name):
        return weights[name].to(device).contiguous()

    norm_w = _w("norm.weight")
    norm_b = _w("norm.bias")
    wl = _w("left_proj.weight")
    wr = _w("right_proj.weight")
    wlg = _w("left_gate.weight")
    wrg = _w("right_gate.weight")
    wog = _w("out_gate.weight")
    to_out_norm_w = _w("to_out_norm.weight")
    to_out_norm_b = _w("to_out_norm.bias")
    to_out_w = _w("to_out.weight")

    bf16 = torch.bfloat16
    use_fused_pre = (hidden == 128 and C % 128 == 0 and C <= 1024)

    if use_fused_pre:
        left_b = torch.empty((B, hidden, N, N), device=device, dtype=bf16)
        right_b = torch.empty((B, hidden, N, N), device=device, dtype=bf16)
        og = torch.empty((B, N, N, hidden), device=device, dtype=bf16)

        if C == 128:
            block_m = 64
            num_warps = 8
        else:
            block_m = 32 if C <= 384 else 16
            num_warps = 8

        grid = (triton.cdiv(N * N, block_m), B)
        mask_ptr = mask if has_mask else x

        if C == 128:
            _pre_c128_kernel[grid](
                x,
                mask_ptr,
                norm_w,
                norm_b,
                wl,
                wr,
                wlg,
                wrg,
                wog,
                left_b,
                right_b,
                og,
                N,
                BLOCK_M=block_m,
                USE_MASK=has_mask,
                num_warps=num_warps,
                num_stages=2,
            )
        else:
            _pre_cbig_kernel[grid](
                x,
                mask_ptr,
                norm_w,
                norm_b,
                wl,
                wr,
                wlg,
                wrg,
                wog,
                left_b,
                right_b,
                og,
                N,
                C,
                BLOCK_M=block_m,
                USE_MASK=has_mask,
                num_warps=num_warps,
                num_stages=2,
            )
    else:
        left_b, right_b, og = _pre_pytorch(
            x, mask, weights, B, N, C, hidden, has_mask, device
        )

    # Core TriMul product:
    #   K[b,h,i,j] = sum_k L[b,h,i,k] * R[b,h,j,k]
    batched = B * hidden
    left_bh = left_b.reshape(batched, N, N)
    right_bh = right_b.reshape(batched, N, N)
    k = torch.empty_like(left_b)
    k_bh = k.reshape(batched, N, N)

    torch.bmm(left_bh, right_bh.transpose(1, 2), out=k_bh)

    del left_b, right_b, left_bh, right_bh

    # Final fused kernel prefers [B, N, N, H].
    k_t = k.permute(0, 2, 3, 1).contiguous()
    del k, k_bh

    out = torch.empty((B, N, N, C), device=device, dtype=torch.float32)
    to_out_w_bf = to_out_w.to(bf16, device=device).contiguous()

    if hidden % 128 == 0:
        h_chunk = 128
    else:
        h_chunk = 64 if hidden >= 64 else max(triton.next_power_of_2(hidden), 16)

    if C % 128 == 0:
        c_block = 128
    elif C % 64 == 0:
        c_block = 64
    else:
        c_block = max(triton.next_power_of_2(C), 16)

    single_h = (hidden == h_chunk and (hidden & (hidden - 1)) == 0)
    block_p = 64

    grid = (triton.cdiv(N * N, block_p), B, triton.cdiv(C, c_block))
    _final_kernel[grid](
        k_t,
        og,
        to_out_w_bf,
        to_out_norm_w,
        to_out_norm_b,
        out,
        N,
        C,
        H=hidden,
        BLOCK_P=block_p,
        H_CHUNK=h_chunk,
        C_BLOCK=c_block,
        SINGLE_H=single_h,
        num_warps=8,
        num_stages=2,
    )

    return out
```
