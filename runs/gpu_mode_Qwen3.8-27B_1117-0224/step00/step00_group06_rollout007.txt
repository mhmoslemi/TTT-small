@triton.jit
def layernorm_to_bf16_kernel(
    x_ptr, w_ptr, b_ptr, y_ptr,
    M, C: tl.constexpr, C_BLOCK: tl.constexpr,
    BLOCK_R: tl.constexpr, EPS: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_r = pid * BLOCK_R + tl.arange(0, BLOCK_R)
    offs_c = tl.arange(0, C_BLOCK)
    c_valid = offs_c < C
    r_valid = offs_r < M
    r_idx = tl.where(r_valid, offs_r, 0)
    c_idx = tl.where(c_valid, offs_c, 0)
    ptrs = r_idx[:, None] * C + c_idx[None, :]
    x = tl.load(x_ptr + ptrs, mask=r_valid[:, None] & c_valid[None, :], other=0.0).to(tl.float32)
    # x invalid c zero
    sum_x = tl.sum(x, axis=1)
    mean = sum_x / C
    shifted = x - mean[:, None]
    shifted = tl.where(c_valid[None, :], shifted, 0.0)
    var = tl.sum(shifted * shifted, axis=1) / C
    inv_std = 1.0 / tl.sqrt(var + EPS)
    w = tl.load(w_ptr + c_idx, mask=c_valid, other=0.0).to(tl.float32)
    b = tl.load(b_ptr + c_idx, mask=c_valid, other=0.0).to(tl.float32)
    y = shifted * inv_std[:, None] * w[None, :] + b[None, :]
    y = tl.where(c_valid[None, :], y, 0.0)
    tl.store(y_ptr + ptrs, y.to(y_ptr.dtype.element_ty), mask=r_valid[:, None] & c_valid[None, :])
```

@triton.jit
def contraction_kernel(l_ptr, r_ptr, o_ptr, N,
                       H: tl.constexpr,
                       BI: tl.constexpr, BJ: tl.constexpr, BD: tl.constexpr,
                       NUM_D: tl.constexpr,
                       ):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)
    b = pid_bd // NUM_D
    pid_d = pid_bd % NUM_D

    offs_i = pid_i * BI + tl.arange(0, BI)
    offs_j = pid_j * BJ + tl.arange(0, BJ)
    offs_d = tl.arange(0, BD)

    i_valid = offs_i < N
    j_valid = offs_j < N
    d_valid = (pid_d * BD + offs_d) < H

    i_idx = tl.where(i_valid, offs_i, 0)
    j_idx = tl.where(j_valid, offs_j, 0)
    d_idx = tl.where(d_valid, pid_d * BD + offs_d, 0)

    base_i = b * N * N + i_idx * N
    base_j = b * N * N + j_idx * N

    # initial pointers for k=0
    l_ptrs = l_ptr + (base_i[:, None] + 0) * H + d_idx[None, :]
    r_ptrs = r_ptr + (base_j[:, None] + 0) * H + d_idx[None, :]

    l_mask = i_valid[:, None] & d_valid[None, :]
    r_mask = j_valid[:, None] & d_valid[None, :]

    acc = tl.zeros((BI, BJ), dtype=tl.float32)
    for k in range(0, N):
       l = tl.load(l_ptrs, mask=l_mask, other=0.0)
       r = tl.load(r_ptrs, mask=r_mask, other=0.0)
       acc += tl.dot(l, tl.trans(r))
       l_ptrs += H
       r_ptrs += H
    # store

import torch
import triton
import triton.language as tl


"""
Fused TriMul forward (outgoing) for AlphaFold3-style pair tensors.

Algorithm:
1. Triton layer-norm over the input channel dimension, casting to bf16.
2. One cuBLAS bf16 GEMM computes all five linear projections/gates.
3. Triton kernels apply sigmoid gates and masks, producing L, R, and out_gate.
4. A custom Triton kernel computes the N^3H contraction:
       out[b,i,j,d] = sum_k L[b,i,k,d] * R[b,j,k,d]
   using tl.dot over hidden chunks.
5. A final Triton kernel fuses LayerNorm(R), multiplication by out_gate,
   and the final hidden_dim -> dim projection, returning float32.
"""

_EPS = 1e-5
_WEIGHT_CACHE = {}
_PROJ_KEYS = (
    "left_proj.weight",
    "right_proj.weight",
    "left_gate.weight",
    "right_gate.weight",
    "out_gate.weight",
)


def _next_pow2(x: int) -> int:
    p = 1
    x = int(x)
    while p < x:
        p <<= 1
    return p


def _get_blas_weights(weights: dict, device: torch.device):
    """Cache converted bf16 GEMM weights to avoid repeated small kernel launches."""
    key_parts = [str(device)]
    for k in _PROJ_KEYS + ("to_out.weight",):
        t = weights[k]
        key_parts.append(t.data_ptr())
        key_parts.append(tuple(t.shape))
        key_parts.append(str(t.dtype))
        key_parts.append(int(getattr(t, "_version", 0)))

    key = tuple(key_parts)
    cached = _WEIGHT_CACHE.get(key)
    if cached is not None:
        return cached

    bf = torch.bfloat16
    all_t_parts = []
    for k in _PROJ_KEYS:
        all_t_parts.append(weights[k].to(bf, non_blocking=True).T)

    # [dim, 5 * hidden_dim], row-major.
    W_all_T = torch.cat(all_t_parts, dim=1).contiguous()

    # [dim, hidden_dim], row-major. The final fused kernel loads it as
    # [hidden_dim, dim] via pointer arithmetic.
    W_to_out = weights["to_out.weight"].to(bf, non_blocking=True).contiguous()

    value = (W_all_T, W_to_out)
    _WEIGHT_CACHE[key] = value
    return value


@triton.jit
def layernorm_to_bf16_kernel(
    x_ptr,
    w_ptr,
    b_ptr,
    y_ptr,
    M,
    C: tl.constexpr,
    C_BLOCK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    EPS: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_r = pid * BLOCK_R + tl.arange(0, BLOCK_R)
    offs_c = tl.arange(0, C_BLOCK)

    r_valid = offs_r < M
    c_valid = offs_c < C
    r_idx = tl.where(r_valid, offs_r, 0)
    c_idx = tl.where(c_valid, offs_c, 0)

    ptrs = r_idx[:, None] * C + c_idx[None, :]
    mask = r_valid[:, None] & c_valid[None, :]

    x = tl.load(x_ptr + ptrs, mask=mask, other=0.0).to(tl.float32)

    mean = tl.sum(x, axis=1) / C
    shifted = x - mean[:, None]
    shifted = tl.where(c_valid[None, :], shifted, 0.0)

    var = tl.sum(shifted * shifted, axis=1) / C
    inv_std = 1.0 / tl.sqrt(var + EPS)

    w = tl.load(w_ptr + c_idx, mask=c_valid, other=0.0).to(tl.float32)
    b = tl.load(b_ptr + c_idx, mask=c_valid, other=0.0).to(tl.float32)

    y = shifted * inv_std[:, None] * w[None, :] + b[None, :]
    y = tl.where(c_valid[None, :], y, 0.0)

    tl.store(y_ptr + ptrs, y.to(y_ptr.dtype.element_ty), mask=mask)


@triton.jit
def apply_left_right_kernel(
    all_ptr,
    left_ptr,
    right_ptr,
    mask_ptr,
    M,
    GH: tl.constexpr,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = tl.arange(0, H)

    m_valid = offs_m < M
    h_valid = offs_h < H
    m_idx = tl.where(m_valid, offs_m, 0)

    mask2 = m_valid[:, None] & h_valid[None, :]
    base = m_idx[:, None] * GH + offs_h[None, :]

    l = tl.load(all_ptr + base, mask=mask2, other=0.0)
    lg = tl.load(all_ptr + base + 2 * H, mask=mask2, other=0.0)
    r = tl.load(all_ptr + base + H, mask=mask2, other=0.0)
    rg = tl.load(all_ptr + base + 3 * H, mask=mask2, other=0.0)

    l = l.to(tl.float32) * tl.sigmoid(lg.to(tl.float32))
    r = r.to(tl.float32) * tl.sigmoid(rg.to(tl.float32))

    if HAS_MASK:
        mm = tl.load(mask_ptr + m_idx, mask=m_valid, other=0.0).to(tl.float32)
        l = l * mm[:, None]
        r = r * mm[:, None]

    out_offsets = m_idx[:, None] * H + offs_h[None, :]
    tl.store(left_ptr + out_offsets, l.to(left_ptr.dtype.element_ty), mask=mask2)
    tl.store(right_ptr + out_offsets, r.to(right_ptr.dtype.element_ty), mask=mask2)


@triton.jit
def apply_out_gate_kernel(
    all_ptr,
    out_ptr,
    M,
    GH: tl.constexpr,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = tl.arange(0, H)

    m_valid = offs_m < M
    h_valid = offs_h < H
    m_idx = tl.where(m_valid, offs_m, 0)
    mask2 = m_valid[:, None] & h_valid[None, :]

    base = m_idx[:, None] * GH + 4 * H + offs_h[None, :]
    x = tl.load(all_ptr + base, mask=mask2, other=0.0)
    y = tl.sigmoid(x.to(tl.float32)).to(out_ptr.dtype.element_ty)

    out_offsets = m_idx[:, None] * H + offs_h[None, :]
    tl.store(out_ptr + out_offsets, y, mask=mask2)


@triton.jit
def contraction_kernel(
    l_ptr,
    r_ptr,
    o_ptr,
    N,
    H: tl.constexpr,
    BI: tl.constexpr,
    BJ: tl.constexpr,
    BD: tl.constexpr,
    NUM_D: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)

    b = pid_bd // NUM_D
    pid_d = pid_bd % NUM_D

    offs_i = pid_i * BI + tl.arange(0, BI)
    offs_j = pid_j * BJ + tl.arange(0, BJ)
    offs_d = tl.arange(0, BD)

    i_valid = offs_i < N
    j_valid = offs_j < N
    d_val = pid_d * BD + offs_d
    d_valid = d_val < H

    i_idx = tl.where(i_valid, offs_i, 0)
    j_idx = tl.where(j_valid, offs_j, 0)
    d_idx = tl.where(d_valid, d_val, 0)

    base_i = b * N * N + i_idx * N
    base_j = b * N * N + j_idx * N

    l_ptrs = l_ptr + (base_i[:, None] + 0) * H + d_idx[None, :]
    r_ptrs = r_ptr + (base_j[:, None] + 0) * H + d_idx[None, :]

    l_mask = i_valid[:, None] & d_valid[None, :]
    r_mask = j_valid[:, None] & d_valid[None, :]

    acc = tl.zeros((BI, BJ), dtype=tl.float32)

    for k in range(0, N):
        l = tl.load(l_ptrs, mask=l_mask, other=0.0)
        r = tl.load(r_ptrs, mask=r_mask, other=0.0)
        acc += tl.dot(l, tl.trans(r)).to(tl.float32)
        l_ptrs += H
        r_ptrs += H

    out_base = b * N * N + i_idx[:, None] * N + j_idx[None, :]
    o_ptrs = o_ptr + out_base[:, :, None] * H + d_idx[None, None, :]
    o_mask = (
        i_valid[:, None, None]
        & j_valid[None, :, None]
        & d_valid[None, None, :]
    )
    tl.store(o_ptrs, acc.to(o_ptr.dtype.element_ty), mask=o_mask)


@triton.jit
def fused_ln_gate_to_out_kernel(
    x_ptr,
    g_ptr,
    ln_w_ptr,
    ln_b_ptr,
    w_ptr,
    y_ptr,
    M,
    C: tl.constexpr,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
    NUM_C: tl.constexpr,
    EPS: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = tl.arange(0, H)

    m_valid = offs_m < M
    h_valid = offs_h < H
    m_idx = tl.where(m_valid, offs_m, 0)

    mask2 = m_valid[:, None] & h_valid[None, :]
    base = m_idx[:, None] * H + offs_h[None, :]

    x = tl.load(x_ptr + base, mask=mask2, other=0.0).to(tl.float32)
    g = tl.load(g_ptr + base, mask=mask2, other=0.0).to(tl.float32)

    mean = tl.sum(x, axis=1) / H
    shifted = x - mean[:, None]
    shifted = tl.where(h_valid[None, :], shifted, 0.0)
    var = tl.sum(shifted * shifted, axis=1) / H
    inv_std = 1.0 / tl.sqrt(var + EPS)

    lw = tl.load(ln_w_ptr + offs_h, mask=h_valid, other=0.0).to(tl.float32)
    lb = tl.load(ln_b_ptr + offs_h, mask=h_valid, other=0.0).to(tl.float32)

    a = shifted * inv_std[:, None] * lw[None, :] + lb[None, :]
    a = a * g[None, :]
    a = tl.where(h_valid[None, :], a, 0.0)
    a_in = a.to(x_ptr.dtype.element_ty)

    for c_block in range(0, NUM_C):
        offs_c = c_block * BLOCK_C + tl.arange(0, BLOCK_C)
        c_valid = offs_c < C
        c_idx = tl.where(c_valid, offs_c, 0)

        w_ptrs = w_ptr + c_idx[None, :] * H + offs_h[:, None]
        w_mask = h_valid[:, None] & c_valid[None, :]
        w = tl.load(w_ptrs, mask=w_mask, other=0.0)

        acc = tl.dot(a_in, w).to(tl.float32)

        y_ptrs = y_ptr + m_idx[:, None] * C + c_idx[None, :]
        y_mask = m_valid[:, None] & c_valid[None, :]
        tl.store(y_ptrs, acc.to(y_ptr.dtype.element_ty), mask=y_mask)


def custom_kernel(data):
    """TriMul forward entrypoint."""
    if len(data) == 4:
        input_tensor, mask, weights, config = data
    else:
        input_tensor, weights, config = data
        mask = None

    B, N, _, C = input_tensor.shape
    H = config.get("hidden_dim", None)
    if H is None:
        H = int(weights["left_proj.weight"].shape[0])
    H = int(H)
    C = int(C)
    M = B * N * N
    device = input_tensor.device

    if not input_tensor.is_contiguous():
        input_tensor = input_tensor.contiguous()
    x_flat = input_tensor.view(M, C)

    W_all_T, W_to_out = _get_blas_weights(weights, device)

    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    to_norm_w = weights["to_out_norm.weight"]
    to_norm_b = weights["to_out_norm.bias"]

    # 1) LayerNorm(input) -> bf16
    x_norm = torch.empty(M, C, dtype=torch.bfloat16, device=device)
    C_BLOCK = _next_pow2(C)
    if C_BLOCK <= 128:
        BLOCK_R = 64
    elif C_BLOCK <= 512:
        BLOCK_R = 16
    else:
        BLOCK_R = 8
    layernorm_to_bf16_kernel[(triton.cdiv(M, BLOCK_R),)](
        x_flat, norm_w, norm_b, x_norm,
        M, C, C_BLOCK, BLOCK_R, _EPS,
        num_warps=4,
    )

    x_flat = None
    input_tensor = None

    # 2) All linear projections/gates in one bf16 GEMM.
    GH = 5 * H
    all_proj = x_norm @ W_all_T
    x_norm = None

    left = torch.empty(M, H, dtype=torch.bfloat16, device=device)
    right = torch.empty(M, H, dtype=torch.bfloat16, device=device)
    out_gate = torch.empty(M, H, dtype=torch.bfloat16, device=device)

    no_mask_flag = bool(config.get("nomask", False))
    has_mask = (mask is not None) and (not no_mask_flag)
    mask_flat = None
    if has_mask:
        mask_flat = mask.reshape(-1)
        if not mask_flat.is_contiguous():
            mask_flat = mask_flat.contiguous()
        if mask_flat.dtype != torch.float32:
            mask_flat = mask_flat.to(torch.float32, non_blocking=True)

    mask_ptr = mask_flat if mask_flat is not None else left

    # 3) Apply sigmoid gates + masks to L/R, and compute out_gate.
    BLOCK_M_LR = 64
    apply_left_right_kernel[(triton.cdiv(M, BLOCK_M_LR),)](
        all_proj, left, right, mask_ptr,
        M, GH, H, BLOCK_M_LR, has_mask,
        num_warps=4,
    )

    BLOCK_M_OG = 128
    apply_out_gate_kernel[(triton.cdiv(M, BLOCK_M_OG),)](
        all_proj, out_gate, M, GH, H, BLOCK_M_OG,
        num_warps=4,
    )
    all_proj = None

    # 4) Custom contraction: out[b,i,j,d] = sum_k L[b,i,k,d]*R[b,j,k,d]
    out = torch.empty(M, H, dtype=torch.bfloat16, device=device)

    if N >= 768:
        BI = 128
        BJ = 128
        BD = 32
        warps = 8
        stages = 3
    else:
        BI = 64
        BJ = 128
        BD = 32
        warps = 4
        stages = 4

    BD = min(BD, _next_pow2(H))
    if BD < 16:
        BD = 16

    NUM_D = triton.cdiv(H, BD)
    num_i = triton.cdiv(N, BI)
    num_j = triton.cdiv(N, BJ)

    contraction_kernel[(num_i, num_j, B * NUM_D)](
        left, right, out, N,
        H, BI, BJ, BD, NUM_D,
        num_warps=warps,
        num_stages=stages,
    )
    left = None
    right = None

    # 5) Fused: LayerNorm(out) * out_gate @ to_out_weight^T -> float32 output.
    final_out = torch.empty(M, C, dtype=torch.float32, device=device)
    BLOCK_C = 128
    NUM_C = triton.cdiv(C, BLOCK_C)

    if C <= 128:
        BLOCK_M = 128
        final_warps = 8
    else:
        BLOCK_M = 64
        final_warps = 4

    fused_ln_gate_to_out_kernel[(triton.cdiv(M, BLOCK_M),)](
        out, out_gate, to_norm_w, to_norm_b, W_to_out, final_out,
        M, C, H, BLOCK_M, BLOCK_C, NUM_C, _EPS,
        num_warps=final_warps,
    )

    return final_out.view(B, N, N, C)
