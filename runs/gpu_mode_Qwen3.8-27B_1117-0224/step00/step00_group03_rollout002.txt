@triton.jit
def _final_ln_gate_kernel(
    out_p_ptr, y_ptr, gamma_ptr, beta_ptr, z_ptr,
    M, N, H, FIVE_H,
    BLOCK_T: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    p = pid * BLOCK_T + tl.arange(0, BLOCK_T)
    p_mask = p < M
    p_i64 = p.to(tl.int64)
    NN = N * N
    b = p_i64 // NN
    ij = p_i64 % NN
    offs_h = tl.arange(0, BLOCK_H)
    hv = offs_h < H
    offs_h_safe = tl.where(hv, offs_h, 0).to(tl.int64)
    op_off = (b[None, :] * H + offs_h_safe[:, None]) * NN + ij[None, :]
    op_mask = hv[:, None] & p_mask[None, :]
    vals = tl.load(out_p_ptr + op_off, mask=op_mask, other=0.0).to(tl.float32)
    # H float
    hsum = tl.sum(vals, axis=0)
    mean = hsum / H
    hsq = tl.sum(vals * vals, axis=0)
    var = hsq / H - mean * mean
    rstd = 1.0 / tl.sqrt(var + 1e-5)
    gamma = tl.load(gamma_ptr + offs_h_safe, mask=hv, other=0.0).to(tl.float32)
    beta = tl.load(beta_ptr + offs_h_safe, mask=hv, other=0.0).to(tl.float32)
    norm = (vals - mean[None, :]) * rstd[None, :]
    norm = norm * gamma[:, None] + beta[None, :]
    # gate
    y_off = p_i64[:, None] * FIVE_H + 4 * H + offs_h_safe[None, :]
    gate_in = tl.load(y_ptr + y_off, mask=p_mask[:, None] & hv[None, :], other=0.0).to(tl.float32)
    gate = _triton_sigmoid(gate_in)
    final = norm * tl.trans(gate)
    # store z [BT, BH]
    z_off = p_i64[:, None] * H + offs_h_safe[None, :]
    z_mask = p_mask[:, None] & hv[None, :]

```python
"""
Fused AlphaFold3-style outgoing TriMul forward pass.

Strategy:
1) LayerNorm + one combined bf16 GEMM to compute all 5 projections from x.
2) A Triton kernel applies masks/gates and writes left/right into channel-first
   bf16 tensors [B, H, N, N] for efficient batched matrix multiplication.
3) Core N^3 operation is C = left @ right^T, launched as a batched GEMM over
   B*H batches.
4) A second Triton kernel performs the final LayerNorm + out_gate sigmoid
   directly from the channel-first core output and the stored projection tensor.
5) Final Linear projection is done with cuBLAS in bf16, cast to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _triton_sigmoid(x):
    e = tl.exp(-tl.abs(x))
    return tl.where(x > 0, 1.0 / (1.0 + e), e / (1.0 + e))


@triton.jit
def _fuse_projected_transpose_kernel(
    y_ptr,
    mask_ptr,
    left_t_ptr,
    right_t_ptr,
    B,
    N,
    H,
    FIVE_H,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_k = tl.program_id(2)

    k0 = pid_k * BLOCK_K
    offs_k = k0 + tl.arange(0, BLOCK_K)
    kv = offs_k < N
    offs_k_safe = tl.where(kv, offs_k, 0).to(tl.int64)

    offs_h = tl.arange(0, BLOCK_H)
    hv = offs_h < H
    offs_h_safe = tl.where(hv, offs_h, 0).to(tl.int64)

    N64 = tl.cast(N, tl.int64)
    H64 = tl.cast(H, tl.int64)
    NN64 = N64 * N64

    base_flat = (
        tl.cast(pid_b, tl.int64) * NN64
        + tl.cast(pid_i, tl.int64) * N64
        + tl.cast(k0, tl.int64)
    )
    token = base_flat + offs_k_safe

    y_off = token[:, None] * FIVE_H + offs_h_safe[None, :]
    load_mask = kv[:, None] & hv[None, :]

    # left projection + left gate
    lp = tl.load(y_ptr + y_off, mask=load_mask, other=0.0).to(tl.float32)
    lg = tl.load(y_ptr + y_off + H, mask=load_mask, other=0.0).to(tl.float32)

    if HAS_MASK:
        mask_off = base_flat + offs_k
        m = tl.load(mask_ptr + mask_off, mask=kv, other=0.0).to(tl.float32)[:, None]
    else:
        m = 1.0

    left = lp * _triton_sigmoid(lg) * m

    b_base = tl.cast(pid_b, tl.int64) * H64 * NN64
    h_off = offs_h_safe[:, None] * NN64
    ik_off = tl.cast(pid_i, tl.int64) * N64 + offs_k_safe[None, :]
    store_mask = hv[:, None] & kv[None, :]

    tl.store(
        left_t_ptr + b_base + h_off + ik_off,
        tl.trans(left.to(tl.bfloat16)),
        mask=store_mask,
    )

    # right projection + right gate
    rp = tl.load(y_ptr + y_off + 2 * H, mask=load_mask, other=0.0).to(tl.float32)
    rg = tl.load(y_ptr + y_off + 3 * H, mask=load_mask, other=0.0).to(tl.float32)

    right = rp * _triton_sigmoid(rg) * m

    tl.store(
        right_t_ptr + b_base + h_off + ik_off,
        tl.trans(right.to(tl.bfloat16)),
        mask=store_mask,
    )


@triton.jit
def _final_ln_gate_kernel(
    out_p_ptr,
    y_ptr,
    gamma_ptr,
    beta_ptr,
    z_ptr,
    M,
    N,
    H,
    FIVE_H,
    BLOCK_T: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    p_raw = pid * BLOCK_T + tl.arange(0, BLOCK_T)
    p_mask = p_raw < M
    p_safe = tl.where(p_mask, p_raw, 0).to(tl.int64)

    N64 = tl.cast(N, tl.int64)
    H64 = tl.cast(H, tl.int64)
    NN64 = N64 * N64

    b = p_safe // NN64
    ij = p_safe % NN64

    offs_h = tl.arange(0, BLOCK_H)
    hv = offs_h < H
    offs_h_safe = tl.where(hv, offs_h, 0).to(tl.int64)

    # out_p is [B, H, N, N], token index is b*N*N + i*N + j
    op_off = (b[None, :] * H64 + offs_h_safe[:, None]) * NN64 + ij[None, :]
    op_mask = hv[:, None] & p_mask[None, :]
    vals = tl.load(out_p_ptr + op_off, mask=op_mask, other=0.0).to(tl.float32)

    hsum = tl.sum(vals, axis=0)
    mean = hsum / H
    hsq = tl.sum(vals * vals, axis=0)
    var = hsq / H - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    gamma = tl.load(gamma_ptr + offs_h_safe, mask=hv, other=0.0).to(tl.float32)
    beta = tl.load(beta_ptr + offs_h_safe, mask=hv, other=0.0).to(tl.float32)

    norm = (vals - mean[None, :]) * rstd[None, :]
    norm = norm * gamma[:, None] + beta[None, :]

    # out_gate is the 4th H-slice of the combined projection tensor y
    y_off = p_safe[:, None] * FIVE_H + 4 * H + offs_h_safe[None, :]
    gate_in = tl.load(y_ptr + y_off, mask=p_mask[:, None] & hv[None, :], other=0.0).to(tl.float32)
    gate = _triton_sigmoid(gate_in)

    final = norm * tl.trans(gate)

    z_off = p_safe[:, None] * H64 + offs_h_safe[None, :]
    z_mask = p_mask[:, None] & hv[None, :]
    tl.store(z_ptr + z_off, tl.trans(final).to(tl.bfloat16), mask=z_mask)


def _next_pow2(n: int) -> int:
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _block_k_for_hidden(h: int) -> int:
    if h <= 128:
        return 64
    if h <= 256:
        return 32
    if h <= 512:
        return 16
    return 8


def _block_t_for_hidden(h: int) -> int:
    if h <= 128:
        return 64
    if h <= 256:
        return 32
    if h <= 512:
        return 16
    return 8


def custom_kernel(data):
    """
    data is either:
      (x, mask, weights, config)   preferred
      (x, weights, config)
    where:
      x: [B, N, N, C]
      mask: [B, N, N] or None
      weights: dict of linear/LayerNorm parameters
      config: dict with hidden_dim (and optionally nomask)

    Returns:
      out: [B, N, N, C] float32
    """
    with torch.inference_mode():
        if len(data) == 4:
            x, mask, weights, config = data
        elif len(data) == 3:
            x, weights, config = data
            mask = None
        else:
            raise ValueError("Expected data of length 3 or 4")

        B, N, _, C = x.shape
        H = int(config.get("hidden_dim", C))
        device = x.device

        use_mask = False
        if mask is not None and not config.get("nomask", False):
            mask = mask.to(device=device, dtype=torch.float32)
            if mask.shape != (B, N, N):
                mask = mask.reshape(B, N, N)
            mask = mask.contiguous()
            use_mask = True

        # LayerNorm in fp32.
        if x.dtype == torch.float32:
            x_f32 = x.reshape(-1, C)
        else:
            x_f32 = x.to(torch.float32).reshape(-1, C)

        norm_w = weights["norm.weight"].to(device=device, dtype=torch.float32)
        norm_b = weights["norm.bias"].to(device=device, dtype=torch.float32)
        x_norm = F.layer_norm(x_f32, (C,), norm_w, norm_b, eps=1e-5)
        x_bf = x_norm.to(torch.bfloat16).contiguous()

        # Combined weight matrix for all five projections.
        def _w(name):
            return weights[name].to(device=device, dtype=torch.bfloat16)

        W_cat = torch.cat(
            [
                _w("left_proj.weight"),
                _w("right_proj.weight"),
                _w("left_gate.weight"),
                _w("right_gate.weight"),
                _w("out_gate.weight"),
            ],
            dim=0,
        )
        W_all_t = W_cat.t().contiguous()

        # [B, N, N, 5*H] bf16
        y = torch.matmul(x_bf.reshape(-1, C), W_all_t).view(B, N, N, 5 * H)

        del x_bf, x_f32, x_norm, W_cat, W_all_t

        # Channel-first bf16 tensors for the core batched GEMM.
        left_t = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
        right_t = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)

        BLOCK_H = _next_pow2(H)
        BLOCK_K = _block_k_for_hidden(H)
        grid = (B, N, triton.cdiv(N, BLOCK_K))

        mask_arg = mask if use_mask else left_t
        _fuse_projected_transpose_kernel[grid](
            y,
            mask_arg,
            left_t,
            right_t,
            B,
            N,
            H,
            5 * H,
            BLOCK_K,
            BLOCK_H,
            use_mask,
            num_warps=4,
        )

        # Core TriMul: out[b,h,i,j] = sum_k left[b,h,i,k] * right[b,h,j,k]
        out_p = torch.matmul(left_t, right_t.transpose(-1, 2))

        del left_t, right_t

        # Final normalized gated tensor, channel-last [B, N, N, H], bf16.
        z = torch.empty((B, N, N, H), dtype=torch.bfloat16, device=device)

        BLOCK_T = _block_t_for_hidden(H)
        M = B * N * N
        grid2 = (triton.cdiv(M, BLOCK_T),)

        final_gamma = weights["to_out_norm.weight"].to(device=device, dtype=torch.float32)
        final_beta = weights["to_out_norm.bias"].to(device=device, dtype=torch.float32)

        _final_ln_gate_kernel[grid2](
            out_p,
            y,
            final_gamma,
            final_beta,
            z,
            M,
            N,
            H,
            5 * H,
            BLOCK_T,
            BLOCK_H,
            num_warps=4,
        )

        del out_p, y

        # Final Linear: [M, H] @ [H, C]
        W_out_t = weights["to_out.weight"].to(device=device, dtype=torch.bfloat16).t().contiguous()
        out = torch.matmul(z.reshape(-1, H), W_out_t).view(B, N, N, C)

        return out.to(torch.float32)
```
