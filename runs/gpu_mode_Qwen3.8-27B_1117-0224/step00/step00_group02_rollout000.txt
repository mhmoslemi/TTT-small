def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_li, stride_lk, stride_ld,  # left strides for [i, k, d]
    stride_rj, stride_rk, stride_rd,  # right strides for [j, k, d]
    stride_oi, stride_oj, stride_od,  # out strides for [i, j, d]
    stride_lb, stride_rb, stride_ob,  # batch strides
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Each block computes out[i_block:i_block+BI, j_block:j_block+BJ, d] for a fixed (b, d)
    pid = tl.program_id(0)
    
    # Decode pid to get b, d, i_block, j_block
    # Grid: (num_bi * num_bj * H * B)
    num_bj = tl.cdiv(N, BLOCK_J)
    num_bi = tl.cdiv(N, BLOCK_I)
    
    # Let's use: pid -> (b, d, ii, jj)
    jj = pid % num_bj
    pid2 = pid // num_bj
    ii = pid2 % num_bi
    pid3 = pid2 // num_bi
    d = pid3 % H
    b = pid3 // H
    
    offs_i = ii * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = jj * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)
    
    # A = left[b, i, k, d] for i in [ii*BI, (ii+1)*BI), k in [0, N), fixed d
    # Strides: i -> N*H, k -> H, d -> 1
    # For the GEMM: A[i, k] at left_ptr + b*stride_lb + i*N*H + k*H + d
    
    a_base = left_ptr + b * stride_lb + d  # + d since stride_ld = 1
    b_base = right_ptr + b * stride_rb + d
    
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)
    
    for k_start in range(0, N, BLOCK_K):
        # Load A tile: [BLOCK_I, BLOCK_K] from left[b, i, k, d]
        a_ptrs = a_base + offs_i[:, None] * stride_li + (k_start + offs_k)[None, :] * stride_lk
        a = tl.load(a_ptrs, mask=(offs_i[:, None] < N) & ((k_start + offs_k)[None, :] < N), other=0.0)
        
        # Load B tile: [BLOCK_K, BLOCK_J] from right[b, j, k, d]
        # B[k, j] = right[b, j, k, d]
        b_ptrs = b_base + offs_j[None, :] * stride_rj + (k_start + offs_k)[:, None] * stride_rk
        b_tile = tl.load(b_ptrs, mask=(offs_j[None, :] < N) & ((k_start + offs_k)[:, None] < N), other=0.0)
        
        acc = tl.dot(a, b_tile, acc)
    
    # Store: out[b, i, j, d]
    out_base = out_ptr + b * stride_ob + d
    c_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    mask = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask)

import triton
import triton.language as tl

@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H, B,
    num_bi, num_bj,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):

    pid_div = pid // num_bj
    ii = pid_div % num_bi
    pid_div = pid_div // num_bi
    d = pid_div % H
    b = pid_div // H

    offs_j = jj * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)

    b_base = right_ptr + batch_d_offset
    
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)
    
    for k_start in range(0, N, BLOCK_K):
        k_offsets = k_start + offs_k
        a_ptrs = a_base + offs_i[:, None] * N + k_offsets[None, :]
        b_ptrs = b_base + offs_j[None, :] * N + k_offsets[:, None]
        
        a = tl.load(a_ptrs, mask=(offs_i[:, None] < N) & (k_offsets[None, :] < N), other=0.0)
        b_tile = tl.load(b_ptrs, mask=(offs_j[None, :] < N) & (k_offsets[:, None] < N), other=0.0)
        
        acc = tl.dot(a, b_tile, acc)
    
    # C[i, j] at out_ptr + (b*H+d)*N2 + i*N + j
    c_ptrs = out_ptr + batch_d_offset + offs_i[:, None] * N + offs_j[None, :]
    tl.store(c_ptrs, acc, mask=(offs_i[:, None] < N) & (offs_j[None, :] < N))
