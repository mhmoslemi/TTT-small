@triton.jit
def _prepare_lr_kernel(
    proj_ptr,
    mask_ptr,
    L_ptr,
    R_ptr,
    og_ptr,
    N2,
    H,
    PWIDTH,
    BLOCK_T: tl.constexpr,
    BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    EVEN: tl.constexpr,
):

    pid_bh64 = tl.cast(tl.program_id(1), tl.int64)
    b64 = tl.cast(tl.program_id(2), tl.int64)

    Hi = tl.cast(H, tl.int64)
    PWi = tl.cast(PWIDTH, tl.int64)

    t = tl.arange(0, BLOCK_T).to(tl.int64)
    hh = tl.arange(0, BLOCK_H).to(tl.int64)

    t_local = pid_t64 * BLOCK_T + t
    h_local = pid_bh64 * BLOCK_H + hh

    t_valid = t_local < N2i
    h_valid = h_local < Hi

    t_safe = tl.minimum(t_local, N2i - 1)
    h_safe = tl.minimum(h_local, Hi - 1)

    gt = b64 * N2i + t_safe

    p_off = gt[:, None] * PWi + h_safe[None, :]
    lr_base = (b64 * Hi + h_safe) * N2i
    og_off = gt[:, None] * Hi + h_safe[None, :]

    if EVEN:

        right = tl.load(proj_ptr + p_off + Hi)

def _final_fused_kernel(
    out_ptr,
    gate_ptr,
    w_ptr,
    lnw_ptr,
    lnb_ptr,

):

def custom_kernel(data):
    """
    TriMul forward implementation.

    Strategy:
    - Use cuBLAS-compatible batched GEMM for the O(N^3) pair core:
      out_{b,i,j,h} = sum_k L_{b,h,i,k} * R_{b,h,j,k}.
    - Fusion kernels in Triton for:
      1) masked/gated left/right projection transform into h-major planes + out_gate,
      2) final LayerNorm + out_gate + final linear projection for H=128.
    """
    if len(data) == 4:
        input_tensor, mask, weights, config = data
    elif len(data) == 3:
        input_tensor, weights, config = data
        mask = None
    else:
        raise ValueError("Unsupported data format")

    B, N, N, C = input_tensor.shape
    H = int(config.get("hidden_dim", 128))
    eps = float(config.get("eps", 1e-5))
    device = input_tensor.device
    M = B * N * N
    N2 = N * N

    def _as_bf16(x):
        return None if x is None else x.to(torch.bfloat16)

    def _as_f32(x, default=None, shape=None):
        if x is None:
            if default is None:
                return None
            return default
        return x.float().contiguous()

    norm_w = _as_bf16(weights.get("norm.weight"))
    norm_b = _as_bf16(weights.get("norm.bias"))

    # Initial LayerNorm in bf16. PyTorch's layernorm for half/bf16 accumulates in fp32.
    x_b = input_tensor.reshape(M, C).to(torch.bfloat16)
    x_norm = F.layer_norm(x_b, (C,), norm_w, norm_b, eps)
    if not x_norm.is_contiguous():
        x_norm = x_norm.contiguous()

    # Combined projection:
    # [left, right, left_gate, right_gate, out_gate]
    W_cat = torch.cat(
        [
            weights["left_proj.weight"].to(torch.bfloat16),
            weights["right_proj.weight"].to(torch.bfloat16),
            weights["left_gate.weight"].to(torch.bfloat16),
            weights["right_gate.weight"].to(torch.bfloat16),
            weights["out_gate.weight"].to(torch.bfloat16),
        ],
        dim=0,
    )
    W_cat_t = W_cat.t().contiguous()

    proj = torch.mm(x_norm, W_cat_t)  # [M, 5H], bf16
    del x_norm, x_b

    # Transformed effective left/right matrices, organized as [B, H, N, N]
    # so that the h-th core GEMM is a standard N x N @ N x N^T batch.
    L = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    R = torch.empty_like(L)
    og = torch.empty((M, H), dtype=torch.bfloat16, device=device)

    has_mask = mask is not None
    if has_mask:
        mask_f = mask.reshape(M).to(torch.float32)
    else:
        mask_f = proj  # dummy pointer; not read when HAS_MASK is False

    BLOCK_T = 128
    BLOCK_H = 64
    num_t = triton.cdiv(N2, BLOCK_T)
    num_bh = triton.cdiv(H, BLOCK_H)
    EVEN = (N2 % BLOCK_T == 0) and (H % BLOCK_H == 0)

    _prepare_lr_kernel[(num_t, num_bh, B)](
        proj,
        mask_f,
        L,
        R,
        og,
        N2,
        H,
        5 * H,
        BLOCK_T=BLOCK_T,
        BLOCK_H=BLOCK_H,
        HAS_MASK=has_mask,
        EVEN=EVEN,
        num_warps=4,
    )
    del proj, mask_f

    # Core: for every (b, h), compute L_{b,h} @ R_{b,h}^T.
    out_h = torch.bmm(
        L.view(B * H, N, N),
        R.view(B * H, N, N).transpose(1, 2),
    )
    del L, R

    # Re-layout for final LayerNorm over H and the final linear layer:
    # [B, H, N, N] -> [B, N, N, H], flattened as [M, H].
    out_core = out_h.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del out_h

    if H == 128 and (C % 128 == 0) and (M % 128 == 0):
        final = torch.empty((M, C), dtype=torch.float32, device=device)

        ln_w_f = weights.get("to_out_norm.weight")
        if ln_w_f is None:
            ln_w_f = torch.ones(H, dtype=torch.float32, device=device)
        else:
            ln_w_f = ln_w_f.float().contiguous()

        ln_b_f = weights.get("to_out_norm.bias")
        if ln_b_f is None:
            ln_b_f = torch.zeros(H, dtype=torch.float32, device=device)
        else:
            ln_b_f = ln_b_f.float().contiguous()

        Wout_t = weights["to_out.weight"].to(torch.bfloat16).t().contiguous()

        BLOCK_M = 128
        BLOCK_C = 128

        _final_fused_kernel[(triton.cdiv(M, BLOCK_M),)](
            out_core,
            og,
            Wout_t,
            ln_w_f,
            ln_b_f,
            final,
            C,
            eps,
            BLOCK_M=BLOCK_M,
            BLOCK_H=128,
            BLOCK_C=BLOCK_C,
            num_warps=8,
            num_stages=2,
        )
    else:
        # Robust path for non-128 hidden dims or non-aligned shapes.
        out_f = out_core.view(M, H).float()

        ln_w = weights.get("to_out_norm.weight")
        ln_b = weights.get("to_out_norm.bias")
        if ln_w is not None:
            ln_w = ln_w.float()
        if ln_b is not None:
            ln_b = ln_b.float()

        out_normed = F.layer_norm(out_f, (H,), ln_w, ln_b, eps)
        out_gated = (out_normed * og.float()).to(torch.bfloat16)

        Wout_b = weights["to_out.weight"].to(torch.bfloat16)
        final = torch.mm(out_gated, Wout_b.t()).float()

    return final.view(B, N, N, C)
