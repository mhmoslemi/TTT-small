@triton.jit
def contraction_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # left[b, i, k, d] strides
    stride_lb, stride_li, stride_lk, stride_ld,
    # right[b, j, k, d] strides
    stride_rb, stride_rj, stride_rk, stride_rd,
    # out[b, i, j, d] strides
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # pid encodes (b, d, i_block, j_block)
    num_i_blocks = tl.cdiv(N, BLOCK_I)
    num_j_blocks = tl.cdiv(N, BLOCK_J)
    num_batches = B * H
    
    pid = tl.program_id(0)
    # Decode: pid → (batch_idx, i_block, j_block)
    # batch_idx = b * H + d
    pid_ij = pid % (num_i_blocks * num_j_blocks)
    batch_idx = pid // (num_i_blocks * num_j_blocks)
    i_block = pid_ij // num_j_blocks
    j_block = pid_ij % num_j_blocks
    
    b = batch_idx // H
    d = batch_idx % H
    
    offs_i = i_block * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = j_block * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)
    
    # Pointers for left: left[b, i, k, d]
    # Address: b*stride_lb + i*stride_li + k*stride_lk + d*stride_ld
    left_ptrs = left_ptr + b * stride_lb + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk + d * stride_ld
    
    # Pointers for right: right[b, j, k, d]
    # Address: b*stride_rb + j*stride_rj + k*stride_rk + d*stride_rd
    right_ptrs = right_ptr + b * stride_rb + offs_j[:, None] * stride_rj + offs_k[None, :] * stride_rk + d * stride_rd
    
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

        k_mask = (k + tl.arange(0, BLOCK_K)) < N
        a_mask = (offs_i < N)[:, None] & k_mask[None, :]
        b_mask = (offs_j < N)[:, None] & k_mask[None, :]
        
        a = tl.load(left_ptrs, mask=a_mask, other=0.0)
        b_mat = tl.load(right_ptrs, mask=b_mask, other=0.0)
