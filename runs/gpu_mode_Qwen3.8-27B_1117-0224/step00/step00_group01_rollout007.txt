def _prep_kernel(
    x_ptr, mask_ptr,
    l_ptr, r_ptr, og_ptr,
    norm_w_ptr, norm_b_ptr,
    wl_ptr, wr_ptr, wlg_ptr, wrg_ptr, wog_ptr,
    S, C, H,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_C: tl.constexpr,
    BLOCK_H: tl.constexpr,
    EPS: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_r = tl.program_id(1)
    P = S * S
    offs_r = pid_r * BLOCK_R + tl.arange(0, BLOCK_R)
    rmask = offs_r < P
    offs_c = tl.arange(0, BLOCK_C)
    cmask = offs_c < C
    x_ptrs = x_ptr + pid_b * P * C + offs_r[:, None] * C + offs_c[None, :]
    x = tl.load(x_ptrs, mask=rmask[:, None] & cmask[None, :], other=0.0).to(tl.float32)
    x_sum = tl.sum(x, axis=1)
    mean = x_sum / C
    sumsq = tl.sum(x * x, axis=1)
    var = sumsq / C - mean * mean
    var = tl.maximum(var, 0.0)
    inv_std = 1.0 / tl.sqrt(var + EPS)
    nw = tl.load(norm_w_ptr + offs_c, mask=cmask, other=0.0).to(tl.float32)
    nb = tl.load(norm_b_ptr + offs_c, mask=cmask, other=0.0).to(tl.float32)
    xhat = (x - mean[:, None]) * inv_std[:, None]
    xhat = xhat * nw[None, :] + nb[None, :]
    xhat = tl.where(cmask[None, :], xhat, 0.0)
    a = xhat.to(tl.bfloat16)

    if HAS_MASK:
        m = tl.load(mask_ptr + pid_b * P + offs_r, mask=rmask, other=0.0).to(tl.float32)
    else:
        m = tl.full((BLOCK_R,), 1.0, tl.float32)
        m = tl.where(rmask, m, 0.0)

    for h0 in range(0, H, BLOCK_H):
        offs_h = h0 + tl.arange(0, BLOCK_H)
        hmask = offs_h < H
        w_mask = cmask[:, None] & hmask[None, :]
        wl = tl.load(wl_ptr + offs_c[:, None] * H + offs_h[None, :], mask=w_mask, other=0.0).to(tl.bfloat16)
        wr = tl.load(wr_ptr + offs_c[:, None] * H + offs_h[None, :], mask=w_mask, other=0.0).to(tl.bfloat16)
        wlg = tl.load(wlg_ptr + offs_c[:, None] * H + offs_h[None, :], mask=w_mask, other=0.0).to(tl.bfloat16)
        wrg = tl.load(wrg_ptr + offs_c[:, None] * H + offs_h[None, :], mask=w_mask, other=0.0).to(tl.bfloat16)
        wog = tl.load(wog_ptr + offs_c[:, None] * H + offs_h[None, :], mask=w_mask, other=0.0).to(tl.bfloat16)

        l = tl.dot(a, wl, out_dtype=tl.float32)
        r = tl.dot(a, wr, out_dtype=tl.float32)
        lg = tl.dot(a, wlg, out_dtype=tl.float32)
        rg = tl.dot(a, wrg, out_dtype=tl.float32)
        og = tl.dot(a, wog, out_dtype=tl.float32)
        lg = 1.0 / (1.0 + tl.exp(-lg))
        rg = 1.0 / (1.0 + tl.exp(-rg))
        og = 1.0 / (1.0 + tl.exp(-og))
        l = l * m[:, None] * lg
        r = r * m[:, None] * rg
        # store [BH, BR]
        store_mask = hmask[:, None] & rmask[None, :]
        base = pid_b * H * P

        og_ptrs = og_ptr + base + offs_h[:, None] * P + offs_r[None, :]
        tl.store(l_ptrs, tl.trans(l.to(tl.bfloat16)), mask=store_mask)

def _core_kernel(l_ptr, r_ptr, y_ptr, S, H, B,
                 BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    P = S*S
    num_m = tl.cdiv(S, BLOCK_I)
    num_n = tl.cdiv(S, BLOCK_J)

    bh = pid // tiles_per_bh
    rem = pid % tiles_per_bh
    pid_m = rem % num_m
    pid_n = rem // num_m
    i0 = pid_m * BLOCK_I
    j0 = pid_n * BLOCK_J
    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)
    imask = offs_i < S
    jmask = offs_j < S
    base_l = l_ptr + bh * P
    base_r = r_ptr + bh * P
    base_y = y_ptr + bh * P
    a_ptrs = base_l + offs_i[:,None]*S + offs_k[None,:]

    c_ptrs = base_y + offs_i[:,None]*S + offs_j[None,:]
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

```python
"""
Fused forward pass for the outgoing TriMul operator.

Layout choice:
  - The intermediate hidden tensors L, R, OG, Y are stored as [B, H, S, S].
    This makes the core `sum_k L[b,h,i,k] * R[b,h,j,k]` a conventional
    batched GEMM in `h`: Y[b,h,i,j] = L[b,h,i,:] @ R[b,h,j,:]^T, with
    coalesced loads/stores.

Kernels:
  1) _prep_kernel:
       - LayerNorm over input channels C
       - five linear projections and two/output gates
       - mask multiplication
       - writes gated L, gated R, and out_gate in [B,H,S,S] bf16
  2) _core_kernel:
       - H batched GEMMs for the TriMul contraction, accumulating in fp32
       - writes pre-normalization Y in [B,H,S,S] bf16
  3) _final_kernel:
       - LayerNorm over hidden H for each output pair (i,j)
       - multiply by out_gate
       - final projection to output channels C
       - writes fp32 output [B,S,S,C]
"""

import torch
import triton
import triton.language as tl

_WEIGHT_CACHE = {}
_WEIGHT_NAMES = (
    "norm.weight",
    "norm.bias",
    "left_proj.weight",
    "right_proj.weight",
    "left_gate.weight",
    "right_gate.weight",
    "out_gate.weight",
    "to_out_norm.weight",
    "to_out_norm.bias",
    "to_out.weight",
)


def _next_pow2(n: int) -> int:
    n = int(n)
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _prepare_weights(weights, device):
    key = tuple(
        (
            weights[name].data_ptr(),
            tuple(weights[name].shape),
            str(weights[name].dtype),
            str(weights[name].device),
            getattr(weights[name], "_version", 0),
        )
        for name in _WEIGHT_NAMES
    )

    cached = _WEIGHT_CACHE.get(key)
    if cached is not None:
        return cached[0]

    fp32 = torch.float32
    bf16 = torch.bfloat16

    norm_w = weights["norm.weight"].to(device=device, dtype=fp32).contiguous()
    norm_b = weights["norm.bias"].to(device=device, dtype=fp32).contiguous()

    Wl = weights["left_proj.weight"].t().contiguous().to(device=device, dtype=bf16)
    Wr = weights["right_proj.weight"].t().contiguous().to(device=device, dtype=bf16)
    Wlg = weights["left_gate.weight"].t().contiguous().to(device=device, dtype=bf16)
    Wrg = weights["right_gate.weight"].t().contiguous().to(device=device, dtype=bf16)
    Wog = weights["out_gate.weight"].t().contiguous().to(device=device, dtype=bf16)

    out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp32).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp32).contiguous()

    # to_out Linear: [C, H] @ x -> we need [H, C] for the final dot.
    Wto_t = weights["to_out.weight"].t().contiguous().to(device=device, dtype=bf16)

    prep = {
        "norm_w": norm_w,
        "norm_b": norm_b,
        "Wl": Wl,
        "Wr": Wr,
        "Wlg": Wlg,
        "Wrg": Wrg,
        "Wog": Wog,
        "out_w": out_norm_w,
        "out_b": out_norm_b,
        "Wto_t": Wto_t,
    }

    _WEIGHT_CACHE[key] = (prep, tuple(weights[name] for name in _WEIGHT_NAMES))
    return prep


@triton.jit
def _prep_kernel(
    x_ptr,
    mask_ptr,
    l_ptr,
    r_ptr,
    og_ptr,
    norm_w_ptr,
    norm_b_ptr,
    wl_ptr,
    wr_ptr,
    wlg_ptr,
    wrg_ptr,
    wog_ptr,
    S,
    C,
    H,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_C: tl.constexpr,
    BLOCK_H: tl.constexpr,
    EPS: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_r = tl.program_id(1)

    P = S * S

    offs_r = pid_r * BLOCK_R + tl.arange(0, BLOCK_R)
    rmask = offs_r < P
    offs_c = tl.arange(0, BLOCK_C)
    cmask = offs_c < C
    load_mask = rmask[:, None] & cmask[None, :]

    x_ptrs = x_ptr + pid_b * P * C + offs_r[:, None] * C + offs_c[None, :]
    x = tl.load(x_ptrs, mask=load_mask, other=0.0).to(tl.float32)

    # Stable-enough LayerNorm: centered mean/variance within this tile.
    x_sum = tl.sum(x, axis=1)
    mean = x_sum / C
    xc = x - mean[:, None]
    xc = tl.where(load_mask, xc, 0.0)
    var = tl.sum(xc * xc, axis=1) / C
    var = tl.where(var > 0.0, var, 0.0)
    inv_std = 1.0 / tl.sqrt(var + EPS)

    nw = tl.load(norm_w_ptr + offs_c, mask=cmask, other=0.0).to(tl.float32)
    nb = tl.load(norm_b_ptr + offs_c, mask=cmask, other=0.0).to(tl.float32)

    xhat = xc * inv_std[:, None] * nw[None, :] + nb[None, :]
    xhat = tl.where(load_mask, xhat, 0.0)
    a = xhat.to(tl.bfloat16)

    if HAS_MASK:
        m = tl.load(mask_ptr + pid_b * P + offs_r, mask=rmask, other=0.0).to(tl.float32)
        m = tl.where(rmask, m, 0.0)
    else:
        m = tl.where(rmask, 1.0, 0.0)

    base = pid_b * H * P

    for h0 in range(0, H, BLOCK_H):
        offs_h = h0 + tl.arange(0, BLOCK_H)
        hmask = offs_h < H
        w_mask = cmask[:, None] & hmask[None, :]

        # Left and left gate
        wl = tl.load(wl_ptr + offs_c[:, None] * H + offs_h[None, :], mask=w_mask, other=0.0).to(tl.bfloat16)
        wlg = tl.load(wlg_ptr + offs_c[:, None] * H + offs_h[None, :], mask=w_mask, other=0.0).to(tl.bfloat16)

        l = tl.dot(a, wl)
        lg = 1.0 / (1.0 + tl.exp(-tl.dot(a, wlg)))
        l = l * m[:, None] * lg

        st_mask = hmask[:, None] & rmask[None, :]
        l_ptrs = l_ptr + base + offs_h[:, None] * P + offs_r[None, :]
        tl.store(l_ptrs, tl.trans(l.to(tl.bfloat16)), mask=st_mask)

        # Right and right gate
        wr = tl.load(wr_ptr + offs_c[:, None] * H + offs_h[None, :], mask=w_mask, other=0.0).to(tl.bfloat16)
        wrg = tl.load(wrg_ptr + offs_c[:, None] * H + offs_h[None, :], mask=w_mask, other=0.0).to(tl.bfloat16)

        r = tl.dot(a, wr)
        rg = 1.0 / (1.0 + tl.exp(-tl.dot(a, wrg)))
        r = r * m[:, None] * rg

        r_ptrs = r_ptr + base + offs_h[:, None] * P + offs_r[None, :]
        tl.store(r_ptrs, tl.trans(r.to(tl.bfloat16)), mask=st_mask)

        # Output gate for this (i,j) pair position
        wog = tl.load(wog_ptr + offs_c[:, None] * H + offs_h[None, :], mask=w_mask, other=0.0).to(tl.bfloat16)
        og = 1.0 / (1.0 + tl.exp(-tl.dot(a, wog)))

        og_ptrs = og_ptr + base + offs_h[:, None] * P + offs_r[None, :]
        tl.store(og_ptrs, tl.trans(og.to(tl.bfloat16)), mask=st_mask)


@triton.jit
def _core_kernel(
    l_ptr,
    r_ptr,
    y_ptr,
    S,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    P = S * S

    num_m = tl.cdiv(S, BLOCK_I)
    num_n = tl.cdiv(S, BLOCK_J)
    tiles_per_bh = num_m * num_n

    bh = pid // tiles_per_bh
    rem = pid % tiles_per_bh

    # Consecutive pids share the same J block and sweep I blocks:
    # R tile tends to be reused across nearby A blocks in L2.
    pid_m = rem % num_m
    pid_n = rem // num_m

    i0 = pid_m * BLOCK_I
    j0 = pid_n * BLOCK_J

    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)

    imask = offs_i < S
    jmask = offs_j < S

    base_l = l_ptr + bh * P
    base_r = r_ptr + bh * P
    base_y = y_ptr + bh * P

    a_ptrs = base_l + offs_i[:, None] * S + offs_k[None, :]
    b_ptrs = base_r + offs_j[:, None] * S + offs_k[None, :]
    c_ptrs = base_y + offs_i[:, None] * S + offs_j[None, :]

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    for k0 in range(0, S, BLOCK_K):
        kmask = (k0 + offs_k) < S
        a = tl.load(a_ptrs, mask=imask[:, None] & kmask[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=jmask[:, None] & kmask[None, :], other=0.0)
        acc += tl.dot(a, tl.trans(b))
        a_ptrs += BLOCK_K
        b_ptrs += BLOCK_K

    tl.store(c_ptrs, acc.to(tl.bfloat16), mask=imask[:, None] & jmask[None, :])


@triton.jit
def _final_kernel(
    y_ptr,
    og_ptr,
    out_ptr,
    wto_ptr,
    norm_w_ptr,
    norm_b_ptr,
    S,
    H,
    C,
    EPS: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)

    P = S * S

    num_j = tl.cdiv(S, BLOCK_J)
    num_c = tl.cdiv(C, BLOCK_C)

    # Decode order: fastest c-chunk, then j-block, then i-row, then batch.
    pid_c = pid % num_c
    rem = pid // num_c
    pid_j = rem % num_j
    rem = rem // num_j
    pid_i = rem % S
    b = rem // S

    j0 = pid_j * BLOCK_J
    c0 = pid_c * BLOCK_C

    offs_j = j0 + tl.arange(0, BLOCK_J)
    offs_c = c0 + tl.arange(0, BLOCK_C)

    jmask = offs_j < S
    cmask = offs_c < C

    base_y = y_ptr + b * H * P
    base_og = og_ptr + b * H * P

    # First pass: LayerNorm statistics over H for each (i, j) in this block.
    sum_y = tl.zeros((BLOCK_J,), dtype=tl.float32)
    sum_sq_y = tl.zeros((BLOCK_J,), dtype=tl.float32)

    for h0 in range(0, H, BLOCK_H):
        offs_h = h0 + tl.arange(0, BLOCK_H)
        hmask = offs_h < H
        valid = hmask[:, None] & jmask[None, :]

        y = tl.load(base_y + offs_h[:, None] * P + pid_i * S + offs_j[None, :], mask=valid, other=0.0).to(tl.float32)
        y = tl.where(valid, y, 0.0)

        sum_y += tl.sum(y, axis=0)
        sum_sq_y += tl.sum(y * y, axis=0)

    mean = sum_y / H
    var = sum_sq_y / H - mean * mean
    var = tl.where(var > 0.0, var, 0.0)
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # Second pass: normalize, apply output gate, project to C.
    acc = tl.zeros((BLOCK_J, BLOCK_C), dtype=tl.float32)

    for h0 in range(0, H, BLOCK_H):
        offs_h = h0 + tl.arange(0, BLOCK_H)
        hmask = offs_h < H
        valid = hmask[:, None] & jmask[None, :]

        y = tl.load(base_y + offs_h[:, None] * P + pid_i * S + offs_j[None, :], mask=valid, other=0.0).to(tl.float32)
        og = tl.load(base_og + offs_h[:, None] * P + pid_i * S + offs_j[None, :], mask=valid, other=0.0).to(tl.float32)

        gamma = tl.load(norm_w_ptr + offs_h, mask=hmask, other=0.0).to(tl.float32)
        beta = tl.load(norm_b_ptr + offs_h, mask=hmask, other=0.0).to(tl.float32)

        yhat = (y - mean[None, :]) * inv_std[None, :] * gamma[:, None] + beta[:, None]
        t = yhat * og
        t = tl.where(valid, t, 0.0)

        w_mask = hmask[:, None] & cmask[None, :]
        w = tl.load(wto_ptr + offs_h[:, None] * C + offs_c[None, :], mask=w_mask, other=0.0).to(tl.bfloat16)

        acc += tl.dot(tl.trans(t.to(tl.bfloat16)), w)

    out_ptrs = out_ptr + (b * S + pid_i) * S * C + offs_j[:, None] * C + offs_c[None, :]
    tl.store(out_ptrs, acc, mask=jmask[:, None] & cmask[None, :])


def custom_kernel(data):
    if len(data) == 4:
        a, b, c, d = data
        if isinstance(c, dict) and not isinstance(b, dict):
            input_tensor, mask, weights, config = a, b, c, d
        elif isinstance(b, dict):
            input_tensor, weights, config = a, b, c
            mask = d if isinstance(d, torch.Tensor) else None
        else:
            input_tensor, mask, weights, config = a, b, c, d
    elif len(data) == 3:
        input_tensor, weights, config = data
        mask = None
    else:
        input_tensor = data[0]
        if isinstance(data[1], dict):
            weights = data[1]
            config = data[2] if len(data) > 2 else {}
            mask = None
        else:
            mask = data[1]
            weights = data[2]
            config = data[3] if len(data) > 3 else {}

    x = input_tensor.contiguous()
    B, S, S2, C = x.shape
    device = x.device

    H = int(config.get("hidden_dim", weights["left_proj.weight"].shape[0]))
    C = int(config.get("dim", x.shape[-1]))

    has_mask = mask is not None
    if has_mask and bool(config.get("nomask", False)):
        has_mask = False
    if has_mask:
        mask = mask.contiguous()
    else:
        mask = None

    prep = _prepare_weights(weights, device)

    P = S * S

    # ---- First stage blocks / sizes ----
    BLOCK_C_PRE = _next_pow2(C)
    # Keep x tile roughly 16K elements.
    BR = max(16, _next_pow2(16384 // max(1, BLOCK_C_PRE)))
    BR = min(128, BR)
    BLOCK_H_PRE = min(128, _next_pow2(H))
    pre_warps = 8 if BR >= 64 else 4

    # Allocate intermediate hidden tensors in [B, H, S, S] layout.
    L = torch.empty((B, H, S, S), dtype=torch.bfloat16, device=device)
    R = torch.empty((B, H, S, S), dtype=torch.bfloat16, device=device)
    OG = torch.empty((B, H, S, S), dtype=torch.bfloat16, device=device)
    Y = torch.empty((B, H, S, S), dtype=torch.bfloat16, device=device)

    mask_arg = mask if has_mask else x
    grid_pre = (B, triton.cdiv(P, BR))
    _prep_kernel[grid_pre](
        x,
        mask_arg,
        L,
        R,
        OG,
        prep["norm_w"],
        prep["norm_b"],
        prep["Wl"],
        prep["Wr"],
        prep["Wlg"],
        prep["Wrg"],
        prep["Wog"],
        S,
        C,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BR,
        BLOCK_C=BLOCK_C_PRE,
        BLOCK_H=BLOCK_H_PRE,
        EPS=1e-5,
        num_warps=pre_warps,
        num_stages=2,
    )

    # ---- Core TriMul contraction ----
    if S >= 512:
        BI = BJ = 128
    else:
        BI = BJ = 64
    BK = 64
    core_warps = 8 if BI >= 128 else 4

    num_i = triton.cdiv(S, BI)
    num_j = triton.cdiv(S, BJ)
    grid_core = (B * H * num_i * num_j,)
    _core_kernel[grid_core](
        L,
        R,
        Y,
        S,
        BLOCK_I=BI,
        BLOCK_J=BJ,
        BLOCK_K=BK,
        num_warps=core_warps,
        num_stages=3,
    )

    # ---- Final LayerNorm, gate, and projection ----
    out = torch.empty((B, S, S, C), dtype=torch.float32, device=device)

    BLOCK_C_FIN = min(_next_pow2(C), 512)
    BLOCK_J_FIN = 128 if BLOCK_C_FIN <= 128 else 64
    BLOCK_H_FIN = min(128, _next_pow2(H))
    fin_warps = 8

    fin_num_j = triton.cdiv(S, BLOCK_J_FIN)
    fin_num_c = triton.cdiv(C, BLOCK_C_FIN)
    grid_final = (B * S * fin_num_j * fin_num_c,)

    _final_kernel[grid_final](
        Y,
        OG,
        out,
        prep["Wto_t"],
        prep["out_w"],
        prep["out_b"],
        S,
        H,
        C,
        EPS=1e-5,
        BLOCK_J=BLOCK_J_FIN,
        BLOCK_H=BLOCK_H_FIN,
        BLOCK_C=BLOCK_C_FIN,
        num_warps=fin_warps,
        num_stages=2,
    )

    return out
```
