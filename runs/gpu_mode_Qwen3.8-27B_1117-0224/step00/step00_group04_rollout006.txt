```python
"""
Fused Triton implementation of the AlphaFold3 outgoing TriMul forward pass.

Pipeline:
  1. Fused LayerNorm + all linear projections/gates:
       left  = x_norm -> W_left  * sigmoid(x_norm -> W_left_gate)  * mask
       right = x_norm -> W_right * sigmoid(x_norm -> W_right_gate) * mask
       gate  = sigmoid(x_norm -> W_out_gate)
     The LayerNorm statistics are computed inside the kernel, so no normalized
     x is materialized.
  2. Batched triangular contraction in Triton:
       out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
  3. Fused LayerNorm + output gating + final linear projection.

The hot contraction is a true batched GEMM over all hidden channels (d), and
the two LayerNorm-heavy linear stages are fused with their surrounding
nonlinearities to avoid materializing normalized tensors.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _sigmoid(x):
    # Numerically stable sigmoid for fp32 inputs.
    return tl.where(
        x >= 0,
        1.0 / (1.0 + tl.exp(-x)),
        tl.exp(x) / (1.0 + tl.exp(x)),
    )


@triton.jit
def init_ln_all_c128_kernel(
    x_ptr,
    mask_ptr,
    wl_ptr,
    wr_ptr,
    wlg_ptr,
    wrg_ptr,
    wog_ptr,
    bwl_ptr,
    bwr_ptr,
    bwlg_ptr,
    bwrg_ptr,
    bwog_ptr,
    gamma_ptr,
    out_l_ptr,
    out_r_ptr,
    out_g_ptr,
    M,
    C,
    H,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
    BLOCK_N: tl.constexpr,
    EPS: tl.constexpr,
):
    """
    Fused LN + left/right/out-gate projections for C <= 128.
    Loads each row of x only once.
    """
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    row_ok = offs_m < M

    offs_c = tl.arange(0, BLOCK_C)
    c_ok = offs_c < C

    x_mask = row_ok[:, None] & c_ok[None, :]
    x = tl.load(x_ptr + offs_m[:, None] * C + offs_c[None, :], mask=x_mask, other=0.0).to(tl.float32)

    mean = tl.sum(x, axis=1) / C
    sumsq = tl.sum(x * x, axis=1) / C
    var = sumsq - mean * mean
    rstd = 1.0 / tl.sqrt(tl.maximum(var, 0.0) + EPS)

    gamma = tl.load(gamma_ptr + offs_c, mask=c_ok, other=0.0).to(tl.float32)
    a = ((x - mean[:, None]) * (rstd[:, None] * gamma[None, :])).to(tl.bfloat16)

    if HAS_MASK:
        mrow = tl.load(mask_ptr + offs_m, mask=row_ok, other=0.0).to(tl.float32)

    for n0 in range(0, H, BLOCK_N):
        offs_n = n0 + tl.arange(0, BLOCK_N)
        n_ok = offs_n < H
        w_mask = n_ok[None, :] & c_ok[:, None]
        store_mask = row_ok[:, None] & n_ok[None, :]

        # left * sigmoid(left_gate)
        wl = tl.load(wl_ptr + offs_n[None, :] * C + offs_c[:, None], mask=w_mask, other=0.0)
        wlg = tl.load(wlg_ptr + offs_n[None, :] * C + offs_c[:, None], mask=w_mask, other=0.0)
        bwl = tl.load(bwl_ptr + offs_n, mask=n_ok, other=0.0).to(tl.float32)
        bwlg = tl.load(bwlg_ptr + offs_n, mask=n_ok, other=0.0).to(tl.float32)

        acc_l = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_l = tl.dot(a, wl, acc_l)
        acc_gl = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_gl = tl.dot(a, wlg, acc_gl)

        val_l = (acc_l + bwl[None, :]) * _sigmoid(acc_gl + bwlg[None, :])
        if HAS_MASK:
            val_l = val_l * mrow[:, None]
        tl.store(out_l_ptr + offs_m[:, None] * H + offs_n[None, :], val_l.to(tl.bfloat16), mask=store_mask)

        # right * sigmoid(right_gate)
        wr = tl.load(wr_ptr + offs_n[None, :] * C + offs_c[:, None], mask=w_mask, other=0.0)
        wrg = tl.load(wrg_ptr + offs_n[None, :] * C + offs_c[:, None], mask=w_mask, other=0.0)
        bwr = tl.load(bwr_ptr + offs_n, mask=n_ok, other=0.0).to(tl.float32)
        bwrg = tl.load(bwrg_ptr + offs_n, mask=n_ok, other=0.0).to(tl.float32)

        acc_r = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_r = tl.dot(a, wr, acc_r)
        acc_gr = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_gr = tl.dot(a, wrg, acc_gr)

        val_r = (acc_r + bwr[None, :]) * _sigmoid(acc_gr + bwrg[None, :])
        if HAS_MASK:
            val_r = val_r * mrow[:, None]
        tl.store(out_r_ptr + offs_m[:, None] * H + offs_n[None, :], val_r.to(tl.bfloat16), mask=store_mask)

        # out_gate = sigmoid(logit)
        wog = tl.load(wog_ptr + offs_n[None, :] * C + offs_c[:, None], mask=w_mask, other=0.0)
        bwog = tl.load(bwog_ptr + offs_n, mask=n_ok, other=0.0).to(tl.float32)

        acc_g = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_g = tl.dot(a, wog, acc_g)

        val_g = _sigmoid(acc_g + bwog[None, :])
        tl.store(out_g_ptr + offs_m[:, None] * H + offs_n[None, :], val_g.to(tl.bfloat16), mask=store_mask)


@triton.jit
def init_ln_all_c384_kernel(

    mask_ptr,
    wl_ptr,
    wr_ptr,
    wlg_ptr,
    wrg_ptr,
    wog_ptr,
    bwl_ptr,
    bwr_ptr,
    bwlg_ptr,
    bwrg_ptr,
    bwog_ptr,
    gamma_ptr,
    out_l_ptr,
    out_r_ptr,
    out_g_ptr,
    M,
    C,
    H,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
    BLOCK_N: tl.constexpr,
    EPS: tl.constexpr,
):

    Uses three static 128-wide chunks, loads x only once.
    """

    c1_off = BLOCK_C + offs_c
    c2_off = 2 * BLOCK_C + offs_c

    c0_ok = c0_off < C
    c1_ok = c1_off < C
    c2_ok = c2_off < C

    x0 = tl.load(x_ptr + offs_m[:, None] * C + c0_off[None, :], mask=row_ok[:, None] & c0_ok[None, :], other=0.0).to(tl.float32)
    x1 = tl.load(x_ptr + offs_m[:, None] * C + c1_off[None, :], mask=row_ok[:, None] & c1_ok[None, :], other=0.0).to(tl.float32)
    x2 = tl.load(x_ptr + offs_m[:, None] * C + c2_off[None, :], mask=row_ok[:, None] & c2_ok[None, :], other=0.0).to(tl.float32)

    mean = (tl.sum(x0, axis=1) + tl.sum(x1, axis=1) + tl.sum(x2, axis=1)) / C
    sumsq = (tl.sum(x0 * x0, axis=1) + tl.sum(x1 * x1, axis=1) + tl.sum(x2 * x2, axis=1)) / C
    var = sumsq - mean * mean
    rstd = 1.0 / tl.sqrt(tl.maximum(var, 0.0) + EPS)

    g0 = tl.load(gamma_ptr + c0_off, mask=c0_ok, other=0.0).to(tl.float32)
    g1 = tl.load(gamma_ptr + c1_off, mask=c1_ok, other=0.0).to(tl.float32)
    g2 = tl.load(gamma_ptr + c2_off, mask=c2_ok, other=0.0).to(tl.float32)

    a0 = ((x0 - mean[:, None]) * (rstd[:, None] * g0[None, :])).to(tl.bfloat16)
    a1 = ((x1 - mean[:, None]) * (rstd[:, None] * g1[None, :])).to(tl.bfloat16)
    a2 = ((x2 - mean[:, None]) * (rstd[:, None] * g2[None, :])).to(tl.bfloat16)

    if HAS_MASK:
        mrow = tl.load(mask_ptr + offs_m, mask=row_ok, other=0.0).to(tl.float32)

    for n0 in range(0, H, BLOCK_N):
        offs_n = n0 + tl.arange(0, BLOCK_N)
        n_ok = offs_n < H
        w0_mask = n_ok[None, :] & c0_ok[:, None]
        w1_mask = n_ok[None, :] & c1_ok[:, None]
        w2_mask = n_ok[None, :] & c2_ok[:, None]
        store_mask = row_ok[:, None] & n_ok[None, :]

        # left
        wl0 = tl.load(wl_ptr + offs_n[None, :] * C + c0_off[:, None], mask=w0_mask, other=0.0)
        wl1 = tl.load(wl_ptr + offs_n[None, :] * C + c1_off[:, None], mask=w1_mask, other=0.0)
        wl2 = tl.load(wl_ptr + offs_n[None, :] * C + c2_off[:, None], mask=w2_mask, other=0.0)
        wlg0 = tl.load(wlg_ptr + offs_n[None, :] * C + c0_off[:, None], mask=w0_mask, other=0.0)
        wlg1 = tl.load(wlg_ptr + offs_n[None, :] * C + c1_off[:, None], mask=w1_mask, other=0.0)
        wlg2 = tl.load(wlg_ptr + offs_n[None, :] * C + c2_off[:, None], mask=w2_mask, other=0.0)
        bwl = tl.load(bwl_ptr + offs_n, mask=n_ok, other=0.0).to(tl.float32)
        bwlg = tl.load(bwlg_ptr + offs_n, mask=n_ok, other=0.0).to(tl.float32)

        acc_l = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_l = tl.dot(a0, wl0, acc_l)
        acc_l = tl.dot(a1, wl1, acc_l)
        acc_l = tl.dot(a2, wl2, acc_l)

        acc_gl = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_gl = tl.dot(a0, wlg0, acc_gl)
        acc_gl = tl.dot(a1, wlg1, acc_gl)
        acc_gl = tl.dot(a2, wlg2, acc_gl)

        val_l = (acc_l + bwl[None, :]) * _sigmoid(acc_gl + bwlg[None, :])
        if HAS_MASK:
            val_l = val_l * mrow[:, None]
        tl.store(out_l_ptr + offs_m[:, None] * H + offs_n[None, :], val_l.to(tl.bfloat16), mask=store_mask)

        # right
        wr0 = tl.load(wr_ptr + offs_n[None, :] * C + c0_off[:, None], mask=w0_mask, other=0.0)
        wr1 = tl.load(wr_ptr + offs_n[None, :] * C + c1_off[:, None], mask=w1_mask, other=0.0)
        wr2 = tl.load(wr_ptr + offs_n[None, :] * C + c2_off[:, None], mask=w2_mask, other=0.0)
        wrg0 = tl.load(wrg_ptr + offs_n[None, :] * C + c0_off[:, None], mask=w0_mask, other=0.0)
        wrg1 = tl.load(wrg_ptr + offs_n[None, :] * C + c1_off[:, None], mask=w1_mask, other=0.0)
        wrg2 = tl.load(wrg_ptr + offs_n[None, :] * C + c2_off[:, None], mask=w2_mask, other=0.0)
        bwr = tl.load(bwr_ptr + offs_n, mask=n_ok, other=0.0).to(tl.float32)
        bwrg = tl.load(bwrg_ptr + offs_n, mask=n_ok, other=0.0).to(tl.float32)

        acc_r = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_r = tl.dot(a0, wr0, acc_r)
        acc_r = tl.dot(a1, wr1, acc_r)
        acc_r = tl.dot(a2, wr2, acc_r)

        acc_gr = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_gr = tl.dot(a0, wrg0, acc_gr)
        acc_gr = tl.dot(a1, wrg1, acc_gr)
        acc_gr = tl.dot(a2, wrg2, acc_gr)

        val_r = (acc_r + bwr[None, :]) * _sigmoid(acc_gr + bwrg[None, :])
        if HAS_MASK:
            val_r = val_r * mrow[:, None]
        tl.store(out_r_ptr + offs_m[:, None] * H + offs_n[None, :], val_r.to(tl.bfloat16), mask=store_mask)

        # out_gate
        wog0 = tl.load(wog_ptr + offs_n[None, :] * C + c0_off[:, None], mask=w0_mask, other=0.0)
        wog1 = tl.load(wog_ptr + offs_n[None, :] * C + c1_off[:, None], mask=w1_mask, other=0.0)
        wog2 = tl.load(wog_ptr + offs_n[None, :] * C + c2_off[:, None], mask=w2_mask, other=0.0)
        bwog = tl.load(bwog_ptr + offs_n, mask=n_ok, other=0.0).to(tl.float32)

        acc_g = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_g = tl.dot(a0, wog0, acc_g)
        acc_g = tl.dot(a1, wog1, acc_g)
        acc_g = tl.dot(a2, wog2, acc_g)

        val_g = _sigmoid(acc_g + bwog[None, :])
        tl.store(out_g_ptr + offs_m[:, None] * H + offs_n[None, :], val_g.to(tl.bfloat16), mask=store_mask)


@triton.jit
def init_ln_all_chunk_kernel(
    x_ptr,
    mask_ptr,
    wl_ptr,
    wr_ptr,
    wlg_ptr,
    wrg_ptr,
    wog_ptr,
    bwl_ptr,
    bwr_ptr,
    bwlg_ptr,
    bwrg_ptr,
    bwog_ptr,
    gamma_ptr,
    out_l_ptr,
    out_r_ptr,
    out_g_ptr,
    M,
    C,
    H,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    EPS: tl.constexpr,
):
    """
    General fallback for larger C: computes LN statistics, then fused GEMMs
    for left/right/out-gate in chunks.
    """
    pid = tl.program_id(0)
    num_n = tl.cdiv(H, BLOCK_N)
    pid_n = pid % num_n
    pid_m = pid // num_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    row_ok = offs_m < M

    mean = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sumsq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    for k0 in range(0, C, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        k_ok = offs_k < C
        xmask = row_ok[:, None] & k_ok[None, :]
        xk = tl.load(x_ptr + offs_m[:, None] * C + offs_k[None, :], mask=xmask, other=0.0).to(tl.float32)
        mean += tl.sum(xk, axis=1)
        sumsq += tl.sum(xk * xk, axis=1)

    mean = mean / C
    var = sumsq / C - mean * mean
    rstd = 1.0 / tl.sqrt(tl.maximum(var, 0.0) + EPS)

    if HAS_MASK:
        mrow = tl.load(mask_ptr + offs_m, mask=row_ok, other=0.0).to(tl.float32)

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    n_ok = offs_n < H
    store_mask = row_ok[:, None] & n_ok[None, :]

    # Left + right in one K sweep.
    acc_l = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_gl = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_r = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_gr = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k0 in range(0, C, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        k_ok = offs_k < C
        xmask = row_ok[:, None] & k_ok[None, :]
        xk = tl.load(x_ptr + offs_m[:, None] * C + offs_k[None, :], mask=xmask, other=0.0).to(tl.float32)

        gk = tl.load(gamma_ptr + offs_k, mask=k_ok, other=0.0).to(tl.float32)
        a = ((xk - mean[:, None]) * (rstd[:, None] * gk[None, :])).to(tl.bfloat16)

        wmask = n_ok[None, :] & k_ok[:, None]

        wl = tl.load(wl_ptr + offs_n[None, :] * C + offs_k[:, None], mask=wmask, other=0.0)
        acc_l = tl.dot(a, wl, acc_l)

        wlg = tl.load(wlg_ptr + offs_n[None, :] * C + offs_k[:, None], mask=wmask, other=0.0)
        acc_gl = tl.dot(a, wlg, acc_gl)

        wr = tl.load(wr_ptr + offs_n[None, :] * C + offs_k[:, None], mask=wmask, other=0.0)
        acc_r = tl.dot(a, wr, acc_r)

        wrg = tl.load(wrg_ptr + offs_n[None, :] * C + offs_k[:, None], mask=wmask, other=0.0)
        acc_gr = tl.dot(a, wrg, acc_gr)

    bwl = tl.load(bwl_ptr + offs_n, mask=n_ok, other=0.0).to(tl.float32)
    bwlg = tl.load(bwlg_ptr + offs_n, mask=n_ok, other=0.0).to(tl.float32)
    bwr = tl.load(bwr_ptr + offs_n, mask=n_ok, other=0.0).to(tl.float32)
    bwrg = tl.load(bwrg_ptr + offs_n, mask=n_ok, other=0.0).to(tl.float32)

    val_l = (acc_l + bwl[None, :]) * _sigmoid(acc_gl + bwlg[None, :])
    val_r = (acc_r + bwr[None, :]) * _sigmoid(acc_gr + bwrg[None, :])

    if HAS_MASK:
        val_l = val_l * mrow[:, None]
        val_r = val_r * mrow[:, None]

    tl.store(out_l_ptr + offs_m[:, None] * H + offs_n[None, :], val_l.to(tl.bfloat16), mask=store_mask)
    tl.store(out_r_ptr + offs_m[:, None] * H + offs_n[None, :], val_r.to(tl.bfloat16), mask=store_mask)

    # Out gate.
    acc_g = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k0 in range(0, C, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        k_ok = offs_k < C
        xmask = row_ok[:, None] & k_ok[None, :]
        xk = tl.load(x_ptr + offs_m[:, None] * C + offs_k[None, :], mask=xmask, other=0.0).to(tl.float32)

        gk = tl.load(gamma_ptr + offs_k, mask=k_ok, other=0.0).to(tl.float32)
        a = ((xk - mean[:, None]) * (rstd[:, None] * gk[None, :])).to(tl.bfloat16)

        wmask = n_ok[None, :] & k_ok[:, None]
        wog = tl.load(wog_ptr + offs_n[None, :] * C + offs_k[:, None], mask=wmask, other=0.0)
        acc_g = tl.dot(a, wog, acc_g)

    bwog = tl.load(bwog_ptr + offs_n, mask=n_ok, other=0.0).to(tl.float32)
    val_g = _sigmoid(acc_g + bwog[None, :])
    tl.store(out_g_ptr + offs_m[:, None] * H + offs_n[None, :], val_g.to(tl.bfloat16), mask=store_mask)


@triton.jit
def contraction_kernel(
    l_ptr,
    r_ptr,
    o_ptr,
    N,
    H,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP: tl.constexpr,
):
    """
    out[b,i,j,d] = sum_k L[b,i,k,d] * R[b,j,k,d]

    L/R/O are flattened as [B*N*N, H] and laid out so that a fixed (b,d) gives
    an N x N matrix over (row, k) or (row, j) with row stride N*H and k stride H.
    """
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    num_ij = num_i * num_j

    pid = tl.program_id(0)

    d = pid % H
    tmp = pid // H
    b = tmp // num_ij
    pid_ij = tmp % num_ij

    # L2-friendly swizzle over the I/J tile grid.
    group_id = pid_ij // (GROUP * num_j)
    first_j = group_id * GROUP
    group_size = tl.minimum(num_j - first_j, GROUP)
    j = first_j + (pid_ij % group_size)
    i = (pid_ij % (GROUP * num_j)) // group_size

    offs_i = i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = j * BLOCK_J + tl.arange(0, BLOCK_J)
    i_ok = offs_i < N
    j_ok = offs_j < N

    base = b * (N * N * H) + d
    row_stride = N * H

    offs_k = tl.arange(0, BLOCK_K)
    l_ptrs = l_ptr + base + offs_i[:, None] * row_stride + offs_k[None, :] * H
    r_ptrs = r_ptr + base + offs_k[:, None] * H + offs_j[None, :] * row_stride

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    for k0 in range(0, N, BLOCK_K):
        k_ok = (k0 + offs_k) < N

        l = tl.load(l_ptrs, mask=i_ok[:, None] & k_ok[None, :], other=0.0)
        r = tl.load(r_ptrs, mask=k_ok[:, None] & j_ok[None, :], other=0.0)

        acc = tl.dot(l, r, acc)

        l_ptrs += BLOCK_K * H
        r_ptrs += BLOCK_K * H

    o_ptrs = o_ptr + base + offs_i[:, None] * row_stride + offs_j[None, :] * H
    tl.store(o_ptrs, acc.to(o_ptr.dtype.element_ty), mask=i_ok[:, None] & j_ok[None, :])


@triton.jit
def final_ln_gate_linear_h128_kernel(
    out_ptr,
    gate_ptr,
    w_ptr,
    gamma_ptr,
    beta_ptr,
    betaw_ptr,
    res_ptr,
    M,
    H,
    C,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_C: tl.constexpr,
    EPS: tl.constexpr,
):
    """
    Fused final LayerNorm + gate + linear for H <= 128.
    Loads each row of `out` and `gate` only once, then loops over output C.
    """
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    row_ok = offs_m < M

    offs_h = tl.arange(0, BLOCK_H)
    h_ok = offs_h < H

    load_mask = row_ok[:, None] & h_ok[None, :]
    o = tl.load(out_ptr + offs_m[:, None] * H + offs_h[None, :], mask=load_mask, other=0.0).to(tl.float32)
    g = tl.load(gate_ptr + offs_m[:, None] * H + offs_h[None, :], mask=load_mask, other=0.0).to(tl.float32)

    mean = tl.sum(o, axis=1) / H
    sumsq = tl.sum(o * o, axis=1) / H
    var = sumsq - mean * mean
    rstd = 1.0 / tl.sqrt(tl.maximum(var, 0.0) + EPS)

    gamma = tl.load(gamma_ptr + offs_h, mask=h_ok, other=0.0).to(tl.float32)
    beta = tl.load(beta_ptr + offs_h, mask=h_ok, other=0.0).to(tl.float32)

    a = ((o - mean[:, None]) * (rstd[:, None] * gamma[None, :]) + beta[None, :]) * g
    a = a.to(tl.bfloat16)

    for c0 in range(0, C, BLOCK_C):
        offs_c = c0 + tl.arange(0, BLOCK_C)
        c_ok = offs_c < C

        wmask = h_ok[:, None] & c_ok[None, :]
        w = tl.load(w_ptr + offs_h[:, None] + offs_c[None, :] * H, mask=wmask, other=0.0)

        acc = tl.zeros((BLOCK_M, BLOCK_C), dtype=tl.float32)
        acc = tl.dot(a, w, acc)

        bw = tl.load(betaw_ptr + offs_c, mask=c_ok, other=0.0).to(tl.float32)
        acc += bw[None, :]

        st_mask = row_ok[:, None] & c_ok[None, :]
        tl.store(res_ptr + offs_m[:, None] * C + offs_c[None, :], acc, mask=st_mask)


@triton.jit
def final_ln_gate_linear_chunk_kernel(
    out_ptr,
    gate_ptr,
    w_ptr,
    gamma_ptr,
    beta_ptr,
    betaw_ptr,
    res_ptr,
    M,
    H,
    C,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_C: tl.constexpr,
    EPS: tl.constexpr,
):
    """
    General final LayerNorm + gate + linear for arbitrary H.
    """
    pid = tl.program_id(0)

    num_c = tl.cdiv(C, BLOCK_C)
    pid_c = pid % num_c
    pid_m = pid // num_c

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    row_ok = offs_m < M

    mean = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sumsq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    for k0 in range(0, H, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        k_ok = offs_k < H
        o = tl.load(out_ptr + offs_m[:, None] * H + offs_k[None, :], mask=row_ok[:, None] & k_ok[None, :], other=0.0).to(tl.float32)
        mean += tl.sum(o, axis=1)
        sumsq += tl.sum(o * o, axis=1)

    mean = mean / H
    var = sumsq / H - mean * mean
    rstd = 1.0 / tl.sqrt(tl.maximum(var, 0.0) + EPS)

    offs_c = pid_c * BLOCK_C + tl.arange(0, BLOCK_C)
    c_ok = offs_c < C

    acc = tl.zeros((BLOCK_M, BLOCK_C), dtype=tl.float32)

    for k0 in range(0, H, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        k_ok = offs_k < H

        o = tl.load(out_ptr + offs_m[:, None] * H + offs_k[None, :], mask=row_ok[:, None] & k_ok[None, :], other=0.0).to(tl.float32)
        g = tl.load(gate_ptr + offs_m[:, None] * H + offs_k[None, :], mask=row_ok[:, None] & k_ok[None, :], other=0.0).to(tl.float32)
        gamma_k = tl.load(gamma_ptr + offs_k, mask=k_ok, other=0.0).to(tl.float32)
        beta_k = tl.load(beta_ptr + offs_k, mask=k_ok, other=0.0).to(tl.float32)

        a = ((o - mean[:, None]) * (rstd[:, None] * gamma_k[None, :]) + beta_k[None, :]) * g
        a = a.to(tl.bfloat16)

        wmask = k_ok[:, None] & c_ok[None, :]
        w = tl.load(w_ptr + offs_k[:, None] + offs_c[None, :] * H, mask=wmask, other=0.0)

        acc = tl.dot(a, w, acc)

    bw = tl.load(betaw_ptr + offs_c, mask=c_ok, other=0.0).to(tl.float32)
    acc += bw[None, :]

    st_mask = row_ok[:, None] & c_ok[None, :]
    tl.store(res_ptr + offs_m[:, None] * C + offs_c[None, :], acc, mask=st_mask)


def custom_kernel(data):
    if len(data) == 3:
        x, weights, config = data
        mask = None
    else:
        x, mask, weights, config = data

    if not x.is_contiguous():
        x = x.contiguous()

    B, N, _, C = x.shape
    M = B * N * N
    H = config.get("hidden_dim", weights["left_proj.weight"].shape[0])
    device = x.device

    has_mask = (mask is not None) and (not bool(config.get("nomask", False)))
    if has_mask:
        mask_arg = mask.to(torch.float32).reshape(-1).contiguous()
    else:
        mask_arg = torch.zeros(1, dtype=torch.float32, device=device)

    # Normalize-layer parameters.
    gamma1 = weights["norm.weight"].float().contiguous()
    beta1 = weights["norm.bias"].float().contiguous()

    # Prepare initial linear weights: bf16 for tl.dot, fp32 betaW epilogue.
    def prep_linear(weight):
        W32 = weight.to(torch.float32).contiguous()
        Wbf = W32.to(torch.bfloat16).contiguous()
        betaW = (beta1[None, :] * W32).sum(dim=1).float().contiguous()
        return Wbf, betaW

    Wl, bwl = prep_linear(weights["left_proj.weight"])
    Wr, bwr = prep_linear(weights["right_proj.weight"])
    Wlg, bwlg = prep_linear(weights["left_gate.weight"])
    Wrg, bwrg = prep_linear(weights["right_gate.weight"])
    Wog, bwog = prep_linear(weights["out_gate.weight"])

    # Output-layer parameters.
    gamma2 = weights["to_out_norm.weight"].float().contiguous()
    beta2 = weights["to_out_norm.bias"].float().contiguous()

    Wout32 = weights["to_out.weight"].to(torch.float32).contiguous()
    Wout = Wout32.to(torch.bfloat16).contiguous()
    betaWout = (Wout32 * beta2[None, :]).sum(dim=1).float().contiguous()

    # Intermediate buffers.
    left = torch.empty(M, H, dtype=torch.bfloat16, device=device)
    right = torch.empty(M, H, dtype=torch.bfloat16, device=device)
    gate = torch.empty(M, H, dtype=torch.bfloat16, device=device)
    mid = torch.empty(M, H, dtype=torch.bfloat16, device=device)
    out = torch.empty(B, N, N, C, dtype=torch.float32, device=device)

    # 1) Fused input LayerNorm + first projections/gates.
    if C <= 128:
        BLOCK_M = 32
        BLOCK_N = 128
        grid = (triton.cdiv(M, BLOCK_M),)
        init_ln_all_c128_kernel[grid](
            x, mask_arg,
            Wl, Wr, Wlg, Wrg, Wog,
            bwl, bwr, bwlg, bwrg, bwog,
            gamma1,
            left, right, gate,
            M, C, H,
            HAS_MASK=has_mask,
            BLOCK_M=BLOCK_M,
            BLOCK_C=128,
            BLOCK_N=BLOCK_N,
            EPS=1e-5,
            num_warps=4,
        )
    elif C <= 384:
        BLOCK_M = 16
        BLOCK_N = 128
        grid = (triton.cdiv(M, BLOCK_M),)
        init_ln_all_c384_kernel[grid](
            x, mask_arg,
            Wl, Wr, Wlg, Wrg, Wog,
            bwl, bwr, bwlg, bwrg, bwog,
            gamma1,
            left, right, gate,
            M, C, H,
            HAS_MASK=has_mask,
            BLOCK_M=BLOCK_M,
            BLOCK_C=128,
            BLOCK_N=BLOCK_N,
            EPS=1e-5,
            num_warps=4,
        )
    else:
        BLOCK_M = 16
        BLOCK_N = 128
        BLOCK_K = 64
        grid = (triton.cdiv(M, BLOCK_M) * triton.cdiv(H, BLOCK_N),)
        init_ln_all_chunk_kernel[grid](
            x, mask_arg,
            Wl, Wr, Wlg, Wrg, Wog,
            bwl, bwr, bwlg, bwrg, bwog,
            gamma1,
            left, right, gate,
            M, C, H,
            HAS_MASK=has_mask,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            EPS=1e-5,
            num_warps=4,
        )

    # 2) Triangular contraction.
    if N <= 128:
        BI = BJ = 32
        BK = 32 if N < 64 else 64
        num_warps_c = 4
    elif N <= 256:
        BI = BJ = 64
        BK = 64
        num_warps_c = 4
    else:
        BI = BJ = 128
        BK = 64
        num_warps_c = 8

    grid = (B * H * triton.cdiv(N, BI) * triton.cdiv(N, BJ),)
    contraction_kernel[grid](
        left, right, mid,
        N, H,
        BLOCK_I=BI,
        BLOCK_J=BJ,
        BLOCK_K=BK,
        GROUP=8,
        num_warps=num_warps_c,
    )

    # 3) Fused final LayerNorm + gate + linear.
    if H <= 128:
        BLOCK_M = 32
        grid = (triton.cdiv(M, BLOCK_M),)
        final_ln_gate_linear_h128_kernel[grid](
            mid, gate, Wout, gamma2, beta2, betaWout, out,
            M, H, C,
            BLOCK_M=BLOCK_M,
            BLOCK_H=128,
            BLOCK_C=128,
            EPS=1e-5,
            num_warps=4,
        )
    else:
        BLOCK_M = 16
        BLOCK_K = 128
        BLOCK_C = 128
        grid = (triton.cdiv(M, BLOCK_M) * triton.cdiv(C, BLOCK_C),)
        final_ln_gate_linear_chunk_kernel[grid](
            mid, gate, Wout, gamma2, beta2, betaWout, out,
            M, H, C,
            BLOCK_M=BLOCK_M,
            BLOCK_K=BLOCK_K,
            BLOCK_C=BLOCK_C,
            EPS=1e-5,
            num_warps=4,
        )

    return out
```
