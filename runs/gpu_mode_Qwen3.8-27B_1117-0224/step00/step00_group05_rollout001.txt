```python
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Grid: B * (N/BLOCK_I) * (N/BLOCK_J) * H
    pid = tl.program_id(0)
    
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    
    # Decode pid
    pid_j = pid % num_j
    pid_temp = pid // num_j
    pid_i = pid_temp % num_i
    pid_temp = pid_temp // num_i
    pid_bh = pid_temp
    
    num_b = B  # We need B passed in
    pid_h = pid_bh % H
    pid_b = pid_bh // H
    
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)
    
    # Pointers
    left_ptrs = left_ptr + pid_b * stride_lb + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk + pid_h * stride_lh
    right_ptrs = right_ptr + pid_b * stride_rb + offs_j[:, None] * stride_rj + offs_k[None, :] * stride_rk + pid_h * stride_rh
    
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)
    
    for k_start in range(0, N, BLOCK_K):
        k_mask = (k_start + offs_k) < N
        a = tl.load(left_ptrs, mask=k_mask[None, :], other=0.0)
        b = tl.load(right_ptrs, mask=k_mask[None, :], other=0.0)
        acc = tl.dot(a, tl.trans(b), acc)
        left_ptrs += BLOCK_K * stride_lk
        right_ptrs += BLOCK_K * stride_rk
    
    # Store
    out_ptrs = out_ptr + pid_b * stride_ob + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj + pid_h * stride_oh
    out_mask = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(out_ptrs, acc, mask=out_mask)
```

```python
import torch
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_lh, stride_li, stride_lk,
    stride_rb, stride_rh, stride_rj, stride_rk,
    stride_ob, stride_oh, stride_oi, stride_oj,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    
    pid_j = pid % num_j
    pid_temp = pid // num_j
    pid_i = pid_temp % num_i
    pid_bh = pid_temp // num_i
    
    pid_h = pid_bh % H
    pid_b = pid_bh // H
    
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)
    
    # Strides for [B, H, N, N] layout
    # left[b, h, i, k]
    left_ptrs = left_ptr + pid_b * stride_lb + pid_h * stride_lh + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
    # right[b, h, j, k]
    right_ptrs = right_ptr + pid_b * stride_rb + pid_h * stride_rh + offs_j[:, None] * stride_rj + offs_k[None, :] * stride_rk
    
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)
    
    i_mask = offs_i < N
    j_mask = offs_j < N
    
    for k_start in range(0, N, BLOCK_K):
        k_mask = (k_start + offs_k) < N
        a = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        b_mat = tl.load(right_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0)
        acc = tl.dot(a, tl.trans(b_mat), acc)
        left_ptrs += BLOCK_K * stride_lk
        right_ptrs += BLOCK_K * stride_rk
    
    out_ptrs = out_ptr + pid_b * stride_ob + pid_h * stride_oh + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    out_mask = i_mask[:, None] & j_mask[None, :]
```

```python
    hidden_dim = config["hidden_dim"]

    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, keepdim=True, unbiased=False)
```

```python
import torch
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    total_batch, N, H,
    stride_lb, stride_lh, stride_li, stride_lk,
    stride_rb, stride_rh, stride_rj, stride_rk,
    stride_ob, stride_oh, stride_oi, stride_oj,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    
    # Decode: pid → (batch, i_block, j_block)
    pid_j = pid % num_j
    pid_temp = pid // num_j
    pid_i = pid_temp % num_i
    pid_bh = pid_temp // num_i
    
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)
    
    # Build pointers
    left_ptrs = left_ptr + pid_bh * stride_lb + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
    right_ptrs = right_ptr + pid_bh * stride_rb + offs_j[:, None] * stride_rj + offs_k[None, :] * stride_rk
    
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)
    
    i_mask = offs_i < N
    j_mask = offs_j < N
    
    for k_start in range(0, N, BLOCK_K):
        k_mask = (k_start + offs_k) < N
        a = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        b_mat = tl.load(right_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0)
        acc = tl.dot(a, tl.trans(b_mat), acc)
        left_ptrs += BLOCK_K * stride_lk
        right_ptrs += BLOCK_K * stride_rk
    
    out_ptrs = out_ptr + pid_bh * stride_ob + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    out_mask = i_mask[:, None] & j_mask[None, :]
    tl.store(out_ptrs, acc.to(tl.bfloat16), mask=out_mask)
```

```python
"""
TriMul (Triangle Multiplicative Update) "outgoing" kernel implementation.
The main O(N^3) einsum operation is implemented as a batched matmul in Triton
with the tensor permuted to [B, H, N, N] layout for coalesced memory access.
LayerNorm, projections, and gating are done in PyTorch.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _trimul_batched_matmul(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_lb, stride_li, stride_lk,
    stride_rb, stride_rj, stride_rk,
    stride_ob, stride_oi, stride_oj,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[bh, i, j] = sum_k left[bh, i, k] * right[bh, j, k]
    where bh = b * H + h is the flattened batch index.
    
    left:  [B*H, N, N] (row-major, strides: [N*N, N, 1])
    right: [B*H, N, N] (row-major, strides: [N*N, N, 1])
    out:   [B*H, N, N] (row-major, strides: [N*N, N, 1])
    """
    pid = tl.program_id(0)

    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)

    pid_j = pid % num_j
    pid_temp = pid // num_j
    pid_i = pid_temp % num_i
    pid_bh = pid_temp // num_i

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers into the batched matrices
    left_ptrs = left_ptr + pid_bh * stride_lb + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
    right_ptrs = right_ptr + pid_bh * stride_rb + offs_j[:, None] * stride_rj + offs_k[None, :] * stride_rk

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    i_mask = offs_i < N
    j_mask = offs_j < N

    for k_start in range(0, N, BLOCK_K):
        k_mask = (k_start + offs_k) < N
        a = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        b_mat = tl.load(right_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0)
        acc = tl.dot(a, tl.trans(b_mat), acc)
        left_ptrs += BLOCK_K * stride_lk
        right_ptrs += BLOCK_K * stride_rk

    out_ptrs = out_ptr + pid_bh * stride_ob + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    out_mask = i_mask[:, None] & j_mask[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


def custom_kernel(data):
    """
    TriMul forward pass with the O(N^3) outgoing update implemented in Triton.
    
    Args:
        data: (input_tensor, mask, weights, config)
    Returns:
        output tensor of shape [B, N, N, dim] in float32
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape
    device = input_tensor.device

    # --- PyTorch: LayerNorm + projections + masking + gating ---
    x = input_tensor.float()
    norm_w = weights["norm.weight"].float()
    norm_b = weights["norm.bias"].float()

    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, keepdim=True, unbiased=False)
    x_norm = (x - mean) / torch.sqrt(var + 1e-5)
    x_norm = x_norm * norm_w + norm_b

    left = torch.matmul(x_norm, weights["left_proj.weight"].float().T)
    right = torch.matmul(x_norm, weights["right_proj.weight"].float().T)

    if mask is not None:
        mask_f = mask.float().unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    left_gate = torch.sigmoid(torch.matmul(x_norm, weights["left_gate.weight"].float().T))
    right_gate = torch.sigmoid(torch.matmul(x_norm, weights["right_gate.weight"].float().T))
    out_gate = torch.sigmoid(torch.matmul(x_norm, weights["out_gate_weight"].float().T))

    left = left * left_gate
    right = right * right_gate

    # --- Permute to [B, H, N, N] and convert to bfloat16 for tensor cores ---
    left_perm = left.permute(0, 3, 1, 2).contiguous().to(torch.bfloat16)   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous().to(torch.bfloat16) # [B, H, N, N]

    total_batch = B * hidden_dim
    out_perm = torch.empty(total_batch, N, N, device=device, dtype=torch.float32)

    # --- Triton kernel: batched matmul out[bh] = left[bh] @ right[bh]^T ---
    BLOCK_I = 128
    BLOCK_J = 128
    BLOCK_K = 64

    num_i = triton.cdiv(N, BLOCK_I)
    num_j = triton.cdiv(N, BLOCK_J)
    grid = (total_batch * num_i * num_j,)

    _trimul_batched_matmul[grid](
        left_perm, right_perm, out_perm,
        N,
        left_perm.stride(0), left_perm.stride(1), left_perm.stride(2),
        right_perm.stride(0), right_perm.stride(1), right_perm.stride(2),
        out_perm.stride(0), out_perm.stride(1), out_perm.stride(2),
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # --- PyTorch: reshape back, final LayerNorm, gating, projection ---
    out = out_perm.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    to_out_norm_w = weights["to_out_norm.weight"].float()
    to_out_norm_b = weights["to_out_norm.bias"].float()
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), to_out_norm_w, to_out_norm_b, eps=1e-5)

    out = out * out_gate
    out = torch.matmul(out, weights["to_out_weight"].float().T)

    return out.float()
```
