import torch
import triton
import triton.language as tl


"""
TriMul forward (outgoing) implementation.

Pipeline:
1. Triton kernel:
   - LayerNorm over the final channel dimension
   - 5 projections + gates in one pass
   - applies mask and sigmoid gates
   - writes left/right in batched-GEMM-friendly layout [B, H, N, N]
   - writes out_gate in token-major layout [B, N*N, H]
2. cuBLAS batched matrix multiply:
   - out_h[b,h,i,j] = sum_k left[b,h,i,k] * right[b,h,j,k]
3. Triton kernel:
   - LayerNorm over H for each (b,i,j)
   - multiply by out_gate
   - final Linear(H -> D)
   - store float32 output
"""


def _next_pow2(n: int) -> int:
    return 1 << (max(1, n) - 1).bit_length()


@triton.jit
def _fused_ln_proj_gate_kernel(
    x_ptr,
    mask_ptr,
    out_l_ptr,
    out_r_ptr,
    out_g_ptr,
    gamma_ptr,
    beta_ptr,
    wl0_ptr,
    wl1_ptr,
    wl2_ptr,
    wr0_ptr,
    wr1_ptr,
    wr2_ptr,

    wo1_ptr,
    wo2_ptr,

    D1: tl.constexpr,

    HAS_MASK: tl.constexpr,

            x_ptr + offs_m[:, None] * D + off_d0[None, :],
            mask=valid[:, None] & d_mask[None, :],
            other=0.0,
        ).to(tl.float32)

        b0 = tl.load(beta_ptr + off_d0, mask=d_mask, other=0.0).to(tl.float32)

        mean = tl.sum(x0, axis=1) / D
        v0 = x0 - mean[:, None]
        var = tl.sum(v0 * v0 * dm_f[None, :], axis=1)
        var = tl.maximum(var, 0.0)

        left_t = tl.trans(left)

        tl.store(out_l_ptrs, left_t, mask=valid[None, :])

        right_t = tl.trans(right)

        w_lg1 = tl.load(wl1_ptr + off_d1[:, None] * N_LG + h_lg_off[None, :])
        acc_lg = tl.dot(n0, w_lg0)
        acc_lg = tl.dot(n1, w_lg1, acc_lg)
        acc_lg_3 = tl.reshape(acc_lg, (BLOCK_M, H, 2))
        l_acc, lg_acc = tl.split(acc_lg_3)
        left = l_acc * mvals[:, None] * (1.0 / (1.0 + tl.exp(-lg_acc)))
        left_t = tl.trans(left)
        out_l_ptrs = out_l_ptr + (b_id[None, :] * H + h_off[:, None]) * M_batch + token[None, :]
        tl.store(out_l_ptrs, left_t, mask=valid[None, :])

        acc_rg = tl.dot(n1, w_rg1, acc_rg)

        g1 = tl.load(gamma_ptr + D0 + off_d1).to(tl.float32)
        b1 = tl.load(beta_ptr + D0 + off_d1).to(tl.float32)

@triton.jit
def _final_ln_gate_linear_kernel(
    out_h_ptr,
    gate_ptr,
    out_ptr,
    gamma_ptr,
    beta_ptr,
    w_final_ptr,
    M_total,
    M_batch,
    D,
    EPS,
    H: tl.constexpr,
    D_BLOCK: tl.constexpr,
    D_CHUNK: tl.constexpr,
    BLOCK_M: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    valid = offs_m < M_total

    b_id = offs_m // M_batch
    local = offs_m % M_batch

    h_off = tl.arange(0, H)

    # Load out_h in [H, BLOCK_M] for coalesced local token reads.
    data_ptrs = (
        out_h_ptr
        + b_id[None, :] * (H * M_batch)
        + h_off[:, None] * M_batch
        + local[None, :]
    )
    data = tl.load(data_ptrs, mask=valid[None, :], other=0.0).to(tl.float32)

    # LayerNorm over H.
    mean = tl.sum(data, axis=0) / H
    data_c = data - mean[None, :]
    var = tl.sum(data_c * data_c, axis=0)
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + EPS)

    g = tl.load(gamma_ptr + h_off).to(tl.float32)
    b = tl.load(beta_ptr + h_off).to(tl.float32)

    norm_h = data_c * rstd[None, :] * g[None, :] + b[None, :]  # [H, BLOCK_M]
    norm = tl.trans(norm_h)                                     # [BLOCK_M, H]

    # Load out_gate token-major.
    gate_ptrs = gate_ptr + (b_id[:, None] * M_batch + local[:, None]) * H + h_off[None, :]
    gate = tl.load(gate_ptrs, mask=valid[:, None], other=0.0).to(tl.float32)

    scaled = (norm * gate).to(tl.bfloat16)
    h_off_w = tl.arange(0, H)

    # Final linear in D chunks to keep shared-memory tile sizes moderate.
    for d_start in range(0, D_BLOCK, D_CHUNK):
        off_d = tl.arange(0, D_CHUNK) + d_start
        w = tl.load(w_final_ptr + h_off_w[:, None] * D_BLOCK + off_d[None, :])
        acc = tl.dot(scaled, w)

        out_ptrs = out_ptr + offs_m[:, None] * D + off_d[None, :]
        d_mask = off_d < D
        tl.store(out_ptrs, acc.to(tl.float32), mask=valid[:, None] & d_mask[None, :])


def _make_chunks(D: int):
    chunks = []
    rem = D
    maxc = 256
    while rem > 0:
        c = 1
        limit = min(maxc, rem)
        while (c << 1) <= limit:
            c <<= 1
        chunks.append(c)
        rem -= c

    # Keep at most 3 chunks in the main production kernel for the allowed
    # c_z range (<=768 gives exactly [256, 256, 256]).
    while len(chunks) > 3:
        s = chunks[-1] + chunks[-2]
        if s in (128, 256, 512, 1024):
            chunks[-2] = s
            chunks.pop()
        else:
            s = sum(chunks[2:])
            if (s & (s - 1)) == 0:
                chunks = chunks[:2] + [s]
            else:
                # Fallback for out-of-scope dimensions: make the last chunk
                # the remaining sum. This is not expected for the stated tests.
                chunks = chunks[:2] + [s]
            break

    return chunks

    norm_b = weights["norm.bias"].to(torch.float32, device=device).contiguous()

    left_w = weights["left_proj.weight"].to(dtype, device=device).T.contiguous()
    right_w = weights["right_proj.weight"].to(dtype, device=device).T.contiguous()
    left_g_w = weights["left_gate.weight"].to(dtype, device=device).T.contiguous()
    right_g_w = weights["right_gate.weight"].to(dtype, device=device).T.contiguous()
    out_g_w = weights["out_gate.weight"].to(dtype, device=device).T.contiguous()
    to_out_w = weights["to_out.weight"].to(dtype, device=device).T.contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float32, device=device).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float32, device=device).contiguous()

    def pair_weight(proj_w, gate_w):
        w = torch.empty((D, 2 * H), dtype=dtype, device=device)
        w[:, 0::2] = proj_w
        w[:, 1::2] = gate_w
        return w

    w_lg = pair_weight(left_w, left_g_w)
    w_rg = pair_weight(right_w, right_g_w)
    w_og = out_g_w.contiguous()

    chunks = _make_chunks(D)
    D0 = chunks[0] if len(chunks) > 0 else 1
    D1 = chunks[1] if len(chunks) > 1 else 0
    D2 = chunks[2] if len(chunks) > 2 else 0
    NCHUNK = min(3, len(chunks))

    def split_w(wmat, idx):
        if idx == 0:
            return wmat[:D0].contiguous()
        if idx == 1:
            if D1 == 0:
                return wmat[:1].contiguous()
            return wmat[D0:D0 + D1].contiguous()
        if D2 == 0:
            return wmat[:1].contiguous()
        return wmat[D0 + D1:D0 + D1 + D2].contiguous()

    wl0 = split_w(w_lg, 0)
    wl1 = split_w(w_lg, 1)
    wl2 = split_w(w_lg, 2)

    wr0 = split_w(w_rg, 0)
    wr1 = split_w(w_rg, 1)
    wr2 = split_w(w_rg, 2)

    wo0 = split_w(w_og, 0)
    wo1 = split_w(w_og, 1)
    wo2 = split_w(w_og, 2)

    if D <= 128:
        BLOCK_M_PROD = 128
    elif D <= 384:
        BLOCK_M_PROD = 64
    elif D <= 512:
        BLOCK_M_PROD = 32
    else:
        BLOCK_M_PROD = 16

    NUM_WARPS_PROD = 8 if BLOCK_M_PROD >= 32 else 4

    left = torch.empty((B * H, N, N), dtype=dtype, device=device)
    right = torch.empty((B * H, N, N), dtype=dtype, device=device)
    out_gate = torch.empty((M_total, H), dtype=dtype, device=device)

    grid_prod = (triton.cdiv(M_total, BLOCK_M_PROD),)
    _fused_ln_proj_gate_kernel[grid_prod](
        x_2d,
        mask_1d,
        left,
        right,
        out_gate,
        norm_w,
        norm_b,
        wl0,
        wl1,
        wl2,
        wr0,
        wr1,
        wr2,
        wo0,
        wo1,
        wo2,
        M_total,
        M_batch,
        D,
        EPS,
        NCHUNK=NCHUNK,
        D0=D0,
        D1=D1,
        D2=D2,
        H=H,
        HAS_MASK=has_mask,
        BLOCK_M=BLOCK_M_PROD,
        num_warps=NUM_WARPS_PROD,
    )

    # Bmm: [B*H, N, N] x [B*H, N, N]^T -> [B*H, N, N]
    A = left
    Bmat = right.transpose(1, 2)
    out_h = torch.bmm(A, Bmat)

    D_BLOCK = _next_pow2(D)
    D_CHUNK = 256 if D_BLOCK >= 256 else D_BLOCK

    if D <= 384:
        BLOCK_M_FINAL = 32
    else:
        BLOCK_M_FINAL = 16

    NUM_WARPS_FINAL = 8 if BLOCK_M_FINAL >= 32 else 4

    w_final = torch.zeros((H, D_BLOCK), dtype=dtype, device=device)
    w_final[:, :D] = to_out_w

    out = torch.empty((M_total, D), dtype=torch.float32, device=device)

    grid_final = (triton.cdiv(M_total, BLOCK_M_FINAL),)
    _final_ln_gate_linear_kernel[grid_final](
        out_h,
        out_gate,
        out,
        to_out_norm_w,
        to_out_norm_b,
        w_final,
        M_total,
        M_batch,
        D,
        EPS,
        H=H,
        D_BLOCK=D_BLOCK,
        D_CHUNK=D_CHUNK,
        BLOCK_M=BLOCK_M_FINAL,
        num_warps=NUM_WARPS_FINAL,
    )

    return out.view(B, N, N, D)
