import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _triuml_einsum(
    left_ptr, right_ptr, out_ptr,
    N: tl.constexpr, HD: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Fused einsum: out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    
    Implemented as batched GEMM: for each (b,d), C = A @ B^T
    where A = left[b,:,:,d] (N x N), B = right[b,:,:,d] (N x N).
    """
    pid = tl.program_id(0)
    num_j_blocks: tl.constexpr = N // BLOCK_N
    num_i_blocks: tl.constexpr = N // BLOCK_M

    j_block = pid % num_j_blocks
    tmp = pid // num_j_blocks
    i_block = tmp % num_i_blocks
    bd = tmp // num_i_blocks
    d = bd % HD
    b = bd // HD

    offs_m = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute base pointers for this (b, d) pair
    left_base = left_ptr + b * N * N * HD + d
    right_base = right_ptr + b * N * N * HD + d
    out_base = out_ptr + b * N * N * HD + d

    # A[i, k] = left[b, i, k, d]: row stride = N*HD, col stride = HD
    a_ptrs = left_base + offs_m[:, None] * (N * HD) + offs_k[None, :] * HD

    # B[k, j] = right[b, j, k, d]: row stride = HD, col stride = N*HD

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    norm_bias = weights["norm.bias"]

    right_gate_w = weights["right_gate.weight"]

    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    left = F.linear(x_norm, left_proj_w)
    right = F.linear(x_norm, right_proj_w)

        right = right * m

    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))

    right = right * right_gate

    right_bf = right.to(torch.bfloat16).contiguous()

    BLOCK_N = 128
    BLOCK_K = 64

    out = F.layer_norm(out, (hidden_dim,), to_out_norm_w, to_out_norm_b, eps=1e-5)
    out = out * out_gate
    out = F.linear(out, to_out_w)

"""
TriMul (Triangle Multiplicative Update) forward pass for AlphaFold3-style models.
Fuses LayerNorm, linear projections, sigmoid gating, and the O(N^3) pairwise
contraction (implemented as a batched GEMM in Triton with bf16 tensor cores).
The core einsum out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d] is
decomposed into B*hidden_dim independent GEMMs of size N x N x N.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _triuml_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    N: tl.constexpr, HD: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel computing out[b,i,j,d] = sum_k left[b,i,k,d]*right[b,j,k,d].
    
    For each (b,d): C[i,j] = A[i,k] * B[k,j] summed over k,
    where A = left[b,:,:,d], B^T = right[b,:,:,d].
    Memory layout: left/right are [B, N, N, HD], out is [B, N, N, HD].
    """
    pid = tl.program_id(0)
    num_j_blocks: tl.constexpr = N // BLOCK_N
    num_i_blocks: tl.constexpr = N // BLOCK_M

    j_block = pid % num_j_blocks
    tmp = pid // num_j_blocks
    i_block = tmp % num_i_blocks
    bd = tmp // num_i_blocks
    d = bd % HD
    b = bd // HD

    offs_m = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Base offset for this (b, d) slice
    # left[b, i, k, d] contiguous stride: b->N*N*HD, i->N*HD, k->HD, d->1
    left_base = left_ptr + b * N * N * HD + d
    right_base = right_ptr + b * N * N * HD + d
    out_base = out_ptr + b * N * N * HD + d

    # A[i, k] = left[b, i, k, d]  (row stride = N*HD, col stride = HD)
    a_ptrs = left_base + offs_m[:, None] * (N * HD) + offs_k[None, :] * HD

    # B[k, j] = right[b, j, k, d]  (row stride = HD, col stride = N*HD)
    b_ptrs = right_base + offs_n[None, :] * (N * HD) + offs_k[:, None] * HD

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for _ in range(N // BLOCK_K):
        a = tl.load(a_ptrs)
        bm = tl.load(b_ptrs)
        acc = tl.dot(a, bm, acc)
        a_ptrs += BLOCK_K * HD
        b_ptrs += BLOCK_K * HD

    # Store: out[b, i, j, d] (row stride = N*HD, col stride = HD)
    out_ptrs = out_base + offs_m[:, None] * (N * HD) + offs_n[None, :] * HD
    tl.store(out_ptrs, acc)

    """TriMul forward pass: LayerNorm -> projections -> mask/gate -> einsum -> LayerNorm -> gate -> out_proj."""
    input_tensor, mask, weights, config = data
    dim = config["dim"]

    B, N, N2, C = input_tensor.shape
    assert N == N2, "Input must have square sequence dimensions"

    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    x_norm = F.layer_norm(x, (dim,), norm_weight, norm_bias, eps=1e-5)

    left = F.linear(x_norm, left_proj_w)

    # Apply mask to zero out invalid positions
    if mask is not None:

    # Sigmoid gates

    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))

    right = right * right_gate

    # Core einsum: out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]

    out = out * out_gate
    out = F.linear(out, to_out_w)
