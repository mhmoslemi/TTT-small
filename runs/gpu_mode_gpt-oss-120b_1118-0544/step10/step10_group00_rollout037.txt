@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Compute indices
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    batch = pid_bh // H
    h = pid_bh % H

    i_start = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_start = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # create masks for out-of-range
    mask_i = i_start < N
    mask_j = j_start < N

    # offsets for output
    out_ptrs = out_ptr + batch * stride_o_batch + h * stride_o_h + i_start[:, None] * stride_o_i + j_start[None, :] * stride_o_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_start < N

        # load left tile (i,k)
        left_ptrs = left_ptr + batch * stride_l_batch + h * stride_l_h + i_start[:, None] * stride_l_i + k_start[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile transposed (k,j)
        right_ptrs = right_ptr + batch * stride_r_batch + h * stride_r_h + k_start[:, None] * stride_r_k + j_start[None, :] * stride_r_j

        # Compute dot: left (M,K) * right (K,N) -> (M,N)
        acc += tl.dot(left, right)

    # store result (cast to fp16)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

    right_out_ptr,   # fp16 *

    hidden_dim,      # i64 scalar = H

):

    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    All tensors are flattened in row‑major order.

    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    mask_idx = offs // hidden_dim

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr  + offs, left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,

    right_gate: torch.Tensor,
    mask: torch.Tensor,

) -> Tuple[torch.Tensor, torch.Tensor]:

@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]"""

    pid_i = tl.program_id(0)   # block row
    pid_j = tl.program_id(1)   # block col
    pid_bh = tl.program_id(2)  # batch * hidden index

    batch = pid_bh // H
    h = pid_bh % H

    i_base = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_base < N
    mask_j = j_base < N

    # pointer to output tile
    out_ptrs = out_ptr + batch * stride_o_batch + h * stride_o_h + \
               i_base[:, None] * stride_o_i + j_base[None, :] * stride_o_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_base < N

        # left tile (i, k)
        left_ptrs = left_ptr + batch * stride_l_batch + h * stride_l_h + \
                    i_base[:, None] * stride_l_i + k_base[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile transposed (k, j)
        right_ptrs = right_ptr + batch * stride_r_batch + h * stride_r_h + \
                     k_base[:, None] * stride_r_k + j_base[None, :] * stride_r_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # accumulate dot product
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N), float32

    # store result
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul(left: torch.Tensor, right: torch.Tensor,
             block_m: int = 32, block_n: int = 32, block_k: int = 32) -> torch.Tensor:
    """
    Batched triangle multiplication using a Triton GEMM.
    left, right: tensors of shape (B, N, N, H) and dtype float16.
    Returns tensor of the same shape and dtype.

    B, N, _, H = left.shape
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    stride_l = left.stride()
    stride_r = right.stride()
    stride_o = out.stride()

    grid = (
        triton.cdiv(N, block_m),
        triton.cdiv(N, block_n),

        left,
        right,
        out,

        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return out
