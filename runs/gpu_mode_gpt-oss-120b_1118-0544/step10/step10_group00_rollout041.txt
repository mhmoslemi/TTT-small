@triton.jit
def tri_mul_kernel(

    a_ptr, b_ptr, c_ptr,

    M, N, K,

    stride_am, stride_ai, stride_ak,

    stride_bm, stride_bk, stride_bn,

    stride_cm, stride_cn, stride_ck,

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr

):
    pid_m = tl.program_id(0)  # batch*hidden
    pid_i = tl.program_id(1)  # block row
    pid_j = tl.program_id(2)  # block col

    # compute offsets
    offs_m = pid_m * BLOCK_M  # not used directly

    # compute the start of output block
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_i = i_offset < N
    mask_j = j_offset < N

    # pointer to output
    c_ptrs = c_ptr + pid_m * stride_cm + i_offset[:, None] * stride_cn + j_offset[None, :] * stride_ck

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # load a block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + pid_m * stride_am + i_offset[:, None] * stride_ai + k_offset[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load b block: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + pid_m * stride_bm + k_offset[:, None] * stride_bk + j_offset[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)  # a: (M,K), b: (K,N)

    # store result
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])

Actually for output we want to store as (M, N, N). So stride_cm = N*N, stride_cn = N, stride_ck = 1.

def tri_mul_torch(left, right):
    # left/right shape (B, N, N, H) fp16
    B, N, _, H = left.shape
    # reshape to (M, N, N)
    M = B * H
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(M, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(M, N, N)
    # allocate output
    out = torch.empty_like(left_flat)
    # get strides
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bk = right_flat.stride(2)  # K dimension (original K)
    stride_bn = right_flat.stride(1)  # N dimension

    stride_cm = out.stride(0)
    stride_cn = out.stride(1)
    stride_ck = out.stride(2)

    # Determine block sizes (maybe 64,64,32)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (M, tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N))
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out,
        M,
        N,
        N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bn,
        stride_cm, stride_cn, stride_ck,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # reshape back
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

"""
TriMul outgoing forward pass (AlphaFold3) using Triton.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (left/right) and gating (left_gate, right_gate, out_gate).
4. Optional mask fusion (element‑wise, performed in PyTorch for simplicity).
5. Core reduction (TriMul):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM kernel in Triton (the k‑dimension is reduced
   in tiles, accumulation is in fp32, result stored in fp16).
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   final linear projection back to `dim`.
7. Cast back to the original dtype.

The kernel processes the batch×hidden dimension as a single Triton axis,
so a launch grid of (M, ceil(N/BM), ceil(N/BN)) where M = B*H.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes
    stride_am, stride_ai, stride_ak,   # strides for A (left)
    stride_bm, stride_bk, stride_bn,   # strides for B (right, treated as transposed)
    stride_cm, stride_cn, stride_ck,   # strides for C (output)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for TriMul:
    C[m, i, j] = Σ_k A[m, i, k] * B[m, j, k]
    where A and B have shape (M, N, K) and C has shape (M, N, N).
    """
    pid_m = tl.program_id(0)  # batch * hidden
    pid_i = tl.program_id(1)  # row block
    pid_j = tl.program_id(2)  # col block

    # Offsets for the output tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # Pointer to the output tile
    c_ptrs = (
        c_ptr
        + pid_m * stride_cm
        + i[:, None] * stride_cn
        + j[None, :] * stride_ck
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < K

        # Load A block: (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + pid_m * stride_am
            + i[:, None] * stride_ai
            + k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B block (transposed): (BLOCK_K, BLOCK_N)
        b_ptrs = (
            b_ptr
            + pid_m * stride_bm
            + k[:, None] * stride_bk
            + j[None, :] * stride_bn
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N)

    # Write back the result in fp16
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched TriMul using the Triton kernel.
    left, right: (B, N, N, H) fp16
    Returns: (B, N, N, H) fp16
    """
    B, N, _, H = left.shape
    M = B * H

    # reshape to (M, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(M, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(M, N, N)

    out = torch.empty_like(left_flat)

    # strides for A
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    # strides for B (treated as transposed)
    stride_bm = right_flat.stride(0)
    stride_bk = right_flat.stride(2)   # K dimension stride (original last dim)
    stride_bn = right_flat.stride(1)   # N dimension stride (original middle dim)

    # strides for C
    stride_cm = out.stride(0)
    stride_cn = out.stride(1)
    stride_ck = out.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (M, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out,
        M,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with required weight tensors (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to fp16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Apply mask and fuse with projection & gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul) via Triton kernel
    # ------------------------------------------------------------------
    out = _tri_mul_torch(left, right)   # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
