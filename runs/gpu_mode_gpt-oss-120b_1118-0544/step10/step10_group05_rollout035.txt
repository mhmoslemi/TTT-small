"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for compute‑heavy parts.
3. Linear projections (left/right) and gating projections (left_gate,
   right_gate, out_gate) – bias‑free.
4. If a mask is present, fuse mask, projection and gating with a
   small Triton kernel; otherwise fuse with simple element‑wise ops.
5. Core reduction (TriMul):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (one matmul per batch×head) via a
   custom Triton kernel.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and apply the final linear projection back to `dim`.
7. Cast the result back to the input dtype.

The kernel operates in fp16, accumulating in fp32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse projection, gating and mask:
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask
    The mask has shape [B, N, N] and is broadcast over the hidden dim.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    # each hidden entry repeats the same mask element
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch Triton kernel that fuses projection, gating and mask.
    All tensors must be fp16, contiguous and on the same device.
    Returns:
        left, right – tensors of shape (B, N, N, H) in fp16.
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],   # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


@triton.jit
def _batched_matmul_kernel(
    left_ptr,   # fp16 *
    right_ptr,  # fp16 *
    out_ptr,    # fp16 *
    N: tl.constexpr,               # seq length
    stride_l_batch, stride_l_n, stride_l_k,
    stride_r_batch, stride_r_n, stride_r_k,
    stride_o_batch, stride_o_n, stride_o_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for TriMul.
    Computes for each batch*head:
        out = left @ right.T
    left/right/out shapes: (B*H, N, N)
    """
    pid_b = tl.program_id(0)   # batch*head index
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: (M, K)
        a_ptr = left_ptr + pid_b * stride_l_batch \
                + offs_m[:, None] * stride_l_n \
                + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right block transposed: (K, N)  -> we load right as (K, N)
        b_ptr = right_ptr + pid_b * stride_r_batch \
                + offs_k[:, None] * stride_r_n \
                + offs_n[None, :] * stride_r_k
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # write back
    c_ptr = out_ptr + pid_b * stride_o_batch \
            + offs_m[:, None] * stride_o_n \
            + offs_n[None, :] * stride_o_k
    tl.store(c_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform batched matrix multiplication per hidden slice using Triton.
    left/right: (B, N, N, H) fp16
    Returns: out of shape (B, N, N, H) fp16
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_flat = torch.empty_like(left_flat)

    # strides (in elements)
    stride_l_batch, stride_l_n, stride_l_k = left_flat.stride()
    stride_r_batch, stride_r_n, stride_r_k = right_flat.stride()
    stride_o_batch, stride_o_n, stride_o_k = out_flat.stride()

    # block sizes – tuned for typical B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_l_batch, stride_l_n, stride_l_k,
        stride_r_batch, stride_r_n, stride_r_k,
        stride_o_batch, stride_o_n, stride_o_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to fp16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Apply mask and fuse with projection & gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16, contiguous and on the same device
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask (or mask ignored): simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul) – batched GEMM via Triton
    # ------------------------------------------------------------------
    out = _batched_matmul(left, right)   # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
