@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # get program ids
    pid_bh = tl.program_id(0) # batch*hidden
    pid_m = tl.program_id(1) # i block
    pid_n = tl.program_id(2) # j block

    # compute batch index and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # Compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-range i,j
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offset
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M, K)
        # left offset = b*stride_left_b + h*stride_left_h + offs_m[:,None]*stride_left_i + offs_k[None,:]*stride_left_k
        left_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile (K, N)
        # right offset = b*stride_right_b + h*stride_right_h + offs_k[:, None]*stride_right_k + offs_n[None, :]*stride_right_j
        right_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Compute dot product
        acc += tl.dot(left, right)  # left (M,K), right (K,N) => (M,N)

    # Store result
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)

    # 2. Cast weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # 3. Linear projections
    left_proj = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # 4. Gating
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # 5. Apply mask and fuse
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 6. Core reduction using Triton kernel
    # Ensure contiguous for stride access
    left = left.contiguous()
    right = right.contiguous()
    # allocate output
    out = torch.empty_like(left)

    B, N, _, H = left.shape

    # Get strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid sizes
    grid = (B * H, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # 7. Output LayerNorm, gating, final linear
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)

    # Cast back to original dtype
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

"""
TriMul outgoing forward pass (AlphaFold3) – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑normalize the input over the channel dimension.
2. Cast to FP16 for compute‑heavy parts.
3. Linear projections (left/right) and gating (left_gate, right_gate, out_gate).
4. Fuse projection, gating and optional mask with a tiny Triton kernel.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM using a custom Triton kernel that
   performs the matrix multiplication per (batch, hidden) pair.
6. Output LayerNorm, element‑wise multiplication with out_gate,
   and final linear projection back to the original dimension.
7. Cast back to the input dtype (float32).

The kernel operates on contiguous FP16 tensors and uses 64×64×64 tiling.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and (optional) mask:
       left_out  = left_proj  * left_gate  * mask
       right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # mask is broadcasted over the hidden dimension → same mask for each hidden channel
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication per (batch, hidden) pair:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    """
    pid_bh = tl.program_id(0)          # batch * hidden
    pid_m  = tl.program_id(1)          # i‑block
    pid_n  = tl.program_id(2)          # j‑block

    # decode batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # coordinate offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (M, K)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_m[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        left = left.to(tl.float32)

        # right tile: (K, N) – note the order K then N for tl.dot
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_k[:, None] * stride_right_k
            + offs_n[None, :] * stride_right_j
        )
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        right = right.to(tl.float32)

        # matrix multiplication fragment
        acc += tl.dot(left, right)          # (M, K) @ (K, N) -> (M, N)

    # write back
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_m[:, None] * stride_out_i
        + offs_n[None, :] * stride_out_j
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict containing:
                * "dim" (int)          – input channel dimension C
                * "hidden_dim" (int)   – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Fuse projection, gating and optional mask via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        # mask shape [B, N, N] → broadcast over hidden dim
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul) using a Triton batched GEMM
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides for the three tensors
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_rj, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_lb,
        s_li,
        s_lk,
        s_lh,
        s_rb,
        s_rj,
        s_rk,
        s_rh,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """Launch the Triton kernel that fuses projection, gating and mask.
    All tensors must be fp16 and contiguous.
    Returns (left_out, right_out) each with shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out
