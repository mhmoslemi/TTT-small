def _tri_mul(left: torch.Tensor, right: torch.Tensor, block_m=64, block_n=64, block_k=64) -> torch.Tensor:
    # left, right shape (BH,N,N) contiguous FP16.
    BH, M, K = left.shape
    _, K2, N = right.shape
    assert K == K2

    out = torch.empty((BH, M, N), dtype=left.dtype, device=left.device)

    # compute strides
    stride_al = K  # left stride for M dimension (row)
    stride_ak = 1  # left stride for K dimension (col)
    stride_br = N  # right stride for K dimension (row)
    stride_bn = 1  # right stride for N dimension (col)
    stride_cl = N  # out stride for M dimension
    stride_cn = 1  # out stride for N dimension

    batch_stride_left = M * K
    batch_stride_right = K * N
    batch_stride_out = M * N

    grid = (BH, triton.cdiv(M, block_m), triton.cdiv(N, block_n))
    tri_mul_kernel[grid](
        left, right, out,
        M, N, K,
        stride_al, stride_ak,
        stride_br, stride_bn,
        stride_cl, stride_cn,
        batch_stride_left, batch_stride_right, batch_stride_out,
        BLOCK_M=block_m, BLOCK_N=block_n, BLOCK_K=block_k,
    )
    return out

"""
TriMul (outgoing) forward pass – Triton‑fused implementation.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections (left/right) and gating (left_gate, right_gate, out_gate)
   are computed in FP16.
4. If a mask is present, a Triton kernel fuses projection, gating and mask
   multiplication for both left and right tensors.
5. Core “tri‑mul”:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden slice `h`.
   A custom Triton kernel performs the GEMM on blocks (M,N,K) with a
   float32 accumulator for numerical stability.
6. Apply a second LayerNorm, multiply by `out_gate`, and a final linear
   projection back to the original `dim`.
7. Cast the result back to the input dtype.

The implementation keeps the heavy reductions on the GPU via Triton while
leveraging PyTorch for the surrounding linear layers and normalizations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask:
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    All tensors are flattened in row‑major order.
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # index of the corresponding (b,i,k) element (mask is broadcast across hidden)
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # 1.0 = no masking

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr  + offs, left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Calls the Triton kernel that fuses projection, gating and mask.
    All inputs must be contiguous FP16 tensors.
    Returns (left, right) each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM:
    for each batch (including hidden slice):
        C = A @ B   where A: (M,K), B: (K,N)
    """
    pid = tl.program_id(0)          # batch * hidden index
    pid_m = tl.program_id(1)        # tile row
    pid_n = tl.program_id(2)        # tile col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_ptr + pid * batch_stride_a + \
                  offs_m[:, None] * stride_am + \
                  offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid * batch_stride_b + \
                  offs_k[:, None] * stride_bk + \
                  offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = c_ptr + pid * batch_stride_c + \
             offs_m[:, None] * stride_cm + \
             offs_n[None, :] * stride_cn

    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _tri_mul(left: torch.Tensor, right_t: torch.Tensor,
             block_m: int = 64, block_n: int = 64, block_k: int = 64) -> torch.Tensor:
    """
    Batched matrix multiplication using Triton.
    left:  (B*H, N, N)    – FP16
    right_t: (B*H, N, N) – FP16 (already transposed, i.e. K->N)
    Returns out of shape (B*H, N, N) – FP16
    """
    batch, M, K = left.shape
    _, K2, N = right_t.shape
    assert K == K2

    out = torch.empty((batch, M, N), dtype=left.dtype, device=left.device)

    # Strides for contiguous tensors (row‑major)
    stride_am = K            # A row stride
    stride_ak = 1            # A col stride
    stride_bk = N            # B row stride (since B is K x N)
    stride_bn = 1            # B col stride
    stride_cm = N            # C row stride
    stride_cn = 1            # C col stride

    batch_stride_a = M * K
    batch_stride_b = K * N
    batch_stride_c = M * N

    grid = (batch,
            triton.cdiv(M, block_m),
            triton.cdiv(N, block_n))
    _tri_mul_kernel[grid](
        left, right_t, out,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=block_m, BLOCK_N=block_n, BLOCK_K=block_k,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights: dict with the following keys (all float32):
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Apply mask and fuse projection & gating (Triton if needed)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # Simple element‑wise fusion when no mask is used
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul) – batched GEMM per hidden slice
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) and then flatten batch*H
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left.shape
    left_flat = left.view(B * H, N, N)
    right_flat = right.view(B * H, N, N)

    # transpose right for GEMM (K dimension first)
    right_t = right_flat.permute(0, 2, 1).contiguous()  # (B*H, N, N)

    out_flat = _tri_mul(left_flat, right_t,
                        block_m=64, block_n=64, block_k=64)  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
