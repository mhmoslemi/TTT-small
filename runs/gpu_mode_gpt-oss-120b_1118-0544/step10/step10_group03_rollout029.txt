def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Cast to fp16
    orig_dtype = x.dtype
    x = x.to(torch.float16)

    # Prepare weight tensors in fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Combine projection and gate weights for left/right
    proj_gate_w = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w], dim=0)  # shape (4*H, dim)

    # Linear transform
    y = F.linear(x, proj_gate_w)   # (B,N,N,4*H)

    # Split
    left_proj, right_proj, left_gate_pre, right_gate_pre = torch.split(y, hidden_dim, dim=-1)

    # Apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # Apply mask if not nomask
    if mask is not None and not nomask:
        # mask shape (B,N,N) -> broadcast to (B,N,N,1)
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Compute out gate
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B,N,N,H)

    # Core reduction
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B*H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1,2))
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layer norm, out gate, final linear
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)

    # Cast back
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out

def apply_gate_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,

    mask = offs < total_elems

    # Load
    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)

    rg_sig = 1.0 / (1.0 + tl.exp(-rg))

    r_out = rp * rg_sig * m

    tl.store(left_out_ptr + offs, l_out, mask=mask)

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the AlphaFold3 reference module but reduces the
number of large GEMM calls by concatenating the four projection/gate weight
matrices into a single linear layer.  A small Triton kernel fuses the
application of the sigmoid gates and the optional mask.  The core cubic
reduction is performed with a batched GEMM (torch.bmm) on fp16 tensors for
maximum throughput.  The final output is returned in the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _apply_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64
    hidden_dim,      # i64
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse sigmoid gating and optional mask:
        left  = left_proj  * sigmoid(left_gate)  * mask
        right = right_proj * sigmoid(right_gate) * mask
    The mask tensor has shape (B, N, N) and is broadcast over the hidden
    dimension.  ``total_elems`` corresponds to B*N*N*hidden_dim.
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Index into mask (which lacks the hidden dimension)
    mask_idx = offs // hidden_dim

    left_proj = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # sigmoid(x) = 1 / (1 + exp(-x))
    left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))

    left_out = left_proj * left_gate_sig * m
    right_out = right_proj * right_gate_sig * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch the Triton kernel that applies sigmoid gates and a binary mask.
    All tensors must be fp16 and contiguous.  Returns tensors of shape
    (B, N, N, hidden_dim) for left and right.
    """
    total = left_proj.numel()
    hidden_dim = left_proj.shape[-1]

    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _apply_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        hidden_dim,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict containing:
                * "dim" (int)        – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive parts
    # ------------------------------------------------------------------
    orig_dtype = x.dtype
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Prepare FP16 weights
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)  # (H, C)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, C)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)  # (H, C)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, C)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (H,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)       # (C, H)

    # ------------------------------------------------------------------
    # 4) Fuse left/right projections and their gates into a single GEMM
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices: [left_proj; right_proj; left_gate; right_gate]
    proj_gate_w = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w], dim=0
    )  # (4*H, C)

    # Linear transform: result shape (B, N, N, 4*H)
    y = F.linear(x, proj_gate_w)

    # Split into the four components
    left_proj, right_proj, left_gate_pre, right_gate_pre = torch.split(
        y, hidden_dim, dim=-1
    )  # each (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply sigmoid gates and optional mask (via Triton)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_gate_mask(
            left_proj, right_proj, left_gate_pre, right_gate_pre, mask_fp16
        )
    else:
        # No mask (or mask ignored)
        left_gate = torch.sigmoid(left_gate_pre)
        right_gate = torch.sigmoid(right_gate_pre)
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Compute out gate (separate linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Core cubic reduction:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM on fp16.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched matrix multiplication
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (left * right^T)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = (
        out_flat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, C)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
