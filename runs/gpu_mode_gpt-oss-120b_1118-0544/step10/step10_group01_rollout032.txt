"""
TriMul (outgoing) forward pass – highly optimized Triton‑fused version.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for the compute‑heavy part.
3. Fuse the two projection Linear layers (proj + gate) for the left and right
   paths by concatenating their weights and performing a single matmul per
   side.  The gating part is passed through a sigmoid.
4. Optionally apply the mask and fuse the projection, gate and mask in a
   Triton kernel, producing `left` and `right` of shape (B,N,N,H).
5. Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM on tensors of shape (B*H, N, N).
6. Output LayerNorm, element‑wise multiplication with the output gate,
   and a final linear projection back to the original dimension.
7. Cast the result back to the input dtype.

Only the projection/gate/mask fusion is performed in Triton; the remaining
operations use highly‑optimized PyTorch primitives (layer‑norm, batched
matrix multiplication, linear layers).  This balance gives excellent
throughput while satisfying the requirement to use a custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and (optional) mask.
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # map linear index to (b,i,j) position for mask lookup
    mask_idx = offs // hidden_dim

    left   = tl.load(left_proj_ptr + offs,  mask=mask, other=0.0)
    right  = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate  = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rgate  = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m      = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # default mask=1

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch the Triton kernel that fuses projection, gating and mask.
    All tensors must be fp16 and contiguous.
    Returns (left_out, right_out) each with shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],   # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with the model parameters (float32)
            - config: dict with:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Prepare FP16 weights
    # ------------------------------------------------------------------
    # Concatenate projection and gate weights to reduce GEMM count
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)      # (H, C)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)      # (H, C)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)     # (H, C)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)     # (H, C)

    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2H, C)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2H, C)

    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, C)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (H,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)       # (C, H)

    # ------------------------------------------------------------------
    # 4) Linear projections (combined with gates)
    # ------------------------------------------------------------------
    # left_comb: (B,N,N,2H) -> split into proj and gate
    left_comb = F.linear(x, left_comb_w)   # (B,N,N,2H)
    left_proj, left_gate = torch.split(left_comb, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate)

    # right_comb: (B,N,N,2H)
    right_comb = F.linear(x, right_comb_w)
    right_proj, right_gate = torch.split(right_comb, hidden_dim, dim=-1)
    right_gate = torch.sigmoid(right_gate)

    # output gate (H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Apply mask and fuse projection & gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is contiguous FP16
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask (or ignored): simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # Batched matrix multiplication (FP16)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 8) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
