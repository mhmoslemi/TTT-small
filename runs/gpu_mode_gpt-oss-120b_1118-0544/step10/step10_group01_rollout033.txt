"""
TriMul (outgoing) forward pass implemented with a fused mask kernel in Triton.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (`dim`).
2. Cast to FP16 for compute‑heavy parts.
3. Linear projections for the left/right paths and their gates (sigmoid) using
   torch.nn.functional.linear (highly‑optimized cuBLAS kernels).
4. If a mask is present, apply it in‑place to the projected tensors with a
   small Triton kernel (no extra memory copies).
5. Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to a batched matrix‑multiplication
        out = left_flat @ right_flatᵀ
   where `left_flat`/`right_flat` are reshaped to (B*H, N, N).
6. Output layer‑norm over the hidden dimension, multiplication by the output
   gate and a final linear projection back to `dim`.
7. Cast the result back to the original dtype.

Only the mask‑application step is performed in Triton; all other heavy
operations rely on PyTorch's optimized kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _apply_mask_kernel(
    left_ptr,      # fp16 *
    right_ptr,     # fp16 *
    mask_ptr,      # fp16 *
    TOTAL_ELEMS: tl.constexpr,   # total number of elements in left/right
    HIDDEN_DIM: tl.constexpr,    # hidden dimension size
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place multiplication of left/right tensors by a broadcast mask.

    Each element of the mask corresponds to one (b,i,j) position and is
    broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds
    mask = offs < TOTAL_ELEMS

    # Index of the mask element for each hidden entry
    mask_idx = offs // HIDDEN_DIM

    left  = tl.load(left_ptr + offs,  mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    m     = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left  = left  * m
    right = right * m

    tl.store(left_ptr + offs,  left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_mask(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor,
                block_size: int = 1024) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch the Triton kernel that multiplies `left` and `right` by `mask`
    in‑place. All tensors must be fp16 and contiguous.
    """
    total = left.numel()
    grid = (triton.cdiv(total, block_size),)

    _apply_mask_kernel[grid](
        left,
        right,
        mask.view(-1),          # flatten mask to 1‑D
        total,
        left.shape[-1],         # hidden dimension
        BLOCK_SIZE=block_size,
    )
    return left, right


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (no bias)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Fuse projection with gate (element‑wise)
    # ------------------------------------------------------------------
    left  = left_proj * left_gate   # (B, N, N, H)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Optional mask application (Triton kernel)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is contiguous FP16
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_mask(left, right, mask_fp16)

    # ------------------------------------------------------------------
    # 8) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B*H, N, N) for batched GEMM
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (FP16)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 10) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
