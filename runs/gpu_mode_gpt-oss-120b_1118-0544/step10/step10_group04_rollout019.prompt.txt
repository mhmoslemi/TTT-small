import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) forward pass – Triton‑accelerated projection‑gate‑mask fusion.

Algorithm
---------
1. Layer‑norm the 4‑D input over the channel dimension `dim`.
2. Cast to FP16 for the compute‑heavy stages.
3. Linear projections `left_proj` / `right_proj` and gating projections
   (`left_gate`, `right_gate`, `out_gate`) – all bias‑free.
4. Fuse mask, projection and gating with a custom Triton kernel
   (only executed when a mask is present and not ignored).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM over the hidden dimension using
   `torch.bmm` (leveraging cuBLAS, which is optimal for dense matmul).
6. Apply a second LayerNorm, multiply by `out_gate`, and a final linear
   projection back to the original channel dimension `dim`.
7. Cast the result back to the input dtype (usually float32).

The kernel in step 4 reduces memory traffic by performing
projection × gate × mask in a single pass.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask:
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds accesses
    mask = offs < total_elems

    # One mask value per (b,i,j) entry, repeated over the hidden dim
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launches the Triton kernel that fuses projection, gating and mask.
    All tensor arguments must be fp16 and contiguous.
    Returns (left, right) each of shape (B, N, N, H).
    """
    assert left_proj.is_contiguous() and right_proj.is_contiguous()
    assert left_gate.is_contiguous() and right_gate.is_contiguous()
    assert mask.is_contiguous()

    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],          # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to FP16 for the compute‑intensive parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast linear weights to FP16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections (FP16)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)   # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 6) Apply mask and fuse with projection & gating
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask (or mask ignored): simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7) Core reduction (TriMul)
    #   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9) Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

Approach and rationale
We are implementing the outgoing TriangleMultiplicativeUpdate (TriMul) operator from AlphaFold3. This involves computing a tensor `out` of shape [B, N, N, C] where each element is computed by a summation over a third index `k` (which is not the last dimension) and conditional multiplication by a mask.

The operations in the TriMul are performed in three main steps (as in the PyTorch code):
1. Apply LayerNorm along the channel dimension.
2. Perform linear projections (left_proj, right_proj) and gating operations (left_gate, right_gate, out_gate).
3. Compute the TriMul using the formula: `out[b,i,j,d] = sum_{k} (left_proj[b,i,k,d] * right_proj[b,j,k,d])`
   Additionally, we conditionally multiply by the mask.

The previous implementation used `torch.bmm` to compute the reduction. However, for large N (up to 1024), this approach is not optimal and can be accelerated with a tiled kernel reducing memory accesses. The outgoing mask (if provided) should also be applied appropriately.

Based on the paper and the problem description, the outgoing TriMul should be computed such that the mask zeros out invalid pairs. Specifically, we'll consider the presence of an outgoing mask and apply element-wise multiplication where necessary.

However, note that the PyTorch code includes:
    left = left * left_gate * mask
    right = right * right_gate * mask
    out = left @ right

The mask is applied to all three dimensions (B, N, N). We will do the same with conditional mask application.

Our approach will be to:
- Compute LayerNorm on the input
- Apply linear projections (left_proj and right_proj) using provided weights
- Apply gating operations (left_gate, right_gate) with sigmoid activation
- If a mask is provided, multiply the projected tensors by the mask element-wise. Note: the mask is [B, N, N].
- Compute the TriMul reduction via a batched matrix multiplication for the values that should be computed. We'll use a tiled kernel to minimize the number of memory accesses.

But note: the reduction must be careful to only consider pairs (i, j) for which the mask is active. The mask is applied after the projections and gating, so if the mask is 0 at any position for a given (b, i, j), then the entire triplet [b, i, j] should be 0.

    2. `left = x_ln` @ `weights_left` (linear projection to hidden_dim)
    3. `right = x_ln` @ `weights_right`
    4. `left_gate = torch.sigmoid(x_ln` @ `weights_left_gate)`
    5. `right_gate = torch.sigmoid(x_ln` @ `weights_right_gate}`
    6. If mask is provided: `left_masked = left * mask`; `right_masked = right * mask`.
       Otherwise, no masking is applied for this step.

    7. TriMul reduction: `out_temp[b,i,j,d] = sum_{k} (left[b,i,k,d] * right[b,j,k,d])`
       If mask is provided: `out_temp[b,i,j,d] = out_temp[b,i,j,d] * mask[b,i,j]`
       Also, apply `out_gate` after reduction: `out_final[b,i,j,d] = out_temp[b,i,j,d] * out_gate[b,i,j,d]`

    8. `out = out_final` @ `weights_out` (finallinear projection from hidden_dim to C)
