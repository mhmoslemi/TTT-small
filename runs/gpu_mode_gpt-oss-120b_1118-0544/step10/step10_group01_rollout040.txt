def batched_matmul(a, b):
    # a, b shape (batch, M, K) and (batch, K, N)
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2
    out = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch, tl.cdiv(M, BLOCK_M), tl.cdiv(N, BLOCK_N))
    batched_matmul_kernel[grid](
        a, b, out,
        M, N, K,
        a.stride(1), a.stride(2),
        b.stride(1), b.stride(2),
        out.stride(1), out.stride(2),
        a.stride(0), b.stride(0), out.stride(0),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (`dim`).
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections for the left/right paths and their gates are done with
   `torch.nn.functional.linear` (fp16).
4. A small Triton kernel fuses the two projections, the two gates and the
   optional mask into two tensors `left` and `right` of shape (B, N, N, H).
5. Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication per hidden slice.
   We flatten the batch and hidden dimensions (B·H, N, N) and run a custom
   Triton kernel that performs the matmul (`left @ rightᵀ`) for all batches
   in one launch, avoiding cuBLAS overhead.
6. Output layer‑norm, output‑gate multiplication and final linear projection.
7. Cast back to the original dtype.

Only the projection‑gate‑mask fusion and the batched matmul are implemented
in Triton; the remaining ops use PyTorch for simplicity and numerical
stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gating and (optional) mask
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """
    Compute:
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask
    where mask has shape (B, N, N) and is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    in_range = offs < total_elems

    # each element belongs to a (b,i,k) location; the mask index is the same for all H entries
    mask_idx = offs // hidden_dim

    left   = tl.load(left_proj_ptr + offs,  mask=in_range, other=0.0)
    right  = tl.load(right_proj_ptr + offs, mask=in_range, other=0.0)
    lgate  = tl.load(left_gate_ptr + offs,  mask=in_range, other=0.0)
    rgate  = tl.load(right_gate_ptr + offs, mask=in_range, other=0.0)
    m      = tl.load(mask_ptr + mask_idx, mask=in_range, other=1.0)

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=in_range)
    tl.store(right_out_ptr + offs, right_out, mask=in_range)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch the Triton kernel that fuses projection, gating and mask.
    All tensors must be fp16 and contiguous.
    Returns (left_out, right_out) each with shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Triton kernel: batched matmul (A @ Bᵀ) for many small matrices
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr,               # fp16 *
    b_ptr,               # fp16 *
    c_ptr,               # fp16 *
    M, N, K,             # i64 scalars
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes, for each batch index `b`:
        C[b] = A[b] @ B[b].T
    where A has shape (M, K) and B has shape (N, K) (i.e. we treat B as transposed).
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for this batch
    a_ptrs = a_ptr + batch_id * batch_stride_a + \
             offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * batch_stride_b + \
             tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k * BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)               # (BLOCK_M, BLOCK_N) accumulation
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    c_ptrs = c_ptr + batch_id * batch_stride_c + \
             offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ Bᵀ using the Triton kernel above.
    Shapes:
        a: (BATCH, M, K)  fp16
        b: (BATCH, N, K)  fp16   (note: we treat the second dim as K, third as N)
    Returns:
        c: (BATCH, M, N)  fp16
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, N, K2 = b.shape
    assert K == K2

    c = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _batched_matmul_kernel[grid](
        a, b, c,
        M, N, K,
        a.stride(1), a.stride(2),
        b.stride(1), b.stride(2),
        c.stride(1), c.stride(2),
        a.stride(0), b.stride(0), c.stride(0),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return c


# ----------------------------------------------------------------------
# Exported entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: (B, N, N, C)   torch.Tensor
            - mask: (B, N, N) or None
            - weights: dict with model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        Tensor of shape (B, N, N, C) with the same dtype as `input_tensor`.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Prepare fp16 weights (no bias)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Fuse mask, projection and gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is contiguous fp16 and flattened
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask (or ignored): simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul) via batched matmul
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # transpose the last two dims of right before flattening
    right_T = right.permute(0, 3, 2, 1).contiguous()
    right_flat = right_T.view(B * H, N, N)

    out_flat = _batched_matmul(left_flat, right_flat)  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
