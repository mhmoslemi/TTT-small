def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # 1) Input LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # (B,N,N,dim)

    # Cast to fp16
    x = x.to(torch.float16)

    # Convert weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections; bias=False
    left_proj = torch.nn.functional.linear(x, left_proj_w)  # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # Apply mask and fuse projections using Triton kernel
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(left_proj, right_proj, left_gate, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Core reduction using Triton
    out = _trmul_core(left, right)  # shape (B,N,N,H)

    # Output layernorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to original dtype
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

def _trmul_core(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    # left, right: shape (B,N,N,H) fp16 contiguous
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # same shape and dtype

    # Get strides
    stride_left = left.stride()  # (N*N*H, N*H, H, 1)
    stride_right = right.stride()
    stride_out = out.stride()

    # Choose block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * H )

    _trmul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

@triton.jit
def _trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_x = tl.program_id(0)  # row tile i
    pid_y = tl.program_id(1)  # col tile j
    pid_z = tl.program_id(2)  # batch*head

    # Determine batch and head
    b = pid_z // H
    h = pid_z % H

    # Offsets
    i_offset = pid_x * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_y * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K)

    # Mask for bounds
    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_k = k_offset < N

    # Compute base offsets for this (b,h)
    # base = b*stride_left_b + h*stride_left_h
    base_left = b * stride_left_b + h * stride_left_h
    base_right = b * stride_right_b + h * stride_right_h
    base_out = b * stride_out_b + h * stride_out_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for kk in range(0, N, BLOCK_K):
        cur_k = kk + k_offset
        cur_mask_k = cur_k < N

        # Load A tile (left): shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + base_left \
                 + i_offset[:, None] * stride_left_i \
                 + cur_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & cur_mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile (right) transposed: shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + base_right \
                 + j_offset[None, :] * stride_right_i \
                 + cur_k[:, None] * stride_right_k
        b = tl.load(b_ptrs, mask=cur_mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Accumulate

"""
TriMul outgoing forward pass (AlphaFold3) – Triton fused implementation.

Algorithm
---------
1. Layer‑norm on the input channel dimension.
2. Linear projections (left/right) and gating (left_gate, right_gate, out_gate) in fp16.
3. Fuse projection, gating and optional mask with a tiny Triton kernel
   (`_proj_gate_mask_kernel`), reducing memory traffic.
4. Core TriMul reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton batched‑GEMM where the second operand is
   implicitly transposed, thus avoiding the expensive torch.bmm.
5. Output layer‑norm, element‑wise out‑gate, final linear projection.
6. Cast back to the original dtype.

The implementation supports both masked and un‑masked inputs and works for
batch sizes 1‑2, sequence lengths up to 1024 and hidden dimensions up to 384.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# 1. Fuse projection, gating and mask
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 = B*N*N*H
    hidden_dim,      # i64 = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    in_bounds = offs < total_elems

    # mask is stored as [B, N, N] → flatten; each hidden element shares the same mask
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=in_bounds, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=in_bounds, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=in_bounds, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=in_bounds, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=in_bounds, other=1.0)

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=in_bounds)
    tl.store(right_out_ptr + offs, right_out, mask=in_bounds)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """Launch the Triton kernel that fuses projection, gating and mask."""
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# 2. Core TriMul reduction (batched GEMM with transposed RHS)
# ----------------------------------------------------------------------
@triton.jit
def _trmul_kernel(
    left_ptr,   # fp16 *
    right_ptr,  # fp16 *
    out_ptr,    # fp16 *
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_x = tl.program_id(0)          # tile row (i)
    pid_y = tl.program_id(1)          # tile col (j)
    pid_z = tl.program_id(2)          # batch * head

    # decode batch and head index
    b = pid_z // H
    h = pid_z % H

    # offsets within the tile
    i_off = pid_x * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_y * BLOCK_N + tl.arange(0, BLOCK_N)
    k_off = tl.arange(0, BLOCK_K)

    mask_i = i_off < N
    mask_j = j_off < N

    # base pointer for this (b, h)
    base_left  = b * stride_left_b  + h * stride_left_h
    base_right = b * stride_right_b + h * stride_right_h
    base_out   = b * stride_out_b   + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_iter in range(num_k):
        k_start = k_iter * BLOCK_K
        cur_k = k_start + k_off
        mask_k = cur_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + base_left \
                 + i_off[:, None] * stride_left_i \
                 + cur_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right tile (transposed) : shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + base_right \
                 + j_off[None, :] * stride_right_i \
                 + cur_k[:, None] * stride_right_k
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # write result
    out_ptrs = out_ptr + base_out \
               + i_off[:, None] * stride_out_i \
               + j_off[None, :] * stride_out_k
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _trmul_core(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Batched TriMul reduction using the Triton kernel above."""
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides for contiguous tensors (B,N,N,H)
    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    _trmul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# 3. Full custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict containing:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast model weights to fp16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections (fp16)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 6) Apply mask and fuse projection & gating
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7) Core reduction (TriMul) – Triton implementation
    # --------------------------------------------------------------
    out = _trmul_core(left, right)   # (B, N, N, H)

    # --------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)    # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9) Cast back to original dtype if needed
    # --------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
