TriMul outgoing (AlphaFold3) forward pass optimized with fused
linear+gate projections and a small Triton kernel for mask‑gate
multiplication. The heavy reduction is performed by a batched
matrix multiplication (torch.bmm) on FP16 tensors.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _mask_gate_fuse_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):

    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    # index of the corresponding mask element (each mask element repeats hidden_dim times)
    mask_idx = offs // hidden_dim

    left = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # default 1.0 = no masking

    left_out = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

def _apply_mask_gate_fuse(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse mask, projection and gating with a single Triton kernel.
    All tensors must be contiguous fp16.
    Returns left, right of shape (B, N, N, H).

    hidden_dim = left_proj.shape[-1]
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _mask_gate_fuse_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        hidden_dim,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: (B, N, N, C) torch.Tensor (float32 or float16)
            - mask: (B, N, N) torch.Tensor or None
            - weights: dict with required parameters (all float32)
            - config: dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape (B, N, N, C) with same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1) Input LayerNorm over channel dimension
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # 2) Cast to FP16 for compute‑intensive part
    x = x.to(torch.float16)

    # 3) Prepare FP16 weights (no bias)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # 4) Fuse projection and gate for left and right via concatenated linear
    left_cat_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, C)
    right_cat_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    left_cat = F.linear(x, left_cat_w)   # (B, N, N, 2*H)
    right_cat = F.linear(x, right_cat_w)

    # split
    left_proj = left_cat[..., :hidden_dim]                     # (B,N,N,H)
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])      # (B,N,N,H)
    right_proj = right_cat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])

    # 5) Apply mask and fuse with projection & gating
    if mask is not None and not nomask:
        # mask shape (B,N,N) -> (B,N,N,1)
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        left, right = _apply_mask_gate_fuse(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # compute left @ right^T using batched bmm
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 7) Output LayerNorm, gating and final projection
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # out_gate
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))
    out = out * out_gate

    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # 8) Cast back to original dtype
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

"""
TriMul outgoing (AlphaFold3) forward pass optimized with fused
linear+gate projections and a small Triton kernel for mask‑gate
multiplication. The heavy reduction is performed by a batched
matrix multiplication (torch.bmm) on FP16 tensors.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _mask_gate_fuse_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse mask, projection and gating:
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask
    All tensors are flattened in row‑major order.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # each mask element repeats `hidden_dim` times across the hidden dimension
    mask_idx = offs // hidden_dim

    left = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # 1.0 = no masking

    left_out = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs, left_out, mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_mask_gate_fuse(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch the Triton kernel that fuses projection, gating and mask.
    All inputs must be contiguous FP16 tensors.
    Returns (left, right) each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    hidden_dim = left_proj.shape[-1]
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _mask_gate_fuse_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        hidden_dim,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: (B, N, N, C) torch.Tensor (float32 or float16)
            - mask: (B, N, N) torch.Tensor or None
            - weights: dict containing the model parameters (all float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape (B, N, N, C) with the same dtype as `input_tensor`.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Fuse projection and gate for left/right via concatenated linear
    # ------------------------------------------------------------------
    left_cat_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, C)
    right_cat_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    left_cat = F.linear(x, left_cat_w)   # (B, N, N, 2*H)
    right_cat = F.linear(x, right_cat_w)

    left_proj = left_cat[..., :hidden_dim]                     # (B, N, N, H)
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])      # (B, N, N, H)
    right_proj = right_cat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])

    # ------------------------------------------------------------------
    # 5) Apply mask and fuse with projection & gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # (B, N, N) -> (B, N, N, 1) in FP16
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        left, right = _apply_mask_gate_fuse(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
