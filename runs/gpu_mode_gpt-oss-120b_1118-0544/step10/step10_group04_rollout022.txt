"""
TriMul (outgoing) forward pass – Triton‑accelerated projection‑gate‑mask fusion.

Algorithm
---------
1. Layer‑norm the 4‑D input over the channel dimension `dim`.
2. Cast to FP16 for the compute‑heavy stages.
3. Linear projections (`left_proj`, `right_proj`) and gating
   (`left_gate`, `right_gate`, `out_gate`) – bias‑free.
4. Fuse mask, projection and gating with a custom Triton kernel
   (executed only when a mask is present and not ignored).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (`torch.bmm`) over the hidden dimension.
6. Apply a second LayerNorm, multiply by `out_gate`, and a final linear
   projection back to the original channel dimension `dim`.
7. Cast the result back to the input dtype (usually float32).

Only the projection‑gate‑mask fusion is done in Triton; the remaining
operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse projection, gating and mask:
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask
    One mask value is shared across the hidden dimension.
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # guard against OOB
    mask_valid = offs < total_elems

    # each mask entry corresponds to (b, i, j) -> one per hidden_dim values
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=mask_valid, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask_valid, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=mask_valid, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask_valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=mask_valid)
    tl.store(right_out_ptr + offs, right_out, mask=mask_valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launches the Triton kernel that fuses projection, gating and mask.
    All tensors must be fp16 and contiguous.
    Returns (left, right) with shape (B, N, N, H).
    """
    assert left_proj.is_contiguous() and right_proj.is_contiguous()
    assert left_gate.is_contiguous() and right_gate.is_contiguous()
    assert mask.is_contiguous()

    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],          # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast linear weights to FP16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections (FP16)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)   # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 6) Apply mask and fuse with projection & gating
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is fp16 and contiguous; mask is broadcast over hidden_dim
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask (or ignored): simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7) Core reduction (TriMul)
    #   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9) Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
