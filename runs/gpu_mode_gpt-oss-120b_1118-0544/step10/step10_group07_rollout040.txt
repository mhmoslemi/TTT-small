def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program id mapping
    batch_id = tl.program_id(0)  # each batch gets its own block per (block_m, block_n)
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block col

    # compute block start indices
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offsets
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # load A and B blocks
        a_ptrs = a_ptr + batch_id * stride_a_batch + off_m[:, None] * stride_a_m + k_offset[None, :] * stride_a_k
        b_ptrs = b_ptr + batch_id * stride_b_batch + k_offset[:, None] * stride_b_k + off_n[None, :] * stride_b_n

        a = tl.load(a_ptrs, mask=(off_m < M)[:, None] & (k_offset < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(k_offset < K)[:, None] & (off_n < N)[None, :], other=0.0)

        acc += tl.dot(a, b)

    # write back C
    c_ptrs = c_ptr + batch_id * stride_c_batch + off_m[:, None] * stride_c_m + off_n[None, :] * stride_c_n
    mask_c = (off_m[:, None] < M) & (off_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)
```

"""
TriMul (outgoing) forward pass implemented with Triton fusion.

Algorithm
---------
1. Layer‑norm over the input channel dimension.
2. Cast to fp16 for compute‑heavy parts.
3. Linear projections (left/right) and gating projections (left_gate/right_gate/out_gate).
4. Fuse projection, gating and optional mask in a Triton kernel.
5. Core reduction: for each batch and hidden slice compute a
   batched matrix multiplication
        out[b*h, i, j] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   implemented as a custom Triton batched GEMM.
6. Reshape back, apply output LayerNorm, multiply by out_gate and final linear projection.
7. Cast back to the original dtype (float32).

All heavy‑weight parts (proj‑gate‑mask fusion and batched GEMM) are
implemented in Triton. The rest leverages PyTorch for simplicity and
numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ------------------------------------------------------------
# Triton kernel: fuse linear projection, gating and optional mask
# ------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # guard
    valid = offs < total_elems

    # each element corresponds to a (b,i,k) index repeated over hidden_dim
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=valid, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=valid, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)  # default 1.0 = no mask

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gating and mask via Triton.
    All tensors must be fp16 and contiguous.
    Returns (left, right) each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ B^T) for many batches
# ------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    BATCH, M, N, K,                     # sizes
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for A and B
        a_ptrs = a_ptr + batch_id * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + batch_id * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # Write back
    c_ptrs = c_ptr + batch_id * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


def _batched_matmul(a: torch.Tensor, b: torch.Tensor,
                    block_m: int = 64, block_n: int = 64, block_k: int = 32) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B (where B is already transposed as KxN).
    A: (batch, M, K), B: (batch, K, N)    -> C: (batch, M, N)
    All tensors must be fp16 and contiguous.
    """
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "Inner dimensions must match"

    c = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

    grid = (batch,
            triton.cdiv(M, block_m),
            triton.cdiv(N, block_n))
    _batched_matmul_kernel[grid](
        a,
        b,
        c,
        batch,
        M,
        N,
        K,
        a.stride(0), a.stride(1), a.stride(2),
        b.stride(0), b.stride(1), b.stride(2),
        c.stride(0), c.stride(1), c.stride(2),
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return c


# ------------------------------------------------------------
# Entry point
# ------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) – can be all‑ones or None.
            - weights: dict containing model parameters (float32)
            - config: dict with keys:
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to fp16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid, fp16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Fuse mask, projection and gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B, N, N) -> fp16 contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask or ignored: simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul) – batched GEMM over the hidden dim
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # transpose right for multiplication
    right_t = right_flat.transpose(1, 2).contiguous()

    out_flat = _batched_matmul(left_flat, right_t)  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(dtype)

    return out
