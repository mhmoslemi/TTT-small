@triton.jit
def proj_gate_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems, hidden_dim,
    MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # compute index for mask: each element corresponds to (b,i,j,k) flatten, but we need index of mask which is (b,i,j). Since hidden_dim is innermost, mask_idx = offs // hidden_dim
    mask_idx = offs // hidden_dim
    # load
    left = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    if MASK:
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    else:
        m = 1.0
    left_out = left * lgate * m
    right_out = right * rgate * m
    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the reference PyTorch module but fuses several
element‑wise operations (masking, gating and the final out‑gate multiplication)
into small Triton kernels.  The heavy N³ reduction is performed with a
batched matrix multiplication (torch.bmm) on fp16 tensors, which leverages
cuBLAS for high throughput.  All linear layers and LayerNorms are executed
in half precision and the result is cast back to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_mask_kernel(
    left_proj_ptr,  # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,  # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,  # fp16 *
    left_out_ptr,  # fp16 *
    right_out_ptr,  # fp16 *
    total_elems,   # i64
    hidden_dim,    # i64
    MASK: tl.constexpr,      # compile‑time flag: 1 → use mask, 0 → ignore
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and optional mask:
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # index of the corresponding mask element (mask has no hidden dim)
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if MASK:
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    else:
        m = 1.0

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


@triton.jit
def elemwise_mul_kernel(
    a_ptr,      # fp16 *
    b_ptr,      # fp16 *
    out_ptr,    # fp16 *
    total_elems,
    BLOCK_SIZE: tl.constexpr,
):
    """Element‑wise multiplication: out = a * b"""
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Helper functions to launch kernels
# ----------------------------------------------------------------------
def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor | None,
    nomask: bool,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """Fuse projection, gating and optional mask using Triton."""
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    # If mask is disabled, pass a dummy pointer (won't be read)
    mask_ptr = mask if (mask is not None and not nomask) else left_proj
    mask_flag = 0 if (mask is None or nomask) else 1

    grid = (triton.cdiv(total, block_size),)
    proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask_ptr,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        MASK=mask_flag,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def _apply_out_gate(
    tensor: torch.Tensor,
    out_gate: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """Multiply tensor by out_gate element‑wise using Triton."""
    total = tensor.numel()
    out = torch.empty_like(tensor)

    grid = (triton.cdiv(total, block_size),)
    elemwise_mul_kernel[grid](
        tensor,
        out_gate,
        out,
        total,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float16)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # (B, N, N, dim), still float32
    orig_dtype = x.dtype
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2) Linear projections and gates (all fp16)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 3) Apply mask and fuse with projection & gating
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
    else:
        mask_fp16 = None

    left, right = _apply_proj_gate_mask(
        left_proj, right_proj, left_gate, right_gate, mask_fp16, nomask
    )

    # --------------------------------------------------------------
    # 4) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM.
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiplication (left * right^T)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5) Output LayerNorm (float16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 6) Multiply by out_gate (fused Triton kernel)
    # --------------------------------------------------------------
    out = _apply_out_gate(out, out_gate)

    # --------------------------------------------------------------
    # 7) Final linear projection back to input dimension
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 8) Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
