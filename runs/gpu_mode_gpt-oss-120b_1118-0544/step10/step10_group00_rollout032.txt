@triton.jit
def tri_mul_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    batch_heads,
    N,
    H,
    stride_lbh, stride_lbn, stride_lbk,  # left stride: batch*H?, N, N
    stride_rbh, stride_rbn, stride_rbk,
    stride_obh, stride_obn, stride_obk,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
   pid_b = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   # compute offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   # shape (BLOCK_M, BLOCK_N)
   # initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   for k in range(0, tl.cdiv(N, BLOCK_K)):
       offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
       
       # compute pointers for left (A) = left[pid_b, offs_m, offs_k]
       a_ptrs = left_ptr + pid_b * stride_lbh + offs_m[:, None] * stride_lbn + offs_k[None, :] * stride_lbk
       # right (B) = right[pid_b, offs_n, offs_k] (since we need transpose, we treat right as B^T with dimensions (N,N): B_T[j,k] = right[pid_b, j, k]; we need B_T for multiplication: A[i,k] * B_T[j,k]
       b_ptrs = right_ptr + pid_b * stride_rbh + offs_n[:, None] * stride_rbn + offs_k[None, :] * stride_rbk
       a = tl.load(a_ptrs, mask=offs_m < N & offs_k < N, other=0.0).to(tl.float32)
       b = tl.load(b_ptrs, mask=offs_n < N & offs_k < N, other=0.0).to(tl.float32)
       # note: we need B^T, but we loaded right as (j,k). To compute sum_k A[i,k] * B[j,k] which is same as A[i,:] dot B[j,:]; that's correct. So no need to transpose.
       acc += tl.dot(a, b)  # dimension (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N) => (BLOCK_M, BLOCK_N)
   # store result
   c_ptrs = out_ptr + pid_b * stride_obh + offs_m[:, None] * stride_obn + offs_n[None, :] * stride_obk
   # cast to output dtype (fp16)
   tl.store(c_ptrs, acc.to(tl.float16), mask=offs_m < N & offs_n < N)

def tri_mul_torch_triton(left, right):

    B,N,_,H = left.shape
    left_flat = left.permute(0,3,1,2).contiguous()
    right_flat = right.permute(0,3,1,2).contiguous()
    batch_heads = B * H
    left_flat = left_flat.view(batch_heads, N, N)
    right_flat = right_flat.view(batch_heads, N, N)

    out_flat = torch.empty_like(left_flat)

    # strides
    stride_lbh = left_flat.stride(0)
    stride_lbn = left_flat.stride(1)
    stride_lbk = left_flat.stride(2)
    stride_rbh = right_flat.stride(0)
    stride_rbn = right_flat.stride(1)
    stride_rbk = right_flat.stride(2)
    stride_obh = out_flat.stride(0)
    stride_obn = out_flat.stride(1)
    stride_obk = out_flat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch_heads, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    # call kernel
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_heads,
        N,
        stride_lbh, stride_lbn, stride_lbk,
        stride_rbh, stride_rbn, stride_rbk,
        stride_obh, stride_obn, stride_obk,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # After kernel, reshape back
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()
    return out

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # cast to fp16
    x = x.to(torch.float16)

    # linear projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16))

    # gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)))

    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # core reduction via Triton
    out = tri_mul_torch_triton(left, right)  # shape [B,N,N,H]

    # to_out_norm (layer norm) on hidden dim
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"].to(torch.float16), bias=weights["to_out_norm.bias"].to(torch.float16))

    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    # cast back to original dtype
    out = out.to(input_tensor.dtype)

    return out

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_heads, N,
    stride_lbh, stride_lbn, stride_lbk,
    stride_rbh, stride_rbn, stride_rbk,
    stride_obh, stride_obn, stride_obk,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for within bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointer offsets
        # left: shape (batch_heads, N, N) -> offsets: batch*stride_lbh + offs_m * stride_lbn + offs_k * stride_lbk
        a_ptrs = left_ptr + pid_b * stride_lbh + offs_m[:, None] * stride_lbn + offs_k[None, :] * stride_lbk
        # right: shape (batch_heads, N, N) -> we need B[j,k] (j = offs_n, k = offs_k)
        b_ptrs = right_ptr + pid_b * stride_rbh + offs_n[:, None] * stride_rbn + offs_k[None, :] * stride_rbk

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = out_ptr + pid_b * stride_obh + offs_m[:, None] * stride_obn + offs_n[None, :] * stride_obk
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul outgoing (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm over the channel dimension (dim) of the input.
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections (left/right) and gating projections (left_gate,
   right_gate, out_gate) – all without bias.
4. Apply mask (if present) and fuse it with projection & gating.
5. Core reduction (tri‑mul):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix‑multiply (A @ Bᵀ) using a custom
   Triton kernel that tiles the sequence dimension.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and apply a final linear projection back to `dim`.
7. Cast the result back to the original dtype.

Only the core reduction is performed in Triton; all other ops use
PyTorch (leveraging its highly‑optimized kernels).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    out_ptr,           # fp16 *
    batch_heads,       # i64 scalar = B * H
    N,                 # i64 scalar = seq_len
    stride_lbh, stride_lbn, stride_lbk,
    stride_rbh, stride_rbn, stride_rbk,
    stride_obh, stride_obn, stride_obk,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication for the TriMul core.

    Computes for each head h and batch b:
        out[b, i, j] = Σ_k left[b, i, k] * right[b, j, k]
    """
    pid_b = tl.program_id(0)   # batch * hidden head
    pid_m = tl.program_id(1)   # block in M (i dimension)
    pid_n = tl.program_id(2)   # block in N (j dimension)

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers to the sub‑tiles
        a_ptrs = left_ptr + pid_b * stride_lbh \
                 + offs_m[:, None] * stride_lbn \
                 + offs_k[None, :] * stride_lbk
        b_ptrs = right_ptr + pid_b * stride_rbh \
                 + offs_n[:, None] * stride_rbn \
                 + offs_k[None, :] * stride_rbk

        # Load with bounds checking
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_n[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate the dot product
        acc += tl.dot(a, b)

    # Write back the result (cast to FP16)
    c_ptrs = out_ptr + pid_b * stride_obh \
             + offs_m[:, None] * stride_obn \
             + offs_n[None, :] * stride_obk
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _tri_mul_torch_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the TriMul core reduction using the Triton kernel.
    left / right: tensors of shape (B, N, N, H) in fp16.
    Returns: tensor of shape (B, N, N, H) in fp16.
    """
    B, N, _, H = left.shape
    # Move hidden dimension next to batch for easier flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_heads = B * H
    left_flat = left_perm.view(batch_heads, N, N)
    right_flat = right_perm.view(batch_heads, N, N)

    out_flat = torch.empty_like(left_flat)

    # Strides (in elements) for contiguous (batch, M, N) layout
    stride_lbh = left_flat.stride(0)
    stride_lbn = left_flat.stride(1)
    stride_lbk = left_flat.stride(2)
    stride_rbh = right_flat.stride(0)
    stride_rbn = right_flat.stride(1)
    stride_rbk = right_flat.stride(2)
    stride_obh = out_flat.stride(0)
    stride_obn = out_flat.stride(1)
    stride_obk = out_flat.stride(2)

    # Tile sizes – chosen to fit typical GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_heads,
        N,
        stride_lbh, stride_lbn, stride_lbk,
        stride_rbh, stride_rbn, stride_rbk,
        stride_obh, stride_obn, stride_obk,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights: dict with the following keys (all float32):
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    # ------------------------------------------------------------------
    # 5) Apply mask and fuse with projection & gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction (TriMul) via Triton
    # ------------------------------------------------------------------
    out = _tri_mul_torch_triton(left, right)   # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
