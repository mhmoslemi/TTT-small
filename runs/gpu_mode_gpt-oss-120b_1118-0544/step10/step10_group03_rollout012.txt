"""
Outgoing Triangle Multiplicative Update (TriMul) – forward pass.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Cast to fp16 for compute‑intensive ops.
3. Linear projections:   left_proj = x·W_left,  right_proj = x·W_right
   Linear gates:          left_gate = σ(x·W_left_gate),  right_gate = σ(x·W_right_gate)
   Output gate:           out_gate  = σ(x·W_out_gate)
4. Fuse mask (if present), gating and projection with a tiny Triton kernel:
        left  = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask
5. Core reduction (the “triangle”):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is performed as a batched matrix‑multiply after reshaping:
        (B·H, N, N) × (B·H, N, N)ᵀ → (B·H, N, N)
6. Output layer‑norm, multiply by out_gate and final linear projection.
7. Cast back to the original dtype (usually float32).

Only step 4 is implemented in Triton; the heavy reduction uses PyTorch’s
highly‑optimized batched GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _gate_mask_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    left_gate_ptr,     # fp16 *
    right_gate_ptr,    # fp16 *
    mask_ptr,          # fp16 *
    left_out_ptr,      # fp16 *
    right_out_ptr,     # fp16 *
    total_elements,    # i64 scalar = B*N*N*H
    hidden_dim,        # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and optional mask.

    left_out  = left  * left_gate  * mask
    right_out = right * right_gate * mask
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elements

    # each element belongs to a (b,i,j) index; mask is shared across the hidden dim
    mask_idx = offs // hidden_dim

    left      = tl.load(left_ptr + offs,      mask=mask, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr + offs,mask=mask, other=0.0)
    m         = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # 1 if no mask

    left_out  = left  * left_gate  * m
    right_out = right * right_gate * m

    tl.store(left_out_ptr  + offs, left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_gate_mask(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch the Triton kernel that fuses mask and gating.
    All tensors must be contiguous fp16.
    """
    total = left.numel()
    hidden = left.shape[-1]

    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        hidden,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: (B, N, N, C) torch.Tensor
            - mask: (B, N, N) torch.Tensor or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (opt) – if True, ignore mask

    Returns:
        Tensor of shape (B, N, N, C) with the same dtype as ``input_tensor``.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the heavy part
    # ------------------------------------------------------------------
    orig_dtype = x.dtype
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Prepare fp16 weights (no bias in linear layers)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask + gating (fused Triton kernel)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure contiguity and fp16
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_gate_mask(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
        )
    else:
        # No mask (or ignored)
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM after reshaping.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, N, N, H) -> (B*H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # Batched matrix multiplication
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
