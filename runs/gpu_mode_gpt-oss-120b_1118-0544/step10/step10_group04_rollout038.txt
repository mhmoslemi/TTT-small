"""
Triton‑accelerated outgoing TriMul forward pass.
...
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # guard
    mask = offs < total_elems

    # index for mask (repeated hidden_dim times)
    mask_idx = offs // hidden_dim

    l = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = l * lg * m
    right_out = r * rg * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gating and mask with a Triton kernel.
    All tensors must be fp16 and contiguous.
    Returns (left, right) each of shape (B, N, N, H).
    """
    assert left_proj.is_contiguous() and right_proj.is_contiguous()
    assert left_gate.is_contiguous() and right_gate.is_contiguous()
    assert mask.is_contiguous()

    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out

    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.
    Input:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [B, N, N, dim] with same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input LayerNorm over last dimension (dim)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # 2. Cast to half precision for heavy compute
    x = x.to(torch.float16)

    # 3. Cast weights to half precision
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)

    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # 4. Linear projections
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)  # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # 6. Fuse mask, projection and gating
    if mask is not None and not nomask:

        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 7. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    B, N, _, H = left.shape

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    out = F.layer_norm(
        out,

        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B,N,N,dim)

    # 9. Cast back to original dtype if needed
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

"""
Triton‑accelerated outgoing TriMul forward pass.
The kernel fuses element‑wise projection, gating and optional mask multiplication.
The heavy N³ reduction is performed with a batched GEMM (torch.bmm), which
leverages cuBLAS for optimal performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask:
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # guard against out‑of‑bounds
    oob = offs >= total_elems

    # mask index (each mask value repeats hidden_dim times)
    mask_idx = offs // hidden_dim

    l  = tl.load(left_proj_ptr + offs,  mask=~oob, other=0.0)
    r  = tl.load(right_proj_ptr + offs, mask=~oob, other=0.0)
    lg = tl.load(left_gate_ptr + offs,  mask=~oob, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=~oob, other=0.0)
    m  = tl.load(mask_ptr + mask_idx,  mask=~oob, other=1.0)

    left_out  = l * lg * m
    right_out = r * rg * m

    tl.store(left_out_ptr + offs,  left_out,  mask=~oob)
    tl.store(right_out_ptr + offs, right_out, mask=~oob)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,

    Launch the Triton kernel that fuses projection, gating and mask.
    All tensors must be fp16 and contiguous.
    Returns (left, right) each of shape (B, N, N, H).
    """
    assert left_proj.is_contiguous() and right_proj.is_contiguous()
    assert left_gate.is_contiguous() and right_gate.is_contiguous()
    assert mask.is_contiguous()

        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],          # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out

    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
```
