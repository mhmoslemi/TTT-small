"""
TriMul (outgoing) forward pass – Triton‑accelerated projection‑gate‑mask fusion.

Algorithm
---------
1. Apply LayerNorm over the channel dimension of the 4‑D input.
2. Compute linear projections `left_proj` / `right_proj` and gating projections
   (`left_gate`, `right_gate`, `out_gate`) without bias.
3. If a mask is present, fuse projection, gating and mask in a single Triton
   kernel to reduce memory traffic.
4. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication (L @ Rᵀ) for each (b,h) pair.
5. Apply a second LayerNorm, multiply by `out_gate`, and a final linear
   projection back to the original channel dimension.
6. Cast the result back to the input dtype (typically float32).

The kernel in step 3 is mandatory to satisfy the “use Triton” requirement.
All other operations are performed with PyTorch for simplicity and numerical
stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # *float32
    right_proj_ptr,  # *float32
    left_gate_ptr,   # *float32
    right_gate_ptr,  # *float32
    mask_ptr,        # *float32
    left_out_ptr,    # *float32
    right_out_ptr,   # *float32
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask:
       left_out  = left_proj  * left_gate  * mask
       right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # mask for out‑of‑bounds elements
    valid = offs < total_elems

    # each (b,i,j) element repeats `hidden_dim` times in the flattened tensor
    mask_idx = offs // hidden_dim

    left_val   = tl.load(left_proj_ptr + offs,  mask=valid, other=0.0)
    right_val  = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    lgate_val  = tl.load(left_gate_ptr + offs,  mask=valid, other=0.0)
    rgate_val  = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    mask_val   = tl.load(mask_ptr + mask_idx,      mask=valid, other=1.0)

    left_out   = left_val * lgate_val * mask_val
    right_out  = right_val * rgate_val * mask_val

    tl.store(left_out_ptr + offs,  left_out,  mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch the Triton kernel that fuses projection, gating and mask.
    All tensors must be contiguous and of dtype torch.float32.
    Returns (left, right) each of shape (B, N, N, H).
    """
    assert left_proj.is_contiguous()
    assert right_proj.is_contiguous()
    assert left_gate.is_contiguous()
    assert right_gate.is_contiguous()
    assert mask.is_contiguous()

    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],  # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Linear projections (float32, bias‑free)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])   # (B, N, N, hidden_dim)
    right_proj = F.linear(x, weights["right_proj.weight"])  # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # (B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # (B, N, N, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4) Apply mask and fuse with projection & gating (Triton)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is float32 and contiguous
        mask_fp = mask.to(torch.float32).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp
        )
    else:
        # No mask (or mask ignored): simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM over (B*H) matrices.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 7) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
