"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the reference PyTorch module but reduces the number of
dense linear layers by concatenating projection and gate weights, runs all
compute‑intensive parts in fp16, and uses a small Triton kernel to apply the
optional mask. The core reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed as a batched GEMM using torch.bmm (cuBLAS), which is highly
optimized on NVIDIA GPUs. The final output is cast back to the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _mask_apply_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    mask_ptr,          # fp16 *
    left_out_ptr,      # fp16 *
    right_out_ptr,     # fp16 *
    total_elems,       # i64 scalar = B*N*N*H
    hidden_dim,        # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Element‑wise mask multiplication:
    left_out  = left  * mask
    right_out = right * mask
    mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # each hidden_dim entries share the same mask element
    mask_idx = offs // hidden_dim

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(left_out_ptr + offs, l * m, mask=mask)
    tl.store(right_out_ptr + offs, r * m, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict containing:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp16)
    # ------------------------------------------------------------------
    norm_weight_fp16 = weights["norm.weight"].to(torch.float16)
    norm_bias_fp16 = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(
        input_tensor.to(torch.float16),
        (dim,),
        weight=norm_weight_fp16,
        bias=norm_bias_fp16,
    )  # (B, N, N, dim)  fp16

    # ------------------------------------------------------------------
    # 2) Combined linear projections for left and right (proj + gate)
    # ------------------------------------------------------------------
    # concatenate proj and gate weights to halve the number of GEMMs
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, C)
    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, C)

    left_comb = F.linear(x, left_comb_weight)   # (B, N, N, 2*H)
    right_comb = F.linear(x, right_comb_weight)  # (B, N, N, 2*H)

    # split into proj and gate parts
    left_proj = left_comb[..., :hidden_dim]
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])  # (B,N,N,H)
    right_proj = right_comb[..., :hidden_dim]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])  # (B,N,N,H)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 3) Optional mask application (fused via Triton)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        total = left.numel()
        left_out = torch.empty_like(left)
        right_out = torch.empty_like(right)

        BLOCK_SIZE = 1024
        grid = (triton.cdiv(total, BLOCK_SIZE),)

        _mask_apply_kernel[grid](
            left,
            right,
            mask_fp16,
            left_out,
            right_out,
            total,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        left = left_out
        right = right_out

    # ------------------------------------------------------------------
    # 4) Out gate (no mask)
    # ------------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5) Core reduction: batched matrix multiplication over the shared axis k
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight_fp16 = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias_fp16 = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight_fp16,
        bias=to_out_norm_bias_fp16,
    )
    out = out * out_gate
    to_out_weight_fp16 = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_weight_fp16)  # (B, N, N, C)

    # ------------------------------------------------------------------
    # 7) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
