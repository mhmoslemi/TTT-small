def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    orig_dtype = x.dtype
    x = x.to(torch.float16)

    # Prepare concat weight for left/right projections and gates
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # concat
    concat_w = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0)
    # flatten input
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)  # (B*N*N, dim)
    combined = F.linear(x_flat, concat_w)  # (B*N*N, hidden_dim*5)
    # split
    splits = torch.split(combined, hidden_dim, dim=1)
    left_proj, right_proj, left_gate, right_gate, out_gate = splits
    # apply sigmoid to gates
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()
    # mask handling
    if mask is not None and not nomask:
        # mask shape (B,N,N)
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(left_proj, right_proj, left_gate, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
    # reshape for batched GEMM
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    # permute to (B, hidden_dim, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # batched matrix multiplication
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # to_out_norm
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    # multiply by out_gate
    out_gate = out_gate.view(B, N, N, hidden_dim)
    out = out * out_gate
    # final linear
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)
    # cast back
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)
    return out
```

"""
Optimized outgoing TriMul implementation.

The algorithm follows the reference PyTorch module but reduces the number of
expensive matrix multiplications:
1. A single fused linear layer computes the two projections (left/right) and
   the three gating vectors (left_gate, right_gate, out_gate) in one GEMM.
2. The gating vectors are passed through a sigmoid.
3. An element‑wise Triton kernel applies the optional mask and multiplies the
   projections with their respective gates.
4. The core triangular update is performed as a batched matrix multiplication
   (left @ rightᵀ) after reshaping the tensors, which is handled by
   torch.bmm (cuBLAS) – this is the most efficient path for the many small
   GEMMs required.
5. Output LayerNorm, out‑gate, and final linear projection complete the
   forward pass.

Only the mask‑gate‑projection fusion is performed in Triton, satisfying the
requirement to use a custom kernel while keeping the overall runtime low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse mask, gating and projection:
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    within = offs < total_elems

    # each element of mask corresponds to hidden_dim entries in the flattened tensors
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,   mask=within, other=0.0)
    right = tl.load(right_proj_ptr + offs,  mask=within, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,   mask=within, other=0.0)
    rgate = tl.load(right_gate_ptr + offs,  mask=within, other=0.0)
    m     = tl.load(mask_ptr + mask_idx,    mask=within, other=1.0)

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=within)
    tl.store(right_out_ptr + offs, right_out, mask=within)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch the Triton kernel that fuses mask, gating and projection.
    All tensors must be fp16 and contiguous.

    Returns:
        left_out, right_out – tensors of shape (B, N, N, hidden_dim)
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],   # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict containing:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive parts
    # ------------------------------------------------------------------
    orig_dtype = x.dtype
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Fuse the four linear projections (left/right + gates) into one GEMM
    # ------------------------------------------------------------------
    # Convert weights to fp16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Concatenate weights along the output dimension = 5 * hidden_dim
    concat_w = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0
    )  # (5*H, dim)

    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)  # (B*N*N, dim)

    # Single fused linear
    combined = F.linear(x_flat, concat_w)  # (B*N*N, 5*H)

    # Split into individual components
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        combined, hidden_dim, dim=1
    )
    # Apply sigmoid to gating vectors
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # ------------------------------------------------------------------
    # 4) Apply mask and fuse with projection & gating (Triton)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Reshape back to (B, N, N, H)
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5) Core reduction (TriMul) via batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Transform to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )
    # reshape out_gate to match out
    out_gate = out_gate.view(B, N, N, hidden_dim)
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 7) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
