@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_lm, stride_li, stride_lk,
    stride_rm, stride_rk, stride_rj,
    stride_om, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0) # block row
    pid_n = tl.program_id(axis=1) # block col
    pid_b = tl.program_id(axis=2) # batch

    # Offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers
    # left: (batch, i, k)
    left_ptrs = left_ptr + pid_b * stride_lm + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk
    # right: (batch, k, j)
    right_ptrs = right_ptr + pid_b * stride_rm + offs_k[:, None] * stride_rk + offs_n[None, :] * stride_rj

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles
        left = tl.load(left_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)
        right = tl.load(right_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(left, right)

        # Advance pointer
        left_ptrs += BLOCK_K * stride_lk
        right_ptrs += BLOCK_K * stride_rk

    # Write output
    out_ptrs = out_ptr + pid_b * stride_om + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the reference PyTorch module but fuses the
core reduction (out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]) into a
custom Triton kernel.  All heavy linear / gating operations are performed
in fp16 and the final result is cast back to the input dtype (float32).

The kernel computes a batched matrix multiplication for each (batch, hidden)
pair:
    out_flat[m, i, j] = Σ_k left_flat[m, i, k] * right_flat[m, j, k]
which is equivalent to the required einsum.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    left_ptr,          # *fp16   left tensor [batch, N, N]
    right_ptr,         # *fp16   right tensor [batch, N, N]
    out_ptr,           # *fp16   output tensor [batch, N, N]
    batch,             # i32     number of (B * hidden) batches
    N,                 # i32     sequence length
    K,                 # i32     reduction dimension (== N)
    stride_l_batch,    # i64     stride for batch dim in left
    stride_l_i,        # i64     stride for i (row) dim in left
    stride_l_k,        # i64     stride for k (col) dim in left
    stride_r_batch,    # i64     stride for batch dim in right
    stride_r_k,        # i64     stride for k dim in right (row in transposed view)
    stride_r_j,        # i64     stride for j dim in right (col in transposed view)
    stride_o_batch,    # i64     stride for batch dim in out
    stride_o_i,        # i64     stride for i dim in out
    stride_o_j,        # i64     stride for j dim in out
    BLOCK_M: tl.constexpr,  # tile size for output rows
    BLOCK_N: tl.constexpr,  # tile size for output cols
    BLOCK_K: tl.constexpr,  # tile size for reduction
):
    """Batched matrix multiplication:
    out[m, i, j] = Σ_k left[m, i, k] * right[m, j, k]
    """
    pid_m = tl.program_id(0)  # output row tile
    pid_n = tl.program_id(1)  # output col tile
    pid_b = tl.program_id(2)  # batch tile

    # Compute offsets for each dimension
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for the current batch
    left_base = left_ptr + pid_b * stride_l_batch + offs_i[:, None] * stride_l_i
    right_base = right_ptr + pid_b * stride_r_batch + offs_j[None, :] * stride_r_j

    # Accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # Load tiles
        left = tl.load(
            left_base + cur_k[None, :] * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Right is accessed as (k, j) -> use stride_r_k for k, stride_r_j for j
        right = tl.load(
            right_base + cur_k[:, None] * stride_r_k,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # Accumulate dot product
        acc += tl.dot(left, right)

    # Write back the result (cast to fp16)
    out_ptrs = out_ptr + pid_b * stride_o_batch + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict containing:
                * "dim" (int)        – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) LayerNorm over input channels
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive ops
    # --------------------------------------------------------------
    orig_dtype = x.dtype
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast model weights to fp16
    # --------------------------------------------------------------
    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections
    # --------------------------------------------------------------
    left_proj = F.linear(x, w_left_proj)   # (B, N, N, H)
    right_proj = F.linear(x, w_right_proj)

    # --------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x, w_out_gate))

    # --------------------------------------------------------------
    # 6) Apply mask (if present) and fuse with projection & gating
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM in Triton.
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # left shape is (B, N, N, H)

    # Rearrange to (B, H, N, N) and merge batch & hidden dims
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch = B * H
    left_flat = left_perm.view(batch, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(batch, N, N) # (B*H, N, N)

    out_flat = torch.empty_like(left_flat)

    # Strides (in element units)
    stride_l_batch, stride_l_i, stride_l_k = left_flat.stride()
    stride_r_batch, stride_r_j, stride_r_k = right_flat.stride()  # note order j,k
    stride_o_batch, stride_o_i, stride_o_j = out_flat.stride()

    # Triton block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),  # rows
        triton.cdiv(N, BLOCK_N),  # cols
        batch,                    # batch dimension
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch,
        N,
        N,  # K == N
        stride_l_batch,
        stride_l_i,
        stride_l_k,
        stride_r_batch,
        stride_r_k,
        stride_r_j,
        stride_o_batch,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
    )
    out = out * out_gate
    out = F.linear(out, w_to_out)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9) Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
