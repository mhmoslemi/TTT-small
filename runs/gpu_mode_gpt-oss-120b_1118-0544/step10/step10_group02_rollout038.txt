    M, N, K,
    stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,

5. Core reduction (TriMul):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched matrix multiplication with a custom Triton kernel.
6. Output LayerNorm, element‑wise multiplication with `out_gate`,
   final linear projection back to `dim`.
7. Cast back to the original dtype.

The Triton kernel performs a blocked batched GEMM (A @ Bᵀ) over the
hidden dimension, avoiding the expensive torch.bmm loop and yielding
significant speed‑ups for large sequence lengths.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_am, stride_ak,               # A strides (M, K)
    stride_bk, stride_bn,               # B strides (K, N)
    stride_cm, stride_cn,               # C strides (M, N)
    batch_stride_a, batch_stride_b, batch_stride_c,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a batch of matrices.

    A: (P, M, K)   fp16
    B: (P, K, N)   fp16  (already transposed)
    C: (P, M, N)   fp16
    """
    pid = tl.program_id(0)

    # number of tiles
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)

    # batch index and tile indices
    batch = pid // (num_tiles_m * num_tiles_n)
    tile_id = pid % (num_tiles_m * num_tiles_n)
    pid_m = tile_id // num_tiles_n
    pid_n = tile_id % num_tiles_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds (M,N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute base pointers for this batch
    a_batch = a_ptr + batch * batch_stride_a
    b_batch = b_ptr + batch * batch_stride_b
    c_batch = c_ptr + batch * batch_stride_c

    # pointers to the start of the tile
    a_ptrs = a_batch + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute remaining K elements for this tile
        k_offset = k * BLOCK_K
        k_valid = (tl.arange(0, BLOCK_K) < (K - k_offset))

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_valid[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back result
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(a: torch.Tensor,
                           b: torch.Tensor,
                           BLOCK_M: int = 64,
                           BLOCK_N: int = 64,
                           BLOCK_K: int = 64) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    a: (P, M, K)  fp16
    b: (P, K, N)  fp16  (already transposed)
    Returns c: (P, M, N) fp16
    """
    assert a.is_cuda and b.is_cuda
    P, M, K = a.shape
    _, _, N = b.shape

    c = torch.empty((P, M, N), dtype=a.dtype, device=a.device)

    # strides in elements (not bytes)
    stride_am = a.stride(1)
    stride_ak = a.stride(2)
    stride_bk = b.stride(1)
    stride_bn = b.stride(2)
    stride_cm = c.stride(1)
    stride_cn = c.stride(2)

    batch_stride_a = a.stride(0)
    batch_stride_b = b.stride(0)
    batch_stride_c = c.stride(0)

    num_tiles_m = (M + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (P * num_tiles_m * num_tiles_n,)

    _batched_matmul_kernel[grid](
        a,
        b,
        c,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict containing:
                * "dim" (int)          – input channel dimension C
                * "hidden_dim" (int)   – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast linear weights to fp16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5) Optional mask application (before gating)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # broadcast mask over hidden dimension
        mask_exp = mask.unsqueeze(-1).to(torch.float16)  # (B, N, N, 1)
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # --------------------------------------------------------------
    # 6) Gating projections
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 7) Fuse projection with gating
    # --------------------------------------------------------------
    left  = left_proj * left_gate   # (B, N, N, H)
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 8) Core reduction (TriMul) using Triton batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (P, N, N) where P = B * H
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # transpose right for A @ Bᵀ
    right_t = right_flat.permute(0, 2, 1).contiguous()

    # Triton batched matmul
    out_flat = _batched_matmul_triton(left_flat, right_t)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 9) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 10) Cast back to original dtype if needed
    # --------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
