We'll implement kernel similarly to typical matmul but with batch dimension.

Pseudo:

```
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bn, stride_bk,
    stride_cm, stride_cn, stride_co,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)   # batch index: 0..M-1
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile col

    block_row = pid_m * BLOCK_M
    block_col = pid_n * BLOCK_N

    # offsets for rows and cols within tile
    offs_m = block_row + tl.arange(0, BLOCK_M)
    offs_n = block_col + tl.arange(0, BLOCK_N)

    # create mask for out-of-bound rows/cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # compute pointers
        a_ptrs = a_ptr + pid_batch * stride_am + offs_m[:, None] * stride_an + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_batch * stride_bm + offs_n[:, None] * stride_bn + offs_k[None, :] * stride_bk  # note: B is transposed, we need B[j,k], and we want B_T[j,k] = B[k,j]; we can load B at (j,k) then dot product across k

        # load A tile
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        # load B tile
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # compute dot product across k dimension
        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_ptr + pid_batch * stride_cm + offs_m[:, None] * stride_cn + offs_n[None, :] * stride_co
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated projection‑gate‑mask fusion
and batched matrix multiplication.

Algorithm
---------
1. Layer‑norm the 4‑D input over the channel dimension.
2. Cast to FP16 for compute‑heavy stages.
3. Linear projections (left/right) and gating projections (left/right/out)
   using the provided weights (bias‑free).
4. Fuse projection, gating and optional mask with a custom Triton kernel.
5. Core reduction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication where for each batch‑head pair
   we compute `C = A @ Bᵀ`. Implemented with a Triton kernel that tiles
   the computation over the sequence dimension.
6. Output layer‑norm, multiply by out‑gate and final linear projection back
   to the original channel dimension.
7. Cast back to the input dtype.

The kernel in step 4 and the batched GEMM in step 5 are the Triton‑accelerated
parts.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gating and mask
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask:
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds
    mask = offs < total_elems

    # One mask value per (b,i,j) entry, repeated over the hidden dim
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launches the Triton kernel that fuses projection, gating and mask.
    All tensor arguments must be fp16 and contiguous.
    Returns (left, right) each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],          # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A, B: (M, N, K) where M = B * hidden_dim, N = seq_len, K = seq_len
#   C: (M, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bn, stride_bk,
    stride_cm, stride_cn, stride_co,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # batch index (0 .. M-1)
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile col

    # Compute tile start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K) -> shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + pid_batch * stride_am \
                 + offs_m[:, None] * stride_an \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load Bᵀ tile (K x N) -> shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + pid_batch * stride_bm \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Compute block matrix product
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_ptr + pid_batch * stride_cm \
             + offs_m[:, None] * stride_cn \
             + offs_n[None, :] * stride_co
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive stages
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast linear weights to FP16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections (FP16)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)   # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 6) Apply mask and fuse with projection & gating
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7) Core reduction (TriMul) – batched GEMM using Triton
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    B, N_seq, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)
    M = B * H
    left_flat = left_perm.view(M, N_seq, N_seq)
    right_flat = right_perm.view(M, N_seq, N_seq)

    out_flat = torch.empty_like(left_flat)

    # Tiling parameters (tuned for typical B200 GPU)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        M,
        triton.cdiv(N_seq, BLOCK_M),
        triton.cdiv(N_seq, BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M,
        N_seq,
        N_seq,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9) Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype:
        out = out.to(dtype)

    return out
