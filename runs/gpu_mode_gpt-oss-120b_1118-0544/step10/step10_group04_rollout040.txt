    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Cast to fp16
    x = x.to(torch.float16)

    # Cast weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Projections
    left_proj = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # Gating
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply mask and fuse with gates
    if mask is not None and not nomask:
        # ensure mask is contiguous and dtype fp16
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_mask_fuse(left_proj, right_proj, left_gate, right_gate, mask_fp16)
    else:
        # element-wise multiply
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Core reduction: batched GEMM
    B, N, _, H = left.shape
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # Output layer norm
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, to_out_w)  # (B,N,N,dim)

    # Cast back to input dtype if needed
    if out.dtype != input_tensor.dtype:

    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated mask‑gate‑projection fusion.

Algorithm
---------
1. Layer‑norm the 4‑D input over the channel dimension `dim`.
2. Cast to fp16 for the compute‑heavy stages.
3. Linear projections (`left_proj`, `right_proj`) and gating projections
   (`left_gate`, `right_gate`, `out_gate`) – all bias‑free.
4. If a mask is present, fuse mask, projection and gating with a small
   Triton kernel (element‑wise multiplication, no extra memory moves).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM via `torch.bmm` (cuBLAS).
6. Apply a second LayerNorm, multiply by `out_gate`, and a final linear
   projection back to the original channel dimension `dim`.
7. Cast the result back to the input dtype.

The kernel in step 4 is required to satisfy the “at least part in Triton”
rule.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _mask_fuse_kernel(
    left_proj_ptr,    # fp16 *
    right_proj_ptr,   # fp16 *
    left_gate_ptr,    # fp16 *
    right_gate_ptr,   # fp16 *
    mask_ptr,         # fp16 *
    left_out_ptr,     # fp16 *
    right_out_ptr,    # fp16 *
    total_elems,      # i64 scalar = B*N*N*H
    hidden_dim,       # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask, projection and gate:
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # Out‑of‑bounds guard
    valid = offs < total_elems

    # One mask value per (b,i,j) entry, repeated over hidden_dim
    mask_idx = offs // hidden_dim

    # Load values (masked)
    l = tl.load(left_proj_ptr + offs,   mask=valid, other=0.0)
    r = tl.load(right_proj_ptr + offs,  mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs,  mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx,    mask=valid, other=1.0)

    # Fuse
    left_out = l * lg * m
    right_out = r * rg * m

    # Store
    tl.store(left_out_ptr + offs,  left_out,  mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_mask_fuse(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """Launch the Triton kernel that fuses mask, projection and gate.
    All tensors must be fp16 and contiguous.
    Returns (left, right) each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _mask_fuse_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for the compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)  # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 6) Apply mask and fuse with projection & gating (Triton)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_mask_fuse(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask (or mask ignored): simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
