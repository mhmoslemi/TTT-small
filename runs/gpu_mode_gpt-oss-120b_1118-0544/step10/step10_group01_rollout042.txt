@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems,
    hidden_dim,
    BLOCK_SIZE: tl.constexpr
):

    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # compute mask index
    mask_idx = offs // hidden_dim

    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)

    right_out = right * rgate * m

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # cast to fp16
    x = x.to(torch.float16)

    # cast weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)  # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # mask handling
    if mask is None or nomask:
        mask_fp16 = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=torch.float16,
            device=input_tensor.device,
        )
    else:
        mask_fp16 = mask.to(torch.float16)

    # Fuse projection, gating, mask using Triton
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_fp16 = mask_fp16.contiguous()

    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    total = left_proj.numel()
    BLOCK_SIZE = 1024  # tune
    grid = (triton.cdiv(total, BLOCK_SIZE),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask_fp16,
        left,
        right,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # core reduction
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # output layernorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    out = out * out_gate

    out = torch.nn.functional.linear(out, to_out_w)

    # cast back
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

"""
TriMul (outgoing) forward pass with fused projection‑gate‑mask using Triton.

Algorithm:
1. Layer‑norm the input over the channel dimension.
2. Cast to fp16 for compute‑heavy parts.
3. Linear projections (left/right) and gating projections are computed with
   torch.nn.functional.linear.
4. A Triton kernel fuses the two projections, the two sigmoids and an optional
   mask into two tensors `left` and `right` of shape (B,N,N,H).  This removes
   two separate element‑wise passes.
5. Core reduction: for each hidden slice h we compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is implemented as a batched matrix multiplication
       out_flat = (B*H, N, N) @ (B*H, N, N)^T.
6. Output layer‑norm, element‑wise multiplication with the output gate and
   a final linear projection back to the original dimension.
7. Cast the result back to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,    # fp16 *
    right_proj_ptr,   # fp16 *
    left_gate_ptr,    # fp16 *
    right_gate_ptr,   # fp16 *
    mask_ptr,         # fp16 *
    left_out_ptr,     # fp16 *
    right_out_ptr,    # fp16 *
    total_elems,      # i64
    hidden_dim,       # i64
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and (optional) mask.

    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # guard
    mask = offs < total_elems

    # each element belongs to a (b,i,j) position; mask has one value per (b,i,j)
    mask_idx = offs // hidden_dim

    left   = tl.load(left_proj_ptr + offs,  mask=mask, other=tl.float16(0.0))
    right  = tl.load(right_proj_ptr + offs, mask=mask, other=tl.float16(0.0))
    lgate  = tl.load(left_gate_ptr + offs,  mask=mask, other=tl.float16(0.0))
    rgate  = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    m      = tl.load(mask_ptr + mask_idx,    mask=mask, other=tl.float16(1.0))

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=mask)

"""
TriMul (outgoing) forward pass with fused projection‑gate‑mask using Triton.

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to fp16 for compute‑heavy parts.
3. Compute left/right linear projections and their gates with torch.nn.functional.
4. A Triton kernel fuses projection, sigmoid gate and an optional mask into two
   tensors `left` and `right` of shape (B,N,N,H), eliminating separate
   element‑wise passes.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is performed as a batched matrix multiplication
       out_flat = (B*H, N, N) @ (B*H, N, N).T
6. Output layer‑norm, element‑wise multiplication with the output gate and a
   final linear projection back to the original dimension.
7. Cast the result back to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,    # fp16 *
    right_proj_ptr,   # fp16 *
    left_gate_ptr,    # fp16 *
    right_gate_ptr,   # fp16 *
    mask_ptr,         # fp16 *
    left_out_ptr,     # fp16 *
    right_out_ptr,    # fp16 *
    total_elems,      # i64
    hidden_dim,       # i64
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and (optional) mask.
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # each element belongs to a (b,i,j) position; mask has one value per (b,i,j)
    mask_idx = offs // hidden_dim

    left   = tl.load(left_proj_ptr  + offs, mask=mask, other=tl.float16(0.0))
    right  = tl.load(right_proj_ptr + offs, mask=mask, other=tl.float16(0.0))
    lgate  = tl.load(left_gate_ptr  + offs, mask=mask, other=tl.float16(0.0))
    rgate  = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    m      = tl.load(mask_ptr + mask_idx,      mask=mask, other=tl.float16(1.0))

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr  + offs, left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)
