```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def apply_mask_kernel(
    inp_ptr,  # fp16 *
    mask_ptr, # fp16 *
    out_ptr,  # fp16 *
    total_elems, # i64
    hidden_dim,  # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # mask index (per element)
    mask_idx = offs // hidden_dim
    val = tl.load(inp_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    tl.store(out_ptr + offs, val * m, mask=mask)

def _apply_mask(tensor: torch.Tensor, mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    # tensor: (B,N,N,H) fp16, contiguous
    # mask: (B,N,N) fp16, contiguous
    total = tensor.numel()
    out = torch.empty_like(tensor)
    grid = (triton.cdiv(total, block_size),)
    apply_mask_kernel[grid](
        tensor,
        mask,
        out,
        total,
        tensor.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    Implements the TriMul operation from AlphaFold3, fusing
    linear projections, gating, and optional masking with a Triton kernel.
    The core reduction is performed as a batched matrix multiplication.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1) Input LayerNorm over channel dimension `dim`
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    orig_dtype = x.dtype

    # 2) Cast to fp16 for compute‑intensive parts
    x = x.to(torch.float16)

    B, N, _, _ = x.shape
    # flatten for linear ops: (B*N*N, dim)
    x_flat = x.reshape(-1, dim)  # (B*N*N, dim)

    # 3) Combine left/right projections with their gates
    # left: [proj | gate], right: [proj | gate]
    left_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    right_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)

    # 4) Linear forward
    left_combined = F.linear(x_flat, left_weight)   # (B*N*N, 2*hidden_dim)
    right_combined = F.linear(x_flat, right_weight) # (B*N*N, 2*hidden_dim)

    # split projections and gates
    left_proj = left_combined[:, :hidden_dim]
    left_gate_pre = left_combined[:, hidden_dim:]
    right_proj = right_combined[:, :hidden_dim]
    right_gate_pre = right_combined[:, hidden_dim:]

    # 5) Apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # 6) Apply mask and fuse projection with gate
    # left = left_proj * left_gate * mask
    # right = right_proj * right_gate * mask
    left = left_proj * left_gate
    right = right_proj * right_gate

    if mask is not None and not nomask:
        # mask shape (B,N,N) -> broadcast to (B,N,N,hidden_dim)
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_mask(left, mask_fp16)
        right = _apply_mask(right, mask_fp16)

    # reshape back to (B,N,N,hidden_dim)
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # 7) Core reduction (TriMul): out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # reshape to (B*hidden_dim, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden_dim)

    # 8) Output LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # 9) Compute out_gate and apply
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_weight)).view(B, N, N, hidden_dim)
    out = out * out_gate

    # 10) Final linear projection back to `dim`
    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_weight)  # (B,N,N,dim)

    # 11) Cast back to original dtype
    out = out.to(orig_dtype)

    return out
```

"""
Outgoing Triangle Multiplicative Update (TriMul) – forward pass.

Implements the TriMul operation from AlphaFold3, fusing linear projections,
gating, and optional masking with a Triton kernel.  The core reduction
computes
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
as a batched matrix multiplication.  All heavy arithmetic is performed in
FP16; the final result is returned in the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def apply_mask_kernel(
    inp_ptr,    # fp16 *
    mask_ptr,   # fp16 *
    out_ptr,    # fp16 *
    total_elems,  # i64
    hidden_dim,   # i64
    BLOCK_SIZE: tl.constexpr,
):
    """Element‑wise multiplication of a tensor with a broadcast mask.

    Tensor shape: (B, N, N, hidden_dim) flattened.
    Mask shape:    (B, N, N)    flattened.
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # each `hidden_dim` consecutive elements share the same mask entry
    mask_idx = offs // hidden_dim

    x = tl.load(inp_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)
    tl.store(out_ptr + offs, x * m, mask=valid)


def _apply_mask(tensor: torch.Tensor, mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Apply a broadcast mask to a 4‑D tensor using the Triton kernel above.

    Args:
        tensor: (B, N, N, H) contiguous FP16 tensor.
        mask:   (B, N, N) contiguous FP16 tensor.
    Returns:
        Tensor of the same shape as `tensor` with the mask applied.
    """
    total = tensor.numel()
    out = torch.empty_like(tensor)
    grid = (triton.cdiv(total, block_size),)
    apply_mask_kernel[grid](
        tensor,
        mask,
        out,
        total,
        tensor.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask:         torch.Tensor of shape [B, N, N] or None
            - weights:      dict of model parameters (float32)
            - config:       dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    orig_dtype = x.dtype
    x = x.to(torch.float16)                     # compute in FP16
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2) Flatten for linear layers
    # ------------------------------------------------------------------
    x_flat = x.reshape(-1, dim)                 # (B*N*N, dim)

    # ------------------------------------------------------------------
    # 3) Combine projection and gate weights (left and right)
    # ------------------------------------------------------------------
    left_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                         # (2*H, dim)
    right_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                         # (2*H, dim)

    # ------------------------------------------------------------------
    # 4) Linear forward (proj + gate in one matmul each)
    # ------------------------------------------------------------------
    left_combined = F.linear(x_flat, left_weight)   # (B*N*N, 2*H)
    right_combined = F.linear(x_flat, right_weight) # (B*N*N, 2*H)

    # split into projection and gate pre‑sigmoid parts
    left_proj = left_combined[:, :hidden_dim]
    left_gate_pre = left_combined[:, hidden_dim:]
    right_proj = right_combined[:, :hidden_dim]
    right_gate_pre = right_combined[:, hidden_dim:]

    # ------------------------------------------------------------------
    # 5) Apply sigmoid to gate tensors
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 6) Fuse projection, gating and optional mask
    # ------------------------------------------------------------------
    left = left_proj * left_gate   # (B*N*N, H)
    right = right_proj * right_gate

    if mask is not None and not nomask:
        # ensure mask is FP16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_mask(left, mask_fp16)
        right = _apply_mask(right, mask_fp16)

    # reshape back to (B, N, N, H)
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # (B, H, N, N) -> (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # ------------------------------------------------------------------
    # 9) Compute out_gate and apply
    # ------------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_weight)).view(B, N, N, hidden_dim)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_weight)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 11) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)
    return out
