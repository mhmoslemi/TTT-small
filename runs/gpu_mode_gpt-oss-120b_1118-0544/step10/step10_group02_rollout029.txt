import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # index of mask dimension (b,i,k)
    mask_idx = offs // hidden_dim

    left = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 2048,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gating, and optional mask in a Triton kernel.
    All tensors must be contiguous fp16.
    Returns (left, right) with same shape as inputs.
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

        mask,

        right_out,
        total,

    )

"""
TriMul outgoing forward pass (AlphaFold3) – Triton‑fused projection & gating.

Algorithm
---------
1. Input LayerNorm over the channel dimension (`dim`).
2. Cast to FP16 for compute‑intensive stages.
3. Linear projections (left/right) and gating (left_gate, right_gate, out_gate).
4. Fuse projection, gating and optional mask with a Triton kernel
   (`_proj_gate_mask_kernel`).  This eliminates extra memory traffic.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   realized as a batched GEMM (left @ rightᵀ) on the reshaped tensors
   (B·H, N, N) – leverages cuBLAS tensor‑core fast path.
6. Output LayerNorm → element‑wise multiplication with `out_gate`
   → final linear projection back to `dim`.
7. Cast back to the input dtype.

The kernel only handles the fused projection/gating/mask; the reduction
uses PyTorch’s highly‑optimized batched matrix multiplication.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask:
       left_out  = left_proj  * left_gate  * mask
       right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds
    valid = offs < total_elems

    # Each hidden dimension repeats the same mask entry
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=valid, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=valid, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m     = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)

    All tensors must be fp16 and contiguous.
    Returns (left_out, right_out) each with shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

        left_proj,

        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],          # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
