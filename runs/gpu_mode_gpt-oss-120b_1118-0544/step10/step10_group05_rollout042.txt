@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_bh, stride_left_i, stride_left_k,
    stride_right_bh, stride_right_j, stride_right_k,
    stride_out_bh, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # row block
    pid_n = tl.program_id(1)  # col block
    pid_bh = tl.program_id(2)  # batch*hidden

    # Compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for boundaries
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left (M x K)
        left = tl.load(left_ptr + pid_bh * stride_left_bh
                        + offs_m[:, None] * stride_left_i
                        + offs_k[None, :] * stride_left_k,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # Load right transposed (K x N) -> need right[j,k] but we want (K,N) for dot
        # right_ptr is stored as (batch, j, k) with stride_right_j (for j) and stride_right_k (for k)
        # To load as (K,N): we load using k as first dim and j as second.
        right = tl.load(right_ptr + pid_bh * stride_right_bh
                         + offs_k[:, None] * stride_right_k
                         + offs_n[None, :] * stride_right_j,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        # Compute dot product, accumulate
        acc += tl.dot(left, right)

    # Convert to fp16
    out = acc.to(tl.float16)

    # Store results
    tl.store(out_ptr + pid_bh * stride_out_bh
                     + offs_m[:, None] * stride_out_i
                     + offs_n[None, :] * stride_out_j,
             out,
             mask=mask_m[:, None] & mask_n[None, :])

The kernel performs a batched matrix multiplication for each batch and hidden
channel:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
All other operations (layer‑norm, linear projections, gating, mask fusion)
are handled in PyTorch, while the heavy N³ reduction is computed in a custom
Triton kernel using shared‑memory tiling.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# -------------------------------------------------------------------------
# Triton kernel: fuse projection, gating and mask (optional)
# -------------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64
    hidden_dim,      # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    within = offs < total_elems
    # each hidden entry repeats the same mask element
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=within, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=within, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=within, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=within, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=within, other=1.0)

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr  + offs, left_out,  mask=within)
    tl.store(right_out_ptr + offs, right_out, mask=within)

def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gating and (optional) mask using a Triton kernel.
    All tensors must be fp16, contiguous and on the same CUDA device.
    Returns left and right tensors of shape (B,N,N,H) in fp16.
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],   # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out

# -------------------------------------------------------------------------
# Triton kernel: core TriMul reduction (batched matrix multiplication)
# -------------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,   # fp16 *
    right_ptr,  # fp16 *
    out_ptr,    # fp16 *
    B, H, N,    # i64 scalars
    stride_left_bh, stride_left_i, stride_left_k,
    stride_right_bh, stride_right_j, stride_right_k,
    stride_out_bh, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # rows block
    pid_n = tl.program_id(1)  # cols block
    pid_bh = tl.program_id(2) # batch*hidden

    # block coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (M, K)
        left = tl.load(
            left_ptr
            + pid_bh * stride_left_bh
            + offs_m[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right tile transposed: we need (K, N)
        # right is stored as (batch, j, k) -> stride_right_j for j, stride_right_k for k
        right = tl.load(
            right_ptr
            + pid_bh * stride_right_bh
            + offs_k[:, None] * stride_right_k
            + offs_n[None, :] * stride_right_j,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        acc += tl.dot(left, right)   # (M,K) dot (K,N) -> (M,N)

    # write back
    out = acc.to(tl.float16)
    tl.store(
        out_ptr
        + pid_bh * stride_out_bh
        + offs_m[:, None] * stride_out_i
        + offs_n[None, :] * stride_out_j,
        out,
        mask=mask_m[:, None] & mask_n[None, :],
    )

# -------------------------------------------------------------------------
# Entry point
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm over channel dimension `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast linear weights to fp16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections (fp16)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 6) Apply mask and fuse with projection & gating
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask (or mask ignored)
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # broadcast mask over hidden dimension
            left = left * mask.unsqueeze(-1)
            right = right * mask.unsqueeze(-1)

    # --------------------------------------------------------------
    # 7) Core TriMul reduction (batched GEMM)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) for a batched GEMM
    B, N, _, H = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)

    # strides (in elements)
    stride_left_bh = left_flat.stride(0)
    stride_left_i  = left_flat.stride(1)
    stride_left_k  = left_flat.stride(2)

    stride_right_bh = right_flat.stride(0)
    stride_right_j  = right_flat.stride(1)   # j dimension
    stride_right_k  = right_flat.stride(2)   # k dimension

    stride_out_bh = out_flat.stride(0)
    stride_out_i  = out_flat.stride(1)
    stride_out_j  = out_flat.stride(2)

    # Block sizes – tuned for typical N up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B,
        H,
        N,
        stride_left_bh,
        stride_left_i,
        stride_left_k,
        stride_right_bh,
        stride_right_j,
        stride_right_k,
        stride_out_bh,
        stride_out_i,
        stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9) Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
