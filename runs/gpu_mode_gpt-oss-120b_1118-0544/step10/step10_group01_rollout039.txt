```python
@triton.jit
def _mask_mul_kernel(
    proj_left_ptr, proj_right_ptr,
    gate_left_ptr, gate_right_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Determine mask index (per (b,i,j))
    mask_idx = offs // hidden_dim

    proj_l = tl.load(proj_left_ptr + offs, mask=mask, other=0.0)
    proj_r = tl.load(proj_right_ptr + offs, mask=mask, other=0.0)
    gate_l = tl.load(gate_left_ptr + offs, mask=mask, other=0.0)
    gate_r = tl.load(gate_right_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out_l = proj_l * gate_l * m
    out_r = proj_r * gate_r * m

    tl.store(out_left_ptr + offs, out_l, mask=mask)
    tl.store(out_right_ptr + offs, out_r, mask=mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16
    x = x.to(torch.float16)

    # Convert weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Combined projection weights
    proj_weight = torch.cat([left_proj_w, right_proj_w], dim=0)  # (2*H, dim)
    proj = torch.nn.functional.linear(x, proj_weight)  # (B,N,N,2H)
    proj_left, proj_right = proj[..., :hidden_dim], proj[..., hidden_dim:]

    # Combined gate weights
    gate_weight = torch.cat([left_gate_w, right_gate_w], dim=0)
    gate = torch.nn.functional.linear(x, gate_weight)
    gate = torch.sigmoid(gate)
    left_gate, right_gate = gate[..., :hidden_dim], gate[..., hidden_dim:]

    # Apply mask and fuse
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        # allocate output tensors
        left = torch.empty_like(proj_left)
        right = torch.empty_like(proj_right)
        total = proj_left.numel()
        grid = (triton.cdiv(total, 1024),)
        _mask_mul_kernel[grid](
            proj_left,
            proj_right,
            left_gate,
            right_gate,
            mask_fp16,
            left,
            right,
            total,
            hidden_dim,
            BLOCK_SIZE=1024,
        )
    else:
        left = proj_left * left_gate
        right = proj_right * right_gate

    # Core reduction
    B, N, _, H = left.shape
    # reshape for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # out gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to original dtype

"""
TriMul (outgoing) forward pass – optimized implementation.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for the compute‑intensive part.
3. Fuse the two linear projections (left/right) into a single matmul and
   similarly fuse the two gating projections.  Splitting the result yields
   `left_proj`, `right_proj`, `left_gate`, `right_gate`.
4. A small Triton kernel multiplies the projections with the gates and the
   optional mask, producing the tensors `left` and `right` of shape
   (B, N, N, H).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication using `torch.bmm`.
6. Apply a second Layer‑norm, multiply by the output gate and project back to
   the original dimension with a final linear layer.
7. Cast the result back to the original dtype.

Only the element‑wise fusion with mask (step 4) is performed in Triton;
all heavy linear algebra uses highly‑optimized PyTorch/cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _mask_mul_kernel(
    proj_left_ptr,   # fp16 *
    proj_right_ptr,  # fp16 *
    gate_left_ptr,   # fp16 *
    gate_right_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    out_left_ptr,    # fp16 *
    out_right_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and (optional) mask.
    out_left  = proj_left  * gate_left  * mask
    out_right = proj_right * gate_right * mask
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # one mask value per (b,i,j) → mask index = element_idx // H
    mask_idx = offs // hidden_dim

    pl = tl.load(proj_left_ptr + offs,  mask=mask, other=0.0)
    pr = tl.load(proj_right_ptr + offs, mask=mask, other=0.0)
    gl = tl.load(gate_left_ptr + offs,  mask=mask, other=0.0)
    gr = tl.load(gate_right_ptr + offs, mask=mask, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out_l = pl * gl * m
    out_r = pr * gr * m

    tl.store(out_left_ptr + offs,  out_l, mask=mask)
    tl.store(out_right_ptr + offs, out_r, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with the model parameters (float32)
            - config: dict with:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Prepare FP16 weights
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Combined linear projections (left & right)
    # ------------------------------------------------------------------
    proj_weight = torch.cat([left_proj_w, right_proj_w], dim=0)  # (2*H, dim)
    proj = F.linear(x, proj_weight)  # (B, N, N, 2*H)
    left_proj, right_proj = proj[..., :hidden_dim], proj[..., hidden_dim:]

    # ------------------------------------------------------------------
    # 5) Combined gating projections (left_gate & right_gate)
    # ------------------------------------------------------------------
    gate_weight = torch.cat([left_gate_w, right_gate_w], dim=0)  # (2*H, dim)
    gate = F.linear(x, gate_weight)
    gate = torch.sigmoid(gate)
    left_gate, right_gate = gate[..., :hidden_dim], gate[..., hidden_dim:]

    # ------------------------------------------------------------------
    # 6) Apply mask and fuse with projection & gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is contiguous FP16
        mask_fp16 = mask.to(torch.float16).contiguous()
        # Allocate output tensors
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        total = left_proj.numel()
        grid = (triton.cdiv(total, 1024),)
        _mask_mul_kernel[grid](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            left,
            right,
            total,
            hidden_dim,
            BLOCK_SIZE=1024,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (FP16)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # output gate (shape B,N,N,hidden_dim)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
