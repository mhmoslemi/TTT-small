"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for the compute‑intensive part.
3. Linear projections (left/right) and gating projections
   (left_gate, right_gate, out_gate) – all bias‑free.
4. Fuse projection, gating and optional mask using a small Triton kernel.
5. Core reduction (TriMul):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (A @ Bᵀ) with a custom Triton kernel.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and apply the final linear projection back to `dim`.
7. Cast the result back to the original dtype.

The implementation mixes PyTorch for easy operations (layer‑norm,
linear) and Triton for the two heavy‑weight fusions: mask‑gate‑projection
and the batched matrix multiplication.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask:
       left_out  = left_proj  * left_gate  * mask
       right_out = right_proj * right_gate * mask
       The mask shape is [B, N, N] and is broadcast over the hidden dim.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    # each hidden entry repeats the same mask element
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch Triton kernel that fuses projection, gating and mask.
    All tensors must be fp16, contiguous and on the same device.
    Returns:
        left, right – tensors of shape (B, N, N, H) in fp16.
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],   # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (M,K)
    stride_bk, stride_bn,          # Bᵀ: (N,K) – note swapped strides
    stride_cm, stride_cn,          # C: (M,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C = A @ Bᵀ   where A,B ∈ (batch, M, K) with M=N, K=N."""
    batch_id = tl.program_id(1)
    pid = tl.program_id(0)

    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Pointers to the start of the batch
    a_batch = a_ptr + batch_id * stride_am * M
    b_batch = b_ptr + batch_id * stride_bk * K

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A block (M x K)
        a = tl.load(
            a_batch
            + (offs_m[:, None] * stride_am)
            + (offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )

        # Load Bᵀ block (N x K) → original B element B[j,k]
        b = tl.load(
            b_batch
            + (offs_n[:, None] * stride_bn)   # note swapped stride
            + (offs_k[None, :] * stride_bk),
            mask=mask_n[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )
        # Compute block dot product
        acc += tl.dot(a, b)

        # Advance pointers (handled implicitly by k loop)

    # Write back result in fp16
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + batch_id * stride_cm * M
        + (offs_m[:, None] * stride_cm)
        + (offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm over channel dimension `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast linear weights to fp16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections (fp16)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 6) Apply mask and fuse with projection & gating (Triton)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7) Core reduction (TriMul) – batched GEMM with Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Strides for the contiguous tensors
    stride_am = left_flat.stride(1)   # = N
    stride_ak = left_flat.stride(2)   # = 1
    # For Bᵀ we swap the original strides
    stride_bk = right_flat.stride(2)  # original col stride = 1
    stride_bn = right_flat.stride(1)  # original row stride = N
    stride_cm = out_flat.stride(1)    # = N
    stride_cn = out_flat.stride(2)    # = 1

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_pid_m * num_pid_n, B * H)

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9) Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
