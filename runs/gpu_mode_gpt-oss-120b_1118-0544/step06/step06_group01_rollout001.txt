"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Input LayerNorm (float32 for stability).
2. Cast to fp16 and compute three linear projections in a fused manner:
   * left_proj + left_gate   (concatenated weight)
   * right_proj + right_gate (concatenated weight)
   * out_gate                (separate weight)
   Apply sigmoid to gate outputs.
3. Fuse mask (if present) and element‑wise gate multiplication using a
   small Triton kernel to keep memory traffic low.
4. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched GEMM (left @ rightᵀ) over the sequence dimension.
5. Output LayerNorm, multiply by out_gate and apply the final linear
   projection back to the original feature dimension.
All heavy arithmetic is performed in fp16; the final result is cast back
to the original dtype (usually float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,            # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,  # same shape as left/right
    mask_ptr,                       # [B,N,N] flattened
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask?
        right = right * right_gate * mask?
    Mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden_dim entries share a mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)  (C == dim)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for compute‑heavy parts
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Load linear weights (fp16) and concatenate where possible
    # -----------------------------------------------------------------
    # Projections
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)   # (hidden_dim, dim)
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)  # (hidden_dim, dim)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)  # (hidden_dim, dim)

    # Gates / final projection
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)    # (hidden_dim, dim)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)  # (hidden_dim,)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)    # (hidden_dim,)
    to_out_w     = weights["to_out.weight"].to(device).to(torch.float16)      # (dim, hidden_dim)

    # -----------------------------------------------------------------
    # 3. Linear projections + gates (fused where possible)
    # -----------------------------------------------------------------
    B, N, _, _ = x.shape
    # flatten batch & sequence dims: (B*N*N, dim)
    x_flat = x.view(-1, dim)  # (BNN, dim)

    # left_proj + left_gate fused
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*hidden_dim, dim)
    left_concat = F.linear(x_flat, left_concat_w)                # (BNN, 2*hidden_dim)
    left      = left_concat[:, :hidden_dim].view(B, N, N, hidden_dim)
    left_gate = left_concat[:, hidden_dim:].view(B, N, N, hidden_dim)

    # right_proj + right_gate fused
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*hidden_dim, dim)
    right_concat = F.linear(x_flat, right_concat_w)                  # (BNN, 2*hidden_dim)
    right      = right_concat[:, :hidden_dim].view(B, N, N, hidden_dim)
    right_gate = right_concat[:, hidden_dim:].view(B, N, N, hidden_dim)

    # out_gate (separate)
    out_gate = F.linear(x_flat, out_gate_w)               # (BNN, hidden_dim)
    out_gate = torch.sigmoid(out_gate).view(B, N, N, hidden_dim)

    # Apply sigmoid to gate tensors
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # -----------------------------------------------------------------
    # 4. Fuse mask and gate multiplication (in‑place) via Triton
    # -----------------------------------------------------------------
    use_mask = (mask is not None) and (not nomask)
    if use_mask:
        # mask: (B,N,N) -> (B*N*N,)
        mask_fp = mask.to(device).to(torch.float16).contiguous().view(-1)

        # Flatten tensors for kernel
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)

        total = left_flat.numel()
        BLOCK_SIZE = 1024

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _mask_gate_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_fp,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # Reshape back (view does not copy)
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ rightᵀ) over sequence dim
    # -----------------------------------------------------------------
    # left/right: (B, N, N, H)
    # reshape to (B*H, N, N) for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out_gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # Cast back to the original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
