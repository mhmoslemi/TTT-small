"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input (float32) using provided weight/bias.
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (bias‑less) → left, right, left_gate, right_gate, out_gate.
4. Fuse mask (if present) and the left/right gates in a single Triton kernel,
   writing the gated tensors back in‑place.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM (left @ rightᵀ) over B·H matrices.
6. Apply a second LayerNorm over the hidden dimension, multiply by out_gate,
   and a final linear projection back to the original channel dimension.
7. Cast back to the input dtype and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: in‑place gate + optional mask fusion
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,            # [B,N,N,H] flattened, fp16
    left_gate_ptr, right_gate_ptr,  # same shape, fp16
    mask_ptr,                       # [B,N,N] flattened, fp16
    total_elems, hidden_dim,        # runtime scalars (int)
    HAS_MASK: tl.constexpr,         # compile‑time bool
    BLOCK_SIZE: tl.constexpr,       # compile‑time int
):
    """
    For each element e in the flattened left/right tensors:
        left[e]  = left[e]  * left_gate[e]  * mask   (if mask present)
        right[e] = right[e] * right_gate[e] * mask   (if mask present)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load
    left       = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right      = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate  = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each mask entry corresponds to hidden_dim consecutive elements
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left  = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr + offs,  left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)  (float32)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32)
            - config: dict containing:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – bool, true if mask should be ignored

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor.to(torch.float32), (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2. Cast to fp16 for the heavy compute path
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (bias‑less) in fp16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left      = F.linear(x, left_proj_w)                     # (B,N,N,H)
    right     = F.linear(x, right_proj_w)                    # (B,N,N,H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))    # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))   # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))     # (B,N,N,H)

    # --------------------------------------------------------------
    # 4. Fuse mask and left/right gates (in‑place) via Triton
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # mask broadcasted over the hidden dimension
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # Flatten tensors for the kernel
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)  # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – simple element‑wise gate fusion
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction via batched GEMM (L @ Rᵀ)
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (uses cuBLAS tensor cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Restore (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating, and final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # --------------------------------------------------------------
    # 7. Cast back to original dtype and return
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
