@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid_batch = tl.program_id(0)  # batch index
    pid_m = tl.program_id(1)  # row tile
    pid_n = tl.program_id(2)  # col tile

    # compute block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # init accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # mask for bounds
        a_valid = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        b_valid = (offs_k < K)[:, None] & (offs_n < N)[None, :]

        # load A and B
        a_ptrs = A_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = B_ptr + pid_batch * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs, mask=a_valid, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=b_valid, other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = C_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    # cast to fp16 before store (or store as fp32 if output dtype is fp16)
    # We'll output fp16; the subsequent layernorm expects fp16 input.
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m[:, None] < M) & (offs_n < N))

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm Overview
------------------
1. Input LayerNorm (float32 for numeric stability).
2. Cast to fp16 for the compute‑heavy path.
3. Linear projections (left/right) and three sigmoid gates (left, right, out).
4. Fuse mask (if present) and the left/right element‑wise gates with a tiny
   Triton kernel (`_gate_mask_kernel`).
5. Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication L @ Rᵀ per hidden channel.
   Implemented with a custom Triton kernel (`_batched_matmul_kernel`).
6. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate and a final linear projection back to the original dimension.
7. Cast back to the original dtype and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,            # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,  # same shape
    mask_ptr,                       # [B,N,N] flattened
    total_elems, hidden_dim,        # runtime scalars
    HAS_MASK: tl.constexpr,         # compile‑time bool
    BLOCK_SIZE: tl.constexpr,       # compile‑time int
):
    """In‑place fusion:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
       mask is broadcast over the hidden dimension H.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden slice shares the same mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # store back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

    """Batched matrix multiplication C = A @ Bᵀ
       A: (batch, M, K)   fp16
       B: (batch, N, K)   fp16 (original layout, we treat rows as K, cols as N)
       C: (batch, M, N)   fp16
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Pointers
        a_ptrs = (A_ptr
                  + pid_batch * stride_a_batch
                  + offs_m[:, None] * stride_a_m
                  + offs_k[None, :] * stride_a_k)
        b_ptrs = (B_ptr
                  + pid_batch * stride_b_batch
                  + offs_k[:, None] * stride_b_k   # row = k (original col)
                  + offs_n[None, :] * stride_b_n)  # col = n (original row)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        # Accumulate dot product
        acc += tl.dot(a, b)

    # Write back result in fp16
    c_ptrs = (C_ptr
              + pid_batch * stride_c_batch
              + offs_m[:, None] * stride_c_m
              + offs_n[None, :] * stride_c_n)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C) (float32)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------ #
    # Unpack inputs
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------ #
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------ #
    # 2. Cast to fp16 for the compute‑heavy part
    # ------------------------------------------------------------------ #
    x = x.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 3. Linear projections (no bias) – fp16
    # ------------------------------------------------------------------ #
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)          # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)        # (B,N,N,hidden_dim)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B,N,N,hidden_dim)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------ #
    # 4. Fuse mask and gates (in‑place) with Triton
    # ------------------------------------------------------------------ #
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten tensors for the kernel
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back to original shapes
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Core contraction: batched matmul L @ Rᵀ using Triton
    # ------------------------------------------------------------------ #
    B, N, _, H = left.shape
    # Rearrange to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Allocate output
    out_mat = torch.empty_like(left_mat)

    # Block sizes (tuned for typical B200 GPUs)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                          # batch dimension
        triton.cdiv(N, BLOCK_M),        # M tiles
        triton.cdiv(N, BLOCK_N),        # N tiles
    )

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        B * H, N, N, N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        # For B we need strides such that we read Bᵀ without an explicit transpose:
        # original right_mat layout is (batch, N, N); we treat rows as K (original col)
        # and cols as N (original row).
        right_mat.stride(0), right_mat.stride(2), right_mat.stride(1),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 6. Output LayerNorm, output gate, and final linear projection
    # ------------------------------------------------------------------ #
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (still fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)      # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------ #
    # 7. Cast back to original dtype
    # ------------------------------------------------------------------ #
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
