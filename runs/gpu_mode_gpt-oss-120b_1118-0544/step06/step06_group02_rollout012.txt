...

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    # compute mask index present in mask tensor (broadcasted over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor, block_size: int = 8192) -> torch.Tensor:
    """

        gate,
        mask,

"""
TriMul (outgoing) forward pass using Triton for gate‑mask fusion.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) with the provided weight/bias.
2. Cast to fp16 for the compute‑intensive portion.
3. Compute four linear projections (left/right and their gates) in fp16.
4. Fuse element‑wise multiplication of projection, gate and optional mask
   with a tiny Triton kernel (`_gate_mask_kernel`). This saves two extra
   memory passes.
5. Core O(N³·H) reduction:
   * reshape left/right from (B,N,N,H) → (B*H,N,N) and use `torch.bmm`
     (batched GEMM) which maps to cuBLAS/Tensor‑core kernels.
6. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate and linearly project back to the original channel dimension.
7. Cast the result back to the input dtype (float32).

The kernel only handles the element‑wise fusion; the heavy cubic reduction
leverages cuBLAS for maximum throughput.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) on a flattened view
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask is broadcasted over the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 8192,
) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using a Triton kernel.

    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],      # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out

# ----------------------------------------------------------------------

        - mask        : torch.Tensor of shape [B, N, N] (or None)

        - config      : dict with keys:
            * "dim" (int) – input channel dimension C
            * "hidden_dim" (int) – hidden dimension H
            * "nomask" (bool, optional) – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input).
    """

    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) → (B*H, N, N)
    left_reshaped  = left.permute(0, 3, 1, 2).contiguous().view(-1, left.shape[1], left.shape[2])
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(-1, right.shape[1], right.shape[2])

    out_reshaped = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))

    # back to (B, N, N, H)
    out = out_reshaped.view(left.shape[0], hidden_dim, left.shape[1], left.shape[2])
    out = out.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Second LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
