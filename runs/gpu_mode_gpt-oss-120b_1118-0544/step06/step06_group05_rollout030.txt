@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block start offsets
    block_m = pid_m * BLOCK_M
    block_n = pid_n * BLOCK_N

    offs_m = block_m + tl.arange(0, BLOCK_M)
    offs_n = block_n + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Mask out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers base for this batch
    a_ptrs = A_ptr + pid_batch * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_batch * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute actual k range offset
        k_offset = k * BLOCK_K
        # Offsets for this block:
        cur_k = k_offset + offs_k
        # mask for current k block
        cur_mask_k = cur_k < K

        # Load A block
        a = tl.load(a_ptrs, mask=mask_m[:, None] & cur_mask_k[None, :], other=0.0)
        # Load B block
        b = tl.load(b_ptrs, mask=cur_mask_k[:, None] & mask_n[None, :], other=0.0)
        # Compute dot product
        acc += tl.dot(a, b)

        # Increment pointers for next k block
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # Write back
    c_ptrs = C_ptr + pid_batch * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    # Cast accumulator to output dtype (float16)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass accelerated with Triton.

Algorithm
---------
1. Layer‑norm the input tensor over the feature dimension `dim`.
2. Cast to float16 and apply linear projections for left/right values and
   three gates (left, right, out).  The gate weights are passed through a sigmoid.
3. Fuse projection, gating and (optional) mask by element‑wise multiplication.
4. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication where for each batch * head we
   compute `left @ rightᵀ`.  The multiplication is performed by a custom
   Triton kernel that tiles the matrices (BLOCK_M×BLOCK_N) and accumulates
   in FP32.
5. Apply a second Layer‑norm over the hidden dimension, multiply by the
   output gate and a final linear projection back to the original `dim`.
6. Cast the result back to the original dtype.

The only Triton component is the batched GEMM kernel; all other
operations use high‑performance PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM computing C = A @ Bᵀ  (A,B ∈ ℝ^{batch,N,N})
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,                         # all equal to sequence length N
    stride_Ab, stride_Am, stride_Ak, # batch, row, col strides for A
    stride_Bb, stride_Bk, stride_Bn, # batch, row, col strides for B (used as Bᵀ)
    stride_Cb, stride_Cm, stride_Cn, # batch, row, col strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)      # batch * hidden dimension
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block column

    # ----- block start offsets -----
    block_m = pid_m * BLOCK_M
    block_n = pid_n * BLOCK_N

    offs_m = block_m + tl.arange(0, BLOCK_M)
    offs_n = block_n + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ----- pointers to the current batch -----
    a_batch_off = pid_batch * stride_Ab
    b_batch_off = pid_batch * stride_Bb
    c_batch_off = pid_batch * stride_Cb

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- iterate over K dimension -----
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k_idx in range(0, num_k_blocks):
        cur_k = k_idx * BLOCK_K
        k_off = cur_k + offs_k                     # actual K indices
        mask_k = k_off < K

        # A: (BLOCK_M, BLOCK_K)   <-- rows i, cols k
        a_ptrs = A_ptr + a_batch_off + offs_m[:, None] * stride_Am + k_off[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Bᵀ: (BLOCK_K, BLOCK_N) <-- rows k, cols j   (note the transposed layout)
        b_ptrs = B_ptr + b_batch_off + k_off[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # ----- store result -----
    c_ptrs = C_ptr + c_batch_off + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C]  (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None (0/1)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """

    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(dtype_in)
    norm_b = weights["norm.bias"].to(dtype_in)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # --------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast all linear weights to FP16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # 4) Linear projections (FP16)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)

    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask exists but caller asked to ignore it – still apply broadcasting
            mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask_fp16
            right = right * mask_fp16

    # --------------------------------------------------------------
    # 6) Core reduction: batched GEMM (left @ rightᵀ) via Triton
    # --------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # reshape to (B*H, N, N) for batched GEMM
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # strides for the kernel
    stride_a_batch = left_flat.stride(0)
    stride_a_row   = left_flat.stride(1)   # M stride
    stride_a_col   = left_flat.stride(2)   # K stride

    stride_b_batch = right_flat.stride(0)
    stride_b_row   = right_flat.stride(1)  # K stride (treated as rows)
    stride_b_col   = right_flat.stride(2)  # N stride (treated as cols)

    stride_c_batch = out_flat.stride(0)
    stride_c_row   = out_flat.stride(1)
    stride_c_col   = out_flat.stride(2)

    # Tile sizes – tuned for typical B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H,                     # batch * hidden
            triton.cdiv(N, BLOCK_M),   # tiles over M
            triton.cdiv(N, BLOCK_N))   # tiles over N

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                     # M, N, K
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(out, (H,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 8) Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(dtype_in)

    return out
