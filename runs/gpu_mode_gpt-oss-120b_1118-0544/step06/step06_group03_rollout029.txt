@triton.jit
def bmm_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # compute pid info
    pid = tl.program_id(0)
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    total_tiles = BATCH * num_pid_m * num_pid_n
    # but each pid corresponds to one tile; we have total size = BATCH * num_pid_m * num_pid_n
    # guard: if pid >= total_tiles: return (should not happen)
    # compute indices:
    batch = pid // (num_pid_m * num_pid_n)
    tile_index = pid % (num_pid_m * num_pid_n)
    pid_m = tile_index // num_pid_n
    pid_n = tile_index % num_pid_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for A and B
    a_ptrs = a_ptr + batch * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    # B transposed: original B shape (BATCH, N, K). We need B_T shape (BATCH, K, N). So we treat B as B_T with stride swapped
    b_ptrs = b_ptr + batch * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load with mask (if out-of-bounds)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # store result
    c_ptrs = c_ptr + batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass – Triton‑accelerated mask/gate fusion and
batched GEMM.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for compute‑heavy ops.
3. Linear projections → left/right tensors of shape (B,N,N,H).
4. Gating (sigmoid) → left_gate, right_gate, out_gate.
5. Fuse projection, gate and optional mask in a Triton kernel
   (proj * gate * mask).  If there is no mask the fusion becomes a
   simple element‑wise product.
6. Core reduction (the `einsum` in the reference):
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM with the right operand transposed,
   using a custom Triton kernel to avoid extra PyTorch permutes.
7. Output LayerNorm on the hidden dimension H, multiply by `out_gate`,
   and a final linear projection back to the original dimension `dim`.
8. Cast the result back to the input dtype (usually float32).

The kernel in step 5 is mandatory; step 6 also uses Triton to keep the
whole computationally‑intensive path on‑device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Kernel 1: fuse projection, gate and (optional) mask
# --------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # int: B*N*N*H
    hidden_dim,        # int: H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # bounds
    mask = offs < total_elems

    # index of the corresponding mask element (broadcast over H)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 2048) -> torch.Tensor:
    """
    Fuse elementwise projection, gate and mask using Triton.
    All tensors must be fp16 contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)   broadcast over H
    Returns proj * gate * mask  (shape (B,N,N,H)).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# --------------------------------------------------------------
# Kernel 2: batched GEMM with right operand transposed
# --------------------------------------------------------------
@triton.jit
def _bmm_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B.T for many independent batches.
    A : (BATCH, M, K)          (row‑major)
    B : (BATCH, N, K)          (row‑major, will be accessed as K×N)
    C : (BATCH, M, N)
    """
    pid = tl.program_id(0)

    # tile decomposition
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    batch = pid // (num_pid_m * num_pid_n)
    tile = pid % (num_pid_m * num_pid_n)
    pid_m = tile // num_pid_n
    pid_n = tile % num_pid_n

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers for A (M×K) and B (K×N) – note the transpose on B
    a_ptrs = a_ptr + batch * stride_a_batch \
                   + offs_m[:, None] * stride_a_m \
                   + offs_k[None, :] * stride_a_k

    # B is stored as (batch, N, K); we read it as (K, N)
    b_ptrs = b_ptr + batch * stride_b_batch \
                   + offs_k[:, None] * stride_b_k \
                   + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # promote to fp32 for better accumulation stability
        a = tl.cast(a, tl.float32)
        b = tl.cast(b, tl.float32)
        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # store result (cast back to fp16)
    c_ptrs = c_ptr + batch * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n

    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _triton_bmm_transpose(A: torch.Tensor,
                          B: torch.Tensor,
                          block_m: int = 128,
                          block_n: int = 128,
                          block_k: int = 32) -> torch.Tensor:
    """
    Wrapper around the Triton kernel above.
    A, B : (batch, N, N)  fp16 contiguous
    Returns C : (batch, N, N)  fp16
    """
    batch, M, K = A.shape
    _, N, _ = B.shape
    assert M == K == N, "A and B must be square matrices"

    C = torch.empty_like(A)

    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_n, stride_b_k = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    num_pid_m = (M + block_m - 1) // block_m
    num_pid_n = (N + block_n - 1) // block_n
    total_grid = batch * num_pid_m * num_pid_n

    grid = (total_grid,)

    _bmm_transpose_kernel[grid](
        A,
        B,
        C,
        batch,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return C


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (float32/float16/bool) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim` (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask + gating (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (A @ Bᵀ) using Triton.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B*H, N, N) for the batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Triton batched GEMM with transpose on the right operand
    out_flat = _triton_bmm_transpose(left_flat, right_flat,
                                     block_m=128, block_n=128, block_k=32)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
