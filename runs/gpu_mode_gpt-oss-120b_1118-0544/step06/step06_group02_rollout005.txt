"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) using the provided
   weight/bias (FP32 → keep in FP32 for numerical stability).
2. Cast the normalized tensor to FP16 for the compute‑heavy part.
3. Four linear projections (left/right + their gates) are performed with
   FP16 weights via `torch.nn.functional.linear`.
4. Fuse element‑wise `proj * gate * mask` (when a mask is present) with a
   small Triton kernel. This saves two extra memory passes.
5. Core O(N³·H) reduction:
   - Reshape ``left``/``right`` to (B*H, N, N) and call ``torch.bmm``.
   - This is equivalent to the ``einsum('... i k d, ... j k d -> ... i j d')``.
6. Apply a second LayerNorm on the hidden dimension, multiply by the
   output gate and finally linearly project back to the original channel
   dimension.
7. Cast the result back to the input dtype (float32).

Only the mask‑gate fusion is implemented in Triton; everything else
leverages highly‑optimized PyTorch kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) on a flattened view
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (total_elems,)
    gate_ptr,          # fp16 * (total_elems,)
    mask_ptr,          # fp16 * (mask_elems,)  where mask_elems = total_elems // hidden_dim
    out_ptr,           # fp16 * (total_elems,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # For each hidden‑dim element we reuse the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse ``proj * gate * mask`` using a Triton kernel.

    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous()
    assert gate.is_contiguous()
    assert mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],          # hidden_dim (last dimension)
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
        - mask        : torch.Tensor of shape [B, N, N] (bool/int) or None
        - weights     : dict of model parameters (float32)
        - config      : dict with keys:
            * "dim" (int)        – input channel dimension C
            * "hidden_dim" (int) – hidden dimension H
            * "nomask" (bool)    – if True, ignore the mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # -------------------------- unpack ------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = x.device
    dtype_in = x.dtype

    # ----------------------- input LayerNorm -----------------------------
    # Use PyTorch's fast implementation (FP32)
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to FP16 for the heavy compute
    x = x.to(torch.float16)

    # -------------------------- cast weights -----------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ---------------------- linear projections ---------------------------
    # (B, N, N, hidden_dim)
    left_proj  = torch.nn.functional.linear(x, left_proj_w)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------ gates -------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()  # (B,N,N,hidden_dim)

    # ------------------- mask & gate fusion (Triton) --------------------
    if mask is not None and not nomask:
        # Convert mask to fp16 (0/1) and ensure contiguity
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------- core O(N³·H) reduction via batched GEMM -------
    # reshape to (B*H, N, N)
    B, N, _, _ = left.shape
    left_reshaped  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    out_reshaped = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))

    # reshape back to (B, N, N, hidden_dim)
    out = out_reshaped.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------ cast back to input dtype -----------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
