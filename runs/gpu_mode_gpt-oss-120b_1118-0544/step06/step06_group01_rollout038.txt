def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # layer norm
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # cast to fp16 for compute
    x = x.to(torch.float16)

    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # apply gates and mask (if any)
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)  # broadcast
        left = left * left_gate * mask
        right = right * right_gate * mask
    else:
        left = left * left_gate
        right = right * right_gate

    # reshape for batched GEMM
    B, N, _, _ = left.shape
    # left: (B, N, N, hidden_dim) -> (B*hidden_dim, N, N)
    left_mat = left.permute(0,3,1,2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0,3,1,2).contiguous().view(B * hidden_dim, N, N)
    # compute right transpose: shape (batch, K, N)
    right_mat_t = right_mat.transpose(1,2).contiguous()
    # allocate output
    out_mat = torch.empty_like(left_mat)  # same dtype fp16

    # launch Triton kernel
    M = N
    Ndim = N
    K = N
    batch = left_mat.shape[0]

    # strides in elements (since Triton uses byte strides? Actually it expects strides in elements, not bytes.)
    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    stride_b_batch = right_mat_t.stride(0)
    stride_b_k = right_mat_t.stride(1)
    stride_b_n = right_mat_t.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_m = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch, (M + BLOCK_M - 1) // BLOCK_M, (Ndim + BLOCK_N - 1) // BLOCK_N)

    batched_gemm_fp16[grid](
        left_mat,
        right_mat_t,
        out_mat,
        M,
        Ndim,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the reference PyTorch code while moving the
heavy N×N matrix multiplication to a custom Triton kernel.  The steps are:

1. Layer‑norm the input (float32 for stability) and cast to fp16.
2. Linear projections (left/right) and gate projections are performed with
   fp16 weights.  Gates are applied element‑wise and optionally masked.
3. The core contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched GEMM (left @ rightᵀ) over the sequence dimension.
   It is executed by `batched_gemm_fp16`, a tiled Triton kernel that
   processes all (batch × hidden) matrices in one launch.
4. An output LayerNorm, out‑gate, and final linear projection return the
   result to the original feature dimension and dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_fp16(
    a_ptr,                # left  [batch, M, K]  (fp16)
    b_ptr,                # right [batch, K, N] (fp16), already transposed
    c_ptr,                # out   [batch, M, N] (fp16)
    M, N, K,              # matrix sizes
    stride_a_batch, stride_a_m, stride_a_k,   # strides of a in elements
    stride_b_batch, stride_b_k, stride_b_n,   # strides of b in elements
    stride_c_batch, stride_c_m, stride_c_n,   # strides of c in elements
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ B   (A: [batch,M,K], B: [batch,K,N]).
    All tensors are in fp16; accumulation uses fp32.
    """
    pid_z = tl.program_id(0)   # batch index
    pid_m = tl.program_id(1)   # tile row
    pid_n = tl.program_id(2)   # tile col

    # Offsets inside the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current batch
    a_batch_ptr = a_ptr + pid_z * stride_a_batch
    b_batch_ptr = b_ptr + pid_z * stride_b_batch
    c_batch_ptr = c_ptr + pid_z * stride_c_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k_tile * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load tiles of A and B with appropriate masks
        a = tl.load(
            a_batch_ptr + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_batch_ptr + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate block matrix product
        acc += tl.dot(a, b)

    # Write the result back (cast to fp16)
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (torch.Tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability) and cast to fp16
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections and gates (all in fp16)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(torch.float16))
    )

    # -----------------------------------------------------------------
    # 3. Apply gates and optional mask
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # broadcast mask over the hidden dimension
        mask = mask.unsqueeze(-1).to(x.dtype)
        left = left * left_gate * mask
        right = right * right_gate * mask
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 4. Core contraction: batched GEMM left @ rightᵀ
    #    left/right: (B, N, N, H) -> (B*H, N, N)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    # bring hidden dim before the two sequence dims
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # transpose right for A @ Bᵀ
    right_mat_t = right_mat.transpose(1, 2).contiguous()
    out_mat = torch.empty_like(left_mat)  # (B*H, N, N), fp16

    # stride information (in elements, not bytes)
    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    stride_b_batch = right_mat_t.stride(0)
    stride_b_k = right_mat_t.stride(1)
    stride_b_n = right_mat_t.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_m = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        left_mat.shape[0],                                 # batch dimension
        (N + BLOCK_M - 1) // BLOCK_M,                      # rows
        (N + BLOCK_N - 1) // BLOCK_N,                      # cols
    )

    batched_gemm_fp16[grid](
        left_mat,
        right_mat_t,
        out_mat,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 5. Reshape back to (B, N, N, H)
    # -----------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast back to the original dtype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
