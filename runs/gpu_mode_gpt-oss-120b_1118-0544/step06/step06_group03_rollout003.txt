"""
TriMul (outgoing) forward – Triton‑accelerated gate & mask fusion.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (float32 for stability).
2. Cast to fp16 for the compute‑intensive part.
3. Linear projections → left/right (B,N,N,H) and three gate projections.
4. Fuse projection, gate and optional mask in a single Triton kernel:
       out = proj * gate * mask    (mask broadcast over H)
5. Core reduction (batched GEMM):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented by reshaping to (B*H, N, N) and using torch.bmm,
   which harnesses cuBLAS/Tensor‑cores.
6. Apply output LayerNorm over the hidden dimension, multiply by out‑gate,
   and a final linear projection back to the original dimension.
7. Cast the result back to the input dtype (usually float32).

Only step 4 is Triton‑based; the remaining steps use highly‑optimized
PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,      # fp16 * (B*N*N*H,)
    gate_ptr,      # fp16 * (B*N*N*H,)
    mask_ptr,      # fp16 * (B*N*N,)
    out_ptr,       # fp16 * (B*N*N*H,)
    total_elems,   # int: B*N*N*H
    hidden_dim,    # int: H
    BLOCK_SIZE: tl.constexpr,
):
    """
    Compute out = proj * gate * mask   (mask broadcast over hidden_dim).
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # index of the mask element (each mask value is reused H times)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 8192) -> torch.Tensor:
    """
    Fuse projection, gate and mask using a Triton kernel.
    All tensors must be fp16 and contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)  (binary 0/1, broadcast over H)
    Returns proj * gate * mask with shape (B,N,N,H).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (bool/float)
                            (may be None)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"          – input channel dimension C
                  * "hidden_dim"  – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim` (float32 for numerical stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy part
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast linear weights to FP16 (biases are None)
    # --------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B,N,N,H)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, w_left_proj)   # (B,N,N,H)
    right_proj = F.linear(x, w_right_proj)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid (still FP16)
    # --------------------------------------------------------------
    left_gate  = F.linear(x, w_left_gate).sigmoid()
    right_gate = F.linear(x, w_right_gate).sigmoid()
    out_gate   = F.linear(x, w_out_gate).sigmoid()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 6. Apply mask + gating (fused via Triton if mask is present)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is fp16, contiguous, with values 0 or 1
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction – batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T  -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
    )
    out = out * out_gate
    out = F.linear(out, w_to_out)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 9. Cast back to the original dtype
    # --------------------------------------------------------------
    out = out.to(dtype_in)

    return out
