import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _proj_gate_mask_kernel(
    # pointers
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # compute index into mask (since mask not permuted across hidden)
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

def _apply_proj_gate_mask(left_proj, right_proj, left_gate, right_gate, mask, BLOCK_SIZE=1024):
    B,N,_ ,H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, BLOCK_SIZE),)

    _proj_gate_mask_kernel[grid](
        left_proj, right_proj,
        left_gate, right_gate,
        mask,
        left_out, right_out,
        total,
        H,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    return left_out, right_out

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm input
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Cast to fp16
    x = x.to(torch.float16)

    # cast weights
    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out = weights["to_out.weight"].to(torch.float16)

    # Linear projections
    left_proj = F.linear(x, w_left_proj)   # shape B,N,N,hidden_dim
    right_proj = F.linear(x, w_right_proj)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x, w_out_gate))

    # Apply mask and gating
    if mask is not None and not nomask:
        # mask is torch.bool? maybe 0/1 float. Ensure fp16 shape.
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(left_proj, right_proj, left_gate, right_gate, mask_fp16)
    else:
        # No mask or ignore
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask present but ignore; still broadcast
            left = left * mask.to(torch.float16).unsqueeze(-1)
            right = right * mask.to(torch.float16).unsqueeze(-1)

    # Core reduction:
    B, N, _, H = left.shape   # Should be hidden_dim
    # reshape for batched matmul
    # Move H to second dimension
    left_b = left.transpose(1,3).reshape(B*H, N, N)
    right_b = right.transpose(1,3).reshape(B*H, N, N)
    out_b = torch.bmm(left_b, right_b.transpose(1,2))
    out = out_b.view(B, H, N, N).permute(0,2,3,1)   # shape B,N,N,H

    # Output layernorm
    out = F.layer_norm(out, (H,), weight=w_to_out_norm, bias=b_to_out_norm)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear
    out = F.linear(out, w_to_out)

    # Cast back to input dtype
    out = out.to(input_tensor.dtype)

    return out

"""
TriMul (outgoing) forward kernel – AlphaFold3.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (`dim`).
2. Cast to FP16 for the compute‑heavy path.
3. Linear projections (no bias) → left_proj, right_proj.
   Gates (sigmoid of linear projections) → left_gate, right_gate, out_gate.
4. Fuse mask, projection and gating for *both* sides in a single Triton
   kernel (only executed when a mask is present and `nomask=False`).
5. Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM: for each hidden slice `h` compute
        out_h = left_h @ right_hᵀ
   Implemented with a single `torch.bmm` on a view of shape (B*H, N, N).
6. Layer‑norm over the hidden dimension, element‑wise multiply by `out_gate`,
   final linear projection back to `dim`.
7. Cast back to the original dtype (usually float32).

The Triton kernel only handles the mask‑gate‑fusion; the bulk matrix
multiplication uses cuBLAS via `torch.bmm`, which is much faster than a
pure‑Triton implementation for large `N`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    # pointers to fp16 tensors
    left_proj_ptr, right_proj_ptr,   # (B,N,N,H)
    left_gate_ptr, right_gate_ptr,   # (B,N,N,H)
    mask_ptr,                        # (B,N,N)
    left_out_ptr, right_out_ptr,     # (B,N,N,H)
    total_elems,                     # int64 = B*N*N*H
    hidden_dim,                      # int64 = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # guard against OOB
    mask = offs < total_elems

    # each hidden element shares the same mask index
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gate and mask for both left and right tensors.
    All tensors must be fp16, contiguous and have shapes:
        (B, N, N, H) for the first four arguments,
        (B, N, N)    for mask.
    Returns:
        left, right : (B, N, N, H) = proj * gate * mask
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C]  (float32 or float16)
            - mask         : torch.Tensor [B, N, N] (0/1) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2) FP16 compute path
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # Cast all linear weights to fp16 (biases are None)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 3) Linear projections (FP16)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 4) Apply mask + gating
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16, contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask (or ignored) → simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask present but we are told to ignore it → still broadcast
            mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask_fp16
            right = right * mask_fp16

    # --------------------------------------------------------------
    # 5) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM over the hidden dimension.
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Reshape to (B*H, N, N) without extra copies
    left_flat  = left.transpose(1, 3).reshape(B * H, N, N)   # (B*H, N, N)
    right_flat = right.transpose(1, 3).reshape(B * H, N, N)  # (B*H, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 7) Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
