    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Input LayerNorm using float32
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16
    x = x.to(torch.float16)

    # Convert weight tensors to fp16
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(torch.float16)

    # Combine left_proj and left_gate weights
    left_comb_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0) # shape (2*H, dim)
    right_comb_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0) # shape (2*H, dim)

    # Compute combined linear for left and right
    # F.linear expects weight shape (out_features, in_features)
    left_comb = torch.nn.functional.linear(x, left_comb_weight) # (B,N,N,2H)
    right_comb = torch.nn.functional.linear(x, right_comb_weight) # (B,N,N,2H)

    # Split
    left_proj, left_gate = left_comb[..., :hidden_dim], left_comb[..., hidden_dim:]
    right_proj, right_gate = right_comb[..., :hidden_dim], right_comb[..., hidden_dim:]

    # Apply sigmoid to gates
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()

    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # out_gate
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid() # (B,N,N,H)

    # Core reduction: For each hidden dim, compute left @ right^T
    B, N, _, H = left.shape

    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

"""
TriMul (outgoing) forward pass – Triton‑accelerated gate‑mask fusion and
projection‑gate weight sharing.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim` (float32 for stability).
2. Cast the normalized tensor to FP16.
3. Fuse the linear projection and its gate into a single matmul by concatenating
   the two weight matrices (`proj` and `gate`).  This halves the number of
   large GEMM calls for the left and right branches.
4. Apply sigmoid to the gate part and fuse the element‑wise product
   `proj * gate * mask` with a tiny Triton kernel (mask is broadcast over the
   hidden dimension).  When `nomask=True` the mask is treated as all‑ones.
5. Compute the core TriMul reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication (`torch.bmm`) over the sequence
   dimension for each hidden channel.
6. Output LayerNorm over the hidden dimension, multiply by the `out_gate`,
   and project back to the original dimension with a final linear layer.
7. Cast the result back to the input dtype (usually float32).

All heavy linear layers use FP16 and cuBLAS; only the gate‑mask fusion is
implemented in Triton, satisfying the requirement to include a custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise proj * gate * mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,          # fp16 * (total_elems,)
    gate_ptr,          # fp16 * (total_elems,)
    mask_ptr,          # fp16 * (mask_elems,)  where mask_elems = total_elems // hidden_dim
    out_ptr,           # fp16 * (total_elems,)
    total_elems,       # i32 scalar
    hidden_dim,        # i32 scalar
    BLOCK_SIZE: tl.constexpr,
):
    """
    out = proj * gate * mask      (mask broadcast over hidden_dim)
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard
    mask = offs < total_elems

    # Index of the corresponding mask element (broadcasted over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # default = 1.0 (no mask)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 16384) -> torch.Tensor:
    """
    Fuse projection, gate and (optional) mask using a Triton kernel.
    All tensors must be contiguous fp16.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)   (binary 0/1, broadcast over H)
    Returns: proj * gate * mask   (shape (B,N,N,H)).
    """
    B, N, _, H = proj.shape
    total = proj.numel()                               # B*N*N*H
    mask_elems = mask.numel()                          # B*N*N
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all weights to FP16 and put them on the right device
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------
    # 4. Fuse projection + gate for left and right branches
    # --------------------------------------------------------------
    # Concatenate weights: [proj; gate]  -> shape (2*H, C)
    left_comb_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, C)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0) # (2*H, C)

    # One linear per side
    left_comb  = F.linear(x, left_comb_w)   # (B, N, N, 2*H)
    right_comb = F.linear(x, right_comb_w)  # (B, N, N, 2*H)

    # Split back
    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # --------------------------------------------------------------
    # 5. Apply mask + gating (fused via Triton when mask is present)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16, contiguous, and broadcastable
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6. Compute out_gate (separate linear + sigmoid)
    # --------------------------------------------------------------
    out_gate = F.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # --------------------------------------------------------------
    # 7. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM (B*H, N, N) @ (B*H, N, N)^T
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # Reshape to (B*H, N, N) for torch.bmm
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
