def proj_gate_mask_kernel( left_proj_ptr, left_gate_ptr, mask_ptr, left_out_ptr,
                           right_proj_ptr, right_gate_ptr, right_out_ptr,
                           total_elems, hidden_dim, BLOCK_SIZE: tl.constexpr ):

    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # compute mask index: each group of hidden_dim corresponds to same mask value:
    mask_idx = offs // hidden_dim

    left_proj = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', False)

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])
    # Cast to fp16
    x = x.to(torch.float16)

    # combine left and right projection weights
    left_proj_w = weights['left_proj.weight']
    right_proj_w = weights['right_proj.weight']
    proj_weights = torch.cat([left_proj_w, right_proj_w], dim=0).to(torch.float16)

    # compute proj concatenated
    proj = F.linear(x, proj_weights)  # shape (B,N,N,2*hidden_dim)

    left_proj, right_proj = torch.split(proj, hidden_dim, dim=-1)

    # combine gating weights
    left_gate_w = weights['left_gate.weight']
    right_gate_w = weights['right_gate.weight']
    gate_weights = torch.cat([left_gate_w, right_gate_w], dim=0).to(torch.float16)

    gates = F.linear(x, gate_weights)  # shape (B,N,N,2*hidden_dim)
    left_gate, right_gate = torch.split(gates, hidden_dim, dim=-1)
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()

    # out gate
    out_gate_w = weights['out_gate.weight'].to(torch.float16)
    out_gate = F.linear(x, out_gate_w).sigmoid() # shape (B,N,N,hidden_dim)

    # apply mask and gating using Triton if needed
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(left_proj, right_proj, left_gate, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask present but ignore? maybe treat as 1
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # core reduction
    B, N, _, H = left.shape
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)  # (B*H,N,N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1,2))
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # final norm and gating
    to_out_norm_w = weights['to_out_norm.weight'].to(torch.float16)
    to_out_norm_b = weights['to_out_norm.bias'].to(torch.float16)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    to_out_w = weights['to_out.weight'].to(torch.float16)
    out = F.linear(out, to_out_w)  # shape (B,N,N,dim)

    # cast back
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
```

"""
TriMul (outgoing) forward pass accelerated with Triton.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for compute‑heavy ops.
3. Fuse the two linear projections (`left_proj` & `right_proj`) into a
   single GEMM by concatenating their weight matrices.
   Do the same for the two gate projections (`left_gate` & `right_gate`).
4. Apply sigmoid to the gates.
5. If a mask is present (and `nomask` is False) fuse mask, projection
   and gate via a Triton kernel that computes
        left  = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask
   (mask is broadcast over the hidden dimension).
   Otherwise perform the element‑wise multiplication in PyTorch.
6. Core reduction (the “TriMul”):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM: reshape to (B*H, N, N) and call
   `torch.bmm`. This leverages cuBLAS which is heavily optimized.
7. Apply LayerNorm over the hidden dimension, multiply by the output gate,
   and linearly project back to the original dimension.
8. Cast the result back to the input dtype (usually float32).

The Triton kernel is only used for the mask‑gate‑projection fusion,
which is the main source of memory traffic. All heavy arithmetic stays
in FP16 for speed while the final output is returned in the original
precision.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fused mask, gate and projection for both left and right
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B*N*N*H,)
    left_gate_ptr,   # fp16 * (B*N*N*H,)
    mask_ptr,        # fp16 * (B*N*N,)
    left_out_ptr,    # fp16 * (B*N*N*H,)

    right_proj_ptr,  # fp16 * (B*N*N*H,)
    right_gate_ptr,  # fp16 * (B*N*N*H,)
    right_out_ptr,   # fp16 * (B*N*N*H,)

    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # mask is broadcast over hidden dimension => same mask for each group of H entries
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(left_out_ptr + offs,  lp * lg * m, mask=mask)
    tl.store(right_out_ptr + offs, rp * rg * m, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse mask, gate and projection for both left and right tensors via Triton.
    All inputs must be contiguous FP16 tensors.
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        left_gate,
        mask,
        left_out,
        right_proj,
        right_gate,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C]  (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None (0/1)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim) – still in input dtype

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Concatenate projection and gate weights (FP16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    proj_w = torch.cat([left_proj_w, right_proj_w], dim=0).to(torch.float16)  # (2*H, dim)

    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    gate_w = torch.cat([left_gate_w, right_gate_w], dim=0).to(torch.float16)  # (2*H, dim)

    out_gate_w = weights["out_gate.weight"].to(torch.float16)               # (H, dim)

    # ------------------------------------------------------------------
    # 4) Linear projections (single fused GEMM for left/right)
    # ------------------------------------------------------------------
    proj = F.linear(x, proj_w)                     # (B, N, N, 2*H)
    left_proj, right_proj = torch.split(proj, hidden_dim, dim=-1)

    # ------------------------------------------------------------------
    # 5) Linear gates (single fused GEMM)
    # ------------------------------------------------------------------
    gates = F.linear(x, gate_w)                    # (B, N, N, 2*H)
    left_gate, right_gate = torch.split(gates, hidden_dim, dim=-1)
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()

    # ------------------------------------------------------------------
    # 6) Output gate (separate)
    # ------------------------------------------------------------------
    out_gate = F.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Apply mask & gating (fused Triton kernel if mask is used)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure contiguous FP16 mask
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask (or mask ignored) – simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # Mask present but we are instructed to ignore it -> treat as all ones
            pass

    # ------------------------------------------------------------------
    # 8) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM over the hidden dimension.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                         # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
