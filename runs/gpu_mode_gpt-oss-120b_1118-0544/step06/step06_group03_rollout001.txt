"""
TriMul (outgoing) forward pass – Triton‑accelerated projection‑gate‑mask fusion.

Algorithm
---------
1. Layer‑norm `input` over the channel dimension `dim` (float32).
2. Cast to FP16 for the compute‑intensive part.
3. Linear projections → `left_proj`, `right_proj` (shape B,N,N,H).
4. Gate projections → `left_gate`, `right_gate`, `out_gate` (sigmoid).
5. Fuse projection, gating and optional mask in a single Triton kernel:
       out = proj * gate * mask   (mask broadcast over H)
   If `mask` is omitted, the kernel reduces to a simple element‑wise product.
6. Core reduction (batched GEMM):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented via `torch.bmm` on a reshaped view (B*H, N, N).
7. Output LayerNorm over hidden dimension H, multiply by `out_gate`,
   and final linear projection back to original dimension `dim`.
8. Return the result in the original input dtype (usually float32).

The only mandatory Triton component is the fusion kernel in step 5;
all other operations use highly‑optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _proj_gate_mask_fuse_kernel(
    proj_ptr,      # fp16 * (B*N*N*H,)
    gate_ptr,      # fp16 * (B*N*N*H,)
    mask_ptr,      # fp16 * (B*N*N,)   (may be dummy if no mask)
    out_ptr,       # fp16 * (B*N*N*H,)
    total_elems,   # int: B*N*N*H
    hidden_dim,    # int: H
    HAS_MASK: tl.constexpr,   # 1 if mask is present, 0 otherwise
    BLOCK_SIZE: tl.constexpr,
):
    """
    Compute out = proj * gate * mask (mask broadcast over hidden_dim).
    If HAS_MASK == 0 the mask load is skipped and treated as 1.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # Load projection and gate
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Compute the corresponding mask index (broadcasted over hidden_dim)
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        out = proj * gate * m
    else:
        out = proj * gate

    tl.store(out_ptr + offs, out, mask=mask)

def _apply_proj_gate_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor | None,
    block_size: int = 2048,
) -> torch.Tensor:
    """
    Fuse projection, gate and optional mask using a Triton kernel.
    All tensors must be fp16, contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N) or None (binary 0/1, broadcast over H)
    Returns proj * gate * mask (shape (B,N,N,H)).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    has_mask = 1 if (mask is not None) else 0
    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask if mask is not None else torch.empty(1, device=proj.device, dtype=proj.dtype),
        out,
        total,
        H,
        HAS_MASK=has_mask,
        BLOCK_SIZE=block_size,
    )
    return out

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (float32/float16/bool) or None
            - weights      : dict of model parameters (float32 tensors)
            - config       : dict with keys:
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. LayerNorm over `dim` (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Fuse projection, gate and optional mask (Triton)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16, contiguous, binary (0/1)
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_proj_gate_mask(left_proj, left_gate, mask_fp16)
        right = _apply_proj_gate_mask(right_proj, right_gate, mask_fp16)
    else:
        # No mask → simple elementwise multiplication (still fuse via triton for consistency)
        left = _apply_proj_gate_mask(left_proj, left_gate, None)
        right = _apply_proj_gate_mask(right_proj, right_gate, None)

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
