@triton.jit
def _batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_stride_left, batch_stride_right, batch_stride_out,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start indices
    block_m = pid_m * BLOCK_M
    block_n = pid_n * BLOCK_N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute offset for batch
    left_batch_ptr = left_ptr + pid_batch * batch_stride_left
    right_batch_ptr = right_ptr + pid_batch * batch_stride_right
    out_batch_ptr = out_ptr + pid_batch * batch_stride_out

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        # mask for k
        mask_k = offs_k < K

        # offsets m and n
        offs_m = block_m + tl.arange(0, BLOCK_M)
        offs_n = block_n + tl.arange(0, BLOCK_N)

        mask_m = offs_m < M
        mask_n = offs_n < N

        # compute pointer addresses
        # left shape: (M,K)
        a_ptrs = left_batch_ptr + (offs_m[:, None] * K + offs_k[None, :])
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right shape: (K,N)
        b_ptrs = right_batch_ptr + (offs_k[:, None] * N + offs_n[None, :])
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # writeout
    out_ptrs = out_batch_ptr + (offs_m[:, None] * N + offs_n[None, :])
    mask_out = mask_m[:, None] & mask_n[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)

            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # after kernel left/right have been modified in-place

        right = right * right_gate

    B, N, _, H = left.shape  # H = hidden_dim

    # transpose right for multiplication
    right_t = right.permute(0, 2, 1, 3).contiguous()

    right_mat = right_t.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    left_mat = left_mat.contiguous()
    right_mat = right_mat.contiguous()

    # allocate output
    out_mat = torch.empty_like(left_mat)

    # compute batch stride (size of each matrix)
    batch_stride = N * N  # number of elements per batch

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

        left_mat,
        right_mat,
        out_mat,
        batch_stride,
        batch_stride,
        batch_stride,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape out_mat to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 4. Output LayerNorm
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    # apply out_gate

    # final linear
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)

    # cast back to original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. LayerNorm on the input (float32) → fp16 for compute.
2. Linear projections (left, right) and gating projections (left_gate, right_gate,
   out_gate) without bias.
3. Fuse mask (if present) and gating for left/right with a small Triton kernel.
4. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which equals a batched GEMM: left @ rightᵀ for each hidden channel.
   Implemented with a dedicated Triton batched‑GEMM kernel that tiles the
   matrices (BLOCK_M×BLOCK_N×BLOCK_K) and accumulates in float32.
5. Reshape result back to (B,N,N,H), apply output LayerNorm, multiply by out_gate,
   and a final linear projection to the original dimension.
6. Cast back to the input dtype.

Only the mask‑gate fusion and the heavy O(N³·H) contraction are executed in
Triton; all other operations use highly‑optimized torch primitives.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: fuse mask and per‑channel gates for left/right tensors
# --------------------------------------------------------------------------- #
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load tensors
    left      = tl.load(left_ptr + offs,      mask=active, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr + offs,mask=active, other=0.0)

    if HAS_MASK:
        # mask is broadcast over the hidden dimension
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)

# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the core contraction (left @ rightᵀ)
# --------------------------------------------------------------------------- #
@triton.jit
def _batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_stride_left, batch_stride_right, batch_stride_out,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # batch index (B*H)
    pid_m = tl.program_id(1)       # row block
    pid_n = tl.program_id(2)       # col block

    # Block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the current batch
    left_batch_ptr  = left_ptr  + pid_batch * batch_stride_left
    right_batch_ptr = right_ptr + pid_batch * batch_stride_right
    out_batch_ptr   = out_ptr   + pid_batch * batch_stride_out

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = left_batch_ptr + (offs_m[:, None] * K + offs_k[None, :])
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (K x N)
        b_ptrs = right_batch_ptr + (offs_k[:, None] * N + offs_n[None, :])
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Write back the result
    out_ptrs = out_batch_ptr + (offs_m[:, None] * N + offs_n[None, :])
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------------------- #
# Entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm (float32) → fp16
    # ------------------------------------------------------------------- #
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 2. Linear projections (bias‑less) and gating
    # ------------------------------------------------------------------- #
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)                     # (B,N,N,H)
    right = F.linear(x, right_proj_w)                   # (B,N,N,H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))  # (B,N,N,H)

    # ------------------------------------------------------------------- #
    # 3. Fuse mask and per‑channel gates (Triton)
    # ------------------------------------------------------------------- #
    if mask is not None and not nomask:
        # Prepare flattened views
        total = left.numel()
        BLOCK = 1024
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_fp        = mask.to(device).to(torch.float16).contiguous().view(-1)

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_fp,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------- #
    # 4. Core O(N³·H) contraction using a Triton batched GEMM
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape  # H == hidden_dim

    # prepare right transposed (k ↔ j)
    right_t = right.permute(0, 2, 1, 3).contiguous()   # (B,N,N,H)

    # reshape to (B*H, N, N) – each row‑major matrix is contiguous
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_t.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # output buffer
    out_mat = torch.empty_like(left_mat)

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    batch_dim = B * H
    grid = (
        batch_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batch_stride = N * N  # elements per (batch, channel) matrix

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch_stride,    # stride between successive (B*H) matrices in left
        batch_stride,    # same for right
        batch_stride,    # same for output
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 5. Output LayerNorm, apply out_gate, final linear projection
    # ------------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # gating (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) fp16

    # ------------------------------------------------------------------- #
    # 6. Cast back to original dtype
    # ------------------------------------------------------------------- #
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
