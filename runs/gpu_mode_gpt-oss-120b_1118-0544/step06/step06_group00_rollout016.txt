"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input (float32 for stability).
2. Cast to fp16 and apply three linear projections:
   * left  = x @ W_left
   * right = x @ W_right
   * left_gate  = sigmoid(x @ W_left_gate)
   * right_gate = sigmoid(x @ W_right_gate)
   * out_gate   = sigmoid(x @ W_out_gate)
3. Fuse masks (if present) and element‑wise gates.
4. Core O(N³·H) contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched mat‑mul (L @ Rᵀ) in a custom Triton kernel.
5. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate and a final linear projection back to the original channel size.
6. Cast back to the input dtype.

The kernel `batched_matmul_kernel` performs the batched matrix multiplication
for all (batch × hidden) pairs in a single launch, using shared‑memory tiling.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    batch, M, N, K,                     # dimensions
    stride_ah, stride_am, stride_ak,    # strides for a (batch, M, K)
    stride_bh, stride_bk, stride_bn,    # strides for b (batch, N, K) – we read as (batch, K, N)
    stride_ch, stride_cm, stride_cn,    # strides for c (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul C = A @ Bᵀ  where
       A, B : (batch, M, K) and (batch, N, K) respectively.
       The kernel loops over K in blocks and accumulates in FP32.
    """
    pid_m = tl.program_id(0)      # block index over M
    pid_n = tl.program_id(1)      # block index over N
    pid_b = tl.program_id(2)      # batch index (B*H)

    # compute start offsets of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑boundary rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to the beginning of the batch slice
    a_batch_ptr = a_ptr + pid_b * stride_ah
    b_batch_ptr = b_ptr + pid_b * stride_bh
    c_batch_ptr = c_ptr + pid_b * stride_ch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # pointers for the current K‑tile
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak
        b_ptrs = b_batch_ptr + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # load tiles with appropriate masks
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0)

        # accumulate
        acc += tl.dot(a, b)

    # write back
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             c,
             mask=(mask_m[:, None] & mask_n[None, :]))


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------ #
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------ #
    # 2. Cast to fp16 for heavy computation
    # ------------------------------------------------------------------ #
    x = x.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 3. Linear projections & gates (no bias)
    # ------------------------------------------------------------------ #
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)                     # (B,N,N,H)
    right = F.linear(x, right_proj_w)                   # (B,N,N,H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,H)

    # ------------------------------------------------------------------ #
    # 4. Fuse mask (if present) and element‑wise gates
    # ------------------------------------------------------------------ #
    if mask is not None and not nomask:
        # mask broadcast over hidden dimension
        mask_ = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * left_gate * mask_
        right = right * right_gate * mask_
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Core contraction via Triton batched matrix multiplication
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------ #
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_mat = torch.empty_like(left_mat)

    # strides (in elements)
    stride_ah, stride_am, stride_ak = left_mat.stride()
    stride_bh, stride_bn, stride_bk = right_mat.stride()
    stride_ch, stride_cm, stride_cn = out_mat.stride()

    # block size choices (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            B * H)

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        B * H,
        N,
        N,
        N,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 6. Output LayerNorm, out‑gate, and final linear projection
    # ------------------------------------------------------------------ #
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (already fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------ #
    # 7. Cast back to original dtype
    # ------------------------------------------------------------------ #
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
