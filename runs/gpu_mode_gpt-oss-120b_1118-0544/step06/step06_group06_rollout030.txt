"""
TriMul (outgoing) forward pass with Triton fusion.

Algorithm
---------
1. Layer‑normalize the input tensor (float32 for stability).
2. Fuse the five linear projections (left‑proj, left‑gate, right‑proj,
   right‑gate, out‑gate) into a single matmul using a concatenated weight
   matrix (shape (5*H, C)).  Slice the result and apply sigmoid to the
   gate parts.
3. Fuse mask application and gate multiplication for the left/right
   tensors in a Triton kernel:
       left_out  = left  * left_gate  * mask
       right_out = right * right_gate * mask
4. Perform the core TriMul reduction:
       out[b,i,j,h] = Σ_k left_out[b,i,k,h] * right_out[b,j,k,h]
   Implemented as a batched GEMM (B*H, N, N) via torch.bmm.
5. Apply a second LayerNorm over the hidden dimension H.
6. Multiply by the output gate and project back to the original channel
   dimension C.
All operations are performed in fp16 where possible; the final result
is cast back to the input dtype (float32 by default).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # const * left   [B,N,N,H] flattened
    right_ptr,         # const * right  [B,N,N,H] flattened
    mask_ptr,          # const * mask   [B,N,N]   flattened
    left_gate_ptr,     # const * left_gate  [B,N,N,H] flattened
    right_gate_ptr,    # const * right_gate [B,N,N,H] flattened
    left_out_ptr,      # * left_out   [B,N,N,H] flattened
    right_out_ptr,     # * right_out  [B,N,N,H] flattened
    H: tl.constexpr,               # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask and gate application:
       left_out  = left  * left_gate  * mask
       right_out = right * right_gate * mask
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load tensors
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Mask repeats for each hidden dimension
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Write results
    tl.store(left_out_ptr + offs, l * lg * m, mask=mask)
    tl.store(right_out_ptr + offs, r * rg * m, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp32 for numeric stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # Cast to fp16 for the heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Fuse all linear projections (left/right + gates + out_gate)
    # ------------------------------------------------------------------
    # All weight tensors are stored as (out_features, in_features) = (H, C)
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, torch.float16)

    # Concatenate along the output dimension: (5*H, C)
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    )

    # Single linear map: (B,N,N,C) -> (B,N,N,5*H)
    combined = F.linear(x, combined_weight)  # fp16

    # ------------------------------------------------------------------
    # 3) Slice and apply sigmoid to gate portions
    # ------------------------------------------------------------------
    B, N, _, _ = combined.shape
    H = hidden_dim
    left      = combined[..., :H]                 # (B,N,N,H)
    left_gate = torch.sigmoid(combined[..., H:2*H])
    right     = combined[..., 2*H:3*H]
    right_gate = torch.sigmoid(combined[..., 3*H:4*H])
    out_gate  = torch.sigmoid(combined[..., 4*H:5*H])

    # ------------------------------------------------------------------
    # 4) Prepare mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_tensor = mask.to(torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 5) Fuse mask and gate multiplication via Triton
    # ------------------------------------------------------------------
    total_elements = left.numel()                     # B * N * N * H
    BLOCK = 1024

    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = (triton.cdiv(total_elements, BLOCK),)
    _mask_gate_fusion_kernel[grid](
        left, right,
        mask_tensor.view(-1),
        left_gate, right_gate,
        left_out, right_out,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction using batched GEMM
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_flat  = left_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # out_flat shape: (B*H, N, N) = left_flat @ right_flat^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(-2, -1))

    # reshape back to (B, N, N, H)
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final projection back to original channel dimension C
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device, torch.float16)
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
