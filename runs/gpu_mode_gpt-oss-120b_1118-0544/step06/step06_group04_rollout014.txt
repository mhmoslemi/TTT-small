"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input (float32) and cast to fp16.
2. Linear projections to the hidden dimension (no bias) and compute three gates
   (sigmoid of linear projections).  All linear ops are performed with torch.
3. Fuse element‑wise mask (if present) and the two input gates with a small
   Triton kernel `_gate_mask_kernel`.  This writes the gated tensors back in‑place.
4. Core O(N³·H) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a blocked batched GEMM in Triton (`_tri_mul_gemm_kernel`),
   processing the (B·H) independent matrix multiplications in a single launch.
5. Apply output LayerNorm over the hidden dimension, multiply by the output gate
   and a final linear projection back to the original channel dimension.
6. Cast the result back to the original input dtype.

The implementation uses fp16 for all heavy arithmetic (Tensor‑core friendly)
and only falls back to PyTorch for the LayerNorm / linear layers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left      = tl.load(left_ptr + offs,      mask=active, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr + offs,mask=active, other=0.0)

    if HAS_MASK:
        # mask is broadcast over hidden dimension
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs,  left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the TriMul contraction.
    Computes for each (b, h):
        out[b, i, j, h] = sum_k left[b, i, k, h] * right[b, j, k, h]
    """
    pid_m = tl.program_id(0)   # i‑block
    pid_n = tl.program_id(1)   # j‑block
    pid_bh = tl.program_id(2)  # batch * hidden

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # decode batch and hidden indices
    bh = pid_bh
    b = bh // H
    h = bh % H

    # base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # tile indices
    i = tl.arange(0, BLOCK_M)
    j = tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    i_idx = i0 + i
    j_idx = j0 + j

    mask_i = i_idx < N
    mask_j = j_idx < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k0 = kk * BLOCK_K
        k_idx = k0 + k
        mask_k = k_idx < N

        # left tile: (BLOCK_M, BLOCK_K)
        left_ptrs = (left_base
                     + (i_idx[:, None] * stride_li)
                     + (k_idx[None, :] * stride_lk))
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile: we need shape (BLOCK_K, BLOCK_N) i.e. (k, j)
        right_ptrs = (right_base
                      + (j_idx[None, :] * stride_ri)
                      + (k_idx[:, None] * stride_rk))
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # store the result
    out_ptrs = (out_base
                + (i_idx[:, None] * stride_oi)
                + (j_idx[None, :] * stride_oj))
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # Configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) and cast to fp16
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, torch.float32)
    norm_b = weights["norm.bias"].to(device, torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑less) + gates (sigmoid)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device, torch.float16)

    left  = F.linear(x, left_proj_w)          # (B,N,N,H)
    right = F.linear(x, right_proj_w)         # (B,N,N,H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3. Fuse mask & input gates (Triton kernel)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure contiguous layout
        mask_fp = mask.to(device, torch.float16).contiguous()
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        _gate_mask_kernel[grid](
            left.view(-1),
            right.view(-1),
            left_gate.view(-1),
            right_gate.view(-1),
            mask_fp.view(-1),
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # After the kernel left/right already contain the gated values
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with a Triton blocked GEMM.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # allocate output tensor
    out = torch.empty_like(left)

    # make tensors contiguous for reliable stride computation
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # strides (in elements)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # block sizes (tuned for B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑blocks
        triton.cdiv(N, BLOCK_N),   # j‑blocks
        B * H,                     # batch‑hidden blocks
    )

    _tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_ri, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------
    # 5. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate    # element‑wise output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device, torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------
    # 6. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
