"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate.

Algorithm
---------
1. Apply LayerNorm on the input tensor (B,N,N,C) using the provided
   weight/bias (float32 → fp16 for compute).
2. Perform four linear projections in fp16:
      left_proj, right_proj : (B,N,N,H)
      left_gate, right_gate, out_gate : (B,N,N,H)
3. Fuse projection, gate and optional mask with a tiny Triton kernel.
   This avoids two extra memory passes.
4. Core O(N³·H) reduction:
   - Reshape left/right to (B*H, N, N) and compute a batched GEMM
     (torch.bmm) which maps to cuBLAS/Tensor‑core kernels.
5. Second LayerNorm on the hidden dimension, multiply by the output gate,
   and linearly project back to the original channel dimension.
6. Cast the result back to the input dtype (usually float32).

Only the element‑wise mask‑gate‑fusion is implemented in Triton; the rest
leverages highly‑optimized PyTorch kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _gate_mask_kernel(
    proj_ptr,      # fp16 * (B*N*N*H,)
    gate_ptr,      # fp16 * (B*N*N*H,)
    mask_ptr,      # fp16 * (B*N*N,)
    out_ptr,       # fp16 * (B*N*N*H,)
    total_elems,   # i32 – total number of elements in proj/gate
    hidden_dim,    # i32 – H (for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # Index of the mask element (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 2048,
) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using a Triton kernel.

    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(
    data: Tuple[
        torch.Tensor,                      # input tensor [B, N, N, C]
        torch.Tensor,                      # mask tensor   [B, N, N] (or None)
        Dict[str, torch.Tensor],           # weights
        Dict[str, Any]                     # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys:
            * "dim" (int)          – input channel dimension C
            * "hidden_dim" (int)   – hidden dimension H
            * "nomask" (bool)      – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ----------------------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ----------------------------------------------------------------------
    # First LayerNorm (over channel dimension C)
    # ----------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for the heavy compute
    x = x.to(torch.float16)

    # ----------------------------------------------------------------------
    # Cast weights to fp16
    # ----------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ----------------------------------------------------------------------
    # Linear projections (B,N,N,H)
    # ----------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()  # (B,N,N,H)

    # ----------------------------------------------------------------------
    # Optional mask & gate fusion via Triton
    # ----------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is contiguous fp16 and flattened
        mask_fp16 = mask.to(torch.float16).contiguous()
        # Apply fused operation
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ----------------------------------------------------------------------
    # Core O(N³·H) reduction using batched GEMM (torch.bmm)
    # ----------------------------------------------------------------------
    B, N, _, _ = left.shape  # left.shape == (B,N,N,H)

    # Reshape to (B*H, N, N)
    left_reshaped  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Compute batch of matrix multiplications: (B*H, N, N) @ (B*H, N, N)^T
    out_reshaped = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))

    # Restore shape (B,N,N,H)
    out = out_reshaped.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------------
    # Second LayerNorm, output gate, and final projection
    # ----------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # Cast back to original dtype if necessary
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
