@triton.jit
def mask_gate_fusion_kernel(
    left_ptr, right_ptr,
    mask_ptr,
    left_gate_ptr, right_gate_ptr,
    hidden_dim: tl.constexpr,
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # mask index
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # store back in-place
    tl.store(left_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm: using fp16 weights
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float16)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float16)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    # Cast to fp16 (input may be fp32)
    x = x.to(torch.float16)

    # Cast linear weights to fp16
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # Projections
    left = F.linear(x, left_proj_w)   # shape B,N,N,hidden_dim
    right = F.linear(x, right_proj_w)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Prepare mask
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_tensor = mask.to(torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # Flatten all tensors for kernel
    total_elements = left.numel()  # B*N*N*hidden_dim
    BLOCK_SIZE = 1024
    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_flat = mask_tensor.view(-1).contiguous()

    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)
    mask_gate_fusion_kernel[grid](
        left,
        right,
        mask_flat,
        left_gate,
        right_gate,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # Now left and right contain the masked gated values
    # Reshape for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Multiply left and right transpose
    out_flat = torch.bmm(left_flat, right_flat.transpose(-2, -1))  # (B*H,N,N)

    # Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Second layer norm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Output gate multiplication
    out = out * out_gate

    # Final projection
    out = F.linear(out, to_out_w)

    # Cast back to original dtype (float32)
    out = out.to(orig_dtype)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (fp16 weights, fp16 activation).
2. Linear projections (left / right) from C → H and three sigmoid gates
   (left, right, out) from the normalized input.
3. Fuse mask (if present) together with the left/right gates in a single
   Triton kernel:
       left  = left  * mask * left_gate
       right = right * mask * right_gate
   The kernel works in‑place to avoid extra buffers.
4. Core reduction: for each hidden channel h we compute
       out_h = left_h @ right_hᵀ
   Implemented as a batched GEMM using torch.bmm (cuBLAS).
5. Second LayerNorm over the hidden dimension.
6. Multiply by the output gate and project back to the original channel
   dimension C with a final linear layer.
7. Cast the result to float32.

Only the mask‑gate fusion is performed in Triton; the heavy matrix
multiplication leverages cuBLAS which is optimal on the target GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,        # fp16*   (B,N,N,H) flattened, will be overwritten
    right_ptr,       # fp16*   (B,N,N,H) flattened, will be overwritten
    mask_ptr,        # fp16*   (B,N,N)   flattened, broadcast over H
    left_gate_ptr,   # fp16*   (B,N,N,H)
    right_gate_ptr,  # fp16*   (B,N,N,H)
    H: tl.constexpr,            # hidden dimension
    TOTAL_ELEMS: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place mask & gate fusion.

    left_out  = left  * mask * left_gate
    right_out = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL_ELEMS

    # Load tensors
    l  = tl.load(left_ptr + offs,        mask=mask)
    r  = tl.load(right_ptr + offs,       mask=mask)
    lg = tl.load(left_gate_ptr + offs,   mask=mask)
    rg = tl.load(right_gate_ptr + offs,  mask=mask)

    # Load mask (repeated for each hidden channel)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Write back in‑place
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)

                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, if True ignore mask (default False)

        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1) Input LayerNorm (fp16 weights, keep activation in fp16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float16)
    norm_bias   = weights["norm.bias"].to(device=device, dtype=torch.float16)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(torch.float16)   # work in fp16 for the heavy part

    # 2) Cast linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)      # (B, N, N, H)
    right = F.linear(x, right_proj_w)    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Prepare mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_tensor = mask.to(torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6) Fuse mask and gates (in‑place) with Triton
    # ------------------------------------------------------------------
    total_elements = left.numel()                # B * N * N * H
    BLOCK = 1024

    # Ensure contiguous memory for all tensors
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_flat = mask_tensor.view(-1).contiguous()

    grid = (triton.cdiv(total_elements, BLOCK),)
    _mask_gate_fusion_kernel[grid](
        left,
        right,
        mask_flat,
        left_gate,
        right_gate,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Core reduction: batched GEMM (left @ rightᵀ) over each hidden channel
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matmul using cuBLAS
    out_flat = torch.bmm(left_flat, right_flat.transpose(-2, -1))   # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10) Final projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 11) Cast to float32 (as required by the specification)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
