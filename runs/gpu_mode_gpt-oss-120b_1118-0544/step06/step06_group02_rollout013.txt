import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, total_elems, hidden_dim, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # compute mask index
    mask_idx = offs // hidden_dim
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

def _apply_gate_mask(proj, gate, mask, block_size=8192):
    # proj,gate: (B,N,N,H) fp16 contiguous
    # mask: (B,N,N,1) fp16 contiguous
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

"""
TriMul (outgoing) forward pass – Triton‑fused mask‑gate.
Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) with the provided weight/bias.
2. Cast to fp16 for compute‑intensive ops.
3. Fuse the five linear projections (left/right + their gates + out‑gate) into a
   single batched GEMM by concatenating the weight matrices.
4. Split the result, apply sigmoid to the three gate tensors.
5. If a mask is present, fuse element‑wise (proj * gate * mask) with a tiny
   Triton kernel for both left and right branches.
6. Compute the O(N³·H) reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is realized by reshaping to (B*H, N, N) and using torch.bmm (cuBLAS).
7. Apply a second LayerNorm, multiply by the output gate and a final linear
   projection back to the original channel dimension.
8. Cast the result back to the input dtype (float32).

The kernel `_gate_mask_kernel` implements the fused multiplication to reduce
memory traffic when a mask is used.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _gate_mask_kernel(
    proj_ptr,               # fp16 * (total_elems,)
    gate_ptr,               # fp16 * (total_elems,)
    mask_ptr,               # fp16 * (mask_elems,)
    out_ptr,                # fp16 * (total_elems,)
    total_elems,            # i32
    hidden_dim,             # i32  (number of hidden channels)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # each hidden element repeats the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 8192) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using a Triton kernel.
    All tensors must be contiguous fp16 on the same CUDA device.
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, C]
        - mask        : torch.Tensor of shape [B, N, N] (or None)
        - weights     : dict of model parameters (float32)
        - config      : dict with keys:
            * "dim" (int)          – input channel dimension C
            * "hidden_dim" (int)   – hidden dimension H
            * "nomask" (bool, optional) – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy computation
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Fuse all linear projections (left/right + gates + out gate)
    # ------------------------------------------------------------------
    # Cast weights to fp16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # Concatenate weight matrices: shape (5*H, C)
    combined_weight = torch.cat(
        [left_proj_w, right_proj_w,
         left_gate_w, right_gate_w,
         out_gate_w],
        dim=0,
    )  # (5*hidden_dim, dim)

    # Combined linear projection
    combined = torch.nn.functional.linear(x, combined_weight)  # (B,N,N,5*H)
    # Split back
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = torch.split(
        combined, hidden_dim, dim=-1
    )
    # Apply sigmoid to gates
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate = out_gate_pre.sigmoid()

    # ------------------------------------------------------------------
    # 3. Optional mask & gate fusion (Triton)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B,N,N) -> (B,N,N,1) -> fp16 contiguous
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4. Core O(N³·H) reduction via batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) for torch.bmm
    left_reshaped = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T  -> (B*H, N, N)
    out_reshaped = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))

    # reshape back to (B,N,N,H)
    out = out_reshaped.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Second LayerNorm + output gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 6. Final linear projection back to original dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # Cast back to the input dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
