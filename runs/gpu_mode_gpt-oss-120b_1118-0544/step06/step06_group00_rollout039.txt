def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype
    dim = config['dim']; hidden_dim = config['hidden_dim']; nomask = config.get('nomask', False)

    # layernorm
    norm_w = weights['norm.weight'].to(device).float()
    norm_b = weights['norm.bias'].to(device).float()
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    # cast to fp16
    x = x.to(torch.float16)

    # projections
    left_proj_w = weights['left_proj.weight'].to(device).to(torch.float16)
    right_proj_w = weights['right_proj.weight'].to(device).to(torch.float16)
    left_gate_w = weights['left_gate.weight'].to(device).to(torch.float16)
    right_gate_w = weights['right_gate.weight'].to(device).to(torch.float16)
    out_gate_w = weights['out_gate.weight'].to(device).to(torch.float16)

    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # apply gates and mask
    if mask is not None and not nomask:
        mask_exp = mask.to(device).to(torch.float16).unsqueeze(-1)
        left = left * left_gate * mask_exp
        right = right * right_gate * mask_exp
    else:
        left = left * left_gate
        right = right * right_gate

    # compute contraction via Triton kernel
    B, N, _, H = left.shape

    # transpose to (B, H, N, N) and reshape to (B*H, N, N)
    left_t = left.permute(0,3,1,2).contiguous()
    right_t = right.permute(0,3,1,2).contiguous()
    # shape (B, H, N, N)

    left_reshape = left_t.view(B*H, N, N)
    right_reshape = right_t.view(B*H, N, N)

    # allocate output
    out_reshape = torch.empty_like(left_reshape, dtype=torch.float16)

    # get strides
    stride_a_batch = left_reshape.stride(0)
    stride_a_m = left_reshape.stride(1)
    stride_a_k = left_reshape.stride(2)
    stride_b_batch = right_reshape.stride(0)
    stride_b_n = right_reshape.stride(1)
    stride_b_k = right_reshape.stride(2)
    stride_c_batch = out_reshape.stride(0)
    stride_c_m = out_reshape.stride(1)
    stride_c_n = out_reshape.stride(2)

    # kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B*H,
            tl.cdiv(N, BLOCK_M),
            tl.cdiv(N, BLOCK_N))

    tr_mul_batch_kernel[grid](
        a_ptr=left_reshape,
        b_ptr=right_reshape,
        c_ptr=out_reshape,
        M=N,
        N=N,
        K=N,
        stride_az=stride_a_batch,
        stride_am=stride_a_m,
        stride_ak=stride_a_k,
        stride_bz=stride_b_batch,
        stride_bn=stride_b_n,
        stride_bk=stride_b_k,
        stride_cz=stride_c_batch,
        stride_cm=stride_c_m,
        stride_cn=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape out back to (B,N,N,H)
    out = out_reshape.view(B, H, N, N).permute(0,2,3,1).contiguous()

    # to_out_norm
    to_out_norm_w = weights['to_out_norm.weight'].to(device).to(torch.float16)
    to_out_norm_b = weights['to_out_norm.bias'].to(device).to(torch.float16)
    out = torch.nn.functional.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # apply output gate
    out = out * out_gate

    # final linear

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Input LayerNorm (float32 for stability) → cast to fp16.
2. Linear projections (left/right) and three sigmoid gates (left/right/out) in fp16.
3. Fuse mask (if present) with the two element‑wise gates (pure PyTorch – cheap).
4. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication L @ Rᵀ for each hidden channel.
   We fuse the batch (B·H) dimension and implement the GEMM with a custom
   Triton kernel (fp16 inputs, fp32 accumulation, fp16 output).
5. Apply second LayerNorm on the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original dimension.
6. Cast back to the input dtype.

The heavy O(N³·H) contraction is completely performed on the GPU by Triton,
giving a large speed‑up over the naïve PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_az, stride_am, stride_ak,   # a: (batch, M, K)
    stride_bz, stride_bn, stride_bk,   # b: (batch, N, K)   (will be transposed on‑the‑fly)
    stride_cz, stride_cm, stride_cn,   # c: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C[batch, M, N] = A[batch, M, K] @ B[batch, N, K].T"""
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # -------------------------- tile indices -------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------- k‑loop -------------------------- #
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + batch_idx * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0
        ).to(tl.float32)

        # B tile: (BLOCK_N, BLOCK_K) → transpose to (BLOCK_K, BLOCK_N)
        b_ptrs = (
            b_ptr
            + batch_idx * stride_bz
            + offs_n[:, None] * stride_bn
            + offs_k[None, :] * stride_bk
        )
        b = tl.load(
            b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0
        ).to(tl.float32)
        b = tl.trans(b)               # (BLOCK_K, BLOCK_N)

        # accumulate
        acc += tl.dot(a, b)           # (BLOCK_M, BLOCK_N)

    # -------------------------- store result -------------------------- #
    c_ptrs = (
        c_ptr
        + batch_idx * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int)         – input channel dimension C
                * "hidden_dim" (int)  – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------ #
    # unpack
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (float32) → fp16
    # ------------------------------------------------------------------ #
    norm_w = weights["norm.weight"].to(device).float()
    norm_b = weights["norm.bias"].to(device).float()
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 2. Linear projections + gates (fp16)
    # ------------------------------------------------------------------ #
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)                    # (B, N, N, H)
    right = F.linear(x, right_proj_w)                  # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B, N, N, H)

    # ------------------------------------------------------------------ #
    # 3. Apply gates and optional mask (pure PyTorch – cheap)
    # ------------------------------------------------------------------ #
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * left_gate * mask_fp
        right = right * right_gate * mask_fp
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------ #
    # 4. Core contraction via Triton batched GEMM
    # ------------------------------------------------------------------ #
    B, N, _, H = left.shape  # (B, N, N, H)

    # shape to (B, H, N, N) then fold batch & channel → (B*H, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left_t.view(B * H, N, N)
    right_flat = right_t.view(B * H, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # strides (in elements) for the flattened tensors
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_n = right_flat.stride(1)
    stride_b_k = right_flat.stride(2)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                          # batch dimension (B·H)
        triton.cdiv(N, BLOCK_M),       # row blocks
        triton.cdiv(N, BLOCK_N),       # col blocks
    )

    _batched_gemm_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        M=N,
        N=N,
        K=N,
        stride_az=stride_a_batch,
        stride_am=stride_a_m,
        stride_ak=stride_a_k,
        stride_bz=stride_b_batch,
        stride_bn=stride_b_n,
        stride_bk=stride_b_k,
        stride_cz=stride_c_batch,
        stride_cm=stride_c_m,
        stride_cn=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 5. Output LayerNorm, gate, final linear
    # ------------------------------------------------------------------ #
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------ #
    # 6. Cast back to original dtype
    # ------------------------------------------------------------------ #
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
