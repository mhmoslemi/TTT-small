    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)  # batch_hidden
    pid_m = tl.program_id(1)  # M block
    pid_n = tl.program_id(2)  # N block
    # compute batch and hidden indices
    batch_idx = pid // H
    hidden_idx = pid % H

"""Triton-accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the PyTorch reference:
1. LayerNorm over input channels (FP32).
2. Linear projections to hidden dimension (FP16).
3. Sigmoid gates and optional mask fused in a Triton kernel.
4. Core contraction `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`
   realized as a batched GEMM over B*hidden matrices using a custom
   Triton kernel (fp16 inputs, fp32 accumulation).
5. LayerNorm over hidden dim, multiply by output gate, final linear
   projection back to original dimension.
All heavy‑weight memory‑bound work is performed in Triton; the rest
leverages PyTorch's highly‑optimized kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: fuse mask and per‑element gates (left/right) in‑place
# -------------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask_active = offs < total_elems

    # Load tensors (fp16)
    left = tl.load(left_ptr + offs, mask=mask_active, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask_active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask_active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask_active, other=0.0)

    if HAS_MASK:
        # each mask entry corresponds to hidden_dim consecutive elements
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask_active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask_active)
    tl.store(right_ptr + offs, right, mask=mask_active)


# -------------------------------------------------------------------------
# Triton kernel: batched GEMM for the core TriMul contraction
#   out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
# -------------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_j, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_batch = tl.program_id(0)   # combined batch * hidden
    pid_m = tl.program_id(1)       # block row (i)
    pid_n = tl.program_id(2)       # block column (j)

    # recover batch and hidden indices
    batch_idx = pid_batch // H
    hidden_idx = pid_batch % H

    # offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this (batch, hidden) slice
    left_base = left_ptr + batch_idx * stride_l_b + hidden_idx * stride_l_h
    right_base = right_ptr + batch_idx * stride_r_b + hidden_idx * stride_r_h
    out_base = out_ptr + batch_idx * stride_o_b + hidden_idx * stride_o_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left block (M x K)
        left = tl.load(
            left_base + offs_m[:, None] * stride_l_i + k_offset[None, :] * stride_l_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load right block transposed (K x N)
        # right layout is (batch, hidden, j, k)
        right = tl.load(
            right_base + k_offset[:, None] * stride_r_k + offs_n[None, :] * stride_r_j,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Multiply‑accumulate
        acc += tl.dot(left, right)  # result is fp32

    # Store output tile
    tl.store(
        out_base + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # -----------------------------------------------------------------
    # 2. Cast to half precision for compute‑heavy part
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (bias‑less) in FP16
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)      # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)    # (B,N,N,hidden_dim)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B,N,N,hidden_dim)

    # -----------------------------------------------------------------
    # 4. Fuse mask and per‑element gates via Triton (in‑place)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten tensors
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # simple element‑wise gating (no mask)
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM via Triton
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Rearrange to (B, H, N, N) and make contiguous
    left_mat = left.permute(0, 3, 1, 2).contiguous()
    right_mat = right.permute(0, 3, 1, 2).contiguous()
    out_mat = torch.empty_like(left_mat)  # (B, H, N, N), fp16

    # compute strides (in elements)
    stride_l_b = left_mat.stride(0)
    stride_l_h = left_mat.stride(1)
    stride_l_i = left_mat.stride(2)
    stride_l_k = left_mat.stride(3)

    stride_r_b = right_mat.stride(0)
    stride_r_h = right_mat.stride(1)
    stride_r_j = right_mat.stride(2)
    stride_r_k = right_mat.stride(3)

    stride_o_b = out_mat.stride(0)
    stride_o_h = out_mat.stride(1)
    stride_o_i = out_mat.stride(2)
    stride_o_j = out_mat.stride(3)

    # launch parameters (tuned for typical sizes)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid = (B * H, grid_m, grid_n)

    _tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        B,
        H,
        N,
        stride_l_b,
        stride_l_h,
        stride_l_i,
        stride_l_k,
        stride_r_b,
        stride_r_h,
        stride_r_j,
        stride_r_k,
        stride_o_b,
        stride_o_h,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) fp16

    # -----------------------------------------------------------------
    # 7. Cast back to original dtype
    # -----------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating, and final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate   # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim), fp16

    # --------------------------------------------------------------
    # 7. Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
