import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_proj_kernel(
    x_ptr,        # *float16  (total_elements, dim)
    mask_ptr,     # *float16  (total_elements)
    left_out_ptr, # *float16  (total_elements, hidden_dim)
    right_out_ptr,
    out_gate_ptr,
    left_proj_w_ptr,   # *float16 (dim, hidden_dim)
    right_proj_w_ptr,
    left_gate_w_ptr,
    right_gate_w_ptr,
    out_gate_w_ptr,
    TOTAL: tl.constexpr,   # total_elements = B * N * N
    DIM: tl.constexpr,     # input channel dim
    HIDDEN: tl.constexpr,  # hidden_dim
    BLOCK_M: tl.constexpr, # rows per program
    BLOCK_N: tl.constexpr, # cols (hidden) per program
    BLOCK_K: tl.constexpr, # inner dim tile size
):
    # Program IDs
    pid_m = tl.program_id(0)  # block of rows
    pid_n = tl.program_id(1)  # block of hidden cols

    # Row indices for this block
    row_range = tl.arange(0, BLOCK_M)
    row_idx = pid_m * BLOCK_M + row_range                # shape (BLOCK_M,)
    row_mask = row_idx < TOTAL

    # Column indices (hidden dimension) for this block
    col_range = tl.arange(0, BLOCK_N)
    col_idx = pid_n * BLOCK_N + col_range                # shape (BLOCK_N,)
    col_mask = col_idx < HIDDEN

    # Load mask values (scalar per row)
    m = tl.load(mask_ptr + row_idx, mask=row_mask, other=0.0)   # (BLOCK_M,)

    # Initialize accumulators in FP32
    acc_left = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_right = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_left_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_right_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_out_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the input channel dimension
    for k in range(0, tl.cdiv(DIM, BLOCK_K)):
        k_start = k * BLOCK_K
        k_offset = k_start + tl.arange(0, BLOCK_K)          # (BLOCK_K,)
        k_mask = k_offset < DIM

        # Load a tile of the input (BLOCK_M x BLOCK_K)
        x = tl.load(
            x_ptr + (row_idx[:, None] * DIM) + k_offset[None, :],
            mask=row_mask[:, None] & k_mask[None, :],
            other=0.0,
        ).to(tl.float32)   # shape (BLOCK_M, BLOCK_K)

        # Load corresponding weight tiles (BLOCK_K x BLOCK_N) for each projection/gate
        def load_weight(w_ptr):
            return tl.load(
                w_ptr + (k_offset[:, None] * HIDDEN) + col_idx[None, :],
                mask=k_mask[:, None] & col_mask[None, :],
                other=0.0,
            ).to(tl.float32)   # (BLOCK_K, BLOCK_N)

        w_left = load_weight(left_proj_w_ptr)
        w_right = load_weight(right_proj_w_ptr)
        w_left_gate = load_weight(left_gate_w_ptr)
        w_right_gate = load_weight(right_gate_w_ptr)
        w_out_gate = load_weight(out_gate_w_ptr)

        # Accumulate dot products
        acc_left += tl.dot(x, w_left)           # (BLOCK_M, BLOCK_N)
        acc_right += tl.dot(x, w_right)
        acc_left_gate += tl.dot(x, w_left_gate)
        acc_right_gate += tl.dot(x, w_right_gate)
        acc_out_gate += tl.dot(x, w_out_gate)

    # Apply sigmoid to gate accumulators
    def sigmoid(x):
        return 1.0 / (1.0 + tl.exp(-x))

    gate_left = sigmoid(acc_left_gate)
    gate_right = sigmoid(acc_right_gate)
    gate_out = sigmoid(acc_out_gate)

    # Broadcast mask to (BLOCK_M, BLOCK_N)
    m_tile = m[:, None]                     # (BLOCK_M, 1)

    # Fuse mask and gates
    left_out = acc_left * m_tile * gate_left
    right_out = acc_right * m_tile * gate_right

    # Store results (cast back to fp16)
    store_mask = row_mask[:, None] & col_mask[None, :]

    tl.store(
        left_out_ptr + (row_idx[:, None] * HIDDEN) + col_idx[None, :],
        left_out.to(tl.float16),
        mask=store_mask,
    )
    tl.store(
        right_out_ptr + (row_idx[:, None] * HIDDEN) + col_idx[None, :],
        right_out.to(tl.float16),
        mask=store_mask,
    )
    tl.store(
        out_gate_ptr + (row_idx[:, None] * HIDDEN) + col_idx[None, :],
        gate_out.to(tl.float16),
        mask=store_mask,
    )

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul) using a
    fused Triton kernel for projection, gating and mask application.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1) First LayerNorm (fp32 for stability)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to fp16 for the heavy part
    x = x.to(torch.float16)

    # 2) Prepare mask (flattened)
    total_elements = bs * N * N
    if mask is None or nomask:
        mask_tensor = torch.ones(total_elements, dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.view(total_elements).to(torch.float16)

    # 3) Transpose weight matrices to (dim, hidden_dim) layout for efficient dot
    def prepare_w(key):
        w = weights[key].to(device=device, dtype=torch.float16)
        return w.t().contiguous()   # now (dim, hidden_dim)

    left_proj_w = prepare_w("left_proj.weight")
    right_proj_w = prepare_w("right_proj.weight")
    left_gate_w = prepare_w("left_gate.weight")
    right_gate_w = prepare_w("right_gate.weight")
    out_gate_w = prepare_w("out_gate.weight")

    # 4) Allocate output buffers (flattened)
    left_out = torch.empty((total_elements, hidden_dim), dtype=torch.float16, device=device)
    right_out = torch.empty_like(left_out)
    out_gate = torch.empty_like(left_out)

    # 5) Launch fused projection/gate kernel
    BLOCK_M = 64   # rows per program
    BLOCK_N = 32   # hidden channels per program
    BLOCK_K = 32   # inner dimension tile

    grid = (
        triton.cdiv(total_elements, BLOCK_M),
        triton.cdiv(hidden_dim, BLOCK_N),
    )

    tri_mul_proj_kernel[grid](
        x,
        mask_tensor,
        left_out,
        right_out,
        out_gate,
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
        TOTAL=total_elements,
        DIM=dim,
        HIDDEN=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # 6) Reshape for batched GEMM
    left = left_out.view(bs, N, N, hidden_dim)
    right = right_out.view(bs, N, N, hidden_dim)

    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)

    # 7) Core reduction: batched matrix multiplication (cuBLAS)
    out_flat = torch.bmm(left_flat, right_flat.transpose(-2, -1))

    # 8) Restore original tensor shape (B, N, N, H)
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 9) Second LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float16),
    )

    # 10) Apply output gate
    out_gate_tensor = out_gate.view(bs, N, N, hidden_dim)
    out = out * out_gate_tensor

    # 11) Final linear projection back to original channels
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)

    # 12) Cast back to original dtype (usually float32)
    out = out.to(orig_dtype)

    return out

"""
TriMul custom Triton kernel (outgoing version).

The kernel fuses:
    * Linear projections from C → H (left/right)
    * Linear gates (left/right/out) with sigmoid
    * Optional binary mask (broadcast over H)

All three outputs (left_out, right_out, out_gate) are written directly in
FP16, avoiding the intermediate tensors used in the reference implementation.
The heavy N³ reduction is performed with a batched GEMM (torch.bmm) on the
reshaped tensors, which lets cuBLAS take full advantage of the B200 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_proj_kernel(
    x_ptr,            # *fp16  (total_elements, dim)
    mask_ptr,         # *fp16  (total_elements)
    left_out_ptr,     # *fp16  (total_elements, hidden_dim)
    right_out_ptr,    # *fp16  (total_elements, hidden_dim)
    out_gate_ptr,     # *fp16  (total_elements, hidden_dim)
    left_proj_w_ptr,  # *fp16  (dim, hidden_dim)
    right_proj_w_ptr, # *fp16  (dim, hidden_dim)
    left_gate_w_ptr,  # *fp16  (dim, hidden_dim)
    right_gate_w_ptr, # *fp16  (dim, hidden_dim)
    out_gate_w_ptr,   # *fp16  (dim, hidden_dim)
    TOTAL: tl.constexpr,   # total_elements = B * N * N
    DIM: tl.constexpr,     # input channel dimension C
    HIDDEN: tl.constexpr,  # hidden dimension H
    BLOCK_M: tl.constexpr, # rows processed per program
    BLOCK_N: tl.constexpr, # hidden‑dim tiles per program
    BLOCK_K: tl.constexpr, # inner‑dim tile size
):
    # ------------------------------------------------------------------
    #  Program IDs and index ranges
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)           # block over rows (elements)
    pid_n = tl.program_id(1)           # block over hidden columns

    row = tl.arange(0, BLOCK_M) + pid_m * BLOCK_M          # (BLOCK_M,)
    col = tl.arange(0, BLOCK_N) + pid_n * BLOCK_N          # (BLOCK_N,)

    row_mask = row < TOTAL
    col_mask = col < HIDDEN

    # Load mask scalar per element (broadcast later)
    m = tl.load(mask_ptr + row, mask=row_mask, other=0.0)  # (BLOCK_M,)

    # ------------------------------------------------------------------
    #  Accumulators (FP32 for precision)
    # ------------------------------------------------------------------

    acc_right = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_left_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_right_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_out_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  Loop over input channel dimension (C)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(DIM, BLOCK_K)):
        k_start = k * BLOCK_K
        k_off = k_start + tl.arange(0, BLOCK_K)                # (BLOCK_K,)
        k_mask = k_off < DIM

        # x tile: (BLOCK_M, BLOCK_K)
        x = tl.load(
            x_ptr + (row[:, None] * DIM) + k_off[None, :],
            mask=row_mask[:, None] & k_mask[None, :],
            other=0.0,
        ).to(tl.float32)

            return tl.load(
                w_ptr + (k_off[:, None] * HIDDEN) + col[None, :],
                mask=k_mask[:, None] & col_mask[None, :],
                other=0.0,
            ).to(tl.float32)

        w_left = load_w(left_proj_w_ptr)
        w_right = load_w(right_proj_w_ptr)
        w_left_gate = load_w(left_gate_w_ptr)
        w_right_gate = load_w(right_gate_w_ptr)
        w_out_gate = load_w(out_gate_w_ptr)

        # Accumulate dot products
        acc_left += tl.dot(x, w_left)
        acc_right += tl.dot(x, w_right)
        acc_left_gate += tl.dot(x, w_left_gate)
        acc_right_gate += tl.dot(x, w_right_gate)
        acc_out_gate += tl.dot(x, w_out_gate)

    # ------------------------------------------------------------------
    #  Sigmoid for gates
    # ------------------------------------------------------------------
    def sigmoid(x):
        return 1.0 / (1.0 + tl.exp(-x))

    gate_left = sigmoid(acc_left_gate)
    gate_right = sigmoid(acc_right_gate)
    gate_out = sigmoid(acc_out_gate)

    # ------------------------------------------------------------------
    #  Fuse mask and gates, write results (cast back to fp16)
    # ------------------------------------------------------------------
    m_tile = m[:, None]                                   # (BLOCK_M,1)
    left_tile = acc_left * m_tile * gate_left
    right_tile = acc_right * m_tile * gate_right

    store_mask = row_mask[:, None] & col_mask[None, :]

    tl.store(
        left_out_ptr + (row[:, None] * HIDDEN) + col[None, :],
        left_tile.to(tl.float16),
        mask=store_mask,
    )
    tl.store(
        right_out_ptr + (row[:, None] * HIDDEN) + col[None, :],
        right_tile.to(tl.float16),
        mask=store_mask,
    )
    tl.store(
        out_gate_ptr + (row[:, None] * HIDDEN) + col[None, :],
        gate_out.to(tl.float16),
        mask=store_mask,
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]  (float32 or float16)
        - mask         : torch.Tensor of shape [B, N, N] (binary) or None
        - weights      : dict of model parameters
        - config       : dict with keys
            * "dim"         – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask"      – bool, if True ignore mask (default False)

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) First LayerNorm (computed in FP32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)            # cast to FP16 for the heavy part

    # ------------------------------------------------------------------
    # 2) Prepare mask (flattened)
    # ------------------------------------------------------------------
    total_elements = bs * N * N
    if mask is None or nomask:
        mask_flat = torch.ones(total_elements, dtype=torch.float16, device=device)
    else:
        mask_flat = mask.view(total_elements).to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Transpose weight matrices to (dim, hidden_dim) layout
    # ------------------------------------------------------------------
    def prep_w(key):
        w = weights[key].to(device=device, dtype=torch.float16)
        return w.t().contiguous()          # (dim, hidden_dim)

    left_proj_w = prep_w("left_proj.weight")
    right_proj_w = prep_w("right_proj.weight")
    left_gate_w = prep_w("left_gate.weight")
    right_gate_w = prep_w("right_gate.weight")
    out_gate_w = prep_w("out_gate.weight")

    # ------------------------------------------------------------------
    # 4) Allocate output buffers (flattened)
    # ------------------------------------------------------------------
    left_out = torch.empty((total_elements, hidden_dim), dtype=torch.float16, device=device)
    right_out = torch.empty_like(left_out)
    out_gate = torch.empty_like(left_out)

    # ------------------------------------------------------------------
    # 5) Launch fused projection/gate kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64   # rows per program (adjustable)
    BLOCK_N = 32   # hidden‑dim columns per program
    BLOCK_K = 32   # inner‑dim tile size

    grid = (
        triton.cdiv(total_elements, BLOCK_M),
        triton.cdiv(hidden_dim, BLOCK_N),
    )

    tri_mul_proj_kernel[grid](
        x,
        mask_flat,
        left_out,
        right_out,
        out_gate,
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
        TOTAL=total_elements,
        DIM=dim,
        HIDDEN=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6) Reshape for batched GEMM (the O(N³) reduction)
    # ------------------------------------------------------------------
    left = left_out.view(bs, N, N, hidden_dim)
    right = right_out.view(bs, N, N, hidden_dim)

    left_flat = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)

    # ------------------------------------------------------------------
    # 7) Core reduction – batched matrix multiplication (cuBLAS)
    # ------------------------------------------------------------------
    out_flat = torch.bmm(left_flat, right_flat.transpose(-2, -1))

    # ------------------------------------------------------------------
    # 8) Restore tensor shape (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9) Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"]
            .to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"]
            .to(device=device, dtype=torch.float16),
    )

    # ------------------------------------------------------------------
    # 10) Apply output gate (produced by the kernel)
    # ------------------------------------------------------------------
    out_gate = out_gate.view(bs, N, N, hidden_dim)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)     # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 12) Cast back to original dtype (usually FP32)
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)
    return out
