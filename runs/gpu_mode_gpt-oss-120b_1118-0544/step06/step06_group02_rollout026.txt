def _proj_gate_mask(proj, gate, mask, block_size=1024):
    assert proj.is_contiguous() and gate.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) (C = dim) with the provided
   weight/bias (FP32).  Cast the result to FP16 for the compute‑intensive part.
2. Compute four linear projections in FP16:
      * left_proj, right_proj     – (B,N,N,H)
      * left_gate_raw, right_gate_raw – (B,N,N,H)
   where H = hidden_dim.
3. Compute the output gate (B,N,N,H) via a linear + sigmoid (FP16).
4. Fuse projection, sigmoid(gate) and optional mask in a single Triton
   kernel (`_proj_gate_mask_kernel`).  This yields the gated tensors
   `left` and `right` in FP16 without extra memory passes.
5. Core cubic reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented with a single `torch.einsum` (batched GEMM) on the
   already‑contiguous tensors, avoiding costly permutes/copies.
6. Apply a second LayerNorm on the hidden dimension (FP16), multiply by the
   output gate, and linearly project back to the original channel dimension.
7. Cast the final result back to the input dtype (float32).

The kernel implements the element‑wise multiplication with sigmoid and mask,
saving memory bandwidth and keeping the compute in half precision.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: proj * sigmoid(gate) * mask   (proj,gate are FP16)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # index of the corresponding mask element (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    # load tensors
    proj = tl.load(proj_ptr + offs, mask=valid, other=tl.float16(0.0))
    gate = tl.load(gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    # sigmoid(gate)
    gate_sig = 1.0 / (1.0 + tl.exp(-gate))

    # load mask (default 1.0 if out‑of‑bounds)
    mask_val = tl.load(mask_ptr + mask_idx,
                       mask=mask_idx < (total_elems // hidden_dim),
                       other=tl.float16(1.0))

    out = proj * gate_sig * mask_val
    tl.store(out_ptr + offs, out, mask=valid)

def _apply_proj_gate_mask(proj: torch.Tensor,
                          gate: torch.Tensor,
                          mask: torch.Tensor,
                          block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * sigmoid(gate) * mask` using a Triton kernel.
    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, C]
        - mask        : torch.Tensor of shape [B, N, N] (or None)
        - weights     : dict of model parameters (float32)
        - config      : dict with keys:
            * "dim" (int)        – input channel dimension C
            * "hidden_dim" (int) – hidden dimension H
            * "nomask" (bool)    – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # LayerNorm (input) – keep in FP32 then cast to FP16
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Cast weights to FP16 for the heavy compute
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj   = torch.nn.functional.linear(x, left_proj_w)          # (B,N,N,H)
    right_proj  = torch.nn.functional.linear(x, right_proj_w)         # (B,N,N,H)
    left_gate_r = torch.nn.functional.linear(x, left_gate_w)          # (B,N,N,H) – raw gate
    right_gate_r= torch.nn.functional.linear(x, right_gate_w)         # (B,N,N,H)
    out_gate    = torch.nn.functional.linear(x, out_gate_w).sigmoid() # (B,N,N,H)

    # ------------------------------------------------------------------
    # Apply mask & sigmoid‑gate fusion
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask → FP16, broadcasted over hidden dimension inside the kernel
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_proj_gate_mask(left_proj, left_gate_r, mask_fp16)
        right = _apply_proj_gate_mask(right_proj, right_gate_r, mask_fp16)
    else:
        left  = left_proj * left_gate_r.sigmoid()
        right = right_proj * right_gate_r.sigmoid()

    # Release intermediate tensors early
    del left_proj, right_proj, left_gate_r, right_gate_r, x

    # ------------------------------------------------------------------
    # Core cubic reduction via einsum (batched GEMM)
    # ------------------------------------------------------------------
    # out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # Release left/right
    del left, right

    # ------------------------------------------------------------------
    # Second LayerNorm (hidden_dim) – keep FP16
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # Cast back to input dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
