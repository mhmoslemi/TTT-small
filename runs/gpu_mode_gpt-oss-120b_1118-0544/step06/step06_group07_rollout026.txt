@triton.jit
def gate_mask_kernel(left_ptr, right_ptr,
                     left_gate_ptr, right_gate_ptr,
                     mask_ptr,
                     total_elems, hidden_dim,
                     HAS_MASK: tl.constexpr,
                     BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        mask_idx = offs // hidden_dim
        # load mask, but mask has shape total_elems // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    # layernorm
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    # Cast to fp16
    x = x.to(torch.float16)
    # linear projections
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16)
        # Flatten tensors
        left_flat = left.reshape(-1)
        right_flat = right.reshape(-1)
        left_gate_flat = left_gate.reshape(-1)
        right_gate_flat = right_gate.reshape(-1)
        mask_flat = mask_fp.reshape(-1)  # shape (B*N*N,)
        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # just multiply by gates
        left = left * left_gate
        right = right * right_gate

    # core contraction via batched GEMM
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * H, N, N)  # transpose
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    # output layernorm
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # apply out_gate
    out = out * out_gate

    # final linear
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)

    # cast back to original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)
    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (float32 for stability).
2. Cast to fp16 for compute‑intensive parts.
3. Linear projections to a hidden dimension (bias‑less):
   - left, right    : shape (B, N, N, H)
   - left_gate, right_gate, out_gate : shape (B, N, N, H) after sigmoid.
4. Fuse mask (if present) and gate multiplications in a single Triton kernel,
   writing the gated tensors back in‑place.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (left @ rightᵀ) using torch.bmm, which
   leverages cuBLAS tensor cores.
6. Apply a second LayerNorm on the hidden dimension, multiply by out_gate,
   and a final linear projection back to the original dimension.
7. Cast back to the input dtype and return.

The kernel `_gate_mask_kernel` handles the element‑wise fusion of mask and gates.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,            # [B*N*N*H] flattened
    left_gate_ptr, right_gate_ptr,  # same shape
    mask_ptr,                       # [B*N*N] flattened
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,         # compile‑time bool
    BLOCK_SIZE: tl.constexpr,       # compile‑time int
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    If HAS_MASK=False the mask term is omitted.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Each mask element is shared across `hidden_dim` consecutive entries.
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Write results back in‑place.
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of parameter tensors (float32)
            - config: dict containing:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, mask is ignored

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (bias‑less) and gates
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)            # (B,N,N,H)
    right = F.linear(x, right_proj_w)          # (B,N,N,H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4. Fuse mask and gate multiplications via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # Make tensors contiguous before flattening
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        # Flatten tensors for the kernel
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_fp.view(-1)  # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back to original layout
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # Simple element‑wise gating without mask
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM (left @ rightᵀ) using torch.bmm.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) where each slice is a matrix
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)          # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * H, N, N)        # (B*H, N, N) with k ↔ j
    out_mat = torch.bmm(left_mat, right_mat)                         # (B*H, N, N)

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------
    # 7. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
