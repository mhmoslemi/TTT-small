def _apply_mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    # total elements = B*N*N*H
    mask_idx = offset // H  # because mask repeats across hidden dimension

    m = tl.load(mask_ptr + mask_idx, mask=offset < total_elements)
    l = tl.load(left_ptr + offset, mask=offset < total_elements)
    r = tl.load(right_ptr + offset, mask=offset < total_elements)
    lg = tl.load(left_gate_ptr + offset, mask=offset < total_elements)

    tl.store(left_ptr + offset, l * m * lg, mask=offset < total_elements)
    tl.store(right_ptr + offset, r * m * rg, mask=offset < total_elements)

@triton.jit
def _apply_mask_gate_kernel(
    left_ptr, right_ptr, mask_ptr, left_gate_ptr, right_gate_ptr,
    H: tl.constexpr,
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply
    tl.store(left_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)
```

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm Overview
------------------
1. Layer‑normalize the input tensor (float32 → fp16).
2. Linear projections `left` and `right` from C → H (hidden_dim) in fp16.
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. If a mask is present, fuse mask, left‑gate and right‑gate with a Triton kernel:
       left  = left  * mask * left_gate
       right = right * mask * right_gate
   (in‑place, no extra temporaries).
   Otherwise multiply by the gates directly.
5. Reshape `left`/`right` to (B*H, N, N) and perform a batched GEMM:
       out[b*h, i, j] = Σ_k left[b*h, i, k] * right[b*h, j, k]
   using torch.bmm (cuBLAS).
6. Reshape back to (B, N, N, H) and apply a second LayerNorm over H.
7. Multiply by the output gate and project back to the original channel dimension C.
8. Cast the result to the original dtype and return.

Only the mask‑gate fusion is written in Triton; the remaining heavy
operations use highly‑optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: in‑place mask & gate fusion
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # fp16*   (B,N,N,H) flattened
    right_ptr,         # fp16*   (B,N,N,H) flattened
    mask_ptr,          # fp16*   (B,N,N)   flattened
    left_gate_ptr,     # fp16*   (B,N,N,H) flattened
    right_gate_ptr,    # fp16*   (B,N,N,H) flattened
    H: tl.constexpr,          # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,      # threads per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # load values
    l  = tl.load(left_ptr + offs, mask=mask)
    r  = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # map linear index to mask index (mask repeats H times)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # in‑place multiply
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (same keys as the reference module)
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, if True ignore the mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (computed in fp32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    ).to(torch.float16)   # cast to fp16 for the remaining ops

    # ------------------------------------------------------------------
    # 2) Cast all weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask and gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask -> fp16, flatten for the kernel
        mask_fp16 = mask.to(device=device, dtype=torch.float16)

        total_elements = left.numel()                # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        # In‑place fusion via Triton
        _mask_gate_fusion_kernel[grid](
            left.view(-1),
            right.view(-1),
            mask_fp16.view(-1),
            left_gate.view(-1),
            right_gate.view(-1),
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # left and right now already contain mask * gate
    else:
        # No mask: simple elementwise multiplication
        left = left * left_gate
        right = right * right_gate

    # Gates no longer needed – free memory
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: batched GEMM (left @ rightᵀ) over the sequence dim
    # ------------------------------------------------------------------
    # Move hidden dim to the second position for a convenient view
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left.shape
    # Collapse batch and hidden dim → (B*H, N, N)
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # Batched matrix multiplication (cuBLAS)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Restore original layout (B, N, N, H)
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)   # (B, N, N, H)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over the hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
