@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_k, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # Compute number of blocks
    num_m_blocks = tl.cdiv(N, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)
    num_bh = B * H
    bh = pid // (num_m_blocks * num_n_blocks)  # batch*head id
    block_i = (pid // num_n_blocks) % num_m_blocks
    block_j = pid % num_n_blocks

    # Compute batch and head indices
    b = bh // H
    h = bh % H

    # Compute start indices
    i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary checks
    mask_i = i < N
    mask_j = j < N

    # Pointers base for this b and h
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load blocks
        a_ptrs = left_base + i[:, None] * stride_l_i + k_offset[None, :] * stride_l_k
        b_ptrs = right_base + k_offset[:, None] * stride_r_k + j[None, :] * stride_r_i

        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Store result
    out_ptrs = out_base + i[:, None] * stride_o_i + j[None, :] * stride_o_k
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

Note: `stride_l_i` is stride on i-dim (second dimension). For left shape (B,N,N,H), strides are:

- stride on dimension 0 (B) = N*N*H maybe
- stride on dimension 1 (i) = N*H (since moving one row changes the second N dimension)
- stride on dimension 2 (k) = H (since moving third dimension changes the inner N)
- stride on dimension 3 (H) = 1

But we just use whatever returned by .stride().

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # layernorm
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    # cast to fp16

    BLOCK_N = 64

    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # cast back to original dtype
    out = out.to(dtype_in)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input (float32) → cast to float16.
2. Linear projections for left/right values and their gates (bias‑less).
3. Apply sigmoid to gates and (optionally) the mask, then fuse gate multiplication.
4. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM using a Triton kernel that processes
   (B·hidden_dim) independent N×N matrix multiplications with tiling.
5. Output LayerNorm on the hidden dimension, multiply by the output gate.
6. Final linear projection back to the original dimension.
7. Cast back to the input dtype.

The heavy O(N³·H) contraction is performed entirely in the Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_k, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute, for each batch b and head h:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    left / right / out are stored as (B, N, N, H) in row‑major order.
    """
    pid = tl.program_id(0)

    # Number of blocks along each dimension
    num_m_blocks = tl.cdiv(N, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)
    # Total number of (batch, head) pairs
    num_bh = B * H

    # Decode program id
    bh = pid // (num_m_blocks * num_n_blocks)          # which (b, h)
    block_i = (pid // num_n_blocks) % num_m_blocks    # i‑tile
    block_j = pid % num_n_blocks                       # j‑tile

    b = bh // H
    h = bh % H

    # Tile offsets
    i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # Base pointers for this (b, h)
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + i[:, None] * stride_l_i + k_offset[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right block transposed: shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + k_offset[:, None] * stride_r_k + j[None, :] * stride_r_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # Store result
    out_ptrs = out_base + i[:, None] * stride_o_i + j[None, :] * stride_o_k
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------- 1. Input LayerNorm (float32) ----------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # Cast to fp16 for the bulk of computation
    x = x.to(torch.float16)

    # -------- 2. Linear projections ----------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left  = F.linear(x, left_proj_w)       # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)      # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # -------- 3. Apply gates (and optional mask) ----------
    if mask is not None and not nomask:
        mask = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * left_gate * mask
        right = right * right_gate * mask
    else:
        left = left * left_gate
        right = right * right_gate

    # Ensure tensors are contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()

    # -------- 4. Core contraction via Triton ----------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # (B,N,N,hidden_dim), dtype fp16

    # Strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_k, stride_o_h = out.stride()

    # Tiling parameters (tuned for modern Ampere GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H *
            ((N + BLOCK_M - 1) // BLOCK_M) *
            ((N + BLOCK_N - 1) // BLOCK_N),)

    _tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_k, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------- 5. Output LayerNorm and gating ----------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (still fp16)

    # -------- 6. Final linear projection ----------
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # -------- 7. Cast back to original dtype ----------
    out = out.to(dtype_in)

    return out
