    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,

    batch_id = tl.program_id(0)

    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)

    # pointers for A and B tiles
    a_ptrs = a_ptr + batch_id * stride_ab + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    norm_b = weights['norm.bias'].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # load weights and cast to fp16
    left_proj_w = weights['left_proj.weight'].to(device).to(torch.float16)
    right_proj_w = weights['right_proj.weight'].to(device).to(torch.float16)
    left_gate_w = weights['left_gate.weight'].to(device).to(torch.float16)
    right_gate_w = weights['right_gate.weight'].to(device).to(torch.float16)
    out_gate_w = weights['out_gate.weight'].to(device).to(torch.float16)
    to_out_norm_w = weights['to_out_norm.weight'].to(device).to(torch.float16)
    to_out_norm_b = weights['to_out_norm.bias'].to(device).to(torch.float16)

    right_gate = torch.sigmoid(F.linear(x, right_gate_w))

    stride_ak = left_mat.stride(2)

    stride_bn = right_t.stride(2)

        BLOCK_N=BLOCK_N,

"""
Outgoing Triangle Multiplicative Update (TriMul) – forward pass.

The implementation follows the reference PyTorch module but
* uses LayerNorm and Linear layers from torch for stability,
* casts heavy compute to fp16,
* fuses the per‑element mask and gate multiplications in a small
  Triton kernel to reduce memory traffic,
* performs the core N×N×N contraction as a batched GEMM via
  torch.bmm (cuBLAS), which is heavily optimized on modern GPUs,
* finally applies the output LayerNorm, out‑gate and a linear projection.

All heavy arithmetic is performed in fp16; the final result is cast
back to the input dtype (typically float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,           # [B,N,N,H] flattened
    right_ptr,          # same
    left_gate_ptr,      # same
    right_gate_ptr,     # same
    mask_ptr,           # [B,N,N] flattened, broadcast over H
    total_elems,        # total number of elements in left/right tensors
    hidden_dim,         # H
    HAS_MASK: tl.constexpr,   # compile‑time bool
    BLOCK_SIZE: tl.constexpr, # compile‑time int
):
    """
    In‑place fusion:
        left  = left  * left_gate  * (mask if present)
        right = right * right_gate * (mask if present)
    The mask is shared across the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load the four tensors
    left      = tl.load(left_ptr      + offs, mask=active, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden_dim entries share a mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for compute‑heavy parts
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, dtype=torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gates (fp16)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 4. Fuse mask and gate multiplication with a Triton kernel
    # -----------------------------------------------------------------
    use_mask = (mask is not None) and (not nomask)
    if use_mask:
        # Ensure contiguous tensors for the kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp = mask.to(device, dtype=torch.float16).contiguous()

        left_flat      = left.view(-1)
        right_flat     = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat= right_gate.view(-1)
        mask_flat      = mask_fp.view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK_SIZE = 1024

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # Tensors are updated in‑place; no further reshaping needed.
    else:
        # Simple element‑wise multiplication when no mask is present
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ rightᵀ) over the sequence dim
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # left: (B, N, N, H)

    # reshape to (B*H, N, N) – each hidden channel is treated as an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # (B*H, N, N)
    # right needs to be transposed on the last two dimensions before multiplication
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)         # (B*H, N, N)

    # batched matrix multiply using cuBLAS (torch.bmm) – highly optimized
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the original dtype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
