@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_heads, N,
    stride_l_bh, stride_l_i, stride_l_k,
    stride_r_bh, stride_r_i, stride_r_k,
    stride_o_bh, stride_o_i, stride_o_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute indices
    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)
    total_blocks_per_bh = num_blocks_i * num_blocks_j
    bh = pid // total_blocks_per_bh
    rem = pid % total_blocks_per_bh
    i0 = (rem // num_blocks_j) * BLOCK_M
    j0 = (rem % num_blocks_j) * BLOCK_N

    # Determine bounds
    i_max = tl.minimum(i0 + BLOCK_M, N)
    j_max = tl.minimum(j0 + BLOCK_N, N)
    m = i_max - i0
    n = j_max - j0

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k0 in range(0, N, BLOCK_K):
        k_max = tl.minimum(k0 + BLOCK_K, N)
        kk = k_max - k0
        # Load left tile (m x kk)
        left = tl.load(
            left_ptr + bh * stride_l_bh + (i0 + tl.arange(0, BLOCK_M, 1)[:, None]) * stride_l_i + (k0 + tl.arange(0, BLOCK_K, 1)[None, :]) * stride_l_k,
            mask = (tl.arange(0, BLOCK_M)[:, None] < m) & (tl.arange(0, BLOCK_K)[None, :] < kk),
            other = 0.0
        )
        # Load right tile transposed (kk x n)
        right_t = tl.load(
            right_ptr + bh * stride_r_bh + (k0 + tl.arange(0, BLOCK_K, 1)[:, None]) * stride_r_i + (j0 + tl.arange(0, BLOCK_N, 1)[None, :]) * stride_r_k,
            mask = (tl.arange(0, BLOCK_K)[:, None] < kk) & (tl.arange(0, BLOCK_N)[None, :] < n),
            other = 0.0
        )
        # Convert to fp32 for accumulation
        acc += tl.dot(left.to(tl.float32), right_t.to(tl.float32))
    # Store result as fp16 (or fp32)
    out = acc.to(tl.float16)
    # Compute store offsets
    out_ptrs = out_ptr + bh * stride_o_bh + (i0 + tl.arange(0, BLOCK_M)[:, None]) * stride_o_i + (j0 + tl.arange(0, BLOCK_N)[None, :]) * stride_o_k
    tl.store(out_ptrs, out, mask = (tl.arange(0, BLOCK_M)[:, None] < m) & (tl.arange(0, BLOCK_N)[None, :] < n))

    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = torch.nn.functional.linear(x, left_proj_w)  # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply mask and gates
    if mask is not None and not nomask:
        # ensure mask to half
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)

        total = left_flat.numel()
        BLOCK = 1024

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # Core contraction using Triton
    # Permute left & right to (B, H, N, N)
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    B, H, N, _ = left_perm.shape

    # Prepare output
    out_perm = torch.empty_like(left_perm)

    # get strides
    stride_l_bh = left_perm.stride(0)
    stride_l_i = left_perm.stride(1)
    stride_l_k = left_perm.stride(2)
    stride_r_bh = right_perm.stride(0)
    stride_r_i = right_perm.stride(1)
    stride_r_k = right_perm.stride(2)
    stride_o_bh = out_perm.stride(0)
    stride_o_i = out_perm.stride(1)
    stride_o_k = out_perm.stride(2)

    # Determine block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_i = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_j = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B * H * num_blocks_i * num_blocks_j

    grid = (total_blocks,)

    _tri_mul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B * H,
        N,
        stride_l_bh, stride_l_i, stride_l_k,
        stride_r_bh, stride_r_i, stride_r_k,
        stride_o_bh, stride_o_i, stride_o_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to (B,N,N,H)
    out = out_perm.permute(0,2,3,1).contiguous()

    # Output layernorm
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply out_gate
    out = out * out_gate

    # Final linear projection
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to original dtype
    out = out.to(dtype_in)

    return out

"""
TriMul (outgoing) forward pass accelerated with Triton.

Algorithm
---------
1. Layer‑norm the input (float32 for stability) → x.
2. Cast x to FP16.
3. Linear projections (bias‑less) to obtain:
       left, right  (shape [B,N,N,H])
       left_gate, right_gate, out_gate (same shape)
4. Fuse mask (if present) and element‑wise gates in a Triton kernel,
   writing the gated tensors back in‑place.
5. Core O(N³·H) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (left @ rightᵀ) with a custom
   Triton kernel that works on blocks of size 64×64×64 and accumulates
   in FP32.
6. Layer‑norm over the hidden dimension, apply out_gate, and final
   linear projection back to the original dimension.
7. Cast the result back to the input dtype (float32) and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. Kernel that fuses mask and gate multiplication
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # One mask value per (b,i,j) location, broadcast over the hidden dim
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


# ----------------------------------------------------------------------
# 2. Kernel that computes the batched matrix multiplication
#    out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_heads, N,
    stride_l_bh, stride_l_i, stride_l_k,
    stride_r_bh, stride_r_i, stride_r_k,
    stride_o_bh, stride_o_i, stride_o_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # program indexing
    # --------------------------------------------------------------
    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)
    blocks_per_head = num_blocks_i * num_blocks_j
    bh = pid // blocks_per_head                 # which (batch, head)
    block_idx = pid % blocks_per_head
    i0 = (block_idx // num_blocks_j) * BLOCK_M   # start row
    j0 = (block_idx %  num_blocks_j) * BLOCK_N   # start col

    # --------------------------------------------------------------
    # tile indices
    # --------------------------------------------------------------
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # main k‑loop
    # --------------------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left tile:   (M, K)
        left = tl.load(
            left_ptr
            + bh * stride_l_bh
            + i[:, None] * stride_l_i
            + k[None, :] * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile transposed: (K, N)
        # We load right as (k, j) which is the transpose of (j, k)
        right_t = tl.load(
            right_ptr
            + bh * stride_r_bh
            + k[:, None] * stride_r_i
            + j[None, :] * stride_r_k,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # accumulate in FP32
        acc += tl.dot(left.to(tl.float32), right_t.to(tl.float32))

    # --------------------------------------------------------------
    # write result (store as FP16 for the next layers)
    # --------------------------------------------------------------
    out = acc.to(tl.float16)
    out_ptrs = (
        out_ptr
        + bh * stride_o_bh
        + i[:, None] * stride_o_i
        + j[None, :] * stride_o_k
    )
    tl.store(out_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])

# 3. Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of parameter tensors (float32)
            - config: dict with keys
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy path
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (bias‑less) in FP16
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)          # (B,N,N,H)
    right = F.linear(x, right_proj_w)        # (B,N,N,H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 4. Fuse mask and gates (in‑place) via Triton
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16 and contiguous
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # Flatten tensors for the kernel
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction via custom Triton GEMM
    # --------------------------------------------------------------
    # Rearrange to (B, H, N, N) for easier batching
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    out_perm = torch.empty_like(left_perm)

    # Strides (in elements)
    stride_l_bh = left_perm.stride(0)
    stride_l_i  = left_perm.stride(1)
    stride_l_k  = left_perm.stride(2)

    stride_r_bh = right_perm.stride(0)
    stride_r_i  = right_perm.stride(1)
    stride_r_k  = right_perm.stride(2)

    stride_o_bh = out_perm.stride(0)
    stride_o_i  = out_perm.stride(1)
    stride_o_k  = out_perm.stride(2)

    # Block sizes (tuned for typical N ≤ 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_i = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_j = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B * H * num_blocks_i * num_blocks_j

    _tri_mul_kernel[(total_blocks,)](
        left_perm,
        right_perm,
        out_perm,
        B * H,
        N,
        stride_l_bh, stride_l_i, stride_l_k,
        stride_r_bh, stride_r_i, stride_r_k,
        stride_o_bh, stride_o_i, stride_o_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Back to (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating, and final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (FP16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in FP16

    # --------------------------------------------------------------
    # 7. Cast back to original dtype and return
    # --------------------------------------------------------------
    out = out.to(dtype_in)
    return out
