@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_Ab, stride_Ai, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Ci, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr = 3,
    ):
   batch_id = tl.program_id(1)
   pid = tl.program_id(0)
   num_pid_m = tl.cdiv(M, BLOCK_M)
   num_pid_n = tl.cdiv(N, BLOCK_N)
   pid_m = pid // num_pid_n
   pid_n = pid % num_pid_n

   # Compute block start offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   # Mask for out-of-bounds
   mask_m = offs_m < M
   mask_n = offs_n < N

   a_ptrs = a_ptr + batch_id * stride_Ab + offs_m[:, None] * stride_Ai + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
   b_ptrs = b_ptr + batch_id * stride_Bb + tl.arange(0, BLOCK_K)[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0,BLOCK_K) < K - k*BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0,BLOCK_K) < K - k*BLOCK_K)[:, None] & mask_n[None,:], other=0.0)
        # dot: using tl.dot
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

   # Store
   c_ptrs = c_ptr + batch_id * stride_Cb + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cn
   tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input (float32 for stability) and cast to fp16.
2. Linear projections (left / right) and three sigmoid gates (left, right, out)
   are applied in fp16.
3. Apply the optional mask and fuse it with the left/right gates.
4. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix multiplication L @ Rᵀ.  A custom Triton kernel
   performs the batched GEMM for all (batch * hidden_dim) slices in a single
   launch, accumulating in FP32 for accuracy.
5. A second LayerNorm on the hidden dimension, multiplication by the output
   gate, and a final linear projection back to the original dimension.
6. Cast the result back to the input dtype.

Only the heavy matrix multiplication (step 4) is written in Triton; the
remaining operations use PyTorch which is already highly optimized.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel for batched matmul: C = A @ B, where A shape (B, M, K),
# B shape (B, K, N) and C shape (B, M, N).  Accumulates in FP32, stores FP16.
# ----------------------------------------------------------------------------- #
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions
    stride_ab, stride_am, stride_ak,   # stride of A: batch, M, K
    stride_bb, stride_bk, stride_bn,   # stride of B: batch, K, N
    stride_cb, stride_cm, stride_cn,   # stride of C: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel.

    Arguments
    ---------
    a_ptr, b_ptr, c_ptr : pointers to fp16 tensors.
    M, N, K : matrix sizes.
    stride_* : strides for each dimension (in elements).
    BLOCK_* : compile‑time tile sizes.
    """
    batch_id = tl.program_id(1)                # batch index
    pid = tl.program_id(0)                     # tile index in M×N grid

    # -------------------------------------------------------------
    # Compute tile coordinates.
    # -------------------------------------------------------------
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # Block start offsets.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # -------------------------------------------------------------
    # Accumulator.
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Loop over K dimension in tiles.
    # -------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for the current tile.
        a_tile_ptr = (
            a_ptr
            + batch_id * stride_ab
            + (offs_m[:, None] * stride_am)
            + (offs_k[None, :] * stride_ak)
        )
        b_tile_ptr = (
            b_ptr
            + batch_id * stride_bb
            + (offs_k[:, None] * stride_bk)
            + (offs_n[None, :] * stride_bn)
        )

        # Load with masking.
        a = tl.load(a_tile_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_tile_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Dot‑product accumulate.
        acc += tl.dot(a, b)

    # -------------------------------------------------------------
    # Write back the result.
    # -------------------------------------------------------------
    c_tile_ptr = (
        c_ptr
        + batch_id * stride_cb
        + (offs_m[:, None] * stride_cm)
        + (offs_n[None, :] * stride_cn)
    )
    tl.store(
        c_tile_ptr,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )

    """

    dtype_in = input_tensor.dtype

    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------------- #
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * left_gate * mask_fp
        right = right * right_gate * mask_fp
    else:
        left = left * left_gate
        right = right * right_gate

    # ----------------------------------------------------------------- #
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM (L @ Rᵀ) using a Triton kernel.
    # ----------------------------------------------------------------- #
    B, N, _, H = left.shape                         # B, seq_len, seq_len, hidden_dim
    # reshape to (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # transpose right for the product L @ Rᵀ
    right_mat_t = right_mat.permute(0, 2, 1).contiguous()

    # allocate output
    out_mat = torch.empty_like(left_mat)

    # strides (in elements)
    stride_ab = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_bb = right_mat_t.stride(0)
    stride_bk = right_mat_t.stride(1)
    stride_bn = right_mat_t.stride(2)

    stride_cb = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # tile sizes – chosen to give good occupancy on B200 (A100‑class)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (num_tiles_m * num_tiles_n, B * H)   # (tiles, batch)

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat_t,
        out_mat,
        N, N, N,                     # M, N, K (all equal to seq_len)
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cb, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------- #
    # 6. Output LayerNorm, output gate, final linear projection.
    # ----------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate                       # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)              # (B, N, N, C)

    # ----------------------------------------------------------------- #
    # 7. Cast back to original dtype.
    # ----------------------------------------------------------------- #
    out = out.to(dtype_in)
    return out
