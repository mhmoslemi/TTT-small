def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device
    dtype_in = x.dtype

    B, N, _, C = x.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = F.layer_norm(x, (C,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # cast to fp16
    x = x.to(torch.float16)

    # Linear projections and gates in fp16
    # Use weight matrices (dim, hidden) shape? In torch.nn.Linear weight shape is (out_features, in_features).
    # left_proj_weight shape: (hidden_dim, dim) from weights dict.
    left = F.linear(x, weights["left_proj.weight"], bias=None)  # B,N,N,H
    right = F.linear(x, weights["right_proj.weight"], bias=None)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"], bias=None))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"], bias=None))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"], bias=None))

    if mask is not None and not nomask:
        # broadcast mask to (B,N,N,1) and cast to fp16
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # shape B,N,N,1
        left = left * left_gate * mask_fp
        right = right * right_gate * mask_fp
    else:
        left = left * left_gate
        right = right * right_gate

    # Core contraction: left @ right^T per batch and hidden channel
    # reshape to (B*H, N, N)
    left_mat = left.permute(0,3,1,2).reshape(B*H, N, N)
    right_mat = right.permute(0,3,1,2).reshape(B*H, N, N)

    out_mat = torch.matmul(left_mat, right_mat.transpose(-2, -1))
    out = out_mat.reshape(B, H, N, N).permute(0,2,3,1)

    # to_out_norm
    out = F.layer_norm(out, (H,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"], bias=None)

    # cast back to original dtype
    if dtype_in != out.dtype:
        out = out.to(dtype_in)

    return out
```

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input tensor (float32 for stability).
2. Cast to fp16 for compute‑heavy parts.
3. Compute left/right projections and three sigmoid gates with a single
   linear + torch.sigmoid per gate.
4. Fuse mask (if present) and the left/right gates in‑place with a tiny
   Triton kernel, avoiding extra Python launches.
5. Perform the core O(N³·H) contraction as a batched matrix multiply:
          out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   by reshaping to (B·H, N, N) and calling torch.matmul (cuBLAS tensor‑cores).
6. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate and linearly project back to the original dimension.
7. Cast the result back to the input dtype.

The kernel only implements the mask‑gate fusion; the heavy linear
projections, matmul and layer‑norms are left to PyTorch’s highly‑optimised
CUDA kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,            # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,  # same shape
    mask_ptr,                       # [B,N,N] flattened
    total_elems, hidden_dim,        # runtime scalars
    HAS_MASK: tl.constexpr,         # compile‑time bool
    BLOCK_SIZE: tl.constexpr,       # compile‑time int
):
    """In‑place fusion of:
         left  = left  * left_gate  * mask
         right = right * right_gate * mask
       mask is broadcast over the hidden dimension H.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # load values
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    if HAS_MASK:
        # each hidden slice shares the same mask entry
        mask_idx = offs // hidden_dim          # index in [B,N,N]
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # store results back in‑place
    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy part
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and sigmoid gates (all fp16)
    # -----------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w)                     # (B,N,N,H)
    right = F.linear(x, right_proj_w)                   # (B,N,N,H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B,N,N,H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,H)

    # -----------------------------------------------------------------
    # 4. Fuse mask and gates (if a mask is supplied)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape (B,N,N) -> (B,N,N,1) and fp16
        mask_fp = mask.to(torch.float16).unsqueeze(-1).contiguous()

        # flatten tensors for the kernel
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK_SIZE = 1024  # must be <= 1024 on CUDA
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – simple element‑wise fusion
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched matmul left @ rightᵀ
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_mat = torch.matmul(left_mat, right_mat.transpose(-2, -1))  # (B*H, N, N)

    # restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating, and final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (still fp16)

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)  # (B,N,N,dim) in fp16

    # -----------------------------------------------------------------
    # 7. Cast back to original dtype and return
    # -----------------------------------------------------------------
    out = out.to(dtype_in)
    return out
