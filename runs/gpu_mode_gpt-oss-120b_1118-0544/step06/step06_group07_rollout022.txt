def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype
    
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    
    # 1. LayerNorm
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    
    # Cast to fp16 for compute heavy.
    x = x.to(torch.float16)
    
    # 2. Linear projections (no bias)
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    
    # Apply linear projections
    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)
    left_gate_lin = torch.nn.functional.linear(x, left_gate_w)
    right_gate_lin = torch.nn.functional.linear(x, right_gate_w)
    out_gate_lin = torch.nn.functional.linear(x, out_gate_w)
    
    # 3. Apply gates and mask via Triton kernel
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate_lin.contiguous().view(-1)
        right_gate_flat = right_gate_lin.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)
        total = left_flat.numel()
        BLOCK = 16384   # choose large block for better occupancy
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
        _gate_mask_sigmoid_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask, just apply sigmoid gates
        left_gate = torch.sigmoid(left_gate_lin)
        right_gate = torch.sigmoid(right_gate_lin)
        left = left * left_gate
        right = right * right_gate
    
    # Compute out_gate (sigmoid)
    out_gate = torch.sigmoid(out_gate_lin)   # (B,N,N,H)
    
    # 4. Core contraction: batched GEMM for each hidden dim
    B, N, _, H = left.shape
    left_mat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_mat = right.permute(0,3,1,2).contiguous().view(B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))  # (B*H,N,N)
    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()
    
    # 5. Output layer norm
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    
    # 6. Apply out_gate and final linear

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)
    
    # 7. Cast back to original dtype
    out = out.to(dtype_in)

@triton.jit
def _gate_mask_sigmoid_kernel(left_ptr, right_ptr,
                              left_gate_ptr, right_gate_ptr,
                              mask_ptr,
                              total_elems, hidden_dim,
                              HAS_MASK: tl.constexpr,
                              BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # Load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate_l = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate_l = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    # Sigmoid
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_l))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_l))
    if HAS_MASK:
        idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + idx, mask=mask, other=1.0)  # broadcast
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate
    # Store
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm (float32) the input tensor [B,N,N,C].
2. Cast to FP16 for the compute‑heavy path.
3. Linear projections (bias‑less) to obtain:
       left, right, left_gate_lin, right_gate_lin, out_gate_lin
   each of shape (B,N,N,hidden_dim).
4. Fuse sigmoid‑gating and optional masking into a single Triton kernel,
   writing the gated tensors back in‑place (left, right).
5. Core contraction: for each hidden dimension h perform a batched GEMM
   `out_h = left_h @ right_hᵀ`.  Implemented with torch.bmm on the
   reshaped tensors (leveraging cuBLAS Tensor‑Cores).
6. Layer‑norm over the hidden dimension, multiply by the sigmoid‑gated
   out_gate, and a final linear projection back to the original dimension.
7. Cast back to the input dtype and return.

Only the gating‑and‑mask step is implemented in Triton; all other
operations use highly‑optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_sigmoid_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape (pre‑sigmoid)
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    """
    In‑place gating:
        left  = left  * sigmoid(left_gate_lin)  * mask
        right = right * sigmoid(right_gate_lin) * mask
    If HAS_MASK=False the mask term is omitted (treated as 1.0).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load tensors (fp16)
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate_lin = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate_lin = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    # Sigmoid (1/(1+exp(-x))) in FP16
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_lin))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_lin))

    if HAS_MASK:
        # Each mask entry corresponds to `hidden_dim` consecutive elements.
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store the gated tensors back in‑place.
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of parameter tensors (float32)
            - config: dict with keys
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # --------------------------------------------------------------
    # 2. Cast to fp16 for the compute‑heavy path
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (bias‑less) in fp16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left      = F.linear(x, left_proj_w)      # (B,N,N,hidden_dim)
    right     = F.linear(x, right_proj_w)     # (B,N,N,hidden_dim)
    left_gate_lin  = F.linear(x, left_gate_w)   # (B,N,N,hidden_dim)
    right_gate_lin = F.linear(x, right_gate_w)  # (B,N,N,hidden_dim)
    out_gate_lin   = F.linear(x, out_gate_w)    # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 4. Fuse sigmoid‑gating and mask via Triton
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is contiguous and fp16
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # Flatten tensors for the kernel
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate_lin.contiguous().view(-1)
        right_gate_flat= right_gate_lin.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 8192  # reasonable block size for fp16 workloads
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_sigmoid_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back to original layout
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – apply sigmoid gates directly
        left_gate  = torch.sigmoid(left_gate_lin)
        right_gate = torch.sigmoid(right_gate_lin)
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Output gate (sigmoid) – needed after the contraction
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_lin)   # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 6. Core contraction via batched GEMM (L @ Rᵀ)
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Reshape to (B*H, N, N) for a batch of GEMMs
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (uses cuBLAS/Tensor Cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Output LayerNorm, gating, and final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # --------------------------------------------------------------
    # 8. Cast back to original dtype and return
    # --------------------------------------------------------------
    out = out.to(dtype_in)
    return out
