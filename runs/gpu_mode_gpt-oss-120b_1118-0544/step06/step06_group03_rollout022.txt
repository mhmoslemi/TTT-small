@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, H,
    stride_left_m, stride_left_i, stride_left_j,
    stride_right_m, stride_right_i, stride_right_j,
    stride_out_m, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_i = tl.program_id(0)  # block i index
    pid_j = tl.program_id(1)  # block j index
    pid_m = tl.program_id(2)  # batch*head index

    # compute global offsets
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    m = pid_m

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K

        # compute offsets for loading A block
        a_ptrs = left_ptr + m * stride_left_m + (i_base + tl.arange(0, BLOCK_M)) * stride_left_i + (k_base + tl.arange(0, BLOCK_K)) * stride_left_j
        # shape (BLOCK_M, BLOCK_K)
        a = tl.load(a_ptrs, mask=(i_base + tl.arange(0, BLOCK_M) < N)[:, None] & (k_base + tl.arange(0, BLOCK_K) < N)[None, :], other=0.0)

        # compute offsets for B block (transposed)
        # we need B as (BLOCK_K, BLOCK_N) where indices (k,j)
        b_ptrs = right_ptr + m * stride_right_m + (k_base + tl.arange(0, BLOCK_K)) * stride_right_i + (j_base + tl.arange(0, BLOCK_N)) * stride_right_j
        b = tl.load(b_ptrs, mask=(k_base + tl.arange(0, BLOCK_K) < N)[None, :] & (j_base + tl.arange(0, BLOCK_N) < N)[:, None], other=0.0)

        # dot product
        # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)

    # write out
    out_ptrs = out_ptr + m * stride_out_m + (i_base + tl.arange(0, BLOCK_M)) * stride_out_i + (j_base + tl.arange(0, BLOCK_N)) * stride_out_j
    tl.store(out_ptrs, acc, mask=(i_base + tl.arange(0, BLOCK_M) < N)[:, None] & (j_base + tl.arange(0, BLOCK_N) < N)[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    # retrieve dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm (float32)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # cast to fp16
    x = x.to(torch.float16)

    # linear projections: left_proj = x @ left_proj_weight.T
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16))

    # gating: produce sigmoid
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # fuse mask/gate/projection
    if not nomask and mask is not None:
        mask_fp = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Now we have left/right shape [B,N,N,H] in fp16

    B, N, _, H = left.shape

    # Permute and reshape to (M,N,N) where M = B*H
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    M = B * H
    left_flat = left_perm.view(M, N, N)
    right_flat = right_perm.view(M, N, N)

    # allocate output tensor
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)  # same dtype fp16

    # compute block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), M)

    # stride values
    stride_left_m, stride_left_i, stride_left_j = left_flat.stride()
    stride_right_m, stride_right_i, stride_right_j = right_flat.stride()
    stride_out_m, stride_out_i, stride_out_j = out_flat.stride()

    # Call kernel
    _tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M,
        N,
        H,
        stride_left_m, stride_left_i, stride_left_j,
        stride_right_m, stride_right_i, stride_right_j,
        stride_out_m, stride_out_i, stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm (float32) over hidden_dim

    # Multiply by out_gate (which is shape [B,N,N,H])

    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast back to original dtype (float32)
    out = out.to(input_tensor.dtype)

    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (float32 for stability).
2. Cast to float16 for compute‑heavy ops.
3. Linear projections → left/right tensors of shape [B, N, N, H].
4. Compute gating logits → sigmoid; fuse projection, gate and optional mask
   in a Triton kernel (element‑wise multiplication with broadcasted mask).
5. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM using a custom Triton kernel that operates
   on the flattened tensor (M = B*H, N, N).  The kernel tiles the matrices in
   64‑x‑64 blocks and accumulates in float32.
6. Output layer‑norm over the hidden dimension, element‑wise multiplication
   with the output gate, and a final linear projection back to the original
   dimension.
7. Cast the result back to the input dtype (usually float32).

Only the mask‑gate fusion and the batched matrix multiplication are written
in Triton; all other ops use highly‑optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and (optional) mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,            # fp16 * (B*N*N*H,)
    gate_ptr,            # fp16 * (B*N*N*H,)
    mask_ptr,            # fp16 * (B*N*N,)
    out_ptr,             # fp16 * (B*N*N*H,)
    total_elems,         # int: B*N*N*H
    hidden_dim,          # int: H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # bounds
    mask = offs < total_elems

    # mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # 1.0 when mask is absent

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 2048) -> torch.Tensor:
    """
    Fuse ``proj * gate * mask`` using a Triton kernel.
    All tensors must be ``float16`` and contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)   (broadcast over H)
    Returns tensor of shape (B, N, N, H).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,               # M = B*H, N = seq_len
    stride_left_m, stride_left_i, stride_left_j,
    stride_right_m, stride_right_i, stride_right_j,
    stride_out_m, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out = left @ right^T for each batch*head (M) slice.
    left/right/out are (M, N, N) in row‑major layout.
    """
    pid_i = tl.program_id(0)   # block row index
    pid_j = tl.program_id(1)   # block column index
    pid_m = tl.program_id(2)   # batch*head index

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    m = pid_m

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K

        # --------------------------------------------------------------
        # Load A block: shape (BLOCK_M, BLOCK_K)
        # --------------------------------------------------------------
        a_ptrs = left_ptr + \
                 m * stride_left_m + \
                 (i_start + tl.arange(0, BLOCK_M)) * stride_left_i + \
                 (k_start + tl.arange(0, BLOCK_K)) * stride_left_j

        a = tl.load(
            a_ptrs,
            mask=(i_start + tl.arange(0, BLOCK_M) < N)[:, None] &
                 (k_start + tl.arange(0, BLOCK_K) < N)[None, :],
            other=0.0,
        )

        # --------------------------------------------------------------
        # Load B block transposed (K, N): shape (BLOCK_K, BLOCK_N)
        # We need right[m, k, j] -> treat k as first dimension.
        # --------------------------------------------------------------
        b_ptrs = right_ptr + \
                 m * stride_right_m + \
                 (k_start + tl.arange(0, BLOCK_K)) * stride_right_i + \
                 (j_start + tl.arange(0, BLOCK_N)) * stride_right_j

        b = tl.load(
            b_ptrs,
            mask=(k_start + tl.arange(0, BLOCK_K) < N)[None, :] &
                 (j_start + tl.arange(0, BLOCK_N) < N)[:, None],
            other=0.0,
        )

        # dot product (accumulates in fp32)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Store result block
    # --------------------------------------------------------------
    out_ptrs = out_ptr + \
               m * stride_out_m + \
               (i_start + tl.arange(0, BLOCK_M)) * stride_out_i + \
               (j_start + tl.arange(0, BLOCK_N)) * stride_out_j

    tl.store(
        out_ptrs,
        acc,
        mask=(i_start + tl.arange(0, BLOCK_M) < N)[:, None] &
             (j_start + tl.arange(0, BLOCK_N) < N)[None, :],
    )

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (bool/float)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to float16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias) in float16
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4. Gate projections + sigmoid (float16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse mask (if present) with projection and gate
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # ensure mask is contiguous float16 (0/1)
        mask_fp = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core reduction: batched GEMM (left @ right^T)
    #    Shapes: left/right -> [B, N, N, H]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # Rearrange to (M, N, N) where M = B * H
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    M = B * H
    left_flat = left_perm.view(M, N, N)
    right_flat = right_perm.view(M, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Block sizes (tuned for typical GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # i blocks
        triton.cdiv(N, BLOCK_N),   # j blocks
        M,                         # batch*head dimension
    )

    # Strides for the flat tensors
    stride_l_m, stride_l_i, stride_l_j = left_flat.stride()
    stride_r_m, stride_r_i, stride_r_j = right_flat.stride()
    stride_o_m, stride_o_i, stride_o_j = out_flat.stride()

    _tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M,
        N,
        stride_l_m, stride_l_i, stride_l_j,
        stride_r_m, stride_r_i, stride_r_j,
        stride_o_m, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm over hidden dimension, apply out_gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 8. Cast back to original dtype (typically float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
