"""
custom_kernel: Triton‑accelerated outgoing TriMul.

Algorithm
---------
1. LayerNorm over the channel dimension (fp16) using torch.nn.functional.layer_norm.
2. Linear projections (left, right) and gated projections (left_gate, right_gate, out_gate) in fp16.
3. Fuse mask, left_gate, right_gate into left/right in‑place via a Triton kernel.
   If no mask, only gate fusion is performed.
4. Core contraction: for each hidden dimension compute left @ rightᵀ using batched FP16 GEMM
   (torch.bmm on a reshaped (B*H, N, N) tensor).
5. Output LayerNorm over hidden dimension, apply out_gate, and final linear projection back to original dim.
6. Cast result back to the input dtype (float32).

All heavy arithmetic is performed in float16 to exploit tensor‑cores.
A small Triton kernel is used for the elementwise gating+mask fusion.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_fuse_kernel(
    left_ptr, right_ptr,          # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,# same shape
    mask_ptr,                     # [B,N,N] flattened, broadcast over H
    total, hidden_dim,            # runtime scalars
    HAS_MASK: tl.constexpr,       # compile‑time bool
    BLOCK_SIZE: tl.constexpr,     # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # load data
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # mask index repeats every hidden_dim elements
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim" (C), "hidden_dim" (H), optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. LayerNorm (fp16) over channel dimension
    # -----------------------------------------------------------------
    # Convert input to fp16
    x = input_tensor.to(torch.float16)

    # LayerNorm weights/bias to fp16
    norm_w = weights["norm.weight"].to(device).to(torch.float16)
    norm_b = weights["norm.bias"].to(device).to(torch.float16)

    # torch.nn.functional.layer_norm works in fp16 (computes stats in fp32 internally)
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2. Linear projections & gates (no bias)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    # Projections
    left  = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    # Gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 3. Fuse mask and gates via Triton (in‑place)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # Flatten tensors for the kernel
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)  # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – simple element‑wise multiplication (in‑place)
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 4. Core contraction: batched GEMM (left @ rightᵀ) over hidden dimension
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiplication (FP16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5. Output LayerNorm, out_gate, final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply output gate (in‑place)
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # -----------------------------------------------------------------
    # 6. Cast back to original dtype
    # -----------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5. Output LayerNorm, out_gate, and final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # -----------------------------------------------------------------
    # 6. Cast back to original dtype and return
    # -----------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
