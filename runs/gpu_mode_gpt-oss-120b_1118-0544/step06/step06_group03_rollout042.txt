@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute start pointers for this block
    a_ptrs = a_ptr + batch_id * batch_stride_a + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * batch_stride_b + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        # Load tiles
        a = tl.load(a_ptrs, mask=offs_m < M, other=0.0) # shape (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=offs_n < N, other=0.0) # shape (BLOCK_K, BLOCK_N)
        # Compute dot product
        acc += tl.dot(a, b)

        # Increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back C
    c_ptrs = c_ptr + batch_id * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m < M) & (offs_n < N))

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_gemm_kernel(...):
    ...

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # get dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # cast to fp16
    x = x.to(torch.float16)

    # cast weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # compute projections
    left_proj = F.linear(x, left_proj_w)   # shape (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # apply mask if needed
    if mask is not None and not nomask:
        mask = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # core reduction using custom kernel
    B, N, _, H = left.shape
    # permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    total_batches = B * H
    # flatten to (total_batches, N, N)
    left_flat = left_perm.view(total_batches, N, N)
    right_flat = right_perm.view(total_batches, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # strides
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)
    stride_bk = right_flat.stride(2)  # col stride of original (=> 1)
    stride_bn = right_flat.stride(1)  # row stride of original (=> N)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)
    batch_stride_a = left_flat.stride(0)
    batch_stride_b = right_flat.stride(0)
    batch_stride_c = out_flat.stride(0)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (total_batches,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # output layernorm
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # out gating
    out = out * out_gate

    # final projection
    out = F.linear(out, to_out_w)  # (B,N,N,dim)

    # cast back to original dtype
    out = out.to(input_tensor.dtype)

    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated core reduction.

Algorithm
---------
1. LayerNorm over the channel dimension `dim`.
2. Cast to FP16 for all compute‑intensive operations.
3. Linear projections → left_proj / right_proj (shape [B,N,N,H]).
4. Gate projections + sigmoid → left_gate / right_gate / out_gate.
5. Fuse projection, gate and optional mask with simple element‑wise ops.
6. Core reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM of (N×N) @ (N×N)^T for each (b,h) pair.
   Implemented with a custom Triton kernel that processes many batches
   (B·H) in parallel and uses FP16 tensor cores (accumulation in FP32).
7. Output LayerNorm over the hidden dimension H, multiply by `out_gate`,
   and a final linear projection back to the original dimension `dim`.
8. Cast back to the input dtype (usually float32).

The function `custom_kernel` follows the exact semantics of the reference
PyTorch module but replaces the O(N³) reduction with a highly‑optimized
Triton kernel.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,        # pointers to fp16 tensors
    M, N, K,                    # matrix sizes (all equal to seq_len)
    stride_am, stride_ak,       # strides for A (row, col)
    stride_bk, stride_bn,       # strides for B (row, col) – B is accessed transposed
    stride_cm, stride_cn,       # strides for C (row, col)
    batch_stride_a, batch_stride_b, batch_stride_c,  # strides between batches
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B^T for a batch of matrices.
    A : (M, K)  fp16
    B : (N, K)  fp16   (original B of shape (N, K); we read it as transposed)
    C : (M, N)  fp16   (accumulated in fp32)

    Each program instance handles one batch (B·H) and a tile of C.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch
    a_batch = a_ptr + batch_id * batch_stride_a
    b_batch = b_ptr + batch_id * batch_stride_b
    c_batch = c_ptr + batch_id * batch_stride_c

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)

        # Pointers to the current tiles of A and B (B is accessed transposed)
        a_tile_ptr = a_batch + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        b_tile_ptr = b_batch + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (cur_k[None, :] < K)
        mask_b = (cur_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_tile_ptr, mask=mask_a, other=0.0)
        b = tl.load(b_tile_ptr, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # Store the result
    c_tile_ptr = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_tile_ptr, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (bool/float)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim` (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast linear weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6. Apply mask & gating (simple elementwise ops; mask broadcast over H)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left_proj * left_gate * mask_fp
        right = right_proj * right_gate * mask_fp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Release intermediate tensors early to reduce peak memory
    del left_proj, right_proj, left_gate, right_gate
    torch.cuda.empty_cache()

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched A @ B^T where A,B ∈ ℝ^{N×N}
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # (B, N, N, H) → (B, H, N, N) → (B·H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    total_batches = B * H

    left_flat = left_perm.view(total_batches, N, N)
    right_flat = right_perm.view(total_batches, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # Strides (all tensors are contiguous, row‑major)
    stride_am = left_flat.stride(1)        # = N
    stride_ak = left_flat.stride(2)        # = 1
    stride_bk = right_flat.stride(2)       # = 1 (col stride of original)
    stride_bn = right_flat.stride(1)       # = N (row stride of original)
    stride_cm = out_flat.stride(1)         # = N
    stride_cn = out_flat.stride(2)         # = 1
    batch_stride_a = left_flat.stride(0)   # = N*N
    batch_stride_b = right_flat.stride(0)  # = N*N
    batch_stride_c = out_flat.stride(0)    # = N*N

    # Tiling parameters – tuned for typical B200 / FP16 Tensor Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)          # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
