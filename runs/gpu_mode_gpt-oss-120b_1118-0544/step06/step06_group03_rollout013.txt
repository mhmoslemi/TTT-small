import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fuse_proj_gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                               total_elems, hidden_dim,
                               BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # mask broadcast: each hidden_dim elements share one mask element
    mask_idx = offs // hidden_dim
    p = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    out = p * g * m
    tl.store(out_ptr + offs, out, mask=mask)

def apply_fused_proj_gate_mask(proj, gate, mask, block_size=1024):
    # proj, gate: (B,N,N,H) float16 contiguous
    # mask: (B,N,N) float16 contiguous
    B,N,_,H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    fuse_proj_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out

def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.
    Implements:
        1. LayerNorm over input channel dim.
        2. Linear projections + sigmoid gates.
        3. Optional mask‑gate fusion via a Triton kernel.
        4. Batched GEMM reduction (Σ_k left * right) using torch.bmm.
        5. Output LayerNorm, out‑gate, final linear projection.
    Returns tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # 1. Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # 2. Cast to FP16 for compute‑intensive part
    x = x.to(torch.float16)

    # 3. Cast linear weights to FP16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # 4. Linear projections (B,N,N,hidden_dim)
    # torch.nn.functional.linear expects weight (out_features, in_features)
    left_proj = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # 5. Gates (sigmoid)
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # 6. Apply mask + gating (fused)
    if mask is not None and not nomask:
        # Ensure mask dtype matches and is contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_fused_proj_gate_mask(left_proj, left_gate, mask_fp16)
        right = apply_fused_proj_gate_mask(right_proj, right_gate, mask_fp16)
    else:
        # No mask: simple elementwise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 7. Core reduction: batched matmul over the sequence dimension
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Compute (B*H, N, N) @ (B*H, N, N).T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 8. Output LayerNorm, out‑gate, final linear projection
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # 9. Cast back to original dtype
    out = out.to(dtype_in)

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _proj_gate_mask_fuse_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # int: B*N*N*H
    hidden_dim,        # int: H
    BLOCK_SIZE: tl.constexpr,
):
    """
    out = proj * gate * mask   (mask broadcast over hidden_dim)
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # bounds
    mask = offs < total_elems

    # mask element index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    p = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = p * g * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_proj_gate_mask(proj: torch.Tensor,
                          gate: torch.Tensor,
                          mask: torch.Tensor,
                          block_size: int = 1024) -> torch.Tensor:
    """
    Fuse projection, gate and mask using Triton.
    proj, gate : (B, N, N, H)  fp16, contiguous
    mask       : (B, N, N)     fp16, contiguous
    Returns: (B, N, N, H) fp16 tensor = proj * gate * mask
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],          # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float32/float16/bool) or None
            - weights: dict of model parameters (float32)
            - config: dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    dtype_in = input_tensor.dtype
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast linear weights to fp16 (biases are None for all but norms)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w).contiguous()
    right_proj = F.linear(x, right_proj_w).contiguous()

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid (still fp16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid().contiguous()
    right_gate = F.linear(x, right_gate_w).sigmoid().contiguous()
    out_gate = F.linear(x, out_gate_w).sigmoid().contiguous()

    # ------------------------------------------------------------------
    # 6. Fuse mask, projection and gate (Triton kernel)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_proj_gate_mask(left_proj, left_gate, mask_fp16)
        right = _apply_proj_gate_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
