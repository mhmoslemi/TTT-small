def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loops over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # compute pointers for A and B sub-blocks
        a_ptrs = a_ptr + batch_id * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch_id * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

```
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers
        a_ptrs = a_ptr + batch_id * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch_id * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # mask for loads
        mask_a = mask_m[:, None] & mask_k[None, :]
        mask_b = mask_k[:, None] & mask_n[None, :]

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_ptr + batch_id * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)
```

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the reference PyTorch module:
1. Layer‑normalize the input (float32 for stability).
2. Linear projections to a hidden dimension `hidden_dim` (float16).
3. Compute three sigmoid gates.
4. Fuse mask, left‑gate and right‑gate in a tiny Triton kernel
   (left_out = left * mask * left_gate,
    right_out = right * mask * right_gate).
5. Core reduction via a batched matrix multiplication:
   out[b,i,j,d] = Σ_k left_out[b,i,k,d] * right_out[b,j,k,d]
   implemented with a custom Triton GEMM kernel that treats the
   hidden dimension as part of the batch axis, avoiding costly
   permutes/copies.
6. Apply a second LayerNorm over the hidden dimension.
7. Multiply by the output gate and linearly project back to the
   original channel dimension.
All heavy arithmetic is done in float16; the final result is cast back
to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # const * left   [B,N,N,H]
    right_ptr,         # const * right  [B,N,N,H]
    mask_ptr,          # const * mask   [B,N,N] (flattened)
    left_gate_ptr,     # const * left_gate
    right_gate_ptr,    # const * right_gate
    left_out_ptr,      # * left_out
    right_out_ptr,     # * right_out
    H: tl.constexpr,               # hidden dimension (compile‑time constant)
    total_elements: tl.int32,      # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask and sigmoid gates:
       left_out  = left  * mask * left_gate
       right_out = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load data (masked)
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # mask repeats for each hidden channel → divide by H
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Apply mask and gates
    tl.store(left_out_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (M, K)
    stride_bk, stride_bn,          # B: (K, N)
    stride_cm, stride_cn,          # C: (M, N)
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: for each batch `b`,
       C[b] = A[b] @ B[b]   with A shape (M,K), B shape (K,N).
       All tensors are in fp16; accumulation is fp32.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets of the block in output matrix C
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to sub‑tiles of A and B
        a_ptrs = a_ptr + batch_id * batch_stride_a + \
                 offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch_id * batch_stride_b + \
                 offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load with bounds checking
        mask_a = mask_m[:, None] & mask_k[None, :]
        mask_b = mask_k[:, None] & mask_n[None, :]
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)   # fp16 → fp32 accumulation

    # Write result back to C
    c_ptrs = c_ptr + batch_id * batch_stride_c + \
             offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters
            - config: dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – bool, if True ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to float16 for the heavy computation
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections (to hidden dimension)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(device=device, dtype=torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(device=device, dtype=torch.float16))

    # ------------------------------------------------------------------
    # 3) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(device=device, dtype=torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(device=device, dtype=torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(device=device, dtype=torch.float16))
    )

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 4) Mask handling & fusion of mask + gates (Triton)
    # ------------------------------------------------------------------
    if mask is None or nomask:
        # No mask → simple elementwise multiplication
        left_out = left * left_gate
        right_out = right * right_gate
    else:
        # Prepare mask tensor (float16) and flatten it
        mask_fp = mask.to(dtype=torch.float16, device=device)
        mask_flat = mask_fp.view(-1)                     # shape B*N*N

        # Allocate output buffers
        left_out = torch.empty_like(left)
        right_out = torch.empty_like(right)

        total_elements = left.numel()                    # B * N * N * H
        BLOCK = 65536                                     # tuneable block size

        grid = (triton.cdiv(total_elements, BLOCK),)
        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            mask_ptr=mask_flat,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            left_out_ptr=left_out,
            right_out_ptr=right_out,
            H=hidden_dim,
            total_elements=total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )

    # ------------------------------------------------------------------
    # 5) Core reduction: batched GEMM (left_out @ right_outᵀ)
    #    We treat each (batch, hidden_dim) pair as a separate batch.
    # ------------------------------------------------------------------
    # Right matrix needs to be transposed on its last two axes.
    right_T = right_out.permute(0, 3, 2, 1).contiguous()   # (B, H, N, N)

    # Reshape to (B*H, N, N) without extra copies for left_out
    left_mat = left_out.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    left_mat = left_mat.view(B * hidden_dim, N, N)
    right_mat = right_T.view(B * hidden_dim, N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat)

    # GEMM block sizes – chosen to fit in shared memory and achieve good occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_M = triton.cdiv(N, BLOCK_M)
    num_blocks_N = triton.cdiv(N, BLOCK_N)

    grid = (B * hidden_dim, num_blocks_M, num_blocks_N)

    _batched_gemm_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_am=left_mat.stride(1),   # stride for M dimension (row) = K
        stride_ak=left_mat.stride(2),   # stride for K dimension (col) = 1
        stride_bk=right_mat.stride(1),  # stride for K dimension
        stride_bn=right_mat.stride(2),  # stride for N dimension
        stride_cm=out_mat.stride(1),
        stride_cn=out_mat.stride(2),
        batch_stride_a=left_mat.stride(0),
        batch_stride_b=right_mat.stride(0),
        batch_stride_c=out_mat.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float16),
    )

    # ------------------------------------------------------------------
    # 7) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].to(device=device, dtype=torch.float16)
    )

    # ------------------------------------------------------------------
    # 9) Cast back to the original dtype of the input
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
