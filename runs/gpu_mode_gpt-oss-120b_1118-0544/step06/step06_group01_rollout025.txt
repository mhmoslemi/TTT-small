           M, N, K,

    # Compute block index
    grid_m = tl.cdiv(M, BLOCK_M)
    grid_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // grid_n
    pid_n = pid % grid_n

        right = right * right_gate

    # permute and reshape for batched matmul
    B, N, _, H = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) transposed k dimension
    # combine batch and hidden dims
    left_mat = left_perm.view(-1, N, N)
    right_mat = right_perm.view(-1, N, N)
    batch_h = left_mat.shape[0]

    # allocate output
    out_mat = torch.empty_like(left_mat)  # same dtype (fp16)

    # define constants for blocks
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # compute strides
    stride_ah = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_bh = right_mat.stride(0)
    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)

    stride_ch = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), batch_h)
    _matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_ah, stride_bh, stride_ch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 6 Output layernorm, gate, linear
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # back to dim

    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the reference PyTorch code but fuses
* mask‑gate multiplication,
* the costly N×N×N contraction (einsum) via a custom Triton batched
  matrix‑multiply kernel.

All heavy arithmetic is performed in fp16 with float32 accumulation.
The final output is returned in the original input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,                # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,      # same shape as left/right
    mask_ptr,                           # [B,N,N] flattened
    total_elems, hidden_dim,            # runtime scalars
    HAS_MASK: tl.constexpr,             # compile‑time bool
    BLOCK_SIZE: tl.constexpr,           # compile‑time int
):
    """In‑place element‑wise fusion:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
       Mask is shared across the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden_dim entries share a mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


@triton.jit
def _matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ah, stride_bh, stride_ch,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiply C = A @ B
    A : (batch, M, K)
    B : (batch, K, N)
    C : (batch, M, N)

    Blocked GEMM with float32 accumulation.
    """
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block column
    pid_b = tl.program_id(2)  # batch index

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        # Load A tile (M x K)
        a = tl.load(
            a_batch
            + (offs_m[:, None] * stride_am)
            + ((cur_k + tl.arange(0, BLOCK_K))[None, :] * stride_ak),
            mask=(offs_m[:, None] < M)
            & ((cur_k + tl.arange(0, BLOCK_K))[None, :] < K),
            other=0.0,
        )
        # Load B tile (K x N)
        b = tl.load(
            b_batch
            + ((cur_k + tl.arange(0, BLOCK_K))[:, None] * stride_bk)
            + (offs_n[None, :] * stride_bn),
            mask=((cur_k + tl.arange(0, BLOCK_K))[:, None] < K)
            & (offs_n[None, :] < N),
            other=0.0,
        )
        acc += tl.dot(a, b)

    # Write back
    c = acc.to(tl.float16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, c, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for compute‑heavy parts
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Load linear weights (fp16)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections and gates (fp16)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_w)            # (B, N, N, H)
    right = F.linear(x, right_proj_w)          # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # --------------------------------------------------------------
    # 4. Fuse mask and gate multiplications (Triton)
    # --------------------------------------------------------------
    use_mask = (mask is not None) and (not nomask)
    if use_mask:
        # mask shape (B, N, N) -> broadcast over hidden dim
        mask_fp = mask.to(device).to(torch.float16)
        # Flatten tensors for kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_f.numel()
        BLOCK_SIZE = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # left/right have been modified in‑place
    else:
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ rightᵀ) via Triton
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, H, N, N) for A, (B, H, N, N) with K‑dimension transposed for B
    A = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    B_mat = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) -> K dimension transposed

    # Collapse batch & hidden dims
    A_mat = A.view(-1, N, N)
    B_mat = B_mat.view(-1, N, N)
    batch_h = A_mat.shape[0]

    # Output buffer
    C_mat = torch.empty_like(A_mat)

    # Block sizes (tuned for typical B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Compute strides for the flattened tensors
    stride_ah = A_mat.stride(0)
    stride_am = A_mat.stride(1)
    stride_ak = A_mat.stride(2)

    stride_bh = B_mat.stride(0)
    stride_bk = B_mat.stride(1)
    stride_bn = B_mat.stride(2)

    stride_ch = C_mat.stride(0)
    stride_cm = C_mat.stride(1)
    stride_cn = C_mat.stride(2)

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_h,
    )
    _matmul_kernel[grid](
        A_mat,
        B_mat,
        C_mat,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_ah, stride_bh, stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to original layout (B, N, N, H)
    out = C_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the input dtype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
