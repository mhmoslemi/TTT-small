import math

class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) forward pass – Triton‑accelerated mask & gate fusion.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension `dim`.
2. Cast to FP16 for compute‑heavy operations.
3. Linear projections → left/right tensors (B,N,N,H).
4. Gate projections (sigmoid) → left_gate/right_gate/out_gate (B,N,N,H).
5. Fuse projection, gating and optional mask in a single Triton kernel
   (proj * gate * mask).  If there is no mask the fusion is a simple
   element‑wise product.
6. Core reduction (batched GEMM):
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented by reshaping to (B*H, N, N) and using torch.bmm,
   which invokes cuBLAS/Tensor‑cores.
7. Output LayerNorm over the hidden dimension H, multiply by `out_gate`,
   and a final linear projection back to the original dimension `dim`.
8. Cast the result back to the input dtype (usually float32).

The kernel in step 5 is the only mandatory Triton component; the rest
leverages highly‑optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # int: B*N*N*H
    hidden_dim,        # int: H
    BLOCK_SIZE: tl.constexpr,
):
    """
    out = proj * gate * mask   (mask broadcast over hidden_dim)
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # bounds check
    mask = offs < total_elems

    # Compute index of the mask element (broadcasted over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 2048) -> torch.Tensor:
    """
    Fuse projection, gate and mask using a Triton kernel.
    All tensors must be fp16, contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)  (binary 0/1, broadcast over H)
    Returns proj * gate * mask (shape (B,N,N,H)).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (float32/float16/bool)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim` (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask + gating (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16, contiguous, binary (0/1)
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        # No mask → simple elementwise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
