@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, H, # or maybe not needed if we have strides
    stride_lm, stride_li, stride_lk,
    stride_rm, stride_rk, stride_rj,
    stride_om, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row (i)
    pid_n = tl.program_id(1)  # block col (j)
    pid_h = tl.program_id(2)  # batch_hidden index (m)

    # compute offsets
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # mask for out-of-bounds
        i_mask = i_offset < N
        j_mask = j_offset < N
        k_mask = k_offset < N

        # pointers
        a_ptrs = left_ptr + pid_h * stride_lm + i_offset[:, None] * stride_li + k_offset[None, :] * stride_lk
        b_ptrs = right_ptr + pid_h * stride_rm + k_offset[:, None] * stride_rk + j_offset[None, :] * stride_rj

        # load
        a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        # dot product accumulate
        acc += tl.dot(a, b)  # a: (BM,BK), b: (BK,BN) -> (BM,BN)

    # write back result
    out_ptrs = out_ptr + pid_h * stride_om + i_offset[:, None] * stride_oi + j_offset[None, :] * stride_oj
    # store with mask
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # config extract
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)
    device = input_tensor.device; dtype_in = input_tensor.dtype

    # 1. Input LayerNorm (float32)
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # cast to fp16
    x = x.to(torch.float16)

    # 2. Linear projections
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)   # shape (B,N,N,H)
    right = F.linear(x, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # 3. Apply mask & gate
    if mask is not None and not nomask:
        # flatten and apply mask/gates with kernel
        # Convert mask to fp16
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)  # (B*N*N)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # 4. Core O(N^3) contraction using Triton
    B, N, _, H = left.shape
    # Permute to (B, H, N, N) and flatten to (M, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    M = B * H
    left_mat = left_perm.view(M, N, N)
    right_mat = right_perm.view(M, N, N)
    out_mat = torch.empty_like(left_mat)  # (M,N,N), fp16

    # strides
    stride_lm, stride_li, stride_lk = left_mat.stride()
    stride_rm, stride_rk, stride_rj = right_mat.stride()
    stride_om, stride_oi, stride_oj = out_mat.stride()

    # block sizes: tuned for best performance
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), M)
    _tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M,
        N,
        stride_lm, stride_li, stride_lk,
        stride_rm, stride_rk, stride_rj,
        stride_om, stride_oi, stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 5. Output LayerNorm, out_gate, final linear projection
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate  # broadcast over hidden dim

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # (B,N,N,dim), fp16

    # cast back to original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

"""
TriMul forward (outgoing) – Triton‑accelerated implementation.

Algorithm
---------
1. Input LayerNorm (float32) → fp16.
2. Linear projections (left/right) and gate‑projections (left_gate, right_gate,
   out_gate) using the provided weights (bias‑less).  All heavy work is in fp16.
3. Fuse element‑wise mask (if present) and gate multiplication with a small
   Triton kernel `_gate_mask_kernel`.  This writes the gated tensors back in‑place.
4. Core O(N³·H) contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication `left @ rightᵀ` for each hidden
   channel d.  We reshape the tensors to (M=N_batch*H, N, N) and compute the
   GEMM with a custom Triton kernel `_tri_mul_gemm_kernel`, using
   block‑tile sizes (128×128) and Tensor‑Core‑compatible fp16→fp32 accumulation.
5. Output LayerNorm over the hidden dimension, element‑wise multiplication by
   `out_gate`, and final linear projection back to the original dimension.
6. Cast the result back to the input dtype (usually float32).

All tensors are kept on the GPU; the only explicit Triton kernels are the
gate‑mask fusion and the batched GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    """In‑place fusion:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
       `mask` is broadcast over the hidden dimension H.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load tensors
    left      = tl.load(left_ptr + offs,      mask=active, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr + offs,mask=active, other=0.0)

    if HAS_MASK:
        # mask index repeats every hidden_dim entries
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs,  left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,                                   # M = B*H, N = seq_len
    stride_lm, stride_li, stride_lk,
    stride_rm, stride_rk, stride_rj,
    stride_om, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for TriMul:
        out[m, i, j] = Σ_k left[m, i, k] * right[m, j, k]
    left and right have shape (M, N, N); out has the same shape.
    """
    pid_m = tl.program_id(2)      # batch × hidden index
    pid_i = tl.program_id(0)      # tile row
    pid_j = tl.program_id(1)      # tile column

    # Compute row / column offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_offset < N
    j_mask = j_offset < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # Load a tile of left: (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + pid_m * stride_lm \
                 + i_offset[:, None] * stride_li \
                 + k_offset[None, :] * stride_lk
        a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # Load a tile of right transposed: (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + pid_m * stride_rm \
                 + k_offset[:, None] * stride_rk \
                 + j_offset[None, :] * stride_rj
        b = tl.load(b_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        # Accumulate dot product (fp16 inputs → fp32 accumulation)
        acc += tl.dot(a, b)

    # Write the result back (cast to fp16)
    out_ptrs = out_ptr + pid_m * stride_om \
               + i_offset[:, None] * stride_oi \
               + j_offset[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) and cast to fp16
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑less) and gate projections
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left  = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)         # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3. Fuse mask (if any) and gates via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Prepare contiguous flattened tensors
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # Simple element‑wise fusion when mask is absent
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core O(N³·H) contraction using a tiled Triton GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Move hidden dim next to batch for efficient contiguous layout
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    M = B * H
    left_mat  = left_perm.view(M, N, N)
    right_mat = right_perm.view(M, N, N)
    out_mat   = torch.empty_like(left_mat)   # (M, N, N), fp16

    # Strides in elements
    stride_lm, stride_li, stride_lk = left_mat.stride()
    stride_rm, stride_rk, stride_rj = right_mat.stride()
    stride_om, stride_oi, stride_oj = out_mat.stride()

    # Tile sizes – tuned for B200 Tensor‑Cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), M)
    _tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M,
        N,
        stride_lm, stride_li, stride_lk,
        stride_rm, stride_rk, stride_rj,
        stride_om, stride_oi, stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore (B, N, N, H) layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B, N, N, C), fp16

    # ------------------------------------------------------------------
    # 6. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
