"""
TriMul (outgoing) forward pass – Triton‑fused mask/gate and batched GEMM.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) using PyTorch.
2. Cast to fp16 for compute‑heavy phases.
3. Linear projections (left/right) and their gates are computed with
   `torch.nn.functional.linear`.  The element‑wise `proj * gate * mask`
   (when a mask is present) is fused in a tiny Triton kernel.
4. The core O(N³·H) reduction:
   - Reshape the projected tensors to (M, N, N) where M = B·H.
   - Run a custom Triton kernel that performs a batched matrix
     multiplication `out = left @ rightᵀ` for each of the M matrices.
   - All accumulations are done in fp32 and the result is stored in fp16.
5. Reshape the output back to (B,N,N,H), apply a second LayerNorm,
   multiply by the output gate and project back to the original channel
   dimension.
6. Cast the final result back to the input dtype (float32).

The implementation mixes PyTorch for highly‑optimized LayerNorm/Linear
with Triton kernels for the mask‑gate fusion and the large batched GEMM,
giving a large speed‑up over a pure‑PyTorch version while remaining
correct.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and optional mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (total_elems,)
    gate_ptr,          # fp16 * (total_elems,)
    mask_ptr,          # fp16 * (mask_elems,)
    out_ptr,           # fp16 * (total_elems,)
    total_elems,       # i32
    hidden_dim,        # i32 (size of last dimension, broadcast factor for mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # mask is broadcasted over the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 2048) -> torch.Tensor:
    """Fuse `proj * gate * mask` with a Triton kernel."""
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],               # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the core TriMul reduction
# out = left @ rightᵀ  where left/right have shape (M, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr,   # fp16 * (M, N, K)  (K == N)
    right_ptr,  # fp16 * (M, N, K)
    out_ptr,    # fp16 * (M, N, N)
    M, N, K,
    stride_left_m, stride_left_n, stride_left_k,
    stride_right_m, stride_right_n, stride_right_k,
    stride_out_m, stride_out_n, stride_out_n2,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile column
    pid_z = tl.program_id(2)          # which matrix in the batch (M dimension)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out‑of‑bounds rows / columns
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension (also N)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (left_ptr +
                  pid_z * stride_left_m +
                  offs_m[:, None] * stride_left_n +
                  offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load right tile transposed: we need (BLOCK_K, BLOCK_N)
        # Original right layout = (M, N, K). We want rightᵀ => (K, N)
        b_ptrs = (right_ptr +
                  pid_z * stride_right_m +
                  offs_k[:, None] * stride_right_k +
                  offs_n[None, :] * stride_right_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # Write back
    out_ptrs = (out_ptr +
                pid_z * stride_out_m +
                offs_m[:, None] * stride_out_n +
                offs_n[None, :] * stride_out_n2)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, C]
        - mask        : torch.Tensor of shape [B, N, N] (or None)
        - weights     : dict of model parameters (float32)
        - config      : dict with keys:
            * "dim" (int)        – input channel dimension C
            * "hidden_dim" (int) – hidden dimension H
            * "nomask" (bool)    – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # First LayerNorm (input)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Use fp16 for the heavy compute
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Cast constant weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w) # (B,N,N,H)

    # ------------------------------------------------------------------
    # Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # Fuse projection, gate and optional mask (Triton)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:

        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Reshape to (M, N, N) where M = B * hidden_dim
    B, N, _, H = left.shape
    # (B, N, N, H) -> (B, H, N, N) -> (M, N, N)
    left_reshaped = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N)

    M = left_reshaped.shape[0]   # = B * H
    K = N                         # because we multiply left @ rightᵀ

    # Allocate output tensor
    out_reshaped = torch.empty_like(left_reshaped)  # (M, N, N), fp16

    # Compute strides for the three tensors (contiguous layout)
    # left/right: (M, N, N) -> stride = (N*N, N, 1)
    stride_left_m, stride_left_n, stride_left_k = left_reshaped.stride()
    stride_right_m, stride_right_n, stride_right_k = right_reshaped.stride()
    # out: (M, N, N) -> same stride pattern
    stride_out_m, stride_out_n, stride_out_n2 = out_reshaped.stride()

    # Tile sizes – tuned for typical B200 resources
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        M,                         # batch*hidden
    )
    _tri_mul_gemm_kernel[grid](
        left_reshaped,
        right_reshaped,
        out_reshaped,
        M, N, K,
        stride_left_m, stride_left_n, stride_left_k,
        stride_right_m, stride_right_n, stride_right_k,
        stride_out_m, stride_out_n, stride_out_n2,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_reshaped.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Second LayerNorm, output gate and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # Cast back to original dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
