def custom_kernel(data):
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # Cast to fp16 for compute
    x = x.to(torch.float16)

    # 2. Get weights cast to fp16
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # 3. Projections
    left = F.linear(x, left_proj_w)   # shape [B,N,N,H]
    right = F.linear(x, right_proj_w) # shape [B,N,N,H]

    # 4. Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    B, N, _, _ = left.shape

    # 5. Prepare mask
    if mask is not None and not nomask:
        mask_tensor = mask.to(torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # 6. Fuse mask and gates (Triton)
    total_elements = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)
    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    _mask_gate_fusion_kernel[grid](
        left,
        right,
        mask_tensor.view(-1),
        left_gate,
        right_gate,
        left_out,
        right_out,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # 7. Core TriMul reduction using custom Triton GEMM
    # permute to bring hidden dim to batch
    left_perm = left_out.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_out.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape  # note shape is (B, H, N, N)
    # flatten batch and hidden dims
    left_flat = left_perm.view(-1, N, N)
    right_flat = right_perm.view(-1, N, N)
    out_flat = torch.empty_like(left_flat)

    # strides
    stride_lb = left_flat.stride(0)
    stride_lm = left_flat.stride(1)
    stride_lk = left_flat.stride(2)

    stride_rb = right_flat.stride(0)
    stride_rn = right_flat.stride(1)
    stride_rk = right_flat.stride(2)

    stride_ob = out_flat.stride(0)
    stride_om = out_flat.stride(1)
    stride_on = out_flat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    NUM_BLOCKS_M = (N + BLOCK_M - 1) // BLOCK_M
    NUM_BLOCKS_N = (N + BLOCK_N - 1) // BLOCK_N
    num_batches = left_flat.shape[0]

    grid = (num_batches * NUM_BLOCKS_M * NUM_BLOCKS_N,)

    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_lb, stride_lm, stride_lk,
        stride_rb, stride_rn, stride_rk,
        stride_ob, stride_om, stride_on,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        NUM_BLOCKS_M=NUM_BLOCKS_M,
        NUM_BLOCKS_N=NUM_BLOCKS_N,
        num_warps=4,
    )

    # reshape back to (B,N,N,H)
    out = (
        out_flat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # 8. Second layer norm over hidden dim H
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 9. Apply output gate
    out = out * out_gate

    # 10. Final linear back to dim
    out = F.linear(out, to_out_w)

    # cast back
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out

   tid = pid % blocks_per_batch
   pid_m = tid // num_blocks_n
   pid_n = tid % num_blocks_n

   # offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

   # Compute mask for out-of-bounds
   mask_m = offs_m < M
   mask_n = offs_n < N

   # base offset for this batch
   batch_offset_l = batch * stride_lb
   batch_offset_r = batch * stride_rb
   batch_offset_o = batch * stride_ob

   # Initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # loop over K
   for k in range(0, tl.cdiv(K, BLOCK_K)):
       offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
       mask_k = offs_k < K

       # load tiles
       a = tl.load(left_ptr + batch_offset_l + offs_m[:, None] * stride_lm + offs_k[None, :] * stride_lk,

                   mask=mask_k[:, None] & mask_n[None, :], other=tl.float16(0.0))

       acc += tl.dot(a, b)

   # write output
   c = acc.to(tl.float16)
   tl.store(out_ptr + batch_offset_o + offs_m[:, None] * stride_om + offs_n[None, :] * stride_on,
            c, mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The forward pass consists of:
1. Layer‑norm the input tensor (float32 for stability).
2. Linear projections (left/right) and three sigmoid gates (left, right, out) in fp16.
3. Fuse mask, left/right projections and their gates in a Triton kernel.
4. Core cubic reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication (B·H independent GEMMs) in a
   custom Triton kernel that directly computes left @ rightᵀ for each hidden
   slice.
5. Layer‑norm over the hidden dimension, apply the output gate and a final
   linear projection back to the original channel dimension.
The final result is cast back to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Mask + gate fusion (vectorized)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,            # const * left  (B,N,N,H) flattened
    right_ptr,           # const * right (B,N,N,H) flattened
    mask_ptr,            # const * mask  (B,N,N)   flattened
    left_gate_ptr,       # const * left_gate  (B,N,N,H) flattened
    right_gate_ptr,      # const * right_gate (B,N,N,H) flattened
    left_out_ptr,        # * left_out  (B,N,N,H) flattened
    right_out_ptr,       # * right_out (B,N,N,H) flattened
    H: tl.constexpr,                 # hidden dimension
    total_elements: tl.constexpr,    # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load raw tensors
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # mask repeats for each hidden channel
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply mask and gates
    tl.store(left_out_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


# ----------------------------------------------------------------------
# Batched matrix multiplication for the TriMul reduction
# Computes out = left @ rightᵀ for each (batch, hidden) slice.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    left_ptr,    # const * left  (B*H, N, N)   fp16
    right_ptr,   # const * right (B*H, N, N)   fp16
    out_ptr,     # * out        (B*H, N, N)   fp16
    M, N, K,     # problem dimensions (all equal to seq_len)
    stride_lb, stride_lm, stride_lk,   # left strides: batch, row, col
    stride_rb, stride_rn, stride_rk,   # right strides: batch, row(N), col(K)
    stride_ob, stride_om, stride_on,   # out strides: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Decompose program id into (batch, block_m, block_n)
    # --------------------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    batch = pid // blocks_per_batch
    tid   = pid % blocks_per_batch
    pid_m = tid // num_blocks_n
    pid_n = tid % num_blocks_n

    # --------------------------------------------------------------
    # Offsets for this block
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base offsets for the current batch slice
    batch_off_l = batch * stride_lb
    batch_off_r = batch * stride_rb
    batch_off_o = batch * stride_ob

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension K
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: (BLOCK_M, BLOCK_K)
        a = tl.load(
            left_ptr + batch_off_l
            + offs_m[:, None] * stride_lm
            + offs_k[None, :] * stride_lk,
            mask=mask_m[:, None] & mask_k[None, :],
            other=tl.float16(0.0),
        )
        # Load B tile as (BLOCK_K, BLOCK_N) -> we read rightᵀ
        b = tl.load(
            right_ptr + batch_off_r
            + offs_k[:, None] * stride_rk
            + offs_n[None, :] * stride_rn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=tl.float16(0.0),
        )
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write result
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        out_ptr + batch_off_o
        + offs_m[:, None] * stride_om
        + offs_n[None, :] * stride_on,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters
            - config: dict containing:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – if True, ignore the mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm (fp32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"]
        .to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"]
        .to(device=device, dtype=torch.float32),
    )
    # Cast to fp16 for the heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 5) Prepare mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_tensor = mask.to(torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6) Fuse mask and gates (Triton)
    # ------------------------------------------------------------------
    total_elements = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    _mask_gate_fusion_kernel[grid](
        left,
        right,
        mask_tensor.view(-1),
        left_gate,
        right_gate,
        left_out,
        right_out,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Core TriMul reduction (batched GEMM via Triton)
    # ------------------------------------------------------------------
    # Move hidden dimension to the batch axis for a packed GEMM
    left_perm  = left_out.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_out.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Flatten batch & hidden dimensions
    left_flat  = left_perm.view(-1, N, N)
    right_flat = right_perm.view(-1, N, N)
    out_flat   = torch.empty_like(left_flat)

    # Strides (int64)
    stride_lb = left_flat.stride(0)
    stride_lm = left_flat.stride(1)
    stride_lk = left_flat.stride(2)

    stride_rb = right_flat.stride(0)
    stride_rn = right_flat.stride(1)
    stride_rk = right_flat.stride(2)

    stride_ob = out_flat.stride(0)
    stride_om = out_flat.stride(1)
    stride_on = out_flat.stride(2)

    # Tiling parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    num_batches  = left_flat.shape[0]

    grid = (num_batches * num_blocks_m * num_blocks_n,)

    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_lb, stride_lm, stride_lk,
        stride_rb, stride_rn, stride_rk,
        stride_ob, stride_om, stride_on,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to (B, N, N, H)
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 8) Second layer‑norm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9) Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10) Final projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)

    # ------------------------------------------------------------------
    # 11) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
