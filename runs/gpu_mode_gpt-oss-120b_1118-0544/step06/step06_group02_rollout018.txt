import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    # mask index broadcast across hidden_dim
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    # proj, gate: [B,N,N,H] fp16 contiguous
    # mask: [B,N,N] fp16 contiguous
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size
    )
    return out

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    TriMul (outgoing) forward pass – fused mask & gate via Triton.
    The heavy O(N³·H) reduction is performed with torch.bmm (cuBLAS).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for compute
    x = x.to(torch.float16)

    # Cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(device, torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device, torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    to_out_w = weights["to_out.weight"].to(device, torch.float16)

    # Concatenate left weights (proj + gate) and right weights similarly
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, dim)
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0)   # (2*H, dim)

    # Linear projections (proj + gate logits) in one go
    left_concat = torch.nn.functional.linear(x, left_concat_w)   # (B,N,N,2*H)
    right_concat = torch.nn.functional.linear(x, right_concat_w) # (B,N,N,2*H)

    # Split
    left_proj = left_concat[..., :hidden_dim]                     # (B,N,N,H)
    left_gate_logit = left_concat[..., hidden_dim:]              # (B,N,N,H)
    left_gate = left_gate_logit.sigmoid()

    right_proj = right_concat[..., :hidden_dim]
    right_gate_logit = right_concat[..., hidden_dim:]
    right_gate = right_gate_logit.sigmoid()

    # Output gate
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # Apply mask and gate
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Core reduction: batched GEMM per head
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_reshaped = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # compute left @ right^T
    out_reshaped = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))
    # reshape back to (B,N,N,H)
    out = out_reshaped.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Second LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection to original dim
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # Cast back to original dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    mask_idx = offs // hidden_dim

    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, dim)
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    left_concat = torch.nn.functional.linear(x, left_concat_w)    # (B,N,N,2*H)
    right_concat = torch.nn.functional.linear(x, right_concat_w)  # (B,N,N,2*H)

    # Split back into proj & gate logits
    left_proj = left_concat[..., :hidden_dim]                     # (B,N,N,H)
    left_gate_logit = left_concat[..., hidden_dim:]              # (B,N,N,H)
    left_gate = left_gate_logit.sigmoid()

    right_proj = right_concat[..., :hidden_dim]
    right_gate_logit = right_concat[..., hidden_dim:]
    right_gate = right_gate_logit.sigmoid()

    # Output gate (separate linear)
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4. Fuse mask & gate (optional) using Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³·H) reduction: batched GEMM per hidden dimension
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, H, N, N) -> (B*H, N, N)
    left_reshaped = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_reshaped = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_reshaped.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Second LayerNorm + output gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to original dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # Cast back to the original input dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
