    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    # layernorm
    norm_w = weights["norm.weight"].to(device).float()
    norm_b = weights["norm.bias"].to(device).float()
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    # cast to fp16
    x = x.to(torch.float16)
    # compute projections/gates
    left_proj_w = weights["left_proj.weight"].to(device).half()
    right_proj_w = weights["right_proj.weight"].to(device).half()
    left_gate_w = weights["left_gate.weight"].to(device).half()
    right_gate_w = weights["right_gate.weight"].to(device).half()
    out_gate_w = weights["out_gate.weight"].to(device).half()

    left = torch.nn.functional.linear(x, left_proj_w)  # shape B,N,N,H
    right = torch.nn.functional.linear(x, right_proj_w)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # apply mask if any and gates
    if mask is not None and not config.get("nomask", False):
        # we fuse mask/gates using Triton kernel (previous)
        # combine with left/right gating
        # But we already have left,right, left_gate, right_gate; we can fuse via in-place kernel.
        # However we may just do elementwise as before (torch) because kernel may not give huge speed.
        # But we need a Triton kernel somewhere else (the contraction kernel qualifies)
        mask_fp = mask.to(device).to(torch.float16)
        left = left * mask_fp.unsqueeze(-1) * left_gate
        right = right * mask_fp.unsqueeze(-1) * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # Ensure left,right contiguous

    stride_bo, stride_io, stride_jo, stride_ho = out.stride()

    # Determine block sizes; empirically set
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 4  # small to reduce loops
    BLOCK_H = 32  # process hidden dim in chunks

    total_blocks_i = (N + BLOCK_M - 1) // BLOCK_M
    total_blocks_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * total_blocks_i * total_blocks_j,)

    tri_mul_kernel[grid](
        left, right, out,
        B, N, hidden,
        stride_bl, stride_il, stride_kl, stride_hl,
        stride_br, stride_jr, stride_kr, stride_hr,
        stride_bo, stride_io, stride_jo, stride_ho,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H
    )
    # out now holds contraction result

    # Apply to_out_norm, out_gate, to_out linear
    to_out_norm_w = weights["to_out_norm.weight"].to(device).half()
    to_out_norm_b = weights["to_out_norm.bias"].to(device).half()
    out = torch.nn.functional.layer_norm(out, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # elementwise

    # final linear
    to_out_w = weights["to_out.weight"].to(device).half()
    out = torch.nn.functional.linear(out, to_out_w)  # shape (B,N,N,dim)

    # Cast back to original dtype
    out = out.to(dtype_in)

    return out

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    b = tl.program_id(0)
    i_block = tl.program_id(1)
    j_block = tl.program_id(2)

    # compute global indices
    i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for i and j
    mask_i = i < N
    mask_j = j < N

    # loop over hidden dim
    for h_start in range(0, H, BLOCK_H):
        # compute h indices for this chunk
        h = h_start + tl.arange(0, BLOCK_H)
        mask_h = h < H

        # accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # loop over k dimension
        for k0 in range(0, N, BLOCK_K):
            # k indices
            k = k0 + tl.arange(0, BLOCK_K)
            mask_k = k < N

            # compute offsets
            i_off = i[:, None, None] * stride_il
            j_off = j[:, None, None] * stride_jr
            k_off = k[None, :, None] * stride_kl  # same stride for left and right k
            h_off = h[None, None, :] * stride_hl   # for left
            # load left tile
            left_ptrs = left_ptr + b * stride_bl + i_off + k_off + h_off
            left_tile = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0,
            )
            # for right, h stride may be same but we use stride_hr for clarity
            h_off_r = h[None, None, :] * stride_hr
            right_ptrs = right_ptr + b * stride_br + j_off + k_off + h_off_r
            right_tile = tl.load(
                right_ptrs,
                mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0,
            )
            # accumulate over k dimension
            for kk in range(0, BLOCK_K):
                # The loop unrolled at compile time; need to ensure we don't exceed N if mask_k[kk] false.
                left_slice = left_tile[:, kk, :]   # (M, H)
                right_slice = right_tile[:, kk, :] # (N, H)
                # outer product across i,j for each h
                acc += left_slice[:, None, :] * right_slice[None, :, :]
        # store result
        out_ptrs = out_ptr + b * stride_bo + i[:, None, None] * stride_io + j[None, :, None] * stride_jo + h[None, None, :] * stride_ho
        # convert to fp16 before store
        tl.store(
            out_ptrs,
            acc.to(tl.float16),

        )

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

The implementation follows the reference PyTorch TriMul:
1. Layer‑norm the input (float32 for stability).
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (left/right) and three sigmoid gates (left/right/out).
4. Apply the optional mask and fuse it with the two element‑wise gates.
5. Compute the core contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   via a custom Triton kernel that processes the sequence dimensions in
   blocks while iterating over the hidden dimension in chunks.
6. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate and a final linear projection back to the original dimension.
7. Cast back to the input dtype.

Only the heavy O(N³·C_hidden) contraction is written in Triton; the rest
uses highly‑optimised torch operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # shape / strides
    N,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    # compile‑time constants
    H: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using block‑wise tiling on (i,j) and chunked processing of the hidden
    dimension `h`.  All tensors are contiguous with layout
    (B, N, N, H) and stride order (B, i, k, h).
    """
    b = tl.program_id(0)          # batch index
    i_block = tl.program_id(1)    # block index on axis i
    j_block = tl.program_id(2)    # block index on axis j

    # Global i/j coordinates for this program
    i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # Loop over hidden dimension in chunks of BLOCK_H
    for h_start in range(0, H, BLOCK_H):
        h = h_start + tl.arange(0, BLOCK_H)
        mask_h = h < H

        # Accumulator for the current hidden chunk
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # Loop over k dimension
        for k0 in range(0, N, BLOCK_K):
            k = k0 + tl.arange(0, BLOCK_K)
            mask_k = k < N

            # Offsets for the three dimensions (i/k/h) – broadcasting builds a (M,K,H) tile
            i_off = i[:, None, None] * stride_il
            j_off = j[:, None, None] * stride_jr
            k_off_l = k[None, :, None] * stride_kl
            k_off_r = k[None, :, None] * stride_kr
            h_off_l = h[None, None, :] * stride_hl
            h_off_r = h[None, None, :] * stride_hr

            # Load left tile  (M, K, H)
            left_tile = tl.load(
                left_ptr
                + b * stride_bl
                + i_off
                + k_off_l
                + h_off_l,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0,
            )
            # Load right tile (N, K, H)
            right_tile = tl.load(
                right_ptr
                + b * stride_br
                + j_off
                + k_off_r
                + h_off_r,
                mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0,
            )

            # Accumulate over the K dimension (unrolled, BLOCK_K is constexpr)
            for kk in range(0, BLOCK_K):
                # left_slice: (M, H), right_slice: (N, H)
                left_slice = left_tile[:, kk, :]   # (M, H)
                right_slice = right_tile[:, kk, :] # (N, H)
                # Outer product across i and j for each hidden channel
                acc += left_slice[:, None, :] * right_slice[None, :, :]

        # Store the result for this hidden slice
        out_ptrs = (
            out_ptr
            + b * stride_bo
            + i[:, None, None] * stride_io
            + j[None, :, None] * stride_jo
            + h[None, None, :] * stride_ho
        )
        tl.store(
            out_ptrs,
            acc.to(tl.float16),
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------ #
    norm_w = weights["norm.weight"].to(device).float()
    norm_b = weights["norm.bias"].to(device).float()
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------ #
    # 2. Cast to fp16 for the heavy part
    # ------------------------------------------------------------------ #
    x = x.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 3. Linear projections (no bias) – all in fp16
    # ------------------------------------------------------------------ #
    left_proj_w = weights["left_proj.weight"].to(device).half()
    right_proj_w = weights["right_proj.weight"].to(device).half()
    left_gate_w = weights["left_gate.weight"].to(device).half()
    right_gate_w = weights["right_gate.weight"].to(device).half()
    out_gate_w = weights["out_gate.weight"].to(device).half()

    left = F.linear(x, left_proj_w)                     # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)                  # (B,N,N,hidden_dim)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------ #
    # 4. Apply mask (if present) and fuse with gates
    # ------------------------------------------------------------------ #
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp * left_gate
        right = right * mask_fp * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Core O(N³·H) contraction via a custom Triton kernel
    # ------------------------------------------------------------------ #
    left = left.contiguous()
    right = right.contiguous()
    B, N_seq, _, H = left.shape
    out = torch.empty((B, N_seq, N_seq, H), device=device, dtype=torch.float16)

    # strides for the tensors (B, i, k, h) layout
    stride_bl, stride_il, stride_kl, stride_hl = left.stride()
    stride_br, stride_jr, stride_kr, stride_hr = right.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out.stride()

    # block sizes – chosen to fit in registers while giving enough parallelism
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 4
    BLOCK_H = 32

    grid = (B, (N_seq + BLOCK_M - 1) // BLOCK_M, (N_seq + BLOCK_N - 1) // BLOCK_N)

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        N_seq,
        stride_bl,
        stride_il,
        stride_kl,
        stride_hl,
        stride_br,
        stride_jr,
        stride_kr,
        stride_hr,
        stride_bo,
        stride_io,
        stride_jo,
        stride_ho,
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------ #
    # 6. Output LayerNorm, gate, and final linear projection
    # ------------------------------------------------------------------ #
    to_out_norm_w = weights["to_out_norm.weight"].to(device).half()
    to_out_norm_b = weights["to_out_norm.bias"].to(device).half()
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (still fp16)

    to_out_w = weights["to_out.weight"].to(device).half()
    out = F.linear(out, to_out_w)  # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------ #
    # 7. Cast back to original dtype and return
    # ------------------------------------------------------------------ #
    out = out.to(dtype_in)

    return out
