def custom_kernel(data):
   input_tensor, mask, weights, config = data
   device = input_tensor.device
   dtype = input_tensor.dtype
   B,N,N,dim = input_tensor.shape
   hidden = config["hidden_dim"]
   dim = config["dim"]

   # 1. LayerNorm
   norm_w = weights["norm.weight"].to(device).to(torch.float32)
   norm_b = weights["norm.bias"].to(device).to(torch.float32)
   x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
   # Cast to fp16 for compute
   x = x.to(torch.float16)

   # 2. Linear projections and gates
   left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
   right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
   left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
   right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
   out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

   # Linear projections (no bias)
   left = torch.nn.functional.linear(x, left_proj_w)
   right = torch.nn.functional.linear(x, right_proj_w)

   left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
   right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
   out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

   # Fuse mask and gates: left = left * left_gate * mask; right = right * right_gate * mask
   use_mask = mask is not None and not config.get("nomask", False)
   if use_mask:
        # convert mask to fp16
        mask_fp = mask.to(device).to(torch.float16)
        # flatten
        left_f = left.reshape(-1)
        right_f = right.reshape(-1)
        left_gate_f = left_gate.reshape(-1)
        right_gate_f = right_gate.reshape(-1)
        mask_f = mask_fp.reshape(-1)
        total = left_f.numel()
        BLOCK_SIZE = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,

            right_gate_f,

            total,
            hidden,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        left = left_f.view_as(left)
        right = right_f.view_as(right)
   else:

        right = right * right_gate

   # 3. Batched GEMM using torch.bmm (fast)
   B_, N_, _, H_ = left.shape

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # Load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

"""
Outgoing Triangle Multiplicative Update (TriMul) – forward pass.

Algorithm
---------
1. Layer‑normalize the input (float32 for stability).
2. Cast to fp16 and apply linear projections (left/right) and their
   multiplicative gates (left_gate/right_gate).  The optional mask is
   fused together with the gate multiplications in a small Triton kernel
   to reduce memory traffic.
3. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched GEMM (left @ rightᵀ) over the sequence dimension.
4. Apply an output LayerNorm, multiply by the out‑gate and a final linear
   projection back to the original feature dimension.
All heavy arithmetic is performed in fp16; the final result is cast back
to the input dtype (usually float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,                # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,      # same shape as left/right
    mask_ptr,                           # [B,N,N] flattened (broadcasted)
    total_elems, hidden_dim,            # runtime scalars
    HAS_MASK: tl.constexpr,             # compile‑time bool
    BLOCK_SIZE: tl.constexpr,           # compile‑time int
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    Mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left      = tl.load(left_ptr      + offs, mask=active, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden_dim entries share a mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    ##################################################################
    # 1. Input LayerNorm (float32 for numerical stability)
    ##################################################################
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    # LayerNorm over the last dimension (= dim)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for compute‑heavy parts
    x = x.to(torch.float16)

    ##################################################################
    # 2. Load linear weights (cast to fp16)
    ##################################################################
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    ##################################################################
    # 3. Linear projections and gates
    ##################################################################
    # Projections (no bias)
    left = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)        # (B, N, N, H)

    # Gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    ##################################################################
    # 4. Fuse mask and gate multiplication with a Triton kernel
    ##################################################################
    use_mask = (mask is not None) and (not nomask)
    if use_mask:
        # Prepare mask in fp16
        mask_fp = mask.to(device).to(torch.float16)

        # Ensure contiguous views for the kernel
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()
        mask_fp   = mask_fp.contiguous()

        # Flatten tensors
        left_flat      = left.view(-1)
        right_flat     = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat= right_gate.view(-1)
        mask_flat      = mask_fp.view(-1)            # (B*N*N,)

        total = left_flat.numel()
        BLOCK_SIZE = 1024

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # Reshape back to original layout
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask: simple element‑wise gate multiplication
        left = left * left_gate
        right = right * right_gate

    ##################################################################
    # 5. Core contraction – batched GEMM (left @ rightᵀ) over sequence dim
    ##################################################################
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) with row‑major layout
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiply: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    ##################################################################
    # 6. Output LayerNorm, out‑gate, final linear projection
    ##################################################################
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the original input dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
