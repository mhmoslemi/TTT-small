def _gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                     total_elems, hidden_dim,
                     BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # Compute index into mask array: each mask corresponds to hidden_dim consecutive proj entries
    mask_idx = offs // hidden_dim
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    maskv = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    out = proj * gate * maskv
    tl.store(out_ptr + offs, out, mask=mask)
```

"""
Optimized TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑normalize the input tensor (float32) with provided weight/bias.
2. Cast to fp16 for the compute‑intensive part.
3. Linear projections (left/right) and gates (left/right/out) in fp16.
4. Fuse projection, gate and optional mask via a tiny Triton kernel
   (`_gate_mask_kernel`). This avoids two extra memory passes.
5. Core reduction:
   - For each hidden dimension h, compute a matrix multiplication
     left_h @ right_hᵀ (size N×N) using batched GEMM (`torch.bmm`).
   - This yields a tensor of shape [B, N, N, hidden_dim].
6. Apply a second LayerNorm on the hidden dimension, multiply by the
   output gate and linearly project back to the original channel dimension.
7. Cast the result back to the input dtype (float32).

At least one operation (gate & mask fusion) is implemented in Triton.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # corresponding mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 2048,
) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using a Triton kernel.
    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],          # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, C]
        - mask        : torch.Tensor of shape [B, N, N] (or None)
        - weights     : dict of model parameters (float32)
        - config      : dict with keys:
            * "dim" (int)         – input channel dimension C
            * "hidden_dim" (int)  – hidden dimension H
            * "nomask" (bool)     – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------- LayerNorm (input) -------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for the heavy part
    x = x.to(torch.float16)

    # -------------------- cast weights to fp16 -------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ----------------------- linear projections -------------------------
    # (B, N, N, hidden_dim)
    left_proj  = torch.nn.functional.linear(x, left_proj_w)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------- gates ----------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B, N, N, hidden_dim)

    # ------------------- mask & gate fusion (Triton) --------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ----------------- core O(N³·H) reduction via batched GEMM ----------
    B, N, _, _ = left.shape
    # reshape to (B*H, N, N) for batched matmul
    left_reshaped  = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)   # (B*H, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N)  # (B*H, N, N)

    # out_{b,i,j,h} = sum_k left_{b,i,k,h} * right_{b,j,k,h}
    out_reshaped = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))   # (B*H, N, N)

    # reshape back to (B, N, N, hidden_dim)
    out = out_reshaped.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------ cast back to input dtype -----------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
