"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate, fp16 batched GEMM.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) (float32 → fp16).
2. Linear projections (left/right) and their gates (all fp16) using torch.nn.functional.linear.
3. Fuse projection, gate and optional mask with a small Triton kernel
   to avoid separate memory passes.
4. Core O(N³·H) reduction:
   - Reshape left/right to (B*H, N, N) and compute a batched matrix multiply
     `out = left @ right.T` with torch.bmm (Tensor‑cores, fp16).
5. Second LayerNorm on the hidden dimension, multiply by the output gate,
   and linearly project back to the original channel dimension.
6. Cast the result back to the input dtype (float32).

Only the element‑wise mask & gate fusion is performed in Triton; all
large GEMM/LayerNorm steps use PyTorch’s highly‑optimized CUDA kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) on a flattened view
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,   # fp16* (B*N*N*H,)
    gate_ptr,   # fp16* (B*N*N*H,)
    mask_ptr,   # fp16* (B*N*N,)
    out_ptr,    # fp16* (B*N*N*H,)
    total_elems,    # i32 - total number of elements in proj/gate
    hidden_dim,     # i32 - H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # mask index broadcasted across hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 2048,
) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using a Triton kernel.

    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],          # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] (or None)
        - weights      : dict of model parameters (float32)
        - config       : dict with keys:
            * "dim" (int)          – input channel dimension C
            * "hidden_dim" (int)   – hidden dimension H
            * "nomask" (bool)      – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Cast weights to fp16 once
    # ------------------------------------------------------------------
    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, w_left_proj)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, w_right_proj) # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, w_left_gate).sigmoid()
    right_gate = torch.nn.functional.linear(x, w_right_gate).sigmoid()
    out_gate = torch.nn.functional.linear(x, w_out_gate).sigmoid()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5. Fuse projection, gate and mask (if present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core O(N³·H) reduction via batched GEMM (fp16)
    #    out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # reshape to (B*H, N, N) for torch.bmm
    left_reshaped = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matmul: (B*H, N, N) @ (B*H, N, N)^T
    out_reshaped = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_reshaped.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Second LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, w_to_out)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8. Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
