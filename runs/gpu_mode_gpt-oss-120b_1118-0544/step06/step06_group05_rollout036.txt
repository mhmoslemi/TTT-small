def batch_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_size, M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # linear grid index
    # compute batch index and block row col
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    # pid = batch_id * num_blocks_m * num_blocks_n + block_m * num_blocks_n + block_n
    batch_id = pid // (num_blocks_m * num_blocks_n)
    block_id = pid % (num_blocks_m * num_blocks_n)
    block_m = block_id // num_blocks_n
    block_n = block_id % num_blocks_n

    # compute offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer offsets for C: base + batch_stride_c * batch_id + offs_m[:,None] * stride_cm + offs_n[None,:] * stride_cn
    c_ptrs = c_ptr + batch_id * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        # Load A block
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_ptr + batch_id * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B block (transposed)

        acc += tl.dot(a, b)

    # Write back
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', False)

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])

    # Cast to fp16
    x = x.to(torch.float16)

    # Load weights
    left_proj_w = weights['left_proj.weight'].to(torch.float16)
    right_proj_w = weights['right_proj.weight'].to(torch.float16)
    left_gate_w = weights['left_gate.weight'].to(torch.float16)
    right_gate_w = weights['right_gate.weight'].to(torch.float16)
    out_gate_w = weights['out_gate.weight'].to(torch.float16)
    to_out_norm_w = weights['to_out_norm.weight'].to(torch.float16)
    to_out_norm_b = weights['to_out_norm.bias'].to(torch.float16)
    to_out_w = weights['to_out.weight'].to(torch.float16)

    # Linear projections: use torch.nn.functional.linear
    left_proj = F.linear(x, left_proj_w)  # shape [B,N,N,H]
    right_proj = F.linear(x, right_proj_w)

    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # mask handling
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(left_proj, right_proj, left_gate, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # core reduction using Triton batched matrix multiply
    B, N, _, H = left.shape
    # now flatten batch*H as separate GEMMs
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # compute strides
    a_stride_b = left_flat.stride(0)
    a_stride_m = left_flat.stride(1)
    a_stride_k = left_flat.stride(2)

    b_stride_b = right_flat.stride(0)
    b_stride_k_orig = right_flat.stride(1)  # row stride = N
    b_stride_n_orig = right_flat.stride(2)  # col stride = 1

    # For transposed view:
    b_stride_k = b_stride_n_orig   # column stride (1)
    b_stride_n = b_stride_k_orig   # row stride (N)

    c_stride_b = out_flat.stride(0)
    c_stride_m = out_flat.stride(1)
    c_stride_n = out_flat.stride(2)

    # Choose block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # compute grid
    total_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    total_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    batch_h = B * H
    grid = (batch_h * total_blocks_m * total_blocks_n,)

    # launch kernel
    batch_matmul_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_h,
        N,
        N,
        N,
        a_stride_m,
        a_stride_k,
        b_stride_k,
        b_stride_n,
        c_stride_m,
        c_stride_n,
        a_stride_b,
        b_stride_b,
        c_stride_b,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,

import torch
import triton
import triton.language as tl
import torch.nn.functional as F
from typing import Tuple, Dict

# Triton kernel for fused projection, gating, mask
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    # each element corresponds to one (b,i,k) location for each hidden dim
    # Compute mask index by dividing by hidden_dim
    mask_idx = offs // hidden_dim

    # Load
    left_proj = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gating and mask using Triton.
    All tensors must be fp16 and contiguous.
    Returns fused left and right tensors of shape (B,N,N,H).
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out

# Triton kernel for batched matmul with transposed B (L @ R^T)
@triton.jit
def _batched_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # Compute number of blocks per dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # Derive batch index and block indices
    batch_id = pid // (num_blocks_m * num_blocks_n)
    block_id = pid % (num_blocks_m * num_blocks_n)
    block_m = block_id // num_blocks_n
    block_n = block_id % num_blocks_n

    # Offsets within the block
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer to output tile
    c_ptrs = c_ptr + batch_id * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = a_ptr + batch_id * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B tile (K x N) – note that we treat B as transposed, so stride_bk corresponds to original column stride
        b_ptrs = b_ptr + batch_id * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # Write-back
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    The implementation:
        1. Layer‑norm the input.
        2. Linear projections + sigmoid gates.
        3. Optional mask fusion via a small Triton kernel.
        4. Core O(N³) reduction computed as a batched matrix multiply L @ Rᵀ using a custom Triton kernel.
        5. Output layer‑norm, gate and final linear projection.
    Returns a tensor of shape [B, N, N, dim] in the original dtype.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # 1) Input LayerNorm over channel dimension
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # Cast to fp16 for compute‑heavy parts
    x = x.to(torch.float16)

    # 2) Cast linear weights to fp16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # 3) Linear projections (no bias) and gates
    left_proj = F.linear(x, left_proj_w)        # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # 4) Apply mask and gates (fused kernel if mask is present)
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # Simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # 5) Core TriMul reduction via batched GEMM L @ Rᵀ
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)

    # Strides (in elements)
    a_stride_b = left_flat.stride(0)
    a_stride_m = left_flat.stride(1)
    a_stride_k = left_flat.stride(2)

    b_stride_b = right_flat.stride(0)
    # original strides for right_flat (batch, row, col)
    b_stride_row = right_flat.stride(1)   # = N
    b_stride_col = right_flat.stride(2)   # = 1
    # For transposed view (K,N)
    b_stride_k = b_stride_col
    b_stride_n = b_stride_row

    c_stride_b = out_flat.stride(0)
    c_stride_m = out_flat.stride(1)
    c_stride_n = out_flat.stride(2)

    # Block sizes – these work well for a wide range of N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    total_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    batch_h = B * H
    grid = (batch_h * total_blocks_m * total_blocks_n,)

    _batched_matmul_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_h,
        N,
        N,
        N,
        a_stride_m,
        a_stride_k,
        b_stride_k,
        b_stride_n,
        c_stride_m,
        c_stride_n,
        a_stride_b,
        b_stride_b,
        c_stride_b,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 6) Output layer‑norm, gate, final linear
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # Cast back to original dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to FP16 and apply linear projections + sigmoid gates.
3. If a mask is provided, fuse projection, gating and mask with a small Triton kernel.

@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr, right_proj_ptr,      # fp16 * (B,N,N,H)
    left_gate_ptr, right_gate_ptr,      # fp16 * (B,N,N,H)
    mask_ptr,                           # fp16 * (B,N,N)
    left_out_ptr, right_out_ptr,        # fp16 * (B,N,N,H)
    total_elems, hidden_dim,            # runtime scalars
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # mask is broadcast over hidden dimension
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    left_out  = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs,  left_out,  mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)

def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """Fuse projection, gating and mask using Triton. All tensors must be fp16."""
    total = left_proj.numel()
    left_out  = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],        # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out

# ----------------------------------------------------------------------
# 2) Batched GEMM kernel (computes L @ Rᵀ for each hidden slice)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_am, stride_ak,          # A: (M,K)
    stride_bk, stride_bn,          # B (transposed view): (K,N)
    stride_cm, stride_cn,          # C: (M,N)
    stride_ab, stride_bb, stride_cb,   # batch strides for A, B, C
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # number of program blocks per dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # decode program id into (batch, block_m, block_n)
    batch_id = pid // (num_blocks_m * num_blocks_n)
    block_id = pid % (num_blocks_m * num_blocks_n)
    block_m = block_id // num_blocks_n
    block_n = block_id % num_blocks_n

    # tile offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to output tile
    c_ptrs = c_ptr + batch_id * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M x K)
        a_ptrs = a_ptr + batch_id * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # B tile (K x N) – we already pass transposed view via strides
        b_ptrs = b_ptr + batch_id * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # write back
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# 3) Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul) using Triton.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    dtype_in = input_tensor.dtype
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast all linear weights to FP16 (no bias)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections and gates
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (fused Triton kernel when mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(left_proj, right_proj, left_gate, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction via batched GEMM L @ Rᵀ
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for the batched GEMM
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_flat = torch.empty_like(left_flat)

    # strides (in elements)
    a_stride_b = left_flat.stride(0)
    a_stride_m = left_flat.stride(1)
    a_stride_k = left_flat.stride(2)

    b_stride_b = right_flat.stride(0)
    # original row/col strides for right_flat (batch, row, col)
    b_stride_row = right_flat.stride(1)   # = N
    b_stride_col = right_flat.stride(2)   # = 1
    # for transposed view (K,N)
    b_stride_k = b_stride_col
    b_stride_n = b_stride_row

    c_stride_b = out_flat.stride(0)
    c_stride_m = out_flat.stride(1)
    c_stride_n = out_flat.stride(2)

    # Tile sizes – work well for the supported sequence lengths
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    total_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    batch_h = B * H
    grid = (batch_h * total_blocks_m * total_blocks_n,)

    _batched_matmul_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_h,
        N,
        N,
        N,
        a_stride_m,
        a_stride_k,
        b_stride_k,
        b_stride_n,
        c_stride_m,
        c_stride_n,
        a_stride_b,
        b_stride_b,
        c_stride_b,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)
    return out
