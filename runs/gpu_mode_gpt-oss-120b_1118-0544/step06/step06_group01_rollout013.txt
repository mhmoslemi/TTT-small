def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # layernorm
    x = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # cast to fp16
    x = x.to(torch.float16)

    # linear projections
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))  # shape B,N,N,H

    # fuse mask and gates using Triton kernel
    if mask is not None and not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # shape B,N,N,1 broadcast
        # flatten
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)  # B*N*N
        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
    else:
        left = left * left_gate
        right = right * right_gate

    # Now left and right shaped B,N,N,H again (if mask case updated in-place). Ensure they are views.
    # reshape to (B*H,N,N)
    B, N, _, H = left.shape
    left_mat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)  # shape (B*H,N,N)
    right_mat = right.permute(0,3,1,2).contiguous().view(B*H, N, N)

    # compute out using Triton kernel
    out_mat = torch.empty_like(left_mat)
    # compute strides
    stride_b, stride_i, stride_k = left_mat.stride()
    stride_br, stride_ir, stride_kr = right_mat.stride()
    stride_ob, stride_io, stride_jo = out_mat.stride()

    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B*H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        stride_left_batch, stride_left_i, stride_left_k,
        stride_right_batch, stride_right_i, stride_right_k,
        stride_out_batch, stride_out_i, stride_out_j,
        N,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
        batch = tl.program_id(0)
        pid_i = tl.program_id(1)
        pid_j = tl.program_id(2)

        # compute base offsets
        i_base = pid_i * BLOCK_M
        j_base = pid_j * BLOCK_N

        # initialize accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # loop over k
        for k in range(0, N, BLOCK_K):

            cur_k = tl.minimum(N - k, BLOCK_K)

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑normalize the input (FP32 for numerical stability).
2. Cast to FP16 and apply linear projections (left/right) and gating
   transformations (left_gate, right_gate, out_gate).
3. Fuse mask, gate and element‑wise multiplication with a small Triton
   kernel to avoid extra memory passes.
4. Compute the core contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a tiled Triton kernel (batched GEMM, essentially Q·Kᵀ) that
   operates on the (batch * hidden) dimension.
5. Apply output LayerNorm, the out‑gate, and a final linear projection
   back to the original feature dimension.
All heavy arithmetic is performed in FP16 with FP32 accumulation.
The final result is returned in the original dtype (normally FP32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,                # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,      # same shape as left/right
    mask_ptr,                           # [B,N,N] flattened
    total_elems, hidden_dim,            # runtime scalars
    HAS_MASK: tl.constexpr,             # compile‑time bool
    BLOCK_SIZE: tl.constexpr,           # compile‑time int
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    The mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left      = tl.load(left_ptr      + offs, mask=active, other=0.0, dtype=tl.float16)
    right     = tl.load(right_ptr     + offs, mask=active, other=0.0, dtype=tl.float16)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0, dtype=tl.float16)
    right_gate= tl.load(right_gate_ptr+ offs, mask=active, other=0.0, dtype=tl.float16)

    if HAS_MASK:
        # each hidden_dim entries share a mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0, dtype=tl.float16)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    N,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out = left @ rightᵀ for each (batch, head) pair.
    left  : [BH, N, N]  (row i, col k)
    right : [BH, N, N]  (row j, col k) – accessed as (k, j) to realise the transpose.
    out   : [BH, N, N]  (row i, col j)
    """
    batch = tl.program_id(0)          # BH dimension
    pid_i = tl.program_id(1)          # tile over rows   (i)
    pid_j = tl.program_id(2)          # tile over cols   (j)

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, N, BLOCK_K):
        cur_k = tl.minimum(N - k, BLOCK_K)

        # ------- load a block of left (i,k) -------
        offs_i = i_base + tl.arange(0, BLOCK_M)
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_i = offs_i < N
        mask_k = offs_k < N

        left = tl.load(
            left_ptr
            + batch * stride_left_batch
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16)

        # ------- load a block of right transposed (k,j) -------
        offs_j = j_base + tl.arange(0, BLOCK_N)
        mask_j = offs_j < N

        right = tl.load(
            right_ptr
            + batch * stride_right_batch
            + offs_k[:, None] * stride_right_i   # k as row
            + offs_j[None, :] * stride_right_k,  # j as column
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
            dtype=tl.float16)

        # dot product and accumulate
        acc += tl.dot(left.to(tl.float32), right.to(tl.float32))

    # ------- write out -------
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    out = acc.to(tl.float16)
    tl.store(
        out_ptr
        + batch * stride_out_batch
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j,
        out,
        mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy parts
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # -----------------------------------------------------------------
    # 4. Linear projections and gates (all fp16)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_w)                     # (B, N, N, H)
    right = F.linear(x, right_proj_w)                   # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 5. Fuse mask and gate multiplication with a Triton kernel
    # -----------------------------------------------------------------
    use_mask = (mask is not None) and (not nomask)
    if use_mask:
        # mask broadcast over hidden dimension
        mask_fp = mask.to(device).to(torch.float16)          # (B, N, N)
        # flatten tensors for the kernel
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)      # (B*N*N,)

        total = left_flat.numel()
        BLOCK_SIZE = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # The tensors are updated in‑place, reshape back to the original view
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Core contraction: batched GEMM (left @ rightᵀ) over the sequence dim
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) with contiguous memory layout
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_mat = torch.empty_like(left_mat)   # fp16 output buffer

    # strides for the kernel
    stride_lb, stride_li, stride_lk = left_mat.stride()
    stride_rb, stride_ri, stride_rk = right_mat.stride()
    stride_ob, stride_oi, stride_oj = out_mat.stride()

    # Tiling parameters (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_lb, stride_li, stride_lk,
        stride_rb, stride_ri, stride_rk,
        stride_ob, stride_oi, stride_oj,
        N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)      # (B, N, N, C)

    # -----------------------------------------------------------------
    # 8. Cast back to the original dtype
    # -----------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
