def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    P, N,
    stride_batch, stride_row, stride_col, 
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute coordinate offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block (M, K)
        left_ptrs = left_ptr + batch_id * stride_batch + offs_m[:, None] * stride_row + offs_k[None, :] * stride_col
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load right block (N, K) as transposed to (K, N)
        # Right originally shape (P, N, N). We need right_T such that we multiply left (M,K) * right_T(K,N)
        # So load right_T as right[:, offs_k, offs_n]?
        right_ptrs = right_ptr + batch_id * stride_batch + offs_k[:, None] * stride_row + offs_n[None, :] * stride_col

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to FP16 for compute‑heavy parts.
3. Linear projections (left/right) and gating projections (sigmoid) are computed
   with torch.nn.functional.linear.
4. If a mask is present (and not disabled) a single Triton kernel fuses
   projection, gating and mask for both left and right tensors.
5. Core TriMul reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (left @ rightᵀ) for each
   (batch, hidden) pair.  It is implemented with a custom Triton kernel
   that tiles the N×N matrices and uses tensor‑core friendly fp16·fp16→fp32
   dot products.
6. Output LayerNorm, multiplicative out‑gate and final linear projection.
7. Cast back to the original dtype (usually float32).

The implementation keeps all large tensor arithmetic in FP16 while
accumulating in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Kernel 1: fuse projection, gate and mask (both left and right)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B*N*N*H,)
    right_proj_ptr,  # fp16 * (B*N*N*H,)
    left_gate_ptr,   # fp16 * (B*N*N*H,)
    right_gate_ptr,  # fp16 * (B*N*N*H,)
    mask_ptr,        # fp16 * (B*N*N,)
    left_out_ptr,    # fp16 * (B*N*N*H,)
    right_out_ptr,   # fp16 * (B*N*N*H,)
    total_elems,     # i64 = B*N*N*H
    hidden_dim,      # i64 = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    in_bounds = offs < total_elems
    # mask is broadcasted over the hidden dimension
    mask_idx = offs // hidden_dim

    left_proj = tl.load(left_proj_ptr + offs, mask=in_bounds, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=in_bounds, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=in_bounds, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=in_bounds, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=in_bounds, other=1.0)

    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=in_bounds)
    tl.store(right_out_ptr + offs, right_out, mask=in_bounds)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """Fuse projection, gate and mask for both left and right tensors."""
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Kernel 2: batched matrix multiplication for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 * (P, N, N)
    right_ptr,         # fp16 * (P, N, N)
    out_ptr,           # fp16 * (P, N, N)
    P,                 # i64 = batch * hidden
    N,                 # i64 = sequence length
    stride_batch,      # i64 = N * N
    stride_row,        # i64 = N
    stride_col,        # i64 = 1
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # index over P = B * H
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block column

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block: shape (M, K)
        left_ptrs = left_ptr + pid_batch * stride_batch \
                    + offs_m[:, None] * stride_row \
                    + offs_k[None, :] * stride_col
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # Load right block transposed: we need (K, N)
        # right is (P, N, N); we read rows = k, cols = n
        right_ptrs = right_ptr + pid_batch * stride_batch \
                     + offs_k[:, None] * stride_row \
                     + offs_n[None, :] * stride_col
        right_T = tl.load(right_ptrs,
                          mask=mask_k[:, None] & mask_n[None, :],
                          other=0.0)

        # Dot-product accumulation (M,K)·(K,N) -> (M,N)
        acc += tl.dot(left, right_T)

    # Store the result tile
    out_ptrs = out_ptr + pid_batch * stride_batch \
               + offs_m[:, None] * stride_row \
               + offs_n[None, :] * stride_col
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation (AlphaFold3).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] (0/1) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast all linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16) and gates
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (fused Triton kernel if needed)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is contiguous FP16
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # Simple element‑wise fusion (mask may still be applied if provided but ignored)
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM using a custom Triton kernel.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Merge batch and hidden dimensions: (B, H, N, N) → (P, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    P = B * H
    left_flat = left_perm.view(P, N, N)
    right_flat = right_perm.view(P, N, N)

    out_flat = torch.empty_like(left_flat)

    # Strides for a contiguous (P, N, N) tensor
    stride_batch = N * N
    stride_row = N
    stride_col = 1

    # Tiling parameters – chosen to fit typical B200 resources
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        P,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        P,
        N,
        stride_batch,
        stride_row,
        stride_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
