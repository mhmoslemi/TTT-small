@triton.jit
def triton_trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_h = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute batch and hidden index
    b = batch_h // H
    h = batch_h % H

    # Compute base offset for this batch/h
    offset = b * stride_left_batch + h * stride_left_h

    # Compute offsets within M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for boundary
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers for A and B tiles
    a_ptrs = left_ptr + offset + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
    b_ptrs = right_ptr + offset + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + offs_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k_offset < N), other=0.0)
        b = tl.load(b_ptrs, mask=(k_offset[:, None] < N) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # Write back result
    c_ptrs = out_ptr + offset + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the reference PyTorch module:
1. LayerNorm over the input channel dimension (fp32).
2. Linear projections to the hidden dimension (fp16).
3. Three sigmoid gates (left, right, out) from the normalized input.
4. Fuse mask (if present) and the left/right gates in a single Triton kernel.
5. Core TriMul reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (A @ Bᵀ) in Triton, where each hidden
   channel is treated as an independent batch.
6. LayerNorm over the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original channel dimension.
All heavy arithmetic is performed in fp16 with fp32 accumulation for
numerical stability.  The final result is returned in the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask with left/right projections and their gates
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,            # const * left  [B,N,N,H]
    right_ptr,           # const * right [B,N,N,H]
    mask_ptr,            # const * mask  [B,N,N] (flattened)
    left_gate_ptr,       # const * left_gate  [B,N,N,H]
    right_gate_ptr,      # const * right_gate [B,N,N,H]
    left_out_ptr,        # * left_out  [B,N,N,H]
    right_out_ptr,       # * right_out [B,N,N,H]
    H: tl.constexpr,                # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load tensors
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Index into the mask (mask repeats for each hidden dim)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply mask and gates
    tl.store(left_out_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


# ----------------------------------------------------------------------
# Kernel 2: batched matrix multiplication for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    left_ptr,          # const * left  [B,N,N,H]
    right_ptr,         # const * right [B,N,N,H]
    out_ptr,           # * out  [B,N,N,H]
    B: tl.constexpr,   # batch size
    N: tl.constexpr,   # sequence length
    H: tl.constexpr,   # hidden dimension
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_h = tl.program_id(0)          # combined batch*hidden index
    pid_m   = tl.program_id(1)          # row tile
    pid_n   = tl.program_id(2)          # col tile

    # Decompose combined index
    b = batch_h // H
    h = batch_h % H

    # Base offset for this (b, h)
    base_l = b * stride_l_batch + h * stride_l_h
    base_r = b * stride_r_batch + h * stride_r_h
    base_o = b * stride_o_batch + h * stride_o_h

    # Tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers to the start of the tile
    a_ptrs = left_ptr + base_l + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
    b_ptrs = right_ptr + base_r + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        cur_k = k * BLOCK_K + offs_k
        k_mask = cur_k < N

        a = tl.load(a_ptrs, mask=mask_i[:, None] & k_mask[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=k_mask[:, None] & mask_j[None, :], other=0.0)
        # dot returns fp16, accumulate in fp32
        acc += tl.dot(a, b).to(tl.float32)

        # Advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_l_k
        b_ptrs += BLOCK_K * stride_r_k

    # Store the result
    c_ptrs = out_ptr + base_o + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, if True ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to fp16 for the heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Prepare mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_tensor = mask.to(torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6) Fuse mask and gates with Triton
    # ------------------------------------------------------------------
    total_elements = left.numel()               # B * N * N * H
    BLOCK = 1024

    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid_fuse = (triton.cdiv(total_elements, BLOCK),)
    _mask_gate_fusion_kernel[grid_fuse](
        left,
        right,
        mask_tensor.view(-1),
        left_gate,
        right_gate,
        left_out,
        right_out,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Core TriMul reduction: batched GEMM (left_out @ right_outᵀ)
    # ------------------------------------------------------------------
    out = torch.empty_like(left_out)

    # Strides for the tensors (B, N, N, H)
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h = left_out.stride()
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h = right_out.stride()
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h = out.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid_matmul = (
        B * hidden_dim,                     # combined batch*hidden
        triton.cdiv(N, BLOCK_M),           # tiles over rows
        triton.cdiv(N, BLOCK_N),           # tiles over cols
    )
    _trimul_matmul_kernel[grid_matmul](
        left_out,
        right_out,
        out,
        B,
        N,
        hidden_dim,
        stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
        stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
        stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 8) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9) Apply output gate and final projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
