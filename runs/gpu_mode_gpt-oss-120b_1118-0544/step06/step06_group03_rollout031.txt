"""
TriMul (outgoing) forward pass – Triton‑accelerated core reduction.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast tensors to fp16 for compute‑intensive ops.
3. Linear projections → `left_proj`, `right_proj`  (shape [B, N, N, H]).
4. Gate projections (`left_gate`, `right_gate`, `out_gate`) with sigmoid.
5. Apply mask (if any) and fuse with gates by simple element‑wise ops.
6. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM over the sequence dimension using a
   custom Triton kernel (one GEMM per (batch, hidden) pair).
7. Output LayerNorm over hidden dimension `H`, multiply by `out_gate`,
   and final linear projection back to the original dimension `dim`.
8. Cast result back to the input dtype (float32).

Only the heavy reduction in step 6 is written in Triton; the remaining
operations use highly‑optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    batch, M, N, K,                      # matrix sizes
    stride_batch_a, stride_am, stride_ak,  # strides for A
    stride_batch_b, stride_bk, stride_bn,  # strides for B (transposed)
    stride_batch_c, stride_cm, stride_cn,  # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for many independent (batch) matrices.

    A : (batch, M, K)   fp16
    B : (batch, N, K)   fp16  (interpreted as transposed inside kernel)
    C : (batch, M, N)   fp16
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for bk in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = bk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile [BLOCK_M, BLOCK_K]
        a_ptrs = (
            a_ptr
            + pid_batch * stride_batch_a
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load B tile [BLOCK_K, BLOCK_N] (B is read transposed)
        b_ptrs = (
            b_ptr
            + pid_batch * stride_batch_b
            + offs_k[:, None] * stride_bk   # k dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bn   # n dimension (col of Bᵀ)
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate
        acc += tl.dot(a, b)

    # Write back C
    c_ptrs = (
        c_ptr
        + pid_batch * stride_batch_c
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (float32/float16/bool) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    dtype_in = input_tensor.dtype
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B, N, N, H)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (+ sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask (if any) and gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16)          # (B,N,N)
        mask_fp16 = mask_fp16.unsqueeze(-1)         # (B,N,N,1) broadcast over H
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM across hidden dimension (out = left @ rightᵀ)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) and flatten batch & hidden into one dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B,H,N,N)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(-1, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(-1, N, N) # (B*H, N, N)

    out_flat = torch.empty_like(left_flat)

    batch = left_flat.shape[0]
    M = N
    N_dim = N
    K = N

    # Strides in elements
    stride_batch_a = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_batch_b = right_flat.stride(0)
    # B will be read transposed: swap the last two strides
    stride_bk = right_flat.stride(2)   # original column stride
    stride_bn = right_flat.stride(1)   # original row stride

    stride_batch_c = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # Tunable block sizes (experimentally good for typical B200 GPUs)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N_dim, BLOCK_N),
    )

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch,
        M,
        N_dim,
        K,
        stride_batch_a,
        stride_am,
        stride_ak,
        stride_batch_b,
        stride_bk,
        stride_bn,
        stride_batch_c,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
