def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bn, stride_bk,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_batch: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    # compute batch index
    batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_idx = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = block_idx // tl.cdiv(N, BLOCK_N)
    pid_n = block_idx % tl.cdiv(N, BLOCK_N)
    # compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        # compute masks
        a_mask = (offs_m < M) & (offs_k < K)
        b_mask = (offs_k < K) & (offs_n < N)
        # load A
        a_ptrs = a_ptr + batch * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=a_mask[:, None], other=0.0)
        # load B (transposed)
        # B original shape (batch, K, N)?? Actually we need B_T shape (batch, N, K) but we load transposed.
        # We treat b_ptr as pointer to right_mat (batch, K, N) = (batch,N,N) but we need B_T[i,j] = right_mat[j,i]
        # So we load using offset: row = offs_k, col = offs_n (since B_T row = k, col = j)
        b_ptrs = b_ptr + batch * batch_stride_b + offs_k[:, None] * stride_bn + offs_n[None, :] * stride_bk
        b = tl.load(b_ptrs, mask=b_mask[None, :], other=0.0)
        acc += tl.dot(a, b)
    # store

"""
TriMul (outgoing) Triton implementation
--------------------------------------
The module computes the outgoing triangle multiplicative update used in AlphaFold3.
Given an input tensor x of shape [B, N, N, C] it performs:

1. LayerNorm on the last dimension.
2. Linear projections (left/right) and gating (left/right/out) without bias.
3. Optional mask + gate fusion via a small Triton kernel.
4. Core contraction:
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   which is a batched GEMM (L @ Rᵀ) over the hidden dimension h.
   This is implemented with a tiled Triton kernel for high throughput.
5. Output LayerNorm, multiplication by out_gate and final linear projection.
All heavy compute is done in float16; the final result is cast back to the
input dtype (usually float32).

Only the mask‑gate fusion and the batched GEMM are written in Triton; the rest
uses efficient PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask and gates (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,          # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,
    mask_ptr,                     # [B,N,N] flattened
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # loads
    left      = tl.load(left_ptr + offs,      mask=active, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr + offs,mask=active, other=0.0)

    if HAS_MASK:
        # each hidden slice shares the same mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs,  left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the core TriMul contraction
#    out = left @ rightᵀ   (per hidden channel)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # tensors shaped (B*H, N, N)
    stride_batch_a, stride_am, stride_ak,    # strides for A
    stride_batch_b, stride_bn, stride_bk,    # strides for B (will be accessed transposed)
    stride_batch_c, stride_cm, stride_cn,    # strides for C
    M, N, K,                                 # all equal to seq_len
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)          # tile over M & N
    batch = tl.program_id(1)        # batch index (B*H)

    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # mask out-of-bounds
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # pointers
        a_ptrs = a_ptr + batch * stride_batch_a \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch * stride_batch_b \
                 + offs_k[:, None] * stride_bn \
                 + offs_n[None, :] * stride_bk  # note: B accessed transposed

        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # store result
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_ptr + batch * stride_batch_c \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # Config
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the rest of the computation
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (bias‑less) and gating
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left  = F.linear(x, left_proj_w)      # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)     # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4. Fuse mask & gates via Triton (in‑place)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Prepare tensors for the kernel
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # Simple element‑wise fusion without mask
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via batched GEMM (TriMul)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for the kernel
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # allocate output tensor
    out_mat = torch.empty_like(left_mat)

    # strides
    stride_batch_a = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_batch_b = right_mat.stride(0)
    stride_bn = right_mat.stride(1)   # row stride (original row)
    stride_bk = right_mat.stride(2)   # column stride

    stride_batch_c = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # block sizes (tuned for B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_tiles_m * num_tiles_n, B * H)

    _tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_batch_a, stride_am, stride_ak,
        stride_batch_b, stride_bn, stride_bk,
        stride_batch_c, stride_cm, stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------
    # 7. Cast back to original dtype and return
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
