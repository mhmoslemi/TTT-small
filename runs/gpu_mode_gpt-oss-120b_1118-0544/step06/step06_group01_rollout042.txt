"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the PyTorch reference but replaces the
computationally‑heavy contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
with a custom Triton kernel.  The kernel computes a batched
matrix multiplication for each (batch, hidden) pair, avoiding the
expensive view/contiguous reshapes used in the reference.
All heavy arithmetic is performed in fp16; LayerNorms and the final
linear projection are done in fp32 for numerical stability.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the core contraction (batched GEMM over the sequence dim)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,               # const * fp16   [B, N, N, H]
    right_ptr,              # const * fp16   [B, N, N, H]
    out_ptr,                # * fp16         [B, N, N, H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: for each (b, h) compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    """
    pid_i = tl.program_id(0)            # tile over output rows (i)
    pid_j = tl.program_id(1)            # tile over output cols (j)
    pid_bh = tl.program_id(2)           # combined batch * hidden index

    # decode batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # offsets within the N dimension
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this (b, h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator in fp32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, N, BLOCK_K):
        k_off = k + tl.arange(0, BLOCK_K)

        # mask for out‑of‑bounds loads
        a_mask = (i_off[:, None] < N) & (k_off[None, :] < N)
        b_mask = (k_off[:, None] < N) & (j_off[None, :] < N)

        # load tiles
        a = tl.load(
            left_base
            + i_off[:, None] * stride_left_i
            + k_off[None, :] * stride_left_k,
            mask=a_mask,
            other=0.0,
        )
        b = tl.load(
            right_base
            + k_off[:, None] * stride_right_k
            + j_off[None, :] * stride_right_j,
            mask=b_mask,
            other=0.0,
        )
        # dot product (uses tensor‑cores when possible)
        acc += tl.dot(a, b)

    # write result back in fp16
    c_mask = (i_off[:, None] < N) & (j_off[None, :] < N)
    tl.store(
        out_base
        + i_off[:, None] * stride_out_i
        + j_off[None, :] * stride_out_j,
        acc.to(tl.float16),
        mask=c_mask,
    )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (torch.Tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)   # fp16 for the heavy part

    # -----------------------------------------------------------------
    # 2. Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gating (all in fp16)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 4. Apply mask (if present) and gates
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_ = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * left_gate * mask_
        right = right * right_gate * mask_
    else:
        left = left * left_gate
        right = right * right_gate

    left = left.contiguous()
    right = right.contiguous()

    # -----------------------------------------------------------------
    # 5. Core contraction via Triton (batched GEMM over sequence dim)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)   # fp16 buffer for the pre‑norm result

    # strides (in elements, not bytes)
    left_stride = left.stride()
    right_stride = right.stride()
    out_stride = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # pid_i
        triton.cdiv(N, BLOCK_N),          # pid_j
        B * H,                            # pid_bh (batch * hidden)
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        left_stride[0],
        left_stride[1],
        left_stride[2],
        left_stride[3],
        right_stride[0],
        right_stride[1],
        right_stride[2],
        right_stride[3],
        out_stride[0],
        out_stride[1],
        out_stride[2],
        out_stride[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out_gate, and final linear projection
    # -----------------------------------------------------------------
    # Convert to fp32 for LayerNorm (more stable)
    out_fp32 = out.to(torch.float32)

    # Output LayerNorm
    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
    )

    # Apply out_gate (casted to fp32)
    out_fp32 = out_fp32 * out_gate.to(torch.float32)

    # Final linear projection back to original feature dimension
    out_fp32 = F.linear(out_fp32, to_out_w.to(torch.float32))   # (B, N, N, dim)

    # Cast back to the original dtype (usually float32)
    if out_fp32.dtype != dtype_in:
        out_fp32 = out_fp32.to(dtype_in)

    return out_fp32
