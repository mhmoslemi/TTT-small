    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"].to(device=device, dtype=torch.float32), bias=weights["norm.bias"].to(device=device, dtype=torch.float32))

    x = x.to(torch.float16)

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"].to(device, torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(device, torch.float16))

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(device, torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(device, torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device, torch.float16)))

    # If mask is None or nomask ignore; else convert mask to float16
    if mask is None or nomask:
        mask_tensor = torch.ones((input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]), dtype=torch.float16, device=device)
    else:

    # Permute and reshape for kernel
    B, N, _, _ = left.shape
    # Left, Right, Gates have shape (B,N,N,H)
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left_gate = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.permute(0, 3, 1, 2).contiguous()
    mask_tensor = mask_tensor.contiguous()  # shape (B,N,N)

    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)
    right_flat = right.view(BH, N, N)
    left_gate_flat = left_gate.view(BH, N, N)
    right_gate_flat = right_gate.view(BH, N, N)
    # out_gate remains (B, H, N, N) for later multiplication; we need same shape after we compute out.

    # Allocate output tensor

```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_ptr,
    B, N, H,
    stride_al, stride_ai, stride_ak,
    stride_bl, stride_bi, stride_bk,
    stride_gl, stride_gi, stride_gk,
    stride_rl, stride_ri, stride_rk,
    stride_ol, stride_oi, stride_ok,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    batch_idx = tl.program_id(0)  # runs over B*H
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Compute starting offset for each batch
    # base offset for left and right and gates
    a_batch_offset = batch_idx * stride_al
    b_batch_offset = batch_idx * stride_bl
    gl_batch_offset = batch_idx * stride_gl
    rg_batch_offset = batch_idx * stride_rl
    out_batch_offset = batch_idx * stride_ol

    # compute batch index for mask: b = batch_idx // H
    b_idx = batch_idx // H

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        k_mask = cur_k < N

        # Load left tile (M,K)
        a = tl.load(
            left_ptr + a_batch_offset + offs_m[:, None] * stride_ai + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & k_mask[None, :],
            other=0.0
        )
        a = a.to(tl.float32)
        # Load left gate
        a_gate = tl.load(
            left_gate_ptr + gl_batch_offset + offs_m[:, None] * stride_gi + cur_k[None, :] * stride_gk,
            mask=mask_m[:, None] & k_mask[None, :],
            other=0.0
        )
        a_gate = a_gate.to(tl.float32)
        # Load mask for left
        mask_left = tl.load(
            mask_ptr + b_idx * N * N + offs_m[:, None] * N + cur_k[None, :],
            mask=mask_m[:, None] & k_mask[None, :],
            other=0.0
        )
        mask_left = mask_left.to(tl.float32)

        a = a * a_gate * mask_left

        # Load right tile transposed (K,N)
        b = tl.load(
            right_ptr + b_batch_offset + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bi,
            mask=k_mask[:, None] & mask_n[None, :],
            other=0.0
        )
        b = b.to(tl.float32)
        # Load right gate
        b_gate = tl.load(
            right_gate_ptr + rg_batch_offset + cur_k[:, None] * stride_rk + offs_n[None, :] * stride_ri,
            mask=k_mask[:, None] & mask_n[None, :],
            other=0.0
        )
        b_gate = b_gate.to(tl.float32)
        # Load mask for right
        mask_right = tl.load(
            mask_ptr + b_idx * N * N + offs_n[None, :] * N + cur_k[:, None],
            mask=mask_n[None, :] & k_mask[:, None],
            other=0.0
        )
        mask_right = mask_right.to(tl.float32)

        b = b * b_gate * mask_right

        # dot product: a (M,K) dot b (K,N) => (M,N)
        acc += tl.dot(a, b)

    # Store result
    tl.store(
        out_ptr + out_batch_offset + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_ok,
        acc,
        mask=mask_m[:, None] & mask_n[None, :]
    )

    )

"""custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the reference PyTorch TriMul module:

1. Layer‑normalize the input tensor.
2. Linear projections (left/right) from C → H.
3. Compute three sigmoid gates (left, right, out).
4. Fuse mask, left‑gate and right‑gate and perform the core
   reduction   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched matrix multiplication using a Triton GEMM kernel.
5. Apply a second LayerNorm on the hidden dimension H.
6. Multiply by the output gate and project back to the original
   channel dimension C.

Only the mask‑gate fusion and the O(N³) reduction are executed on
Triton; all other cheap operations are done with PyTorch for clarity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr,            # (B*H, N, N) fp16
    left_gate_ptr, right_gate_ptr,  # (B*H, N, N) fp16
    mask_ptr,                       # (B, N, N)   fp16
    out_ptr,                        # (B*H, N, N) fp32
    B, N, H,                        # problem sizes
    # strides for the tensors
    stride_al, stride_ai, stride_ak,   # left  batch, i, k
    stride_br, stride_bi, stride_bk,   # right batch, j, k
    stride_gl, stride_gi, stride_gk,   # left_gate
    stride_rl, stride_ri, stride_rk,   # right_gate
    stride_ml, stride_mi, stride_mk,   # mask
    stride_ol, stride_oi, stride_ok,   # out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM with per‑element mask and gate fusion.

    Computes for each batch (= original batch × hidden dim):
        C[i, j] = Σ_k (A[i, k] * gA[i, k] * m[i, k]) *
                 (B[j, k] * gB[j, k] * m[j, k])
    where A/B are the left/right projections, gA/gB the sigmoid gates,
    and m the binary mask.
    """
    batch_idx = tl.program_id(0)          # 0 … B*H-1
    pid_m = tl.program_id(1)              # row tile
    pid_n = tl.program_id(2)              # col tile

    # -------------------- tile offsets --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base offsets for the current batch
    a_batch_off = batch_idx * stride_al
    b_batch_off = batch_idx * stride_br
    gl_batch_off = batch_idx * stride_gl
    rl_batch_off = batch_idx * stride_rl
    out_batch_off = batch_idx * stride_ol

    # original batch index (before hidden‑dim split) needed for mask
    bidx = batch_idx // H

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- reduction over k --------------------
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        kmask = cur_k < N

        # ---- left tile (M × K) ----
        a = tl.load(
            left_ptr + a_batch_off
                     + offs_m[:, None] * stride_ai
                     + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & kmask[None, :],
            other=0.0
        )
        a = a.to(tl.float32)

        a_gate = tl.load(
            left_gate_ptr + gl_batch_off
                         + offs_m[:, None] * stride_gi
                         + cur_k[None, :] * stride_gk,
            mask=mask_m[:, None] & kmask[None, :],
            other=0.0
        )
        a_gate = a_gate.to(tl.float32)

        mask_left = tl.load(
            mask_ptr + bidx * stride_ml
                     + offs_m[:, None] * stride_mi
                     + cur_k[None, :] * stride_mk,
            mask=mask_m[:, None] & kmask[None, :],
            other=0.0
        )
        mask_left = mask_left.to(tl.float32)

        a = a * a_gate * mask_left

        # ---- right tile transposed (K × N) ----
        b = tl.load(
            right_ptr + b_batch_off
                      + cur_k[:, None] * stride_bk
                      + offs_n[None, :] * stride_bi,
            mask=kmask[:, None] & mask_n[None, :],
            other=0.0
        )
        b = b.to(tl.float32)

        b_gate = tl.load(
            right_gate_ptr + rl_batch_off
                           + cur_k[:, None] * stride_rk
                           + offs_n[None, :] * stride_ri,
            mask=kmask[:, None] & mask_n[None, :],
            other=0.0
        )
        b_gate = b_gate.to(tl.float32)

        mask_right = tl.load(
            mask_ptr + bidx * stride_ml
                     + offs_n[None, :] * stride_mi
                     + cur_k[:, None] * stride_mk,
            mask=mask_n[None, :] & kmask[:, None],
            other=0.0
        )
        mask_right = mask_right.to(tl.float32)

        b = b * b_gate * mask_right

        # ---- accumulate block result ----
        acc += tl.dot(a, b)   # a: (M,K), b: (K,N) → (M,N)

    # -------------------- write back --------------------
    tl.store(
        out_ptr + out_batch_off
                 + offs_m[:, None] * stride_oi
                 + offs_n[None, :] * stride_ok,
        acc,
        mask=mask_m[:, None] & mask_n[None, :]
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters
            - config: dict with keys
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, if True ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))     # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Prepare mask (broadcasted over hidden dimension)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is None or nomask:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(torch.float16)

    # ------------------------------------------------------------------
    # 5) Rearrange tensors for batched GEMM (shape: (B*H, N, N))
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()        # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left_gate = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)
    right_flat = right.view(BH, N, N)
    left_gate_flat = left_gate.view(BH, N, N)
    right_gate_flat = right_gate.view(BH, N, N)
    out_gate_flat = out_gate.view(BH, N, N)

    # ------------------------------------------------------------------
    # 6) Allocate output tensor for the reduction (float32)
    # ------------------------------------------------------------------
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7) Triton kernel launch (batched GEMM + mask/gate fusion)
    # ------------------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    grid = (BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_tensor,
        out_flat,
        B, N, hidden_dim,
        # strides
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),# B
        left_gate_flat.stride(0), left_gate_flat.stride(1), left_gate_flat.stride(2),  # G_left
        right_gate_flat.stride(0), right_gate_flat.stride(1), right_gate_flat.stride(2),# G_right
        mask_tensor.stride(0), mask_tensor.stride(1), mask_tensor.stride(2),          # mask
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),                  # output
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9) Second LayerNorm over the hidden dimension H
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float32)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # ------------------------------------------------------------------
    # 10) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final projection back to original channel dimension C
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float32)
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 12) Cast back to original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
