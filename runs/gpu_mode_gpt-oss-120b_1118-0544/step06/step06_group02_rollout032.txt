@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_Ab, stride_Al, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute masks
    mask_m = off_m < M
    mask_n = off_n < N

    # pointers to start of A and B for this batch
    A_batch_ptr = A_ptr + batch_idx * stride_Ab
    B_batch_ptr = B_ptr + batch_idx * stride_Bb

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        off_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # load A block [M, K]
        a = tl.load(A_batch_ptr + off_m[:, None] * stride_Al + off_k[None, :] * stride_Ak,
                    mask=mask_m[:, None] & (off_k < K), other=0.0)

        # load B block [K, N]
        b = tl.load(B_batch_ptr + off_k[:, None] * stride_Bk + off_n[None, :] * stride_Bn,
                    mask=(off_k < K)[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # write back
    c = acc.to(tl.float16)
    C_batch_ptr = C_ptr + batch_idx * stride_Cb
    tl.store(C_batch_ptr + off_m[:, None] * stride_Cm + off_n[None, :] * stride_Cn,
             c,
             mask=mask_m[:, None] & mask_n[None, :])
```

@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_Ab, stride_Al, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute masks for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to batch start
    a_ptr = A_ptr + batch_idx * stride_Ab
    b_ptr = B_ptr + batch_idx * stride_Bb

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        # range of k indices
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # mask for k within K
        mask_k = offs_k < K

        # load A block [BLOCK_M, BLOCK_K]
        a = tl.load(
            a_ptr + offs_m[:, None] * stride_Al + offs_k[None, :] * stride_Ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B block [BLOCK_K, BLOCK_N]
        b = tl.load(
            b_ptr + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # accumulate
        acc += tl.dot(a, b)

    # write back result
    c = acc.to(tl.float16)
    c_ptr = C_ptr + batch_idx * stride_Cb
    tl.store(
        c_ptr + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )

"""
TriMul (outgoing) forward pass – Triton‑fused batched GEMM.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) with provided LN parameters.
2. Cast to fp16 for the compute‑intensive part.
3. Linear projections → left/right tensors of shape (B,N,N,H).
4. Compute sigmoid gates and (optionally) apply the pairwise mask.
5. Fuse the O(N³·H) reduction using a custom Triton batched‑gemm kernel:
     out[b,h,i,j] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   The hidden dimension H is merged into the batch dimension, yielding
   a single kernel that computes B*H independent GEMMs of size N×N.
6. Reshape back to (B,N,N,H) and apply a second LayerNorm.
7. Multiply by the output gate and a final linear projection to C.
8. Cast the result back to the input dtype (float32).

The kernel uses fp16 inputs, fp32 accumulation and Tensor‑Core‑friendly
tiling (64×64×32) to achieve high throughput on an NVIDIA B200.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_Ab, stride_Al, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   (A: [batch,M,K], B: [batch,K,N])"""
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_ptr = A_ptr + batch_idx * stride_Ab
    b_ptr = B_ptr + batch_idx * stride_Bb

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        mask_k = offs_k < K

        # A block [BLOCK_M, BLOCK_K]
        a = tl.load(
            a_ptr + offs_m[:, None] * stride_Al + offs_k[None, :] * stride_Ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B block [BLOCK_K, BLOCK_N]
        b = tl.load(
            b_ptr + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # accumulate (Tensor Core fp16·fp16→fp32)
        acc += tl.dot(a, b)

    # write back in fp16
    c = acc.to(tl.float16)
    c_ptr = C_ptr + batch_idx * stride_Cb
    tl.store(
        c_ptr + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, C]
        - mask        : torch.Tensor of shape [B, N, N] (or None)
        - weights     : dict of model parameters (float32)
        - config      : dict with keys:
            * "dim" (int)        – input channel dimension C
            * "hidden_dim" (int) – hidden dimension H
            * "nomask" (bool)    – if True, ignore mask

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ---------------------- unpack ----------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ---------------------- first LayerNorm ----------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)  # fp16 for the heavy part

    # ---------------------- cast weights ----------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ---------------------- linear projections ----------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ---------------------- gates ----------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()  # (B,N,N,H)

    # ---------------------- mask handling ----------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)          # (B,N,N,1)
        left = left_proj * mask_fp16 * left_gate
        right = right_proj * mask_fp16 * right_gate
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ---------------------- prepare for batched GEMM ----------------------
    B, N, _, H = left.shape
    # move hidden dim to second axis and make contiguous
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch = B * H
    left = left.view(batch, N, N)                 # (B*H, N, N)
    right = right.view(batch, N, N)               # (B*H, N, N)
    # transpose right for standard GEMM A @ B
    right_T = right.transpose(2, 1).contiguous()  # (B*H, N, N)

    # output buffer
    out = torch.empty_like(left)  # (B*H, N, N), fp16

    # strides (int)
    a_stride_b, a_stride_m, a_stride_k = left.stride()
    b_stride_b, b_stride_k, b_stride_n = right_T.stride()
    c_stride_b, c_stride_m, c_stride_n = out.stride()

    # ---------------------- launch batched GEMM kernel ----------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_gemm_kernel[grid](
        left,
        right_T,
        out,
        batch,
        N,
        N,
        N,
        a_stride_b,
        a_stride_m,
        a_stride_k,
        b_stride_b,
        b_stride_k,
        b_stride_n,
        c_stride_b,
        c_stride_m,
        c_stride_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------------------- reshape back ----------------------
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ---------------------- second LayerNorm ----------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ---------------------- output gating ----------------------
    out = out * out_gate

    # ---------------------- final linear projection ----------------------
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # ---------------------- cast back to original dtype ----------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
