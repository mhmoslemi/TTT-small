```python
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    GROUPS, N,
    stride_g_left, stride_i_left, stride_k_left,
    stride_g_right, stride_j_right, stride_k_right,
    stride_g_out, stride_i_out, stride_j_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    group_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute starting indices
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    # Masks for out of bounds
    mask_i = i < N
    mask_j = j < N

    # Pointers for left (M x K)
    left_ptrs = left_ptr + group_idx * stride_g_left + i[:, None] * stride_i_left + (k[None, :] + 0) * stride_k_left
    # Pointers for right (K x N) (since we need transpose)
    # We want tile shape (K, N) where first dimension is k, second is j.
    right_ptrs = right_ptr + group_idx * stride_g_right + (k[:, None]) * stride_k_right + j[None, :] * stride_j_right

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles
    for start_k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = start_k * BLOCK_K + k
        k_mask = cur_k < N

        # Load left tile
        left = tl.load(left_ptrs, mask=mask_i[:, None] & k_mask[None, :], other=0.0)
        # Load right tile
        right = tl.load(right_ptrs, mask=k_mask[:, None] & mask_j[None, :], other=0.0)

        # Cast to fp32 for accumulation
        left = left.to(tl.float32)
        right = right.to(tl.float32)

        acc += tl.dot(left, right)

        # Update pointers for next k tile
        left_ptrs += BLOCK_K * stride_k_left
        right_ptrs += BLOCK_K * stride_k_right

    # Store result
    out_ptrs = out_ptr + group_idx * stride_g_out + i[:, None] * stride_i_out + j[None, :] * stride_j_out
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the PyTorch reference but moves the two
memory‑intensive parts to Triton:

1. **Mask & gate fusion** – a 1‑D kernel multiplies the projected
   tensors `left` / `right` by their respective gates and (optionally)
   by the pairwise mask, writing the results back in‑place.

2. **Core contraction** – a batched GEMM kernel computes
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] for every
   batch `b` and hidden channel `d`.  The tensor is reshaped to
   (B·H, N, N) so each group is a standard matrix‑multiply with
   left @ rightᵀ.

All other operations (LayerNorms, linear projections, the final
output gate and projection) are performed with PyTorch (float16
computations, cast back to the input dtype at the end).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left = tl.load(left_ptr + offs, mask=active, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=active, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=tl.float16(0.0))

    if HAS_MASK:
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_l_g, stride_l_i, stride_l_k,
    stride_r_g, stride_r_j, stride_r_k,
    stride_o_g, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    group_id = tl.program_id(0)          # 0 … (B*H-1)
    pid_m = tl.program_id(1)             # block row
    pid_n = tl.program_id(2)             # block col

    # start offsets for this block
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left tile: (M, K)
        left_ptrs = (
            left_ptr
            + group_id * stride_l_g
            + offs_i[:, None] * stride_l_i
            + cur_k[None, :] * stride_l_k
        )
        left = tl.load(
            left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=tl.float16(0.0)
        )
        left = left.to(tl.float32)

        # right tile transposed: (K, N) -> we load as (K, N)
        right_ptrs = (
            right_ptr
            + group_id * stride_r_g
            + cur_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_j
        )
        right = tl.load(
            right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=tl.float16(0.0)
        )
        right = right.to(tl.float32)

        acc += tl.dot(left, right)

    # store result
    out_ptrs = (
        out_ptr
        + group_id * stride_o_g
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)        (float32)
            - mask:        torch.Tensor of shape (B, N, N) or None
            - weights:     dict of model parameters (float32)
            - config:      dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, optional (True ⇒ ignore mask)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (bias‑less) in FP16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)          # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)        # (B,N,N,hidden_dim)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4. Fuse mask and gates (Triton kernel) – optional
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure contiguity
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp = mask.to(device).to(torch.float16).contiguous()

        # flatten tensors for the 1‑D kernel
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_fp.view(-1)  # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )

        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # simple element‑wise gating when mask is absent
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ rightᵀ) via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # reshape to (B*H, N, N) where the hidden dimension is merged with batch
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    G = B * H
    left_mat = left_perm.view(G, N, N)
    right_mat = right_perm.view(G, N, N)

    out_mat = torch.empty_like(left_mat)  # (G, N, N), fp16

    # strides for the reshaped tensors (they are contiguous)
    stride_l_g, stride_l_i, stride_l_k = left_mat.stride()
    stride_r_g, stride_r_j, stride_r_k = right_mat.stride()
    stride_o_g, stride_o_i, stride_o_j = out_mat.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        G,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        stride_l_g,
        stride_l_i,
        stride_l_k,
        stride_r_g,
        stride_r_j,
        stride_r_k,
        stride_o_g,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (still FP16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # (B,N,N,dim) in FP16

    # ------------------------------------------------------------------
    # 7. Cast back to original dtype and return
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
