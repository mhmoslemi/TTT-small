"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor x ∈ [B,N,N,C] (fp32 for stability).
2. Cast to fp16 and apply linear projections (left/right) → hidden_dim.
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. Fuse mask (if present) and left/right gates in a Triton kernel:
       left_out  = left  * mask * left_gate
       right_out = right * mask * right_gate
5. Perform the core TriMul reduction:
       out[b,i,j,h] = Σ_k left_out[b,i,k,h] * right_out[b,j,k,h]
   Implemented as a batched fp16 GEMM (left_out @ right_outᵀ) using a custom
   Triton kernel that processes (B*hidden_dim) independent matrix
   multiplications.
6. Apply a second LayerNorm over the hidden dimension.
7. Multiply by the output gate and project back to the original channel
   dimension C with a final linear layer.
8. Cast the result back to the input dtype.

Only the mask‑gate fusion and the batched GEMM are written in Triton; all
other operations use PyTorch (which internally calls cuBLAS for the
linear layers). This mix provides both correctness and high performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask and gates (element‑wise)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # const fp16* left   (B,N,N,H) flattened
    right_ptr,         # const fp16* right  (B,N,N,H) flattened
    mask_ptr,          # const fp16* mask   (B,N,N)    flattened
    left_gate_ptr,     # const fp16* left_gate
    right_gate_ptr,    # const fp16* right_gate
    left_out_ptr,      # fp16* left_out
    right_out_ptr,     # fp16* right_out
    H: tl.constexpr,          # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # Load values (masked)
    l  = tl.load(left_ptr + offs,   mask=valid)
    r  = tl.load(right_ptr + offs,  mask=valid)
    lg = tl.load(left_gate_ptr + offs,  mask=valid)
    rg = tl.load(right_gate_ptr + offs, mask=valid)

    # The mask repeats for each hidden channel ⇒ integer division by H
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=valid)

    # Apply mask and gates
    tl.store(left_out_ptr + offs,  l * m * lg, mask=valid)
    tl.store(right_out_ptr + offs, r * m * rg, mask=valid)


# ----------------------------------------------------------------------
# Kernel 2: batched matrix multiplication (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,      # pointers to A, B, C
    M, N, K,                  # matrix dims (M×K)·(K×N) → (M×N)
    stride_ai, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_bi, stride_bk, stride_bn,   # strides for B (batch, K, N)
    stride_ci, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    pid = tl.program_id(0)

    # Compute batch, block‑row, block‑col indices
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    batch = tl.cdiv(pid, num_pid_m * num_pid_n)               # which batch
    pid_rem = pid - batch * (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n
    pid_n = pid_rem % num_pid_n

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Create a mask for out‑of‑bounds threads
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator in fp32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for the current tile of A and B
        a_ptrs = a_ptr + batch * stride_ai + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        b_ptrs = b_ptr + batch * stride_bi + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        # Load tiles (apply masks)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute dot product and accumulate
        acc += tl.dot(a, b)

    # Write back the result (cast to fp16)
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + batch * stride_ci + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (LayerNorm weights/biases,
                       linear weights for all projections and output)
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, if True ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    B, N, _, C = input_tensor.shape
    assert C == dim, "config dim does not match input tensor last dimension"

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to fp16 for the bulk of the computation
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Prepare mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_tensor = mask.to(torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6) Fuse mask and gates with Triton
    # ------------------------------------------------------------------
    total_elements = left.numel()                # B * N * N * H
    BLOCK = 1024

    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = (triton.cdiv(total_elements, BLOCK),)
    _mask_gate_fusion_kernel[grid](
        left,
        right,
        mask_tensor.view(-1),        # flatten mask (B*N*N)
        left_gate,
        right_gate,
        left_out,
        right_out,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Core reduction: batched GEMM (left_out @ right_outᵀ)
    # ------------------------------------------------------------------
    # Reshape to (batch, M, K) and (batch, K, N)
    left_mat = left_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (B*H, M, K)
    right_mat = right_out.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)  # (B*H, K, N)

    # Allocate output tensor
    out_mat = torch.empty_like(left_mat)

    # Strides for the contiguous tensors
    stride_ai, stride_am, stride_ak = left_mat.stride()
    stride_bi, stride_bk, stride_bn = right_mat.stride()
    stride_ci, stride_cm, stride_cn = out_mat.stride()

    # Block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_batches = B * hidden_dim
    grid = (total_batches * num_pid_m * num_pid_n,)

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                         # M, N, K (all equal N)
        stride_ai, stride_am, stride_ak,
        stride_bi, stride_bk, stride_bn,
        stride_ci, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9) Second LayerNorm over hidden dimension (use fp32 weights)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)  # layernorm in higher precision
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w.to(device, torch.float32),
        bias=to_out_norm_b.to(device, torch.float32),
    )
    out = out.to(torch.float16)

    # ------------------------------------------------------------------
    # 10) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 12) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
