"""
TriMul outgoing kernel with Triton mask+gate fusion.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_fusion(
    left_ptr, right_ptr,
    mask_ptr,
    left_gate_ptr, right_gate_ptr,
    left_out_ptr, right_out_ptr,
    H: tl.constexpr,
    total_elem: tl.constexpr,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elem

    # Load
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # mask index: remove H dimension
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    tl.store(left_out_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)

def custom_kernel(data):
    """
    Forward pass of outgoing Triangle Multiplicative Update (TriMul).
    Input:
        data = (x, mask, weights, config)
        x: torch.Tensor [B, N, N, C] float32/float16 on CUDA
        mask: torch.Tensor [B, N, N] (binary) or None
        weights: dict of parameters
        config: dict with "dim", "hidden_dim", "nomask" (optional)
    Returns:
        torch.Tensor [B, N, N, C] same dtype as input.
    """
    x, mask, weights, config = data
    device = x.device
    orig_dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. LayerNorm (fp32 for stability)
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # Cast to fp16 for compute intensive ops
    x = x.to(torch.float16)

    # 2. Load weights (fp16)
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # 3. Linear projections
    # shape: (B,N,N,hidden_dim)
    left = F.linear(x, left_proj_w)
    right = F.linear(x, right_proj_w)

    # 4. Gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # 5. Prepare mask (broadcast over hidden dim)
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16)
    else:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # 6. Fuse mask and gates via Triton
    total_elements = left.numel()  # B * N * N * hidden_dim
    BLOCK = 1024  # adjust block size for good occupancy
    grid = (triton.cdiv(total_elements, BLOCK),)

    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    _mask_gate_fusion[grid](
        left, right,
        mask_fp16.view(-1),
        left_gate, right_gate,
        left_out, right_out,
        hidden_dim,
        total_elements,
        BLOCK,
        num_warps=4,
    )

    # 7. Core reduction: batched matrix multiplication (B*hidden_dim, N, N)
    # reshape to (B*hidden_dim, N, N)
    left_mat = left_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(-2, -1))  # (B*hidden_dim, N, N)

    # reshape back to (B, N, N, hidden_dim)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # 8. Second LayerNorm over hidden dim
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 9. Apply output gate
    out = out * out_gate

    # 10. Final linear projection back to original dim
    out = F.linear(out, to_out_w)  # (B,N,N,dim)

    # Cast back to original dtype if needed
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out

"""
TriMul (outgoing) forward pass with Triton‑accelerated mask & gate fusion.

Algorithm
---------
1. Layer‑normalize the input tensor x ∈ [B, N, N, C] (fp32 for stability).
2. Cast to fp16 and apply linear projections:
   left  = x @ left_proj_weightᵀ   → [B, N, N, H]
   right = x @ right_proj_weightᵀ  → [B, N, N, H]
3. Compute sigmoid gates (left_gate, right_gate, out_gate) from the same
   normalized input using separate linear layers.
4. Fuse mask (if present) and the left/right gates in a Triton kernel:
   left_out  = left  * mask * left_gate
   right_out = right * mask * right_gate
5. Core reduction: for each hidden dimension h,
   out[b,i,j,h] = Σ_k left_out[b,i,k,h] * right_out[b,j,k,h].
   This is performed as a batched matrix multiplication
   (B·H, N, N) via torch.bmm.
6. Apply a second LayerNorm over the hidden dimension H.
7. Multiply by the output gate and linearly project back to the original
   channel dimension C.
8. Cast the result back to the original dtype of the input.

Only the mask‑gate fusion is implemented in Triton; all other ops use
high‑performance torch kernels (cuBLAS for GEMM, cuDNN for LayerNorm).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion(
    left_ptr,               # const * left  [B,N,N,H]
    right_ptr,              # const * right [B,N,N,H]
    mask_ptr,               # const * mask  [B,N,N] flattened
    left_gate_ptr,          # const * left_gate  [B,N,N,H]
    right_gate_ptr,         # const * right_gate [B,N,N,H]
    left_out_ptr,           # * left_out  [B,N,N,H]
    right_out_ptr,          # * right_out [B,N,N,H]
    H: tl.constexpr,        # hidden dimension
    TOTAL_ELEMS: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask and sigmoid gates:
    left_out  = left  * mask * left_gate
    right_out = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL_ELEMS

    # Load data
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Compute mask index (mask repeats for each hidden dim)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply mask and gates
    tl.store(left_out_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (x, mask, weights, config)
            - x: torch.Tensor [B, N, N, C] (float32 or float16) on CUDA
            - mask: torch.Tensor [B, N, N] (binary) or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, C] with the same dtype as the input.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    orig_dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp32 for numeric stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the compute‑intensive portion
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Load linear weights (fp16)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Prepare mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16)
    else:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 7) Fuse mask and gates using Triton
    # ------------------------------------------------------------------
    total_elements = left.numel()                # B * N * N * H
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    _mask_gate_fusion[grid](
        left,
        right,
        mask_fp16.view(-1),
        left_gate,
        right_gate,
        left_out,
        right_out,
        hidden_dim,
        total_elements,
        BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 8) Core reduction via batched GEMM (left_out @ right_outᵀ)
    #    Reshape to (B*H, N, N) for torch.bmm.
    # ------------------------------------------------------------------
    left_mat  = left_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(-2, -1))   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 9) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)   # (B, N, N, H)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 10) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 11) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 12) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 13) Cast back to the original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
