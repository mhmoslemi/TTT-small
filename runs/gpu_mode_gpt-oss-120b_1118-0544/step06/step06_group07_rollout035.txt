@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_Ab, stride_Ai, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Ci, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile
        a_ptrs = a_ptr + batch_id * stride_Ab + offs_m[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)

        # Load B tile (K,N)
        b_ptrs = b_ptr + batch_id * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + batch_id * stride_Cb + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cn
    c = acc.to(tl.float16)  # store as fp16
    tl.store(c_ptrs, c, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. LayerNorm (float32) on the input tensor [B,N,N,C].
2. Cast to FP16 for the compute‑heavy path.
3. Linear projections (bias‑less) to hidden dimension H:
   left, right, left_gate, right_gate, out_gate.
4. Apply sigmoid to the gate tensors.
5. Fuse mask (if present) and gate multiplication with a single
   element‑wise PyTorch operation (cheap compared to the O(N³) step).
6. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (left @ rightᵀ) using a custom Triton
   kernel (batch = B*H, M = N, N = N, K = N).
7. LayerNorm on the hidden dimension, multiply by out_gate, and final
   linear projection back to the original dimension.
8. Cast back to the input dtype and return.

The heavy O(N³) part is performed entirely on the GPU with a fused
Triton matmul, satisfying the requirement to implement a non‑trivial
kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ Bᵀ) for many batches.
# A: (batch, M, K)    (float16)
# B: (batch, K, N)    (float16) – already the transposed version of right
# C: (batch, M, N)    (float16)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    BATCH, M, N, K,                     # runtime shapes
    stride_Ab, stride_Ai, stride_Ak,    # strides for A (batch, M, K)
    stride_Bb, stride_Bk, stride_Bn,    # strides for B (batch, K, N)
    stride_Cb, stride_Ci, stride_Cn,    # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)   # which batch
    pid_m = tl.program_id(1)      # block row
    pid_n = tl.program_id(2)      # block column

    # Compute absolute coordinates of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + batch_id * stride_Ab
            + offs_m[:, None] * stride_Ai
            + offs_k[None, :] * stride_Ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            b_ptr
            + batch_id * stride_Bb
            + offs_k[:, None] * stride_Bk
            + offs_n[None, :] * stride_Bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate
        acc += tl.dot(a, b)

    # Write back result in FP16
    c_ptrs = (
        c_ptr
        + batch_id * stride_Cb
        + offs_m[:, None] * stride_Ci
        + offs_n[None, :] * stride_Cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using Triton for the
    O(N³) contraction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of parameter tensors (float32)
            - config: dict with keys
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (bias‑less) in fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left  = F.linear(x, left_proj_w)          # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)         # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4. Apply mask and gates (element‑wise, cheap compared to N³)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape (B,N,N) -> broadcast over hidden_dim
        mask_exp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * left_gate * mask_exp
        right = right * right_gate * mask_exp
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via custom Triton batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                         # H == hidden_dim
    batch = B * H

    # Reshape to (batch, N, N) where batch = B*H
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(batch, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(batch, N, N)

    # Transpose the second operand (right) on‑the‑fly by creating a view
    right_mat_T = right_mat.transpose(1, 2).contiguous()

    # Output buffer
    out_mat = torch.empty_like(left_mat)   # (batch, N, N), fp16

    # Strides (in elements)
    stride_Ab, stride_Ai, stride_Ak = left_mat.stride()
    stride_Bb, stride_Bk, stride_Bn = right_mat_T.stride()
    stride_Cb, stride_Ci, stride_Cn = out_mat.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        batch,
        N,
        N,
        N,
        stride_Ab,
        stride_Ai,
        stride_Ak,
        stride_Bb,
        stride_Bk,
        stride_Bn,
        stride_Cb,
        stride_Ci,
        stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate                     # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)            # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------
    # 7. Cast back to original dtype and return
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
