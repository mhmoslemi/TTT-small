@triton.jit
def _gate_mask_fuse_kernel(proj_left_ptr, gate_left_ptr, proj_right_ptr, gate_right_ptr, mask_ptr, out_left_ptr, out_right_ptr, total_elems, hidden_dim, BLOCK_SIZE: tl.constexpr):

   pid = tl.program_id(0)
   block_start = pid * BLOCK_SIZE
   offs = block_start + tl.arange(0, BLOCK_SIZE)

   mask = offs < total_elems

   # Compute mask index: each element corresponds to a location in [B,N,N], each repeated hidden_dim times.
   mask_idx = offs // hidden_dim

   proj_l = tl.load(proj_left_ptr + offs, mask=mask, other=0.0)
   gate_l = tl.load(gate_left_ptr + offs, mask=mask, other=0.0)
   proj_r = tl.load(proj_right_ptr + offs, mask=mask, other=0.0)
   gate_r = tl.load(gate_right_ptr + offs, mask=mask, other=0.0)
   mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

   out_l = proj_l * gate_l * mask_val
   out_r = proj_r * gate_r * mask_val

   tl.store(out_left_ptr + offs, out_l, mask=mask)
   tl.store(out_right_ptr + offs, out_r, mask=mask)

def _apply_gate_mask_fused(left_proj, left_gate, right_proj, right_gate, mask, block_size=65536):
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_fuse_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Cast to fp16 for compute
    x = x.to(torch.float16)

    # Cast weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear proj
    left_proj = torch.nn.functional.linear(x, left_proj_w)        # shape [B,N,N,H]
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # Apply mask and gates
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_gate_mask_fused(left_proj, left_gate, right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Core reduction: batched matmul per hidden dim
    # Permute to [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2)
    right_perm = right.permute(0, 3, 1, 2)

    B, H, N, _ = left_perm.shape

    # reshape to (B*H, N, N)
    left_flat = left_perm.reshape(B * H, N, N)
    right_flat = right_perm.reshape(B * H, N, N)

    # batched matmul: left @ right^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B,N,N,H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Second LayerNorm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply out_gate
    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to original dtype if needed
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) using the provided LN weights.
2. Cast to fp16 for the compute‑intensive part.
3. Compute four linear projections (left/right) and three gates (left/right/out)
   in fp16 via `torch.nn.functional.linear`.
4. Fuse projection, gate and optional mask with a tiny Triton kernel
   (`_gate_mask_fuse_kernel`). This reduces two extra memory passes.
5. Core O(N³·H) reduction:
   - Permute tensors to shape (B, H, N, N).
   - Perform a batched matrix multiplication:
       out = left @ rightᵀ   (hence sum over the sequence index k).
6. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate, and linearly project back to the original channel dimension.
7. Cast the result back to the input dtype (float32).

Only the mask‑gate fusion is implemented in Triton; the remaining heavy
operations use PyTorch’s highly optimized cuBLAS/LN kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _gate_mask_fuse_kernel(
    proj_left_ptr,   # fp16 * (B*N*N*H,)
    gate_left_ptr,   # fp16 * (B*N*N*H,)
    proj_right_ptr,  # fp16 * (B*N*N*H,)
    gate_right_ptr,  # fp16 * (B*N*N*H,)
    mask_ptr,        # fp16 * (B*N*N,)
    out_left_ptr,    # fp16 * (B*N*N*H,)
    out_right_ptr,   # fp16 * (B*N*N*H,)
    total_elems,     # i32 – total number of elements in proj/gate tensors
    hidden_dim,      # i32 – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask is broadcasted over the hidden dimension
    mask_idx = offs // hidden_dim

    proj_l = tl.load(proj_left_ptr + offs, mask=valid, other=0.0)
    gate_l = tl.load(gate_left_ptr + offs, mask=valid, other=0.0)
    proj_r = tl.load(proj_right_ptr + offs, mask=valid, other=0.0)
    gate_r = tl.load(gate_right_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out_l = proj_l * gate_l * m
    out_r = proj_r * gate_r * m

    tl.store(out_left_ptr + offs, out_l, mask=valid)
    tl.store(out_right_ptr + offs, out_r, mask=valid)


def _apply_gate_mask_fused(
    proj_left: torch.Tensor,
    gate_left: torch.Tensor,
    proj_right: torch.Tensor,
    gate_right: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 65536,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse (proj * gate * mask) for both left and right tensors using Triton.
    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    total = proj_left.numel()
    out_left = torch.empty_like(proj_left)
    out_right = torch.empty_like(proj_right)

    grid = (triton.cdiv(total, block_size),)
    _gate_mask_fuse_kernel[grid](
        proj_left,
        gate_left,
        proj_right,
        gate_right,
        mask,
        out_left,
        out_right,
        total,
        proj_left.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out_left, out_right


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, C]
        - mask        : torch.Tensor of shape [B, N, N] (or None)
        - weights     : dict of model parameters (fp32)
        - config      : dict with keys:
            * "dim" (int)        – input channel dimension C
            * "hidden_dim" (int) – hidden dimension H
            * "nomask" (bool)    – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32 -> FP16)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Cast all weights to fp16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (B,N,N,H) in fp16
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, H]
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Gates (fp16) and sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse mask & gate (Triton) or simple elementwise if no mask
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_gate_mask_fused(
            left_proj, left_gate, right_proj, right_gate, mask_fp16
        )
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core O(N³·H) reduction: batched matmul per hidden channel
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as: out = left_perm @ right_permᵀ  (batch dims B,H)
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N)
    left_perm  = left.permute(0, 3, 1, 2)   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2)  # [B, H, N, N]

    # Batched matrix multiplication over the last two dimensions
    out_perm = torch.matmul(left_perm, right_perm.transpose(-1, -2))  # [B, H, N, N]

    # Return to original layout (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7. Second LayerNorm on hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8. Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 10. Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
