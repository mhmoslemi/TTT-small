@triton.jit
def batched_matmul_kernel(a_ptr, b_ptr, c_ptr,
                          batch, M, N, K,
                          stride_am, stride_ak,
                          stride_bk, stride_bn,
                          stride_cm, stride_cn,
                          stride_cbatch,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
                          BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets for block
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    off_k = tl.arange(0, BLOCK_K)

    # Initialize accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # k offset
        k_offset = k * BLOCK_K
        # Compute loads
        a_ptrs = a_ptr + batch_id * stride_cbatch + off_m[:, None] * stride_am + (k_offset + off_k)[None, :] * stride_ak
        b_ptrs = b_ptr + batch_id * stride_cbatch + (k_offset + off_k)[:, None] * stride_bk + off_n[None, :] * stride_bn

        # Load (with mask)
        a = tl.load(a_ptrs, mask=(off_m < M)[:, None] & (k_offset + off_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(k_offset + off_k < K)[:, None] & (off_n < N)[None, :], other=0.0)

        # Cast to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        # Dot product
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + batch_id * stride_cbatch + off_m[:, None] * stride_cm + off_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=(off_m < M)[:, None] & (off_n < N)[None, :])
```

    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_cb, # batch stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        off_k = cur_k + tl.arange(0, BLOCK_K)

        # Compute pointers
        a_ptrs = a_ptr + batch * stride_cb + off_m[:, None] * stride_am + off_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch * stride_cb + off_k[:, None] * stride_bk + off_n[None, :] * stride_bn

        # Load with mask
        a = tl.load(a_ptrs, mask=(off_m < M)[:, None] & (off_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(off_k < K)[:, None] & (off_n < N)[None, :], other=0.0)

        # Cast to fp32
        a = a.to(tl.float32)

        acc += tl.dot(a, b)

    # Write out
    c_ptrs = c_ptr + batch * stride_cb + off_m[:, None] * stride_cm + off_n[None, :] * stride_cn

"""
TriMul (outgoing) forward pass – Triton‑accelerated kernels
==========================================================

This implementation follows the PyTorch reference for the outgoing
Triangle Multiplicative Update (TriMul) used in AlphaFold3.  The
computation consists of:

1. Layer‑norm over the channel dimension.
2. Linear projections to a hidden dimension *H* (no bias).
3. Sigmoid gates.
4. Optional mask – fused together with the projections and gates by a
   custom Triton kernel.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication (B*H independent GEMMs).
   The reduction is performed by a second Triton kernel that computes
   all GEMMs in one launch.
6. Output LayerNorm, multiplication by the output gate and a final
   linear projection back to the original dimension.
7. Result is cast back to the input dtype (float32).

Both kernels are written for fp16 compute (accumulating in fp32) and
work for any of the test configurations listed in the task description.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse projection, gate and (optional) mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,          # fp16 * (total_elems,)
    gate_ptr,          # fp16 * (total_elems,)
    mask_ptr,          # fp16 * (mask_elems,)
    out_ptr,           # fp16 * (total_elems,)
    total_elems,       # int: B*N*N*H
    hidden_dim,        # int: H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Bounds
    mask = offs < total_elems

    # Index of the corresponding mask element (broadcast over H)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 2048) -> torch.Tensor:
    """
    Fuse projection, gate and mask using a Triton kernel.
    All tensors must be fp16 and contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)  (binary 0/1, broadcast over H)
    Returns proj * gate * mask (shape (B,N,N,H)).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel 2: batched matrix multiplication (A @ B) where
#   A : (batch, M, K)  (fp16)
#   B : (batch, K, N)  (fp16)
#   C : (batch, M, N)  (fp16, computed in fp32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    batch, M, N, K,                     # dimensions
    stride_am, stride_ak,               # A strides (M, K)
    stride_bk, stride_bn,               # B strides (K, N)
    stride_cm, stride_cn,               # C strides (M, N)
    stride_cb,                          # batch stride (common for A, B, C)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # pointers with proper offset and batch stride
        a_ptrs = a_ptr + batch_id * stride_cb + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch_id * stride_cb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # masks for out‑of‑bounds loads
        a_mask = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        b_mask = (offs_k < K)[:, None] & (offs_n < N)[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # promote to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # write back results (cast to fp16)
    c_ptrs = c_ptr + batch_id * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (float32/float16/bool) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask"     – if True, ignore the mask (optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to fp16 (no bias terms in the linear layers)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections -> (B, N, N, H)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid -> (B, N, N, H)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Fuse projection, gate and optional mask
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM across the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape tensors to (B*H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # transpose right so that we can use a standard GEMM (K,N)
    right_T_flat = right_flat.transpose(1, 2).contiguous()

    # allocate output tensor
    out_flat = torch.empty_like(left_flat)

    # strides (all tensors are contiguous)
    batch = B * H
    M = N
    N_dim = N
    K = N

    stride_am = left_flat.stride(1)      # stride for M dimension
    stride_ak = left_flat.stride(2)      # stride for K dimension
    stride_bk = right_T_flat.stride(1)   # stride for K dimension in B
    stride_bn = right_T_flat.stride(2)   # stride for N dimension in B
    stride_cm = out_flat.stride(1)       # stride for M dimension in C
    stride_cn = out_flat.stride(2)       # stride for N dimension in C
    stride_cb = left_flat.stride(0)      # batch stride (same for all)

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N_dim, BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_T_flat,
        out_flat,
        batch,
        M,
        N_dim,
        K,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
