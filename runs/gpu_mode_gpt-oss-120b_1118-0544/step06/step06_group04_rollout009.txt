def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    # compute total blocks per batch dimensions
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_blocks_m * num_blocks_n)
    block_in_batch = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_in_batch // num_blocks_n
    pid_n = block_in_batch % num_blocks_n

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # compute pointers for a and b
        a_ptrs = a_ptr + batch * batch_stride_a + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_ptr + batch * batch_stride_b + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Input LayerNorm (float32) → cast to float16.
2. Linear projections (bias‑less) to hidden dimension and corresponding gates.
3. Fuse element‑wise mask (if any) and gate multiplication with a small Triton
   kernel `_gate_mask_kernel`.
4. Core contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication L @ Rᵀ for each hidden channel.
   Implemented with a custom Triton kernel `_batched_matmul_kernel` that
   processes all (batch * hidden_dim) matrices in a blocked fashion.
5. Output LayerNorm over hidden dimension, element‑wise multiplication by the
   output gate, and final linear projection back to the original dimension.
6. Cast result to the input dtype and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    """Fuse mask, left‑gate and right‑gate in‑place."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left      = tl.load(left_ptr + offs,      mask=active, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # Each hidden slice shares the same mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs,  left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication:
        C[batch, i, j] = Σ_k A[batch, i, k] * B[batch, k, j]
    A shape (batch, M, K), B shape (batch, K, N) (note B is stored as (batch, N, K)
    and we use stride_bk/stride_bn to treat it as transposed).
    """
    pid = tl.program_id(0)

    # Number of blocks in M and N dimensions
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # Derive batch index and block coordinates
    batch = pid // (num_blocks_m * num_blocks_n)
    block_in_batch = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_in_batch // num_blocks_n
    pid_n = block_in_batch % num_blocks_n

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for A and B
        a_ptrs = a_ptr + batch * batch_stride_a + (offs_m[:, None] * stride_am
                                                    + offs_k[None, :] * stride_ak)
        b_ptrs = b_ptr + batch * batch_stride_b + (offs_k[:, None] * stride_bk
                                                    + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Matrix‑multiply‑accumulate
        acc += tl.dot(a, b)   # result is float32

    # Write back
    c_ptrs = c_ptr + batch * batch_stride_c + (offs_m[:, None] * stride_cm
                                                + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C) (float32 or float16)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # Config
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → fp16
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)  # work in fp16 from now on

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑less) and gates
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left  = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)         # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3. Fuse mask and gates (Triton)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Prepare flattened views
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        total = left.numel()
        BLOCK = 1024

        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # Simple element‑wise gating (no mask)
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core contraction: batched matmul via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Permute to (B, H, N, N) and make contiguous so each (b,h) slice is contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Collapse batch and hidden dims
    batch_h = B * H
    left_mat  = left_perm.view(batch_h, N, N)   # (batch_h, M=N, K=N)
    right_mat = right_perm.view(batch_h, N, N)  # (batch_h, N, K=N) – will be accessed transposed
    out_mat = torch.empty_like(left_mat)        # allocate output

    # Strides (all tensors are contiguous)
    batch_stride_a = left_mat.stride(0)
    batch_stride_b = right_mat.stride(0)
    batch_stride_c = out_mat.stride(0)

    stride_am = left_mat.stride(1)   # N
    stride_ak = left_mat.stride(2)   # 1
    # For Bᵀ we treat B as (K,N)
    stride_bk = right_mat.stride(2)  # 1   (k stride)
    stride_bn = right_mat.stride(1)  # N   (n stride)

    stride_cm = out_mat.stride(1)    # N
    stride_cn = out_mat.stride(2)    # 1

    # Block sizes (tuned for typical GPUs)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)
    total_programs = batch_h * num_blocks_m * num_blocks_n

    grid = (total_programs,)

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        N,   # M
        N,   # N
        N,   # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B, N, N, dim) in fp16

    # ------------------------------------------------------------------
    # 6. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
