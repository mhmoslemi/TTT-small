import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    l_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * l_gate * m
        right = right * r_gate * m
    else:
        left = left * l_gate
        right = right * r_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The algorithm reproduces the PyTorch reference:
1. Layer‑norm the input (FP32 for stability).
2. Cast to FP16 and apply bias‑less linear projections to obtain `left` and `right`.
3. Compute left/right gates, optionally fuse with the mask in‑place using a Triton
   kernel (`_gate_mask_kernel`).
4. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is expressed as a batched GEMM (left @ rightᵀ) over the hidden dimension,
   leveraging cuBLAS.
5. Apply an output layer‑norm, multiply by the output gate, and project back to
   the original channel dimension.
6. Cast the result back to the input dtype and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    If HAS_MASK=False the mask term is omitted (treated as 1.0).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load operands
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    l_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    r_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # One mask value spans `hidden_dim` consecutive elements
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * l_gate * m
        right = right * r_gate * m
    else:
        left = left * l_gate
        right = right * r_gate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)
