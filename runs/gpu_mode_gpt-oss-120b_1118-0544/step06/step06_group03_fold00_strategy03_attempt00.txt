                                         N, hidden_dim, BLOCK_SIZE=4096, PARTITION=256,)
    
    # Reshape back before output LayerNorm
    out_tensor = out_flat.view(BATCH_SIZE, N, N, hidden_dim)
    
    #------------------------------------------------------------------#
    # 4. Output transformation: projection back to original dimension
    #------------------------------------------------------------------#
    # Apply output LayerNorm
    out_tensor = out_tensor.permute(0, 3, 1, 2).contiguous()
    h, w, hdim = out_tensor.shape[1], out_tensor.shape[2], hidden_dim
    BLOCK = 256  # Use block that fits in L1/SRAM and maintains compute intensity
    
    # Write output LayerNorm using higher-level triton operations
    out_norm = F.layer_norm(out_tensor, (hidden_dim,), 
                           weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    
    # Apply out_gate and linear projection
    out_norm = out_norm * out_gate_out
    out_full = F.linear(out_norm, weights["to_out.weight"]).to(dtype_in)
    
    #------------------------------------------------------------------#
    # Output tensor
    #------------------------------------------------------------------#
    return out_full.contiguous()
