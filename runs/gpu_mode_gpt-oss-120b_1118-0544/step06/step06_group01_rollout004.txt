    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # 1. LayerNorm
    norm_weight = weights["norm.weight"].to(device).to(torch.float32)
    norm_bias = weights["norm.bias"].to(device).to(torch.float32)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # 2. Cast to fp16
    x = x.to(torch.float16)

    # 3. Linear projections (weights)
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # 4. Compute left, right and gates
    # Use torch.nn.functional.linear for fast: out = input @ weight.T
    # Input shape: (B,N,N,dim), weight shape: (hidden_dim, dim)
    # F.linear expects input shape (..., in_features) and weight shape (out_features,in_features)
    # So left = F.linear(x, left_proj_weight) yields (..., hidden_dim)
    left = torch.nn.functional.linear(x, left_proj_weight) # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

        right = right.contiguous()

            left_gate, right_gate,
            mask_fp,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )

        left = left * left_gate
        right = right * right_gate

    # 6. Core contraction
    B,N,_,H = left.shape
    # combine B and H
    left_mat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_mat = right.permute(0,3,1,2).contiguous().view(B*H, N, N)

    out = out.to(input_tensor.dtype)

"""
Outgoing TriMul (Triangle Multiplicative Update) forward pass.

Algorithm
---------
1. Layer‑normalize the input (float32 for stability).
2. Cast to fp16 and apply three linear projections:
   - left  = x @ W_left   (hidden_dim)
   - right = x @ W_right  (hidden_dim)
   - left_gate, right_gate, out_gate = sigmoid(x @ W_gate) (hidden_dim)
3. Fuse element‑wise mask (if present) and gate multiplications with a
   small Triton kernel, writing the results back in‑place.
4. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM (left @ rightᵀ) for each (b,d) pair and is
   performed with torch.bmm (cuBLAS) on fp16 data.
5. Apply output LayerNorm, the out‑gate, and a final linear projection
   back to the original feature dimension.
6. Cast the result back to the input dtype (usually float32).

Only the mask+gate fusion is done in Triton; the remaining heavy
operations use highly‑optimized PyTorch/cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # *fp16  [B,N,N,H] flattened
    right_ptr,              # *fp16  same shape
    left_gate_ptr,          # *fp16  same shape
    right_gate_ptr,         # *fp16  same shape
    mask_ptr,               # *fp16  [B,N,N] flattened, broadcasted over H
    total_elems,            # i64    total number of elements in left/right
    hidden_dim,             # i64    H
    HAS_MASK: tl.constexpr, # bool
    BLOCK_SIZE: tl.constexpr # i32
):
    """In‑place fusion of (left * left_gate * mask) and (right * right_gate * mask)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden_dim entries share the same mask entry
        mask_index = offs // hidden_dim          # i64 vector
        mask_val   = tl.load(mask_ptr + mask_index, mask=mask, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy parts
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # -----------------------------------------------------------------
    # 4. Linear projections and gates (fp16)
    # -----------------------------------------------------------------
    left  = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)         # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 5. Fuse mask and gate multiplication with a Triton kernel
    # -----------------------------------------------------------------
    use_mask = (mask is not None) and (not nomask)
    if use_mask:
        # Ensure contiguous tensors for the kernel
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()
        mask_fp   = mask.to(device).to(torch.float16).contiguous()

        total = left.numel()
        BLOCK_SIZE = 1024  # number of threads per program

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left,
            right,
            left_gate,
            right_gate,
            mask_fp,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # left and right are now masked & gated in‑place
    else:
        # No mask (or nomask flag): simple element‑wise gate multiplication
        left  = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Core contraction: batched GEMM (left @ rightᵀ) over the sequence dim
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) for a single batched GEMM
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiply: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the original dtype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
