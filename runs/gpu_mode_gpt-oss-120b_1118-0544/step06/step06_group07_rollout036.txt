"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input (float32 for stability).
2. Cast to FP16 for the compute‑heavy path.
3. Linear projections → left, right, left_gate, right_gate, out_gate (bias‑less).
4. Fuse mask (if present) and the left/right gates in‑place via a tiny Triton kernel.
5. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (left @ rightᵀ) with a custom Triton kernel
   over the (B·hidden_dim) matrix batch.
6. Layer‑norm over the hidden dimension, apply out_gate, and final linear projection.
7. Cast back to the original dtype and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. Gate + mask fusion (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # load
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # each mask entry corresponds to hidden_dim consecutive elements
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # store back in‑place
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


# ----------------------------------------------------------------------
# 2. Batched GEMM: C = A @ Bᵀ  (A, B : (batch, M, K), (batch, K, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                 # pointers to FP16 tensors
    M, N, K,                             # matrix sizes
    stride_am, stride_ak,                # A strides (row, col)
    stride_bk, stride_bn,                # B strides (row, col)
    stride_cm, stride_cn,                # C strides (row, col)
    batch_stride_a, batch_stride_b, batch_stride_c,  # strides between batches
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    batch_id = tl.program_id(2)  # batch index

    # tile start indices
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds
    mask_i = i < M
    mask_j = j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, K, BLOCK_K):
        k_idx = k + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        # A: (batch, M, K)
        a_ptrs = a_ptr + batch_id * batch_stride_a \
                 + i[:, None] * stride_am \
                 + k_idx[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # B: (batch, K, N)
        b_ptrs = b_ptr + batch_id * batch_stride_b \
                 + k_idx[:, None] * stride_bk \
                 + j[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        acc += tl.dot(a, b)  # FP16 * FP16 -> FP32 accumulation

    # write result
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + batch_id * batch_stride_c \
             + i[:, None] * stride_cm \
             + j[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# 3. Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Cast to FP16
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (bias‑less) in fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left  = F.linear(x, left_proj_w)          # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)         # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4. Fuse mask and gates (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B,N,N) -> FP16, broadcast later inside kernel
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten everything
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 131072  # 128k elements per program
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # no mask → simple element‑wise gate fusion
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via batched GEMM (Triton)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape          # H == hidden_dim
    # bring hidden dim to batch dimension: (B, H, N, N) → (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous()
    right_mat = right.permute(0, 3, 1, 2).contiguous()
    batch = B * H
    left_mat = left_mat.view(batch, N, N)
    right_mat = right_mat.view(batch, N, N)

    # Allocate output tensor
    out_mat = torch.empty_like(left_mat)   # FP16, (batch, N, N)

    # Strides for a contiguous (batch, M, K) layout where M=K=N
    M = N
    K = N
    N_out = N
    stride_am = K          # row stride for A
    stride_ak = 1          # col stride for A
    stride_bk = K          # row stride for B
    stride_bn = 1          # col stride for B
    stride_cm = N_out      # row stride for C
    stride_cn = 1          # col stride for C
    batch_stride_a = M * K
    batch_stride_b = K * N_out
    batch_stride_c = M * N_out

    # Tile sizes – chosen to fit in shared memory (tuned empirically)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N_out, BLOCK_N),
        batch,
    )
    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M, N_out, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (still FP16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in FP16

    # ------------------------------------------------------------------
    # 7. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
