def triton_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_head,
    stride_b_batch, stride_b_k, stride_b_i, stride_b_head,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_head,
    H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_head = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute batch and head indices
    b_idx = batch_head // H
    h_idx = batch_head - b_idx * H

    # base pointers
    a_base = a_ptr + b_idx * stride_a_batch + h_idx * stride_a_head
    b_base = b_ptr + b_idx * stride_b_batch + h_idx * stride_b_head
    c_base = c_ptr + b_idx * stride_c_batch + h_idx * stride_c_head

    # mask for out-of-bounds of M,N
    mask_m = offs_m < M
    mask_n = offs_n < N
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # mask for K dimension
        mask_k = offs_k < K

        # load blocks
        a_ptrs = a_base + offs_m[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_i

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

"""
TriMul (outgoing) forward pass – Triton‑accelerated core reduction.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (`dim`).
2. Cast to fp16 for compute‑heavy parts.
3. Linear projections → left/right tensors (B,N,N,H).
4. Gate projections + sigmoid → left_gate, right_gate, out_gate (B,N,N,H).
5. Fuse projection, gating and optional mask in a small Triton kernel.
6. Core reduction (batched GEMM):
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented by a custom Triton matmul kernel that works on the
   strided (B,N,N,H) layout, avoiding the expensive view+copy used in
   the reference implementation.
7. Output LayerNorm over the hidden dimension H, multiply by `out_gate`,
   and a final linear projection back to the original dimension `dim`.
8. Cast the result back to the input dtype (usually float32).

All heavy‑compute steps (mask‑gate fusion and the N³ reduction) are
implemented in Triton; the remaining lightweight ops use PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1. Fusion of projection, gate and (optional) mask
# --------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,            # fp16 * (B*N*N*H,)
    gate_ptr,            # fp16 * (B*N*N*H,)
    mask_ptr,            # fp16 * (B*N*N,)
    out_ptr,             # fp16 * (B*N*N*H,)
    total_elems,         # i64: B*N*N*H
    hidden_dim,          # i64: H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    # index of the corresponding mask element (broadcast over H)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 2048) -> torch.Tensor:
    """
    Fuse projection, gate and mask using a Triton kernel.
    All tensors must be fp16 and contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)    (binary 0/1, broadcast over H)
    Returns proj * gate * mask  (shape (B,N,N,H)).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# --------------------------------------------------------------
# 2. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# --------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,               # fp16 pointers
    M, N, K,                           # i64: dimensions (all = seq_len)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_head,
    stride_b_batch, stride_b_k, stride_b_i, stride_b_head,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_head,
    H,                                 # i64: hidden dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM where each (b, h) pair is a separate matrix:
        C[b,h] = A[b,h] @ B[b,h].T
    A, B, C are stored with shape (B, N, N, H) and the strides
    are passed explicitly.
    """
    batch_head = tl.program_id(0)          # 0 .. B*H-1
    pid_m = tl.program_id(1)               # block row
    pid_n = tl.program_id(2)               # block col

    # ------- compute block start indices -------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ------- decode batch and head ----------
    b_idx = batch_head // H
    h_idx = batch_head - b_idx * H

    # base pointers for this (b, h)
    a_base = a_ptr + b_idx * stride_a_batch + h_idx * stride_a_head
    b_base = b_ptr + b_idx * stride_b_batch + h_idx * stride_b_head
    c_base = c_ptr + b_idx * stride_c_batch + h_idx * stride_c_head

    # masks for out-of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for the current k‑block
        a_ptrs = a_base + offs_m[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_i

        # load with masks
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


# --------------------------------------------------------------
# 3. Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (float32/float16/bool) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ----------------------------------------------------------
    # unpack arguments
    # ----------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ----------------------------------------------------------
    # 1. Input LayerNorm over `dim` (float32 for stability)
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ----------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy parts
    # ----------------------------------------------------------
    x = x.to(torch.float16)

    # ----------------------------------------------------------
    # 3. Cast model weights to fp16 (no bias terms)
    # ----------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ----------------------------------------------------------
    # 4. Linear projections (B,N,N,H)
    # ----------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # ----------------------------------------------------------
    # 5. Gate projections + sigmoid (B,N,N,H)
    # ----------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ----------------------------------------------------------
    # 6. Apply mask + gating (fused via Triton if mask present)
    # ----------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is fp16, contiguous, binary (0/1)
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ----------------------------------------------------------
    # 7. Core reduction (batched GEMM) using Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ----------------------------------------------------------
    B, N, _, H = left.shape
    # allocate output tensor
    out = torch.empty_like(left, dtype=torch.float16)

    # strides for the (B,N,N,H) layout
    stride_a_batch, stride_a_i, stride_a_k, stride_a_head = left.stride()
    stride_b_batch, stride_b_i, stride_b_k, stride_b_head = right.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_head = out.stride()

    # Triton kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _tri_mul_kernel[grid](
        left, right, out,
        N, N, N,                                 # M, N, K
        stride_a_batch, stride_a_i, stride_a_k, stride_a_head,
        stride_b_batch, stride_b_k, stride_b_i, stride_b_head,  # note swapped k/i for B^T
        stride_c_batch, stride_c_i, stride_c_j, stride_c_head,
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ----------------------------------------------------------
    # 9. Cast back to the original dtype
    # ----------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
