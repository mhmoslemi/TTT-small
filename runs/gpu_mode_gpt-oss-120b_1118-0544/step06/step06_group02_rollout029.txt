def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # row tile
    pid_n = tl.program_id(1)  # col tile
    pid_z = tl.program_id(2)  # batch*head

    # compute batch index and head index
    b = pid_z // H
    h = pid_z % H

    # compute start offsets of each block
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # create masks for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # compute base pointers
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute actual k offset
        k_offset = k * BLOCK_K
        cur_k = k_offset + offs_k
        # bounds for k
        valid_k = cur_k < N
        # maybe incorporate in mask for loading
        # compute pointers for left block (i,k)
        left_ptrs = left_base + (offs_i[:, None] * stride_left_i) + (cur_k[None, :] * stride_left_k)
        # compute pointers for right block (j,k) but as transposed: we need right[j,k] -> address as (j,cur_k)

        left = tl.load(left_ptrs, mask=mask_i[:, None] & valid_k[None, :], other=0.0)

        # dot
        acc += tl.dot(left, right)  # left shape (M,K) right shape (N,K) => dot treated as sum over K, producing (M,N)
        # advance k? not needed due to indexing via cur_k

    # apply masks for i and j
    out_ptrs = out_base + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)

    pid_z = tl.program_id(2)

"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate and
batched GEMM in a custom kernel.

Algorithm
---------
1. Layer‑normalize the input.
2. Cast to fp16 for compute‑intensive parts.
3. Four linear projections (left/right + their gates) are done with
   fp16 weights.
4. If a mask is present, fuse mask, projection and gate with a tiny
   Triton kernel (`_gate_mask_kernel`).
5. Core O(N³·H) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   Implemented as a fused GEMM kernel (`tri_mul_kernel`) that computes
   many independent (N×N) matrix multiplies (one per batch×head) on
   the GPU, avoiding costly permutes.
6. Apply a second LayerNorm, multiply by the output gate and a final
   linear projection back to the original dimension.
7. Cast the result back to the input dtype (float32).

All heavy element‑wise work is performed in Triton, while the
LayerNorms and linear layers use PyTorch’s highly‑optimised kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) on a flattened view
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask is broadcasted over the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 2048,
) -> torch.Tensor:
    """Fuse `proj * gate * mask` using a Triton kernel."""
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the core TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,     # fp16 * (B, N, N, H)
    right_ptr,    # fp16 * (B, N, N, H)
    out_ptr,      # fp16 * (B, N, N, H)
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # tile row (i)
    pid_n = tl.program_id(1)  # tile col (j)
    pid_z = tl.program_id(2)  # batch*head index

    # decompose pid_z -> batch b and head h
    b = pid_z // H
    h = pid_z % H

    # start offsets for this tile
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k_start = kb * BLOCK_K
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left block: (M, K) -> indices (i, k)
        left_ptrs = left_base + (offs_i[:, None] * stride_l_i) + (cur_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right block (transposed) : (K, N) -> indices (k, j)
        right_ptrs = right_base + (cur_k[:, None] * stride_r_k) + (offs_j[None, :] * stride_r_i)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left, right)   # (M,K)·(K,N) -> (M,N)

    # write result
    out_ptrs = out_base + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, C]
        - mask        : torch.Tensor of shape [B, N, N] (or None)
        - weights     : dict of model parameters (float32)
        - config      : dict with keys:
            * "dim" (int)         – input channel dimension C
            * "hidden_dim" (int)  – hidden dimension H
            * "nomask" (bool)     – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------- LayerNorm (input) -------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for compute‑intensive part
    x = x.to(torch.float16)

    # ---------------------------- cast weights ---------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ----------------------- linear projections -------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------- gates ----------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------- mask & gate fusion (Triton) --------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------- core TriMul -----------------------------
    B, N, _, H = left.shape
    assert H == hidden_dim, "Hidden dimension mismatch"

    # allocate output tensor
    out = torch.empty_like(left)

    # strides (B,N,N,H) layout
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------ cast back to input dtype ----------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
