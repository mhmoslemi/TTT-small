@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_a_batch,
    stride_bk, stride_bn, stride_b_batch,
    stride_cm, stride_cn, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # pid encodes batch and tile indices
    # number of blocks per matrix = ceil(M/BLOCK_M) * ceil(N/BLOCK_N)
    # total number of blocks per batch = num_blocks_m * num_blocks_n
    # we compute batch index and block indices
    batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_id = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = block_id // tl.cdiv(N, BLOCK_N)
    pid_n = block_id % tl.cdiv(N, BLOCK_N)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # pointers to start of each batch
    a_batch_ptr = a_ptr + batch * stride_a_batch
    b_batch_ptr = b_ptr + batch * stride_b_batch
    c_batch_ptr = c_ptr + batch * stride_c_batch

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offset
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load B tile
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # Layernorm
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    # Cast to fp16
    x = x.to(torch.float16)

    # Linear projects
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    # compute left, right, gates
    left = torch.nn.functional.linear(x, left_proj_w)  # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse mask and gates
    if mask is not None and not config.get('nomask', False):
        # mask shape [B,N,N], unsqueeze for broadcast -> [B,N,N,1]
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten for kernel
        total = left.numel() # B*N*N*H
        # Use kernel _gate_mask_kernel previously defined
        # The kernel multiplies left * left_gate * mask and right * right_gate * mask.
        # mask index: each element of mask corresponds to H consecutive elements.
        # Provide hidden_dim as runtime argument.
        BLOCK = 65536  # maybe bigger
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']), )
        _gate_mask_kernel[grid](
            left.contiguous().view(-1),
            right.contiguous().view(-1),
            left_gate.contiguous().view(-1),
            right_gate.contiguous().view(-1),
            mask_fp.contiguous().view(-1),
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # no need to reshape, they already in correct shape (left,right)
    else:
        left = left * left_gate
        right = right * right_gate

    # Now compute core contraction via Triton GEMM
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose right for multiplication
    right_t = right_perm.permute(0, 1, 3, 2).contiguous()
    # Flatten batch dimension
    batch = bs * hidden_dim
    left_mat = left_perm.view(batch, N, N)
    right_mat = right_t.view(batch, N, N)
    # Allocate output
    out_mat = torch.empty_like(left_mat)

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # compute needed grid size

    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

        right_mat,
        out_mat,
        N,
        N,

    out = torch.nn.functional.linear(out, to_out_w)

    # cast back

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm (float32) the [B,N,N,C] input.
2. Cast to fp16 for the compute‑heavy path.
3. Bias‑less linear projections → left, right and three gates.
4. Fuse mask (if present) and the left/right gates in a single Triton kernel,
   writing the gated tensors back in‑place.
5. Core O(N³·H) contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   implemented as a batched GEMM (left @ rightᵀ) over B·H matrices with a
   custom Triton kernel.
6. Layer‑norm over the hidden dimension, multiply by the output gate, and a
   final linear projection back to the original dimension.
7. Cast back to the input dtype and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. Gate + mask fusion kernel (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # load tensors
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # each mask entry corresponds to `hidden_dim` consecutive elements
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # write back in‑place
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)

            - mask         : torch.Tensor of shape (B, N, N) or None

            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    """

    dtype_in = input_tensor.dtype

    hidden_dim = config["hidden_dim"]

    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # --------------------------------------------------------------
    # 2. Cast to fp16 for the heavy path
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (bias‑less) in fp16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left  = F.linear(x, left_proj_w)    # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)   # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 4. Fuse mask and gates (in‑place) via Triton
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure contiguous fp16 tensors
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        total = left.numel()
        BLOCK = 8192  # number of elements processed per program
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left.contiguous().view(-1),
            right.contiguous().view(-1),
            left_gate.contiguous().view(-1),
            right_gate.contiguous().view(-1),
            mask_fp.view(-1),
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # tensors are already updated in‑place
    else:
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # batch, seq_len, seq_len, hidden_dim

    # reshape to (B, H, N, N) and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()        # (B, H, N, N)
    # transpose the last two dimensions of right for A @ Bᵀ
    right_t = right_perm.transpose(2, 3).contiguous()          # (B, H, N, N)

    # collapse batch and hidden_dim into a single batch dimension for the kernel
    batch_dim = B * hidden_dim
    left_mat = left_perm.view(batch_dim, N, N)
    right_mat = right_t.view(batch_dim, N, N)

    # allocate output
    out_mat = torch.empty_like(left_mat)

    # kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch_dim * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # --------------------------------------------------------------
    # 7. Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
