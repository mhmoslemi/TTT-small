def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    x = x.to(torch.float16)  # compute in fp16

    # load weights to fp16
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)

    # projections and gates
    left = F.linear(x, left_proj_w)  # (B,N,N,H)
    right = F.linear(x, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # apply mask if any
    if mask is not None and not nomask:
        mask = mask.to(device).to(torch.float16).unsqueeze(-1)  # B,N,N,1
        left = left * mask
        right = right * mask

    # reshape for kernel
    B, N, _, H = left.shape
    # Permute to (B, H, N, N) and then reshape to (B*H, N, N)
    left_resh = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_resh = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # output placeholder
    out_resh = torch.empty_like(left_resh)  # shape (B*H, N, N), fp16

    # strides
    stride_lb, stride_lm, stride_lk = left_resh.stride()
    stride_rb, stride_rn, stride_rk = right_resh.stride()
    stride_ob, stride_om, stride_on = out_resh.stride()

    # launch kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N), B * H)
    tri_mul_kernel[grid](
        left_resh,
        right_resh,
        out_resh,
        N, N, N,
        stride_lb, stride_lm, stride_lk,
        stride_rb, stride_rn, stride_rk,
        stride_ob, stride_om, stride_on,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back
    out = out_resh.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,H)

    # output layernorm
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # apply out_gate
    out = out * out_gate

    # final projection
    out = F.linear(out, to_out_w)  # shape (B,N,N,dim)
    out = out.to(dtype)

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the PyTorch reference but off‑loads the
O(N³) contraction to a custom Triton kernel.   The steps are:

1. Layer‑normalize the input (float32 for stability).
2. Cast to fp16 and apply linear projections and gate layers.
3. Optionally apply a pairwise mask (broadcasted over the hidden dimension).
4. Fuse the core contraction:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   which is a batched GEMM (left @ rightᵀ) over the sequence dimension.
   This GEMM is performed by the `tri_mul_kernel` Triton kernel.
5. Apply the output LayerNorm, the output gate and a final linear projection.
6. Cast the result back to the input dtype (usually float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,                         # all equal to seq_len
    stride_lb, stride_lm, stride_lk, # left: (batch, M, K)
    stride_rb, stride_rn, stride_rk, # right: (batch, N, K) – we read it transposed
    stride_ob, stride_om, stride_on, # out: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for TriMul: out = left @ rightᵀ

    left  : [batch, M, K]
    right : [batch, N, K] (interpreted as [batch, K, N] inside)
    out   : [batch, M, N]
    """
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_b = tl.program_id(2)          # batch (combined B * hidden_dim)

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    left_batch  = left_ptr  + pid_b * stride_lb
    right_batch = right_ptr + pid_b * stride_rb
    out_batch   = out_ptr   + pid_b * stride_ob

    # Accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in tiles
    for k in range(0, K, BLOCK_K):
        k_range = tl.arange(0, BLOCK_K)
        k_valid = (k + k_range) < K

        # left[i, k]  -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_batch \
                    + (offs_m[:, None] * stride_lm) \
                    + ((k + k_range)[None, :] * stride_lk)
        left_tile = tl.load(left_ptrs,
                            mask=mask_m[:, None] & k_valid[None, :],
                            other=0.0)

        # right[j, k] read as transposed: rightᵀ[k, j]  -> shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_batch \
                     + ((k + k_range)[:, None] * stride_rk) \
                     + (offs_n[None, :] * stride_rn)
        right_tile = tl.load(right_ptrs,
                             mask=k_valid[:, None] & mask_n[None, :],
                             other=0.0)

        acc += tl.dot(left_tile, right_tile)

    # Write the result (cast back to fp16)
    out_ptrs = out_batch \
               + (offs_m[:, None] * stride_om) \
               + (offs_n[None, :] * stride_on)
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    # -----------------------------------------------------------------
    # unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for compute‑heavy parts
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gates
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 4. Apply gates (and optional mask) element‑wise
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    if mask is not None and not nomask:
        # mask shape: (B, N, N) → broadcast over hidden dimension
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_fp
        right = right * mask_fp

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ rightᵀ) over the sequence dim
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Reorder to (B, H, N, N) and collapse batch & hidden into a single dim
    left_resh  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_resh = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Allocate output tensor
    out_resh = torch.empty_like(left_resh)

    # Strides (in elements)
    stride_lb, stride_lm, stride_lk = left_resh.stride()
    stride_rb, stride_rn, stride_rk = right_resh.stride()
    stride_ob, stride_om, stride_on = out_resh.stride()

    # Triton kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),    # blocks over rows
        triton.cdiv(N, BLOCK_N),    # blocks over cols
        B * H                       # combined batch * hidden dimension
    )

    tri_mul_kernel[grid](
        left_resh,
        right_resh,
        out_resh,
        N, N, N,                         # M, N, K
        stride_lb, stride_lm, stride_lk,
        stride_rb, stride_rn, stride_rk,
        stride_ob, stride_om, stride_on,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B, N, N, H)
    out = out_resh.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)                # (B, N, N, C)

    # Cast back to the original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
