import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_fuse_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, total, hidden_dim, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE
    offs = offset + tl.arange(0, BLOCK_SIZE)
    mask = offs < total
    # compute mask index
    mask_idx = offs // hidden_dim
    p = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    # Gate is expected to have sigmoid applied before calling this kernel
    out = p * g * m
    tl.store(out_ptr + offs, out, mask=mask)

def _apply_mask_and_gate(proj, gate, mask, block_size=16384):
    # proj, gate: (B,N,N,H) float16, mask: (B,N,N) float16
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _mask_fuse_kernel[grid](
        proj, gate, mask, out,
        total, H,
        BLOCK_SIZE=block_size,
    )
    return out

def custom_kernel(data):
    """
    TriMul (outgoing) forward pass – Triton‑accelerated projection + mask & gate fusion.

    Steps:
    1) LayerNorm on input (float32).
    2) Cast to FP16.
    3) Fused linear projections for left/right (proj + gate) using concatenated weight matrices.
    4) Apply sigmoid to gates.
    5) Optional mask‑gate fusion via a small Triton kernel.
    6) Batched GEMM (left * right^T) using torch.bmm on (B*H, N, N).
    7) Output LayerNorm, out‑gate, final linear projection.
    8) Cast back to original dtype.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # 1. Input layer norm (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # 2. Cast to fp16 for heavy compute
    x = x.to(torch.float16)

    # 3. Prepare concatenated weights for left and right projections + gates
    # Convert weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (hidden_dim, dim)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

    # Concatenate for fused linear: shape (2*hidden_dim, dim)
    left_fused_w = torch.cat([left_proj_w, left_gate_w], dim=0)
    right_fused_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # Reshape input for linear: (B*N*N, dim)
    B, N, _, _ = x.shape
    x_flat = x.view(B * N * N, dim)

    # 4. Compute fused projections
    left_fused = F.linear(x_flat, left_fused_w)   # (B*N*N, 2*hidden_dim)
    right_fused = F.linear(x_flat, right_fused_w)

    # Split projections and gates
    left_proj = left_fused[:, :hidden_dim].view(B, N, N, hidden_dim)
    left_gate = left_fused[:, hidden_dim:].view(B, N, N, hidden_dim).sigmoid()
    right_proj = right_fused[:, :hidden_dim].view(B, N, N, hidden_dim)
    right_gate = right_fused[:, hidden_dim:].view(B, N, N, hidden_dim).sigmoid()

    # 5. Compute out_gate
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = F.linear(x_flat, out_gate_w).view(B, N, N, hidden_dim).sigmoid()

    # 6. Apply mask and element‑wise gate
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_mask_and_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_and_gate(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and nomask:
            # mask is ignored, still need to broadcast if mask tensor exists
            # No operation needed.
            pass

    # 7. Core reduction: batched GEMM per hidden_dim
    # reshape to (B*hidden_dim, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Compute left * right^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 8. Output LayerNorm over hidden_dim
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # 9. Apply out_gate
    out = out * out_gate

    # 10. Final linear projection back to dim
    to_out_w = weights["to_out.weight"].to(torch.float16)
    # Reshape for linear: (B*N*N, hidden_dim)
    out_flat2 = out.view(B * N * N, hidden_dim)
    out_proj = F.linear(out_flat2, to_out_w)   # (B*N*N, dim)
    out = out_proj.view(B, N, N, dim)

    # 11. Cast back to original dtype (float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated projection & mask‑gate fusion.

Algorithm
---------
1. Layer‑norm the input (float32 for stability).
2. Cast to fp16 for compute‑heavy stages.
3. Fuse each projection + its gate into a single linear call by concatenating
   the two weight matrices (proj and gate).  This halves the number of memory
   loads for the two linear layers.
4. Apply sigmoid to the gate part.
5. If a mask is present, fuse mask, projection and gate in a tiny Triton
   kernel: out = proj * gate * mask   (mask broadcast over the hidden dim).
6. Core reduction: for every batch element and hidden channel compute
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication L * Rᵀ.  Implemented with
   `torch.bmm` on tensors of shape (B*H, N, N) – this uses cuBLAS/Tensor‑cores.
7. Output LayerNorm over the hidden dimension, multiply by an out‑gate and a
   final linear projection back to the original channel size.
8. Cast the result back to the input dtype (usually float32).

The only mandatory Triton component is the mask‑gate‑fusion kernel, which
eliminates a separate element‑wise pass when a mask is used.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr,          # fp16* (B*N*N*H,)
    gate_ptr,          # fp16* (B*N*N*H,)
    mask_ptr,          # fp16* (B*N*N,)
    out_ptr,           # fp16* (B*N*N*H,)
    total_elems,       # int: B*N*N*H
    hidden_dim,        # int: H
    BLOCK_SIZE: tl.constexpr,
):
    """
    Compute: out = proj * gate * mask
    `mask` is broadcasted over the hidden dimension.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # Index of the mask element (every hidden_dim consecutive entries belong to the same mask)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_mask_and_gate(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 16384) -> torch.Tensor:
    """
    Fuse projection, gate and mask using a Triton kernel.
    All tensors must be fp16 contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)  (binary 0/1, broadcast over H)
    Returns proj * gate * mask (same shape as proj).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _mask_gate_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (bool/float)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numeric stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Prepare fused projection+gate weights (fp16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)   # (H, C)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

    # Concatenate projection & gate weights -> (2*H, C)
    left_fused_w = torch.cat([left_proj_w, left_gate_w], dim=0)
    right_fused_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # ------------------------------------------------------------------
    # 4. Linear projections (fused) for left/right
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    x_flat = x.view(B * N * N, dim)          # (B*N*N, C)

    left_fused = F.linear(x_flat, left_fused_w)   # (B*N*N, 2*H)
    right_fused = F.linear(x_flat, right_fused_w)

    # Split into proj / gate and apply sigmoid to gates
    left_proj = left_fused[:, :hidden_dim].view(B, N, N, hidden_dim)
    left_gate = left_fused[:, hidden_dim:].view(B, N, N, hidden_dim).sigmoid()
    right_proj = right_fused[:, :hidden_dim].view(B, N, N, hidden_dim)
    right_gate = right_fused[:, hidden_dim:].view(B, N, N, hidden_dim).sigmoid()

    # ------------------------------------------------------------------
    # 5. Out‑gate (separate linear)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = F.linear(x_flat, out_gate_w).view(B, N, N, hidden_dim).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask & gate (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_mask_and_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_and_gate(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        # mask is ignored when nomask == True

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM for each hidden channel
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Equivalent to (left_h) @ (right_h)^T for each h.
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9. Apply out‑gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10. Final linear projection back to original dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (C, H)
    out_flat2 = out.view(B * N * N, hidden_dim)            # (B*N*N, H)
    out_proj = F.linear(out_flat2, to_out_w)               # (B*N*N, C)
    out = out_proj.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 11. Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
