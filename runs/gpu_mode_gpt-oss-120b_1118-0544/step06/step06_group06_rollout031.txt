"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input (fp32 for stability) and cast to fp16.
2. Linear projections → left/right (dim → hidden_dim) and compute three sigmoid gates.
3. Fuse mask (if present) with left/right gates using simple PyTorch ops.
4. Core O(N³·H) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a *batched* GEMM on tensors of shape (B·H, N, N) with a
   custom Triton kernel that processes the batch dimension together with the
   M×N tiles, thereby re‑using the K‑tiles of left/right across many i/j pairs.
5. Second LayerNorm over the hidden dimension.
6. Multiply by the output gate and project back to the original channel dimension.
7. Cast back to the original dtype.

The kernel (`batched_gemm_kernel`) performs a fused GEMM for all batches
simultaneously, eliminating the overhead of launching many small cuBLAS calls.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (int)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B  where A has shape (batch, M, K) and B has shape (batch, K, N).
    The batch dimension = total number of (original batch) × hidden_dim.
    All tensors are contiguous and stored in row‑major order.
    Accumulation is performed in fp32 and the result is stored as fp16.
    """
    pid = tl.program_id(0)

    # number of tiles along M and N
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    # decode batch, tile_m, tile_n from linear pid
    batch = pid // (num_pid_m * num_pid_n)
    pid_rem = pid % (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n
    pid_n = pid_rem % num_pid_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers to the beginning of the current batch
    a_batch = a_ptr + batch * stride_a_batch
    b_batch = b_ptr + batch * stride_b_batch
    c_batch = c_ptr + batch * stride_c_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # A tile: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k),
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )
        # B tile: (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_batch + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n),
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        # accumulate
        acc += tl.dot(a, b)

    # write back the result as fp16
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n),
        c,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )

    """

    """

    # ------------------------------------------------------------------

        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to fp16 for the heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Load weights in fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask and gates (pure PyTorch – inexpensive)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_f = mask.to(dtype=x.dtype, device=device).unsqueeze(-1)  # (B,N,N,1)
    else:
        mask_f = torch.ones((B, N, N, 1), dtype=x.dtype, device=device)

    left = left * mask_f * left_gate
    right = right * mask_f * right_gate

    # ------------------------------------------------------------------
    # 6) Prepare tensors for batched GEMM
    #    left  : (B, N, N, H)  →  (B, H, N, N) → (B*H, N, N)
    #    right : (B, N, N, H)  →  (B, H, N, N) (transposed K) → (B*H, N, N)
    # ------------------------------------------------------------------
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  (K=N, N=j)

    B, H, M, K = left_t.shape          # here M == K == N
    total_batch = B * H

    a = left_t.view(total_batch, M, K)   # (B*H, N, N)
    b = right_t.view(total_batch, K, M)  # (B*H, N, N)  note: K==N, M==N

    # output tensor
    c = torch.empty_like(a)

    # ------------------------------------------------------------------
    # 7) Launch Triton batched GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batch
        * ((M + BLOCK_M - 1) // BLOCK_M)
        * ((K + BLOCK_N - 1) // BLOCK_N),
    )

    # strides in number of elements (will be multiplied by element size internally)
    stride_a_batch = a.stride(0)
    stride_a_m = a.stride(1)
    stride_a_k = a.stride(2)

    stride_b_batch = b.stride(0)
    stride_b_k = b.stride(1)
    stride_b_n = b.stride(2)

    stride_c_batch = c.stride(0)
    stride_c_m = c.stride(1)
    stride_c_n = c.stride(2)

    batched_gemm_kernel[grid](
        a,
        b,
        c,
        M,
        K,
        K,                     # N == K for our square case
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = (
        c.view(B, H, M, K)           # (B, H, N, N)
        .permute(0, 2, 3, 1)         # (B, N, N, H)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 9) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 10) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 12) Cast back to original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
```
