def batched_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B_H, N, HIDDEN,  # not needed stride?
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_j, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Compute batch index
    batch_id = tl.program_id(0)  # 0 to B_h-1
    # Compute block row and col via pid1 and pid2
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # Compute offsets
    # row offset in output
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    # col offset
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Initialize accumulator

    # Loop over k dimension

        # Load left tile: shape (BLOCK_M, BLOCK_K)

def batched_matmul_transpose_kernel(
    left_ptr, right_ptr, out_ptr,
    B_H, N,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_j,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load left tile [M,K]
        left = tl.load(
            left_ptr + batch_id * stride_l_batch + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k,
            mask=(offs_i[:, None] < N) & (offs_k[None, :] < N),
            other=0.0
        )
        # Load right tile transposed [K,N] (since right transposed)
        right = tl.load(
            right_ptr + batch_id * stride_r_batch + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j,
            mask=(offs_k[:, None] < N) & (offs_j[None, :] < N),
            other=0.0
        )
        # Dot product accumulate
        acc += tl.dot(left, right)
    # Cast to fp16 for output
    out = acc.to(tl.float16)
    tl.store(
        out_ptr + batch_id * stride_o_batch + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j,
        out,
        mask=(offs_i[:, None] < N) & (offs_j[None, :] < N)
    )
```

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Key ideas
---------
1. LayerNorm and all linear layers are executed in PyTorch (fp16) for
   simplicity and numerical stability.
2. A small Triton kernel fuses mask broadcasting and gate multiplication,
   reducing memory traffic.
3. The expensive contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is implemented as a batched matrix multiplication (left @ rightᵀ) with a
   custom Triton GEMM kernel that works on the merged batch‐hidden dimension.
   This avoids the O(N³) Python loop and leverages block‑wise tiling.
4. The rest (output LayerNorm, out‑gate, final linear) is performed in PyTorch.
   The final tensor is returned in the original dtype (usually float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Kernel: fuse mask + gates (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,                # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,      # same shape as left/right
    mask_ptr,                           # [B,N,N] flattened
    total_elems, hidden_dim,            # runtime scalars
    HAS_MASK: tl.constexpr,             # compile‑time bool
    BLOCK_SIZE: tl.constexpr,           # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left      = tl.load(left_ptr      + offs, mask=active, other=0.0, dtype=tl.float16)
    right     = tl.load(right_ptr     + offs, mask=active, other=0.0, dtype=tl.float16)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0, dtype=tl.float16)
    right_gate= tl.load(right_gate_ptr+ offs, mask=active, other=0.0, dtype=tl.float16)

    if HAS_MASK:
        # each hidden_dim entries share a mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0, dtype=tl.float16)
        left  = left  * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)

# ----------------------------------------------------------------------
# Kernel: batched matmul with right matrix transposed
#          out = left @ rightᵀ
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_transpose_kernel(
    left_ptr,    # [B*H, N, N] flattened, fp16
    right_ptr,   # [B*H, N, N] flattened, fp16
    out_ptr,     # [B*H, N, N] flattened, fp16
    B_H: tl.constexpr,      # B * hidden_dim
    N: tl.constexpr,        # sequence length
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_j,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Guard against out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M x K)
        left = tl.load(
            left_ptr + batch_id * stride_l_batch
                     + offs_i[:, None] * stride_l_i
                     + offs_k[None, :] * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # Load right tile transposed (K x N)
        # rightᵀ[k, j] = right[j, k]
        right = tl.load(
            right_ptr + batch_id * stride_r_batch
                      + offs_k[:, None] * stride_r_k
                      + offs_j[None, :] * stride_r_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        acc += tl.dot(left, right)

    # Write result
    out = acc.to(tl.float16)
    tl.store(
        out_ptr + batch_id * stride_o_batch
                + offs_i[:, None] * stride_o_i
                + offs_j[None, :] * stride_o_j,
        out,
        mask=mask_i[:, None] & mask_j[None, :],
    )

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    ##################################################################
    # 1. Input LayerNorm (float32 for stability)
    ##################################################################
    norm_weight = weights["norm.weight"].to(device).to(torch.float32)
    norm_bias   = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Cast to fp16 for the heavy part
    x = x.to(torch.float16)

    ##################################################################
    # 2. Load linear weights (fp16)
    ##################################################################
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    ##################################################################
    # 3. Projections and gates
    ##################################################################
    left = F.linear(x, left_proj_w)            # (B, N, N, H)
    right = F.linear(x, right_proj_w)          # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    ##################################################################
    # 4. Fuse mask + gate multiplication (Triton)
    ##################################################################
    use_mask = (mask is not None) and (not nomask)
    if use_mask:
        # Ensure contiguous tensors for flattening
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()
        mask_fp   = mask.to(device).to(torch.float16).contiguous()

        total = left.numel()
        BLOCK_SIZE = 1024

        # Flatten all tensors
        left_flat       = left.view(-1)
        right_flat      = right.view(-1)
        left_gate_flat  = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat       = mask_fp.view(-1)  # (B*N*N,)

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # Tensors have been updated in‑place; reshape back
        left  = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # Simple element‑wise gate multiplication
        left = left * left_gate
        right = right * right_gate

    ##################################################################
    # 5. Core contraction: batched GEMM (left @ rightᵀ)
    ##################################################################
    B, N, _, H = left.shape  # H == hidden_dim
    # Merge batch and hidden dimensions -> (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_mat   = torch.empty_like(left_mat)

    # Strides for the flattened tensors (row‑major)
    stride_l_batch, stride_l_i, stride_l_k = left_mat.stride()
    stride_r_batch, stride_r_i, stride_r_k = right_mat.stride()
    stride_o_batch, stride_o_i, stride_o_k = out_mat.stride()

    # For the transposed access of the right matrix:
    #   row stride  -> original column stride (stride_r_k)
    #   col stride  -> original row stride    (stride_r_i)
    stride_r_k_trans = stride_r_k   # corresponds to k (row in transposed)
    stride_r_j_trans = stride_r_i   # corresponds to j (col in transposed)

    # Tiling parameters (tuned for typical N up to 1024)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                           # batch * hidden
        triton.cdiv(N, BLOCK_M),         # rows
        triton.cdiv(N, BLOCK_N),         # cols
    )

    _batched_matmul_transpose_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        B * H,
        N,
        stride_l_batch, stride_l_i, stride_l_k,
        stride_r_batch, stride_r_k_trans, stride_r_j_trans,
        stride_o_batch, stride_o_i, stride_o_k,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    ##################################################################
    # 6. Output LayerNorm, out‑gate, final linear projection
    ##################################################################
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # Cast back to the original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
