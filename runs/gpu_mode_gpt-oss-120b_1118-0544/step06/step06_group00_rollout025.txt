```
@triton.jit
def triang_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,
    stride_l_m, stride_l_i, stride_l_k,
    stride_r_m, stride_r_i, stride_r_k,
    stride_o_m, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute block start indices
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        off_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = off_k < N

        # load left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + pid_m * stride_l_m + offs_i[:,None] * stride_l_i + off_k[None,:] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_i[:,None] & mask_k[None,:], other=0.0)

        # load right tile (transposed): we need right[j, k] -> we want tile shape (BLOCK_N, BLOCK_K)
        # Right transposed: B[k, j] = right[pid_m, j, k]
        # So loads: stride_r_m + off_k[:,None] * stride_r_k + offs_j[None,:] * stride_r_i
        b_ptrs = right_ptr + pid_m * stride_r_m + off_k[:,None] * stride_r_k + offs_j[None,:] * stride_r_i
        b = tl.load(b_ptrs, mask=mask_k[:,None] & mask_j[None,:], other=0.0)

        # accumulate
        acc += tl.dot(a, b)

    # Store output
    out_ptrs = out_ptr + pid_m * stride_o_m + offs_i[:,None] * stride_o_i + offs_j[None,:] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:,None] & mask_j[None,:])
```

@triton.jit
def triang_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,
    stride_l_m, stride_l_i, stride_l_k,
    stride_r_m, stride_r_i, stride_r_k,
    stride_o_m, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute start indices
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left tile (M,i,k)
        a_ptrs = left_ptr + pid_m * stride_l_m + i_offset[:,None] * stride_l_i + k_offset[None,:] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_i[:,None] & mask_k[None,:], other=0.0)

        # Load right tile transposed: we need right[j,k] => right[pid_m, j, k] used as B[k, j] for multiplication.
        b_ptrs = right_ptr + pid_m * stride_r_m + k_offset[:,None] * stride_r_k + j_offset[None,:] * stride_r_i
        b = tl.load(b_ptrs, mask=mask_k[:,None] & mask_j[None,:], other=0.0)

        # dot: a (BLOCK_M, BLOCK_K) dot b (BLOCK_K, BLOCK_N) => (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)   # default dtype is float32 for accumulation

    out_ptrs = out_ptr + pid_m * stride_o_m + i_offset[:,None] * stride_o_i + j_offset[None,:] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:,None] & mask_j[None,:])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for mask and gate fusion
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # load values
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden slice shares same mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

# Triton kernel for batched matrix multiplication left @ right^T
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,
    stride_l_m, stride_l_i, stride_l_k,
    stride_r_m, stride_r_i, stride_r_k,
    stride_o_m, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)   # batch*hidden index
    pid_i = tl.program_id(1)   # row block
    pid_j = tl.program_id(2)   # col block

    # compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + pid_m * stride_l_m + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load B tile (transposed): we need B[k, j] = right[pid_m, j, k]
        b_ptrs = right_ptr + pid_m * stride_r_m + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # store result
    out_ptrs = out_ptr + pid_m * stride_o_m + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton-accelerated outgoing Triangle Multiplicative Update (TriMul).
    Steps:
      1. LayerNorm on input (float32).
      2. Linear projections to hidden dim and sigmoid gates.
      3. Fuse mask (if present) and left/right gates via a Triton kernel.
      4. Core contraction out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
         performed as a batched GEMM (left @ right^T) with a custom Triton kernel.
      5. Output LayerNorm, output gate, and final linear projection back to original dim.
      6. Cast result back to the input dtype.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------ #
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------ #
    # 2. Cast to fp16 for heavy compute
    # ------------------------------------------------------------------ #
    x = x.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 3. Linear projections & gates (fp16)
    # ------------------------------------------------------------------ #
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)      # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)    # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------ #
    # 4. Fuse mask and left/right gates
    # ------------------------------------------------------------------ #
    if mask is not None and not nomask:
        # mask broadcast to (B,N,N,1) and fp16
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten tensors
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_fp.contiguous().view(-1)  # (B*N*N,)

        total = left_f.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask – simple element‑wise fusion
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Core contraction via custom Triton GEMM (left @ right^T)
    # ------------------------------------------------------------------ #
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_mat = torch.empty_like(left_mat)  # fp16

    # strides for Triton (in elements)
    stride_l_m, stride_l_i, stride_l_k = left_mat.stride()
    stride_r_m, stride_r_i, stride_r_k = right_mat.stride()
    stride_o_m, stride_o_i, stride_o_j = out_mat.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    # Grid: (M, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        B * H,
        N,
        stride_l_m, stride_l_i, stride_l_k,
        stride_r_m, stride_r_i, stride_r_k,
        stride_o_m, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 6. Output LayerNorm, output gate and final linear projection
    # ------------------------------------------------------------------ #
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------ #
    # 7. Cast back to original dtype
    # ------------------------------------------------------------------ #
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
