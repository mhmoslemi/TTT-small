"""
TriMul (outgoing) forward pass – Triton‑accelerated mask‑gate fusion
and a custom batched GEMM kernel.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for compute‑heavy parts.
3. Linear projections → left/right tensors (B,N,N,H).
4. Gate projections (sigmoid) → left_gate/right_gate/out_gate (B,N,N,H).
5. Fuse projection, gating and optional mask in a Triton kernel
   (`_gate_mask_fuse_kernel`).  If there is no mask the fusion reduces
   to an element‑wise product.
6. Core reduction (batched GEMM):
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented by a custom Triton kernel (`bmm_fp16_kernel`) that
   computes a batch of FP16 matrix multiplications (B*H, N, N) using
   tensor‑cores.
7. Output LayerNorm, multiply by `out_gate`, final linear projection
   back to the original dimension `dim`.
8. Cast the result back to the input dtype (usually float32).

Only the mask‑gate fusion and the batched GEMM are performed in Triton;
all other ops use highly‑optimized PyTorch primitives.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask (mask broadcast over hidden)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # int: B*N*N*H
    hidden_dim,        # int: H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # bounds
    valid = offs < total_elems

    # index into mask (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 2048) -> torch.Tensor:
    """Fuse projection, gate and optional mask (broadcast over hidden_dim)."""
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched FP16 GEMM (C = A @ B) where A,B are (batch, M, K) and (batch, K, N)
# ----------------------------------------------------------------------
@triton.jit
def bmm_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,   # A: (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # B: (batch, K, N)
    stride_cm, stride_cn, stride_cb,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m    = tl.program_id(1)
    pid_n    = tl.program_id(2)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile (k = 0)
    a_block_ptr = a_ptr + batch_id * stride_ab + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_block_ptr = b_ptr + batch_id * stride_bb + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(a_block_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_block_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

        # Move to the next K‑block
        a_block_ptr += BLOCK_K * stride_ak
        b_block_ptr += BLOCK_K * stride_bk

    # Write back the result
    c_block_ptr = c_ptr + batch_id * stride_cb + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_block_ptr,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def batched_gemm_fp16(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication for FP16 tensors:
        C[b] = A[b] @ B[b]   (A: (batch, M, K), B: (batch, K, N))
    Returns C of shape (batch, M, N) in FP16.
    """
    assert A.shape == B.shape, "A and B must have the same shape"
    batch, M, K = A.shape
    N = K

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    C = torch.empty_like(A)

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    bmm_fp16_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        A.stride(1), A.stride(2), A.stride(0),   # stride_am, stride_ak, stride_ab
        B.stride(1), B.stride(2), B.stride(0),   # stride_bk, stride_bn, stride_bb
        C.stride(1), C.stride(2), C.stride(0),   # stride_cm, stride_cn, stride_cb
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (optional)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask + gating (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for the custom batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T  =>  (B*H, N, N)
    out_flat = batched_gemm_fp16(left_flat, right_flat)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
