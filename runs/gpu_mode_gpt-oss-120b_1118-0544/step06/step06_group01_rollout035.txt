    stride_right_b, stride_right_j, stride_right_k, stride_right_h,

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = num_tiles_m * num_tiles_n
    batch_idx = pid // total_tiles
    tile_idx = pid % total_tiles
    pid_m = tile_idx // num_tiles_n
    pid_n = tile_idx % num_tiles_n

    # compute base offsets for batch
    b = batch_idx // H
    h = batch_idx % H

    # base offset contributions
    offset_left = b * stride_left_b + h * stride_left_h
    offset_right = b * stride_right_b + h * stride_right_h
    offset_out = b * stride_out_b + h * stride_out_h

    # row and col indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    num_tiles_k = (N + BLOCK_K - 1) // BLOCK_K
    for k in range(num_tiles_k):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left submatrix
        # left shape (M,K)
        a_ptrs = left_ptr + offset_left + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right submatrix
        # right shape (N,K)
        b_ptrs = right_ptr + offset_right + offs_n[:, None] * stride_right_j + offs_k[None, :] * stride_right_k
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Dot product with transpose of right (so shape (K,N))
        acc += tl.dot(a, b, trans_b=True)

    # Store result
    out_ptrs = out_ptr + offset_out + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    # layernorm
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    # cast to fp16
    x = x.to(torch.float16)
    # Linear projections
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)

    # Linear layers
    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # fuse gate and mask
    use_mask = (mask is not None) and (not nomask)
    if use_mask:
        # mask shape (B,N,N) -> unsqueeze dim for broadcasting
        mask_fp = mask.to(device).to(torch.float16)
        # flatten for kernel
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)   # shape B*N*N
        total = left_flat.numel()
        BLOCK_SIZE = 8192   # maybe 8192 to have many threads
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # Core contraction via Triton
    B, N, _, H = left.shape
    # allocate output
    out = torch.empty_like(left)  # shape (B,N,N,H), same dtype (fp16)

    # compute strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    # Since right indices are j and k, we need stride for j dimension: it's the second dimension (index 1)
    # Actually shape is (B,N,N,H) -> dims: dim0 = batch, dim1 = i, dim2 = k?, but for right we treat second dim as j (same as i dimension) and third as k.
    # So stride_right_b = stride_right[0], stride_right_j = stride_right[1], stride_right_k = stride_right[2], stride_right_h = stride_right[3].
    # We'll pass accordingly.

    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * H * num_tiles_m * num_tiles_n,)

    tri_mul_kernel[grid](
        left, right, out,
        B, H, N,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Convert out to float32 for subsequent ops
    out_fp32 = out.to(torch.float32)
    # Output LayerNorm
    out_norm = torch.nn.functional.layer_norm(out_fp32, (H,), weight=to_out_norm_w.to(torch.float32), bias=to_out_norm_b.to(torch.float32))
    # Multiply by out_gate (need matching dtype)
    out_gate_fp32 = out_gate.to(torch.float32)
    out_norm = out_norm * out_gate_fp32
    # Final linear projection
    out_final = torch.nn.functional.linear(out_norm, to_out_w.to(torch.float32))
    # Cast back to input dtype
    if out_final.dtype != dtype:
        out_final = out_final.to(dtype)
    return out_final
```

- Ensure that we pass correct block size and other arguments.

"""
Highly‑optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the input (float32 for stability).
2. Cast to fp16 and apply linear projections and sigmoid gates.
3. Fuse mask, left‑gate and right‑gate element‑wise with a tiny Triton kernel.
4. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   implemented as a batched GEMM (left @ rightᵀ) with a custom Triton kernel.
   The kernel tiles the i/j dimensions (BLOCK_M × BLOCK_N) and iterates over k
   in blocks (BLOCK_K), using Tensor‑Core math (fp16) via `tl.dot`.
5. Apply output LayerNorm, the output gate, and a final linear projection back
   to the original feature dimension.
All heavy arithmetic is performed in fp16; the final result is returned in the
original input dtype (usually float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Fuse mask and gate multiplication (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,                # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,      # same shape as left/right
    mask_ptr,                           # [B,N,N] flattened
    total_elems, hidden_dim,            # runtime scalars
    HAS_MASK: tl.constexpr,             # compile‑time bool
    BLOCK_SIZE: tl.constexpr,           # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left      = tl.load(left_ptr      + offs, mask=active, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden_dim entries share one mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


# ----------------------------------------------------------------------
# 2) Core TriMul contraction (batched GEMM left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of tiles per output matrix
    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    tiles_per_batch = num_tiles_m * num_tiles_n

    batch_idx = pid // tiles_per_batch                     # 0 .. B*H-1
    tile_idx  = pid % tiles_per_batch
    pid_m = tile_idx // num_tiles_n
    pid_n = tile_idx % num_tiles_n

    # decompose combined batch index into (b, h)
    b = batch_idx // H
    h = batch_idx % H

    # base offsets for this (b, h) slice
    off_left  = b * stride_left_b  + h * stride_left_h
    off_right = b * stride_right_b + h * stride_right_h
    off_out   = b * stride_out_b   + h * stride_out_h

    # tile start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # iterate over K dimension in blocks
    num_tiles_k = (N + BLOCK_K - 1) // BLOCK_K
    for k in range(num_tiles_k):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + off_left + \
                 offs_m[:, None] * stride_left_i + \
                 offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # right tile: shape (BLOCK_N, BLOCK_K)
        b_ptrs = right_ptr + off_right + \
                 offs_n[:, None] * stride_right_i + \
                 offs_k[None, :] * stride_right_k
        b = tl.load(b_ptrs,
                    mask=mask_n[:, None] & mask_k[None, :],
                    other=0.0)

        # accumulate a @ bᵀ   (b is (N,K) -> transposed inside dot)
        acc += tl.dot(a, b, trans_b=True)

    # store result
    out_ptrs = out_ptr + off_out + \
               offs_m[:, None] * stride_out_i + \
               offs_n[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 3) Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple):
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for the heavy arithmetic
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Load linear weights (fp16)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections and gating
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)           # (B, N, N, H)
    right = F.linear(x, right_proj_w)         # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4. Fuse mask and gate multiplications (Triton)
    # ------------------------------------------------------------------
    use_mask = (mask is not None) and (not nomask)
    if use_mask:
        mask_fp = mask.to(device).to(torch.float16)

        # flatten tensors for kernel
        left_f    = left.contiguous().view(-1)
        right_f   = right.contiguous().view(-1)
        lgate_f   = left_gate.contiguous().view(-1)
        rgate_f   = right_gate.contiguous().view(-1)
        mask_f    = mask_fp.contiguous().view(-1)      # (B*N*N,)

        total = left_f.numel()
        BLOCK_SIZE = 8192  # tuneable

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
        _gate_mask_fuse_kernel[grid](
            left_f, right_f,
            lgate_f, rgate_f,
            mask_f,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # Views are updated in‑place; reshape back for later use
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via custom Triton GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # (B, N, N, H), fp16

    # strides (must be integers)
    sL_b, sL_i, sL_k, sL_h = left.stride()
    sR_b, sR_i, sR_k, sR_h = right.stride()
    sO_b, sO_i, sO_j, sO_h = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * H * num_tiles_m * num_tiles_n,)

    _tri_mul_kernel[grid](
        left, right, out,
        B, H, N,
        sL_b, sL_i, sL_k, sL_h,
        sR_b, sR_i, sR_k, sR_h,
        sO_b, sO_i, sO_j, sO_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)   # cast for stable LayerNorm
    out_norm = F.layer_norm(
        out_fp32,
        (H,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32)
    )
    out_norm = out_norm * out_gate.to(torch.float32)
    out_final = F.linear(out_norm, to_out_w.to(torch.float32))

    # Cast back to the original dtype
    if out_final.dtype != dtype_in:
        out_final = out_final.to(dtype_in)

    return out_final
