@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    stride_gate_batch, stride_gate_i, stride_gate_k, stride_gate_d,
    stride_mask_batch, stride_mask_i, stride_mask_k,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Compute program ids
    batch_h = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # derive b and d
    b = batch_h // H
    d = batch_h % H

    # Compute start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Create range vectors
    i_range = i_start + tl.arange(0, BLOCK_M)[:, None]  # (M,1)
    j_range = j_start + tl.arange(0, BLOCK_N)[None, :]  # (1,N)
    # Loop over k tiles
    for k_start in range(0, tl.cdiv(N, BLOCK_K)):
        k_range = k_start * BLOCK_K + tl.arange(0, BLOCK_K)[None, :]  # (1,K)

        # Compute masks for valid indices
        i_mask = i_range < N
        j_mask = j_range < N
        k_mask = k_range < N

        # Offsets for left (M,K)
        left_offset = (
            b * stride_left_batch
            + i_range * stride_left_i
            + k_range * stride_left_k
            + d * stride_left_d
        )
        # left_gate offsets
        left_gate_offset = (
            b * stride_gate_batch
            + i_range * stride_gate_i
            + k_range * stride_gate_k
            + d * stride_gate_d
        )
        # mask offset for left (M,K)
        mask_offset_left = (
            b * stride_mask_batch
            + i_range * stride_mask_i
            + k_range * stride_mask_k
        )

        # Load tensors
        left = tl.load(left_ptr + left_offset, mask=i_mask & k_mask, other=0.0)
        left_gate = tl.load(left_gate_ptr + left_gate_offset, mask=i_mask & k_mask, other=0.0)
        mask_left = tl.load(mask_ptr + mask_offset_left, mask=i_mask & k_mask, other=1.0)

        left_fused = left * mask_left * left_gate

        # Offsets for right (K,N) -> note we need shape (K,N)
        # We'll compute offset using k_range (rows) and j_range (cols)
        right_offset = (
            b * stride_right_batch
            + j_range * stride_right_i
            + k_range * stride_right_k
            + d * stride_right_d
        )
        right_gate_offset = (
            b * stride_gate_batch
            + j_range * stride_gate_i
            + k_range * stride_gate_k
            + d * stride_gate_d
        )
        mask_offset_right = (
            b * stride_mask_batch
            + j_range * stride_mask_i
            + k_range * stride_mask_k
        )

        # Load tensors
        right = tl.load(right_ptr + right_offset, mask=j_mask & k_mask, other=0.0)
        right_gate = tl.load(right_gate_ptr + right_gate_offset, mask=j_mask & k_mask, other=0.0)
        mask_right = tl.load(mask_ptr + mask_offset_right, mask=j_mask & k_mask, other=1.0)

        right_fused = right * mask_right * right_gate

        # Compute dot product and accumulate
        # left_fused shape (M,K), right_fused shape (K,N)
        acc += tl.dot(left_fused, right_fused)

    # Store result
    # Compute out offset (M,N)
    out_offset = (
        b * stride_out_batch
        + i_range * stride_out_i
        + j_range * stride_out_j
        + d * stride_out_d
    )
    # Convert accumulator to fp16 for store
    acc_fp16 = acc.to(tl.float16)
    tl.store(out_ptr + out_offset, acc_fp16, mask=i_mask & j_mask)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The kernel fuses mask application, sigmoid gates and the core
`out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]` reduction.
Each (batch, hidden) pair is treated as an independent GEMM where the
right operand is accessed transposed so that the sum over `k` is performed
inside the kernel (no intermediate Python calls).  The surrounding
LayerNorms, linear projections and the final projection are kept in
PyTorch for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    # pointers
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr, out_ptr,
    # sizes
    B, N, H,
    # strides (contiguous layout)
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    stride_gate_batch, stride_gate_i, stride_gate_k, stride_gate_d,
    stride_mask_batch, stride_mask_i, stride_mask_k,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute one tile of the TriMul reduction for a specific (batch, hidden)
    pair.  The algorithm follows a standard tiled GEMM where the right matrix
    is accessed in transposed order so that the product implements the
    required sum over the sequence dimension `k`.
    """
    batch_h = tl.program_id(0)      # combined batch*hidden index
    pid_i   = tl.program_id(1)      # tile index over i (rows)
    pid_j   = tl.program_id(2)      # tile index over j (cols)

    # decode batch and hidden indices
    b = batch_h // H
    d = batch_h % H

    # start indices of the current tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # ------------------------------------------------------------------
    # masks for the tile in the i and j dimensions (static for the tile)
    # ------------------------------------------------------------------
    i_valid = i_start + tl.arange(0, BLOCK_M) < N   # (M,)
    j_valid = j_start + tl.arange(0, BLOCK_N) < N   # (N,)

    # ------------------------------------------------------------------
    # accumulator in fp32 for numerical stability
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # iterate over the k dimension in tiles
    # ------------------------------------------------------------------
    for k_start in range(0, tl.cdiv(N, BLOCK_K)):
        # k indices for the current tile
        k_valid = k_start + tl.arange(0, BLOCK_K) < N   # (K,)

        # ------------------------------------------------------------------
        #  left tile  : shape (M, K)
        # ------------------------------------------------------------------
        i_idx = i_start + tl.arange(0, BLOCK_M)[:, None]          # (M,1)
        k_idx = k_start + tl.arange(0, BLOCK_K)[None, :]          # (1,K)

        left_off = (
            b * stride_left_batch
            + i_idx * stride_left_i
            + k_idx * stride_left_k
            + d * stride_left_d
        )
        left_gate_off = (
            b * stride_gate_batch
            + i_idx * stride_gate_i
            + k_idx * stride_gate_k
            + d * stride_gate_d
        )
        mask_off_left = (
            b * stride_mask_batch
            + i_idx * stride_mask_i
            + k_idx * stride_mask_k
        )

        left_mask = i_valid[:, None] & k_valid[None, :]
        l = tl.load(left_ptr + left_off,      mask=left_mask,      other=0.0)
        lg = tl.load(left_gate_ptr + left_gate_off, mask=left_mask, other=0.0)
        m_l = tl.load(mask_ptr + mask_off_left,   mask=left_mask, other=1.0)

        left_fused = l * m_l * lg                     # (M, K)

        # ------------------------------------------------------------------
        #  right tile : shape (K, N)  (the matrix is accessed transposed)
        # ------------------------------------------------------------------
        k_idx_T = k_start + tl.arange(0, BLOCK_K)[:, None]      # (K,1)
        j_idx   = j_start + tl.arange(0, BLOCK_N)[None, :]      # (1,N)

        right_off = (
            b * stride_right_batch
            + k_idx_T * stride_right_k
            + j_idx   * stride_right_i
            + d * stride_right_d
        )
        right_gate_off = (
            b * stride_gate_batch
            + k_idx_T * stride_gate_k
            + j_idx   * stride_gate_i
            + d * stride_gate_d
        )
        mask_off_right = (
            b * stride_mask_batch
            + k_idx_T * stride_mask_k
            + j_idx   * stride_mask_i
        )

        right_mask = k_valid[:, None] & j_valid[None, :]
        r = tl.load(right_ptr + right_off,       mask=right_mask, other=0.0)
        rg = tl.load(right_gate_ptr + right_gate_off, mask=right_mask, other=0.0)
        m_r = tl.load(mask_ptr + mask_off_right,      mask=right_mask, other=1.0)

        right_fused = r * m_r * rg                    # (K, N)

        # ------------------------------------------------------------------
        #  accumulate the GEMM product   (M,K) @ (K,N) -> (M,N)
        # ------------------------------------------------------------------
        acc += tl.dot(left_fused, right_fused)        # fp32 accumulation

    # ------------------------------------------------------------------
    #  store the result tile
    # ------------------------------------------------------------------
    out_i = i_start + tl.arange(0, BLOCK_M)[:, None]   # (M,1)
    out_j = j_start + tl.arange(0, BLOCK_N)[None, :]   # (1,N)

    out_off = (
        b * stride_out_batch
        + out_i * stride_out_i
        + out_j * stride_out_j
        + d * stride_out_d
    )
    store_mask = i_valid[:, None] & j_valid[None, :]
    tl.store(out_ptr + out_off, acc.to(tl.float16), mask=store_mask)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operation (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, if True ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # unpack and prepare
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)    # rest of the pipeline works in fp16

    # ------------------------------------------------------------------
    # 2) Cast weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Mask handling (broadcast over hidden dim)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_tensor = mask.to(device=device, dtype=torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction – fused Triton kernel
    # ------------------------------------------------------------------
    out = torch.empty_like(left)   # (B, N, N, H) fp16

    # Strides for contiguous tensors
    stride_batch = N * N * hidden_dim
    stride_i = N * hidden_dim
    stride_k = hidden_dim
    stride_d = 1

    stride_mask_batch = N * N
    stride_mask_i = N
    stride_mask_k = 1

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                              # batch*hidden
        (N + BLOCK_M - 1) // BLOCK_M,                # i tiles
        (N + BLOCK_N - 1) // BLOCK_N,                # j tiles
    )

    _tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_tensor,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_batch=stride_batch,
        stride_left_i=stride_i,
        stride_left_k=stride_k,
        stride_left_d=stride_d,
        stride_right_batch=stride_batch,
        stride_right_i=stride_i,
        stride_right_k=stride_k,
        stride_right_d=stride_d,
        stride_gate_batch=stride_batch,
        stride_gate_i=stride_i,
        stride_gate_k=stride_k,
        stride_gate_d=stride_d,
        stride_mask_batch=stride_mask_batch,
        stride_mask_i=stride_mask_i,
        stride_mask_k=stride_mask_k,
        stride_out_batch=stride_batch,
        stride_out_i=stride_i,
        stride_out_j=stride_k,
        stride_out_d=stride_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
