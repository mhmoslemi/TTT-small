    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
    batch_idx,

@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_ci, stride_cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A
        a = tl.load(a_ptr + pid_batch * stride_am
                     + offs_m[:, None] * stride_ai
                     + offs_k[None, :] * stride_ak,
                     mask=(offs_m < M) & (offs_k < K), other=0.0)
        # Load B (note we need B_T)
        b = tl.load(b_ptr + pid_batch * stride_bm
                     + offs_k[:, None] * stride_bk
                     + offs_n[None, :] * stride_bn,
                     mask=(offs_k < K) & (offs_n < N), other=0.0)
        # Accumulate
        acc += tl.dot(a, b)

    # Write back
    c = acc.to(tl.float16)
    tl.store(c_ptr + pid_batch * stride_cm
                + offs_m[:, None] * stride_ci
                + offs_n[None, :] * stride_cn,
                c,
                mask=(offs_m < M) & (offs_n < N))
```

"""
custom_kernel: Triton‑accelerated outgoing TriMul (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Cast to fp16 and cast all linear weights to fp16.
3. Linear projections (no bias) produce left/right values and three gates.
4. Optionally fuse a mask with the left/right projections.
5. Core “TriMul” reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (left @ rightᵀ) with batch
   size = B * hidden_dim.  It is implemented with a custom Triton kernel
   that tiles the i‑, j‑ and k‑dimensions and uses fp16 dot‑products with
   fp32 accumulation.
6. Layer‑norm over the hidden dimension, apply the out‑gate and a final
   linear projection back to the original channel dimension.
7. Cast back to the input dtype (usually float32) and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_am, stride_ai, stride_ak,   # strides for A (M, I, K)
    stride_bm, stride_bk, stride_bn,   # strides for B (M, K, N)  – B is already transposed
    stride_cm, stride_ci, stride_cn,   # strides for C (M, I, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[M,I,N] = A[M,I,K] @ B[M,K,N] (B is the transposed right matrix)."""
    pid_batch = tl.program_id(0)   # batch index = B * hidden_dim
    pid_i = tl.program_id(1)       # tile over the first output dimension (i)
    pid_j = tl.program_id(2)       # tile over the second output dimension (j)

    # Offsets for the output tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load a tile of A: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_batch * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak,
            mask=(offs_i[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )
        # Load a tile of B (already transposed): shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_batch * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bn,
            mask=(offs_k[:, None] < K) & (offs_j[None, :] < N),
            other=0.0,
        )
        # Accumulate the dot product
        acc += tl.dot(a, b)

    # Write back result in fp16
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_batch * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cn,
        c,
        mask=(offs_i[:, None] < M) & (offs_j[None, :] < N),
    )


def _batched_matmul(a: torch.Tensor, b: torch.Tensor,
                    block_m: int = 64, block_n: int = 64, block_k: int = 32) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    a, b: tensors of shape (M, N, K) and (M, K, N) respectively, dtype float16.
    Returns c of shape (M, N, N) dtype float16.
    """
    assert a.is_contiguous() and b.is_contiguous()
    M, N, K = a.shape[0], a.shape[1], a.shape[2]
    # Output tensor
    c = torch.empty_like(a)

    # Compute strides (in elements)
    stride_am, stride_ai, stride_ak = a.stride()
    stride_bm, stride_bk, stride_bn = b.stride()
    stride_cm, stride_ci, stride_cn = c.stride()

    grid = (
        M,                                     # batch dimension
        triton.cdiv(N, block_m),               # tile rows
        triton.cdiv(N, block_n),               # tile cols
    )
    _batched_matmul_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_ci,
        stride_cn,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return c


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul layer using Triton for the
    heavy batched‑GEMM reduction.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] (0/1) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """

    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension (`dim`)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim) – still in input dtype

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast all linear weights to fp16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections and gates (no bias)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)          # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Apply mask (if any) and combine projection with gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask exists but we are instructed to ignore it → still apply broadcasting
            mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask_fp16
            right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (left @ rightᵀ) with a Triton kernel.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                         # B, seq_len, seq_len, hidden_dim
    # reshape to (M, N, N) where M = B * H
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (M, N, N)
    # right must be transposed on the last two dimensions for the GEMM
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N) # (M, N, N) == (M, K, N)

    out_flat = _batched_matmul(left_flat, right_flat)   # (M, N, N) fp16

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)       # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
