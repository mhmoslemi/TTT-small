"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor `x ∈ [B,N,N,C]`.
2. Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
   into a single matrix multiplication:
        [L,R,LG,RG,OG] = x @ W_fused   where W_fused has shape (5*H, C).
   The result is reshaped to (B,N,N,5*H) and split.
3. Apply sigmoid to the three gate tensors.
4. Fuse mask, left/right gates and the projected tensors with a tiny
   Triton kernel:
        left_out  = left  * mask * left_gate
        right_out = right * mask * right_gate
5. Core reduction (outgoing TriMul):
        out[b,i,j,h] = Σ_k left_out[b,i,k,h] * right_out[b,j,k,h]
   Implemented with `torch.einsum`, which internally maps to a batched GEMM
   and avoids extra memory copies.
6. Second LayerNorm over the hidden dimension H.
7. Multiply by the output gate and project back to the original channel
   dimension C with a final linear layer.
8. Cast the result back to the original dtype.

All heavy linear work is performed by a single fused GEMM, and the
mask‑gate fusion is executed in a custom Triton kernel, satisfying the
requirement to use Triton while keeping the overall runtime low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,            # const * left   (B,N,N,H) flattened
    right_ptr,           # const * right  (B,N,N,H) flattened
    mask_ptr,            # const * mask   (B,N,N)   flattened
    left_gate_ptr,       # const * left_gate  (B,N,N,H) flattened
    right_gate_ptr,      # const * right_gate (B,N,N,H) flattened
    left_out_ptr,        # * left_out   (B,N,N,H) flattened
    right_out_ptr,       # * right_out  (B,N,N,H) flattened
    H: tl.constexpr,                 # hidden dimension
    total_elements: tl.constexpr,   # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask and gates:
       left_out  = left  * mask * left_gate
       right_out = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load values
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Compute index into the (B,N,N) mask (repeated H times)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Write fused results
    tl.store(left_out_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (fp32 or fp16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, if True ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # 0) Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for all subsequent heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Fuse five linear projections into one GEMM
    # ------------------------------------------------------------------
    #   left_proj, right_proj, left_gate, right_gate, out_gate
    #   each has weight shape (hidden_dim, dim)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)  # shape (5*H, C)

    B, N, _, _ = input_tensor.shape
    x_flat = x.view(-1, dim)                     # (B*N*N, C)
    fused_out = F.linear(x_flat, fused_weight)   # (B*N*N, 5*H)

    # reshape to (B, N, N, 5*H) and split
    fused_out = fused_out.view(B, N, N, 5 * hidden_dim)
    left = fused_out[..., 0:hidden_dim]
    right = fused_out[..., hidden_dim : 2 * hidden_dim]
    left_gate = fused_out[..., 2 * hidden_dim : 3 * hidden_dim]
    right_gate = fused_out[..., 3 * hidden_dim : 4 * hidden_dim]
    out_gate = fused_out[..., 4 * hidden_dim : 5 * hidden_dim]

    # ------------------------------------------------------------------
    # 4) Apply sigmoid to gate tensors
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 5) Prepare mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_tensor = mask.to(device=device, dtype=torch.float16)  # (B,N,N)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6) Fuse mask and gates with a tiny Triton kernel
    # ------------------------------------------------------------------
    total_elements = left.numel()  # B * N * N * H
    BLOCK = 1024

    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = (triton.cdiv(total_elements, BLOCK),)
    _mask_gate_fusion_kernel[grid](
        left,
        right,
        mask_tensor.view(-1),      # flatten mask (B*N*N)
        left_gate,
        right_gate,
        left_out,
        right_out,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Core TriMul reduction: out[b,i,j,h] = Σ_k left_out[b,i,k,h] * right_out[b,j,k,h]
    #    Implemented via torch.einsum which maps to a batched GEMM.
    # ------------------------------------------------------------------
    out = torch.einsum("b i k h, b j k h -> b i j h", left_out, right_out)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"]
        .to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"]
        .to(device=device, dtype=torch.float16),
    )

    # ------------------------------------------------------------------
    # 9) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10) Final projection back to original channel dimension C
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_weight)  # (B,N,N,C)

    # ------------------------------------------------------------------
    # 11) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
