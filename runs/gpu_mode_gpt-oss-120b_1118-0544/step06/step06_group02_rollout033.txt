@triton.jit
def triton_trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_ri, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_task = tl.program_id(2)  # batch*hidden
    pid_row = tl.program_id(0)
    pid_col = tl.program_id(1)

    # compute batch and hidden index
    b = pid_task // H
    d = pid_task % H

    # compute tile start indices
    i = pid_row * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_col * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_i = i < N
    mask_j = j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k blocks
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        # k block start
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # compute pointer offsets
        # left: left[b,i,k,d]

"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate and
Triton‐based batched matmul (no data copies).

Algorithm
---------
1. Layer‑normalize the input tensor (float32) using the provided weight/bias.
2. Cast to fp16 for the compute‑intensive part.
3. Linear projections (left/right) and their gates are computed with
   `torch.nn.functional.linear` in fp16.
4. Fuse projection, gate and optional mask with a tiny Triton kernel
   (`_gate_mask_kernel`). This removes two memory passes.
5. Core O(N³·H) reduction is a batched matrix multiplication:
   for each hidden slice `d` we compute
        out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
   Implemented as a custom Triton kernel (`_trimul_kernel`) that works on
   the original (B,N,N,H) layout – no costly `permute(...).contiguous()`.
6. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate and linearly project back to the original channel dimension.
7. Cast the result back to the input dtype (float32).

Only the mask‑gate fusion and the O(N³·H) reduction are written in Triton,
satisfying the requirement of “part of the operations in a kernel”.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask (element‑wise)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask is broadcasted over the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 2048,
) -> torch.Tensor:
    """Fuse `proj * gate * mask` using a Triton kernel."""
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,          # fp16 * (B, N, N, H)
    right_ptr,         # fp16 * (B, N, N, H)
    out_ptr,           # fp16 * (B, N, N, H)
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_ri, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)      # block row (i)
    pid_n = tl.program_id(1)      # block col (j)
    pid_t = tl.program_id(2)      # batch * hidden index

    # derive batch and hidden indices
    b = pid_t // H
    d = pid_t % H

    # compute start offsets for this tile
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N

    # range of indices inside the tile
    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)

    # boundary masks
    mask_i = i < N
    mask_j = j < N

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute base pointer offsets that are constant for the whole tile
    base_left = b * stride_lb + d * stride_ld
    base_right = b * stride_rb + d * stride_rd
    base_out = b * stride_ob + d * stride_od

    # iterate over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left tile: shape (BLOCK_M, BLOCK_K) -> left[b, i, k, d]
        left_ptrs = left_ptr + base_left + i[:, None] * stride_li + k[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile transposed: we need (BLOCK_K, BLOCK_N) for dot
        # right[b, j, k, d] -> transpose to (k, j)
        right_ptrs_T = right_ptr + base_right + k[:, None] * stride_rk + j[None, :] * stride_ri
        right_T = tl.load(right_ptrs_T, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product over k dimension
        acc += tl.dot(left, right_T)

    # write out the result (cast back to fp16)
    out_ptrs = out_ptr + base_out + i[:, None] * stride_oi + j[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def _trimul_reduce(left: torch.Tensor, right: torch.Tensor,
                   BLOCK_M=64, BLOCK_N=64, BLOCK_K=64) -> torch.Tensor:
    """
    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    using a custom Triton kernel (no data copies).
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # same shape [B,N,N,H]

    # strides in element counts (not bytes)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_tasks = B * H

    grid = (num_blocks_m, num_blocks_n, total_tasks)

    _trimul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_ri, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys:
            * "dim" (int) – input channel dimension C
            * "hidden_dim" (int) – hidden dimension H
            * "nomask" (bool, optional) – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input, float32)
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------- LayerNorm (input) -------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for the compute‑heavy part
    x_fp16 = x.to(torch.float16)

    # ------------------- cast weights to fp16 ----------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------- linear projections ------------------------------
    left_proj = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B,N,N,H]
    right_proj = torch.nn.functional.linear(x_fp16, right_proj_w)

    # ------------------- gates -------------------------------------------
    left_gate = torch.nn.functional.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, out_gate_w).sigmoid()   # [B,N,N,H]

    # ------------------- mask & gate fusion (Triton) --------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------- core O(N³·H) reduction via Triton -------------
    out = _trimul_reduce(left, right)   # shape [B,N,N,H] fp16

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ------------------- final linear projection ------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # ------------------- cast back to input dtype -----------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
