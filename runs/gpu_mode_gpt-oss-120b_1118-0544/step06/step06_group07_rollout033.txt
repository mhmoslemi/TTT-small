    device = input_tensor.device
    dtype_in = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input layernorm (float32)
    norm_w = weights["norm.weight"].to(device).float()
    norm_b = weights["norm.bias"].to(device).float()
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b) # B,N,N,C

    # Cast to fp16
    x = x.to(torch.float16)

    # 2. Linear projections
    left_proj_w = weights["left_proj.weight"].to(device).half()
    right_proj_w = weights["right_proj.weight"].to(device).half()

    right_gate_w = weights["right_gate.weight"].to(device).half()
    out_gate_w = weights["out_gate.weight"].to(device).half()

    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_w) # ...

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm (FP32) on the [B,N,N,C] input.
2. Cast to FP16 for the compute‑heavy path.
3. Linear projections (left/right) and gates (left/right/out) with bias‑less
   fully‑connected layers.
4. Fuse mask (if present) and gate multiplications in a single Triton kernel
   (in‑place on left/right tensors).
5. Core O(N³·H) contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (left @ rightᵀ) using a custom Triton kernel
   that processes B·H independent matrices.
6. Output LayerNorm, gating with out_gate and final linear projection back to C.
7. Cast back to the original dtype (float32) and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask and gate (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # load
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # each mask element is shared across `hidden_dim` consecutive values
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the O(N³) contraction
#   Computes out[b,d] = left[b,d] @ right[b,d]^T   where
#   left, right : (B*H, N, N)  (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak,   # A: batch, row, col
    stride_bb, stride_bk, stride_bn,   # B (interpreted as K,N)
    stride_cb, stride_cm, stride_cn,   # C: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch index (B*H)

    # --------------------------------------------------------------
    # Offsets within the block
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # --------------------------------------------------------------
    # Pointers for this batch
    # --------------------------------------------------------------
    a_ptrs = a_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # --------------------------------------------------------------
    # Masks for partial tiles
    # --------------------------------------------------------------
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K‑loop
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_remaining = K - k * BLOCK_K
        mask_k = offs_k < k_remaining

        mask_a = mask_m[:, None] & mask_k[None, :]
        mask_b = mask_k[:, None] & mask_n[None, :]

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # dot returns fp16; cast to fp32 for stable accumulation
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

        # advance K‑dim pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Write back
    # --------------------------------------------------------------
    c_ptrs = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).float()
    norm_b = weights["norm.bias"].to(device).float()
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # --------------------------------------------------------------
    # 2. Cast to fp16 for the heavy path
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (bias‑less) and gates
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).half()
    right_proj_w  = weights["right_proj.weight"].to(device).half()
    left_gate_w   = weights["left_gate.weight"].to(device).half()
    right_gate_w  = weights["right_gate.weight"].to(device).half()
    out_gate_w    = weights["out_gate.weight"].to(device).half()

    left = F.linear(x, left_proj_w)                     # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)                   # (B,N,N,hidden_dim)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 4. Fuse mask and gate (in‑place) via Triton
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape -> (B,N,N,1) for broadcasting
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1).contiguous()

        B, N, _, H = left.shape
        total = B * N * N * H

        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)   # (B*N*N,)

        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # no mask – simple element‑wise gating
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core O(N³·H) contraction: batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_mat = torch.empty_like(left_mat)

    # strides
    stride_ab, stride_am, stride_ak = left_mat.stride()      # (batch, row, col)
    stride_bb, stride_bk, stride_bn = right_mat.stride()     # (batch, col(K), row(N))
    stride_cb, stride_cm, stride_cn = out_mat.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cb, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).half()
    to_out_norm_b = weights["to_out_norm.bias"].to(device).half()
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).half()
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # --------------------------------------------------------------
    # 7. Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(dtype_in)

    return out
