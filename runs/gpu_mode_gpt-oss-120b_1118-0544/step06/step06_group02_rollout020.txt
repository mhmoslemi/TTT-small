"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) using PyTorch.
2. Cast to fp16 for the compute‑intensive portion.
3. Fuse each linear projection with its gate (two weights merged) via a single
   `torch.nn.functional.linear`.  The result is split into projection and gate.
4. Apply mask, gate and sigmoid in a tiny Triton kernel that computes
   `proj * gate * mask` in one pass, avoiding extra memory traffic.
5. Core O(N³·H) reduction: for each hidden channel d we need
   `out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]`.  This is exactly a
   batched matrix multiplication `left_d @ right_dᵀ`.  By reshaping we obtain
   a single `torch.bmm` on tensors of shape (B*H, N, N), which is handled by
   cuBLAS/Tensor‑cores.
6. Apply a second LayerNorm, multiply by the output gate and a final linear
   projection back to the original channel dimension.
7. Cast the result back to the input dtype (float32).

Only the mask‑&‑gate fusion is implemented in Triton, satisfying the
requirement of having at least one custom kernel.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: (proj * gate * mask) – mask is broadcast over the hidden dim
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,      # fp16*  (B*N*N*H,)
    gate_ptr,      # fp16*  (B*N*N*H,)
    mask_ptr,      # fp16*  (B*N*N,)
    out_ptr,       # fp16*  (B*N*N*H,)
    total_elems,   # i32 – total number of elements in proj/gate
    hidden_dim,    # i32 – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask index = floor(offs / H) because H is the fastest‑changing dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using a Triton kernel.

    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],               # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] (or None)
        - weights      : dict of model parameters (float32)
        - config       : dict with keys:
            * "dim" (int)        – input channel dimension C
            * "hidden_dim" (int) – hidden dimension H
            * "nomask" (bool)    – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------- first LayerNorm ---------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for the heavy part
    x = x.to(torch.float16)

    # ------------------- prepare combined weights ------------------------
    # left: proj + gate
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, C)
    # right: proj + gate
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, C)
    # out_gate
    out_gate_w = weights["out_gate.weight"].to(torch.float16)  # (H, C)
    # to_out_norm
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    # final linear
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (C, H)

    # ------------------- combined left projection + gate -----------------
    left_combined = torch.nn.functional.linear(x, left_combined_w)  # (B,N,N,2*H)
    left_proj, left_gate = torch.split(left_combined, hidden_dim, dim=-1)
    left_gate = left_gate.sigmoid()  # (B,N,N,H)

    # ------------------- combined right projection + gate ----------------
    right_combined = torch.nn.functional.linear(x, right_combined_w)  # (B,N,N,2*H)
    right_proj, right_gate = torch.split(right_combined, hidden_dim, dim=-1)
    right_gate = right_gate.sigmoid()  # (B,N,N,H)

    # ------------------- output gate ------------------------------------
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()  # (B,N,N,H)

    # ------------------- mask & gate fusion (Triton) --------------------
    if mask is not None and not nomask:
        # mask broadcasted to hidden dim
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------- core O(N³·H) reduction via batched GEMM -------
    # reshape to (B*H, N, N) for batched matmul
    B, N, _, H = left.shape
    left_reshaped = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # out_reshaped shape: (B*H, N, N)
    out_reshaped = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_reshaped.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- second LayerNorm & output gate ---------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # Cast back to original dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
