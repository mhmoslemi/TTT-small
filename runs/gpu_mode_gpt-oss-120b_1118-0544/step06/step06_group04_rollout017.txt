@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B_H, N,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_m, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start indices
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # compute mask for valid
        k_valid = k_offset < N
        m_valid = off_m < N
        n_valid = off_n < N

        # pointers
        left_ptrs = left_ptr + pid_bh * stride_l_batch \
                    + off_m[:, None] * stride_l_m \
                    + k_offset[None, :] * stride_l_k
        right_ptrs = right_ptr + pid_bh * stride_r_batch \
                     + off_n[:, None] * stride_r_m \
                     + k_offset[None, :] * stride_r_k

        # Load
        left = tl.load(left_ptrs, mask=m_valid[:, None] & k_valid[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=n_valid[:, None] & k_valid[None, :], other=0.0)

        # Compute dot
        # left shape (BLOCK_M, BLOCK_K), right shape (BLOCK_N, BLOCK_K)
        # we need left @ right^T => use trans_b=True
        acc += tl.dot(left, right, trans_b=True)

    # Write out
    out_ptrs = out_ptr + pid_bh * stride_o_batch \
               + off_m[:, None] * stride_o_m \
               + off_n[None, :] * stride_o_n

    tl.store(out_ptrs, acc.to(tl.float16), mask=m_valid[:, None] & n_valid[None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr,  # *float16
    right_ptr, # *float16
    out_ptr,   # *float16
    B_H: tl.int32,
    N: tl.int32,
    stride_l_batch: tl.int32,
    stride_l_m: tl.int32,
    stride_l_k: tl.int32,
    stride_r_batch: tl.int32,
    stride_r_m: tl.int32,
    stride_r_k: tl.int32,
    stride_o_batch: tl.int32,
    stride_o_m: tl.int32,
    stride_o_n: tl.int32,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets for current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
                    pid_bh * stride_l_batch + \
                    offs_m[:, None] * stride_l_m + \
                    offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # Load right tile: shape (BLOCK_N, BLOCK_K)
        right_ptrs = right_ptr + \
                     pid_bh * stride_r_batch + \
                     offs_n[:, None] * stride_r_m + \
                     offs_k[None, :] * stride_r_k
        right = tl.load(right_ptrs,
                        mask=mask_n[:, None] & mask_k[None, :],
                        other=0.0)

        # Accumulate dot product: left (M,K) * right (N,K)^T
        acc += tl.dot(left, right, trans_b=True)

    # Write result
    out_ptrs = out_ptr + \
               pid_bh * stride_o_batch + \
               offs_m[:, None] * stride_o_m + \
               offs_n[None, :] * stride_o_n
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

    The heavy O(N³·H) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is performed by a custom batched GEMM kernel written in Triton.
    All other linear / normalization steps use PyTorch for simplicity.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 2. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 3. Cast to fp16 and compute linear projections & gates
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # Linear weights (no bias)
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left  = F.linear(x_h, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_h, right_proj_w)  # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4. Apply mask (if any) and gates – plain PyTorch (cheap)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B,N,N) -> (B,N,N,1) broadcast
        mask = mask.to(device).to(torch.float16).unsqueeze(-1)
        left = left * mask
        right = right * mask

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton kernel
    # ------------------------------------------------------------------
    # Rearrange tensors to (B, hidden_dim, N, N) and flatten batch*hidden
    B, N_, _, H = left.shape
    assert N_ == N, "sequence length mismatch"
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch and hidden dimensions
    BH = B * H
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    # Allocate output tensor
    out_flat = torch.empty_like(left_flat)  # same dtype (fp16)

    # Strides (int)
    s_l_batch = left_flat.stride(0)
    s_l_m = left_flat.stride(1)
    s_l_k = left_flat.stride(2)

    s_r_batch = right_flat.stride(0)
    s_r_m = right_flat.stride(1)
    s_r_k = right_flat.stride(2)

    s_o_batch = out_flat.stride(0)
    s_o_m = out_flat.stride(1)
    s_o_n = out_flat.stride(2)

    # Choose tile sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (BH,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        N,
        s_l_batch, s_l_m, s_l_k,
        s_r_batch, s_r_m, s_r_k,
        s_o_batch, s_o_m, s_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, H, N, N) then permute to (B, N, N, H)
    out_perm = out_flat.view(B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------
    # 7. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
