def batch_matmul_kernel(A_ptr, B_ptr, C_ptr,
                        BATCH, M, N, K,
                        stride_batch_a, stride_m_a, stride_k_a,
                        stride_batch_b, stride_n_b, stride_k_b,
                        stride_batch_c, stride_m_c, stride_n_c,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # make mask for out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute pointer base for A: (batch_id, m, k)
    a_ptrs = A_ptr + batch_id * stride_batch_a + offs_m[:, None] * stride_m_a + tl.arange(0, BLOCK_K)[None, :] * stride_k_a

    # B: (batch_id, n, k) – using B's transposed layout (k as inner dimension).
    # We'll load b as shape (BLOCK_K, BLOCK_N) to multiply with A (M,K) dot (N,K) where dot uses k dimension.
    b_ptrs = B_ptr + batch_id * stride_batch_b + offs_n[None, :] * stride_n_b + tl.arange(0, BLOCK_K)[:, None] * stride_k_b

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load blocks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k*BLOCK_K + tl.arange(0, BLOCK_K) < K), other=0.0)

        acc += tl.dot(a, b)

        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_k_a
        b_ptrs += BLOCK_K * stride_k_b

    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the 4‑D input (float32 for stability) and cast to fp16.
2. Linear projections for left/right tensors and three sigmoid gates.
3. Optional mask multiplication (broadcast over the hidden dimension).
4. Element‑wise gating of left/right tensors.
5. Batched matrix multiplication:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a fused Triton kernel that processes all
   (batch * hidden_dim) matrices in parallel.
6. Second LayerNorm over the hidden dimension, multiplication by the output gate,
   and final linear projection back to the original dimension.
7. Cast the result to float32 (the model’s default dtype) and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_n_b, stride_k_b,
    stride_batch_c, stride_m_c, stride_n_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute many independent (M×K)·(K×N) products.
    Each program ID (batch, block_m, block_n) produces a BLOCK_M×BLOCK_N tile.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current tile.
    a_ptrs = a_ptr + batch_id * stride_batch_a + \
             offs_m[:, None] * stride_m_a + tl.arange(0, BLOCK_K)[None, :] * stride_k_a
    b_ptrs = b_ptr + batch_id * stride_batch_b + \
             offs_n[None, :] * stride_n_b + tl.arange(0, BLOCK_K)[:, None] * stride_k_b

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a_mask = mask_m[:, None] & mask_k[None, :]
        b_mask = mask_k[:, None] & mask_n[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_N), accumulate in fp32

        a_ptrs += BLOCK_K * stride_k_a
        b_ptrs += BLOCK_K * stride_k_b

    # Store the tile.
    c_ptrs = c_ptr + batch_id * stride_batch_c + \
             offs_m[:, None] * stride_m_c + offs_n[None, :] * stride_n_c
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim" (int) and "hidden_dim" (int), optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32) and cast to fp16 for compute.
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias) and gates.
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)          # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)        # (B,N,N,hidden_dim)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim) and gating.
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp
        right = right * mask_fp

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4. Core contraction via Triton batched matmul.
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    total_batch = B * H
    left_mat = left_perm.view(total_batch, N, N)
    right_mat = right_perm.view(total_batch, N, N)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Strides (in elements)
    stride_batch_a = left_mat.stride(0)
    stride_m_a = left_mat.stride(1)
    stride_k_a = left_mat.stride(2)

    stride_batch_b = right_mat.stride(0)
    stride_n_b = right_mat.stride(1)
    stride_k_b = right_mat.stride(2)

    stride_batch_c = out_mat.stride(0)
    stride_m_c = out_mat.stride(1)
    stride_n_c = out_mat.stride(2)

    # Tile sizes – chosen to fit well in shared memory.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batch_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        total_batch, N, N, N,
        stride_batch_a, stride_m_a, stride_k_a,
        stride_batch_b, stride_n_b, stride_k_b,
        stride_batch_c, stride_m_c, stride_n_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5. Output LayerNorm, output gate, final linear projection.
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (half‑precision)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # (B,N,N,dim) in fp16

    # -----------------------------------------------------------------
    # 6. Cast back to float32 and return.
    # -----------------------------------------------------------------
    out = out.to(torch.float32) if out.dtype != torch.float32 else out
    return out
