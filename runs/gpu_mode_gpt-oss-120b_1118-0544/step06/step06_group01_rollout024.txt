```
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr, total_elems, hidden_dim,

    BLOCK_SIZE: tl.constexpr,

    right = tl.load(right_ptr + offs, mask=mask, other=0.0)

        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val

        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

```
@triton.jit
def tri_mul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)  # block i
    pid_j = tl.program_id(1)  # block j
    pid_batch = tl.program_id(2)  # batch * hidden

    # compute batch and head indices
    b = pid_batch // H
    h = pid_batch % H

    # compute offsets for i and j
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute mask for i/j within N
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile (M,K)
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k + \
            h * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile transposed (K,N) -> shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            offs_j[None, :] * stride_right_j + \
            offs_k[:, None] * stride_right_k + \
            h * stride_right_h
        # right shape (K,N)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product
        acc += tl.dot(left, right)  # left (M,K), right (K,N) -> (M,N)

    # store result
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j + \
        h * stride_out_h
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
Outgoing Triangle Multiplicative Update (TriMul) – forward pass.

Algorithm
---------
1. Layer‑normalize the input (float32 for stability) and cast to fp16.
2. Apply linear projections (left/right) and gate linear layers.
3. Fuse element‑wise mask (if present) and gate multiplications with a
   Triton kernel to reduce memory traffic.
4. Compute the core contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batched GEMM using a custom Triton kernel (no extra copies).
5. Apply output LayerNorm, the out‑gate, and a final linear projection back
   to the original feature dimension.
All heavy arithmetic is performed in fp16, accumulating in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # [B,N,N,H] flattened
    right_ptr,              # same
    left_gate_ptr,          # same
    right_gate_ptr,         # same
    mask_ptr,               # [B,N,N] flattened (broadcasted over H)
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,   # compile‑time bool
    BLOCK_SIZE: tl.constexpr, # compile‑time int
):
    """In‑place fusion of left/right * gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left      = tl.load(left_ptr      + offs, mask=active, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=active, other=0.0)

    if HAS_MASK:
        # each H entries share the same mask element
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


@triton.jit
def _tri_mul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]."""
    pid_i = tl.program_id(0)   # i‑block
    pid_j = tl.program_id(1)   # j‑block
    pid_h = tl.program_id(2)   # batch*head index

    # decode batch and head
    b = pid_h // H
    h = pid_h % H

    # offsets for i, j and k
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (M, K)
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k + \
            h * stride_left_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile transposed: (K, N)
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            offs_j[None, :] * stride_right_j + \
            offs_k[:, None] * stride_right_k + \
            h * stride_right_h
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # store the result
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j + \
        h * stride_out_h
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, dim)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, dim) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    use_mask = (mask is not None) and (not nomask)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for compute‑heavy parts
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Load model weights (fp16 where appropriate)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gates (all fp16)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_w)                 # (B,N,N,H)
    right = F.linear(x, right_proj_w)                # (B,N,N,H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,H)

    # -----------------------------------------------------------------
    # 4. Fuse mask and gate multiplications (Triton)
    # -----------------------------------------------------------------
    if use_mask:
        mask_fp = mask.to(device).to(torch.float16)
        total = left.numel()
        BLOCK_SIZE = 1024

        grid = (triton.cdiv(total, BLOCK_SIZE),)

        _gate_mask_fuse_kernel[grid](
            left,
            right,
            left_gate,
            right_gate,
            mask_fp,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction – batched GEMM via Triton
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)   # (B,N,N,H), fp16

    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),      # i‑blocks
        triton.cdiv(N, BLOCK_N),      # j‑blocks
        B * H                         # batch * hidden_dim
    )

    _tri_mul_contract_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # Cast back to the original dtype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
