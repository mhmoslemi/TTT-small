@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bc, stride_cd,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_row = tl.program_id(0) # block row
    pid_col = tl.program_id(1) # block col
    pid_batch = tl.program_id(2) # batch index

    # compute offsets for this block
    a_batch_ptr = a_ptr + pid_batch * stride_ab
    b_batch_ptr = b_ptr + pid_batch * stride_bc
    c_batch_ptr = c_ptr + pid_batch * stride_cd

    # Compute start of tiles
    offs_m = pid_row * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_col * BLOCK_N + tl.arange(0, BLOCK_N)
    # for K loop
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        # load a tile
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        # accumulate
        acc += tl.dot(a, b)
    # after loop, store
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])
```

    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bc, stride_cd,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,

    pid_n = tl.program_id(1)
    pid_b = tl.program_id(2)

    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bc, stride_cd,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_n = tl.program_id(1)
    pid_b = tl.program_id(2)

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    b_batch = b_ptr + pid_b * stride_bc

def custom_kernel(data):
   input_tensor, mask, weights, config = data
   dim = config["dim"]
   hidden_dim = config["hidden_dim"]
   nomask = config.get("nomask", False)

   # LayerNorm
   x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
   x = x.to(torch.float16)

   # Cast weights
   left_proj_w = weights["left_proj.weight"].to(torch.float16)
   right_proj_w = weights["right_proj.weight"].to(torch.float16)
   left_gate_w = weights["left_gate.weight"].to(torch.float16)
   right_gate_w = weights["right_gate.weight"].to(torch.float16)
   out_gate_w = weights["out_gate.weight"].to(torch.float16)
   to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
   to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
   to_out_w = weights["to_out.weight"].to(torch.float16)

   # Projections
   left_proj = F.linear(x, left_proj_w)  # shape (B,N,N,H)
   right_proj = F.linear(x, right_proj_w)

   # Gates
   left_gate = F.linear(x, left_gate_w).sigmoid()
   right_gate = F.linear(x, right_gate_w).sigmoid()
   out_gate = F.linear(x, out_gate_w).sigmoid()

   # Apply mask & gating
   if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # shape (B,N,N,1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
   else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask_fp16
            right = right * mask_fp16

   # permute to (B, H, N, N)
   left = left.permute(0, 3, 1, 2).contiguous()
   right = right.permute(0, 3, 1, 2).contiguous()

   # flatten batch and hidden
   B, H, N, _ = left.shape
   batch_h = B * H
   left_flat = left.view(batch_h, N, N)
   right_flat = right.view(batch_h, N, N)

   # Compute core reduction using Triton
   out_flat = _batched_matmul_triton(left_flat, right_flat)

   # reshape back
   out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

   # Output LayerNorm
   out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
   out = out * out_gate  # broadcasting ok, out_gate shape [B,N,N,H]
   out = F.linear(out, to_out_w)

   # Cast back to original dtype
   out = out.to(input_tensor.dtype)
   return out

"""
TriMul (outgoing) forward pass accelerated with Triton.

Algorithm
---------
1. Layer‑norm the 4‑D input over the channel dimension.
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (left/right) and gating (left/right/out) – all fp16.
4. Apply optional mask and element‑wise gating.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden slice h:
       out_h = left_h @ right_hᵀ
   Implemented with a custom Triton kernel that fuses the K‑loop and
   accumulates in fp32.
6. Output layer‑norm over the hidden dimension, multiply by the out‑gate,
   and final linear projection back to the original dimension.
7. Cast the result back to the input dtype.

The Triton kernel processes a batch*hidden slice as a separate
matrix‑multiplication task, using three program‑id dimensions
(block‑row, block‑col, batch*hidden).  Blocking sizes are tunable.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_am, stride_ak,              # strides for A (row, col)
    stride_bk, stride_bn,              # strides for B (row, col) – B is provided transposed
    stride_cm, stride_cn,              # strides for C (row, col)
    stride_ab, stride_bc, stride_cd,   # strides for batch dimension of A, B, C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_b = tl.program_id(2)  # batch*hidden index

    # offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bc
    c_batch = c_ptr + pid_b * stride_cd

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # main loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # mask for loading A and B
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # compute pointers
        a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        # load tiles
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)

    # store the result
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def _batched_matmul_triton(
    left: torch.Tensor,
    right: torch.Tensor,
    BLOCK_M: int = 64,
    BLOCK_N: int = 64,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    left, right: (B*H, N, N) fp16 tensors.
    Returns: (B*H, N, N) fp16 tensor = left @ rightᵀ
    """
    batch_h, N, _ = left.shape
    M = N
    K = N

    # allocate output
    out = torch.empty_like(left)

    # strides (in elements, not bytes)
    stride_am = left.stride(1)
    stride_ak = left.stride(2)
    stride_ab = left.stride(0)

    # right is used transposed: rows correspond to K, cols to N
    stride_bc = right.stride(0)
    stride_bk = right.stride(2)  # original column stride -> row stride of transposed
    stride_bn = right.stride(1)  # original row stride -> col stride of transposed

    stride_cm = out.stride(1)
    stride_cn = out.stride(2)
    stride_cd = out.stride(0)

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_h,
    )

    _batched_matmul_kernel[grid](
        left,
        right,
        out,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ab,
        stride_bc,
        stride_cd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3).
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32 or float16)
            - mask: torch.Tensor [B, N, N] (0/1) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to fp16 (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)      # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)    # (B, N, N, H)

    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask & gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask exists but user wants to ignore it – still apply for completeness
            mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask_fp16
            right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction via Triton batched GEMM
    #    left and right are (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    # ------------------------------------------------------------------
    # bring hidden dimension to second axis
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left.shape
    batch_h = B * H
    left_flat = left.view(batch_h, N, N)
    right_flat = right.view(batch_h, N, N)

    # Triton GEMM where right is implicitly transposed
    out_flat = _batched_matmul_triton(left_flat, right_flat)

    # restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to the original dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
