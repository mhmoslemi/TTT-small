@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, batch_H,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0) # batch_h index
    pid_m = tl.program_id(1) # tile row
    pid_n = tl.program_id(2) # tile col

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for m and n
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Compute base pointers for the given batch index
    base_l = left_ptr + pid_batch * stride_l_batch
    base_r = right_ptr + pid_batch * stride_r_batch
    base_o = out_ptr + pid_batch * stride_o_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left block (M,K)
        l_ptrs = base_l + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(l_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right block (N,K)
        r_ptrs = base_r + offs_n[:, None] * stride_r_i + offs_k[None, :] * stride_r_k
        # right shape (N,K)
        right = tl.load(r_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # accumulate dot product with reduction over K
        acc += tl.dot(left, right)

    # store result
    out_ptrs = base_o + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nominal_mask = config.get("nomask", False)

    # Layernorm:
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # using torch.nn.functional.layer_norm (float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to float16 for performance
    x = x.to(torch.float16)

    # Load weight tensors
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # Linear projections
    # Use F.linear which does (x @ weight.T) + bias (bias None)
    left = F.linear(x, left_proj_w)  # shape (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply mask and gates
    # mask shape (B,N,N) -> unsqueeze(-1) for broadcast to hidden_dim
    if mask is not None and not nominal_mask:
        # Using Triton kernel fuse
        # Flatten as before
        # Use mask cast to float16
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_fp = mask.to(device).to(torch.float16).contiguous().view(-1)

        total_elems = left_flat.numel()
        BLOCK_SIZE = 1024
        grid = lambda meta: (triton.cdiv(total_elems, meta['BLOCK_SIZE']),)
        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_fp,
            total_elems,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # after this, left and right updated in-place
    else:
        # broadcast mask absence
        left = left * left_gate
        right = right * right_gate

    # If mask is no mask but we still need to multiply with gates (they might have been computed)
    if mask is None or nominal_mask:
        # we already did left=left*left_gate, right=right*right_gate
        pass

    # Core contraction using Triton
    B, N, _, H = left.shape
    # flatten batch & head dims
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_mat = torch.empty_like(left_mat)  # dtype float16

    # get strides
    stride_l = left_mat.stride()
    stride_r = right_mat.stride()
    stride_o = out_mat.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32  # may choose 32

    tri_mul_kernel[grid](

        right_mat,
        out_mat,
        N,
        batch_h,

        stride_r[0], stride_r[1], stride_r[2],
        stride_o[0], stride_o[1], stride_o[2],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape out_mat to (B,N,N,H)

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Apply LayerNorm over the input feature dimension (float32 for stability).
2. Cast to fp16 and compute linear projections (left/right) and gates.
3. Fuse mask (if any) with the gated projections using simple element‑wise ops.
4. Compute the core contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   via a tiled Triton kernel that performs many independent matrix
   multiplications (one per batch * hidden dimension) in fp32 accumulation
   and stores fp16 results.
5. Apply output LayerNorm, the output gate and a final linear projection
   back to the original feature dimension.
All heavy arithmetic is done in fp16, accumulation in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # const *fp16  (B*H, N, N)
    right_ptr,         # const *fp16  (B*H, N, N)
    out_ptr,           # *fp16        (B*H, N, N)
    N,                 # int32 sequence length
    stride_l_batch,    # int32 stride for batch dimension of left
    stride_l_i,        # int32 stride for i dimension of left
    stride_l_k,        # int32 stride for k dimension of left
    stride_r_batch,    # int32 stride for batch dimension of right
    stride_r_i,        # int32 stride for i dimension of right (j in formula)
    stride_r_k,        # int32 stride for k dimension of right
    stride_o_batch,    # int32 stride for batch dimension of out
    stride_o_i,        # int32 stride for i dimension of out
    stride_o_j,        # int32 stride for j dimension of out
    BLOCK_M: tl.constexpr,  # tile size for output rows (i)
    BLOCK_N: tl.constexpr,  # tile size for output cols (j)
    BLOCK_K: tl.constexpr,  # tile size for reduction dimension (k)
):
    pid_batch = tl.program_id(0)   # batch * hidden index
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile col

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this batch element
    left_batch_ptr = left_ptr + pid_batch * stride_l_batch
    right_batch_ptr = right_ptr + pid_batch * stride_r_batch
    out_batch_ptr = out_ptr + pid_batch * stride_o_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M x K)
        l_ptrs = left_batch_ptr + (offs_m[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k)
        mask_l = mask_m[:, None] & mask_k[None, :]
        left = tl.load(l_ptrs, mask=mask_l, other=0.0).to(tl.float32)

        # Load right tile (N x K)  (note: we need right[j,k])
        r_ptrs = right_batch_ptr + (offs_n[:, None] * stride_r_i) + (offs_k[None, :] * stride_r_k)
        mask_r = mask_n[:, None] & mask_k[None, :]
        right = tl.load(r_ptrs, mask=mask_r, other=0.0).to(tl.float32)

        # Dot product over K -> (M,N)
        acc += tl.dot(left, right)

    # Store the result in fp16
    out_ptrs = out_batch_ptr + (offs_m[:, None] * stride_o_i) + (offs_n[None, :] * stride_o_j)
    mask_o = mask_m[:, None] & mask_n[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_o)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for compute‑heavy parts
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Load linear weights (fp16)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections and gates
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4. Apply mask and gate element‑wise (simple torch ops)
    # ------------------------------------------------------------------
    if (mask is not None) and (not nomask):
        # mask: (B, N, N) -> (B, N, N, 1) for broadcasting
        mask_ = mask.to(device).to(torch.float16).unsqueeze(-1)
        left = left * left_gate * mask_
        right = right * right_gate * mask_
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: batched GEMM using Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) with contiguous memory layout
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_mat = torch.empty_like(left_mat)  # fp16 output

    # strides for the three tensors (batch, i, k)
    s_l_batch, s_l_i, s_l_k = left_mat.stride()
    s_r_batch, s_r_i, s_r_k = right_mat.stride()
    s_o_batch, s_o_i, s_o_j = out_mat.stride()

    # Tiling parameters (tuned for typical N up to 1024)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                                 # batch * hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,          # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,          # tiles over j
    )
    _tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        s_l_batch, s_l_i, s_l_k,
        s_r_batch, s_r_i, s_r_k,
        s_o_batch, s_o_i, s_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # Cast back to the original dtype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
