@triton.jit
def triton_trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_mat = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    B_int = B
    N_int = N
    H_int = H

    batch = pid_mat // H_int
    hidden = pid_mat % H_int

    # compute start indices
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N_int
    mask_j = offs_j < N_int

    # allocate accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loops over k dimension
    num_k_blocks = tl.cdiv(N_int, BLOCK_K)
    for k in range(num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N_int

        # load left tile: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_ptr
            + batch * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + hidden * stride_left_h,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        ).to(tl.float32)

        # load right tile transposed: shape (BLOCK_K, BLOCK_N)
        # we want right_T[k, j] = right[batch, j, k, hidden]
        right_T = tl.load(
            right_ptr
            + batch * stride_right_b
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i
            + hidden * stride_right_h,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        ).to(tl.float32)

        # dot: (M,K) x (K,N) -> (M,N)
        acc += tl.dot(left, right_T)

    # store result as fp16 (or same dtype as left/right)
    out = acc.to(tl.float16)

    tl.store(
        out_ptr
        + batch * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + hidden * stride_out_h,
        out,
        mask=mask_i[:, None] & mask_j[None, :]
    )

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input (float32 for stability).
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (left/right) and three sigmoid gates (left, right, out).
4. Fuse mask (if present) with left/right gates using a tiny Triton kernel.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM (one GEMM per (batch, hidden) pair) with Triton.
6. Apply a second LayerNorm on the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original dimension.
7. Cast back to the input dtype and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# 1. Kernel: fuse mask and gate for left / right tensors (in‑place)
# --------------------------------------------------------------------------- #
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,            # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,  # same shape
    mask_ptr,                       # [B,N,N] flattened
    total_elems, hidden_dim,        # runtime scalars
    HAS_MASK: tl.constexpr,         # compile‑time bool
    BLOCK_SIZE: tl.constexpr,       # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each H‑wide slice shares the same mask entry
        mask_idx = offs // hidden_dim          # index in [B,N,N]
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # store in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


# --------------------------------------------------------------------------- #
# 2. Kernel: batched GEMM for the outgoing TriMul contraction
#    Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# --------------------------------------------------------------------------- #
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_mat = tl.program_id(0)   # batch * hidden index
    pid_m   = tl.program_id(1)   # tile row
    pid_n   = tl.program_id(2)   # tile col

    # decode batch and hidden dimension
    batch = pid_mat // H
    hidden = pid_mat % H

    # compute i and j offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (M, K)
        left = tl.load(
            left_ptr
            + batch * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + hidden * stride_left_h,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile transposed: (K, N)
        # we need right_T[k, j] = right[batch, j, k, hidden]
        right_T = tl.load(
            right_ptr
            + batch * stride_right_b
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i
            + hidden * stride_right_h,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # dot product accumulates into fp32
        acc += tl.dot(left, right_T)

    # store result as fp16
    out = acc.to(tl.float16)
    tl.store(
        out_ptr
        + batch * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + hidden * stride_out_h,
        out,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# --------------------------------------------------------------------------- #
# 3. Entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------- #
    # 1. Input LayerNorm (float32 for stability)
    # -------------------------------------------------- #
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim] fp32

    # -------------------------------------------------- #
    # 2. Cast to fp16 for the heavy path
    # -------------------------------------------------- #
    x = x.to(torch.float16)

    # -------------------------------------------------- #
    # 3. Linear projections (bias=False) – fp16
    # -------------------------------------------------- #
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)          # (B,N,N,hidden_dim) fp16
    right = F.linear(x, right_proj_w)        # (B,N,N,hidden_dim) fp16

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # fp16
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # fp16
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # fp16

    # -------------------------------------------------- #
    # 4. Fuse mask and element‑wise gates (if a mask is present)
    # -------------------------------------------------- #
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten everything for the kernel
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left  = left_flat.view_as(left)
        right = right_flat.view_as(right)
        # left_gate / right_gate no longer needed
        del left_gate, right_gate
    else:
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # -------------------------------------------------- #
    # 5. Core contraction via Triton GEMM
    # -------------------------------------------------- #
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # (B,N,N,H) fp16, contiguous

    # strides
    sL_b, sL_i, sL_k, sL_h = left.stride()
    sR_b, sR_i, sR_k, sR_h = right.stride()
    sO_b, sO_i, sO_j, sO_h = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        sL_b,
        sL_i,
        sL_k,
        sL_h,
        sR_b,
        sR_i,
        sR_k,
        sR_h,
        sO_b,
        sO_i,
        sO_j,
        sO_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------- #
    # 6. Output LayerNorm, gate, and final linear projection
    # -------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # (B,N,N,dim) fp16

    # -------------------------------------------------- #
    # 7. Cast back to original dtype
    # -------------------------------------------------- #
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
