@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_bi, stride_bk,  # left stride: batch, i (row), k (col)
    stride_br, stride_bk2, stride_bn, # right stride: batch, k (row), n (col) - after transposition
    stride_ob, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0) # each grid in first dim is a batch instance (b * H)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)  # i-axis
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)  # j-axis

    # mask for bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load A
        a_ptrs = left_ptr + batch_idx * stride_bl + offs_i[:, None] * stride_bi + k_offset[None, :] * stride_bk
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load B (right transposed)
        b_ptrs = right_ptr + batch_idx * stride_br + k_offset[:, None] * stride_bk2 + offs_j[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # write output
    c_ptrs = out_ptr + batch_idx * stride_ob + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program id
    batch_idx = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute start offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)

"""
TriMul (outgoing) forward pass implemented with Triton kernels.

Algorithm
---------
1. Layer‑normalize the input (float32 for stability).
2. Cast to fp16 and apply linear projections (left/right) and gate
   linear layers (left_gate, right_gate, out_gate).
3. Fuse mask (if present) and gate multiplications into a single Triton
   kernel to reduce memory traffic.
4. Compute the core contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batched GEMM where each (b,d) slice performs a matrix multiply
   of size (N,N) @ (N,N)ᵀ.  This is realized with a custom Triton kernel
   that tiles the i‑j plane and reduces over k.
5. Apply output LayerNorm, the out‑gate, and a final linear projection
   back to the original feature dimension.
All heavy arithmetic is performed in fp16 with FP32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,                # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,      # same shape as left/right
    mask_ptr,                           # [B,N,N] flattened (broadcast over H)
    total_elems, hidden_dim,            # runtime scalars
    HAS_MASK: tl.constexpr,             # compile‑time bool
    BLOCK_SIZE: tl.constexpr,           # compile‑time int
):
    """In‑place fusion of left/right with their gates and optional mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left      = tl.load(left_ptr      + offs, mask=active, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden_dim entries share the same mask element
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


@triton.jit
def _trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_row, stride_right_col,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication:
    out[b,i,j] = Σ_k left[b,i,k] * right[b,k,j]   (right is accessed transposed)
    """
    batch_idx = tl.program_id(0)          # 0 … batch‑1
    pid_i = tl.program_id(1)              # block row
    pid_j = tl.program_id(2)              # block col

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (left)
        a_ptrs = left_ptr \
                 + batch_idx * stride_left_batch \
                 + offs_i[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (right transposed)
        b_ptrs = right_ptr \
                 + batch_idx * stride_right_batch \
                 + offs_k[:, None] * stride_right_row \
                 + offs_j[None, :] * stride_right_col
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Write back
    c_ptrs = out_ptr \
             + batch_idx * stride_out_batch \
             + offs_i[:, None] * stride_out_i \
             + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2) Cast to FP16 for heavy compute
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Load linear weights (FP16)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections and gates
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # --------------------------------------------------------------
    # 5) Fuse mask and gate multiplication (Triton)
    # --------------------------------------------------------------
    use_mask = (mask is not None) and (not nomask)
    if use_mask:
        # Prepare mask as fp16
        mask_fp = mask.to(device).to(torch.float16)

        # Flatten tensors for kernel
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK_SIZE = 1024

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # Reshape back (views are already updated)
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # Simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 6) Core contraction: batched GEMM via Triton
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Rearrange to (B, H, N, N) for convenient flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch = B * H
    left_flat = left_perm.view(batch, N, N)   # (batch, N, N)
    right_flat = right_perm.view(batch, N, N) # (batch, N, N)

    out_flat = torch.empty_like(left_flat)

    # Strides for left (row i, col k)
    stride_left_batch = left_flat.stride(0)
    stride_left_i = left_flat.stride(1)
    stride_left_k = left_flat.stride(2)

    # Strides for right (we need it as [k, j] i.e. transposed)
    stride_right_batch = right_flat.stride(0)
    stride_right_row = right_flat.stride(2)   # original k stride (row after transpose)
    stride_right_col = right_flat.stride(1)   # original j stride (col after transpose)

    # Strides for output
    stride_out_batch = out_flat.stride(0)
    stride_out_i = out_flat.stride(1)
    stride_out_j = out_flat.stride(2)

    # Choose block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _trimul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch, N,
        stride_left_batch, stride_left_i, stride_left_k,
        stride_right_batch, stride_right_row, stride_right_col,
        stride_out_batch, stride_out_i, stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
