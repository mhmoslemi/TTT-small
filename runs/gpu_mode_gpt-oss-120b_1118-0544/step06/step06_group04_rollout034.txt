"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑normalize the input (float32) and cast to fp16.
2. Compute linear projections and sigmoid gates (no bias) with torch.nn.functional.linear.
3. Fuse optional mask with the left/right gates using pure PyTorch (tiny cost compared to O(N³)).
4. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (left @ rightᵀ) using a custom Triton kernel that
   tiles the i/j dimensions (BLOCK_M×BLOCK_N) and iterates over k in blocks (BLOCK_K).
   Accumulation is performed in fp32 for stability, then stored as fp16.
5. Apply output LayerNorm, multiply by the output gate, and a final linear projection
   back to the original channel dimension.
6. Cast the result back to the input dtype.

The kernel handles arbitrary batch size (≤2) and hidden dimensions (≤768) and works
with or without a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the O(N³·H) contraction (batched GEMM: left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,          # fp16, [B, N, N, H] flattened
    right_ptr,         # fp16, same shape
    out_ptr,           # fp16, same shape (output)
    # strides for left tensor
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right tensor
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # strides for out tensor
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    B, N, H,           # runtime scalars
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_i = tl.program_id(0)          # tile along i dimension
    pid_j = tl.program_id(1)          # tile along j dimension
    pid_bh = tl.program_id(2)         # combined batch * hidden channel

    # compute start indices of the tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    bh = pid_bh
    b = bh // H                       # batch index
    h = bh % H                        # hidden channel index

    # offsets within the tile
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds i / j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in blocks
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers to left[b, i, k, h] and right[b, j, k, h]
        a_ptrs = left_ptr \
            + b * stride_left_b + h * stride_left_h \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        b_ptrs = right_ptr \
            + b * stride_right_b + h * stride_right_h \
            + offs_j[:, None] * stride_right_i \
            + offs_k[None, :] * stride_right_k

        # load tiles, applying masks
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # accumulate dot product (A @ Bᵀ)
        acc += tl.dot(a, b)

    # store the result tile
    c_ptrs = out_ptr \
        + b * stride_out_b + h * stride_out_h \
        + i_start * stride_out_i + j_start * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

    left = F.linear(x, left_proj_w)            # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)          # (B,N,N,hidden_dim)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))    # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,hidden_dim)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))      # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3. Apply optional mask and gates (pure PyTorch – cheap compared to O(N³))
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # broadcast mask over hidden dimension
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp * left_gate
        right = right * mask_fp * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # Ensure contiguous layout for Triton
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 4. Core contraction via Triton batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    out = torch.empty_like(left)  # (B,N,N,hidden_dim)

    # strides for left / right / out (B,N,N,H) layout
    s_left = left.stride()   # (stride_b, stride_i, stride_k, stride_h)
    s_right = right.stride()
    s_out = out.stride()

    # block sizes – chosen to fit typical GPU shared memory / registers
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M,
          (N + BLOCK_N - 1) // BLOCK_N,
          B * hidden_dim )
    )

    _trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        stride_left_b=s_left[0], stride_left_i=s_left[1],
        stride_left_k=s_left[2], stride_left_h=s_left[3],
        stride_right_b=s_right[0], stride_right_i=s_right[1],
        stride_right_k=s_right[2], stride_right_h=s_right[3],
        stride_out_b=s_out[0], stride_out_i=s_out[1],
        stride_out_j=s_out[2], stride_out_h=s_out[3],
        B=B, N=N, H=hidden_dim,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------
    # 6. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
