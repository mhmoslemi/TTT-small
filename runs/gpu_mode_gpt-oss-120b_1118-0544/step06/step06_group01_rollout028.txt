@triton.jit
def tri_mul_matmul_kernel(
    out_ptr, left_ptr, right_ptr,
    batch, M, N, K,
    stride_out_batch, stride_out_m, stride_out_n,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_n, stride_r_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    batch_idx = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_rem = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_i = pid_rem // tl.cdiv(N, BLOCK_N)
    block_j = pid_rem % tl.cdiv(N, BLOCK_N)

    # Compute start positions
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # Mask for out of bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for out
    out_ptrs = out_ptr + batch_idx * stride_out_batch + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32) # accumulate in fp32

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load left tile (M x K)
        left_ptrs = left_ptr + batch_idx * stride_l_batch + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        # Load right tile (N x K) because we need right^T -> (N x K?) Actually right is [batch,N,N], we want its transpose on last two dims: we need right[b, j, k] (where j is row, k is column). For multiplication left[i,k] * right[j,k] we need right tile arranged as (N,K) with same k dimension as column. So we can load right transposed: pointer offset: stride_r_batch + offs_n[:, None] * stride_r_n + offs_k[None, :] * stride_r_k.
        right_ptrs = right_ptr + batch_idx * stride_r_batch + offs_n[:, None] * stride_r_n + offs_k[None, :] * stride_r_k
        right = tl.load(right_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # Outer product accumulate
        acc += tl.dot(left, right)  # shape (BLOCK_M, BLOCK_N)
    # Write back result
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the PyTorch reference but off‑loads the
heavy N³ contraction to a custom Triton kernel.  The steps are:

1. Layer‑normalize the input (float32 for stability).
2. Cast to fp16 and apply the left/right projections and gate
   linear layers.
3. Apply the optional mask and gate multiplications (pure PyTorch,
   inexpensive).
4. Compute the core contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batched GEMM (left @ rightᵀ) using a tiled Triton kernel.
5. Output LayerNorm, out‑gate, and final linear projection back to the
   original feature dimension.
All heavy arithmetic is in fp16 with fp32 accumulation, and the final
tensor is returned in the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    out_ptr, a_ptr, b_ptr,               # pointers
    BATCH, M, N, K,                      # dimensions
    stride_out_batch, stride_out_m, stride_out_n,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM computing out = a @ bᵀ."""
    pid = tl.program_id(0)

    # Determine which batch element and which tile of (M,N)
    batch_idx = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    rem = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_i = rem // tl.cdiv(N, BLOCK_N)
    block_j = rem % tl.cdiv(N, BLOCK_N)

    # Offsets within the tile
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer to the output tile
    out_ptrs = (
        out_ptr
        + batch_idx * stride_out_batch
        + offs_m[:, None] * stride_out_m
        + offs_n[None, :] * stride_out_n
    )

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M × K)
        a_ptrs = (
            a_ptr
            + batch_idx * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile transposed (K × N)
        # Original B is (batch, N, K); we need its transpose.
        b_ptrs = (
            b_ptr
            + batch_idx * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Store the result in fp16
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )

            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for the heavy part
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 2. Load linear weights (fp16)
    # -------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)

    # -------------------------------------------------
    # 3. Linear projections and gates
    # -------------------------------------------------
    left = F.linear(x, left_proj_w)   # (B, N, N, hidden)
    right = F.linear(x, right_proj_w)  # (B, N, N, hidden)

    # -------------------------------------------------
    # 4. Optional mask
    # -------------------------------------------------
    if (mask is not None) and (not nomask):
        mask_exp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # -------------------------------------------------
    # 5. Gate values
    # -------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6. Core contraction via Triton (batched GEMM)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -------------------------------------------------
    B, N, _, _ = left.shape  # left shape is (B, N, N, hidden)

    # Bring hidden dimension into the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, hidden, N, N)

    batch = B * hidden_dim
    left_mat = left_perm.view(batch, N, N)   # (batch, M, K)
    right_mat = right_perm.view(batch, N, N)  # (batch, N, K)

    out_mat = torch.empty_like(left_mat)  # (batch, N, N), fp16

    # Strides (in elements)
    stride_out_batch = out_mat.stride(0)
    stride_out_m = out_mat.stride(1)
    stride_out_n = out_mat.stride(2)

    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_n = right_mat.stride(1)   # row stride (N dimension)
    stride_b_k = right_mat.stride(2)   # column stride (K dimension)

    # Kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = lambda meta: (
        batch
        * ((N + meta["BLOCK_M"] - 1) // meta["BLOCK_M"])
        * ((N + meta["BLOCK_N"] - 1) // meta["BLOCK_N"]),
    )

    _tri_mul_gemm_kernel[grid](
        out_mat.data_ptr(),
        left_mat.data_ptr(),
        right_mat.data_ptr(),
        batch,
        N,
        N,
        N,
        stride_out_batch,
        stride_out_m,
        stride_out_n,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 7. Output LayerNorm, out‑gate and final projection
    # -------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # Cast back to the original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
