"""
custom_kernel: Outgoing Triangle Multiplicative Update (TriMul) accelerated with Triton.

Algorithm
---------
1. Layer‑norm the input (float32 for stability).
2. Cast to fp16 for compute‑heavy path.
3. Linear projections (left, right) and three sigmoid gates (left, right, out).
4. Fuse mask (if present) and element‑wise gates for *left* and *right* with a
   tiny Triton kernel that works in‑place on the fp16 tensors.
5. Compute the core contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batched matrix multiplication L @ Rᵀ per hidden channel using
   torch.bmm (cuBLAS tensor‑cores).
6. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate, and finally project back to the original dimension.
7. Cast the result back to the original dtype (float32) and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,                     # fp16 pointer, shape (B,N,N,H) flattened
    right_ptr,                    # fp16 pointer, same shape
    left_gate_ptr,                # fp16 pointer, same shape
    right_gate_ptr,               # fp16 pointer, same shape
    mask_ptr,                     # fp16 pointer, shape (B,N,N) flattened
    total_elem, hidden_dim,       # runtime scalars
    HAS_MASK: tl.constexpr,       # compile‑time bool
    BLOCK_SIZE: tl.constexpr,     # compile‑time int
):
    """In‑place fusion:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
       mask is broadcast over the hidden dimension H.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elem

    # Load values (masked load returns 0.0 for out‑of‑range)
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each H‑wide slice shares the same mask entry
        mask_idx = offs // hidden_dim                # index in [B,N,N]
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    # Store the fused results back in‑place
    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """

            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for the compute‑heavy part
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections + gate logits (all fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left      = F.linear(x, left_proj_w)          # (B,N,N,H)
    right     = F.linear(x, right_proj_w)         # (B,N,N,H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,H)

    # -----------------------------------------------------------------
    # 4. Fuse mask (if any) and the two element‑wise gates
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # Convert mask to fp16 and make it contiguous
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # Flatten tensors for the kernel (they must be contiguous)
        left_flat       = left.contiguous().view(-1)
        right_flat      = right.contiguous().view(-1)
        left_gate_flat  = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat       = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total_elems = left_flat.numel()
        BLOCK_SIZE = 8192                       # reasonable block size
        grid = lambda meta: (triton.cdiv(total_elems, meta['BLOCK_SIZE']),)

        _mask_gate_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total_elems,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # Reshape back to original tensors
        left  = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # Simple element‑wise fusion when no mask is needed
        left  = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM L @ Rᵀ per hidden channel
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, H, N, N) -> (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (cuBLAS tensor‑cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate                     # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)            # (B,N,N,dim) in fp16

    # -----------------------------------------------------------------
    # 7. Cast back to the original dtype
    # -----------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
