    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # config

    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for compute
    x = x.to(torch.float16)

    # Load linear weights
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)

    # Linear projections
    left = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Mask and gate fusion
    use_mask = (mask is not None) and (not nomask)
    if use_mask:
        mask_fp = mask.to(device).to(torch.float16)
        # flatten
        left_flat = left.reshape(-1)
        right_flat = right.reshape(-1)
        left_gate_flat = left_gate.reshape(-1)
        right_gate_flat = right_gate.reshape(-1)

            BLOCK_SIZE=BLOCK_SIZE,

        # left and right are already updated in-place

    stride_a_k = left_mat.stride(2)

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm Overview
------------------
1. **Layer‑norm** the 4‑D input (float32 for stability).
2. **Linear projections** (`left_proj`, `right_proj`) and **gates**
   (`left_gate`, `right_gate`, `out_gate`) computed in fp16.
3. **Fuse mask & gate** multiplication with a tiny Triton kernel
   (in‑place on ``left`` and ``right``) to cut memory traffic.
4. **Core contraction**:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM ``left @ rightᵀ`` over the sequence dimension.
   Implemented with a custom Triton kernel that processes the batch
   dimension (B·hidden) and tiles the N×N matrices.
5. **Output LayerNorm**, multiply by ``out_gate`` and a final linear
   projection back to the original feature dimension.
All heavy arithmetic is performed in fp16 with fp32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    ``mask`` is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left      = tl.load(left_ptr      + offs, mask=active, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=active, other=0.0)

    if HAS_MASK:
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


@triton.jit
def _tri_mul_kernel(
    out_ptr, a_ptr, b_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication for the outgoing TriMul:
        out[b] = a[b] @ b[b].T   (a: (M,K), b: (N,K) -> out: (M,N))
    ``a`` has shape (batch, M, K) and ``b`` has shape (batch, K, N)
    (i.e. ``b`` is already transposed w.r.t. the original ``right``).
    """
    pid_b = tl.program_id(0)    # batch index
    pid_m = tl.program_id(1)    # block row
    pid_n = tl.program_id(2)    # block col

    # Block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for A (a) and B (b)
        a_ptrs = a_ptr + pid_b * stride_a_batch + \
                 offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_b * stride_b_batch + \
                 offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Cast to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = out_ptr + pid_b * stride_o_batch + \
             offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) (or None)
            - weights: dict of model parameters (torch.Tensors)
            - config: dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as ``input_tensor``.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numeric stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for compute‑heavy parts
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gates
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 4. Fuse mask and gate multiplication (in‑place)
    # -----------------------------------------------------------------
    use_mask = (mask is not None) and (not nomask)
    if use_mask:
        mask_fp = mask.to(device).to(torch.float16)

        # Flatten tensors for the kernel
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)

        total = left_flat.numel()
        BLOCK_SIZE = 1024

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # ``left`` and ``right`` have been updated in‑place.
    else:
        # No mask: simple element‑wise gate multiplication
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ rightᵀ)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape          # left: (B, N, N, H)
    batch_h = B * H

    # (B, N, N, H) -> (B, H, N, N) and make contiguous
    left_mat = left.permute(0, 3, 1, 2).contiguous()
    # Right needs to be transposed on the last two dims: (j, k) -> (k, j)
    right_mat_T = right.permute(0, 3, 2, 1).contiguous()

    # Collapse batch and hidden dimensions for the kernel
    left_mat = left_mat.view(batch_h, N, N)
    right_mat_T = right_mat_T.view(batch_h, N, N)

    # Output placeholder
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Strides
    stride_a_batch = left_mat.stride(0)
    stride_a_m     = left_mat.stride(1)
    stride_a_k     = left_mat.stride(2)

    stride_b_batch = right_mat_T.stride(0)
    stride_b_k     = right_mat_T.stride(1)   # K dimension
    stride_b_n     = right_mat_T.stride(2)   # N dimension

    stride_o_batch = out_mat.stride(0)
    stride_o_m     = out_mat.stride(1)
    stride_o_n     = out_mat.stride(2)

    # Tiling parameters (tuned for typical B200 memory hierarchy)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (batch_h, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    _tri_mul_kernel[grid](
        out_mat,
        left_mat,
        right_mat_T,
        N, N, N,               # M, N, K
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_o_batch, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore the original 4‑D layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the original dtype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
