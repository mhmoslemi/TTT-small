"""
TriMul (outgoing) forward pass – Triton‑fused mask & gating
================================================================
The module implements the outgoing Triangle Multiplicative Update used in
AlphaFold‑3.  The heavy part of the computation is the contraction

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which is a batch of `hidden_dim` matrix multiplications of size
(N, N).  We perform the following steps

1.  Layer‑norm the input (float32) → fp16.
2.  Four linear projections (left/right + their gates) in fp16.
3.  Fuse projection·gate·mask with a tiny Triton kernel
    (avoids two extra memory passes).
4.  Reshape and permute to obtain tensors of shape
    (B*hidden_dim, N, N) and run a batched GEMM
    (`torch.bmm`) which maps to cuBLAS/Tensor‑Core kernels.
5.  Restore the original layout, apply a second LayerNorm,
    multiply by the output gate and project back to the original
    channel dimension.
All heavy element‑wise work (step 3) is performed in a Triton kernel,
satisfying the requirement to use Triton.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) on a flattened view
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (total_proj,)
    gate_ptr,          # fp16 * (total_proj,)
    mask_ptr,          # fp16 * (total_mask,)
    out_ptr,           # fp16 * (total_proj,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – hidden dimension (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # mask is broadcasted over the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 4096) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using a Triton kernel.
    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],          # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (or None)
        - weights      : dict of model parameters (float32)
        - config       : dict containing:
            * "dim"          – input channel dimension C
            * "hidden_dim"   – hidden dimension H
            * "nomask"       – if True, ignore the mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------- unpack ------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = x.device
    dtype_in = x.dtype

    # -------------------------- LayerNorm (input) -----------------------
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for the compute‑intensive part
    x = x.to(torch.float16)

    # -------------------------- cast weights ----------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------- linear projections -----------------------
    # [B, N, N, hidden_dim]
    left_proj = torch.nn.functional.linear(x, left_proj_w)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ----------------------------- gates -------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # [B,N,N,hidden_dim]

    # ------------------------ mask & gate fusion -----------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------- core O(N³·H) reduction via batched GEMM ------
    # Bring hidden_dim to the batch dimension:
    # left_perm / right_perm : [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose the last two dims of right to match the contraction pattern
    right_perm_t = right_perm.transpose(2, 3).contiguous()

    # Merge batch and hidden_dim dimensions -> (B*H, N, N)
    B = x.shape[0]
    left_mat = left_perm.view(B * hidden_dim, x.shape[1], x.shape[2])
    right_mat = right_perm_t.view(B * hidden_dim, x.shape[1], x.shape[2])

    # Batched matrix multiplication: (i,k) @ (k,j) -> (i,j) for each slice
    out_mat = torch.bmm(left_mat, right_mat)

    # Restore shape [B, hidden_dim, N, N] -> [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, x.shape[1], x.shape[2])
    out = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]

    # -------------------------- second LayerNorm -----------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ---------------------- final linear projection --------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim]

    # Cast back to original dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
