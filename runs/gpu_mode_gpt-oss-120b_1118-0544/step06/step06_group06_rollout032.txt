def tri_mul_batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bk, stride_bj,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)  # batch index
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute base pointers for this batch
    a_batch_ptr = a_ptr + pid * stride_ah
    b_batch_ptr = b_ptr + pid * stride_bh
    c_batch_ptr = c_ptr + pid * stride_ch

    # compute block offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load a block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load b block: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj)

        # accumulate
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_batch_ptr + (offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor (float32 → fp16 for the heavy part).
2. Linear projections (left/right) from C → H (hidden_dim) using fp16 weights.
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. Fuse mask (if present) and left/right gates in a Triton kernel:
       left_out  = left  * mask * left_gate
       right_out = right * mask * right_gate
5. Core reduction: batched matrix multiplication over the sequence dimension.
   For each hidden channel we compute:
       out[b,i,j,h] = Σ_k left_out[b,i,k,h] * right_out[b,j,k,h]
   Implemented as a custom Triton GEMM with fp16 inputs and fp32 accumulation.
6. Second LayerNorm over the hidden dimension (fp32 for stability).
7. Multiply by the output gate and project back to the original channel
   dimension C with a final linear layer.
8. Cast the result back to the original dtype of the input tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # const * left   (B,N,N,H) flattened
    right_ptr,         # const * right  (B,N,N,H) flattened
    mask_ptr,          # const * mask   (B,N,N)   flattened
    left_gate_ptr,     # const * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # const * right_gate (B,N,N,H) flattened
    left_out_ptr,      # * left_out   (B,N,N,H) flattened
    right_out_ptr,     # * right_out  (B,N,N,H) flattened
    H: tl.constexpr,              # hidden dimension
    total_elements: tl.constexpr, # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask and gates:
       left_out  = left  * mask * left_gate
       right_out = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # mask is stored per (b,i,k); broadcast over hidden dim
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    tl.store(left_out_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,

    stride_ch: tl.int32, stride_ci: tl.int32, stride_cj: tl.int32,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ B^T
    A: (M, N, K) where K == N
    B: (M, K, N) (already transposed)
    C: (M, N, N)
    Grid: (M, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    """
    pid = tl.program_id(0)      # batch index (M)
    pid_i = tl.program_id(1)    # block row index
    pid_j = tl.program_id(2)    # block col index

    # Pointers to the current batch
    a_batch = a_ptr + pid * stride_ah
    b_batch = b_ptr + pid * stride_bh
    c_batch = c_ptr + pid * stride_ch

    # Block offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B block: (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_batch + (offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, if True ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float32)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(torch.float16)  # cast to fp16 for the heavy part

    # ------------------------------------------------------------------
    # 2) Load linear weights in fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float32)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w) # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Prepare mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_tensor = mask.to(torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6) Fuse mask and gates using Triton
    # ------------------------------------------------------------------
    total_elements = left.numel()  # B * N * N * H
    BLOCK = 1024

    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = (triton.cdiv(total_elements, BLOCK),)
    _mask_gate_fusion_kernel[grid](
        left,
        right,
        mask_tensor.view(-1),
        left_gate,
        right_gate,
        left_out,
        right_out,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Batched matrix multiplication (core TriMul reduction)
    #    left_out: (B, N, N, H) -> (B, H, N, N) -> view as (M, N, K)
    #    right_out transposed: (B, H, N, N) where axes are (k, j)
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N)
    left_perm = left_out.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) -> (i,k)
    right_perm = right_out.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) -> (k,j)

    M = B * hidden_dim
    left_mat = left_perm.view(M, N, N)   # (M, N, K)
    right_mat = right_perm.view(M, N, N) # (M, K, N)

    out_mat = torch.empty_like(left_mat)  # (M, N, N)

    # Strides (int64)
    stride_ah = left_mat.stride(0)
    stride_ai = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_bh = right_mat.stride(0)
    stride_bk = right_mat.stride(1)
    stride_bj = right_mat.stride(2)

    stride_ch = out_mat.stride(0)
    stride_ci = out_mat.stride(1)
    stride_cj = out_mat.stride(2)

    # Triton block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        M,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        N=N,
        stride_ah=stride_ah,
        stride_ai=stride_ai,
        stride_ak=stride_ak,
        stride_bh=stride_bh,
        stride_bk=stride_bk,
        stride_bj=stride_bj,
        stride_ch=stride_ch,
        stride_ci=stride_ci,
        stride_cj=stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Second LayerNorm over hidden dimension (fp32 for stability)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 9) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10) Final projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 11) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
