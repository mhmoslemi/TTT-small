"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) using PyTorch.
2. Cast to fp16 for the compute‑heavy portion.
3. Compute four linear projections (left/right and their gates) in fp16.
4. Fuse element‑wise mask, gate and projection with a small Triton kernel.
   This saves two extra memory passes compared to naive PyTorch.
5. Core O(N³·H) reduction:
   - Reshape left/right to (B*H, N, N) and use a single torch.bmm
     (batched GEMM) which maps to cuBLAS/Tensor‑core kernels.
6. Apply a second LayerNorm on the hidden dimension, multiply by the
   output gate, and linearly project back to the original channel dimension.
7. Cast the result back to the original dtype (float32).

Only the mask‑gate fusion is performed in Triton; the rest leverages
highly‑optimized PyTorch/cuBLAS kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (total_elems,)
    gate_ptr,          # fp16 * (total_elems,)
    mask_ptr,          # fp16 * (mask_elems,)   (B*N*N)
    out_ptr,           # fp16 * (total_elems,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    """Element‑wise: out = proj * gate * mask   (mask broadcast over H)."""
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    # mask index = element // hidden_dim (broadcast)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 2048,
) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using a Triton kernel.

    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys:
            * "dim" (int)         – input channel dimension C
            * "hidden_dim" (int)  – hidden dimension H
            * "nomask" (bool)     – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (B,N,N,H) – fp16
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6) Fuse mask & gate (Triton) if needed
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is contiguous fp16
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core O(N³·H) reduction via batched GEMM (cuBLAS)
    #    Reshape to (B*H, N, N) and compute left @ right^T
    # ------------------------------------------------------------------
    # left/right: (B, N, N, H) -> (B, H, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_t.shape

    left_flat = left_t.view(B * H, N, N)
    right_flat = right_t.view(B * H, N, N)

    # batched matrix multiplication
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Second LayerNorm on hidden dimension, apply output gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype (float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
