def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)

    # cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N, H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Reshape for batched matmul
    B, N, _, H = left.shape
    # (B, H, N, N)
    left_resh = left.permute(0, 3, 1, 2).contiguous()
    right_resh = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_resh.view(-1, N, N)   # (B*H, N, N)
    right_flat = right_resh.view(-1, N, N) # (B*H, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Launch batched matmul kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    batch = left_flat.shape[0]
    M = N
    K = N
    N_dim = N

    grid = (batch, triton.cdiv(M, BLOCK_M), triton.cdiv(N_dim, BLOCK_N))

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch,
        M,
        N_dim,
        K,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Second LayerNorm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply out_gate
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to original dtype
    out = out.to(input_tensor.dtype)

    return out

"""
TriMul (outgoing) forward pass – Triton‑fused mask‑gate and batched GEMM.

Algorithm
---------
1. Layer‑normalize the 4‑D input (B,N,N,C).
2. Cast to fp16 for the heavy compute part.
3. Four linear projections (left/right) and three gating projections are
   performed with fp16 weights.
4. Fuse `proj * gate * mask` (when a mask is present) with a tiny Triton
   kernel – saves two extra memory passes.
5. Core O(N³·H) reduction:
   - reshape tensors to (B·H, N, N) and run a custom batched GEMM kernel
     that computes `out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]`.
   - the kernel is a tiled matmul with float32 accumulation for accuracy.
6. Second LayerNorm, output‑gate scaling and final linear projection back to
   the original channel dimension.
7. Cast the result back to the input dtype.

The implementation keeps all heavy arithmetic in fp16 (except the
accumulators) and uses Triton for the two fusion points to reduce
memory traffic and kernel launch overhead.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) on a flattened view
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask broadcasted over the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,

    mask: torch.Tensor,

    """
    Fuse `proj * gate * mask` using a Triton kernel.
    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the einsum
#   out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # fp16 pointers
    batch, M, N, K,                           # dimensions (int)
    stride_ab, stride_am, stride_ak,          # strides for A (batch, M, K)
    stride_bb, stride_bn, stride_bk,          # strides for B (batch, N, K)
    stride_cb, stride_cm, stride_cn,          # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)   # batch dimension
    pid_m = tl.program_id(1)   # tile along M
    pid_n = tl.program_id(2)   # tile along N

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in blocks
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )

        # B tile: shape (BLOCK_K, BLOCK_N)   (notice K is the second axis)
        b_ptrs = (
            b_ptr
            + pid_b * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # write back result in fp16
    c = acc.to(tl.float16)
    c_ptrs = (
        c_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, c, mask=mask)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, C]
        - mask        : torch.Tensor of shape [B, N, N] (or None)
        - weights     : dict of model parameters (float32)
        - config      : dict with keys:
            * "dim" (int)         – input channel dimension C
            * "hidden_dim" (int)  – hidden dimension H
            * "nomask" (bool)     – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------- LayerNorm (input) -------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------- cast weights ---------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ----------------------- linear projections -------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------- gates ----------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------- mask & gate fusion (Triton) --------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ----------------- core O(N³·H) reduction via custom batched GEMM ----------
    B, N, _, H = left.shape
    # reshape to (B, H, N, N) → (B*H, N, N)
    left_resh = left.permute(0, 3, 1, 2).contiguous()
    right_resh = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_resh.view(-1, N, N)    # (B*H, N, N)
    right_flat = right_resh.view(-1, N, N)  # (B*H, N, N)

    out_flat = torch.empty_like(left_flat)

    # launch the batched GEMM kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    batch = left_flat.shape[0]
    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch,
        N,              # M
        N,              # N
        N,              # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------ cast back to input dtype ----------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
