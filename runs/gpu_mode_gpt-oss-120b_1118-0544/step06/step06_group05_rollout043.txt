```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # get tile indices
    pid = tl.program_id(0)
    batch_f = tl.program_id(1)

    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_tiles_j
    pid_n = pid % num_tiles_j

    # Compute starting indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Compute b and h
    b = batch_f // H
    h = batch_f % H

    # Base pointers
    left_base = left_ptr + b * stride_left_batch + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base = out_ptr + b * stride_out_batch + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        K_start = k * BLOCK_K
        offs_k = K_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + (offs_m[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load right block transposed: we need right[k, j]
        # right base index: b * ... + h*... + (offs_n[:, None] * stride_right_i) + (offs_k[None, :] * stride_right_k)
        # To get right[j,k] we need to load B correctly: we want B[k,j] for dot. So we load b_T = right_base + (offs_k[:, None] * stride_right_k) + (offs_n[None, :] * stride_right_i)
        b_ptrs = right_base + (offs_k[:, None] * stride_right_k) + (offs_n[None, :] * stride_right_i)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)  # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)

    # Write result
    out_ptrs = out_base + (offs_m[:, None] * stride_out_i) + (offs_n[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
High‑performance outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for compute‑heavy parts.
3. Linear projections:
      left_proj  = x @ W_left
      right_proj = x @ W_right
   Gate projections (sigmoid) and element‑wise gating.
4. Optional mask multiplication (broadcasted over the hidden dimension).
5. Core reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication (`left @ rightᵀ`) using a
   custom Triton kernel. The hidden dimension `hidden_dim` is folded into the
   batch axis, so a single kernel launch processes B·hidden_dim independent
   matrix multiplications.
6. Output layer‑norm, multiply by `out_gate`, final linear projection back to
   the original `dim`.
7. Cast the result back to the input dtype (usually float32).

The Triton kernel follows the persistent‑kernel matmul pattern from the
reference implementation, with an extra batch dimension for the hidden
channels.  All heavy arithmetic stays in FP16 with FP32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    left_ptr,                     # fp16 * (B, N, N, H)
    right_ptr,                    # fp16 * (B, N, N, H)
    out_ptr,                      # fp16 * (B, N, N, H)
    B, N, H,                      # int64 scalars
    # strides for left tensor
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    # strides for right tensor
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    # strides for output tensor
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute `out = left @ rightᵀ` for each (batch, hidden) pair.
    left/ right/out have shape (B, N, N, H) with layout (batch, i, k, h).
    """
    pid = tl.program_id(0)          # tile index within a matrix
    batch_f = tl.program_id(1)      # combined batch·hidden index

    # --------------------------------------------------------------
    # Derive tile coordinates (i‑tile, j‑tile)
    # --------------------------------------------------------------
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # Row offsets (split into two halves for better register reuse)
    offs_m_1 = pid_m * BLOCK_M + tl.arange(0, BLOCK_M // 2)
    offs_m_2 = pid_m * BLOCK_M + tl.arange(BLOCK_M // 2, BLOCK_M)
    # Column offsets
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # K‑dimension offsets
    offs_k = tl.arange(0, BLOCK_K)

    # --------------------------------------------------------------
    # Decode combined batch index into (batch, hidden)
    # --------------------------------------------------------------
    b = batch_f // H
    h = batch_f % H

    # Base pointers for this (b, h) slice
    left_base = left_ptr + b * stride_left_batch + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base = out_ptr + b * stride_out_batch + h * stride_out_h

    # --------------------------------------------------------------
    # Pointers for the current tile
    # --------------------------------------------------------------
    a_ptrs_1 = left_base + (offs_m_1[:, None] * stride_left_i
                           + offs_k[None, :] * stride_left_k)
    a_ptrs_2 = left_base + (offs_m_2[:, None] * stride_left_i
                           + offs_k[None, :] * stride_left_k)
    # right is accessed as transposed: we need right[k, j, h]
    b_ptrs = right_base + (offs_k[:, None] * stride_right_k
                           + offs_n[None, :] * stride_right_i)

    # --------------------------------------------------------------
    # Accumulators (FP32)
    # --------------------------------------------------------------
    acc_1 = tl.zeros((BLOCK_M // 2, BLOCK_N), dtype=tl.float32)
    acc_2 = tl.zeros((BLOCK_M // 2, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K‑loop
    # --------------------------------------------------------------
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        a_1 = tl.load(a_ptrs_1,
                      mask=(offs_m_1[:, None] < N) & (offs_k[None, :] < N),
                      other=0.0)
        a_2 = tl.load(a_ptrs_2,
                      mask=(offs_m_2[:, None] < N) & (offs_k[None, :] < N),
                      other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < N) & (offs_n[None, :] < N),
                    other=0.0)

        # Multiply‑accumulate
        acc_1 += tl.dot(a_1, b)
        acc_2 += tl.dot(a_2, b)

        # Advance pointers to the next K‑block
        a_ptrs_1 += BLOCK_K * stride_left_k
        a_ptrs_2 += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # --------------------------------------------------------------
    # Write results (cast back to FP16)
    # --------------------------------------------------------------
    c_1 = acc_1.to(tl.float16)
    c_2 = acc_2.to(tl.float16)

    c_ptrs_1 = out_base + (offs_m_1[:, None] * stride_out_i
                           + offs_n[None, :] * stride_out_j)
    c_ptrs_2 = out_base + (offs_m_2[:, None] * stride_out_i
                           + offs_n[None, :] * stride_out_j)

    mask_1 = (offs_m_1[:, None] < N) & (offs_n[None, :] < N)
    mask_2 = (offs_m_2[:, None] < N) & (offs_n[None, :] < N)

    tl.store(c_ptrs_1, c_1, mask=mask_1)
    tl.store(c_ptrs_2, c_2, mask=mask_2)


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None (0/1)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the compute‑intensive part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections and gates (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5. Fuse projection, gating and optional mask
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate
    if mask is not None and not nomask:
        # mask shape: (B, N, N) → broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_fp16
        right = right * mask_fp16

    # Ensure contiguous tensors for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape

    # ------------------------------------------------------------------
    # 6. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (left @ rightᵀ) via Triton.
    # ------------------------------------------------------------------
    out = torch.empty_like(left)

    # Strides (in elements)
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Block sizes – tuned for typical sequence lengths on B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid: (tiles over M×N, batch*hidden)
    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_tiles_m * num_tiles_n, B * H)

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
        stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
        stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
