@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Get program IDs: batch, pid_m, pid_n
 
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2);
    
    # Compute offsets: base pointers for this batch
    a_batch_ptr = a_ptr + batch * batch_stride_a
    b_batch_ptr = b_ptr + batch * batch_stride_b
    c_batch_ptr = c_ptr + batch * batch_stride_c

    # Compute row/col offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to each block
    a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_batch_ptr + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load a and b tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k*BLOCK_K + tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k*BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
        # dot product
        acc += tl.dot(a, b)
        # Increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store result
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

@triton.jit
def tri_mul_gemm_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_am, stride_ak,
                        stride_bk, stride_bn,
                        stride_cm, stride_cn,
                        batch_stride_a, batch_stride_b, batch_stride_c,
                        BLOCK_M: tl.constexpr,
                        BLOCK_N: tl.constexpr,
                        BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute mask
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute base pointers for this batch
    a_ptr = A_ptr + batch_id * batch_stride_a
    b_ptr = B_ptr + batch_id * batch_stride_b
    c_ptr = C_ptr + batch_id * batch_stride_c

    # pointer offsets for A and B
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offsets within K dimension
        k_offset = k * BLOCK_K
        cur_k = tl.arange(0, BLOCK_K) + k_offset
        # mask for K
        mask_k = cur_k < K

        # pointers to A block
        a_offs = offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        a = tl.load(a_ptr + a_offs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # pointers to B block
        b_offs = cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptr + b_offs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot
        acc += tl.dot(a, b).to(tl.float32)

    # write result
    c = acc.to(tl.float16)
    c_offs = offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptr + c_offs, c, mask=mask_m[:, None] & mask_n[None, :])

custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input (float32) → fp16.
2. Linear projections for left/right values and three gates (no bias).
3. Fuse mask (if present) and gate multiplications with a small Triton kernel.
4. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM `left @ rightᵀ` for every hidden channel.
   We implement a custom Triton GEMM that processes all batches
   (batch * hidden_dim) in one launch, using fp16 arithmetic and
   fp32 accumulation.
5. Output layer‑norm over the hidden dimension, apply the output gate,
   and a final linear projection back to the original dimensionality.
6. Cast the result back to the original input dtype (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Fuse mask & gate multiplication (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [total] flattened fp16
    left_gate_ptr, right_gate_ptr,     # same shape
    mask_ptr,                          # [B*N*N] flattened fp16
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    """
    In‑place:
        left  = left  * left_gate  * (mask if present)
        right = right * right_gate * (mask if present)
    `mask` is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # load tensors
    left      = tl.load(left_ptr + offs,      mask=active, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr + offs,mask=active, other=0.0)

    if HAS_MASK:
        # each hidden slice shares the same mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    # store back in‑place
    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


# ----------------------------------------------------------------------
# 2) Batched GEMM kernel: C = A @ Bᵀ   (A: (M,K), B: (K,N))
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # fp16 pointers
    stride_am, stride_ak,                # A strides
    stride_bk, stride_bn,                # B strides (after transpose)
    stride_cm, stride_cn,                # C strides
    batch_stride_a, batch_stride_b, batch_stride_c,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    M: tl.constexpr,                     # = N (rows of A)
    N: tl.constexpr,                     # = N (cols of B)
    K: tl.constexpr,                     # = N (shared dimension)
):
    batch_id = tl.program_id(0)
    pid_m    = tl.program_id(1)
    pid_n    = tl.program_id(2)

    # base pointers for this batch
    a_batch = a_ptr + batch_id * batch_stride_a
    b_batch = b_ptr + batch_id * batch_stride_b
    c_batch = c_ptr + batch_id * batch_stride_c

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # pointers to the beginning of the tile
    a_ptrs = a_batch + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    k_offset = 0
    while k_offset < K:
        cur_k = tl.arange(0, BLOCK_K) + k_offset

        # mask for the K dimension
        mask_k = cur_k < K

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product, accumulate in fp32
        acc += tl.dot(a, b).to(tl.float32)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
        k_offset += BLOCK_K

    # write back
    c = acc.to(tl.float16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 3) Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (float32 or bool) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # Config
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32) → fp16
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑less) – fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left      = F.linear(x, left_proj_w)      # (B,N,N,hidden_dim)
    right     = F.linear(x, right_proj_w)     # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3) Fuse mask & gates (Triton)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten tensors
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left  = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # simple element‑wise fusion without mask
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Core contraction – batched GEMM with Triton
    #    left:  (B,N,N,hidden) → (B, hidden, N, N)
    #    right: (B,N,N,hidden) → (B, hidden, N, N) then transpose K dimension
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # bring hidden dim in front and make contiguous
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    # transpose the last two axes of right for the GEMM
    right_t = right.permute(0, 1, 3, 2).contiguous()   # (B, H, N, N) == (B,H,K,J)

    # flatten batch & hidden into a single batch dimension
    batch_dim = B * hidden_dim
    A = left.view(batch_dim, N, N)
    B_mat = right_t.view(batch_dim, N, N)
    C = torch.empty_like(A)

    # strides (row‑major)
    stride_am, stride_ak = A.stride(1), A.stride(2)   # (K, 1)
    stride_bk, stride_bn = B_mat.stride(1), B_mat.stride(2)   # (N, 1)
    stride_cm, stride_cn = C.stride(1), C.stride(2)   # (N, 1)

    batch_stride_a = A.stride(0)
    batch_stride_b = B_mat.stride(0)
    batch_stride_c = C.stride(0)

    # block sizes – tuned for fp16 Tensor Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _tri_mul_gemm_kernel[grid](
        A,
        B_mat,
        C,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        M=N,
        N=N,
        K=N,
    )

    # restore original layout: (B, N, N, hidden_dim)
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5) Output LayerNorm, output gate and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate   # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------
    # 6) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
