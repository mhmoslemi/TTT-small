"""
TriMul (outgoing) custom kernel
--------------------------------
Implements the forward pass of the Triangle Multiplicative Update (TriMul)
as described in AlphaFold3. The core steps are:

1. Layer‑norm the input (fp32 for stability, then cast to fp16).
2. Fuse the left/right linear projections with their respective gates by
   concatenating the projection and gate weights (single matmul per side).
3. Apply mask (if present) and gates in a Triton kernel (in‑place).
4. Perform the O(N³) reduction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   realized as a batched GEMM (B*D, N, N) @ (B*D, N, N)ᵀ using torch.bmm.
5. Second layer‑norm over the hidden dimension.
6. Multiply by an output gate.
7. Linear projection back to the original channel dimension.
All heavy compute uses fp16; layer‑norms use fp32 parameters. The mask‑gate
fusion is performed in Triton, satisfying the requirement to include a
custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,            # fp16 * left (B,N,N,H) flattened
    right_ptr,           # fp16 * right (B,N,N,H) flattened
    mask_ptr,            # fp16 * mask (B,N,N) flattened
    left_gate_ptr,       # fp16 * left_gate (B,N,N,H) flattened
    right_gate_ptr,      # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,     # hidden dimension
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fuse mask and sigmoid gates:
       left = left * mask * left_gate
       right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load values
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Compute index into mask (mask repeats for each hidden dim)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply mask and gates, store back in‑place
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (tensors on the same device)
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, if True ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to fp16 for the bulk of the computation
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Prepare weights (fp16)
    # ------------------------------------------------------------------
    # Concatenate projection and gate weights to fuse two matmuls per side
    left_proj_gate_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)   # (2*H, C)

    right_proj_gate_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)  # (2*H, C)

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float32)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections + gate pre‑activations (single matmul per side)
    # ------------------------------------------------------------------
    # Result shape: (B,N,N,2*H)
    left_comb = F.linear(x, left_proj_gate_w)   # (B,N,N,2*H)
    right_comb = F.linear(x, right_proj_gate_w) # (B,N,N,2*H)

    # Split into projection and gate parts
    left_proj = left_comb[..., :hidden_dim]          # (B,N,N,H)
    left_gate_pre = left_comb[..., hidden_dim:]      # (B,N,N,H)
    right_proj = right_comb[..., :hidden_dim]        # (B,N,N,H)
    right_gate_pre = right_comb[..., hidden_dim:]    # (B,N,N,H)

    left_gate = torch.sigmoid(left_gate_pre)   # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_pre) # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4) Prepare mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    B, N, _, _ = left_proj.shape
    if mask is not None and not nomask:
        mask_tensor = mask.to(device=device, dtype=torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 5) Fuse mask and gates in‑place via Triton
    # ------------------------------------------------------------------
    total_elements = left_proj.numel()                # B * N * N * H
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    _mask_gate_fusion_kernel[grid](
        left_proj,
        right_proj,
        mask_tensor.view(-1),   # flatten mask (B*N*N)
        left_gate,
        right_gate,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6) Core reduction: batched GEMM (L @ Rᵀ) per (batch, hidden)
    # ------------------------------------------------------------------
    # Reshape to (B*H, N, N) for torch.bmm
    left_flat = left_proj.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right_proj.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Batched matrix multiplication (cuBLAS)
    out_flat = torch.bmm(left_flat, right_flat.transpose(-2, -1))   # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Output gate
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,H)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,C)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
```
