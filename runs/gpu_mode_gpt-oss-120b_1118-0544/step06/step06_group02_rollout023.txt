```
@triton.jit
def _gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, total_elems, hidden_dim, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # mask index is offset // hidden_dim
    mask_idx = offs // hidden_dim
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=mask)
```

"""
TriMul (outgoing) forward pass.

- LayerNorm on input (FP16)
- Linear projections + gates (FP16)
- Optional mask & gate fusion via a small Triton kernel
- Core O(N³·H) reduction realized as a batched matrix multiplication:
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
- Second LayerNorm, output gating and final linear projection.
- Returns float32 tensor matching the reference implementation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused (proj * gate * mask)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # broadcast mask over hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(out_ptr + offs, proj * gate * mval, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 2048) -> torch.Tensor:
    """
    Element‑wise (proj * gate * mask) using a Triton kernel.
    All tensors must be contiguous on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] (or None)
        - weights      : dict of model parameters (float32)
        - config       : dict with keys:
            * "dim" (int)        – input channel dimension C
            * "hidden_dim" (int) – hidden dimension H
            * "nomask" (bool)    – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # --------------------------- unpack ----------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # --------------------------- LayerNorm (input) ----------------------
    # Cast weights to match compute dtype (float16)
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)

    # LayerNorm over the channel dimension `dim`
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # Use FP16 for the heavy part
    x = x.to(torch.float16)

    # --------------------------- Linear projections --------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)  # (hidden_dim, dim)

    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_w)  # (B,N,N,hidden_dim)

    # --------------------------- Gates ---------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)   # (hidden_dim, dim)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)  # (hidden_dim, dim)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)    # (hidden_dim, dim)

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()   # (B,N,N,hidden_dim)
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()     # (B,N,N,hidden_dim)

    # --------------------------- Mask & Gate Fusion --------------------
    if mask is not None and not nomask:
        # Ensure mask is contiguous fp16
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right, right_gate, mask_fp16)
    else:
        left = left * left_gate
        right = right * right_gate

    # --------------------------- Core O(N^3·H) reduction ---------------
    # left: (B,N,N,H)   right: (B,N,N,H)
    # Compute out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]

    # Bring hidden dim to 2nd axis for batched GEMM
    # left_perm: (B, H, N, N) where dim order is (b, d, i, k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # right needs to be (B, H, N, N) where the N dimensions are (k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape

    left_reshaped = left_perm.view(B * H, N, N)   # (B*H, N, N) with axes (i, k)
    right_reshaped = right_perm.view(B * H, N, N)  # (B*H, N, N) with axes (k, j)

    # Batched matrix multiplication: sum over k
    out_reshaped = torch.bmm(left_reshaped, right_reshaped)  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_reshaped.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------- Second LayerNorm ----------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Apply output gate
    out = out * out_gate

    # --------------------------- Final linear -------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # Cast back to original dtype if needed
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
