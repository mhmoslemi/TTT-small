Now kernel:

```
@triton.jit
def _tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    total_heads,
    stride_head, stride_i, stride_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # get program IDs
    pid_h = tl.program_id(2) # head index
    pid_m = tl.program_id(0) # row tile
    pid_n = tl.program_id(1) # col tile

    # offsets
    row_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    col_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # masks
    row_mask = row_offset < N
    col_mask = col_offset < N

    # pointer to head base
    head_offset = pid_h * stride_head

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # left submatrix MxK
        left_ptrs = left_ptr + head_offset + row_offset[:, None] * stride_i + k_offset[None, :] * stride_j
        left = tl.load(left_ptrs, mask=row_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

        # right submatrix KxN (transposed version) -> original right matrix indexed by (k, col)
        right_ptrs = right_ptr + head_offset + k_offset[:, None] * stride_i + col_offset[None, :] * stride_j
        right = tl.load(right_ptrs, mask=k_mask[:, None] & col_mask[None, :], other=0.0).to(tl.float32)

        # matmul
        acc += tl.dot(left, right)  # left (M,K), right (K,N) -> (M,N)
    # write back
    out_ptrs = out_ptr + head_offset + row_offset[:, None] * stride_i + col_offset[None, :] * stride_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=row_mask[:, None] & col_mask[None, :])
```

"""
Triton‑accelerated outgoing TriMul forward pass (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input over the channel dimension (`dim`).
2. Cast to fp16 for compute‑intensive parts.
3. Linear projections (no bias) → `left_proj`, `right_proj`.
   Linear gates → `left_gate`, `right_gate`, `out_gate` (sigmoid).
4. If a mask is present and not ignored, fuse mask, projection and gate
   for both left and right tensors in a single Triton kernel.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication for all
   (batch × hidden) pairs in one Triton kernel.
6. Output layer‑norm over the hidden dimension, multiply by `out_gate`,
   and final linear projection back to the original `dim`.
7. Cast back to the original dtype (usually float32).

The implementation uses two Triton kernels:
* `_proj_gate_mask_kernel` – fuses projection, gate and mask.
* `_tri_mul_matmul_kernel` – computes the batched GEMM for the
  TriMul reduction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Kernel 1: fuse projection, gate and mask for left and right tensors
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B,N,N,H)
    right_proj_ptr,  # fp16 * (B,N,N,H)
    left_gate_ptr,   # fp16 * (B,N,N,H)
    right_gate_ptr,  # fp16 * (B,N,N,H)
    mask_ptr,        # fp16 * (B,N,N)
    left_out_ptr,    # fp16 * (B,N,N,H)
    right_out_ptr,   # fp16 * (B,N,N,H)
    total_elems,     # int64 scalar = B*N*N*H
    hidden_dim,      # int64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # guard
    mask = offs < total_elems

    # index of the mask element (broadcast over H)
    mask_idx = offs // hidden_dim

    left_proj = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 2048,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gate and mask for both left and right tensors.
    All tensors must be fp16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)
    Returns:
        left, right : (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_matmul_kernel(
    left_ptr,   # fp16 * (total_heads, N, N)
    right_ptr,  # fp16 * (total_heads, N, N)
    out_ptr,    # fp16 * (total_heads, N, N)
    N: tl.int64,
    total_heads: tl.int64,
    stride_head: tl.int64,
    stride_i: tl.int64,
    stride_j: tl.int64,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(2)          # head index (0 .. total_heads-1)
    pid_m = tl.program_id(0)          # row tile index
    pid_n = tl.program_id(1)          # col tile index

    # Offsets for the current tile
    row_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    col_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    row_mask = row_offset < N
    col_mask = col_offset < N

    # Base pointer for this head
    head_offset = pid_h * stride_head

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # left sub‑matrix: (M, K)
        left_ptrs = left_ptr + head_offset + row_offset[:, None] * stride_i + k_offset[None, :] * stride_j
        left = tl.load(left_ptrs,
                       mask=row_mask[:, None] & k_mask[None, :],
                       other=0.0).to(tl.float32)

        # right sub‑matrix (transposed for GEMM): (K, N)
        right_ptrs = right_ptr + head_offset + k_offset[:, None] * stride_i + col_offset[None, :] * stride_j
        right = tl.load(right_ptrs,
                        mask=k_mask[:, None] & col_mask[None, :],
                        other=0.0).to(tl.float32)

        acc += tl.dot(left, right)   # (M,K) x (K,N) -> (M,N)

    # Store result
    out_ptrs = out_ptr + head_offset + row_offset[:, None] * stride_i + col_offset[None, :] * stride_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=row_mask[:, None] & col_mask[None, :])


def _tri_mul_core(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the batched matrix multiplication:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    Input tensors must have shape (B, N, N, H) and be fp16.
    Returns tensor of the same shape.
    """
    B, N, _, H = left.shape
    total_heads = B * H

    # Rearrange to (B, H, N, N) → (total_heads, N, N) contiguous
    left_flat = left.permute(0, 3, 1, 2).contiguous()
    right_flat = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_flat.view(total_heads, N, N)
    right_flat = right_flat.view(total_heads, N, N)

    out_flat = torch.empty_like(left_flat)

    stride_head = left_flat.stride(0)
    stride_i = left_flat.stride(1)
    stride_j = left_flat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        total_heads,
    )

    _tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        total_heads,
        stride_head,
        stride_i,
        stride_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None (0/1)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # --------------------------- unpack ---------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ---------------------- 1) input LayerNorm -------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B,N,N,dim)

    # ---------------------- 2) cast to fp16 ----------------------------
    x = x.to(torch.float16)

    # ---------------------- 3) cast weights ----------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ---------------------- 4) linear projections -----------------------
    left_proj  = F.linear(x, left_proj_w)          # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ---------------------- 5) mask & gating fusion --------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # simple element‑wise fusion (mask optional)
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask provided but should be ignored → still broadcast if user wants
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # free unneeded tensors early
    del left_proj, right_proj, left_gate, right_gate

    # ---------------------- 6) core TriMul reduction -------------------
    out = _tri_mul_core(left, right)   # (B,N,N,hidden_dim)

    # free left/right
    del left, right

    # ---------------------- 7) output LayerNorm, out‑gate, linear -----
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ---------------------- 8) cast back to original dtype ------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
