@triton.jit
def gate_mask_fusion_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, total_elems, hidden_dim, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < total_elems
    # compute mask index
    mask_idx = offsets // hidden_dim
    proj = tl.load(proj_ptr + offsets, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offsets, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    out = proj * gate * mask_val
    tl.store(out_ptr + offsets, out, mask=mask)

"""
TriMul (outgoing) forward pass – Triton‑accelerated mask & gate fusion.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim` (float32 for stability).
2. Cast to FP16 for compute‑heavy operations.
3. Linear projections → `left_proj` and `right_proj` (B,N,N,H).
4. Gate projections (sigmoid) → `left_gate`, `right_gate`, `out_gate` (B,N,N,H).
5. Fuse projection, gating and optional mask in a single Triton kernel:
       fused = proj * gate * mask   (mask broadcast over the hidden dim).
   If no mask is present, simply multiply `proj * gate`.
6. Core reduction (batched GEMM):
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented via a batched torch.bmm on FP16 tensors (which uses
   Tensor‑cores).  The right operand is transposed on the `k` dimension
   before the multiplication.
7. Output LayerNorm over the hidden dimension `hidden_dim`, multiply by
   `out_gate`, and a final linear projection back to the original dimension `dim`.
8. Cast the result back to the input dtype (usually float32).

Only the mask‑gate‑fusion is performed with a Triton kernel; all other
operations use highly‑optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,      # fp16 * (B*N*N*H,)
    gate_ptr,      # fp16 * (B*N*N*H,)
    mask_ptr,      # fp16 * (B*N*N,)
    out_ptr,       # fp16 * (B*N*N*H,)
    total_elems,   # i64: total number of elements in proj/gate (B*N*N*H)
    hidden_dim,    # i64: H
    BLOCK_SIZE: tl.constexpr,
):
    """Fused element‑wise: out = proj * gate * mask (mask broadcast over H)."""
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds
    mask = offs < total_elems

    # Index of the mask element (broadcasted over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 65536) -> torch.Tensor:
    """
    Fuse projection, gate and (optional) mask using a Triton kernel.
    All tensors must be contiguous fp16.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)  (binary 0/1, broadcast over H)
    Returns proj * gate * mask   (shape (B, N, N, H)).
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (float32/float16/bool)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim` (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid (still FP16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask + gating (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16, contiguous, and binary (0/1)
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        # No mask → simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Prepare tensors for batched matmul:
    #   left : (B, H, i, k)
    #   right: (B, H, k, j)   (transpose on the k dimension)
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    B, H, N, _ = left.shape
    left_flat = left.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right.view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication (FP16 → FP16)
    out_flat = torch.bmm(left_flat, right_flat)  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm (over hidden_dim), multiply by out_gate,
    #    and final linear projection back to `dim`.
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
