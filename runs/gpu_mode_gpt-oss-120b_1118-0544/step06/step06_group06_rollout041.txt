"""
TriMul (outgoing) – Triton‑accelerated kernel.
Steps
1. LayerNorm on the input (fp32 for stability).
2. Linear projections (left/right) → hidden_dim and three gate projections.
3. Apply optional mask and sigmoid gates (elementwise, torch).
4. Fuse hidden dimension into the batch dimension and run a
   custom Triton batched‑matrix‑multiply kernel that computes
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
5. Reshape back to (B,N,N,hidden_dim), apply a second LayerNorm,
   multiply by the output gate and a final projection to dim.
All heavy arithmetic is in fp16, accumulation in fp32,
final result returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr,                     # left  (B*H, N, N)  fp16
    b_ptr,                     # right (B*H, N, N)  fp16 (already transposed)
    c_ptr,                     # out   (B*H, N, N)  fp16
    M, N, K,                   # matrix sizes (all equal to seq_len)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_b = tl.program_id(2)          # batch * hidden

    # Offsets for the output block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offs = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offs < K

        # ---- Load A block (M x K) ----
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + k_offs[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # ---- Load B block (K x N) ----
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + k_offs[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # ---- Write back ----
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] (binary) or None
            - weights: dict of model parameters
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, if True ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    out_dtype = torch.float32

    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    # ------------------------------------------------------------------
    # Load weights (convert to appropriate dtype)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device=device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device=device, dtype=torch.float32)

    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(inp, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)   # switch to fp16 for the heavy work

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 4) Apply mask (if any) and gates
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_f = mask.to(device=device, dtype=torch.float16).unsqueeze(-1)   # (B,N,N,1)
    else:
        mask_f = torch.ones((B, N, N, 1), dtype=torch.float16, device=device)

    left  = left * mask_f * left_gate
    right = right * mask_f * right_gate

    # ------------------------------------------------------------------
    # 5) Rearrange tensors for batched matmul (B*H, N, N)
    # ------------------------------------------------------------------
    # Move hidden dim to the second axis and make it contiguous
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left.shape
    batch_h = B * H

    left_flat  = left.view(batch_h, N, N)          # (B*H, N, N)

    # Transpose right for the matmul (K x N)
    right_T = right_flat.transpose(-2, -1).contiguous()   # (B*H, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------

    b_batch, b_k, b_n = right_T.stride()

        batch_h                    # batch * hidden
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        N, N, N,                     # M, N, K
        a_batch, a_m, a_k,
        b_batch, b_k, b_n,
        c_batch, c_m, c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
