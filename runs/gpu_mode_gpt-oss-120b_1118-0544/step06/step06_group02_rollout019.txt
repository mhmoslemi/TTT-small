"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) using the provided weight/bias.
2. Cast to fp16 for the compute‑intensive part.
3. Linear projections (left/right) and their gates are computed in fp16.
4. Fuse projection, gate and optional mask with a tiny Triton kernel
   (`_gate_mask_kernel`). This reduces two memory passes.
5. Core O(N³·H) reduction:
   - Permute tensors to (B, H, N, N) and use a single `torch.matmul`
     (batched GEMM) which maps to cuBLAS/Tensor‑core kernels.
6. Apply a second LayerNorm on the hidden dimension, multiply by the
   output gate, and linearly project back to the original channel dimension.
7. Cast the result back to the original input dtype (float32).

Only the element‑wise mask & gate fusion is implemented in Triton; the
large matrix multiplication leverages highly‑optimized cuBLAS kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) on a flattened view
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # each hidden_dim consecutive elements share the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 8192,
) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using a Triton kernel.
    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],   # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return out

    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # First LayerNorm (input)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for the compute‑heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Cast weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    # shape: (B, N, N, hidden_dim)
    left_proj  = torch.nn.functional.linear(x, left_proj_w)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # Fuse mask, projection and gate (Triton)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is contiguous fp16
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·H) reduction via batched GEMM (torch.matmul)
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) – no extra copy, just a view with different strides
    left_perm  = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2)  # (B, H, N, N)

    # Batched matrix multiplication: out = left @ right^T  → (B, H, N, N)
    out = torch.matmul(left_perm, right_perm.transpose(-1, -2))
    # Back to (B, N, N, H)
    out = out.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Second LayerNorm + output gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # Final linear projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # Cast back to original dtype (float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
