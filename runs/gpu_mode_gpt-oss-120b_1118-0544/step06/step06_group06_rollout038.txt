    N: tl.constexpr,

    stride_b_batch, stride_b_k, stride_b_n, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,

    num_pid_m = tl.cdiv(N, BLOCK_M)

    pid_n = block_idx % num_pid_n

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

"""
Optimized outgoing Triangular Multiplicative Update (TriMul) forward pass.
The kernel fuses the O(N^3) reduction using a batched GEMM implemented in
Triton, while the surrounding linear layers and LayerNorms are performed
with PyTorch. Mixed‑precision (FP16) is used for the heavy path and the
final output is returned in FP32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_k, stride_b_n, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of thread blocks that cover the M and N dimensions
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    blocks_per_head = num_pid_m * num_pid_n

    # decode batch / head index from program id
    bh = pid // blocks_per_head               # 0 .. B*H-1
    block_idx = pid % blocks_per_head
    pid_m = block_idx // num_pid_n
    pid_n = block_idx % num_pid_n

    b = bh // H
    h = bh % H

    # base pointers for this (b, h) pair
    a_base = a_ptr + b * stride_a_batch + h * stride_a_h
    b_base = b_ptr + b * stride_b_batch + h * stride_b_h
    c_base = c_ptr + b * stride_c_batch + h * stride_c_h

    # offsets inside the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # pointers to the current tile of A and B (B is read transposed)
    a_ptrs = a_base + offs_m[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
    b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # write back the result
    c_ptrs = c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # Cast to FP16 for the heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float32)

    if mask is not None and not nomask:
        mask_tensor = mask.to(device=device, dtype=torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    left = left * mask_tensor
    right = right * mask_tensor
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous layout for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 4) Core O(N³) reduction using a Triton batched GEMM
    #    out_hidden[b, i, j, h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------

    # Strides
    a_batch, a_i, a_k, a_h = left.stride()
    b_batch, b_i, b_k, b_h = right.stride()
    c_batch, c_i, c_j, c_h = out_hidden.stride()

    BLOCK_N = 64
    BLOCK_K = 64

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B * hidden_dim * num_pid_m * num_pid_n

    batched_tri_mul_kernel[grid](
        left,
        right,
        out_hidden,
        B,
        N,
        hidden_dim,
        a_batch, a_i, a_k, a_h,
        b_batch, b_k, b_i, b_h,   # note: b_k is stride over k, b_i is stride over j (after transposition)

        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,

    # ------------------------------------------------------------------
    # 5) Post‑processing: LayerNorm over hidden dim, output gate, final linear
    # ------------------------------------------------------------------

        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    if orig_dtype != torch.float32:
        output = output.to(orig_dtype)

    return output

"""
Optimized outgoing Triangular Multiplicative Update (TriMul).

We keep the LayerNorms and linear layers in PyTorch (FP32/FP16) and
accelerate the O(N³·H) reduction with a custom Triton kernel that
performs a batched GEMM:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
The kernel iterates over the reduction dimension K in tiles,
uses fp32 accumulation and stores the result in fp16. The final
output is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_k, stride_b_n, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of tiles covering the i and j dimensions
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    blocks_per_head = num_pid_m * num_pid_n

    # decode (batch, head) and tile coordinates from pid
    bh = pid // blocks_per_head                # 0 .. B*H-1
    block_idx = pid % blocks_per_head
    pid_m = block_idx // num_pid_n
    pid_n = block_idx % num_pid_n

    b = bh // H
    h = bh % H

    # base pointers for this (b, h) pair
    a_base = a_ptr + b * stride_a_batch + h * stride_a_h
    b_base = b_ptr + b * stride_b_batch + h * stride_b_h
    c_base = c_ptr + b * stride_c_batch + h * stride_c_h

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the tiles of A (left) and B (right transposed)
    a_ptrs = a_base + offs_m[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
    b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_offset = k_tile * BLOCK_K
        cur_k = k_offset + offs_k
        mask_k = cur_k < N

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)

        # advance to the next K‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # write the result tile
    c_ptrs = c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[:, None])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict containing model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]          # input channel dimension C
    hidden_dim = config["hidden_dim"]  # hidden dimension H
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)   # FP16 for the heavy path

    # ------------------------------------------------------------------
    # 2) Linear projections & gates (FP16)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float32)
    to_out_w      = weights["to_out.weight"].to(device, dtype=torch.float32)

    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 3) Apply mask (if any) and element‑wise gates
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_tensor = mask.to(device=device, dtype=torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)
    mask_tensor = mask_tensor.unsqueeze(-1)   # (B, N, N, 1)

    left = left * mask_tensor
    right = right * mask_tensor
    left = left * left_gate
    right = right * right_gate

    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 4) Core O(N³·H) reduction via Triton batched GEMM
    # ------------------------------------------------------------------
    out_hidden = torch.empty_like(left)   # (B, N, N, H), FP16

    # Strides (as python ints)
    a_batch, a_i, a_k, a_h = left.stride()
    b_batch, b_i, b_k, b_h = right.stride()
    c_batch, c_i, c_j, c_h = out_hidden.stride()

    # Tile sizes – chosen to fit typical GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B * hidden_dim * num_pid_m * num_pid_n
    grid = (total_blocks,)

    batched_tri_mul_kernel[grid](
        left,
        right,
        out_hidden,
        B,
        N,
        hidden_dim,
        a_batch, a_i, a_k, a_h,
        b_batch, b_k, b_i, b_h,   # b_k = stride over k, b_i = stride over j after transposition
        c_batch, c_i, c_j, c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 5) Post‑processing: LayerNorm over hidden dim, output gate, final linear
    # ------------------------------------------------------------------
    out = out_hidden.to(torch.float32)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate.to(torch.float32)

    output = F.linear(out, to_out_w)   # (B, N, N, C), FP32

    # Cast back to the original dtype if it was not FP32
    if orig_dtype != torch.float32:
        output = output.to(orig_dtype)

    return output
