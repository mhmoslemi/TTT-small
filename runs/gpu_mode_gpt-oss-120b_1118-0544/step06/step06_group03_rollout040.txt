```
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_aa_batch, stride_aa_m, stride_aa_k,
    stride_bb_batch, stride_bb_k, stride_bb_n,
    stride_cc_batch, stride_cc_m, stride_cc_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # create mask for M,N bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to block of A and B
    a_ptrs = a_ptr + batch * stride_aa_batch + offs_m[:, None] * stride_aa_m + tl.arange(0, BLOCK_K)[None,:] * stride_aa_k
    b_ptrs = b_ptr + batch * stride_bb_batch + tl.arange(0, BLOCK_K)[:, None] * stride_bb_k + offs_n[None,:] * stride_bb_n

    # initialize acc
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load A sub-block
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K - k*BLOCK_K), other=0.0) # need proper mask
        # load B sub-block
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K) < K - k*BLOCK_K)[:, None] & mask_n[None, :], other=0.0)
        # dot
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_aa_k
        b_ptrs += BLOCK_K * stride_bb_k
    # store result
    c_ptrs = c_ptr + batch * stride_cc_batch + offs_m[:, None] * stride_cc_m + offs_n[None,:] * stride_cc_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

    stride_b_batch, stride_b_k, stride_b_n,

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid m,n
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute pointer to start of this batch
    a_batch_ptr = a_ptr + batch * stride_a_batch
    b_batch_ptr = b_ptr + batch * stride_b_batch

            a_batch_ptr + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k,

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension `dim`.
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections → left/right tensors (B,N,N,H).
4. Gate projections (sigmoid) → left_gate/right_gate/out_gate (B,N,N,H).
5. Fuse projection, gating and optional mask in a single Triton kernel
   (proj * gate * mask).  If `nomask` is True the fusion is a simple
   element‑wise product.
6. Core reduction (batched GEMM):
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented by a custom Triton batched‑matmul kernel that treats the
   combined batch*hidden dimension as the batch axis.
7. Output LayerNorm over the hidden dimension H, multiply by `out_gate`,
   and a final linear projection back to the original dimension `dim`.
8. Cast the result back to the original dtype (usually float32).

The implementation mixes PyTorch (for LayerNorms and linear layers) with
two Triton kernels: one for the projection‑gate‑mask fusion and one for the
batched matrix multiplication that dominates the compute cost.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse projection, gate and mask (proj * gate * mask)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # int: B*N*N*H
    hidden_dim,        # int: H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # bounds
    mask = offs < total_elems

    # index in mask (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 65536) -> torch.Tensor:
    """
    Fuse projection, gate and mask using a Triton kernel.
    All tensors must be fp16, contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)  (binary 0/1, broadcast over H)
    Returns proj * gate * mask  (shape (B,N,N,H)).
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    _proj_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel 2: batched matrix multiplication for the core reduction
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Block offset coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out-of-range rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the current batch tile
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # Accumulator in float32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K sized chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # K indices for this iteration
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        # Load A sub‑block (M x K)
        a = tl.load(
            a_batch + offs_m[:, None] * stride_a_m + k_idx[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B sub‑block (K x N) – note B is logically transposed
        b = tl.load(
            b_batch + k_idx[:, None] * stride_b_k + offs_n[None, :] * stride_b_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Dot product
        acc += tl.dot(a, b)

    # Write result back in fp16
    c = acc.to(tl.float16)
    tl.store(
        c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (float32/float16/bool)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over channel dimension `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy part
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast linear weights to fp16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B,N,N,H)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid (still fp16)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 6. Fuse projection, gating and optional mask
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        # No mask → simple elementwise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # Rearrange to (B, H, N, N) and make contiguous
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch and hidden dimensions
    batch = B * H
    left_flat = left.view(batch, N, N)
    right_flat = right.view(batch, N, N)

    # Allocate output tensor for the batched matmul
    out_flat = torch.empty_like(left_flat)

    # Strides for the contiguous tensors (batch, M, K) and (batch, K, N)
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_m, stride_b_n = right_flat.stride()  # m = row, n = col
    # For the B side we need stride over K (col) and N (row) after transpose
    stride_b_k = stride_b_n          # original column stride (1)
    stride_b_n = stride_b_m          # original row stride (N)

    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # Blocking parameters (tuned for typical A200 size)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        math.ceil(N / BLOCK_M),
        math.ceil(N / BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 9. Cast back to the original dtype
    # --------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
