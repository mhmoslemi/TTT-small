def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # Combine projection weights
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    combined_w = torch.cat([left_proj_w, right_proj_w,
                            left_gate_w, right_gate_w,
                            out_gate_w], dim=0)  # shape (5*hidden_dim, dim)

    proj = torch.nn.functional.linear(x, combined_w)  # shape (B,N,N,5*hidden_dim)

    # reshape
    B, N, _, _ = proj.shape
    proj = proj.view(B, N, N, 5, hidden_dim)  # shape (B,N,N,5,H)

    left = proj[...,0,:]  # (B,N,N,H)
    right = proj[...,1,:]
    left_gate = proj[...,2,:]
    right_gate = proj[...,3,:]
    out_gate = proj[...,4,:]

    # Sigmoid gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Apply gating and mask
    if mask is not None and not nomask:
        mask_fp = mask.to(dtype=torch.float16, device=device).contiguous()
        # flatten all tensors
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)  # (B*N*N,)
        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
        _apply_gates_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask; multiply gates directly
        left = left * left_gate
        right = right * right_gate

    # Core contraction
    B, N, _, H = left.shape
    left_mat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_mat = right.permute(0,3,1,2).contiguous().view(B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()

    # Output layernorm
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Gating with out_gate
    out = out * out_gate

    # Final linear
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)  # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back
    if dtype_in != out.dtype:
        out = out.to(dtype_in)
    return out

    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Steps
    -----
    1. Layer‑norm the [B,N,N,C] input (float32).
    2. Cast to fp16 and compute *all* five linear projections (left, right,
       left_gate, right_gate, out_gate) with a single fused matmul.
    3. Apply sigmoid to the three gates.
    4. Fuse mask (if present) and the left/right gates in a Triton kernel,
       writing the gated tensors back in‑place.
    5. Core contraction: for each batch & hidden dimension compute
       out = left @ rightᵀ   (batched GEMM on half‑precision tensors).
    6. Output layer‑norm, multiply by out_gate and a final linear projection.
    7. Cast back to the original dtype and return.

    hidden_dim = config["hidden_dim"]

    # 1. Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)   # half precision for the heavy path

    # ------------------------------------------------------------------
    # 2. Fused linear projection for left, right, gates (single matmul)
    # ------------------------------------------------------------------
    # Each weight: (hidden_dim, dim) – concatenate along the output axis
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    combined_w = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )   # (5*hidden_dim, dim)

    # Single linear: shape (B, N, N, 5*hidden_dim)
    proj = F.linear(x, combined_w)

    B, N, _, _ = proj.shape
    proj = proj.view(B, N, N, 5, hidden_dim)   # (B,N,N,5,hidden_dim)

    left = proj[..., 0, :]   # (B,N,N,hidden_dim)
    right = proj[..., 1, :]
    left_gate = proj[..., 2, :]
    right_gate = proj[..., 3, :]
    out_gate = proj[..., 4, :]

    # ------------------------------------------------------------------
    # 3. Sigmoid gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 4. Fuse mask and gates (in‑place) using Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # Flatten everything for the kernel
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _apply_gates_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Restore original shapes
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – simple element‑wise fusion
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape      # H == hidden_dim
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, gating and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate   # apply output gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 7. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: in‑place multiplication of left/right tensors
# with their gates and (optionally) a mask.
# --------------------------------------------------------------
@triton.jit
def _apply_gates_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elements

    # Load values
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # each mask entry is shared by `hidden_dim` consecutive elements
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Write the results back in‑place
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)

    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Algorithm
    ---------
    1. Layer‑norm the [B,N,N,C] input (float32 for numerical stability).
    2. Cast to fp16 and compute all five linear projections (left, right,
       left_gate, right_gate, out_gate) with a single fused matmul.
    3. Apply sigmoid to the three gate tensors.
    4. Fuse mask (if present) and the left/right gates in a Triton kernel,
       writing the gated tensors back in‑place.
    5. Core contraction: for each batch & hidden dimension compute
