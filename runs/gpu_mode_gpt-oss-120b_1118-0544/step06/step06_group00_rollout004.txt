"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Input LayerNorm in FP32 (stability).
2. Cast to FP16 for compute‑heavy parts.
3. Fuse the five linear projections into a single GEMM:
      concat = Linear(x, W_concat)  # W_concat = [left, right, left_gate,
                                    #            right_gate, out_gate] concatenated
   Split the result into left, right, left_gate, right_gate, out_gate.
4. Apply sigmoid to the three gate tensors.
5. Fuse mask (if present) and the two element‑wise gates for *left* and *right*
   with a small Triton kernel that updates the tensors in‑place.
6. Core contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication L @ Rᵀ for each hidden channel.
7. Output LayerNorm on the hidden dimension, multiply by out_gate and project
   back to the original dimension.
8. Cast the result back to the input dtype and return.

Only the mask‑gate fusion is performed in Triton; the remaining heavy work
leverages highly‑optimised Torch GEMM (cuBLAS) and batched GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,      # *fp16   [total]
    right_ptr,     # *fp16   [total]
    left_gate_ptr, # *fp16   [total]
    right_gate_ptr,# *fp16   [total]
    mask_ptr,      # *fp16   [mask_total] (optional)
    total_elems,   # i32 scalar: B*N*N*H
    hidden_dim,    # i32 scalar: H
    HAS_MASK: tl.constexpr,   # 0/1 compile‑time
    BLOCK_SIZE: tl.constexpr, # e.g. 1024
):
    """In‑place fusion:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load the tensors
    left      = tl.load(left_ptr + offs,        mask=mask, other=0.0)
    right     = tl.load(right_ptr + offs,       mask=mask, other=0.0)
    left_g    = tl.load(left_gate_ptr + offs,   mask=mask, other=0.0)
    right_g   = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden‑dim slice shares the same mask entry
        mask_idx = offs // hidden_dim                # index in [B,N,N]
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left  = left  * left_g  * mask_val
        right = right * right_g * mask_val
    else:
        left  = left  * left_g
        right = right * right_g

    # Store back in‑place
    tl.store(left_ptr + offs,  left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor (B, N, N, C)
        - mask         : torch.Tensor (B, N, N) or None
        - weights      : dict of torch.Tensors (float32)
        - config       : dict with keys
            * "dim"          : int  (C)
            * "hidden_dim"   : int  (H)
            * "nomask"       : bool (optional, default False)

    Returns
    -------
    torch.Tensor
        Tensor of shape (B, N, N, C) with the same dtype as the input.
    """
    # ------------------------------------------------------------------ #
    # unpack arguments
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------ #
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------ #
    # 2. Cast to FP16 for the compute‑heavy part
    # ------------------------------------------------------------------ #
    x = x.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 3. Fused linear projections (5 heads) => single GEMM
    # ------------------------------------------------------------------ #
    # concatenate weights: shape (dim, hidden_dim * 5)
    w_left   = weights["left_proj.weight"]
    w_right  = weights["right_proj.weight"]
    w_lgate  = weights["left_gate.weight"]
    w_rgate  = weights["right_gate.weight"]
    w_outgate= weights["out_gate.weight"]

    # each weight is (hidden_dim, dim) in PyTorch Linear (out_features, in_features)
    # but torch.nn.functional.linear expects (in_features, out_features)
    # therefore we need to transpose
    w_concat = torch.cat([
        w_left.t(),
        w_right.t(),
        w_lgate.t(),
        w_rgate.t(),
        w_outgate.t()
    ], dim=1).contiguous()                     # (dim, hidden_dim*5)

    w_concat = w_concat.to(device).to(torch.float16)

    # flatten x to (B*N*N, dim)
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                  # (B*N*N, dim)

    # single linear
    proj = F.linear(x_flat, w_concat)         # (B*N*N, hidden_dim*5)

    # reshape and split
    proj = proj.view(B, N, N, 5, hidden_dim)   # (B,N,N,5,H)
    left      = proj[..., 0, :].contiguous()
    right     = proj[..., 1, :].contiguous()
    left_gate = proj[..., 2, :].contiguous()
    right_gate= proj[..., 3, :].contiguous()
    out_gate  = proj[..., 4, :].contiguous()

    # apply sigmoid to gates
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------ #
    # 4. Mask & gate fusion (Triton kernel)
    # ------------------------------------------------------------------ #
    if mask is not None and not nomask:
        # mask shape (B,N,N) -> broadcast over hidden_dim
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten tensors for kernel
        total = left.numel()                     # B*N*N*H
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _mask_gate_fusion_kernel[grid](
            left.reshape(-1),
            right.reshape(-1),
            left_gate.reshape(-1),
            right_gate.reshape(-1),
            mask_fp.reshape(-1),
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # after kernel, left/right already contain the fused values
    else:
        # No mask – simple element‑wise fusion
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    This is a batched GEMM: (B*H, N, N) @ (B*H, N, N)^T
    # ------------------------------------------------------------------ #
    # reshape to (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # restore shape (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 6. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------ #
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate                       # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)  # (dim, hidden_dim)
    # torch.nn.functional.linear expects (in_features, out_features) transpose
    out = F.linear(out, to_out_w)               # (B,N,N,dim)

    # ------------------------------------------------------------------ #
    # 7. Cast back to original dtype
    # ------------------------------------------------------------------ #
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
