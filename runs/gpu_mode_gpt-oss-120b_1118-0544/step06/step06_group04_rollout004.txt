"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input (float32) → cast to float16.
2. Linear projections (no bias) to hidden dimension H:
      left  = x @ W_left          right = x @ W_right
      left_gate  = sigmoid(x @ W_left_gate)
      right_gate = sigmoid(x @ W_right_gate)
      out_gate   = sigmoid(x @ W_out_gate)
3. Fuse mask (if present) and the two gate multiplications with a small Triton
   kernel that works on the flattened tensors and writes the gated values back
   in‑place.
4. Core contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication (L @ Rᵀ) for each hidden channel.
   We reshape to (B*H, N, N) and use torch.bmm (cuBLAS) which is highly
   optimized.
5. Output layer‑norm, element‑wise multiplication with out_gate, and final
   linear projection back to the original dimension.
6. Cast the result to float32 and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,            # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,  # same shape
    mask_ptr,                       # [B,N,N] flattened (optional)
    total_elems: tl.int64,
    hidden_dim: tl.int64,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """
    In‑place fusion:
        left  = left  * left_gate  * (mask if present)
        right = right * right_gate * (mask if present)
    The mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load data
    left      = tl.load(left_ptr + offs,      mask=mask, other=tl.float16(0.0))
    right     = tl.load(right_ptr + offs,     mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate= tl.load(right_gate_ptr + offs,mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # Each hidden slice shares the same mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr + offs,  left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C), dtype float32
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape (B, N, N, C) with dtype float32.
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # -----------------------------------------------------------------
    # Config
    # -----------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32) → FP16
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections (bias‑less) in FP16
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    # Projections
    left  = F.linear(x, left_proj_w)      # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)     # (B,N,N,hidden_dim)

    # Gates
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # -----------------------------------------------------------------
    # 3. Fuse mask & gates via Triton (in‑place)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure contiguous tensors for the kernel
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # Simple element‑wise gating without mask
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 4. Core contraction: batch GEMM (L @ Rᵀ) over the sequence dimension
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 5. Output LayerNorm, gate, final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # element‑wise gating (FP16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in FP16

    # -----------------------------------------------------------------
    # 6. Cast back to float32 and return
    # -----------------------------------------------------------------
    out = out.to(torch.float32)
    return out
```
