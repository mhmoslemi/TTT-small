    stride_bz, stride_bm, stride_bn,

    N: tl.constexpr,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).
    Implements the core reduction with a fused mask/gate kernel and
    batched GEMM in a single Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1) LayerNorm (FP32 for stability)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)

    # 2) Linear projections (FP16)
    left = F.linear(x, weights["left_proj.weight"].to(device=device, dtype=torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(device=device, dtype=torch.float16))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(device=device, dtype=torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(device=device, dtype=torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device=device, dtype=torch.float16)))

    # 3) Rearrange to (B, H, N, N) and flatten batch*H
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left_gate = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    left_flat = left.view(B * H, N, N)
    right_flat = right.view(B * H, N, N)
    left_gate_flat = left_gate.view(B * H, N, N)
    right_gate_flat = right_gate.view(B * H, N, N)

    # 4) Prepare mask (fp16)
    if mask is not None and not nomask:
        mask_tensor = mask.to(torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    out_flat = torch.empty_like(left_flat)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    stride_az, stride_am, stride_ak = left_flat.stride()
    stride_bz, stride_bm, stride_bn = right_flat.stride()
    stride_cz, stride_cm, stride_cn = out_flat.stride()
    stride_gaz, stride_gam, stride_gak = left_gate_flat.stride()
    stride_gbz, stride_gbm, stride_gbn = right_gate_flat.stride()
    stride_mz, stride_mi, stride_mj = mask_tensor.stride()

    grid = (B * H, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    outgoing_trimul_kernel[grid](
        left_flat, right_flat, out_flat,
        left_gate_flat, right_gate_flat,
        mask_tensor,
        stride_az, stride_am, stride_ak,
        stride_bz, stride_bm, stride_bn,
        stride_cz, stride_cm, stride_cn,
        stride_gaz, stride_gam, stride_gak,
        stride_gbz, stride_gbm, stride_gbn,
        stride_mz, stride_mi, stride_mj,
        N,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # 7) Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 8) Second LayerNorm over hidden dim
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float16),
    )

    # 9) Apply output gate
    out = out * out_gate

    # 10) Final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"].to(device=device, dtype=torch.float16))

    # 11) Cast back to original dtype if needed
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out

"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The kernel fuses:
  1. Masking + left/right sigmoid gates.
  2. The core ∑_k left[i,k] * right[j,k] reduction.
All heavy arithmetic is performed in fp16 with fp32 accumulation.
The surrounding LayerNorms, linear projections and final projection are
handled in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _outgoing_trimul_kernel(
    # pointers
    A_ptr, B_ptr, C_ptr,          # left, right, output (B*H, N, N)
    A_gate_ptr, B_gate_ptr,       # left_gate, right_gate (B*H, N, N)
    mask_ptr,                     # mask (B, N, N) – broadcast over H
    # strides (int64)
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bm, stride_bn,
    stride_cz, stride_cm, stride_cn,
    stride_gaz, stride_gam, stride_gak,
    stride_gbz, stride_gbm, stride_gbn,
    stride_mz, stride_mi, stride_mj,
    # compile‑time constants
    N: tl.constexpr,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes:
        out[b,i,j] = Σ_k (A[b,i,k] * A_gate[b,i,k] * mask[b,i,k]) *
                     (B[b,j,k] * B_gate[b,j,k] * mask[b,j,k])
    where `b` indexes the combined batch*hidden dimension.
    """
    pid_z = tl.program_id(0)          # combined batch * hidden index
    pid_m = tl.program_id(1)          # block row (i)
    pid_n = tl.program_id(2)          # block col (j)

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Original batch index for mask (mask has no hidden dim)
    b = pid_z // H                     # integer division

    # Loop over the reduction dimension k
    for k_outer in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_outer * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left side (i,k) -----
        a_ptrs = A_ptr + pid_z * stride_az + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        a_gate_ptrs = A_gate_ptr + pid_z * stride_gaz + offs_m[:, None] * stride_gam + offs_k[None, :] * stride_gak
        a_gate = tl.load(a_gate_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        mask_a_ptrs = mask_ptr + b * stride_mz + offs_m[:, None] * stride_mi + offs_k[None, :] * stride_mj
        mask_a = tl.load(mask_a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        a_scaled = a.to(tl.float32) * a_gate.to(tl.float32) * mask_a.to(tl.float32)

        # ----- right side (j,k) – we need the transpose of B (j,k) → (k,j) for GEMM -----
        b_ptrs = B_ptr + pid_z * stride_bz + offs_k[:, None] * stride_bm + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        b_gate_ptrs = B_gate_ptr + pid_z * stride_gbz + offs_k[:, None] * stride_gbm + offs_n[None, :] * stride_gbn
        b_gate = tl.load(b_gate_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        mask_b_ptrs = mask_ptr + b * stride_mz + offs_n[:, None] * stride_mi + offs_k[None, :] * stride_mj
        mask_b = tl.load(mask_b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        b_scaled = b.to(tl.float32) * b_gate.to(tl.float32) * mask_b.to(tl.float32)

        # ----- accumulate -----
        acc += tl.dot(a_scaled, b_scaled)

    # Store result (fp16)
    c_ptrs = C_ptr + pid_z * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Input tuple:
        - input_tensor : torch.Tensor[bs, seq_len, seq_len, dim]
        - mask         : torch.Tensor[bs, seq_len, seq_len] or None
        - weights      : dict of model parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor[bs, seq_len, seq_len, dim] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (keep in fp32 for numeric stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden_dim (fp16)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(device=device, dtype=torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(device=device, dtype=torch.float16))

    # ------------------------------------------------------------------
    # 3) Sigmoid gates (fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(device=device, dtype=torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(device=device, dtype=torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(device=device, dtype=torch.float16))
    )

    # ------------------------------------------------------------------
    # 4) Re‑arrange tensors to (B, H, N, N) and flatten batch*hidden
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left_gate = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    left_flat = left.view(B * H, N, N)
    right_flat = right.view(B * H, N, N)
    left_gate_flat = left_gate.view(B * H, N, N)
    right_gate_flat = right_gate.view(B * H, N, N)

    # ------------------------------------------------------------------
    # 5) Mask (fp16) – broadcasted over hidden dimension
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_tensor = mask.to(torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6) Allocate output and launch Triton kernel
    # ------------------------------------------------------------------
    out_flat = torch.empty_like(left_flat)   # (B*H, N, N), fp16

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides for each tensor
    stride_az, stride_am, stride_ak = left_flat.stride()
    stride_bz, stride_bm, stride_bn = right_flat.stride()
    stride_cz, stride_cm, stride_cn = out_flat.stride()
    stride_gaz, stride_gam, stride_gak = left_gate_flat.stride()
    stride_gbz, stride_gbm, stride_gbn = right_gate_flat.stride()
    stride_mz, stride_mi, stride_mj = mask_tensor.stride()

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _outgoing_trimul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        left_gate_flat,
        right_gate_flat,
        mask_tensor,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bm,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        stride_gaz,
        stride_gam,
        stride_gak,
        stride_gbz,
        stride_gbm,
        stride_gbn,
        stride_mz,
        stride_mi,
        stride_mj,
        N,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"]
        .to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"]
        .to(device=device, dtype=torch.float16),
    )

    # ------------------------------------------------------------------
    # 9) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10) Final projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].to(device=device, dtype=torch.float16),
    )

    # ------------------------------------------------------------------
    # 11) Cast back to the original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
