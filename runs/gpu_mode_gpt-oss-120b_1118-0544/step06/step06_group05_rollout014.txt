@triton.jit
def _apply_mask_and_gate_kernel(
    left_proj_ptr, left_gate_ptr, right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    H, total_elems,
    BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # Compute idx for mask: off // H (since total_elems = B*N*N*H)
    mask_idx = offs // H

    left_proj = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left = left_proj * left_gate * m
    right = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left, mask=mask)
    tl.store(right_out_ptr + offs, right, mask=mask)
```

"""
custom_kernel: Triton‑accelerated outgoing TriMul forward (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input over the channel dimension (`dim`).
2. Cast to FP16 for the heavy compute.
3. Fuse the five linear projections (left_proj, right_proj, left_gate,
   right_gate, out_gate) into one matmul using a concatenated weight matrix.
   Split the result and apply sigmoid to the three gate slices.
4. If a mask is provided (and `nomask` is False) fuse mask, projection and
   gate element‑wise multiplication for **both** left and right tensors in a
   single Triton kernel.  Otherwise perform the element‑wise multiplication
   in PyTorch.
5. Core TriMul reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM by reshaping to (B*H, N, N) and using
   `torch.bmm`.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and final linear projection back to the original `dim`.
7. Cast the result back to the input dtype.

The kernel in step 4 reduces memory traffic and ensures at least one
Triton component is used.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _gate_mask_fuse_kernel(
    left_proj_ptr,   # fp16 *
    left_gate_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse: left = left_proj * left_gate * mask,
            right = right_proj * right_gate * mask."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    within = offs < total_elems

    # mask is broadcast along the hidden dimension
    mask_index = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=within, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=within, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=within, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=within, other=0.0)
    m = tl.load(mask_ptr + mask_index, mask=within, other=1.0)

    left = lp * lg * m
    right = rp * rg * m

    tl.store(left_out_ptr + offs, left, mask=within)
    tl.store(right_out_ptr + offs, right, mask=within)


def _apply_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """Launch the Triton kernel that fuses mask and gate multiplication."""
    assert left_proj.shape == right_proj.shape == left_gate.shape == right_gate.shape
    total = left_proj.numel()
    hidden_dim = left_proj.shape[-1]

    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_fuse_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        hidden_dim,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, C] (float32/float16)
            mask         : torch.Tensor [B, N, N] or None (0/1)
            weights      : dict of model parameters (float32)
            config       : dict with keys:
                "dim"          – input channel dimension C
                "hidden_dim"   – hidden dimension H
                "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim) – still in input dtype

    # ------------------------------------------------------------------
    # 2) Cast everything to FP16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Fuse the five linear projections into a single matmul
    # ------------------------------------------------------------------
    # concatenate weights: [left_proj, right_proj, left_gate, right_gate, out_gate]
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # (5*H, dim)

    # single linear: result shape (B, N, N, 5*H)
    proj_all = F.linear(x, concat_weight)  # FP16

    H = hidden_dim
    # Slice the concatenated result
    left_proj = proj_all[..., :H]
    right_proj = proj_all[..., H : 2 * H]
    left_gate = proj_all[..., 2 * H : 3 * H].sigmoid()
    right_gate = proj_all[..., 3 * H : 4 * H].sigmoid()
    out_gate = proj_all[..., 4 * H : 5 * H].sigmoid()

    # ------------------------------------------------------------------
    # 4) Apply mask & gating (fused Triton kernel if mask is used)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure contiguous and FP16
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_gate_mask(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
        )
    else:
        # simple element‑wise fusion (mask is all ones)
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask exists but we are told to ignore it -> treat as all ones
            # (no effect, so we skip)
            pass

    # free intermediate tensors to keep memory modest
    del left_proj, right_proj, left_gate, right_gate, proj_all

    # ------------------------------------------------------------------
    # 5) Core TriMul reduction (batched GEMM)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 7) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
