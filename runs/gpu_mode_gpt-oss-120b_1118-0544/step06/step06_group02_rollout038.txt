```python
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_B, stride_left_N1, stride_left_N2, stride_left_H,
    stride_right_B, stride_right_N1, stride_right_N2, stride_right_H,
    stride_out_B, stride_out_N1, stride_out_N2, stride_out_H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0) # batch * hidden
    pid1 = tl.program_id(1) # block_i
    pid2 = tl.program_id(2) # block_j

    b = pid0 // H
    h = pid0 % H

    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N

    # Create offsets
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    # mask for out of bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)

        # mask for k
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        # pointer: base + b*stride_left_B + h*stride_left_H + offs_i[:,None]*stride_left_N1 + offs_k[None,:]*stride_left_N2
        left_ptrs = left_ptr + b*stride_left_B + h*stride_left_H + offs_i[:,None] * stride_left_N1 + offs_k[None,:] * stride_left_N2
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile: we need shape (BLOCK_K, BLOCK_N) (k,j)
        right_ptrs = right_ptr + b*stride_right_B + h*stride_right_H + offs_j[None,:] * stride_right_N1 + offs_k[:,None] * stride_right_N2
        # This yields shape (BLOCK_K, BLOCK_N) if we arrange correctly: first dim is k, second is j.
        # Let's compute as:
        right_ptrs = right_ptr + b*stride_right_B + h*stride_right_H + offs_k[:,None] * stride_right_N2 + offs_j[None,:] * stride_right_N1
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Multiply-accumulate
        acc += tl.dot(left, right)  # left (M,K), right (K,N) -> (M,N)

    # Store result
    out_ptrs = out_ptr + b*stride_out_B + h*stride_out_H + offs_i[:,None] * stride_out_N1 + offs_j[None,:] * stride_out_N2
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_B, stride_left_N1, stride_left_N2, stride_left_H,
    stride_right_B, stride_right_N1, stride_right_N2, stride_right_H,
    stride_out_B, stride_out_N1, stride_out_N2, stride_out_H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)  # batch * hidden
    pid1 = tl.program_id(1)  # i block
    pid2 = tl.program_id(2)  # j block

    # Compute batch and hidden indices
    b = pid0 // H
    h = pid0 % H

    # Starting coordinates
    i_start = pid1 * BLOCK_M
    j_start = pid2 * BLOCK_N

    # Offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # Masks for in-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

"""
TriMul (outgoing) – fused mask/gate (Triton) + batched O(N³) reduction (Triton).

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C).
2. Cast to fp16 for the compute‑heavy part.
3. Four linear projections (left/right + gates) are done with torch.nn.functional.linear.
4. Fuse projection, gate and (optional) mask in a tiny Triton kernel
   (`_gate_mask_kernel`). This saves two memory passes.
5. Core O(N³·H) reduction:
   - For each batch b and hidden dimension h we compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   - Implemented as a tiled GEMM‑style kernel (`tri_mul_kernel`) that
     loads tiles of left (i,k) and right (k,j) in fp16, accumulates in fp32
     and writes the result in fp16.
6. Apply a second LayerNorm on the hidden dimension, multiply by the output gate.
7. Linear projection back to the original channel dimension.
8. Cast the result back to the input dtype (float32).

Only the mask‑gate fusion and the O(N³) reduction are written in Triton,
the remaining ops use highly‑optimized PyTorch kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) on a flattened view
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask is broadcasted over the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 2048,
) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using a Triton kernel.
    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],          # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: core TriMul reduction (outgoing)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # fp16 * (B,N,N,H)
    right_ptr,         # fp16 * (B,N,N,H)
    out_ptr,           # fp16 * (B,N,N,H)
    B, N, H,
    stride_left_B, stride_left_N1, stride_left_N2, stride_left_H,
    stride_right_B, stride_right_N1, stride_right_N2, stride_right_H,
    stride_out_B, stride_out_N1, stride_out_N2, stride_out_H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)  # batch * hidden
    pid1 = tl.program_id(1)  # i‑tile
    pid2 = tl.program_id(2)  # j‑tile

    # Decode batch and hidden indices
    b = pid0 // H
    h = pid0 % H

    # Starting indices of the tile
    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N

    # Offsets within the matrix
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load left tile: shape (BLOCK_M, BLOCK_K) ----
        left_ptrs = left_ptr + \
            b * stride_left_B + h * stride_left_H + \
            offs_i[:, None] * stride_left_N1 + \
            offs_k[None, :] * stride_left_N2
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ---- load right tile transposed: shape (BLOCK_K, BLOCK_N) ----
        right_ptrs = right_ptr + \
            b * stride_right_B + h * stride_right_H + \
            offs_k[:, None] * stride_right_N2 + \
            offs_j[None, :] * stride_right_N1
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # ---- accumulate ----
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # Store result back to fp16
    out_ptrs = out_ptr + \
        b * stride_out_B + h * stride_out_H + \
        offs_i[:, None] * stride_out_N1 + \
        offs_j[None, :] * stride_out_N2
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, C]
        - mask        : torch.Tensor of shape [B, N, N] (or None)
        - weights     : dict of model parameters (float32)
        - config      : dict with keys:
            * "dim" (int)         – input channel dimension C
            * "hidden_dim" (int)  – hidden dimension H
            * "nomask" (bool)     – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------- LayerNorm (input) -------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for compute‑intensive part
    x = x.to(torch.float16)

    # --------------------------- cast weights ---------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ----------------------- linear projections -------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)          # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)         # (B,N,N,H)

    # --------------------------- gates ----------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------- mask & gate fusion (Triton) --------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = (left_proj * left_gate).contiguous()
        right = (right_proj * right_gate).contiguous()

    # ----------------- core O(N³·H) reduction via Triton -----------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # (B,N,N,H), fp16

    # Strides for contiguous tensors (B,N,N,H)
    stride_left_B, stride_left_N1, stride_left_N2, stride_left_H = left.stride()
    stride_right_B, stride_right_N1, stride_right_N2, stride_right_H = right.stride()
    stride_out_B, stride_out_N1, stride_out_N2, stride_out_H = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_left_B, stride_left_N1, stride_left_N2, stride_left_H,
        stride_right_B, stride_right_N1, stride_right_N2, stride_right_H,
        stride_out_B, stride_out_N1, stride_out_N2, stride_out_H,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------ cast back to input dtype ----------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
