def batched_triton_matmul(left, right, block_m=64, block_n=64, block_k=64):
    # left, right: shape [B, N, N, H], dtype fp16
    B, N, _, H = left.shape
    # Permute to [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # reshape to [B*H, N, N]
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)
    # Allocate output tensor
    out_flat = torch.empty_like(left_flat)  # same dtype
    # Compute strides
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)
    stride_batch_a = left_flat.stride(0)
    stride_bk = right_flat.stride(2)  # K stride (since we need B's K stride)
    stride_bn = right_flat.stride(1)  # N stride for B (the transposed dimension)
    stride_batch_b = right_flat.stride(0)
    stride_cm = out_flat .stride(1)
    stride_cn = out_flat .stride(2)
    stride_batch_c = out_flat .stride(0)
    # grid
    num_blocks_m = (N + block_m - 1) // block_m
    num_blocks_n = (N + block_n - 1) // block_n
    batch = B * H
    grid = (batch, num_blocks_m, num_blocks_n)
    # Launch kernel
    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    # Reshape output back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
```

@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)  # batch * hidden
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Compute base pointers for this batch
    a_ptrs = a_ptr + pid_z * stride_batch_a + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
    b_ptrs = b_ptr + pid_z * stride_batch_b + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load a and b
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        # Increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_z * stride_batch_c + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, c, mask=mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # Cast to fp16
    x = x.to(torch.float16)

    # Cast weight matrices to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # projections
    left_proj = F.linear(x, left_proj_w)  # shape [B,N,N,hidden_dim]
    right_proj = F.linear(x, right_proj_w)

    # gates
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # apply mask if needed
    if mask is not None and not nomask:
        # Ensure mask shape: [B,N,N,1] broadcast over hidden dim
        mask = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # core reduction using Triton kernel
    out = _batched_matmul_triton(left, right)

    # output layernorm
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # multiply by out_gate
    out = out * out_gate

    # final linear
    out = F.linear(out, to_out_w)  # shape [B,N,N,dim]

    # cast back to original dtype
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)
    return out

@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_batch_a,
    stride_bk, stride_bn, stride_batch_b,
    stride_cm, stride_cn, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_z = tl.program_id(0)  # batch index (B*H)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute starting offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers
    a_ptrs = a_ptr + pid_z * stride_batch_a + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
    b_ptrs = b_ptr + pid_z * stride_batch_b + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_offset = k * BLOCK_K
        # compute mask for K dimension
        mask_k_a = (k_offset + offs_k) < K
        mask_k_b = mask_k_a

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k_a[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k_b[:, None] & mask_n[None, :], other=0.0)

def _batched_matmul_triton(left, right):
    # left, right: shape [B,N,N,H], fp16
    B, N, _, H = left.shape
    # Permute and reshape
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)
    out_flat = torch.empty_like(left_flat)
    # Strides
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)
    stride_batch_a = left_flat.stride(0)
    stride_bk = right_flat.stride(2)  # since we treat B as (N, K) for B^T, K stride is last dim
    stride_bn = right_flat.stride(1)  # N stride
    stride_batch_b = right_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)
    stride_batch_c = out_flat.stride(0)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    batch = B * H
    grid = (batch, num_blocks_m, num_blocks_n)

    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_am, stride_ak, stride_batch_a,
        stride_bk, stride_bn, stride_batch_b,
        stride_cm, stride_cn, stride_batch_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

TriMul (outgoing) forward pass – Triton‑accelerated core reduction.

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (left/right) and gate projections (left/right/out)

4. Apply optional mask and fuse gate multiplication (elementwise).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (A @ Bᵀ) with a custom Triton kernel.
   The tensors are reshaped to (B·H, N, N) so each kernel instance processes
   one (batch, hidden) pair.
6. Output LayerNorm, out‑gate multiplication and final linear projection
   back to the original dimension.
7. Cast back to the original dtype (usually float32).

The Triton kernel performs the batched matrix multiplication in fp16
with fp32 accumulation, exploiting tensor cores on the B200 GPU.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for batched GEMM: C = A @ Bᵀ
# A : (batch, M, K)   (here M = N, K = N)
# B : (batch, N, K)   (interpreted as transposed inside the kernel)
# C : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_batch_a,
    stride_bk, stride_bn, stride_batch_b,
    stride_cm, stride_cn, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)   # batch * hidden_dim index
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block column

    #--------------------------------------------------------------------------------------------------
    # Offsets within the tiles
    #--------------------------------------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_ptrs = a_ptr + pid_z * stride_batch_a + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
    b_ptrs = b_ptr + pid_z * stride_batch_b + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_offset = k * BLOCK_K
        # Masks for the K‑dimension
        mask_k = (k_offset + offs_k) < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)          # fp16·fp16 → fp32 accumulation

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write the result back (store as fp16)
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_z * stride_batch_c + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, c, mask=mask)


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    left, right: tensors of shape [B, N, N, H] and dtype fp16.
    Returns: tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape

    # Rearrange to (B, H, N, N) → (B*H, N, N)
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_ = torch.empty_like(left_)

    # Strides
    stride_am = left_.stride(1)          # stride over M dimension
    stride_ak = left_.stride(2)          # stride over K dimension
    stride_batch_a = left_.stride(0)

    stride_bk = right_.stride(2)         # stride over K dimension of B (transposed view)
    stride_bn = right_.stride(1)         # stride over N dimension of B
    stride_batch_b = right_.stride(0)

    stride_cm = out_.stride(1)
    stride_cn = out_.stride(2)
    stride_batch_c = out_.stride(0)

    # Tunable block sizes (good defaults for B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * H, num_blocks_m, num_blocks_n)

    _batched_gemm_kernel[grid](
        left_,                               # a_ptr
        right_,                              # b_ptr
        out_,                                # c_ptr
        N, N, N,                            # M, N, K
        stride_am, stride_ak, stride_batch_a,
        stride_bk, stride_bn, stride_batch_b,
        stride_cm, stride_cn, stride_batch_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (bool/float)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to fp16 (no bias for linear layers)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (produces [B, N, N, H])
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # shape: (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask (if present) and fuse with gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: [B, N, N] → [B, N, N, 1] for broadcasting over hidden_dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction using the Triton batched GEMM kernel
    # ------------------------------------------------------------------
    out = _batched_matmul_triton(left, right)   # shape: [B, N, N, H], fp16

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # shape: [B, N, N, C]

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
