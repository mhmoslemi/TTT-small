        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # cast to fp16 for speed
    x = x.to(torch.float16)

    # 2) Linear projections (concatenated for left and right)
    left_concat_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(device=device, dtype=torch.float16)
    right_concat_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(device=device, dtype=torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # Compute concatenated outputs
    left_concat = F.linear(x, left_concat_weight)  # (B,N,N,2*H)
    right_concat = F.linear(x, right_concat_weight)  # (B,N,N,2*H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # (B,N,N,H)

    # Split projection and gate
    left_proj, left_gate_raw = left_concat.chunk(2, dim=-1)
    right_proj, right_gate_raw = right_concat.chunk(2, dim=-1)

    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # 3) Mask handling
    B, N, _, _ = left_proj.shape
    if mask is None or nomask:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(torch.float16)

    # 4) Fuse mask and gates using Triton
    total_elements = left_proj.numel()  # B*N*N*H
    BLOCK = 8192  # adjust
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total_elements, BLOCK),)
    _mask_gate_fusion_kernel[grid](
        left_proj,
        right_proj,
        mask_tensor.view(-1),  # flatten
        left_gate,
        right_gate,
        left_out,
        right_out,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # 5) Core reduction via batched matmul
    # reshape to (B*H, N, N)
    left_flat = left_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 6) Second LayerNorm over hidden_dim
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float16),
    )

    # 7) Apply out_gate
    out = out * out_gate

    # 8) Final projection to original dim
    to_out_weight = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_weight)

    # 9) Cast back to original dtype
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out

"""
TriMul (outgoing) forward pass with heavy parts fused in Triton.

Algorithm
---------
1. Layer‑normalize the input tensor (float32 → float16 for speed).
2. Fuse the two linear projections (proj + gate) for the left and right
   branches by concatenating their weights, reducing the number of GEMM calls
   from 5 to 3.
3. Apply sigmoid to the gate halves.
4. Fuse mask (if any) and the gating into a single Triton kernel producing
   `left_out` and `right_out` (both shape [B, N, N, H]).
5. Core multiplicative update:
   out[b,i,j,h] = Σ_k left_out[b,i,k,h] * right_out[b,j,k,h]
   Implemented as a batched matrix multiplication:
   (B*H, N, N) @ (B*H, N, N)^T via `torch.bmm`.
6. Layer‑normalize over the hidden dimension H.
7. Multiply by the output gate.
8. Final linear projection back to the original channel dimension C.
9. Cast back to the original dtype (float32).

Only the mask‑gate fusion is written in Triton; the remaining linear
operations use PyTorch’s highly‑optimized GEMM kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # const * left  (B,N,N,H) flattened
    right_ptr,         # const * right (B,N,N,H) flattened
    mask_ptr,          # const * mask  (B,N,N)   flattened
    left_gate_ptr,     # const * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # const * right_gate (B,N,N,H) flattened
    left_out_ptr,      # * left_out  (B,N,N,H) flattened
    right_out_ptr,     # * right_out (B,N,N,H) flattened
    H: tl.constexpr,               # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,      # thread block size
):
    """Fuse mask and gate:
       left_out  = left  * mask * left_gate
       right_out = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load data (masked)
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Compute index into the 2‑D mask (repeated for each hidden dim)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Write fused results
    tl.store(left_out_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)

    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, if True ignore mask (default False)

    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    x = F.layer_norm(

        (dim,),

        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    right_proj, right_gate_raw = right_concat.split(hidden_dim, dim=-1)

    right_gate = torch.sigmoid(right_gate_raw)

    # ------------------------------------------------------------------

    if mask is None or nomask:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    total_elements = left_proj.numel()                     # B * N * N * H

        left_gate,

        left_out,

        hidden_dim,

        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 5) Core reduction: batched GEMM (left_out @ right_outᵀ)
    #    Reshape to (B*H, N, N) for torch.bmm.
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_flat = left_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication using cuBLAS (fp16 -> fp16)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)  # (B, N, N, H)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float16),
    )

    # ------------------------------------------------------------------
    # 7) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final projection back to original channel dimension C
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
