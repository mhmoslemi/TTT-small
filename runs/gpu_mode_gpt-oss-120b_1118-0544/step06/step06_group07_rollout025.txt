custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1️⃣ Input LayerNorm (float32) on [B, N, N, C].
2️⃣ Cast to FP16 for compute‑heavy path.
3️⃣ Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)

7️⃣ Hidden‑dim LayerNorm (float32), multiply by out_gate, and final linear
   projection back to the original channel dimension.
8️⃣ Cast back to the input dtype and return.

All heavy‑weight tensor operations are performed in FP16 except the
LayerNorms (FP32) for numerical stability.  At least one operation
(gate‑mask fusion) is implemented in Triton.

import triton.language as tl

@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,          # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,# same shape
    mask_ptr,                     # [B,N,N] flattened (optional)
    total_elems, hidden_dim,      # runtime scalars
    HAS_MASK: tl.constexpr,       # compile‑time bool
    BLOCK_SIZE: tl.constexpr,     # compile‑time int
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    If HAS_MASK=False the mask term is omitted (treated as 1.0).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # mask is broadcast over the hidden dimension
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim" (C) and "hidden_dim" (H);
                      optional "nomask" (bool).

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for compute path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse five linear projections into one matmul
    # ------------------------------------------------------------------
    # Stack weight matrices: each has shape (hidden_dim, dim)
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    # Concatenate along the output dimension -> (5*H, dim)
    stacked_w = torch.cat([
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
    ], dim=0)  # shape (5*H, dim)

    B, N, _, _ = x.shape
    # Flatten input to (B*N*N, dim)
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)
    y_flat = F.linear(x_flat, stacked_w)         # (B*N*N, 5*H)
    y = y_flat.view(B, N, N, 5 * hidden_dim)    # (B,N,N,5*H)

    # Split projections and gates
    left       = y[..., 0                 : hidden_dim]                # (B,N,N,H)
    right      = y[..., hidden_dim        : 2*hidden_dim]              # (B,N,N,H)
    left_gate  = torch.sigmoid(y[..., 2*hidden_dim    : 3*hidden_dim])  # (B,N,N,H)
    right_gate = torch.sigmoid(y[..., 3*hidden_dim    : 4*hidden_dim])  # (B,N,N,H)
    out_gate   = torch.sigmoid(y[..., 4*hidden_dim    : 5*hidden_dim])  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse mask and gates (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure contiguity
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)  # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # Simple element‑wise fusion when mask is absent
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # Prepare tensors of shape (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Batched matrix multiplication (uses cuBLAS tensor‑cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm, output gating, final projection
    # ------------------------------------------------------------------
    # LayerNorm in float32 for stability
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out = out.to(torch.float32)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply output gate (still in float32 -> cast to fp16 for next matmul)
    out = out.to(torch.float16) * out_gate

    # Final linear projection back to original dim
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)  # (dim, hidden_dim)
    out = F.linear(out, to_out_w)  # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to original dtype and return
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
