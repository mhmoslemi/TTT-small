@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_bh, stride_left_m, stride_left_k,
    stride_right_bh, stride_right_k, stride_right_n,
    stride_out_bh, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)  # batch index (B*H)
    pid_m = tl.program_id(1)  # row block
    pid_n = tl.program_id(2)  # col block

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load left block (M,K)
        left_ptrs = left_ptr + pid_b * stride_left_bh + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask = (offs_m[:, None] < N) & (offs_k[None, :] < N), other=0.0)

        # Load right block (N,K) but we need transposed version: right_T has shape (K,N)
        # We will load right as (K,N) by swapping indices: i.e., load right[j,k] -> we need right_T[k,j]
        # Original right shape is (B*H, N, N) with strides (stride_right_bh, stride_right_m, stride_right_k)
        # To get right_T[k,j], we index k as row (first dim of right_T) and j as column.
        # This corresponds to original right at (j,k). So we compute pointer:
        right_ptrs = right_ptr + pid_b * stride_right_bh + offs_n[:, None] * stride_right_m + offs_k[None, :] * stride_right_k
        # Here right_ptrs shape (BLOCK_N, BLOCK_K) but we need shape (BLOCK_K, BLOCK_N) to multiply.
        # We'll load then transpose.
        right = tl.load(right_ptrs, mask = (offs_n[:, None] < N) & (offs_k[None, :] < N), other=0.0)
        # Now right shape (BLOCK_N, BLOCK_K). We want (BLOCK_K, BLOCK_N), so transpose:
        right_T = tl.trans(right)

        # Accumulate
        acc += tl.dot(left, right_T)

    # Store result
    out_ptrs = out_ptr + pid_b * stride_out_bh + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc.to(tl.float16), mask = (offs_m[:, None] < N) & (offs_n[None, :] < N))
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = F.layer_norm(...)

    # Cast to fp16
    x = x.to(torch.float16)

    # Linear projections
    left = F.linear(x, left_proj_weight)   # (B,N,N,H)
    right = F.linear(x, right_proj_weight)
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Mask handling
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_tensor = mask.to(torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # Permute to (B,H,N,N) and reshape to (B*H,N,N)
    left = left.permute(0,3,1,2).contiguous()
    right = right.permute(0,3,1,2).contiguous()
    left_gate = left_gate.permute(0,3,1,2).contiguous()
    right_gate = right_gate.permute(0,3,1,2).contiguous()

    batch_h = B * hidden_dim
    left = left.view(batch_h, N, N)
    right = right.view(batch_h, N, N)
    left_gate = left_gate.view(batch_h, N, N)
    right_gate = right_gate.view(batch_h, N, N)

    # Output allocation
    out = torch.empty_like(left)  # shape (batch_h, N, N), dtype fp16

    # Strides
    stride_l_b = left.stride(0)
    stride_l_m = left.stride(1)
    stride_l_k = left.stride(2)

    # same for right
    stride_r_b = right.stride(0)
    stride_r_m = right.stride(1)
    stride_r_k = right.stride(2)

    # gates
    stride_lg_b = left_gate.stride(0)
    stride_lg_m = left_gate.stride(1)
    stride_lg_k = left_gate.stride(2)

    stride_rg_b = right_gate.stride(0)
    stride_rg_m = right_gate.stride(1)
    stride_rg_k = right_gate.stride(2)

    # mask stride
    stride_mask_b = mask_tensor.stride(0)
    stride_mask_i = mask_tensor.stride(1)
    stride_mask_k = mask_tensor.stride(2)

    # out strides
    stride_out_b = out.stride(0)
    stride_out_m = out.stride(1)
    stride_out_n = out.stride(2)

    # kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch_h, tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N))
    tri_mul_forward_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_tensor,
        out,
        N,
        hidden_dim,
        stride_l_b, stride_l_m, stride_l_k,
        stride_r_b, stride_r_m, stride_r_k,
        stride_lg_b, stride_lg_m, stride_lg_k,
        stride_rg_b, stride_rg_m, stride_rg_k,
        stride_mask_b, stride_mask_i, stride_mask_k,
        stride_out_b, stride_out_m, stride_out_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # Reshape out to (B, N, N, H)
    out = out.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # Second LayerNorm
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply out gate
    out = out * out_gate

    # Final projection
    out = F.linear(out, to_out_w)

    # Cast back to original dtype if needed
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out

Kernel signature:

```
@triton.jit
def tri_mul_forward_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_ptr,
    N, H,
    stride_l_b, stride_l_m, stride_l_k,
    stride_r_b, stride_r_m, stride_r_k,
    stride_lg_b, stride_lg_m, stride_lg_k,
    stride_rg_b, stride_rg_m, stride_rg_k,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    batch_idx = pid_b // H
    # hidden_idx = pid_b % H  # not used

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid rows/cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block and its gate and mask
        left_ptrs = left_ptr + pid_b * stride_l_b + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        left_gate_ptrs = left_gate_ptr + pid_b * stride_lg_b + offs_m[:, None] * stride_lg_m + offs_k[None, :] * stride_lg_k
        left_gate = tl.load(left_gate_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        mask_ptrs_left = mask_ptr + batch_idx * stride_mask_b + offs_m[:, None] * stride_mask_i + offs_k[None, :] * stride_mask_k
        mask_val_left = tl.load(mask_ptrs_left, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        left_val = left * left_gate * mask_val_left

        # Load right block and its gate and mask (note right uses offs_n as rows)
        right_ptrs = right_ptr + pid_b * stride_r_b + offs_n[:, None] * stride_r_m + offs_k[None, :] * stride_r_k
        right = tl.load(right_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)
        right_gate_ptrs = right_gate_ptr + pid_b * stride_rg_b + offs_n[:, None] * stride_rg_m + offs_k[None, :] * stride_rg_k
        right_gate = tl.load(right_gate_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)
        mask_ptrs_right = mask_ptr + batch_idx * stride_mask_b + offs_n[:, None] * stride_mask_i + offs_k[None, :] * stride_mask_k
        mask_val_right = tl.load(mask_ptrs_right, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        right_val = right * right_gate * mask_val_right

        # Transpose right_val to (K,N)
        right_val_T = tl.trans(right_val)

        # Accumulate dot product
        acc += tl.dot(left_val, right_val_T)

    # Store result
    out_ptrs = out_ptr + pid_b * stride_out_b + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor (fp32 stats) and cast to fp16.
2. Linear projections (left/right) and three sigmoid gates (left, right, out) to the
   hidden dimension `hidden_dim` (all fp16).
3. Fuse mask, left/right gates and the core TriMul reduction into a single
   Triton kernel:
       out[b,i,j,h] = Σ_k (left[b,i,k,h] * left_gate[b,i,k,h] *
                           mask[b,i,k]) *
                       (right[b,j,k,h] * right_gate[b,j,k,h] *
                           mask[b,j,k])
   The kernel performs a blocked matrix multiplication for each
   (batch * hidden_dim) pair, accumulating in fp32.
4. Reshape the result back to (B, N, N, hidden_dim).
5. Second LayerNorm over the hidden dimension.
6. Apply the output gate and a final linear projection back to the original
   channel dimension `dim`.
7. Cast the result to the original input dtype.

The heavy O(N³·hidden_dim) work is done entirely on the GPU in the Triton
kernel, avoiding extra PyTorch memory copies and achieving a large speed‑up
over a naïve implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_forward_kernel(
    # tensors
    left_ptr, right_ptr,               # (B*H, N, N) left/right projections
    left_gate_ptr, right_gate_ptr,     # (B*H, N, N) gate tensors
    mask_ptr,                          # (B, N, N) binary mask
    out_ptr,                           # (B*H, N, N) output
    # compile‑time constants
    N: tl.constexpr,                   # sequence length
    H: tl.constexpr,                   # hidden dimension
    # strides (int64)
    stride_l_b, stride_l_m, stride_l_k,
    stride_r_b, stride_r_m, stride_r_k,
    stride_lg_b, stride_lg_m, stride_lg_k,
    stride_rg_b, stride_rg_m, stride_rg_k,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_m, stride_out_n,
    # block sizes (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix‑multiply with mask & gate fusion for the outgoing TriMul."""
    pid_b = tl.program_id(0)            # batch * hidden index
    pid_m = tl.program_id(1)            # block row (i)
    pid_n = tl.program_id(2)            # block col (j)

    # batch index for mask (mask has no hidden dimension)
    batch_idx = pid_b // H

    # offsets within the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for kk in range(0, num_k):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---------- left side ----------
        left_ptrs = (left_ptr + pid_b * stride_l_b
                     + offs_m[:, None] * stride_l_m
                     + offs_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        left_gate_ptrs = (left_gate_ptr + pid_b * stride_lg_b
                          + offs_m[:, None] * stride_lg_m
                          + offs_k[None, :] * stride_lg_k)
        left_gate = tl.load(left_gate_ptrs,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0)

        mask_left_ptrs = (mask_ptr + batch_idx * stride_mask_b
                          + offs_m[:, None] * stride_mask_i
                          + offs_k[None, :] * stride_mask_k)
        mask_left = tl.load(mask_left_ptrs,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0)

        left_val = left * left_gate * mask_left

        # ---------- right side ----------
        right_ptrs = (right_ptr + pid_b * stride_r_b
                      + offs_n[:, None] * stride_r_m
                      + offs_k[None, :] * stride_r_k)
        right = tl.load(right_ptrs,
                        mask=mask_n[:, None] & mask_k[None, :],
                        other=0.0)

        right_gate_ptrs = (right_gate_ptr + pid_b * stride_rg_b
                           + offs_n[:, None] * stride_rg_m
                           + offs_k[None, :] * stride_rg_k)
        right_gate = tl.load(right_gate_ptrs,
                             mask=mask_n[:, None] & mask_k[None, :],
                             other=0.0)

        mask_right_ptrs = (mask_ptr + batch_idx * stride_mask_b
                           + offs_n[:, None] * stride_mask_i
                           + offs_k[None, :] * stride_mask_k)
        mask_right = tl.load(mask_right_ptrs,
                             mask=mask_n[:, None] & mask_k[None, :],
                             other=0.0)

        right_val = right * right_gate * mask_right

        # transpose right_val to (K, N) and perform dot product
        acc += tl.dot(left_val, tl.trans(right_val))

    # store accumulator (cast back to fp16)
    out_ptrs = (out_ptr + pid_b * stride_out_b
                + offs_m[:, None] * stride_out_m
                + offs_n[None, :] * stride_out_n)
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (torch.Tensor)
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, if True ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float32)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float32)
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # Cast to fp16 for the heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)     # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Prepare mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_tensor = mask.to(device=device, dtype=torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6) Rearrange tensors to (B*H, N, N) layout for the kernel
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    left_gate = left_gate.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output tensor
    out = torch.empty_like(left)

    # ------------------------------------------------------------------
    # 7) Strides for the kernel
    # ------------------------------------------------------------------
    stride_l_b, stride_l_m, stride_l_k = left.stride()
    stride_r_b, stride_r_m, stride_r_k = right.stride()
    stride_lg_b, stride_lg_m, stride_lg_k = left_gate.stride()
    stride_rg_b, stride_rg_m, stride_rg_k = right_gate.stride()
    stride_mask_b, stride_mask_i, stride_mask_k = mask_tensor.stride()
    stride_out_b, stride_out_m, stride_out_n = out.stride()

    # ------------------------------------------------------------------
    # 8) Triton kernel launch (batched matmul with gating & mask)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _tri_mul_forward_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_tensor,
        out,
        N, hidden_dim,
        stride_l_b, stride_l_m, stride_l_k,
        stride_r_b, stride_r_m, stride_r_k,
        stride_lg_b, stride_lg_m, stride_lg_k,
        stride_rg_b, stride_rg_m, stride_rg_k,
        stride_mask_b, stride_mask_i, stride_mask_k,
        stride_out_b, stride_out_m, stride_out_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 9) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 10) Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # ------------------------------------------------------------------
    # 11) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 12) Final projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)

    # ------------------------------------------------------------------
    # 13) Cast back to the original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
