                     BATCH, M, N, K,
                     stride_l_batch, stride_l_m, stride_l_k,
                     stride_r_batch, stride_r_n, stride_r_k,
                     stride_o_batch, stride_o_m, stride_o_n,
                     BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load left tile
        left_ptrs = left_ptr + batch_id * stride_l_batch + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k

        # cast to float32
        a = a.to(tl.float32)
        b = b.to(tl.float32)

    # store result
    out_ptrs = out_ptr + batch_id * stride_o_batch + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n
    # convert acc to fp16 for storage
    acc = acc.to(tl.float16)

import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH, M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_n, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load left tile [M, K]
        left_ptrs = left_ptr + batch_id * stride_l_batch + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k
        a = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load right tile transposed: we need [K, N]
        right_ptrs = right_ptr + batch_id * stride_r_batch + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_n
        b = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Cast to float32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # Write back
    out_ptrs = out_ptr + batch_id * stride_o_batch + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n

def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    Implements the core contraction using a custom Triton kernel for
    the N‑by‑N‑by‑N tensor multiplication. All other operations
    (layer‑norms, linear projections, gates, mask handling) are
    performed with PyTorch (fp16) for simplicity.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input LayerNorm (float32 for stability)
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)  # switch to fp16 for the heavy parts

    # 2. Stack linear weights (no bias)
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    weight_all = torch.cat(
        [left_proj_w, right_proj_w,
         left_gate_w, right_gate_w,
         out_gate_w],
        dim=0
    )  # (5*hidden_dim, dim)

    # 3. Joint linear projection
    B, N, _, _ = input_tensor.shape
    x_flat = x.view(-1, dim)                      # (B*N*N, dim)
    proj = F.linear(x_flat, weight_all)           # (B*N*N, 5*hidden_dim)
    proj = proj.view(B, N, N, 5, hidden_dim)      # (B, N, N, 5, hidden_dim)

    left       = proj[..., 0, :]   # (B, N, N, H)
    right      = proj[..., 1, :]
    left_gate  = torch.sigmoid(proj[..., 2, :])
    right_gate = torch.sigmoid(proj[..., 3, :])
    out_gate   = torch.sigmoid(proj[..., 4, :])

    # 4. Mask and gate fusion (pure PyTorch, cheap)
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * left_gate * mask_fp
        right = right * right_gate * mask_fp
    else:
        left = left * left_gate
        right = right * right_gate

    # 5. Core contraction using Triton
    # reshape to (B*H, N, N) where each batch corresponds to a hidden channel
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)
    B_, H_, N_, _ = left_mat.shape
    batch = B_ * H_
    left_mat = left_mat.view(batch, N, N)
    right_mat = right_mat.view(batch, N, N)

    out_mat = torch.empty_like(left_mat)  # fp16

    # strides in element units
    stride_l_batch = left_mat.stride(0)
    stride_l_m = left_mat.stride(1)
    stride_l_k = left_mat.stride(2)

    stride_r_batch = right_mat.stride(0)
    stride_r_n = right_mat.stride(1)
    stride_r_k = right_mat.stride(2)

    stride_o_batch = out_mat.stride(0)
    stride_o_m = out_mat.stride(1)
    stride_o_n = out_mat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch,
        N,
        N,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_n,
        stride_r_k,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Output LayerNorm, out_gate, final linear
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # cast back to original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the PyTorch reference but fuses the
expensive N×N×N contraction with a custom Triton kernel.  All
operations are performed in fp16 (except the first LayerNorm for
numeric stability) and the final result is cast back to the input
dtype (usually float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH, M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_n, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication:
    out[b, i, j] = Σ_k left[b, i, k] * right[b, j, k]
    where left/right are (BATCH, M, K) and (BATCH, N, K) respectively.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile (M × K)
        left_ptrs = (
            left_ptr
            + batch_id * stride_l_batch
            + offs_m[:, None] * stride_l_m
            + offs_k[None, :] * stride_l_k
        )
        a = tl.load(
            left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0
        )

        # right tile transposed (K × N)
        right_ptrs = (
            right_ptr
            + batch_id * stride_r_batch
            + offs_k[:, None] * stride_r_k
            + offs_n[None, :] * stride_r_n
        )
        b = tl.load(
            right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0
        )

        # Promote to float32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)

    # Write back the result
    out_ptrs = (
        out_ptr
        + batch_id * stride_o_batch
        + offs_m[:, None] * stride_o_m
        + offs_n[None, :] * stride_o_n
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (tensors)
            - config: dict with keys "dim", "hidden_dim" and optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Stack all linear weights (biases are all False)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    # (5*hidden_dim, dim)
    weight_all = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0
    )

    # -----------------------------------------------------------------
    # 3. Joint projection + split
    # -----------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)
    proj = F.linear(x_flat, weight_all)          # (B*N*N, 5*hidden_dim)
    proj = proj.view(B, N, N, 5, hidden_dim)     # (B, N, N, 5, hidden_dim)

    left = proj[..., 0, :]       # (B, N, N, H)
    right = proj[..., 1, :]      # (B, N, N, H)
    left_gate = torch.sigmoid(proj[..., 2, :])
    right_gate = torch.sigmoid(proj[..., 3, :])
    out_gate = torch.sigmoid(proj[..., 4, :])

    # -----------------------------------------------------------------
    # 4. Apply mask (if any) and gate multiplications
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * left_gate * mask_fp
        right = right * right_gate * mask_fp
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core N×N×N contraction via Triton
    # -----------------------------------------------------------------
    # Rearrange to (B*H, N, N) where each slice is a matrix for one hidden channel
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B_, H_, N_, _ = left_mat.shape
    batch = B_ * H_
    left_mat = left_mat.view(batch, N, N)
    right_mat = right_mat.view(batch, N, N)

    out_mat = torch.empty_like(left_mat)  # fp16 output

    # Strides (in element units)
    stride_l_batch = left_mat.stride(0)
    stride_l_m = left_mat.stride(1)
    stride_l_k = left_mat.stride(2)

    stride_r_batch = right_mat.stride(0)
    stride_r_n = right_mat.stride(1)
    stride_r_k = right_mat.stride(2)

    stride_o_batch = out_mat.stride(0)
    stride_o_m = out_mat.stride(1)
    stride_o_n = out_mat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch,
        N,
        N,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_n,
        stride_r_k,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # Cast back to the original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
