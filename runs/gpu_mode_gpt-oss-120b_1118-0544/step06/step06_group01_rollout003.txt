"""
Outgoing Triangle Multiplicative Update (TriMul) – forward pass.

Algorithm
---------
1. Input LayerNorm (float32 for stability) → cast to FP16.
2. Linear projections:
   * left  = x @ W_left   (dim → hidden_dim)
   * right = x @ W_right  (dim → hidden_dim)
   * left_gate  = sigmoid(x @ W_left_gate)
   * right_gate = sigmoid(x @ W_right_gate)
   * out_gate   = sigmoid(x @ W_out_gate)       # used after output LN
3. Fuse mask (if present) and gate multiplications with a tiny Triton kernel.
   The kernel performs in‑place:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
   (mask broadcast over the hidden dimension)
4. Core contraction (batched GEMM over the sequence dimension):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiply on FP16 tensors.
5. Output LayerNorm over the hidden dimension, apply out‑gate, and final
   linear projection back to the original feature dimension.
6. Cast the result back to the input dtype (usually float32).

Only the mask‑gate fusion is executed in Triton; the remaining ops use
high‑performance PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,                # [B,N,N,H] flattened
    right_ptr,               # same as left
    left_gate_ptr,           # same shape
    right_gate_ptr,          # same shape
    mask_ptr,                # [B,N,N] flattened
    total_elems,             # total number of elements in left/right (B*N*N*H)
    hidden_dim,              # H
    HAS_MASK: tl.constexpr,  # compile‑time bool
    BLOCK_SIZE: tl.constexpr # compile‑time int
):
    """
    In‑place fusion of mask and gate multiplications:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    When HAS_MASK is false the mask term is omitted.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load data
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Each hidden_dim entries share the same mask entry.
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    # Store the fused results back in‑place
    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (torch.Tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32) and cast to fp16 for the heavy ops
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Load linear weights (cast to fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gate values
    # -----------------------------------------------------------------
    left  = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)         # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 4. Fuse mask and gate multiplication with a Triton kernel
    # -----------------------------------------------------------------
    use_mask = (mask is not None) and (not nomask)
    if use_mask:

        mask_fp = mask.to(device).to(torch.float16)

        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # The tensors are modified in‑place; no further reshaping needed.
    else:
        # No mask → simple element‑wise gate multiplication
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM over the sequence dimension
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # Reshape to (B*H, N, N) for batched matmul
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # Compute left @ rightᵀ for each head
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the original dtype (typically float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
