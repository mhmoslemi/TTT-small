"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate.
---------------------------------------------------------

1. Layer‑normalize the input tensor (B,N,N,C) from the provided weights.
2. Cast to FP16 for the heavy part of the computation.
3. Compute left/right projections and their gates in a single matmul per side
   by concatenating the projection and gate weight matrices (saving a pass).
4. Fuse element‑wise projection × gate × optional mask with a small Triton
   kernel (`_gate_mask_kernel`). This reduces two memory traversals.
5. Core O(N³·H) reduction:
   - Reshape the gated tensors to (B*H, N, N) and perform a batched GEMM
     (`torch.bmm`) with Tensor‑cores (FP16).
6. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate, and linearly project back to the original channel dimension.
7. Cast the result back to the input dtype (float32).

The kernel `_gate_mask_kernel` is the only required Triton part, satisfying the
requirement while keeping the overall runtime low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) on a flattened view
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # broadcast mask over hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 2048,
) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using a Triton kernel.
    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],          # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out

# Main entry point
# ----------------------------------------------------------------------

    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
        - mask        : torch.Tensor of shape [B, N, N] (float32) or None
        - weights     : dict of model parameters (float32)
        - config      : dict with keys:
            * "dim" (int)          – input channel dimension C
            * "hidden_dim" (int)   – hidden dimension H
            * "nomask" (bool)      – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------- LayerNorm (input) -------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for compute‑intensive part
    x = x.to(torch.float16)

    # --------------------------- cast weights ---------------------------
    # Projection + gate weights (concatenated to save a matmul)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*H, dim)

    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    left_gate = left_gate.sigmoid()                # (B,N,N,H)

    right_proj, right_gate = torch.split(right_combined, hidden_dim, dim=-1)
    right_gate = right_gate.sigmoid()                # (B,N,N,H)

    out_gate = F.linear(x, out_gate_w).sigmoid()    # (B,N,N,H)

    if mask is not None and not nomask:

        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:

        right = right_proj * right_gate

    # Reshape to (B*H, N, N) for torch.bmm

    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_reshaped = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched GEMM: (B*H, N, N) x (B*H, N, N)^T
    out_reshaped = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = out_reshaped.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- second LayerNorm & output gate -----------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------ cast back to input dtype ----------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
