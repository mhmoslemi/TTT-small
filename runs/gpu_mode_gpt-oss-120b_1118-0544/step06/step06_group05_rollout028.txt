def tri_mul_kernel(...):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # compute batch indices
    b = pid_bh // H
    h = pid_bh % H

    # offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds: mask_i = offs_i < N, mask_j = offs_j < N

    # compute pointer for left and right base
    left_base = left_ptr + b*stride_left_batch + h*stride_left_h
    right_base = right_ptr + b*stride_right_batch + h*stride_right_h
    out_base = out_ptr + b*stride_out_batch + h*stride_out_h

    # iterate over K dimension in tiles of size BLOCK_K
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offset range
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load A tile: left[i, k]
        # compute pointer: left_base + offs_i[:, None]*stride_left_i + k_offset[None, :]*stride_left_k
        a_ptrs = left_base + offs_i[:, None]*stride_left_i + k_offset[None, :]*stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load B tile: right[j, k] but we need B^T (k, j). So we load right[j, k] and treat as transposed.
        # B pointer: right_base + offs_j[:, None]*stride_right_j + k_offset[None, :]*stride_right_k
        # but we want to compute dot over k, so we need B values for each j and k.
        # We can load B similarly as a matrix (j x k), then do dot product with a (i x k) and B (j x k) across k.
        # We'll compute a dot b: a[i,k] * b[j,k] (k dimension). So we need to broadcast b across i dimension.
        # We can compute dot product as tl.dot(a, b.T?) Actually tl.dot expects shape (M,K) and (K,N). It does sum over K dimension.
        # Our a is (BLOCK_M, BLOCK_K), b is (BLOCK_N, BLOCK_K) after transposition.
        # So we need to load b as shape (BLOCK_N, BLOCK_K) and then compute tl.dot(a, b.T) -> shape (BLOCK_M, BLOCK_N).
        # But tl.dot only works with (M,K) and (K,N). So we can load b_T as shape (BLOCK_K, BLOCK_N) and pass: tl.dot(a, b_T)
        # To get b_T, we can load b from right_ptr at shape (BLOCK_N, BLOCK_K) and then transpose via .t. Or we load directly with swapped offsets.
        # However easiest: load b (BLOCK_N, BLOCK_K) and then compute tl.dot(a, b.t())
        # But dot expects 2D tensors, not 3D. We can compute using tl.dot for each i, j pair? 
        # In the matmul kernel example, they compute dot product of a (BLOCK_M, BLOCK_K) and b (BLOCK_K, BLOCK_N). 
        # So we need to load b as shape (BLOCK_K, BLOCK_N) directly using pointer offsets: 
        # b_ptrs = right_base + k_offset[:, None]*stride_right_k + offs_j[None, :]*stride_right_j
        # That yields shape (BLOCK_K, BLOCK_N). Then tl.dot(a, b) gives shape (BLOCK_M,BLOCK_N).
        b_ptrs = right_base + k_offset[:, None]*stride_right_k + offs_j[None, :]*stride_right_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # compute dot product
        acc += tl.dot(a, b)  # a: (M,K), b: (K,N) => (M,N)

        # increment pointers if needed (since we compute offsets each iteration from base, no need to inc)

    # store result
    out_ptrs = out_base + offs_i[:, None]*stride_out_i + offs_j[None, :]*stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward implementation with Triton acceleration.

Algorithm
---------
1. Apply LayerNorm over the channel dimension `dim`.
2. Cast to FP16 for compute‑heavy operations.
3. Linear projections (left/right) and gating (left/right/out) are performed
   with FP16 weights.
4. Optional mask is fused by simple broadcasting (mask is 0/1).
5. Core TriMul reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication
   (B*H independent GEMMs of size N×N).  It is implemented as a
   custom Triton kernel that tiles the i, j and k dimensions (BLOCK_M,
   BLOCK_N, BLOCK_K) and accumulates in FP32.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   out‑gate and project back to `dim`.
7. Cast the result back to the original dtype (usually FP32).

The kernel works with arbitrary batch size (≤2), sequence length (≤1024)
and hidden dimension (≤768) without any data copies.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,                     # fp16 * (B, N, N, H)
    right_ptr,                    # fp16 * (B, N, N, H)
    out_ptr,                      # fp16 * (B, N, N, H)
    B, H, N,                      # program constants
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)          # block index over M (i dimension)
    pid_j = tl.program_id(1)          # block index over N (j dimension)
    pid_bh = tl.program_id(2)         # combined batch * hidden index

    # Resolve batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # Compute absolute offsets for i and j
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (b, h) slice
    left_base = left_ptr + b * stride_left_batch + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base = out_ptr + b * stride_out_batch + h * stride_out_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile from left: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load a transposed tile from right: shape (BLOCK_K, BLOCK_N)
        # (k, j) layout so that dot(a, b) performs Σ_k left[i,k]*right[j,k]
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # Matrix multiply‑accumulate
        acc += tl.dot(a, b)   # (M, K) @ (K, N) -> (M, N)

    # Write the result back in FP16
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper to launch the Triton kernel
# ----------------------------------------------------------------------
def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the TriMul reduction using the Triton kernel.
    left, right: tensors of shape [B, N, N, H] and dtype fp16.
    Returns a tensor of the same shape and dtype.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Strides (in elements)
    slb, sli, slk, slh = left.stride()
    srb, srj, srk, srh = right.stride()
    sob, soi, soj, soh = out.stride()

    # Block sizes (tune if needed)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        H,
        N,
        slb,
        sli,
        slk,
        slh,
        srb,
        srj,
        srk,
        srh,
        sob,
        soi,
        soj,
        soh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] (0/1) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, dim] with the same dtype as the input.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim) – still in input dtype

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast all linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections & gating (all FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)       # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask (if any) and combine with gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B, N, N) -> (B, N, N, 1) for broadcasting
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:   # mask present but ignored by config
            mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask_fp16
            right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction via Triton
    # ------------------------------------------------------------------
    out = _tri_mul_triton(left, right)   # (B, N, N, H) fp16

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
