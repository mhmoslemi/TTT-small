@triton.jit
def tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_A_m, stride_A_k,
    stride_Bb, stride_B_k, stride_B_n,
    stride_Cb, stride_C_m, stride_C_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_b = tl.program_id(2)  # batch

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # masks for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to A and B for this batch
    a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_A_m + tl.arange(0, BLOCK_K)[None, :] * stride_A_k

def _tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for compute‑heavy parts.
3. Linear projections (left/right) and gating (sigmoid) are performed with
   torch.nn.functional.linear (FP16).  Mask is fused as an element‑wise
   multiplication after gating.
4. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (A @ Bᵀ) for every hidden slice.
   It is implemented with a tiled Triton kernel (`_tri_mul_matmul_kernel`)
   that processes each (batch, hidden) pair as an independent GEMM.
5. Output LayerNorm, multiplication by the output gate and final linear
   projection back to the original `dim`.
6. Cast the result back to the original dtype (usually float32).

The Triton kernel dramatically reduces the launch overhead of many small
matrix‑multiplications (one per hidden slice) and keeps the data on‑device,
yielding a noticeable speed‑up over a naïve `torch.bmm` implementation.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F
from typing import Tuple, Dict


@triton.jit
def _tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ Bᵀ
    A: (batch, M, K)   – row‑major
    B: (batch, K, N)   – row‑major (already transposed w.r.t. the original
                         right tensor)
    C: (batch, M, N)
    All tensors are FP16; accumulation is FP32.
    """
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_b = tl.program_id(2)          # batch index (batch * hidden)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the block in A and B for this batch
    a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_b * stride_Bb + tl.arange(0, BLOCK_K)[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_valid = k_offset < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_valid[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (fused element‑wise)
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    if mask is not None and not nomask:
        # mask: (B, N, N) -> (B, N, N, 1) for broadcasting
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (A @ Bᵀ) via a Triton kernel.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # Rearrange so that the hidden dimension becomes part of the batch:
    # left: (B, H, N, N) where dims are (batch, hidden, i, k)
    # right: (B, H, N, N) where dims are (batch, hidden, k, j)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    batch_hidden = left_perm.shape[0] * left_perm.shape[1]   # B * H
    left_flat  = left_perm.view(batch_hidden, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(batch_hidden, N, N)  # (B*H, N, N)
    out_flat   = torch.empty_like(left_flat)          # (B*H, N, N)

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # tiles along M (i)
        triton.cdiv(N, BLOCK_N),   # tiles along N (j)
        batch_hidden,              # batch * hidden
    )

    _tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,          # M
        N,          # N
        N,          # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
