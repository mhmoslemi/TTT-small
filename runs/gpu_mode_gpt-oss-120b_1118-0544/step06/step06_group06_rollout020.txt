"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1️⃣ Layer‑normalize the input tensor `x ∈ [B,N,N,C]`.
2️⃣ Cast to fp16 for the heavy part.
3️⃣ Linear projections (left/right) from C → H (hidden_dim) without bias.
4️⃣ Compute three sigmoid gates (left, right, out) from the normalized input.
5️⃣ Fuse mask (if present) with left/right gates in a Triton kernel:
       left  = left  * mask * left_gate
       right = right * mask * right_gate
   The kernel works in‑place on the `left` and `right` tensors to save memory.
6️⃣ Core reduction: for every hidden channel we need
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication:
       (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N)
7️⃣ Reshape back to (B,N,N,H) and apply a second LayerNorm over H.
8️⃣ Multiply by the output gate.
9️⃣ Final linear projection H → C.
🔟 Cast back to the original dtype of the input tensor.

Only the mask‑gate fusion is performed with Triton; the O(N³·H) reduction
leverages cuBLAS via `torch.bmm`, which is optimal on the target B200 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,        # const * left  (B,N,N,H) flattened
    right_ptr,       # const * right (B,N,N,H) flattened
    mask_ptr,        # const * mask  (B,N,N)   flattened, dtype fp16
    left_gate_ptr,   # const * left_gate  (B,N,N,H) flattened
    right_gate_ptr,  # const * right_gate (B,N,N,H) flattened
    left_out_ptr,    # * left_out  (B,N,N,H) flattened (in‑place)
    right_out_ptr,   # * right_out (B,N,N,H) flattened (in‑place)
    H: tl.constexpr,            # hidden dimension
    TOTAL_ELEMS: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask and sigmoid gates:
       left_out  = left  * mask * left_gate
       right_out = right * mask * right_gate
       The kernel works in‑place on `left` and `right`.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL_ELEMS

    # Load values (masked)
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Compute index into the mask (mask repeats for each hidden dimension)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply mask and gates, write back in‑place
    tl.store(left_out_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see code for required keys)
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, if True ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Apply mask and gates
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is None or nomask:
        # No mask – simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate
    else:
        # Prepare mask (fp16) and launch Triton kernel
        mask_fp16 = mask.to(torch.float16)
        total_elements = left.numel()                     # B * N * N * H
        BLOCK = 1024

        # The kernel works in‑place; we reuse `left` and `right` for the outputs
        grid = (triton.cdiv(total_elements, BLOCK),)
        _mask_gate_fusion_kernel[grid](
            left,                     # left_ptr
            right,                    # right_ptr
            mask_fp16.view(-1),       # mask_ptr (flattened)
            left_gate,                # left_gate_ptr
            right_gate,               # right_gate_ptr
            left,                     # left_out_ptr (in‑place)
            right,                    # right_out_ptr (in‑place)
            hidden_dim,               # H constexpr
            total_elements,           # TOTAL_ELEMS constexpr
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )

    # ------------------------------------------------------------------
    # 7) Core reduction: batched GEMM (left @ rightᵀ) per hidden channel
    # ------------------------------------------------------------------
    # Reshape to (B*H, N, N) to use torch.bmm
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication (cuBLAS)
    out_flat = torch.bmm(left_flat, right_flat.transpose(-2, -1))   # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)   # (B, N, N, H)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 8) Second LayerNorm over hidden dimension H (fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10) Final projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 11) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
```
