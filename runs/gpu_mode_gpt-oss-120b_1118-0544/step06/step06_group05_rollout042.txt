    """
    left, right: torch tensors of shape (B, H, N, N) contiguous float16
    Returns out: same shape

    out = torch.empty_like(left)

@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_aa, stride_ai, stride_ak,
    stride_ba, stride_bi, stride_bk,
    stride_ca, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute range offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointer to start of this batch
    a_batch_ptr = a_ptr + pid_batch * stride_aa
    b_batch_ptr = b_ptr + pid_batch * stride_ba
    c_batch_ptr = c_ptr + pid_batch * stride_ca

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        b_ptrs = b_batch_ptr + offs_n[:, None] * stride_bi + offs_k[None, :] * stride_bk

        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_n[:, None] < N) & (offs_k[None, :] < K)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0).to(tl.float32)

        # dot product: (M,K) x (N,K).T -> (M,N)
        acc += tl.dot(a, b, trans_b=True)

    # Write result
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, c, mask=mask_c)
```

def _tri_mul_matmul(left: torch.Tensor, right: torch.Tensor, block_m=128, block_n=128, block_k=32):
    """
    left, right: tensors of shape (B, H, N, N), dtype torch.float16, contiguous.
    Returns out of same shape.
    """
    B, H, N, _ = left.shape
    total = B * H
    # flatten batch dimension
    left_flat = left.view(total, N, N)
    right_flat = right.view(total, N, N)

    out = torch.empty_like(left_flat)

    stride_a_batch = left_flat.stride(0)
    stride_a_i = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_i = right_flat.stride(1)
    stride_b_k = right_flat.stride(2)

    stride_c_j = out.stride(2)

        stride_c_batch, stride_c_i, stride_c_j,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    # reshape back
    out = out.view(B, H, N, N)
    return out

"""
custom_kernel: Triton‑accelerated outgoing TriMul forward (AlphaFold3)

Algorithm
---------
1. Layer‑norm the input over the channel dimension (dim).
2. Cast to FP16 for the heavy compute parts.
3. Linear projections (left/right) and gating projections (left/right/out)
   – all in FP16, gates are passed through sigmoid().
4. Fuse mask, projection and gate for the left/right tensors with a small
   Triton kernel (only executed when a mask is present and not ignored).
5. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication per hidden slice:
   for each (b,h) → out_h = left_h @ right_hᵀ.
   Implemented with a custom Triton GEMM that processes many small
   (N×N) matrices in a single launch (B·H “batches”).
6. Layer‑norm over the hidden dimension, apply the out‑gate and a final
   linear projection back to the original channel size.
7. Cast the result back to the input dtype (typically float32).

The implementation uses two Triton kernels:
* ``_proj_gate_mask_kernel``  – fuses mask, projection and gate.
* ``_tri_mul_matmul_kernel`` – batched GEMM for the O(N³) reduction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Kernel 1: fuse projection, gate and optional mask
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B*N*N*H,)
    right_proj_ptr,  # fp16 * (B*N*N*H,)
    left_gate_ptr,   # fp16 * (B*N*N*H,)
    right_gate_ptr,  # fp16 * (B*N*N*H,)
    mask_ptr,        # fp16 * (B*N*N,)
    left_out_ptr,    # fp16 * (B*N*N*H,)
    right_out_ptr,   # fp16 * (B*N*N*H,)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # guard
    valid = offs < total_elems
    # mask is broadcast over hidden_dim
    mask_idx = offs // hidden_dim

    left_proj  = tl.load(left_proj_ptr + offs,  mask=valid, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    left_gate  = tl.load(left_gate_ptr + offs,  mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    left_out  = left_proj  * left_gate  * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gate and mask for both left and right tensors.
    All tensors must be FP16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)
    Returns:
        left, right : (B, N, N, H) = proj * gate * mask
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_aa, stride_ai, stride_ak,
    stride_ba, stride_bi, stride_bk,
    stride_ca, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # batch * hidden slice
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile col

    # offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers to the start of the current batch
    a_batch = a_ptr + pid_batch * stride_aa
    b_batch = b_ptr + pid_batch * stride_ba
    c_batch = c_ptr + pid_batch * stride_ca

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # A: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        # B: (BLOCK_N, BLOCK_K) – we will transpose inside tl.dot
        b_ptrs = b_batch + offs_n[:, None] * stride_bi + offs_k[None, :] * stride_bk

        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_n[:, None] < N) & (offs_k[None, :] < K)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0).to(tl.float32)

        # dot product with transposed B: (M,K) @ (N,K).T -> (M,N)
        acc += tl.dot(a, b, trans_b=True)

    # write out
    c = acc.to(tl.float16)
    c_ptrs = c_batch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, c, mask=mask_c)


def _tri_mul_matmul(
    left: torch.Tensor,
    right: torch.Tensor,
    block_m: int = 128,
    block_n: int = 128,
    block_k: int = 32,
) -> torch.Tensor:
    """
    Batched matrix multiplication for TriMul:
        out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
    Input tensors must be FP16, contiguous and have shape (B, H, N, N).
    Returns tensor of the same shape and dtype.
    """
    B, H, N, _ = left.shape
    total = B * H

    # flatten batch*hidden → (total, N, N)
    a = left.reshape(total, N, N)
    b = right.reshape(total, N, N)
    c = torch.empty_like(a)

    # strides (elements, not bytes)
    stride_aa = a.stride(0)
    stride_ai = a.stride(1)
    stride_ak = a.stride(2)

    stride_ba = b.stride(0)
    stride_bi = b.stride(1)
    stride_bk = b.stride(2)

    stride_ca = c.stride(0)
    stride_ci = c.stride(1)
    stride_cj = c.stride(2)

    grid = (
        total,
        (N + block_m - 1) // block_m,
        (N + block_n - 1) // block_n,
    )

    _tri_mul_matmul_kernel[grid](
        a,
        b,
        c,
        N, N, N,                     # M, N, K
        stride_aa,
        stride_ai,
        stride_ak,
        stride_ba,
        stride_bi,
        stride_bk,
        stride_ca,
        stride_ci,
        stride_cj,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )

    return c.reshape(B, H, N, N)

    """

    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------

        (dim,),

        bias=weights["norm.bias"],

    # 2) Cast to FP16 for compute‑heavy parts

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
