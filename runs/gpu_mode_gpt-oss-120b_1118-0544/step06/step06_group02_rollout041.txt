```
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_d,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_d,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    bh_id = tl.program_id(0)  # from 0 to B*H-1
    pid_i = tl.program_id(1)  # row block
    pid_j = tl.program_id(2)  # col block

    # compute batch and hidden indices
    b = bh_id // H
    d = bh_id % H

    # compute start indices
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # compute offsets
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    # mask out-of-range
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute d offset in bytes
    d_offset = d * stride_l_d

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile: shape (M, K)
        # left_ptr + b*stride_l_batch + offs_i[:,None] * stride_l_i + offs_k[None,:] * stride_l_k + d_offset
        left_ptrs = left_ptr + b * stride_l_batch + (offs_i[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k) + d_offset
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile transposed: shape (K, N)
        # right_ptr + b*stride_r_batch + (offs_k[:, None] * stride_r_k) + (offs_j[None, :] * stride_r_j) + d_offset
        right_ptrs = right_ptr + b * stride_r_batch + (offs_k[:, None] * stride_r_k) + (offs_j[None, :] * stride_r_j) + d_offset
        right_t = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot
        acc += tl.dot(left, right_t)

    # store result to out

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # 1. layernorm (input)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # 2. cast to fp16
    x = x.to(torch.float16)

    # 3. cast weights to fp16
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(torch.float16)

    # 4. linear projections and gates
    left_proj = torch.nn.functional.linear(x, left_proj_weight)  # shape [bs, N, N, hidden_dim]
    right_proj = torch.nn.functional.linear(x, right_proj_weight)
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()  # shape [bs,N,N,hidden_dim]

    # 5. apply mask (if any) and gate
    if mask is not None and not config.get("nomask", False):
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [bs,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    left = left_proj * left_gate
    right = right_proj * right_gate

    # 6. core reduction using Triton kernel
    # allocate output tensor
    out = torch.empty_like(left)  # shape [bs,N,N,hidden_dim], fp16

    # compute strides
    stride_l_batch, stride_l_i, stride_l_k, stride_l_d = left.stride()
    stride_r_batch, stride_r_j, stride_r_k, stride_r_d = right.stride()
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d = out.stride()
    B = bs
    N = seq_len
    H = hidden_dim

    # set block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H, (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)
    tri_mul_gemm_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_batch, stride_l_i, stride_l_k, stride_l_d,
        stride_r_batch, stride_r_j, stride_r_k, stride_r_d,
        stride_o_batch, stride_o_i, stride_o_j, stride_o_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # 7. second layernorm across hidden_dim
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    # 8. output gate
    out = out * out_gate

    # 9. final linear projection
    out = torch.nn.functional.linear(out, to_out_weight)

    # Cast back to input dtype (float32)
    out = out.to(input_tensor.dtype)

    return out

import triton
import triton.language as tl
import torch

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_d,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_d,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    bh_id = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    # compute b,d
    b = bh_id // H
    d = bh_id % H

    # compute base offsets
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # compute offsets
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute stride offsets (in elements)
    d_off = d * stride_l_d  # same for left,right,out

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # main loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        cur_k = k_base + offs_k
        mask_k = cur_k < N

        # left tile: (M,K)
        left_ptrs = left_ptr + b * stride_l_batch + (offs_i[:, None] * stride_l_i) + (cur_k[None, :] * stride_l_k) + d_off
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile transposed: (K,N) where rows are k and columns are j
        right_ptrs = right_ptr + b * stride_r_batch + (cur_k[:, None] * stride_r_k) + (offs_j[None, :] * stride_r_j) + d_off
        right_t = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product
        acc += tl.dot(left, right_t).to(tl.float32)

    # store result
    out_ptrs = out_ptr + b * stride_o_batch + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_j) + d_off
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass – Triton‑fused core reduction.
Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C).
2. Cast to fp16 and compute four linear projections (left/right and their gates)
   using PyTorch’s fast `linear`.  Apply the optional mask and the sigmoid gates.
3. Compute the O(N³·H) reduction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   with a custom Triton kernel that performs a batched GEMM over the (B·H)
   dimension, eliminating the costly `permute/contiguous` copy used in the
   reference implementation.
4. Apply a second LayerNorm on the hidden dimension, multiply by the output gate,
   and linearly project back to the original channel dimension.
5. Cast the result back to the input dtype (float32).

At least the heavy reduction is executed in Triton, satisfying the requirement
while keeping the overall runtime low.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul reduction
# Computes out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,          # fp16 * (B,N,N,H)
    right_ptr,         # fp16 * (B,N,N,H)
    out_ptr,           # fp16 * (B,N,N,H)
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_d,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_d,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    bh_id = tl.program_id(0)   # 0 .. B*H-1
    pid_i = tl.program_id(1)   # rows
    pid_j = tl.program_id(2)   # cols

    # Decompose bh_id into batch index b and hidden index d
    b = bh_id // H
    d = bh_id % H

    # Compute start offsets for the current tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Offsets within the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Offsets for the hidden dimension (same for all tensors)
    d_off_l = d * stride_l_d
    d_off_r = d * stride_r_d
    d_off_o = d * stride_o_d

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        cur_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Load left tile (M x K)
        left_ptrs = left_ptr \
            + b * stride_l_batch \
            + (offs_i[:, None] * stride_l_i) \
            + (cur_k[None, :] * stride_l_k) \
            + d_off_l
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # Load right tile transposed (K x N)  – we need right[j,k] as (k,j)
        right_ptrs = right_ptr \
            + b * stride_r_batch \
            + (cur_k[:, None] * stride_r_k) \
            + (offs_j[None, :] * stride_r_j) \
            + d_off_r
        right_t = tl.load(right_ptrs,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0)

        # Dot product: (M,K) * (K,N) -> (M,N)
        acc += tl.dot(left, right_t).to(tl.float32)

    # Store the result
    out_ptrs = out_ptr \
        + b * stride_o_batch \
        + (offs_i[:, None] * stride_o_i) \
        + (offs_j[None, :] * stride_o_j) \
        + d_off_o
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, C]
        - mask        : torch.Tensor of shape [B, N, N] or None
        - weights     : dict of model parameters (float32)
        - config      : dict with keys:
            * "dim" (int)         – input channel dimension C
            * "hidden_dim" (int)  – hidden dimension H
            * "nomask" (bool)     – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------- LayerNorm (input) -------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for compute‑heavy part
    x = x.to(torch.float16)

    # --------------------------- cast weights ---------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ----------------------- linear projections -------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,H]
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------- gates ----------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # [B,N,N,H]

    # ------------------- mask handling (optional) -----------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # Apply elementwise gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ----------------- core O(N³·H) reduction via Triton -----------------
    # Allocate output tensor
    out = torch.empty_like(left)   # [B,N,N,H], fp16

    # Strides (in number of elements)
    stride_l_batch, stride_l_i, stride_l_k, stride_l_d = left.stride()
    stride_r_batch, stride_r_j, stride_r_k, stride_r_d = right.stride()
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d = out.stride()

    B = input_tensor.shape[0]
    N = input_tensor.shape[1]
    H = hidden_dim

    # Tile sizes – chosen to fit in shared memory and use Tensor Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_gemm_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_batch, stride_l_i, stride_l_k, stride_l_d,
        stride_r_batch, stride_r_j, stride_r_k, stride_r_d,
        stride_o_batch, stride_o_i, stride_o_j, stride_o_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]

    # ------------------------ cast back to input dtype ----------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
