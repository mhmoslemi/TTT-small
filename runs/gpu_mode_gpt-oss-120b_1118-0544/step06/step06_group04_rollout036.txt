    left_ptr, right_ptr, out_ptr,
    batch, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,

    # block_i = program_id(1) -> block in rows
    # block_j = program_id(2) -> block in cols

    pid1 = tl.program_id(1) # block row
    pid2 = tl.program_id(2) # block col

    # compute global batch index b = pid0 // H, hidden idx h = pid0 % H
    b = pid0 // H
    h = pid0 % H

    # Offsets within the matrix
    i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for i,j within N
    mask_i = i < N
    mask_j = j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offsets
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K) # shape (BLOCK_K)
        mask_k = k_offset < N

        # Load left: shape (BLOCK_M, BLOCK_K)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Get weights (tensors)
    # Convert to appropriate dtype (float16 for compute)
    # weight treats as linear: weight shape [hidden_dim, dim] (for linear with bias=False)
    # For layernorm we use weight shape [dim] etc.

    # Input LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Cast to fp16
    x = x.to(torch.float16)

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    # Apply mask and gates
    if mask is not None and not nomask:
        mask = mask.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
        # Broadcast mask
        left = left * left_gate * mask
        right = right * right_gate * mask
    else:
        left = left * left_gate
        right = right * right_gate

    # Move to (B, H, N, N) for kernel
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    # Ensure contiguity and dtype half
    # For kernel stride info
    B, H, N, _ = left.shape

    # Allocate output
    out = torch.empty_like(left)

    # Get strides (in elements)
    # left: (B, H, N, N)
    stride_left_b, stride_left_h, stride_left_i, stride_left_k = left.stride()
    stride_right_b, stride_right_h, stride_right_j, stride_right_k = right.stride()
    stride_out_b, stride_out_h, stride_out_i, stride_out_j = out.stride()

    # Launch kernel
    # Grid dims: (B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (B * H, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Now out shape [B, H, N, N] (fp16)
    # Permute back to [B,N,N,H]
    out = out.permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm (hidden_dim) and gating
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out * out_gate

    # Final linear projection
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast back to input dtype
    out = out.to(dtype_in)

    return out

    B, N, H,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_j, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Derive batch and hidden indexes
    b = pid_batch_h // H
    h = pid_batch_h % H

    # Coordinates in the output matrix
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Load left block (i, k)
        left_ptrs = left_ptr + \
                    b * stride_l_b + h * stride_l_h + \
                    i[:, None] * stride_l_i + k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right block (j, k)
        right_ptrs = right_ptr + \
                     b * stride_r_b + h * stride_r_h + \
                     j[:, None] * stride_r_j + k[None, :] * stride_r_k

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. LayerNorm on the input (float32) and cast to float16 for the bulk of the
   computation.
2. Bias‑less linear projections for “left”, “right” and the three gates, all in
   FP16.
3. Apply the optional mask and the left/right gates with ordinary PyTorch
   element‑wise ops.
4. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (L @ Rᵀ) for every hidden channel.
   It is implemented in a custom Triton kernel that processes one (batch,hidden)
   matrix per program‑id and tiles the i, j and k dimensions.
5. Output LayerNorm over the hidden dimension, multiply by the output gate and
   a final linear projection back to the original feature dimension.
6. Cast the result back to the dtype of the input tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # runtime scalars
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_j, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute for each (b, h):
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    The tensors are laid out as (B, H, N, N) after a permute.
    """
    pid_batch_h = tl.program_id(0)   # combined batch * hidden index
    pid_i = tl.program_id(1)         # tile row
    pid_j = tl.program_id(2)         # tile column

    # decompose batch and hidden indices
    b = pid_batch_h // H
    h = pid_batch_h % H

    # tile start indices
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator in FP32 for higher precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left block: (i, k)
        left_off = (
            b * stride_l_b
            + h * stride_l_h
            + i[:, None] * stride_l_i
            + k[None, :] * stride_l_k
        )
        left = tl.load(left_ptr + left_off,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right block: (j, k)
        right_off = (
            b * stride_r_b
            + h * stride_r_h
            + j[:, None] * stride_r_j
            + k[None, :] * stride_r_k
        )
        right = tl.load(right_ptr + right_off,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0)

        # cast to FP32, accumulate
        left_f32 = tl.cast(left, tl.float32)
        right_f32 = tl.cast(right, tl.float32)
        acc += tl.dot(left_f32, right_f32)

    # write back the result (cast to FP16)
    out_off = (
        b * stride_o_b
        + h * stride_o_h
        + i[:, None] * stride_o_i
        + j[None, :] * stride_o_j
    )
    tl.store(out_ptr + out_off,
             tl.cast(acc, tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32) and cast to FP16
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections (bias‑less) and gate values
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(device).to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(device).to(torch.float16))

    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(device).to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(device).to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(device).to(torch.float16))
    )

    # -----------------------------------------------------------------
    # 3. Apply mask (if any) and gates
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * left_gate * mask
        right = right * right_gate * mask
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 4. Core contraction via custom Triton kernel
    #    left/right: (B, N, N, H) -> (B, H, N, N)
    # -----------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left.shape

    out = torch.empty_like(left)

    # strides in element units
    stride_l_b, stride_l_h, stride_l_i, stride_l_k = left.stride()
    stride_r_b, stride_r_h, stride_r_j, stride_r_k = right.stride()
    stride_o_b, stride_o_h, stride_o_i, stride_o_j = out.stride()

    # tiling parameters (tuned for typical sequence lengths)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_h,
        stride_l_i,
        stride_l_k,
        stride_r_b,
        stride_r_h,
        stride_r_j,
        stride_r_k,
        stride_o_b,
        stride_o_h,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back to original layout (B, N, N, H)
    out = out.permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5. Output LayerNorm, gate, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device).to(torch.float16),
        bias=weights["to_out_norm.bias"].to(device).to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(device).to(torch.float16),
    )

    # -----------------------------------------------------------------
    # 6. Cast back to original dtype
    # -----------------------------------------------------------------
    out = out.to(dtype_in)
    return out
