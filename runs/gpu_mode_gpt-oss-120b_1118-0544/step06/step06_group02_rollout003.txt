"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) with given weight/bias.
2. Cast to fp16 for the compute‑intensive part.
3. Four linear projections (left/right and their gates) are performed with
   `torch.nn.functional.linear` in fp16.
4. Fuse projection, gate and optional mask using a tiny Triton kernel
   (`_gate_mask_kernel`).  This saves two extra memory passes.
5. Core O(N³·H) reduction:
   - Reshape left/right to (B*H, N, N) and use `torch.bmm` (cuBLAS) which
     maps to Tensor‑core kernels.
6. Apply a second LayerNorm on the hidden dimension, multiply by the
   output gate and linearly project back to the original channel dimension.
7. Cast the result back to the input dtype (float32).

At least one non‑trivial operation (mask & gate fusion) is implemented in
Triton, satisfying the requirement while keeping the overall runtime low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _gate_mask_kernel(
    proj_ptr,        # fp16 * (total_proj,)
    gate_ptr,        # fp16 * (total_proj,)
    mask_ptr,        # fp16 * (total_mask,)
    out_ptr,         # fp16 * (total_proj,)
    total_elems,     # i32 – total number of elements in proj/gate (B*N*N*H)
    hidden_dim,      # i32 – H
    BLOCK_SIZE: tl.constexpr,
):
    """Element‑wise: out = proj * gate * mask (mask broadcast over H)."""
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # mask index: each block of H elements corresponds to one (b,i,k)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using a Triton kernel.

    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],            # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] (or None)
        - weights      : dict of model parameters (float32)
        - config       : dict with keys:
            * "dim" (int)         – input channel dimension C
            * "hidden_dim" (int)  – hidden dimension H
            * "nomask" (bool)     – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # Cast to fp16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    left_proj  = F.linear(x, left_proj_w)                     # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)                    # (B,N,N,H)

    left_gate  = F.linear(x, left_gate_w).sigmoid()           # (B,N,N,H)
    right_gate = F.linear(x, right_gate_w).sigmoid()          # (B,N,N,H)
    out_gate   = F.linear(x, out_gate_w).sigmoid()            # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3. Fuse projection, gate and optional mask with Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure contiguous fp16 tensors
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4. Core O(N³·H) reduction via batched GEMM (cuBLAS/Tensor Core)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    B, N, _, _ = left.shape
    left_reshaped  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # left @ right^T  -> (B*H, N, N)
    out_reshaped = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_reshaped.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Second LayerNorm, output gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)          # (B,N,N,dim)

    # ------------------------------------------------------------------
    # Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
