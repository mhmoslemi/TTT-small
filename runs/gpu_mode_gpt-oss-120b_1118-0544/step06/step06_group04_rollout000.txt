"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Input LayerNorm (float32) → cast to float16.
2. Linear projections (bias‑less) to hidden dimension H:
       left  = X @ W_left
       right = X @ W_right
3. Compute gates (sigmoid of linear projections) and optionally apply a mask.
   A Triton kernel fuses the element‑wise multiplications:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
   The mask is broadcast over the hidden dimension.
4. Core contraction (outgoing update):
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM: for each hidden channel d,
   out[b,:,:,d] = left[b,:,:,d] @ right[b,:,:,d].T .
   We reshape to (B*H, N, N) and use torch.bmm (cuBLAS).
5. Output LayerNorm over hidden dimension H, multiply by out_gate,
   and a final linear projection back to original dimension C.
6. Cast back to the original input dtype and return.

The only required Triton kernel is the mask‑gate fusion; all other heavy
operations rely on highly‑optimised PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if any) and gate multiplications in‑place
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fusion_kernel(
    left_ptr, right_ptr,           # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr, # same shape
    mask_ptr,                      # [B,N,N] flattened (broadcasted)
    total_elems, hidden_dim,       # runtime scalars
    HAS_MASK: tl.constexpr,        # compile‑time bool
    BLOCK_SIZE: tl.constexpr,      # compile‑time int
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask?
        right = right * right_gate * mask?
    `mask` is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load values
    left      = tl.load(left_ptr + offs,      mask=active, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr + offs,mask=active, other=0.0)

    if HAS_MASK:
        # Compute corresponding mask index (mask is not repeated over H)
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs,  left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # Config
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → fp16 for compute
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑less) to hidden dimension
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left  = F.linear(x, left_proj_w)      # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)     # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3. Fuse mask and gates via Triton (in‑place)
    # ------------------------------------------------------------------
    # Prepare contiguous flattened tensors for the kernel
    left_flat      = left.contiguous().view(-1)
    right_flat     = right.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_gate_flat= right_gate.contiguous().view(-1)

    total = left_flat.numel()                     # B * N * N * H
    BLOCK = 1024                                   # reasonable block size
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        mask_flat = mask_fp.view(-1)               # (B*N*N,)
        HAS_MASK = True
    else:
        # dummy mask (won't be read)
        mask_flat = left_flat      # any tensor with enough size
        HAS_MASK = False

    _gate_mask_fusion_kernel[grid](
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        total,
        hidden_dim,
        HAS_MASK=HAS_MASK,
        BLOCK_SIZE=BLOCK,
    )

    # Reshape back to original layout
    left = left_flat.view_as(left)
    right = right_flat.view_as(right)

    # ------------------------------------------------------------------
    # 4. Core contraction: batched matrix multiplication per hidden channel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, H, N, N) → (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # out_mat[b*h] = left_mat[b*h] @ right_mat[b*h].T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Restore (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate                     # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)            # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------
    # 6. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
