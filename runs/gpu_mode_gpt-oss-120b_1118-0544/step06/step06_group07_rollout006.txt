"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm (float32) the [B,N,N,C] input.
2. Cast to fp16 for the compute‑heavy path.
3. Linear projections (bias‑less) → left, right, left_gate, right_gate, out_gate.
4. Fuse mask (if present) and the left/right gates in a single Triton kernel,
   writing the gated tensors back in‑place.
5. Core contraction: for each hidden dimension d,
       out[...,i,j,d] = Σ_k left[...,i,k,d] * right[...,j,k,d],
   which is a batched GEMM (left @ rightᵀ) over B·H matrices.
6. Layer‑norm over the hidden dimension, multiply by out_gate, and final linear
   projection back to the original dimension.
7. Cast back to the original dtype and return.

Only the gating/mask fusion is performed in Triton; the rest uses highly‑
optimized PyTorch operators.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr,               # *fp16  [B,N,N,H] flattened
    right_ptr,              # *fp16  same shape
    left_gate_ptr,          # *fp16  same shape
    right_gate_ptr,         # *fp16  same shape
    mask_ptr,               # *fp16  [B,N,N] flattened (optional)
    total_elems,            # i64    total number of elements in left/right
    hidden_dim,             # i64    size of H dimension
    HAS_MASK: tl.constexpr, # bool   compile‑time flag
    BLOCK_SIZE: tl.constexpr, # i32   block size
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    If HAS_MASK=False the mask term is omitted (treated as 1.0).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load data
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # Each mask element corresponds to hidden_dim consecutive values
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * lgate * m
        right = right * rgate * m
    else:
        left = left * lgate
        right = right * rgate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # --------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy path
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (bias‑less) in fp16
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device, dtype=torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device, dtype=torch.float16)

    left = F.linear(x, left_proj_w)                     # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)                   # (B,N,N,hidden_dim)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B,N,N,hidden_dim)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))     # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 4. Fuse mask and gates via a Triton kernel (in‑place)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure contiguous tensors
        mask_fp = mask.to(device, dtype=torch.float16).contiguous()
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        # Flatten to 1‑D for the kernel
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_fp.view(-1)   # (B*N*N,)

        total = left_flat.shape[0]
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – element‑wise fuse gates directly (fast torch path)
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction via batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiply (uses cuBLAS/Tensor Cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Restore original layout: (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gate, final linear projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate   # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # --------------------------------------------------------------
    # 7. Cast back to the input dtype and return
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
