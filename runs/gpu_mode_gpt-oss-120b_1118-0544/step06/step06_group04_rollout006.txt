        right = right_flat.view_as(right)

    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=w_to_out_norm, bias=b_to_out_norm)

    out = torch.nn.functional.linear(out, w_to_out)  # (B,N,N,dim)

    # Cast back to original dtype (float32)
    out = out.to(dtype)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Input LayerNorm (FP32) → cast to FP16.
2. Linear projections (bias‑free) to hidden dimension H:
       left  = x @ W_left
       right = x @ W_right
   Gates (sigmoid) are computed from separate linear layers.
3. Fuse element‑wise mask, left‑gate and right‑gate with a small Triton kernel.
   The kernel works on the flattened tensors and writes the gated values back
   in‑place, avoiding extra passes.
4. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication L @ Rᵀ for each hidden channel.
   We reshape to (B·H, N, N) and use torch.bmm (cuBLAS tensor‑cores).
5. Output LayerNorm over the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original dimension.
6. Cast back to the input dtype (usually FP32) and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate.
# -------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,                # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,      # same shape
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,            # runtime scalars
    HAS_MASK: tl.constexpr,             # compile‑time bool
    BLOCK_SIZE: tl.constexpr,           # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # load tensors
    l  = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r  = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden slice shares the same mask entry
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        l = l * lg * m
        r = r * rg * m
    else:
        l = l * lg
        r = r * rg

    tl.store(left_ptr + offs,  l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # -----------------------------------------------------------------
    # Configuration & weight loading (avoid extra copies)
    # -----------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm params (float32)
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)

    # Linear weights – cast once to FP16 for compute
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32) → FP16
    # -----------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections & gates (bias‑less)
    # -----------------------------------------------------------------
    left  = F.linear(x, left_proj_w)      # (B,N,N,H)
    right = F.linear(x, right_proj_w)

    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # -----------------------------------------------------------------
    # 3. Fuse mask and gates (Triton) or simple elementwise multiply
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16 and contiguous
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # Flatten everything for the kernel
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left  = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask → simple element‑wise fusion (highly optimized in PyTorch)
        left  = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 4. Core O(N³·H) contraction via batched GEMM (cuBLAS)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, H, N, N) → (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)               # (B,N,N,dim) in FP16

    # -----------------------------------------------------------------
    # 6. Cast back to original dtype (usually FP32) and return
    # -----------------------------------------------------------------
    out = out.to(dtype_in)
    return out
