"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. **Input LayerNorm** (float32) – stabilises the input.
2. Cast to **fp16** for the heavy compute path.
3. **Linear projections** (left, right) and **gate** projections (left_gate,
   right_gate, out_gate) – all in fp16, gates are passed through a sigmoid.
4. **Fuse mask and gates** with a tiny Triton kernel that updates `left`
   and `right` in‑place:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
   (mask is broadcast over the hidden dimension).
5. **Core contraction** – compute
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched matrix multiplication (B·H, N, N) using
   `torch.bmm`, which leverages cuBLAS tensor‑cores on the B200.
6. **Output LayerNorm**, element‑wise multiplication by `out_gate`,
   and final linear projection back to the original dimension.
7. Cast the result back to the input dtype.

Only the mask‑gate fusion is performed with Triton; the rest uses
high‑performance PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr,               # *float16* left tensor   (B,N,N,H) flattened
    right_ptr,              # *float16* right tensor  (B,N,N,H) flattened
    left_gate_ptr,          # *float16* left gate     (B,N,N,H) flattened
    right_gate_ptr,         # *float16* right gate    (B,N,N,H) flattened
    mask_ptr,               # *float16* mask          (B,N,N)   flattened
    total_elems,            # total number of elements in left/right (B*N*N*H)
    hidden_dim,             # H
    HAS_MASK: tl.constexpr, # compile‑time bool
    BLOCK_SIZE: tl.constexpr, # compile‑time int
):
    """In‑place fusion of mask and gates:
           left  = left  * left_gate  * mask
           right = right * right_gate * mask
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load the values
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden slice shares the same mask entry
        mask_idx = offs // hidden_dim               # index in [B,N,N]
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape (B, N, N, C)
        - mask         : torch.Tensor of shape (B, N, N) or None
        - weights      : dict of model parameters (float32 tensors)
        - config       : dict with keys:
            * "dim"         – input channel dimension C
            * "hidden_dim"  – hidden dimension H
            * "nomask" (optional) – if True, mask is ignored

    Returns
    -------
    torch.Tensor
        Output tensor of shape (B, N, N, C) with the same dtype as the input.
    """
    # --------------------------------------------------------------------- #
    # 0. Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------- #
    # 1. Input LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------------- #
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------------- #
    # 2. Cast to fp16 for compute‑heavy path
    # --------------------------------------------------------------------- #
    x = x.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 3. Linear projections and gate computations (fp16, no bias)
    # --------------------------------------------------------------------- #
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # --------------------------------------------------------------------- #
    # 4. Fuse mask and gates (if mask is present and not disabled)
    # --------------------------------------------------------------------- #
    if mask is not None and not nomask:
        # Convert mask to fp16 and make it contiguous
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # Flatten everything for the kernel
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back to original layout
        left  = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – simple element‑wise fusion
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented via batched GEMM (B*H, N, N)
    # --------------------------------------------------------------------- #
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched matmul
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (uses cuBLAS tensor‑cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 6. Output LayerNorm, gate, and final linear projection
    # --------------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, C) in fp16

    # --------------------------------------------------------------------- #
    # 7. Cast back to original dtype and return
    # --------------------------------------------------------------------- #
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
