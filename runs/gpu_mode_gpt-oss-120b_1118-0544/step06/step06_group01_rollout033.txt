@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B_H, M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds in M and N
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offsets
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + pid_batch * stride_left_batch \
                 + offs_m[:, None] * stride_left_m \
                 + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + pid_batch * stride_right_batch \
                 + offs_k[:, None] * stride_right_k \
                 + offs_n[None, :] * stride_right_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # write back
    c = acc.to(tl.float16)
    out_ptrs = out_ptr + pid_batch * stride_out_batch \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
tri_mul (outgoing) – Fast Triton implementation
------------------------------------------------
Implements the outgoing Triangle Multiplicative Update (TriMul) from AlphaFold3.
The forward pass consists of:

1. LayerNorm on the input (float32 for stability).
2. Linear projections to a hidden dimension (fp16) together with sigmoid gates.
3. Optional mask application (broadcast over the hidden dimension).
4. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   realised as a batched GEMM (left @ rightᵀ) using a custom Triton kernel.
5. Output LayerNorm, multiplication with the output gate and a final linear
   projection back to the original feature dimension.
All heavy arithmetic is performed in fp16; the final result is cast back to the
input dtype (usually float32).
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication:
       out[b, i, j] = Σ_k left[b, i, k] * right[b, k, j]
       (b corresponds to batch*hidden_dim in our use‑case)
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + pid_batch * stride_left_batch \
                 + offs_m[:, None] * stride_left_m \
                 + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B tile: (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + pid_batch * stride_right_batch \
                 + offs_k[:, None] * stride_right_k \
                 + offs_n[None, :] * stride_right_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    c = acc.to(tl.float16)
    out_ptrs = out_ptr + pid_batch * stride_out_batch \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (torch.Tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy parts
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # -----------------------------------------------------------------
    # 4. Linear projections and gates (fp16)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_w)            # (B, N, N, H)
    right = F.linear(x, right_proj_w)          # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 5. Fuse mask and gate multiplication (elementwise)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * left_gate * mask_fp
        right = right * right_gate * mask_fp
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Core contraction using a Triton batched GEMM
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Prepare tensors for the kernel: (B*H, N, N) and (B*H, N, N) where the
    # second operand is already transposed on the last two dimensions.
    left_t = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N) -> (i,k)
    right_t = right.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N) -> (k,j)

    left_flat = left_t.view(B * H, N, N)
    right_flat = right_t.view(B * H, N, N)
    out_flat = torch.empty_like(left_flat)

    # Strides for the kernel (batch, row, col)
    stride_l_batch, stride_l_m, stride_l_k = left_flat.stride()
    stride_r_batch, stride_r_k, stride_r_n = right_flat.stride()
    stride_o_batch, stride_o_m, stride_o_n = out_flat.stride()

    # Block sizes – tuned for typical GPUs (must be multiples of 32)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        N,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_k,
        stride_r_n,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)                     # (B, N, N, C)

    # Cast back to the original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
