def triton_tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                         B, N, H,
                         stride_batch, stride_m, stride_k,
                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # compute
    pid = tl.program_id(0)
    total_batch = B * H
    num_tiles_M = tl.cdiv(N, BLOCK_M)
    num_tiles_N = tl.cdiv(N, BLOCK_N)
    batch_idx = pid // (num_tiles_M * num_tiles_N)
    tile_idx = pid % (num_tiles_M * num_tiles_N)
    tile_m = tile_idx // num_tiles_N
    tile_n = tile_idx % num_tiles_N

    # compute offsets
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Load mask for out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + batch_idx * stride_batch + offs_m[:, None] * stride_m + offs_k[None, :] * stride_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    out_ptrs = out_ptr + batch_idx * stride_batch + offs_m[:, None] * stride_m + offs_n[None, :] * stride_k

    dtype_in = input_tensor.dtype

    nomask = config.get("nomask", False)

    # Norm
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)

    # Linear proj weights (out_features, in_features)
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)

    # 1. LayerNorm over dim
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # cast to FP16
    x = x.to(torch.float16)

    # 2. Linear projections
    left = F.linear(x, left_proj_w)  # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    if mask is not None and not nomask:

        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)

        mask_flat = mask_fp.contiguous().view(-1)  # shape [B*N*N]

            BLOCK_SIZE=BLOCK,

        right = right_flat.view_as(right)

        right = right * right_gate

    right_perm = right.permute(0, 3, 1, 2).contiguous()

    right_mat = right_perm.view(bs * hidden_dim, N, N)

    BLOCK_N = 64
    BLOCK_K = 64

    if out.dtype != dtype_in:

@triton.jit
def _tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                    B, N, H,
                    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    total_batches = B * H
    # number of tiles in each dimension
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    batch_idx = pid // (num_tiles_m * num_tiles_n)
    tile_idx = pid % (num_tiles_m * num_tiles_n)
    tile_m = tile_idx // num_tiles_n
    tile_n = tile_idx % num_tiles_n

    # compute offsets
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # strides for contiguous (B*H, N, N)
    stride_batch = N * N
    stride_m = N
    stride_k = 1

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block (M x K)
        left_ptrs = left_ptr + batch_idx * stride_batch + offs_m[:, None] * stride_m + offs_k[None, :] * stride_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right block transposed (K x N)
        # we need right[batch, k, n] -> use k as row, n as column
        right_ptrs = right_ptr + batch_idx * stride_batch + offs_k[:, None] * stride_m + offs_n[None, :] * stride_k
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # write back
    out_ptrs = out_ptr + batch_idx * stride_batch + offs_m[:, None] * stride_m + offs_n[None, :] * stride_k
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input (float32 for stability).
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (left/right) and three sigmoid gates (left/right/out).
4. Fuse mask and the two element‑wise gates into the left/right tensors with a
   small Triton kernel (in‑place).
5. Core O(N³·H) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   performed as a batched GEMM (left @ rightᵀ) using a custom Triton kernel.
6. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate, and a final linear projection back to the original dimension.
7. Cast back to the input dtype and return.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,            # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,  # same shape
    mask_ptr,                       # [B,N,N] flattened
    total_elems, hidden_dim,        # runtime scalars
    HAS_MASK: tl.constexpr,         # compile‑time bool
    BLOCK_SIZE: tl.constexpr,       # compile‑time int
):
    """In‑place fusion:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
       mask is broadcast over the hidden dimension H.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each H‑wide slice shares the same mask entry
        mask_idx = offs // hidden_dim                # index in [B,N,N]
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where left/right are stored as (B*H, N, N) contiguous tensors.
    The kernel processes one output block (BLOCK_M x BLOCK_N) for a given
    (batch * hidden) slice.
    """
    pid = tl.program_id(0)
    total_batches = B * H

    # number of tiles in M and N dimensions
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)

    batch_idx = pid // (num_tiles_m * num_tiles_n)
    tile_idx = pid % (num_tiles_m * num_tiles_n)
    tile_m = tile_idx // num_tiles_n
    tile_n = tile_idx % num_tiles_n

    # offsets within the current tile
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # strides for a contiguous (B*H, N, N) tensor
    stride_batch = N * N
    stride_m = N
    stride_k = 1

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + batch_idx * stride_batch \
                    + offs_m[:, None] * stride_m \
                    + offs_k[None, :] * stride_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right block (transposed): shape (BLOCK_K, BLOCK_N)
        # we read right[k, n] which corresponds to original right[n, k]
        right_ptrs = right_ptr + batch_idx * stride_batch \
                     + offs_k[:, None] * stride_m \
                     + offs_n[None, :] * stride_k
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # store the result back (FP16)
    out_ptrs = out_ptr + batch_idx * stride_batch \
               + offs_m[:, None] * stride_m \
               + offs_n[None, :] * stride_k
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int)         – input channel dimension C
                * "hidden_dim" (int)  – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------ #
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (C,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------ #
    # 2. Cast to fp16 for compute‑heavy part
    # ------------------------------------------------------------------ #
    x = x.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 3. Linear projections (no bias) and gates (sigmoid)
    # ------------------------------------------------------------------ #
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)      # (B,N,N,H)
    right = F.linear(x, right_proj_w)    # (B,N,N,H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,H)

    # ------------------------------------------------------------------ #
    # 4. Fuse mask and the two element‑wise gates (in‑place) using Triton
    # ------------------------------------------------------------------ #
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten tensors
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (math.ceil(total / meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            H,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM (left @ rightᵀ) with a Triton kernel.
    # ------------------------------------------------------------------ #
    # reshape to (B*H, N, N) for the kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    out_mat = torch.empty_like(left_mat)  # fp16 output buffer

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B * H * num_tiles_m * num_tiles_n
    grid = (total_blocks,)

    _tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        B,
        N,
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 6. Output LayerNorm, output gate, and final linear projection
    # ------------------------------------------------------------------ #
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate                         # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)                # (B,N,N,C) in fp16

    # ------------------------------------------------------------------ #
    # 7. Cast back to original dtype
    # ------------------------------------------------------------------ #
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
