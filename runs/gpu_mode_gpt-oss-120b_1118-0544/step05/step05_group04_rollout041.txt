def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype
    B,N,N,dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
                                      bias=weights["norm.bias"].to(device=device, dtype=torch.float32))
    # Cast to fp16
    x = x.to(torch.float16)

    # Linear projections
    left_proj = F.linear(x, weights["left_proj.weight"].to(device=device, dtype=torch.float16))
    right_proj = F.linear(x, weights["right_proj.weight"].to(device=device, dtype=torch.float16))

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(device=device, dtype=torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(device=device, dtype=torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device=device, dtype=torch.float16)))

    # Prepare mask
    if mask is not None and not config.get("nomask", False):
        mask_fp = mask.to(torch.float16)
    else:
        mask_fp = torch.ones((B,N,N), dtype=torch.float16, device=device)
    # Flatten mask for kernel
    mask_flat = mask_fp.view(-1)

    # Allocate output tensors for gated left/right
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    total_elements = left_proj.numel()
    BLOCK = 1024  # adjust

    grid = (triton.cdiv(total_elements, BLOCK),)
    _apply_mask_gate_kernel[grid](
        left_proj, right_proj,
        mask_flat,
        left_gate, right_gate,
        left_out, right_out,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4
    )

    # Batched GEMM
    # Permute/reshape
    left_flat = left_out.permute(0,3,1,2).contiguous().view(B*hidden_dim, N, N)
    right_flat = right_out.permute(0,3,1,2).contiguous().view(B*hidden_dim, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Strides
    a_batch_stride, a_row_stride, a_col_stride = left_flat.stride()
    b_batch_stride, b_row_stride, b_col_stride = right_flat.stride()
    c_batch_stride, c_row_stride, c_col_stride = out_flat.stride()

    # For transposed B: use row stride as col stride (since we want B^T)
    # Actually b as original shape (batch, N, N). For transpose, we treat row stride = b_col_stride, col stride = b_row_stride.
    stride_b_k = b_col_stride   # stride for k dimension (original col)
    stride_b_n = b_row_stride   # stride for n dimension (original row)

    # Launch kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_N), triton.cdiv(N, BLOCK_M), B*hidden_dim)
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B*hidden_dim, N, N, N,
        a_batch_stride, a_row_stride, a_col_stride,
        b_batch_stride, stride_b_k, stride_b_n,
        c_batch_stride, c_row_stride, c_col_stride,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,H]

    # Output layernorm
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"].to(device=device, dtype=torch.float16),
                       bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float16))

    # Apply out_gate
    out = out * out_gate

    # Final linear
    out = F.linear(out, weights["to_out.weight"].to(device=device, dtype=torch.float16))

    # Cast to original dtype
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
```

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_n = tl.program_id(0)  # block column
    pid_m = tl.program_id(1)  # block row
    pid_b = tl.program_id(2)  # batch

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute batch offset
    batch_offset_a = pid_b * stride_a_batch
    batch_offset_b = pid_b * stride_b_batch
    batch_offset_c = pid_b * stride_c_batch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # K range
        k_range = tl.arange(0, BLOCK_K) + k_offset

        # Load A tile
        a_ptrs = a_ptr + batch_offset_a + (offs_m[:, None] * stride_a_m) + (k_range[None, :] * stride_a_k)
        mask_a = (offs_m[:, None] < M) & (k_range[None, :] < K)

        # Load B tile (transposed)
        b_ptrs = b_ptr + batch_offset_b + (k_range[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        mask_b = (k_range[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

    c_ptrs = c_ptr + batch_offset_c + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    mask_c = mask_m[:, None] & mask_n[None, :]

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the reference PyTorch module but moves the
heavy O(N³·H) reduction into a custom Triton kernel.  The steps are:

1. Layer‑normalize the input.
2. Linear projections to a hidden dimension H (left/right) and three
   sigmoid gates (left, right, out).
3. Fuse mask, left‑gate and right‑gate with a tiny Triton kernel.
4. Perform the core reduction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a batched GEMM kernel written in Triton (no cuBLAS).
5. Apply a second LayerNorm over H, multiply by the output gate and
   project back to the original dimension.
All heavy arithmetic is done in fp16; the final result is cast back to
the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask and left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,            # const fp16* left,   shape [B,N,N,H] flattened
    right_ptr,           # const fp16* right,  same layout
    mask_ptr,            # const fp16* mask,   shape [B,N,N] flattened
    left_gate_ptr,       # const fp16* left_gate,  same layout as left
    right_gate_ptr,      # const fp16* right_gate, same layout as right
    left_out_ptr,        # fp16* left_out
    right_out_ptr,       # fp16* right_out
    H: tl.constexpr,     # hidden dimension
    TOTAL_ELEMS: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,   # thread‑block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL_ELEMS

    # load raw tensors
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # mask is broadcast over the hidden dimension
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(left_out_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the TriMul reduction (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr,               # fp16*   A, shape [BATCH, M, K] (row‑major)
    b_ptr,               # fp16*   B, shape [BATCH, K, N] (row‑major, we will treat it as transposed)
    c_ptr,               # fp16*   C, shape [BATCH, M, N]
    BATCH: tl.constexpr,
    M: tl.constexpr,
    N: tl.constexpr,
    K: tl.constexpr,
    stride_a_batch: tl.constexpr,
    stride_a_m: tl.constexpr,
    stride_a_k: tl.constexpr,
    stride_b_batch: tl.constexpr,
    stride_b_k: tl.constexpr,   # stride for the K dimension (original column)
    stride_b_n: tl.constexpr,   # stride for the N dimension (original row)
    stride_c_batch: tl.constexpr,
    stride_c_m: tl.constexpr,
    stride_c_n: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_n = tl.program_id(0)   # column block
    pid_m = tl.program_id(1)   # row block
    pid_b = tl.program_id(2)   # batch index

    # ------------------------------------------------------------------
    # offsets for this block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # batch offsets
    a_batch_off = pid_b * stride_a_batch
    b_batch_off = pid_b * stride_b_batch
    c_batch_off = pid_b * stride_c_batch

    # accumulator in fp32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # main K-loop
    # ------------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        cur_k = k
        k_range = tl.arange(0, BLOCK_K) + cur_k

        # A tile: [BLOCK_M, BLOCK_K]
        a_ptrs = a_ptr + a_batch_off \
                 + (offs_m[:, None] * stride_a_m) \
                 + (k_range[None, :] * stride_a_k)
        mask_a = (offs_m[:, None] < M) & (k_range[None, :] < K)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # B tile (treated as transposed): [BLOCK_K, BLOCK_N]
        b_ptrs = b_ptr + b_batch_off \
                 + (k_range[:, None] * stride_b_k) \
                 + (offs_n[None, :] * stride_b_n)
        mask_b = (k_range[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # write back
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + c_batch_off \
             + (offs_m[:, None] * stride_c_m) \
             + (offs_n[None, :] * stride_c_n)
    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] (binary) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # cast to fp16 for the compute‑heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections (fp16)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(device=device, dtype=torch.float16),
    )  # [B,N,N,H]
    right = F.linear(
        x,
        weights["right_proj.weight"].to(device=device, dtype=torch.float16),
    )  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(
            x,
            weights["left_gate.weight"].to(device=device, dtype=torch.float16),
        )
    )  # [B,N,N,H]
    right_gate = torch.sigmoid(
        F.linear(
            x,
            weights["right_gate.weight"].to(device=device, dtype=torch.float16),
        )
    )  # [B,N,N,H]
    out_gate = torch.sigmoid(
        F.linear(
            x,
            weights["out_gate.weight"].to(device=device, dtype=torch.float16),
        )
    )  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4) Fuse mask and left/right gates
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_fp = mask.to(torch.float16)
    else:
        mask_fp = torch.ones((B, N, N), dtype=torch.float16, device=device)

    total_elements = left.numel()                     # B * N * N * H
    BLOCK = 1024                                      # tunable

    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = (triton.cdiv(total_elements, BLOCK),)
    _apply_mask_gate_kernel[grid](
        left,
        right,
        mask_fp.view(-1),
        left_gate,
        right_gate,
        left_out,
        right_out,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 5) Core TriMul reduction via a batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # reshape to (BATCH, N, N) where BATCH = B * H
    left_flat = left_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # allocate output buffer
    out_flat = torch.empty_like(left_flat)

    # strides for A
    a_batch_stride, a_row_stride, a_col_stride = left_flat.stride()
    # strides for B (we will read it as transposed)
    b_batch_stride, b_row_stride, b_col_stride = right_flat.stride()
    # strides for C
    c_batch_stride, c_row_stride, c_col_stride = out_flat.stride()

    # For the transposed view of B we swap the row/col strides
    stride_b_k = b_col_stride      # stride for the K dimension (original column)
    stride_b_n = b_row_stride      # stride for the N dimension (original row)

    # launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_N),   # columns
        triton.cdiv(N, BLOCK_M),   # rows
        B * hidden_dim,            # batch dimension
    )
    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * hidden_dim,
        N,
        N,
        N,
        a_batch_stride,
        a_row_stride,
        a_col_stride,
        b_batch_stride,
        stride_b_k,
        stride_b_n,
        c_batch_stride,
        c_row_stride,
        c_col_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"]
        .to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"]
        .to(device=device, dtype=torch.float16),
    )

    # ------------------------------------------------------------------
    # 7) Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].to(device=device, dtype=torch.float16),
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 9) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
