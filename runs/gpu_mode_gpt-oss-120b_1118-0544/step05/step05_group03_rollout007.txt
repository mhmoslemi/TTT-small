"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate multiplication.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for compute‑heavy ops.
3. Linear projections:
      left  = x @ W_left   (B,N,N,H)
      right = x @ W_right
   Gate projections + sigmoid:
      left_gate  = σ(x @ W_left_gate)
      right_gate = σ(x @ W_right_gate)
      out_gate   = σ(x @ W_out_gate)
4. Fuse projection‑gate‑mask (if a mask is present) with a small Triton kernel.
5. Core reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented via a single torch.einsum (no extra copies).
6. Output LayerNorm, multiply by `out_gate`, final linear back to `dim`.
7. Cast back to the original dtype (usually float32).

The Triton kernel only handles the element‑wise (proj * gate * mask) fusion,
leaving the O(N³) reduction to highly‑optimized torch.einsum / cuBLAS.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i64  total number of elements = B*N*N*H
    hidden_dim,        # i64  H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # guard
    mask = offs < total_elems

    # mask index (broadcast over hidden dimension)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 16384) -> torch.Tensor:
    """
    proj, gate : (B, N, N, H)   fp16, contiguous
    mask       : (B, N, N)       fp16, contiguous

    Returns proj * gate * mask (broadcast mask over H).
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (bool/float), may be ignored
            - weights      : dict of model parameters (float32)
            - config       : dict with keys
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask"     – if True, mask is ignored

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim` (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for heavy compute
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all linear weights to FP16 (no bias terms in these layers)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B,N,N,H)
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections (same shape) + sigmoid
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 6. Apply mask & gate (fused via Triton if mask is used)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is half and contiguous (0/1 values)
        mask_half = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left, left_gate, mask_half)
        right = _apply_gate_and_mask(right, right_gate, mask_half)
    else:
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Use einsum to avoid extra permutes/copies.
    out = torch.einsum('bikd,bjkd->bijd', left, right)   # (B,N,N,H)

    # --------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection back to dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)      # (B,N,N,dim)

    # --------------------------------------------------------------
    # 9. Cast back to the original dtype (usually float32)
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
