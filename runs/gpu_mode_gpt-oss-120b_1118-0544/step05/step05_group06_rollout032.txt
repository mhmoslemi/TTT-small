@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program_id(0) : batch hidden index (bh)
    # program_i = program_id(1) row block index
    # program_j = program_id(2) col block index
    bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch and hidden index
    b = bh // H
    h = bh % H

    # compute starting offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # bound checking masks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # calculate base pointers for this batch and hidden channel
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < N

        # load left tile
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + cur_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # load right tile (but we need to load transposed)
        right_ptrs = right_base + offs_j[:, None] * stride_right_j + cur_k[None, :] * stride_right_k
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # we need right as (K,N) to multiply; but we can compute dot of left (M,K) and right (N,K)??? Actually right shape is (N,K). For tl.dot we want (M,K) x (K,N). So we need to transpose right tile. 
        # We can just load right as transposed by swapping dimensions in pointer calculations: load right_tile_T where we treat j index as rows and k as columns:
        # We can load with pointer: right_ptrs_T = right_base + cur_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        # That will give shape (K,N). We'll compute dot.
        right_ptrs_T = right_base + cur_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        right_T = tl.load(right_ptrs_T, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # compute block multiplication
        acc += tl.dot(left, right_T)

    # store result
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', False)

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])

    # Cast to FP16
    x = x.to(torch.float16)

    # Cast weight tensors to fp16
    left_proj_w = weights['left_proj.weight'].to(torch.float16)
    right_proj_w = weights['right_proj.weight'].to(torch.float16)
    left_gate_w = weights['left_gate.weight'].to(torch.float16)
    right_gate_w = weights['right_gate.weight'].to(torch.float16)
    out_gate_w = weights['out_gate.weight'].to(torch.float16)
    to_out_norm_w = weights['to_out_norm.weight'].to(torch.float16)
    to_out_norm_b = weights['to_out_norm.bias'].to(torch.float16)
    to_out_w = weights['to_out.weight'].to(torch.float16)

    # compute linear projections
    left = torch.nn.functional.linear(x, left_proj_w)  # shape B,N,N,H
    right = torch.nn.functional.linear(x, right_proj_w)

    # gates
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    if mask is not None and not nomask:
        # apply gate and mask using triton kernel
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right, right_gate, mask_fp16)
    else:
        left = left * left_gate
        right = right * right_gate

    # core reduction using triton kernel
    B, N, _, H = left.shape

    # Compute strides
    left_stride_b, left_stride_i, left_stride_k, left_stride_h = left.stride()
    right_stride_b, right_stride_j, right_stride_k, right_stride_h = right.stride()
    out_shape = (B, N, N, H)
    out = torch.empty(out_shape, dtype=left.dtype, device=left.device)  # allocate output

    out_stride_b, out_stride_i, out_stride_j, out_stride_h = out.stride()

    # Launch kernel
    # decide block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # grid dims
    grid = (B*H, (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)

    triton_tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        left_stride_b, left_stride_i, left_stride_k, left_stride_h,
        right_stride_b, right_stride_j, right_stride_k, right_stride_h,
        out_stride_b, out_stride_i, out_stride_j, out_stride_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Continue with output layers
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # shape B,N,N,dim
    # Cast back to original dtype
    out = out.to(input_tensor.dtype)

    return out
```

"""
TriMul forward (outgoing) implementation with Triton acceleration.
The kernel fuses the O(N³·H) reduction:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
where left/right are the projected and gated pairwise tensors.
Mask‑gate multiplication is also performed with a small Triton kernel.
All heavy compute is done in FP16 with FP32 accumulation for accuracy.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise proj * gate * mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # *[total_elems]   fp16
    gate_ptr,          # *[total_elems]   fp16
    mask_ptr,          # *[mask_elems]    fp16
    out_ptr,           # *[total_elems]   fp16
    total_elems,       # i64
    hidden_dim,        # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    within = offs < total_elems

    # mask index is the flat index divided by hidden_dim (broadcast)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=within, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=within, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=within, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=within)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 4096) -> torch.Tensor:
    """
    proj, gate: FP16 tensors of shape (B,N,N,H)
    mask: FP16 tensor of shape (B,N,N) – broadcast over H.
    Returns proj * gate * mask.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched matrix‑multiply for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)  # batch*hidden index
    pid1 = tl.program_id(1)  # i‑block
    pid2 = tl.program_id(2)  # j‑block

    bh = pid0
    b = bh // H
    h = bh % H

    # starting offsets for this tile
    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for the selected (b, h)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        cur_k = kk * BLOCK_K + offs_k
        mask_k = cur_k < N

        # left[i, k]
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + cur_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right_T[k, j]  (we load transposed to match tl.dot)
        right_ptrs_T = right_base + cur_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right_T = tl.load(right_ptrs_T,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0)

        # accumulate product
        acc += tl.dot(left.to(tl.float32), right_T.to(tl.float32))

    # write back result (convert to FP16)
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict containing all Linear and LayerNorm parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the compute‑intensive stages
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. FP16 weights
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gates + optional mask (Triton fused)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right, right_gate, mask_fp16)
    else:
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core TriMul reduction (batched GEMM) via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # left shape (B, N, N, H)

    # allocate output tensor
    out = torch.empty((B, N, N, H), dtype=left.dtype, device=left.device)

    # strides (in elements) for each tensor
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # block sizes – tuned for typical B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        math.ceil(N / BLOCK_M),
        math.ceil(N / BLOCK_N),
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_lb,
        s_li,
        s_lk,
        s_lh,
        s_rb,
        s_ri,
        s_rk,
        s_rh,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gating, and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
