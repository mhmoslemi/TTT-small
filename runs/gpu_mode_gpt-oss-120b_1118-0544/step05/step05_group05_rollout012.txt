    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # 1. LayerNorm

    norm_b = weights["norm.bias"].to(device)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    # cast to fp16
    x = x.to(torch.float16)

    # 2. Linear proj
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)

    left = torch.nn.functional.linear(x, left_proj_w)    # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply mask if required
    if mask is not None and not nomask:
        # mask shape (B,N,N)
        mask = mask.to(device).to(torch.float16).unsqueeze(-1)  # broadcast over H
        left = left * left_gate * mask
        right = right * right_gate * mask
    else:
        left = left * left_gate
        right = right * right_gate

    # core contraction via batched GEMM
    B, N, _, H = left.shape
    left_mat = left.permute(0,3,1,2).reshape(B*H, N, N)
    right_mat = right.permute(0,3,1,2).reshape(B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()

    # output layernorm
    out = torch.nn.functional.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # apply out_gate using Triton kernel
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    total = out_flat.numel()
    BLOCK = 1024

        out_gate_flat,
        total,
        BLOCK_SIZE=BLOCK,
    )
    out = out_flat.view_as(out)

    # final linear projection
    out = torch.nn.functional.linear(out, to_out_w)

    # back to original dtype
    out = out.to(dtype)
    return out

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the input (float32 for stability).
2. Cast to fp16 and compute linear projections and associated sigmoid gates.
3. Broadcast‑multiply the optional mask and the gates (pure PyTorch, which is faster
   than a Triton kernel for large tensors).
4. Perform the core O(N³·H) contraction via a batched GEMM:
   out₍b,i,j,h₎ = Σₖ left₍b,i,k,h₎ * right₍b,j,k,h₎ .
   This is realised by reshaping to (B·H, N, N) and using torch.bmm.
5. Apply the output LayerNorm.
6. Multiply by the output gate using a tiny Triton kernel (required to use Triton).
7. Final linear projection back to the original feature dimension.
8. Cast the result back to the input dtype.

All heavy arithmetic is carried out in fp16; the final output is returned in the
original dtype (usually float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_gate_kernel(
    out_ptr,            # *fp16   output tensor (flattened)
    gate_ptr,           # *fp16   gate tensor   (flattened)
    total,              # i64     total number of elements
    BLOCK_SIZE: tl.constexpr,  # compile‑time block size
):
    """Elementwise out *= gate in‑place."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    out_val = tl.load(out_ptr + offs, mask=mask, other=0.0)
    gate_val = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    out_val = out_val * gate_val
    tl.store(out_ptr + offs, out_val, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, torch.float32)
    norm_b = weights["norm.bias"].to(device, torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for compute‑heavy parts
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gates (all fp16)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 4. Apply optional mask and element‑wise gate multiplication
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B, N, N) -> (B, N, N, 1) broadcast over hidden dimension
        mask_fp = mask.to(device, torch.float16).unsqueeze(-1)
        left = left * left_gate * mask_fp
        right = right * right_gate * mask_fp
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ rightᵀ) over the sequence dim
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # Rearrange to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # -----------------------------------------------------------------
    # 7. Apply output gate using a tiny Triton kernel
    # -----------------------------------------------------------------
    out_flat = out.view(-1)
    gate_flat = out_gate.view(-1)
    total = out_flat.numel()
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)
    _apply_gate_kernel[grid](
        out_flat,
        gate_flat,
        total,
        BLOCK_SIZE=BLOCK,
    )
    out = out_flat.view_as(out)

    # -----------------------------------------------------------------
    # 8. Final linear projection back to original dim
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # Cast back to the original dtype (usually float32)
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
