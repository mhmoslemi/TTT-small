"""
Highly‑optimized forward pass for the outgoing TriMul operator (AlphaFold3 style).

Algorithm
---------
1. Layer‑normalize the 4‑D input (B,N,N,C) in FP32.
2. Cast to FP16 and compute linear projections and gate values (no bias).
3. Fuse gate*mask (if a mask is provided) with a tiny Triton kernel.
4. Core O(N³·H) reduction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton batched GEMM where each (b,h) pair is a
   separate “batch”. The kernel tiles the i‑ and j‑dimensions (BLOCK_M×BLOCK_N)
   and iterates over k in blocks (BLOCK_K), using fp32 accumulation.
5. Apply a second LayerNorm over the hidden dimension, multiply by the output
   gate and project back to the original channel dimension.
6. Cast the result back to the input dtype (usually FP32).

Only the mask‑gate fusion and the O(N³·H) reduction are written in Triton;
all other operations use fast PyTorch kernels.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,      # *[total_elems]   fp16
    gate_ptr,      # *[total_elems]   fp16
    mask_ptr,      # *[mask_elems]    fp16
    out_ptr,       # *[total_elems]   fp16
    total_elems,   # i64
    hidden_dim,    # i64  (size of the last dimension H)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    within = offs < total_elems

    # broadcast mask: each hidden entry repeats the same mask value
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=within, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=within, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=within, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=within)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """Element‑wise `proj * gate * mask` where `mask` is (B,N,N) and broadcast
    over the hidden dimension H. All tensors must be FP16 and contiguous."""
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],   # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: core TriMul reduction (batched GEMM per hidden channel)
# ----------------------------------------------------------------------
@triton.jit
def _trmul_kernel(
    left_ptr,   # *[B,N,N,H] fp16
    right_ptr,  # *[B,N,N,H] fp16
    out_ptr,    # *[B,N,N,H] fp16
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Grid: (block_i, block_j, batch_hidden)
    pid_bh = tl.program_id(2)   # combined batch * hidden
    pid_m  = tl.program_id(0)   # i‑block
    pid_n  = tl.program_id(1)   # j‑block

    b = pid_bh // H
    h = pid_bh % H

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    i_range = i0 + tl.arange(0, BLOCK_M)
    j_range = j0 + tl.arange(0, BLOCK_N)

    i_mask = i_range < N
    j_mask = j_range < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in blocks
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = kk * BLOCK_K
        k_range = k0 + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # ----- load left tile (M,K) -----
        # left[b, i, k, h]
        left_ptrs = (left_ptr
                     + b * stride_l_b
                     + i_range[:, None] * stride_l_i
                     + k_range[None, :] * stride_l_k
                     + h * stride_l_h)
        left = tl.load(left_ptrs,
                       mask=(i_mask[:, None] & k_mask[None, :]),
                       other=0.0)

        # ----- load right tile (K,N) (transposed view) -----
        # right[b, j, k, h] -> we need (K,N) where first dim is k, second is j
        right_ptrs = (right_ptr
                      + b * stride_r_b
                      + k_range[:, None] * stride_r_k
                      + j_range[None, :] * stride_r_i
                      + h * stride_r_h)
        right = tl.load(right_ptrs,
                        mask=(k_mask[:, None] & j_mask[None, :]),
                        other=0.0)

        # ----- accumulate -----
        acc += tl.dot(left, right)        # (M,K)·(K,N) -> (M,N)

    # ----- store result -----
    out_ptrs = (out_ptr
                + b * stride_o_b
                + i_range[:, None] * stride_o_i
                + j_range[None, :] * stride_o_j
                + h * stride_o_h)
    tl.store(out_ptrs,
              acc.to(tl.float16),
              mask=(i_mask[:, None] & j_mask[None, :]))

    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B,N,N,hidden_dim)
    # --------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Apply mask + gate (Triton kernel) or just gate
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure tensors are contiguous for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape  # (B,N,N,H)

    # --------------------------------------------------------------
    # 7. Core O(N³·H) reduction via a custom Triton kernel
    # --------------------------------------------------------------
    out = torch.empty_like(left)  # (B,N,N,H), fp16

    # Strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Block sizes – tuned for fp16 matmul on the B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        math.ceil(N / BLOCK_M),
        math.ceil(N / BLOCK_N),
        B * H,
    )

    _trmul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 8. Output LayerNorm, gate, and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B,N,N,dim), still fp16

    # --------------------------------------------------------------
    # 9. Cast back to original dtype
    # --------------------------------------------------------------
    return out.to(dtype)
