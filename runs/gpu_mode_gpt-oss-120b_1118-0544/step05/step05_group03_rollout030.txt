import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for batched matrix multiplication (TriMul reduction)
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # i tile
    pid_n = tl.program_id(1)  # j tile
    pid_bh = tl.program_id(2) # batch*hidden

    # Decode batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # Compute global indices for the tile
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_k):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block: shape (BLOCK_M, BLOCK_K)
        left_offset = b * stride_left_b + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k + h * stride_left_h
        left = tl.load(left_ptr + left_offset,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # Load right block transposed: shape (BLOCK_K, BLOCK_N)
        right_offset = b * stride_right_b + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j + h * stride_right_h
        right = tl.load(right_ptr + right_offset,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # Dot product accumulate
        acc += tl.dot(left, right)  # (M,N) + (K,N) -> (M,N)

    # Write output
    out_offset = b * stride_out_b + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j + h * stride_out_h
    tl.store(out_ptr + out_offset, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

def _tri_mul_forward(input_tensor: torch.Tensor,
                     mask: torch.Tensor,
                     weights: dict,
                     config: dict) -> torch.Tensor:
    """
    Compute the outgoing TriMul update using a fused Triton kernel for the
    O(N³) reduction.  All linear layers are performed in FP16; the
    reduction accumulates in FP32 for numerical stability.
    """
    device = input_tensor.device
    dtype_in = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive work
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # -----------------------------------------------------------------
    # 4. Apply gating (and optional mask) – elementwise in PyTorch
    # -----------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    if mask is not None and not nomask:
        # mask: [B, N, N] → broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # Ensure contiguous layout for Triton
    left = left.contiguous()
    right = right.contiguous()

    # -----------------------------------------------------------------
    # 5. Core TriMul reduction via Triton (batched GEMM over K)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # [B, N, N, H] fp16

    # Strides (in elements) for left/right/out tensors
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Choose tile sizes – these work well for the target shapes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # smaller K works for large N without excess registers

    grid = (triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            B * H)

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out,
                       (H,),
                       weight=weights["to_out_norm.weight"].to(torch.float16),
                       bias=weights["to_out_norm.bias"].to(torch.float16))

    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 7. Cast back to original dtype
    # -----------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

def custom_kernel(data):
    """
    Entry point for the Triton‑accelerated outgoing TriMul operator.
    ``data`` is a tuple (input_tensor, mask, weights, config) as described
    in the problem statement.
    """
    input_tensor, mask, weights, config = data
    return _tri_mul_forward(input_tensor, mask, weights, config)

"""
Outgoing TriMul (AlphaFold3) – Triton‑accelerated implementation.

The heavy O(N³) reduction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
is performed by a custom Triton kernel that treats the hidden dimension
as an extra batch axis (B·H).  All linear projections, LayerNorm and
gate computations are done in FP16, while the reduction accumulates in
FP32 for numerical stability.  The final output is cast back to the
input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # shapes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where left/right have shape (B,N,N,H).

    Grid:
        (ceil(N/BLOCK_M), ceil(N/BLOCK_N), B*H)
    """
    pid_m = tl.program_id(0)          # tile along i
    pid_n = tl.program_id(1)          # tile along j
    pid_bh = tl.program_id(2)         # batch * hidden index

    # decode batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # global i/j offsets for this tile
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis k
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_k):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left block (M x K) -----
        left_offset = (
            b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(
            left_ptr + left_offset,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # ----- load right block transposed (K x N) -----
        # we need right[b, j, k, h] → treat as B[K, N] for matmul
        right_offset = (
            b * stride_right_b
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
            + h * stride_right_h
        )
        right = tl.load(
            right_ptr + right_offset,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # ----- accumulate -----
        acc += tl.dot(left, right)   # (M,K) @ (K,N) → (M,N)

    # ----- write result -----
    out_offset = (
        b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(
        out_ptr + out_offset,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper: full forward pass
# ----------------------------------------------------------------------
def _tri_mul_forward(input_tensor: torch.Tensor,
                     mask: torch.Tensor,
                     weights: dict,
                     config: dict) -> torch.Tensor:
    """
    Compute the outgoing TriMul update.
    """
    device = input_tensor.device
    dtype_in = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm over channel dimension `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to fp16 for the heavy compute path
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --------------------------------------------------------------
    # 4. Apply gating (and optional mask) – elementwise
    # --------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    if mask is not None and not nomask:
        # mask shape (B,N,N) → broadcast over hidden_dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    left = left.contiguous()
    right = right.contiguous()

    # --------------------------------------------------------------
    # 5. Core O(N³) reduction via Triton (batched GEMM)
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)      # (B,N,N,H), fp16

    # Strides in element counts
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_rj, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Tile sizes – tuned for the target GPU
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_lb,
        s_li,
        s_lk,
        s_lh,
        s_rb,
        s_rj,
        s_rk,
        s_rh,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 7. Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask        : torch.Tensor [B, N, N] (or None)
            - weights     : dict of model parameters (float32)
            - config      : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    return _tri_mul_forward(input_tensor, mask, weights, config)
