@triton.jit
def matmul_bn_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_hidden, M, N, K,
    stride_ab, stride_ah, stride_ak,
    stride_bb, stride_bh, stride_bk,
    stride_cb, stride_ch, stride_cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    num_blocks = num_blocks_m * num_blocks_n
    bh = pid // num_blocks
    block_id = pid % num_blocks
    pid_m = block_id // num_blocks_n
    pid_n = block_id % num_blocks_n

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # check bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # compute a_ptr
        a_ptrs = a_ptr + bh * stride_ab + (offs_m[:, None] * stride_ah) + (offs_k[None, :] * stride_ak)
        # compute b_ptr (transposed)
        b_ptrs = b_ptr + bh * stride_bb + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bh)  # careful: we want shape (BLOCK_K, BLOCK_N) where second dim is n offset: row dimension is k, column dimension is n (after transpose)
        # Actually stride_bh is row stride for B (N dimension). In original layout B is (batch_hidden, N, K). So to load B^T as (K,N) we need pointer = base + bh*stride_bb + (offs_k[:,None] * stride_bk) + (offs_n[None,:] * stride_bh)
        # load
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product:
        acc += tl.dot(a, b, out_dtype=tl.float32)

        # increment pointers: not needed because we recompute each loop.

    # store result
    c_ptrs = c_ptr + bh * stride_cb + (offs_m[:, None] * stride_ch) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul forward (outgoing version) with Triton‑accelerated batched GEMM.

The kernel performs:
  1. LayerNorm over the channel dimension C.
  2. Two linear projections to hidden dimension H (left/right).
  3. Three sigmoid gates (left, right, out).
  4. Optional mask + gate fusion (element‑wise, done in PyTorch).
  5. Core reduction   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
     implemented as a batched GEMM (A @ Bᵀ) with a custom Triton kernel.
  6. LayerNorm over the hidden dimension, gate multiplication and final
     linear projection back to C.

Only the heavy O(N³·H) reduction is off‑loaded to Triton; all other ops stay
in PyTorch (they are cheap compared to the GEMM). The kernel works for
batch size 1‑2, sequence length up to 1024, and hidden dimension 128.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_ab, stride_am, stride_ak,   # A: (batch, M, K)
    stride_bb, stride_bn, stride_bk,   # B (original): (batch, N, K)
    stride_cb, stride_cm, stride_cn,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  where A,B shape = (batch, M, K) and (batch, N, K)."""
    pid = tl.program_id(0)

    # number of blocks in the M and N dimensions
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    # decode pid into (batch, block_i, block_j)
    batch_id = pid // blocks_per_batch
    pid_in_batch = pid % blocks_per_batch
    pid_m = pid_in_batch // num_blocks_n
    pid_n = pid_in_batch % num_blocks_n

    # offsets for the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + batch_id * stride_ab + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Bᵀ tile: (BLOCK_K, BLOCK_N)
        # B original layout is (batch, N, K). To get Bᵀ we load with K stride first.
        b_ptrs = b_ptr + batch_id * stride_bb + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product and accumulate
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # write out C tile (FP16)
    c_ptrs = c_ptr + batch_id * stride_cb + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys
                * "dim"          – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask"       – bool, if True the mask is ignored

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # Work in fp16 for the bulk of the computation
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w) # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5) Apply mask (if any) and element‑wise gates
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
    else:
        mask_fp = torch.ones((B, N, N, 1), dtype=torch.float16, device=device)

    left = left * mask_fp * left_gate
    right = right * mask_fp * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: batched GEMM (A @ Bᵀ) in Triton
    #    Shape handling:
    #       left  : (B,N,N,H) -> (B*H, N, K) with K=N
    #       right : (B,N,N,H) -> (B*H, N, K) with K=N
    #    Result out_flat: (B*H, N, N)
    # ------------------------------------------------------------------
    # Move hidden dim to batch dimension
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # Strides (all tensors are contiguous)
    stride_ab = left_flat.stride(0)   # batch stride = N*N
    stride_am = left_flat.stride(1)   # row stride = N
    stride_ak = left_flat.stride(2)   # col stride = 1

    stride_bb = right_flat.stride(0)  # batch stride = N*N
    stride_bn = right_flat.stride(1)  # row stride = N   (used as stride for N dimension)
    stride_bk = right_flat.stride(2)  # col stride = 1   (used for K dimension)

    stride_cb = out_flat.stride(0)    # batch stride = N*N
    stride_cm = out_flat.stride(1)    # row stride = N
    stride_cn = out_flat.stride(2)    # col stride = 1

    batch_hidden = B * hidden_dim
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch_hidden * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_hidden,
        N,
        N,
        N,
        stride_ab,
        stride_am,
        stride_ak,
        stride_bb,
        stride_bn,
        stride_bk,
        stride_cb,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10) Final linear projection back to original dim C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,C)

    # ------------------------------------------------------------------
    # 11) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
