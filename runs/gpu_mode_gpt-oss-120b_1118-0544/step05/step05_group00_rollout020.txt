@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # block indices
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # batch and hidden index
    bh = pid_bh
    batch = bh // H
    hid = bh % H

    # masks for boundary
    mask_i = i < N
    mask_j = j < N

    # compute base offsets
    # offset for batch and hidden
    batch_offset = batch * stride_lb
    hid_offset_left = hid * stride_lh
    hid_offset_right = hid * stride_rh
    hid_offset_out = hid * stride_oh

    # compute index offsets for i and j dimensions
    offset_i = i[:, None] * stride_li
    offset_j = j[None, :] * stride_rj  # but stride_rj same as stride_li

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # left tile: shape (M,K)
        left_ptrs = left_ptr + batch_offset + hid_offset_left + offset_i + k_idx[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # right tile: shape (K,N) (transposed)
        right_ptrs = right_ptr + batch_offset + hid_offset_right + k_idx[:, None] * stride_rk + offset_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float16)

        acc += tl.dot(left, right)

    # store
    out_ptrs = out_ptr + batch_offset + hid_offset_out + offset_i + offset_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input (float32 for stability) and cast to fp16.
2. Linear projections to a hidden dimension `H`:
   * left  = x @ left_proj_weight
   * right = x @ right_proj_weight
   * left_gate  = sigmoid(x @ left_gate_weight)
   * right_gate = sigmoid(x @ right_gate_weight)
   * out_gate   = sigmoid(x @ out_gate_weight)
3. Fuse element‑wise mask (if present) and the two gates in a single
   Triton kernel (`_gate_mask_kernel`), writing the gated tensors back
   in‑place.
4. Core O(N³·H) contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton kernel (`tri_mul_outgoing_kernel`)
   that performs a batched matrix multiplication L @ Rᵀ for each
   (batch, hidden) pair using block‑wise tiling.
5. Apply a second LayerNorm over the hidden dimension, multiply by
   `out_gate`, and project back to the original dimension with a final
   linear layer.
6. Cast the result back to the input dtype.

The implementation uses mixed precision (fp16 compute, fp32 accumulation)
and keeps the heavy O(N³) work on the GPU via Triton, satisfying the
requirement of having at least one Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,            # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,  # same shape
    mask_ptr,                       # [B,N,N] flattened
    total_elems: tl.int64, hidden_dim: tl.int64,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
       `mask` is broadcast over the hidden dimension `H`.
    """
    pid = tl.program_id(0)
    off = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = off < total_elems

    # load tensors
    left = tl.load(left_ptr + off, mask=active, other=0.0)
    right = tl.load(right_ptr + off, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + off, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + off, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden slice shares the same mask entry
        mask_idx = off // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # write back in‑place
    tl.store(left_ptr + off, left, mask=active)
    tl.store(right_ptr + off, right, mask=active)


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Arguments
    ---------
    left_ptr, right_ptr, out_ptr : pointers to fp16 tensors of shape [B,N,N,H]
    stride_*                    : Strides (in elements) of the corresponding
                                 tensor dimensions.
    B, N, H                     : batch size, sequence length, hidden dim.
    BLOCK_M, BLOCK_N, BLOCK_K   : tile sizes.
    """
    pid_i = tl.program_id(0)      # block over i dimension
    pid_j = tl.program_id(1)      # block over j dimension
    pid_bh = tl.program_id(2)     # combined batch & hidden index

    # ----- block indices -----
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # ----- batch & hidden indices -----
    bh = pid_bh
    batch = bh // H
    hid = bh % H

    mask_i = i < N
    mask_j = j < N

    # offsets for batch and hidden dimension
    batch_off = batch * stride_lb
    hid_off_l = hid * stride_lh
    hid_off_r = hid * stride_rh
    hid_off_o = hid * stride_oh

    # offsets for i and j dimensions
    off_i = i[:, None] * stride_li          # (M,1)
    off_j = j[None, :] * stride_rj          # (1,N)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # left tile: shape (M,K)
        left_ptrs = left_ptr + batch_off + hid_off_l + off_i + k_idx[None, :] * stride_lk
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float16)

        # right tile: shape (K,N) – we load transposed so that we can use tl.dot
        right_ptrs = right_ptr + batch_off + hid_off_r + k_idx[:, None] * stride_rk + off_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float16)

        # dot: (M,K) x (K,N) -> (M,N)
        acc += tl.dot(left, right)

    # write back result
    out_ptrs = out_ptr + batch_off + hid_off_o + off_i + off_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------ #
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # ------------------------------------------------------------------ #
    # 2. Cast to fp16 for compute‑heavy path
    # ------------------------------------------------------------------ #
    x = x.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 3. Linear projections (bias‑less) – fp16
    # ------------------------------------------------------------------ #
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left  = F.linear(x, left_proj_w)          # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)         # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------ #
    # 4. Fuse mask and gates (if a mask is present)
    # ------------------------------------------------------------------ #
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten tensors for the kernel
        B, N, _, H = left.shape
        total = B * N * N * H
        BLOCK = 1024

        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B,N,N)

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left  = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Core O(N³·H) contraction via a Triton kernel
    # ------------------------------------------------------------------ #
    B, N, _, H = left.shape
    out = torch.empty_like(left)   # (B,N,N,H)

    # strides (in elements) for the contiguous tensors
    stride_lb = left.stride(0)
    stride_li = left.stride(1)
    stride_lk = left.stride(2)
    stride_lh = left.stride(3)

    # we reuse the same strides for right and out because they have identical layout
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    tri_mul_outgoing_kernel[grid](
        left, right, out,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_lb, stride_li, stride_lk, stride_lh,   # right strides (identical)
        stride_lb, stride_li, stride_lk, stride_lh,   # out strides (identical)
        B, N, H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------ #
    # 6. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------ #
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------ #
    # 7. Cast back to the original dtype
    # ------------------------------------------------------------------ #
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
