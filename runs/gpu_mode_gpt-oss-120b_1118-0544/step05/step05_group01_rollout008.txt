@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # Compute number of tiles per matrix
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_matrix = num_tiles_m * num_tiles_n

    # Determine batch-head index and tile coordinates
    bh_idx = pid // tiles_per_matrix
    tile_idx = pid % tiles_per_matrix
    pid_m = tile_idx // num_tiles_n
    pid_n = tile_idx % num_tiles_n

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for boundary
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator (fp32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + bh_idx * stride_left_batch + (offs_m[:, None] * stride_left_m) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile: shape (BLOCK_K, BLOCK_N)
        # right is transposed: we need (k, n)
        b_ptrs = right_ptr + bh_idx * stride_right_batch + (offs_k[:, None] * stride_right_k) + (offs_n[None, :] * stride_right_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Dot product
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = out_ptr + bh_idx * stride_out_batch + (offs_m[:, None] * stride_out_m) + (offs_n[None, :] * stride_out_n)
    # Cast back to fp16 for storage
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass implemented with Triton.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension `dim`.
2. Cast to fp16 for the compute‑intensive portion.
3. Compute linear projections and gates:
   * left_proj  = x @ W_left
   * right_proj = x @ W_right
   * left_gate  = sigmoid(x @ W_left_gate)
   * right_gate = sigmoid(x @ W_right_gate)
   * out_gate   = sigmoid(x @ W_out_gate)
4. Fuse mask, projection and gate (if a mask is present) with a small
   Triton kernel (`_apply_gate_mask_kernel`).  The mask is broadcast over
   the hidden dimension.
5. Core reduction: for each batch `b` and hidden slice `h`
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication `L @ Rᵀ`.  It is carried out
   by a custom Triton GEMM kernel (`_tri_mul_kernel`) that processes
   tiles of size (BLOCK_M, BLOCK_N) and accumulates in fp32.
6. Apply a second layer‑norm over the hidden dimension, multiply by
   `out_gate`, and project back to the original dimension with a final
   linear layer.
7. Cast the result back to the input dtype (normally fp32).

The implementation uses mixed‑precision (fp16 for storage / arithmetic,
fp32 for accumulation) and keeps the heavy inner loops inside Triton
kernels, yielding a substantial speed‑up over a pure‑PyTorch reference.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Kernel 1: fuse projection * gate * mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B,N,N,H)
    gate_ptr,          # fp16 * (B,N,N,H)
    mask_ptr,          # fp16 * (B,N,N)
    out_ptr,           # fp16 * (B,N,N,H)
    total_elems,       # B*N*N*H
    hidden_dim,        # H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # every hidden_dim consecutive elements correspond to one mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Element‑wise: proj * sigmoid(gate) * mask   (mask broadcast over hidden dim).
    All tensors must be fp16 & contiguous.
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel 2: core TriMul reduction (batched GEMM L @ Rᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,           # fp16 * (B*H, N, N)
    right_ptr,          # fp16 * (B*H, N, N)
    out_ptr,            # fp16 * (B*H, N, N)
    B, N, H,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Tile decomposition: (batch_head, tile_m, tile_n)
    # --------------------------------------------------------------
    tiles_m = tl.cdiv(N, BLOCK_M)
    tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_matrix = tiles_m * tiles_n

    bh_idx = pid // tiles_per_matrix               # 0 .. B*H-1
    tile_idx = pid % tiles_per_matrix
    pid_m = tile_idx // tiles_n
    pid_n = tile_idx % tiles_n

    # Offsets within the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = (
            left_ptr
            + bh_idx * stride_left_batch
            + (offs_m[:, None] * stride_left_m)
            + (offs_k[None, :] * stride_left_k)
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # B tile: (BLOCK_K, BLOCK_N) – note the transpose
        b_ptrs = (
            right_ptr
            + bh_idx * stride_right_batch
            + (offs_k[:, None] * stride_right_k)
            + (offs_n[None, :] * stride_right_n)
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write back (cast to fp16)
    # --------------------------------------------------------------
    c_ptrs = (
        out_ptr
        + bh_idx * stride_out_batch
        + (offs_m[:, None] * stride_out_m)
        + (offs_n[None, :] * stride_out_n)
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask         : torch.Tensor [B, N, N] (binary) or None
            - weights      : dict of FP32 model parameters
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over channel dimension `dim`
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim), dtype = orig_dtype

    # --------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all linear weights to fp16 (no bias)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections and gates (FP16)
    # --------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)      # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 5. Apply mask and gating (fused via Triton if mask present)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6. Core reduction: out = left @ rightᵀ   (batched GEMM)
    #    Shapes: left/right  (B,N,N,H) -> (B*H, N, N)
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) where the hidden dimension is fused into the batch axis
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    out_flat = torch.empty_like(left_flat)

    # strides (in elements)
    stride_l_batch = left_flat.stride(0)
    stride_l_m     = left_flat.stride(1)
    stride_l_k     = left_flat.stride(2)

    stride_r_batch = right_flat.stride(0)
    # for the transposed view we need: k‑stride = stride over last dim, n‑stride = stride over middle dim
    stride_r_k     = right_flat.stride(2)
    stride_r_n     = right_flat.stride(1)

    stride_o_batch = out_flat.stride(0)
    stride_o_m     = out_flat.stride(1)
    stride_o_n     = out_flat.stride(2)

    # Triton grid
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    tiles_per_matrix = tiles_m * tiles_n
    grid = (B * H * tiles_per_matrix,)

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B,
        N,
        H,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_k,
        stride_r_n,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 8. Cast back to original dtype (usually float32)
    # --------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
