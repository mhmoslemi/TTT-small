@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # each program identifies (batch_id, block_m, block_n)
    batch_id = tl.program_id(2)
    block_m = tl.program_id(0)
    block_n = tl.program_id(1)

    # compute start coordinates
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute pointers
    a_ptrs = a_ptr + batch_id * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + batch_id * stride_b_batch + offs_n[None, :] * stride_b_n + offs_k[:, None] * stride_b_k  # note B transposed

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K - k*BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=mask_n[None, :] & (offs_k < K - k*BLOCK_K), other=0.0)
        # compute dot
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # write back
    c_ptrs = c_ptr + batch_id * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated outgoing TriMul forward pass.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (`dim`).
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (no bias) → left/right tensors of shape
   (B, N, N, hidden_dim).
4. Compute gating tensors (sigmoid of linear projections) and apply the
   optional mask (broadcast over the hidden dimension).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication A @ Bᵀ where
   A and B have shape (B*hidden_dim, N, N).  It is implemented with a
   custom Triton kernel that tiles the M, N and K dimensions.
6. Layer‑norm over the hidden dimension, multiply by the output gate,
   and a final linear projection back to `dim`.
7. Cast the result back to the original dtype.

Only the reduction (step 5) is performed in Triton, satisfying the
requirement to use a custom kernel while keeping the surrounding code
in PyTorch for simplicity and readability.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------
# Triton kernel: batched GEMM computing C = A @ Bᵀ
#   A : (batch, M, K)   – left tensor
#   B : (batch, N, K)   – right tensor (will be accessed as Bᵀ)
#   C : (batch, M, N)
# --------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    a_ptr,                # fp16 * (batch, M, K)
    b_ptr,                # fp16 * (batch, N, K)
    c_ptr,                # fp16 * (batch, M, N)
    batch, M, N, K,       # dimensions (int)
    stride_a_batch, stride_a_m, stride_a_k,   # strides for A
    stride_b_batch, stride_b_n, stride_b_k,   # strides for B
    stride_c_batch, stride_c_m, stride_c_n,   # strides for C
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # 3‑D grid: (block_m, block_n, batch_id)
    batch_id = tl.program_id(2)
    block_m = tl.program_id(0)
    block_n = tl.program_id(1)

    # offsets inside the block
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # predicates for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch

    # accumulate in fp32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        # global K offset for this tile
        k_off = k_tile * BLOCK_K

        # load A tile (M x K)
        a = tl.load(
            a_base + offs_m[:, None] * stride_a_m + (offs_k + k_off)[None, :] * stride_a_k,
            mask=mask_m[:, None] & ((offs_k + k_off) < K),
            other=0.0,
        )
        # load B tile (N x K) – note we need Bᵀ, thus K is the row dimension
        b = tl.load(
            b_base + offs_n[None, :] * stride_b_n + (offs_k + k_off)[:, None] * stride_b_k,
            mask=mask_n[None, :] & ((offs_k + k_off) < K),
            other=0.0,
        )
        # dot product over K
        acc += tl.dot(a, b)

    # write result
    c = c_ptr + batch_id * stride_c_batch
    c_ptrs = c + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _tri_mul_reduction(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated reduction:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

    Args:
        left, right: fp16 tensors of shape (B, N, N, H)

    Returns:
        out: fp16 tensor of shape (B, N, N, H)
    """
    B, N, _, H = left.shape
    batch = B * H
    # reshape to (batch, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(batch, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(batch, N, N)
    out_flat = torch.empty_like(left_flat)

    # block sizes – chosen to fit typical B200 constraints
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # grid dimension for M
        triton.cdiv(N, BLOCK_N),   # grid dimension for N
        batch,                     # one block per logical batch
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch,
        N,
        N,
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]
            - mask         : torch.Tensor [B, N, N] (or None)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, dim] (same dtype as input).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all linear weights to fp16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections
    # --------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # --------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 6. Apply mask (if present) and gating
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is fp16 and broadcastable
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction (batched A @ Bᵀ) via Triton
    # --------------------------------------------------------------
    out = _tri_mul_reduction(left, right)   # (B, N, N, H)

    # --------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
