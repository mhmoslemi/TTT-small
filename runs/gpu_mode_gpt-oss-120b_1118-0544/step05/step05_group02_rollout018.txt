    stride_ob, stride_om, stride_ok,

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)

"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the 4‑D input (B, N, N, C).
2. Cast to fp16 and apply three linear projections:
   - left_proj, right_proj  -> (B, N, N, H)
   - left_gate, right_gate, out_gate -> (B, N, N, H) after sigmoid
3. Fuse mask, projection and gate with a tiny Triton kernel (if a mask is
   provided).  The result are tensors `left` and `right`.
4. Compute the expensive reduction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   i.e. for each (b,h) a dense matrix multiplication (N×N)·(N×N)ᵀ.
   This is performed by a custom Triton GEMM kernel that processes the
   (B·H) independent matrix multiplications in parallel.
5. Apply a second LayerNorm on the hidden dimension, multiply by `out_gate`,
   and linearly project back to the original dimension.
6. Cast the result back to the input dtype (float32).

The implementation uses Triton for two parts:
   * `gate_mask_kernel` – element‑wise (proj * gate * mask) fusion.
   * `trimul_gemm_kernel` – batched fp16 GEMM for the O(N³·H) core.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask (broadcast mask over H)
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    proj_ptr,          # *[total_elems]   fp16
    gate_ptr,          # *[total_elems]   fp16
    mask_ptr,          # *[mask_elems]    fp16
    out_ptr,           # *[total_elems]   fp16
    total_elems,       # int
    hidden_dim,        # int
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # map linear index to mask index (mask is not expanded over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using Triton.
    All tensors must be fp16 and contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)
    mask_flat = mask.view(-1)  # (B*N*N,)

    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the core reduction
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr,   # *[BATCH, M, K]  fp16
    right_ptr,  # *[BATCH, M, K]  fp16   (will be accessed as transposed)
    out_ptr,    # *[BATCH, M, N]  fp16   (N == M)
    BATCH: tl.constexpr,
    M: tl.constexpr,
    N: tl.constexpr,
    stride_lb, stride_lm, stride_lk,   # left strides
    stride_rb, stride_rm, stride_rk,   # right strides (same layout as left)
    stride_ob, stride_om, stride_ok,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # which (batch * hidden) slice
    pid_m = tl.program_id(1)       # row tile
    pid_n = tl.program_id(2)       # col tile

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + pid_batch * stride_lb \
                    + offs_m[:, None] * stride_lm \
                    + offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile (transposed on‑the‑fly): we need (BLOCK_K, BLOCK_N)
        # original layout: [batch, row (M), col (K)]
        # after transpose we want rows = K, cols = N (where N == M)
        right_ptrs = right_ptr + pid_batch * stride_rb \
                     + offs_k[:, None] * stride_rm \
                     + offs_n[None, :] * stride_rk
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # Accumulate product
        acc += tl.dot(left, right)   # (M,K)·(K,N) -> (M,N)

    # Write back
    out_ptrs = out_ptr + pid_batch * stride_ob \
               + offs_m[:, None] * stride_om \
               + offs_n[None, :] * stride_ok
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) or None
            - weights: dict of model parameters (float32)
            - config: dict containing 'dim', 'hidden_dim' and optional 'nomask'

    Returns:
        Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6. Fuse mask, projection and gate (Triton) or simple multiplication
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate   # (B, N, N, H)
        right = right_proj * right_gate

    # Ensure contiguity before feeding to Triton GEMM
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction using a Triton batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # Rearrange to (B, H, N, N) -> flatten to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_mat = torch.empty_like(left_mat)

    # Strides for the flattened tensors (batch, row, col)
    stride_lb, stride_lm, stride_lk = left_mat.stride()
    stride_rb, stride_rm, stride_rk = right_mat.stride()
    stride_ob, stride_om, stride_ok = out_mat.stride()

    # Tuning parameters – reasonable defaults for B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                   # batch*hidden dimension
        math.ceil(N / BLOCK_M),                  # rows
        math.ceil(N / BLOCK_N),                  # cols
    )

    trimul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        B * H,
        N,
        N,
        stride_lb, stride_lm, stride_lk,
        stride_rb, stride_rm, stride_rk,
        stride_ob, stride_om, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)               # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype (float32) and return
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
