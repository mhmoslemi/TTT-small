"""
TriMul (outgoing) forward pass – Triton‑accelerated mask/gate fusion
and a custom batched GEMM for the O(N³) reduction.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections for left/right and their gates.
4. Fuse (proj * gate * mask) with a Triton kernel (mask is optional).
5. Compute the reduction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a custom batched GEMM Triton kernel (no extra memory copies,
   FP32 accumulation, FP16 storage).
6. Output LayerNorm → out‑gate multiplication → final linear projection.
7. Cast back to the original dtype.

The implementation follows the same semantics as the reference
PyTorch version and supports all configurations required by the
tests.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. Gate‑mask fusion kernel
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fusion_kernel(
    proj_ptr,      # fp16 * (total,)
    gate_ptr,      # fp16 * (total,)
    mask_ptr,      # fp16 * (mask_total,)
    out_ptr,       # fp16 * (total,)
    total_elems,   # int
    hidden_dim,    # int
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # map each element to the corresponding mask element (mask broadcast over hidden)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 2048) -> torch.Tensor:
    """
    Fuse projection, gate and (optional) mask.
    All tensors must be fp16, contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)  (binary 0/1, broadcast over H)
    Returns proj * gate * mask (shape (B,N,N,H)).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_fusion_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# 2. Batched GEMM kernel (C = A @ Bᵀ) with FP32 accumulation
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_az, stride_am, stride_ak,      # batch, m, k strides for A
    stride_bz, stride_bk, stride_bn,      # batch, k, n strides for B (B is accessed transposed)
    stride_cz, stride_cm, stride_cn,      # batch, m, n strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ   (A,B,C are FP16, accumulation in FP32)."""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers for this batch
    a_batch_ptr = a_ptr + pid_batch * stride_az
    b_batch_ptr = b_ptr + pid_batch * stride_bz
    c_batch_ptr = c_ptr + pid_batch * stride_cz

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K‑loop
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)

        # masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (cur_k[None, :] < K)
        b_mask = (cur_k[:, None] < K) & (offs_n[None, :] < N)

        # load A (M×K) and B (K×N) – B is logically transposed
        a = tl.load(a_batch_ptr + (offs_m[:, None] * stride_am
                                   + cur_k[None, :] * stride_ak),
                    mask=a_mask, other=0.0)
        b = tl.load(b_batch_ptr + (cur_k[:, None] * stride_bk
                                   + offs_n[None, :] * stride_bn),
                    mask=b_mask, other=0.0)

        # dot product, accumulation in FP32
        acc += tl.dot(a, b)

    # store result
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_batch_ptr + (offs_m[:, None] * stride_cm
                            + offs_n[None, :] * stride_cn),
             acc.to(tl.float16), mask=c_mask)


def _batched_gemm(a: torch.Tensor, b: torch.Tensor, block_m=64, block_n=64, block_k=32):
    """
    Compute C = A @ Bᵀ for FP16 tensors.
    A, B : (BATCH, M, K) , (BATCH, K, N)
    Returns C : (BATCH, M, N)
    """
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "Inner dimensions must match"

    # strides (in elements)
    stride_az = a.stride(0)
    stride_am = a.stride(1)
    stride_ak = a.stride(2)

    stride_bz = b.stride(0)
    # B is interpreted transposed: swap k and n strides
    stride_bk = b.stride(2)   # original n stride (1)
    stride_bn = b.stride(1)   # original k stride (K)

    c = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

    stride_cz = c.stride(0)
    stride_cm = c.stride(1)
    stride_cn = c.stride(2)

    grid = (batch,
            triton.cdiv(M, block_m),
            triton.cdiv(N, block_n))

    _batched_gemm_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        stride_az, stride_am, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_cm, stride_cn,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return c


# ----------------------------------------------------------------------
# 3. Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"         – input channel dimension C
                  * "hidden_dim"  – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy part
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Load weights (cast to FP16)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # --------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Fuse projection, gate and optional mask via Triton
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction: batched GEMM (C = A @ Bᵀ)
    #    left, right : (B, N, N, H)  ->  (B*H, N, N)
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # move hidden dimension to batch axis for strided GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_flat  = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    # batched GEMM (right is implicitly transposed)
    out_flat = _batched_gemm(left_flat, right_flat,
                             block_m=64, block_n=64, block_k=32)   # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out_perm = out_flat.view(B, H, N, N)          # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 8. Output LayerNorm, gating and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
