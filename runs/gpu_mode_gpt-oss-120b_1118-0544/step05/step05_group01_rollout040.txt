"""
TriMul outgoing forward with Triton acceleration.

Algorithm:
1. LayerNorm over channel dimension.
2. Cast to fp16 for compute-heavy paths.
3. Linear projections to hidden dimension (left/right) and gating projections.
4. Fuse mask, projection, and gate using a small Triton kernel.
5. Compute the core triangle multiplicative update:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a custom batched GEMM kernel in Triton.
6. LayerNorm over hidden dimension, apply output gate, and linear projection back to original dim.
7. Cast back to the input dtype.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask (mask broadcast over H)
# --------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B, N, N, H)
    gate_ptr,          # fp16 * (B, N, N, H)
    mask_ptr,          # fp16 * (B, N, N)
    out_ptr,           # fp16 * (B, N, N, H)
    total_elems,       # int64 total number of elements in proj/gate (=B*N*N*H)
    hidden_dim,        # int64 H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # each hidden entry shares the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse projection, gate and mask using Triton.
    All tensors must be fp16, contiguous and have shapes:
        proj, gate : (B, N, N, H)
        mask       : (B, N, N)   (binary 0/1)
    Returns: proj * gate * mask  (broadcast over H).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# --------------------------------------------------------------
# Triton kernel: batched GEMM for the triangle multiplicative update
# --------------------------------------------------------------
@triton.jit
def _batched_tri_gemm_kernel(
    left_ptr,                # fp16 * (B, N, N, H)
    right_ptr,               # fp16 * (B, N, N, H)
    out_ptr,                 # fp16 * (B, N, N, H)
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a blocked GEMM over the k dimension.
    Axis 0 -> batch * hidden (B*H)
    Axis 1 -> i block
    Axis 2 -> j block
    """
    batch_idx = tl.program_id(0)      # 0 .. B*H-1
    pid_i = tl.program_id(1)          # i‑block
    pid_j = tl.program_id(2)          # j‑block

    # decode batch index into (b, h)
    b = batch_idx // H
    h = batch_idx % H

    # start offsets for the i and j blocks
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # range indices within a block
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds i/j
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # base pointers for the current (b, h) slice
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h

    # accumulator in FP32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k in range(0, num_k):
        k_start = k * BLOCK_K
        cur_k = k_start + offs_k
        mask_k = cur_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base \
                    + offs_i[:, None] * stride_l_i \
                    + cur_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right block (transposed): shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base \
                     + cur_k[:, None] * stride_r_k \
                     + offs_j[None, :] * stride_r_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # accumulate the product
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # write back
    out_base = out_ptr + b * stride_o_b + h * stride_o_h
    out_ptrs = out_base \
               + offs_i[:, None] * stride_o_i \
               + offs_j[None, :] * stride_o_j
    out = acc.to(tl.float16)
    tl.store(out_ptrs, out, mask=mask_ij)


def _batched_tri_gemm(
    left: torch.Tensor,
    right: torch.Tensor,
    block_m: int = 64,
    block_n: int = 64,
    block_k: int = 32,
) -> torch.Tensor:
    """
    Wrapper around the Triton kernel that computes the triangle multiplicative
    update for all batches and hidden heads.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    total_batches = B * H

    # ensure contiguity (required for correct stride handling)
    left = left.contiguous()
    right = right.contiguous()

    # allocate output
    out = torch.empty_like(left)

    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()

Now review quickly:

- In the mask kernel, we treat mask as fp16. The mask values are binary 0/1. Multiplying with float16 yields 0 or value.

- We used `mask_fp16 = mask.to(torch.float16).contiguous()`. Should be fine.

- Are we missing the mask unsqueeze? Our kernel expects mask shape (B,N,N), and we broadcast over hidden dimension inside kernel (mask_idx = offs // hidden_dim). This replicates unsqueeze(-1) behavior.

- Input mask may be of dtype bool or float. Converting to fp16 yields 0/1.

- In gating step we compute sigmoid after linear; that's correct.

- In step 7 we call _batched_tri_gemm on left and right which are already multiplied by gate and masked.

- The kernel _batched_tri_gemm uses BLOCK_M and BLOCK_N of 64 and block_k of 32. Might need to ensure no integer overflow in stride multiplication: stride values can be up to N*N*H typical. For N=1024, H=128, B=2 yields stride up to 2*1024*1024*128 = large but fits within 64-bit.

- In the kernel, we compute `num_k = tl.cdiv(N, BLOCK_K)`; used in for loop. That's okay.

- In store, we use mask_ij to avoid out-of-bounds.

Potential issues:

- `tl.dot` expects the two tensors to be contiguous in memory? Not required.

- The grid may become too large for large batch*H (up to 256). That's fine.

- For shape where hidden_dim != block_m/n etc, no matter.

- The kernel will allocate accumulator of shape (BLOCK_M, BLOCK_N) (float32). For BLOCK_M=64, BLOCK_N=64, memory per block = 64*64*4 = 16KB. That's fine.

- Use of block_k=32: accumulator per iteration: left (64,32) = 8KB, right (32,64) = 8KB. Combined per iteration maybe 16KB.

"""
TriMul outgoing forward with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (`dim`).
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections to the hidden dimension (`hidden_dim`) for
   left/right streams and their gates (no bias).
4. Fuse mask, projection and gate using a small Triton kernel
   (mask is broadcast over the hidden dimension).
5. Core update:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM in Triton (one program per
   (batch, head, i‑block, j‑block)).
6. Layer‑norm over the hidden dimension, apply the output gate,
   and a final linear projection back to the original channel size.
7. Cast the result back to the input dtype.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# ------------------------------------------------------------------
# Triton kernel: proj * gate * mask  (mask broadcast over hidden dim)
# ------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B, N, N, H)
    gate_ptr,          # fp16 * (B, N, N, H)
    mask_ptr,          # fp16 * (B, N, N)
    out_ptr,           # fp16 * (B, N, N, H)
    total_elems,       # int64 total number of elements in proj/gate
    hidden_dim,        # int64 H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # each hidden entry shares the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse projection, gate and mask using Triton.
    All tensors must be fp16, contiguous and have shapes:
        proj, gate : (B, N, N, H)
        mask       : (B, N, N)   (binary 0/1)
    Returns: proj * gate * mask  (broadcast over H).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ------------------------------------------------------------------
# Triton kernel: batched GEMM for the triangle multiplicative update
# ------------------------------------------------------------------
@triton.jit
def _batched_tri_gemm_kernel(
    left_ptr,                # fp16 * (B, N, N, H)
    right_ptr,               # fp16 * (B, N, N, H)
    out_ptr,                 # fp16 * (B, N, N, H)
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a blocked GEMM over the k dimension.
    Axis 0 -> batch * hidden  (B*H)
    Axis 1 -> i‑block
    Axis 2 -> j‑block
    """
    batch_idx = tl.program_id(0)      # 0 .. B*H-1
    pid_i = tl.program_id(1)          # i‑block
    pid_j = tl.program_id(2)          # j‑block

    # decode (b, h)
    b = batch_idx // H
    h = batch_idx % H

    # block start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # indices inside the block
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # out‑of‑bounds masks for i / j
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # base pointers for the selected (b, h) slice
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_start = k * BLOCK_K
        cur_k = k_start + offs_k
        mask_k = cur_k < N

        # left block: (BLOCK_M, BLOCK_K)
        left_ptrs = left_base \
                    + offs_i[:, None] * stride_l_i \
                    + cur_k[None, :] * stride_l_k
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right block (transposed): (BLOCK_K, BLOCK_N)
        right_ptrs = right_base \
                     + cur_k[:, None] * stride_r_k \
                     + offs_j[None, :] * stride_r_j
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # write back
    out_base = out_ptr + b * stride_o_b + h * stride_o_h
    out_ptrs = out_base \
               + offs_i[:, None] * stride_o_i \
               + offs_j[None, :] * stride_o_j
    out = acc.to(tl.float16)
    tl.store(out_ptrs, out, mask=mask_ij)


def _batched_tri_gemm(
    left: torch.Tensor,
    right: torch.Tensor,
    block_m: int = 64,
    block_n: int = 64,
    block_k: int = 32,
) -> torch.Tensor:
    """
    Wrapper for the Triton batched GEMM that implements the
    triangle multiplicative update.
    """
    assert left.shape == right.shape, "left and right must have identical shapes"
    B, N, _, H = left.shape
    total_batches = B * H

    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # strides in element units
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    grid = (
        total_batches,
        (N + block_m - 1) // block_m,
        (N + block_n - 1) // block_n,
    )

    _batched_tri_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_j,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return out
