"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑normalize the 4‑D input tensor.
2. Linear projections (left/right) and their sigmoid gates are computed in fp16.
3. A Triton kernel fuses element‑wise mask (if present) with the gate
   multiplication, reducing memory traffic.
4. The core O(N³·H) contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is performed as a batched matrix multiplication:
       (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N)
   which is highly optimized in cuBLAS.
5. Output LayerNorm, element‑wise output‑gate and final linear projection back
   to the original feature dimension.
All heavy arithmetic is carried out in fp16; the result is cast back to the
input dtype (usually float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,       # fp16*  [B,N,N,H] flattened
    right_ptr,      # fp16*  same shape as left
    left_gate_ptr,  # fp16*  same shape as left
    right_gate_ptr, # fp16*  same shape as left
    mask_ptr,       # fp16*  [B,N,N] flattened
    total_elems,    # i64 scalar = B*N*N*H
    hidden_dim,     # i64 scalar = H
    HAS_MASK: tl.constexpr,   # compile‑time bool
    BLOCK_SIZE: tl.constexpr, # compile‑time int
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask (if present)
        right = right * right_gate * mask (if present)
    The mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load operands
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_g    = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_g   = tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden_dim consecutive entries share one mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left  = left  * left_g  * mask_val
        right = right * right_g * mask_val
    else:
        left  = left  * left_g
        right = right * right_g

    # Store the fused results back in‑place
    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple):
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (tensors)
            - config: dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # -----------------------------------------------------------------
    # Config & flags
    # -----------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for the compute‑heavy part
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections & gates (fp16)
    # -----------------------------------------------------------------
    left     = F.linear(x, left_proj_w)          # (B, N, N, H)
    right    = F.linear(x, right_proj_w)         # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 4. Fuse mask and gate multiplication with a Triton kernel
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask dtype matches computation dtype
        mask_fp = mask.to(device).to(torch.float16)  # (B, N, N)

        # Flatten tensors for the kernel
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left  = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask → simple element‑wise multiplication with gates
        left  = left  * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ rightᵀ) over the sequence dim
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # Reshape to (B*H, N, N) for batched matmul
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the original dtype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
