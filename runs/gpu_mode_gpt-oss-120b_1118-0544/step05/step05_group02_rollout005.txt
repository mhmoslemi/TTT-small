"""
TriMul (outgoing) forward – Triton‑accelerated kernel.

Algorithm
---------
1. Layer‑norm the 4‑D input (B, N, N, C) in float32.
2. Cast to fp16 for the compute‑intensive part.
3. Fuse the two linear projections per side (proj + gate) into a single
   matmul to reduce the number of GEMMs.
4. Apply sigmoid to the gate tensors.
5. If a mask is present, a small Triton kernel fuses
       proj * gate * mask
   broadcasting the mask over the hidden dimension H.
6. Core O(N³·H) reduction:
       out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   is performed as a batched matrix multiplication
   (B·H, N, N) × (B·H, N, N)ᵀ using torch.bmm.
7. Output LayerNorm, multiplication by the output gate and a final linear
   projection back to the original channel dimension.
8. Cast the result back to the input dtype.

At least one Triton kernel (gate‑mask fusion) is used, while the heavy
batched GEMM leverages cuBLAS for maximal throughput.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def gate_mask_kernel(
    proj_ptr,   # *[total]   fp16
    gate_ptr,   # *[total]   fp16
    mask_ptr,   # *[mask]    fp16
    out_ptr,    # *[total]   fp16
    total_elems,  # int
    hidden_dim,   # int
    BLOCK_SIZE: tl.constexpr,
):
    """
    out = proj * gate * mask
    mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Elementwise: proj * gate * mask  (mask broadcast on hidden dim)
    All tensors must be fp16, contiguous and on the same device.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    # flatten mask (shape B*N*N)
    mask_flat = mask.view(-1)

    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],          # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (float32) or None
            - weights: dict of model parameters (all tensors on GPU)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Prepare fp16 weights (ensure contiguity)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16).contiguous()
    left_gate_w   = weights["left_gate.weight"].to(torch.float16).contiguous()
    right_proj_w  = weights["right_proj.weight"].to(torch.float16).contiguous()
    right_gate_w  = weights["right_gate.weight"].to(torch.float16).contiguous()
    out_gate_w    = weights["out_gate.weight"].to(torch.float16).contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16).contiguous()
    to_out_w      = weights["to_out.weight"].to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 4. Fuse proj+gate linear layers per side
    # ------------------------------------------------------------------
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, C)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, C)

    # (B, N, N, 2*H)
    left_comb = F.linear(x, left_comb_w)
    right_comb = F.linear(x, right_comb_w)

    # Split into projection and gate tensors
    left_proj = left_comb[..., :hidden_dim]                      # (B,N,N,H)
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])       # (B,N,N,H)

    right_proj = right_comb[..., :hidden_dim]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])

    # Output gate (B,N,N,H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5. Apply mask + gate (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    use_mask = (mask is not None) and (not nomask)

    if use_mask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core reduction: batched GEMM (left @ rightᵀ) over the k‑dimension
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)ᵀ  -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate multiplication and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 8. Cast back to the original dtype
    # ------------------------------------------------------------------
    out = out.to(dtype_in)
    return out
