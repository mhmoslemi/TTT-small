@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    batch_id = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_id = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = block_id // tl.cdiv(N, BLOCK_N)
    pid_n = block_id % tl.cdiv(N, BLOCK_N)

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for this batch
    a_ptrs = A_ptr + batch_id * stride_Ab + offs_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptrs = B_ptr + batch_id * stride_Bb + tl.arange(0, BLOCK_K)[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k*BLOCK_K + tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k*BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
        # Cast to fp16 before dot if needed; but a,b are half.
        acc += tl.dot(a, b)

        # Advance pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # Write out
    c_ptrs = C_ptr + batch_id * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the 4‑D input (float32 for numeric stability).
2. Cast to fp16 for the compute‑heavy path.
3. Linear projections for left/right tensors and three sigmoid gates.
4. Fuse the gates with an optional mask (simple PyTorch broadcast; a Triton
   kernel is still used for the O(N³) contraction).
5. Compute the core contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication L @ Rᵀ for each hidden channel.
   This is performed by a custom Triton kernel that tiles the matrix
   multiplication and processes all (batch × hidden) groups in one launch.
6. Apply the output layer‑norm, multiply by the output gate, and a final
   linear projection back to the original channel dimension.
7. Return the result in the original dtype (float32).

The implementation keeps the heavy linear algebra in half‑precision while
maintaining numerical stability for the normalisation steps.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers to (batch, M, K), (batch, K, N), (batch, M, N)
    BATCH, M, N, K,                     # dimensions
    stride_Ab, stride_Am, stride_Ak,    # strides for A
    stride_Bb, stride_Bk, stride_Bn,    # strides for B (note: B is accessed as K‑dimension first → Rᵀ)
    stride_Cb, stride_Cm, stride_Cn,    # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ B (where B is already the transposed
    right tensor). Each batch corresponds to one (batch, hidden) pair.
    """
    pid = tl.program_id(0)

    # compute indices of this block
    num_m_blocks = tl.cdiv(M, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_m_blocks * num_n_blocks

    batch_id = pid // blocks_per_batch
    block_id = pid % blocks_per_batch
    pid_m = block_id // num_n_blocks
    pid_n = block_id % num_n_blocks

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for the current batch
    a_batch_off = batch_id * stride_Ab
    b_batch_off = batch_id * stride_Bb
    c_batch_off = batch_id * stride_Cb

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load A block (M x K)
        a_ptrs = A_ptr + a_batch_off \
                 + offs_m[:, None] * stride_Am \
                 + k_offset[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B block (K x N) – B is stored as (K,N) already (right matrix transposed)
        b_ptrs = B_ptr + b_batch_off \
                 + k_offset[:, None] * stride_Bk \
                 + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Cast to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # Write back the result
    c_ptrs = C_ptr + c_batch_off \
             + offs_m[:, None] * stride_Cm \
             + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim" and "hidden_dim" (ints) and optional "nomask"
    Returns:
        torch.Tensor of shape (B, N, N, C) with same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------ #
    norm_w = weights["norm.weight"].to(device, torch.float32)
    norm_b = weights["norm.bias"].to(device, torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # ------------------------------------------------------------------ #
    # 2. Cast to fp16 for the compute‑heavy part
    # ------------------------------------------------------------------ #
    x = x.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 3. Linear projections (bias‑less) and sigmoid gates
    # ------------------------------------------------------------------ #
    left_proj_w  = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device, torch.float16)

    left  = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)  # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------ #
    # 4. Fuse gates with optional mask (simple broadcast; cheap)
    # ------------------------------------------------------------------ #
    if mask is not None and not nomask:
        # mask: (B,N,N) -> (B,N,N,1)
        mask_fp = mask.to(device, torch.float16).unsqueeze(-1)
        left  = left  * left_gate  * mask_fp
        right = right * right_gate * mask_fp
    else:
        left  = left  * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Core contraction: batched GEMM L @ Rᵀ using Triton
    # ------------------------------------------------------------------ #
    B, N, _, H = left.shape          # (B, N, N, hidden_dim)
    # reshape to (B*H, N, N) for the kernel
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (batch, M, K)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (batch, K, N)

    batch = B * H
    M = N
    K = N
    N_out = N

    # Allocate output tensor
    out_mat = torch.empty((batch, M, N_out), dtype=torch.float16, device=device)

    # Tuning parameters – chosen to fit into shared memory comfortably
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_m_blocks = (M + BLOCK_M - 1) // BLOCK_M
    num_n_blocks = (N_out + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch * num_m_blocks * num_n_blocks
    grid = (total_blocks,)

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch, M, N_out, K,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 6. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------ #
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate                     # apply output gate
    to_out_w = weights["to_out.weight"].to(device, torch.float16)
    out = F.linear(out, to_out_w)            # (B,N,N,dim)

    # ------------------------------------------------------------------ #
    # 7. Cast back to the original dtype (usually float32) and return
    # ------------------------------------------------------------------ #
    out = out.to(dtype_in)
    return out
