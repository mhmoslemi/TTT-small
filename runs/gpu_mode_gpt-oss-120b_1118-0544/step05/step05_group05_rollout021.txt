@triton.jit
def _apply_gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr, total_elems,
    hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # compute mask index: each hidden_dim entries per mask element
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

Return.

"""
TriMul outgoing update (AlphaFold3) forward pass.

The implementation follows the reference PyTorch module but fuses several
element‑wise operations (mask, gate, and projection scaling) in a Triton
kernel and reduces the number of linear layers by concatenating the
projection and gate weights.  The core triangle multiplicative contraction
is performed as a batched GEMM (B*H independent matrix multiplications)
using Torch’s highly optimized cuBLAS kernels.  All heavy arithmetic is
done in FP16; the final result is returned in the input dtype (typically
FP32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,               # [total] FP16 tensors
    left_gate_ptr, right_gate_ptr,     # [total] FP16 tensors
    mask_ptr,                         # [mask_len] FP16 tensor (optional)
    total_elems: tl.int64,            # total number of elements in left/right
    hidden_dim: tl.int64,             # size of the hidden dimension H
    HAS_MASK: tl.constexpr,           # compile‑time bool
    BLOCK_SIZE: tl.constexpr,         # compile‑time int
):
    """
    Fuse element‑wise operations:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    Mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left      = tl.load(left_ptr      + offs, mask=active, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden_dim entries share the same mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor (torch.Tensor): shape (B, N, N, C)
            - mask (torch.Tensor or None): shape (B, N, N)
            - weights (dict): model parameters
            - config (dict): must contain "dim" and "hidden_dim";
              optional "nomask" (bool)

    Returns:
        torch.Tensor: shape (B, N, N, C), same dtype as input_tensor.
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # Load weights (convert to appropriate dtype)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)

    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)   # (H, C)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)  # (H, C)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)   # (H, C)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)  # (H, C)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)    # (H, C)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)    # (H,)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)      # (C, H)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability) and cast to fp16
    # -----------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Combine projection + gate weights to reduce number of matmuls
    # -----------------------------------------------------------------
    left_comb_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2H, C)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0) # (2H, C)

    # Linear projections: produce (B, N, N, 2H)
    left_comb = F.linear(x, left_comb_w)    # (B, N, N, 2H)
    right_comb = F.linear(x, right_comb_w)  # (B, N, N, 2H)

    # Split into projection and gate parts
    H = hidden_dim
    left_proj = left_comb[..., :H]                     # (B, N, N, H)
    left_gate = torch.sigmoid(left_comb[..., H:])      # (B, N, N, H)
    right_proj = right_comb[..., :H]                   # (B, N, N, H)
    right_gate = torch.sigmoid(right_comb[..., H:])    # (B, N, N, H)

    # -----------------------------------------------------------------
    # 3. Output gate (computed after the contraction)
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 4. Fuse mask (if any) and gate multiplication via Triton
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and flatten
        mask_fp = mask.to(device).to(torch.float16)
        # Flatten tensors for the kernel
        left_flat      = left_proj.reshape(-1)
        right_flat     = right_proj.reshape(-1)
        left_gate_flat = left_gate.reshape(-1)
        right_gate_flat= right_gate.reshape(-1)
        mask_flat      = mask_fp.reshape(-1)  # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            H,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left = left_flat.view_as(left_proj)
        right = right_flat.view_as(right_proj)
    else:
        # No mask: simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM over the sequence dimension
    #    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, N, N, H) -> (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # Batched matrix multiply: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)      # (B, N, N, C)

    # Cast back to the original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
