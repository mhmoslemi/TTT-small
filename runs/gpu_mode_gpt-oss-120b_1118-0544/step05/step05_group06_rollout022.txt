"""
TriMul (outgoing) forward pass optimized with Triton.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Cast to fp16 for compute‑intensive stages.
3. Linear projections (left/right) and gating (left/right/out) using
   fp16 weights; apply sigmoid to gates.
4. Fuse mask (if provided) and gate by simple broadcasting multiplication.
5. Core O(N³·H) reduction:
   For each hidden channel h we need  out[b,i,j,h] = Σₖ left[b,i,k,h] *
   right[b,j,k,h].  This is a batched matrix multiplication
   L @ Rᵀ where L,R ∈ ℝ^{N×N}.  We implement a fused Triton kernel
   that processes all (batch × hidden) matrices in parallel using
   tiled GEMM with fp16 tensor‑core dot products and fp32 accumulation.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   out‑gate, and project back to the original channel dimension.
7. Cast the result to the original dtype (usually float32).

The heavy reduction is performed entirely in Triton, satisfying the
requirement to use a custom kernel.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ Bᵀ) for many independent matrices.
# A: (B*H, N, N)  – row‑major
# B: (B*H, N, N)  – row‑major, but interpreted as transposed inside the kernel
# C: (B*H, N, N)  – result
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix dims (all equal to seq_len)
    stride_am, stride_ak, stride_a_batch,   # A strides
    stride_bk, stride_bn, stride_b_batch,   # B strides (note: B is read as transposed)
    stride_cm, stride_cn, stride_c_batch,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # batch * hidden index
    pid_m = tl.program_id(1)              # block row
    pid_n = tl.program_id(2)              # block col

    # Compute start offsets for the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Compute pointers for this tile
        a_ptrs = a_ptr + pid_batch * stride_a_batch \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_batch * stride_b_batch \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Load tiles (fp16)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Accumulate dot product (uses tensor cores when possible)
        acc += tl.dot(a, b)

    # Write back the result with proper masking
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_c_batch \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, c, mask=c_mask)


def _batched_gemm(a: torch.Tensor, b: torch.Tensor, block_m=64, block_n=64, block_k=64) -> torch.Tensor:
    """
    Batched GEMM where each slice performs C = A @ Bᵀ.
    A, B: fp16 tensors of shape (batch, N, N)
    Returns C of the same shape and dtype.
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, N, K = a.shape
    assert K == N, "square matrices expected"

    # Output tensor
    c = torch.empty_like(a)

    # Strides
    stride_am, stride_ak, stride_a_batch = a.stride()
    stride_bm, stride_bn, stride_b_batch = b.stride()   # original row/col strides

    # For Bᵀ we swap the inner strides
    stride_bk = stride_bn        # stride for K dimension (original column stride)
    stride_bn = stride_bm        # stride for N dimension (original row stride)

    # Output strides (same layout as A)
    stride_cm, stride_cn, stride_c_batch = c.stride()

    grid = (batch, math.ceil(N / block_m), math.ceil(N / block_n))
    _batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_a_batch,
        stride_bk,
        stride_bn,
        stride_b_batch,
        stride_cm,
        stride_cn,
        stride_c_batch,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return c


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy stages
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all linear weights to fp16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections
    # --------------------------------------------------------------
    left_proj = F.linear(x_fp16, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Apply mask (if any) and combine with gates
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Cast mask to fp16 and broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core O(N³·H) reduction via batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # left shape: (B, N, N, H)

    # Rearrange to (B*H, N, N) for the batched kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Perform batched GEMM with Triton
    out_flat = _batched_gemm(left_flat, right_flat, block_m=64, block_n=64, block_k=64)

    # Restore original layout (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
