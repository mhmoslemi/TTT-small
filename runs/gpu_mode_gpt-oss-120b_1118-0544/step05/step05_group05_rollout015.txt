import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    The implementation follows the PyTorch reference but fuses the
    element‑wise mask and gate multiplications in a Triton kernel
    and uses a batched GEMM to evaluate the O(N³·H) contraction.
    All heavy arithmetic is performed in fp16; the final result
    is returned in the input dtype (usually float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # ---------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for the rest of the operations
    x = x.to(torch.float16)

    # ---------------------------------------------------------------
    # 2. Load linear weights (fp16)
    # ---------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # ---------------------------------------------------------------
    # 3. Linear projections + gates (still fp16)
    # ---------------------------------------------------------------
    left = F.linear(x, left_proj_w)                 # (B, N, N, H)
    right = F.linear(x, right_proj_w)               # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # ---------------------------------------------------------------
    # 4. Fuse mask & gate multiplication with Triton
    # ---------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape: (B, N, N)
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # Flatten tensors for kernel
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024  # reasonable block size

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – just multiply by gates
        left = left * left_gate
        right = right * right_gate

    # ---------------------------------------------------------------
    # 5. Core contraction via batched GEMM (left @ rightᵀ)
    # ---------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matmul
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------------------------------------------------------------
    # 6. Output LayerNorm, gate, final projection
    # ---------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # Cast back to original dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,          # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,
    mask_ptr,                     # [B,N,N] flattened, broadcast over H
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,       # compile‑time bool
    BLOCK_SIZE: tl.constexpr,     # compile‑time int
):
    """In‑place fusion of mask and gate multiplication."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left  = tl.load(left_ptr  + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    lgate = tl.load(left_gate_ptr  + offs, mask=active, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # each group of `hidden_dim` entries shares one mask value
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * lgate * mask_val
        right = right * rgate * mask_val
    else:
        left  = left  * lgate
        right = right * rgate

    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Processed tensor of shape (B, N, N, dim) matching the dtype of
        ``input_tensor``.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------
    # 2. Cast to fp16 for heavy computation
    # -----------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------
    # 3. Load linear weights (fp16)
    # -----------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # -----------------------------------------------------------
    # 4. Linear projections and gate values (fp16)
    # -----------------------------------------------------------
    left = F.linear(x, left_proj_w)                 # (B, N, N, H)
    right = F.linear(x, right_proj_w)               # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------
    # 5. Fuse mask and gate multiplication in Triton
    # -----------------------------------------------------------
    if mask is not None and not nomask:
        # mask broadcasted over the hidden dimension
        mask_fp = mask.to(device).to(torch.float16).contiguous()   # (B,N,N)

        # Flatten tensors for the kernel
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.view(-1)                         # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024  # reasonable block size for fp16

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back to original layout
        left  = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask: simple element‑wise gate multiplication
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------
    # 6. Core contraction: batched GEMM (left @ rightᵀ)
    # -----------------------------------------------------------
    B, N, _, H = left.shape
    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (each hidden slice independent)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------
    # 7. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # -----------------------------------------------------------
    # 8. Cast back to the input dtype (usually float32)
    # -----------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```

This solution respects the required interface, uses a Triton kernel for the mask‑gate fusion, leverages a batched GEMM for the O(N³·H) contraction, and keeps the final output in the original data type.
