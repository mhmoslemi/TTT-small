@triton.jit
def batched_gemm_kernel(
    A, B, C,
    stride_aba, stride_amm, stride_ak,
    stride_bba, stride_bbk, stride_bbn,
    stride_cca, stride_ccm, stride_ccn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)
    block_m = tl.program_id(1)
    block_n = tl.program_id(2)

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid coordinates
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # pointers for A and B (starting points for this block)
    a_ptrs = A + batch * stride_aba + offs_m[:, None] * stride_amm + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B + batch * stride_bba + tl.arange(0, BLOCK_K)[:, None] * stride_bbk + offs_n[None, :] * stride_bbn

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K - k*BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] < K - k*BLOCK_K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)

        # increment pointers to next k block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bbk

    c_ptrs = C + batch * stride_cca + offs_m[:, None] * stride_ccm + offs_n[None, :] * stride_ccn

"""
custom_kernel: Triton‑accelerated outgoing TriMul forward.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for all subsequent heavy compute.
3. Linear projections (left/right) and their gates (sigmoid) using the
   provided weights.
4. Optional mask multiplication (broadcast over the hidden dimension).
5. Core reduction `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   Implemented as a batched GEMM (`left @ right^T`) in a custom Triton
   kernel that processes each `(batch, hidden)` pair as an independent
   GEMM tile.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and final linear projection back to `dim`.
7. Cast the result back to the input dtype (usually float32).

The kernel fuses the O(N³) reduction and provides a large speed‑up
relative to a pure PyTorch `bmm` implementation.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B^T) for TriMul reduction
# --------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A,                # fp16 * (batch, M, K)
    B,                # fp16 * (batch, K, N)  (already transposed)
    C,                # fp16 * (batch, M, N)  output
    stride_aba, stride_amm, stride_ak,   # strides for A
    stride_bba, stride_bbk, stride_bbn,   # strides for B
    stride_cca, stride_ccm, stride_ccn,   # strides for C
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds checking
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the tile in A and B
    a_ptrs = A + batch_id * stride_aba \
                 + offs_m[:, None] * stride_amm \
                 + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B + batch_id * stride_bba \
                 + tl.arange(0, BLOCK_K)[:, None] * stride_bbk \
                 + offs_n[None, :] * stride_bbn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A and B tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K - k*BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] < K - k*BLOCK_K) & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

        # advance pointers to next K‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bbk

    # write back
    c = acc.to(tl.float16)
    c_ptrs = C + batch_id * stride_cca \
                 + offs_m[:, None] * stride_ccm \
                 + offs_n[None, :] * stride_ccn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def _batched_gemm(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 64,
    BLOCK_N: int = 64,
    BLOCK_K: int = 32,
) -> torch.Tensor:
    """
    Compute C = A @ B   where
        A : (batch, M, K)
        B : (batch, K, N)
    using a Triton kernel. All tensors must be fp16 and contiguous.
    Returns C of shape (batch, M, N) in fp16.
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, Kb, N = B.shape
    assert K == Kb

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_aba, stride_amm, stride_ak = A.stride()
    stride_bba, stride_bbk, stride_bbn = B.stride()
    stride_cca, stride_ccm, stride_ccn = C.stride()

    grid = (batch, triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N))
    _batched_gemm_kernel[grid](
        A, B, C,
        stride_aba, stride_amm, stride_ak,
        stride_bba, stride_bbk, stride_bbn,
        stride_cca, stride_ccm, stride_ccn,
        M, N, K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C

# --------------------------------------------------------------
# Custom kernel entry point
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict with model weights (float32)
            - config: dict with keys "dim" and "hidden_dim" (int), optional key "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast weights to fp16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (no bias) and gates
    # --------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 5. Apply mask and gating
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B, N, N) -> (B, N, N, 1) for broadcasting
        mask = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Release intermediate tensors no longer needed
    del left_proj, right_proj, left_gate, right_gate, x

    # --------------------------------------------------------------
    # 6. Core reduction: batched GEMM (left @ right^T) across hidden dim
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    batch_total = B * H

    # reshape to (batch_total, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(batch_total, N, N)
    # transpose right before flattening so that we compute left @ right^T
    right_t = right.permute(0, 3, 2, 1).contiguous().view(batch_total, N, N)

    # Triton batched GEMM
    out_flat = _batched_gemm(left_flat, right_t, BLOCK_M=64, BLOCK_N=64, BLOCK_K=32)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Output LayerNorm, gate, final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 8. Cast back to the original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim)

    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
