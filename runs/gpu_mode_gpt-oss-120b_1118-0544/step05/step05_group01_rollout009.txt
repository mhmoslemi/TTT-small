@triton.jit
def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0) # linear over (batch, block_m, block_n)
    # compute batch index and block row/col
    batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_index = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = block_index // tl.cdiv(N, BLOCK_N)
    pid_n = block_index % tl.cdiv(N, BLOCK_N)
    # calculate offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # compute a and b pointers for this block, using batch offset
        a_ptrs = a_ptr + batch * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)
        # cast to fp32
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)
    # write output
    c_ptrs = c_ptr + batch * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program id
    pid = tl.program_id(0)
    # compute grid dims
    total_m = tl.cdiv(M, BLOCK_M)
    total_n = tl.cdiv(N, BLOCK_N)
    batch_idx = pid // (total_m * total_n)
    block_idx = pid % (total_m * total_n)
    pid_m = block_idx // total_n
    pid_n = block_idx % total_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # for boundary
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # a ptr: batch offset + row offset * stride_am + k offset * stride_ak
        a_ptrs = a_ptr + batch_idx * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak

    c_ptrs = c_ptr + batch_idx * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

"""
custom_kernel: Triton‑accelerated outgoing TriMul forward pass.

Algorithm
---------
1. Layer‑norm the input along the feature axis.
2. Cast to FP16 for heavy compute.
3. Linear projections (left/right) and gating projections (left/right/out)
   are performed with torch.nn.functional.linear.
4. If a mask is present, fuse mask, projection and gate element‑wise
   using a small Triton kernel.
5. Core O(N³) reduction:
   For each batch and hidden slice compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is implemented as a batched GEMM (left @ rightᵀ) with a custom
   Triton kernel that processes many independent N×N matrix multiplications
   in parallel.
6. Layer‑norm over the hidden dimension, multiply by the output gate and
   apply a final linear projection back to the original dimension.
7. Cast the result back to the input dtype (usually FP32).

Only the mask‑gate fusion and the batched matrix multiplication are
implemented in Triton; the remaining operations use highly‑optimized
PyTorch kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask (mask broadcast over hidden)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,      # fp16 * (B,N,N,H)
    gate_ptr,      # fp16 * (B,N,N,H)
    mask_ptr,      # fp16 * (B,N,N)
    out_ptr,       # fp16 * (B,N,N,H)
    total_elems,   # B*N*N*H
    hidden_dim,    # H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    mask_idx = offs // hidden_dim                           # same mask for H values

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    proj, gate : (B, N, N, H)  fp16
    mask       : (B, N, N)    fp16 (binary 0/1)
    Returns proj * gate * mask  (broadcast over H)
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B) where A,B shape = (batch, M, K) and (batch, K, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # grid dimensions
    total_m = tl.cdiv(M, BLOCK_M)
    total_n = tl.cdiv(N, BLOCK_N)

    batch_idx = pid // (total_m * total_n)
    block_idx = pid % (total_m * total_n)
    pid_m = block_idx // total_n
    pid_n = block_idx % total_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers with batch offset
        a_ptrs = a_ptr + batch_idx * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch_idx * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # compute small matrix multiply; result in fp32
        acc += tl.dot(a, b)

    # write result
    c_ptrs = c_ptr + batch_idx * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul(A: torch.Tensor, B: torch.Tensor,
                    block_m: int = 64, block_n: int = 64, block_k: int = 64) -> torch.Tensor:
    """
    Batched GEMM: C = A @ B   where
        A : (batch, M, K)
        B : (batch, K, N)
    All tensors must be contiguous, fp16 and on the same CUDA device.
    Returns C of shape (batch, M, N) (fp16).
    """
    assert A.shape[0] == B.shape[0]
    assert A.shape[2] == B.shape[1]

    batch, M, K = A.shape
    _, _, N = B.shape

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # strides in element units
    stride_am, stride_ak = A.stride()[1], A.stride()[2]
    stride_ah = A.stride()[0]

    stride_bk, stride_bn = B.stride()[1], B.stride()[2]
    stride_bh = B.stride()[0]

    stride_cm, stride_cn = C.stride()[1], C.stride()[2]
    stride_ch = C.stride()[0]

    grid = (batch * triton.cdiv(M, block_m) * triton.cdiv(N, block_n),)

    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_ah,
        stride_bk,
        stride_bn,
        stride_bh,
        stride_cm,
        stride_cn,
        stride_ch,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return C


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C]
            - mask         : torch.Tensor of shape [B, N, N] (or None)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over dim
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)  float32

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)    # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Apply mask and gating (fused via Triton when mask present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is fp16, contiguous and broadcastable
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    This is batched GEMM across the hidden dimension.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for A and (B*H, N, N) for Bᵀ
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # (batch, M, K)
    right_flat_t = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)      # (batch, K, N)

    # batched matrix multiplication via Triton
    out_flat = _batched_matmul(left_flat, right_flat_t,
                               block_m=64, block_n=64, block_k=64)           # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()            # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate multiplication, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
