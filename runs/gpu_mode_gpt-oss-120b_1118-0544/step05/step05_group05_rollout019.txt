```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # Load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    if HAS_MASK:
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    Implements the sequence:
      1. LayerNorm on input.
      2. Linear projections (left/right) and sigmoid gates.
      3. Optional mask + gate fusion via a Triton kernel.
      4. Batched matrix multiply over the sequence dimension (left @ right^T)
         performed per hidden channel.
      5. Output LayerNorm, gating and final linear projection back to the original
         feature dimension.

    All heavy arithmetic is carried out in fp16; the final result is cast back
    to the dtype of the input tensor.
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float32)
    # Using torch.float32 for layernorm regardless of input dtype
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Convert to fp16 for compute‑intensive ops
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, dtype=torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gates (fp16)
    # -----------------------------------------------------------------
    # Shape: (B, N, N, hidden_dim)
    left = F.linear(x, left_proj_w)       # left_proj
    right = F.linear(x, right_proj_w)     # right_proj

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 4. Fuse mask and gates with a Triton kernel (if mask present)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape: (B, N, N)
        mask_fp = mask.to(device, dtype=torch.float16)
        # Flatten tensors for kernel processing
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)  # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 131072  # 128K elements per program; adjust for occupancy
        grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask: just multiply by gates
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ rightᵀ) per hidden channel
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # Prepare for batched matmul: (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # Compute left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # shape (B, N, N, dim)

    # Cast back to original dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32 for stability).
2. Compute left/right projections and their sigmoid gates (fp16).
3. Fuse optional mask, gate and projection multiplication in a Triton kernel
   to cut memory traffic.
4. Core O(N³·H) contraction performed as a batched matrix multiply
   (left @ rightᵀ) over the sequence dimension, using torch.matmul which
   maps to cuBLAS/Tensor‑core GEMM.
5. Apply output LayerNorm, output gate and a final linear projection back
   to the original feature dimension.  All heavy arithmetic stays in fp16;
   the result is cast back to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,          # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,
    mask_ptr,                     # [B,N,N] flattened (broadcast over H)
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    If HAS_MASK is False the mask term is omitted.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden_dim entries share a mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for compute‑heavy parts
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, dtype=torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gates (all fp16)
    # -----------------------------------------------------------------
    # (B, N, N, hidden_dim)
    left = F.linear(x, left_proj_w)
    right = F.linear(x, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 4. Fuse mask + gate multiplication via Triton (if mask present)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device, dtype=torch.float16)
        # Flatten everything for the kernel
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)  # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Restore original shapes
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask → just apply gates
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ rightᵀ) per hidden channel
    # -----------------------------------------------------------------
    # Shape after permute: (B, H, N, N)
    left_mat = left.permute(0, 3, 1, 2)
    right_mat = right.permute(0, 3, 1, 2)

    # Batched matrix multiplication (uses cuBLAS/Tensor‑core GEMM)
    out = torch.matmul(left_mat, right_mat.transpose(-1, -2))   # (B, H, N, N)

    # Back to (B, N, N, H)
    out = out.permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # Cast back to the original dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
