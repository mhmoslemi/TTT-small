def fuse_proj_gate_mask_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, H,
    total_elements,
    BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    stride_b = N * N * H
    stride_i = N * H
    stride_j = H

    b = tl.div(offs, stride_b)
    rem = offs - b * stride_b

    i = tl.div(rem, stride_i)
    rem2 = rem - i * stride_i

    j = tl.div(rem2, stride_j)
    h = rem2 - j * stride_j

    # offsets

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fuse_gate_mask_inplace_kernel(proj1_ptr, gate1_ptr,
                                  proj2_ptr, gate2_ptr,
                                  mask_ptr,
                                  B, N, H,
                                  total_elements,
                                  BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # compute strides
    stride_b = N * N * H
    stride_i = N * H
    stride_j = H

    b = tl.div(offs, stride_b)
    rem = offs - b * stride_b
    i = tl.div(rem, stride_i)
    rem2 = rem - i * stride_i
    j = tl.div(rem2, stride_j)
    h = rem2 - j * stride_j

    idx = b * stride_b + i * stride_i + j * stride_j + h
    mask_idx = b * (N * N) + i * N + j

    # load values
    proj1 = tl.load(proj1_ptr + idx, mask=mask, other=tl.float16(0.0))
    gate1 = tl.load(gate1_ptr + idx, mask=mask, other=tl.float16(0.0))
    proj2 = tl.load(proj2_ptr + idx, mask=mask, other=tl.float16(0.0))
    gate2 = tl.load(gate2_ptr + idx, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))

    # compute gated & masked
    out1 = proj1 * gate1 * m
    out2 = proj2 * gate2 * m

    tl.store(proj1_ptr + idx, out1, mask=mask)
    tl.store(proj2_ptr + idx, out2, mask=mask)

def custom_kernel(data):
    '''
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

    The implementation fuses:
    * two linear projections (left/right) into a single matmul,
    * two gate projections (left/right) into a single matmul,
    * element‑wise gate, mask and projection multiplication via a Triton kernel,
    * the O(N³) reduction using a batched half‑precision GEMM (torch.bmm).

    All heavy kernels run in fp16; the final result is cast back to the
    input dtype.
    '''
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input layernorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # 2. Cast weights to fp16
    dtype = torch.float16
    left_proj_w = weights["left_proj.weight"].to(dtype)
    right_proj_w = weights["right_proj.weight"].to(dtype)
    left_gate_w = weights["left_gate.weight"].to(dtype)
    right_gate_w = weights["right_gate.weight"].to(dtype)
    out_gate_w = weights["out_gate.weight"].to(dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)
    to_out_w = weights["to_out.weight"].to(dtype)

    # 3. Combined linear projections (proj and gate) to reduce kernel launches
    # concat left/right projection weights
    proj_w = torch.cat([left_proj_w, right_proj_w], dim=0)          # (2*H, dim)
    gate_w = torch.cat([left_gate_w, right_gate_w], dim=0)         # (2*H, dim)

    # linear: (B,N,N,dim) -> (B,N,N,2*H)
    proj = F.linear(x, proj_w)   # (B, N, N, 2*H)
    gate = F.linear(x, gate_w)   # (B, N, N, 2*H)

    # split
    left_proj, right_proj = torch.split(proj, hidden_dim, dim=-1)
    left_gate, right_gate = torch.split(gate, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # 4. Output gate
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,H)

    # 5. Apply mask & gating (fused kernel). If mask is absent or nomask, skip mask.
    if mask is not None and not nomask:
        mask_fp16 = mask.to(dtype).contiguous()
        # Ensure contiguous memory
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left_proj.numel()
        BLOCK = 1024
        grid = ( (total + BLOCK - 1) // BLOCK, )
        fuse_gate_mask_inplace_kernel[grid](
            left_proj, left_gate,
            right_proj, right_gate,
            mask_fp16,
            left_proj.shape[0], left_proj.shape[1], hidden_dim,
            total,
            BLOCK_SIZE=BLOCK,
        )
        # left_proj and right_proj are now gated & masked in‑place
        left = left_proj
        right = right_proj
    else:
        # simple elementwise multiplication (mask omitted)
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 6. Core O(N³) reduction using batched GEMM (half‑precision)
    B, N, _, H = left.shape
    # reshape for batched matmul
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N).T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 7. Output layernorm
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 8. Apply out gate
    out = out * out_gate

    # 9. Final projection back to dim
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # 10. Cast back to original dtype
    out = out.to(input_tensor.dtype)
    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

The kernel fuses:
  * two linear projections (left/right) into one matmul,
  * two gate projections (left/right) into one matmul,
  * element‑wise gating, masking and projection multiplication via a Triton
    kernel (in‑place on the projection tensors).

The O(N³) reduction is performed with a half‑precision batched GEMM
( torch.bmm ) which is already highly optimized on NVIDIA GPUs.
All heavy computation runs in fp16; the final result is cast back to the
input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_inplace_kernel(
    proj1_ptr,   # *[total_elems]   fp16   (left projection)
    gate1_ptr,   # *[total_elems]   fp16   (left gate)
    proj2_ptr,   # *[total_elems]   fp16   (right projection)
    gate2_ptr,   # *[total_elems]   fp16   (right gate)
    mask_ptr,    # *[mask_elems]    fp16   (B,N,N)
    B, N, H,    # int32 scalars
    total_elems,  # int32
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place: proj ← proj * gate * mask   (applied to left & right)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # strides for tensors of shape (B, N, N, H) in row‑major order
    stride_b = N * N * H
    stride_i = N * H
    stride_j = H

    # decode (b, i, j, h) from linear index
    b = tl.div(offs, stride_b)
    rem = offs - b * stride_b
    i = tl.div(rem, stride_i)
    rem2 = rem - i * stride_i
    j = tl.div(rem2, stride_j)
    h = rem2 - j * stride_j

    idx = b * stride_b + i * stride_i + j * stride_j + h          # offset in proj/gate
    mask_idx = b * (N * N) + i * N + j                           # offset in mask

    # load values (masked loads return 0 / 1 for out‑of‑range elements)
    p1 = tl.load(proj1_ptr + idx, mask=active, other=0.0)
    g1 = tl.load(gate1_ptr + idx, mask=active, other=0.0)
    p2 = tl.load(proj2_ptr + idx, mask=active, other=0.0)
    g2 = tl.load(gate2_ptr + idx, mask=active, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)

    # gated & masked result
    out1 = p1 * g1 * m
    out2 = p2 * g2 * m

    # write back in‑place
    tl.store(proj1_ptr + idx, out1, mask=active)
    tl.store(proj2_ptr + idx, out2, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    data = (input_tensor, mask, weights, config)
    Returns tensor of shape [bs, seq_len, seq_len, dim] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 → fp16)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Cast all weights to fp16
    # --------------------------------------------------------------
    dtype = torch.float16
    left_proj_w = weights["left_proj.weight"].to(dtype)
    right_proj_w = weights["right_proj.weight"].to(dtype)
    left_gate_w = weights["left_gate.weight"].to(dtype)
    right_gate_w = weights["right_gate.weight"].to(dtype)
    out_gate_w = weights["out_gate.weight"].to(dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)
    to_out_w = weights["to_out.weight"].to(dtype)

    # --------------------------------------------------------------
    # 3. Combined projections to reduce kernel launches
    # --------------------------------------------------------------
    proj_w = torch.cat([left_proj_w, right_proj_w], dim=0)      # (2*H, dim)
    gate_w = torch.cat([left_gate_w, right_gate_w], dim=0)      # (2*H, dim)

    # (B, N, N, 2*H)
    proj = F.linear(x, proj_w)
    gate = F.linear(x, gate_w)

    # split projections and gates
    left_proj, right_proj = torch.split(proj, hidden_dim, dim=-1)
    left_gate, right_gate = torch.split(gate, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # --------------------------------------------------------------
    # 4. Output gate (separate linear)
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 5. Fuse mask, gates and projections (in‑place)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(dtype).contiguous()
        # Ensure tensors are contiguous before passing to Triton
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left_proj.numel()
        BLOCK = 1024
        grid = ( (total + BLOCK - 1) // BLOCK, )
        _gate_mask_inplace_kernel[grid](
            left_proj, left_gate,
            right_proj, right_gate,
            mask_fp16,
            left_proj.shape[0], left_proj.shape[1], hidden_dim,
            total,
            BLOCK_SIZE=BLOCK,
        )
        left = left_proj
        right = right_proj
    else:
        # simple element‑wise multiplication (mask omitted)
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6. Core O(N³·H) reduction via batched half‑precision GEMM
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T  → (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Output LayerNorm, output gate and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 8. Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
