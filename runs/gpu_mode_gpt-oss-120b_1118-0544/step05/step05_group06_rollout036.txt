def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, N,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bk, stride_bn,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    
    # mask for out-of-range
    mask_m = offs_m < N
    mask_n = offs_n < N

    a_ptrs = a_ptr + pid_h * stride_ah + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_h * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load A and B
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < N), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < N) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
    # convert to fp16
    acc = acc.to(tl.float16)

    # store
    c_ptrs = c_ptr + pid_h * stride_ch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj

TriMul forward (outgoing) implementation in Triton.
Algorithm:
1. Layer‑normalize the 4‑D input.
2. FP16 cast for the heavy part.

   C = A @ Bᵀ for each hidden channel using a custom Triton GEMM kernel
   (tri_mul_kernel).  This replaces the O(N³·H) einsum.
6. Reshape back to (B, N, N, H), apply the second LayerNorm, multiply by the
   out‑gate and a final linear projection.
7. Cast the result back to the original dtype.
All heavy compute (steps 4‑5) is performed on the GPU with Triton kernels.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Kernel 1: fuse projection * gate * mask (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,                # *[total_elems]   fp16
    gate_ptr,                # *[total_elems]   fp16
    mask_ptr,                # *[mask_elems]    fp16
    out_ptr,                 # *[total_elems]   fp16
    total_elems,             # i64
    hidden_dim,              # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    mask_idx = offs // hidden_dim                     # broadcast over hidden dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Element‑wise ``proj * gate * mask`` where ``mask`` has shape (B, N, N)
    and is broadcast over the hidden dimension H.
    All tensors must be FP16 and contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)          # (B*N*N,)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],                              # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the core O(N³·H) reduction
# Compute C = A @ Bᵀ for each batch (B·H) where A, B, C are (batch, N, N)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    batch, N,                          # scalars
    stride_ah, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # B strides (batch, k, j)  (B is transposed)
    stride_ch, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # batch index
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block col

    # --------------------------------------------------------------
    # Compute offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the current tiles of A and Bᵀ
    a_ptrs = a_ptr + pid_batch * stride_ah + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_batch * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the K dimension
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_tiles):
        k_offset = k * BLOCK_K
        mask_k = (k_offset + offs_k) < N

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Store the result (cast to fp16)
    # --------------------------------------------------------------
    c_ptrs = c_ptr + pid_batch * stride_ch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict with LayerNorm and Linear parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy stages
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all weights to fp16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B, N, N, H)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Fuse projection * gate * optional mask
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right, right_gate, mask_fp16)
    else:
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 7. Core O(N³·H) reduction using a batched GEMM Triton kernel
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # Strides (all tensors are contiguous)
    a_batch_stride, a_i_stride, a_k_stride = left_flat.stride()   # (N*N, N, 1)
    b_batch_stride, b_i_stride, b_k_stride = right_flat.stride()  # (N*N, N, 1)
    c_batch_stride, c_i_stride, c_j_stride = out_flat.stride()    # same as A

    # Block sizes – tuned for typical B200 memory hierarchy
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,
        math.ceil(N / BLOCK_M),
        math.ceil(N / BLOCK_N),
    )
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        a_batch_stride,
        a_i_stride,
        a_k_stride,
        b_batch_stride,
        # for Bᵀ we swap the last two strides: k‑stride = column stride (1),
        # n‑stride = row stride (N)
        b_k_stride,
        b_i_stride,
        c_batch_stride,
        c_i_stride,
        c_j_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, gating, and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)                     # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9. Cast back to the original dtype
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
