def outgoing_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
   pid0 = tl.program_id(0)  # head idx
   pid1 = tl.program_id(1)  # block_i
   pid2 = tl.program_id(2)  # block_j

   # compute batch & hidden index
   total_heads = B * H
   b = pid0 // H
   h = pid0 % H

   # compute i, j base
   i_base = pid1 * BLOCK_M
   j_base = pid2 * BLOCK_N

   # range within tile
   i = i_base + tl.arange(0, BLOCK_M)
   j = j_base + tl.arange(0, BLOCK_N)
   k = tl.arange(0, BLOCK_K)

   # mask for i,j inside N
   mask_i = i < N
   mask_j = j < N

   # accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # loop over k
   for k_offset in range(0, N, BLOCK_K):
       cur_k = k_offset + k
       mask_k = cur_k < N

       # compute pointers
       left_offset = b * stride_lb + i[:, None] * stride_li + cur_k[None, :] * stride_lk + h * stride_lh
       right_offset = b * stride_rb + j[:, None] * stride_rj + cur_k[None, :] * stride_rk + h * stride_rh

       left = tl.load(left_ptr + left_offset, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
       right = tl.load(right_ptr + right_offset, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

       # accumulate dot product (left (M,K) * right (N,K).T)
       acc += tl.dot(left, right, trans_b=True)

   # store result
   out_offset = b * stride_ob + i[:, None] * stride_oi + j[None, :] * stride_oj + h * stride_oh

"""
custom_kernel: Triton‑accelerated outgoing Triangular Multiplicative Update (TriMul)

Algorithm (forward only)
-----------------------
1. Input LayerNorm (float32 for numeric stability).
2. Cast to fp16 and apply three linear projections:
      left  = x @ W_left
      right = x @ W_right
   Compute three sigmoid gates (left_gate, right_gate, out_gate).
3. Fuse gates (and optional mask) with element‑wise multiplication in
   regular PyTorch (cheap compared to the O(N³) contraction).
4. Core contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a custom Triton kernel that performs a batched
   matrix multiplication L @ Rᵀ for each (batch, head) pair, accumulating
   in float32 and storing the result in fp16.
5. Output LayerNorm, apply the output gate, and final linear projection
   back to the original dimension.
6. Cast back to the input dtype.

The Triton kernel fuses the K‑loop of the einsum, reducing memory traffic
and avoiding an explicit PyTorch `torch.bmm` over the hidden dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel for the O(N³·H) contraction (outgoing TriMul)
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers (fp16)
    B, N, H,                                    # runtime scalars
    stride_lb, stride_li, stride_lk, stride_lh, # left strides (elements)
    stride_rb, stride_rj, stride_rk, stride_rh, # right strides
    stride_ob, stride_oi, stride_oj, stride_oh, # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = Σ_k left[...,k,:] * right[...,k,:] for each batch & head."""
    pid_head = tl.program_id(0)      # flattened (batch * hidden)
    pid_i    = tl.program_id(1)      # block row index (i)
    pid_j    = tl.program_id(2)      # block col index (j)

    # -------------------------------------------------------
    # decode batch and hidden (head) indices
    # -------------------------------------------------------
    b = pid_head // H                # batch index
    h = pid_head %  H                # hidden (head) index

    # -------------------------------------------------------
    # base indices for the i/j tiles
    # -------------------------------------------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    mask_i = i < N
    mask_j = j < N

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------
    # loop over the reduction dimension k
    # -------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + k
        mask_k = cur_k < N

        # left[b,i,k,h]  -> shape (BLOCK_M, BLOCK_K)
        left_offset = (
            b * stride_lb
            + i[:, None] * stride_li
            + cur_k[None, :] * stride_lk
            + h * stride_lh
        )
        left = tl.load(
            left_ptr + left_offset,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right[b,j,k,h] -> shape (BLOCK_N, BLOCK_K)
        right_offset = (
            b * stride_rb
            + j[:, None] * stride_rj
            + cur_k[None, :] * stride_rk
            + h * stride_rh
        )
        right = tl.load(
            right_ptr + right_offset,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # matrix multiply over the K dimension (left @ rightᵀ)
        acc += tl.dot(left, right, trans_b=True)

    # -------------------------------------------------------
    # write back the result (cast to fp16)
    # -------------------------------------------------------
    out_offset = (
        b * stride_ob
        + i[:, None] * stride_oi
        + j[None, :] * stride_oj
        + h * stride_oh
    )
    tl.store(
        out_ptr + out_offset,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# --------------------------------------------------------------------------- #
# Python entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)  (float32/float16)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------- #
    # Config
    # ------------------------------------------------------------------- #
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------- #
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------- #
    # 2. Cast to fp16 for the compute‑heavy part
    # ------------------------------------------------------------------- #
    x = x.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 3. Linear projections (bias‑free) and gate computations
    # ------------------------------------------------------------------- #
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left  = F.linear(x, left_proj_w)        # (B,N,N,H)
    right = F.linear(x, right_proj_w)       # (B,N,N,H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,H)

    # ------------------------------------------------------------------- #
    # 4. Apply gates (and optional mask) – cheap elementwise work
    # ------------------------------------------------------------------- #
    if mask is not None and not nomask:
        # mask shape (B,N,N) -> (B,N,N,1) for broadcasting over hidden dim
        mask_exp = mask.to(device).to(torch.float16).unsqueeze(-1)
        left = left * left_gate * mask_exp
        right = right * right_gate * mask_exp
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Core O(N³·H) contraction via Triton kernel
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape  # re‑extract in case of shape changes
    left  = left.contiguous()
    right = right.contiguous()
    out   = torch.empty_like(left)  # (B,N,N,H), fp16

    # Strides in *elements* (not bytes)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Kernel block sizes – tuned for typical B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                               # one program per (batch, head)
        triton.cdiv(N, BLOCK_M),             # i‑tiles
        triton.cdiv(N, BLOCK_N),             # j‑tiles
    )

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------- #
    # 6. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------- #
    # 7. Cast back to the original dtype
    # ------------------------------------------------------------------- #
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
