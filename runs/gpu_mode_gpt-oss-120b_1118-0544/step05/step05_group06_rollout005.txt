"""
Highly‑optimized forward pass for the outgoing TriMul operator (AlphaFold3 style).

Algorithm
---------
1. Layer‑normalize the 4‑D input (B, N, N, C) across the channel dimension.
2. Cast to FP16 for compute‑intensive stages.
3. Linear projections (left/right) and gate projections (left/right/out) are
   computed with `torch.nn.functional.linear` (cuBLAS).
4. If a mask is supplied (and `nomask` is False) a Triton kernel fuses the
   element‑wise multiplication of projection, gate and mask, broadcasting the
   mask over the hidden dimension H.
5. Core reduction:
      out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   which is equivalent to a batched matrix multiplication `left @ rightᵀ`
   for each (b, h).  We reshape to (B·H, N, N) and use `torch.bmm`.
6. Apply a second LayerNorm over the hidden dimension, multiply by `out_gate`,
   and project back to the original channel dimension.
7. Cast the result back to the input dtype (usually float32).

At least one operation (gate × mask) is implemented in Triton.
"""
import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused projection * gate * mask (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,                # *[total_elems]   fp16
    gate_ptr,                # *[total_elems]   fp16
    mask_ptr,                # *[mask_elems]    fp16
    out_ptr,                 # *[total_elems]   fp16
    total_elems,             # i64
    hidden_dim,              # i64 (size of the last dimension H)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # each block of `hidden_dim` elements shares the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 4096) -> torch.Tensor:
    """
    Element‑wise ``proj * gate * mask`` where ``mask`` has shape (B, N, N)
    and is broadcast over the hidden dimension H.
    All tensors must be FP16 and contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (math.ceil(total / block_size),)

    # Flatten mask to 1‑D (B*N*N,)
    mask_flat = mask.contiguous().view(-1)

    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],   # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict containing all linear and LayerNorm parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask", optional "eps"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6. Apply mask + gate (Triton kernel) or just gate
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM (left @ rightᵀ) per head
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # left shape (B, N, N, H)

    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiplication with transpose on the second operand
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)
    return out
```
