```python
"""
Triton‑accelerated outgoing Triangle‑Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to FP16 for compute‑heavy parts.
3. Linear projections (no bias) to a hidden dimension H:
   - left_proj, right_proj
   - left_gate, right_gate, out_gate (followed by sigmoid)
4. Fuse mask, projection and gate (proj * gate * mask) with a small Triton kernel.
5. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM in Triton (one GEMM per (b,h) pair).
6. Layer‑norm over the hidden dimension, multiply by out_gate,
   and a final linear projection back to the original dimension.
7. Cast back to the original dtype (typically float32).
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Kernel 1: fuse projection, gate and mask (proj * gate * mask)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B,N,N,H)
    gate_ptr,          # fp16 * (B,N,N,H)
    mask_ptr,          # fp16 * (B,N,N)
    out_ptr,           # fp16 * (B,N,N,H)
    total_elems,       # total number of elements in proj/gate (=B*N*N*H)
    hidden_dim,        # H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # each hidden_dim consecutive entries share the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(out_ptr + offs, proj * gate * m, mask=valid)

def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    proj, gate : (B, N, N, H)  fp16
    mask       : (B, N, N)     fp16 (binary 0/1)
    Returns proj * gate * mask   (broadcast over H)
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out

# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the core reduction
# Computes, for each (batch, hidden) pair:
#   out[i, j] = Σ_k left[i, k] * right[j, k]   (right is transposed)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr,   # fp16 * (B,N,N,H)
    right_ptr,  # fp16 * (B,N,N,H)
    out_ptr,    # fp16 * (B,N,N,H)
    M, N, K,    # all equal to seq_len
    stride_left_i, stride_left_k,       # strides for left (i, k)
    stride_right_k, stride_right_j,     # strides for right (k, j) – note transposed
    stride_out_i, stride_out_j,         # strides for out   (i, j)
    hidden_dim: tl.int32,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # Tile identifiers
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)   # tile over i
    pid_n = tl.program_id(1)   # tile over j
    pid_h = tl.program_id(2)   # combined batch*hidden index

    # --------------------------------------------------------------
    # Decode batch and hidden indices
    # --------------------------------------------------------------
    batch_id = pid_h // hidden_dim
    hid = pid_h - batch_id * hidden_dim   # same as pid_h % hidden_dim

    # --------------------------------------------------------------
    # Compute base pointers for this (batch, hidden) slice
    # --------------------------------------------------------------
    # each batch occupies M*K*hidden_dim elements in memory
    batch_stride = M * K * hidden_dim          # = N * N * H
    left_base = left_ptr + batch_id * batch_stride + hid
    right_base = right_ptr + batch_id * batch_stride + hid
    out_base = out_ptr + batch_id * batch_stride + hid

    # --------------------------------------------------------------
    # Offsets within the tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Accumulator
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main loop over K dimension
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k_iter in range(num_k_blocks):
        cur_k = k_iter * BLOCK_K

        # global k index validity
        k_valid = (cur_k + offs_k) < K

        # Load A tile (left) : shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + (offs_m[:, None] * stride_left_i
                              + (cur_k + offs_k)[None, :] * stride_left_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_valid[None, :],
                    other=0.0)

        # Load B tile (right transposed) : shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + ((cur_k + offs_k)[:, None] * stride_right_k
                               + offs_n[None, :] * stride_right_j)
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # --------------------------------------------------------------
    # Write back result (cast to fp16)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = out_base + (offs_m[:, None] * stride_out_i
                         + offs_n[None, :] * stride_out_j)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C]
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast weights to FP16 (no bias in linear layers)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (shape [B, N, N, H])
    # --------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # --------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Apply mask and gating (fused via Triton if mask present)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # mask is [B, N, N]; broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask present but we are in a branch where it was not
            # (should not happen); fallback to broadcast multiplication.
            mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask_fp16
            right = right * mask_fp16

    # --------------------------------------------------------------
    # 7. Core reduction: batched GEMM (left @ right^T) per hidden channel
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # Allocate output tensor
    out = torch.empty_like(left)   # (B, N, N, H) fp16

    # Strides (in elements) for the tensors
    # left: (B,N,N,H) -> strides (N*N*H, N*H, H, 1)
    left_stride = left.stride()
    right_stride = right.stride()
    out_stride = out.stride()

    # GEMM tile sizes – tuned for the test hardware
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        N, N, N,
        left_stride[1], left_stride[2],           # stride_left_i, stride_left_k
        right_stride[2], right_stride[1],         # stride_right_k, stride_right_j (transposed)
        out_stride[1], out_stride[2],             # stride_out_i, stride_out_j
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out

"""
Triton‑accelerated outgoing Triangle‑Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to FP16 for compute‑heavy parts.
3. Linear projections (no bias) to a hidden dimension H:
   - left_proj, right_proj
   - left_gate, right_gate, out_gate (followed by sigmoid)
4. Fuse mask, projection and gate (proj * gate * mask) with a small
   Triton kernel.
5. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM in Triton (one GEMM per (b,h) pair).
6. Layer‑norm over the hidden dimension, multiply by out_gate,
   and a final linear projection back to the original dimension.
7. Cast back to the original dtype (usually float32).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Kernel 1: fuse projection, gate and mask (proj * gate * mask)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B,N,N,H)
    gate_ptr,          # fp16 * (B,N,N,H)
    mask_ptr,          # fp16 * (B,N,N)
    out_ptr,           # fp16 * (B,N,N,H)
    total_elems,       # total number of elements in proj/gate (=B*N*N*H)
    hidden_dim,        # H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # each H consecutive entries share the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(out_ptr + offs, proj * gate * m, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    proj, gate : (B, N, N, H)  fp16
    mask       : (B, N, N)     fp16 (binary 0/1)
    Returns proj * gate * mask   (broadcast over H)
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the core reduction
# Computes, for each (batch, hidden) pair:
#   out[i, j] = Σ_k left[i, k] * right[j, k]   (right is transposed)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr,   # fp16 * (B,N,N,H)
    right_ptr,  # fp16 * (B,N,N,H)
    out_ptr,    # fp16 * (B,N,N,H)
    M, N, K,    # all equal to seq_len
    stride_left_i, stride_left_k,       # left: (i, k) strides
    stride_right_k, stride_right_j,     # right: (k, j) strides (transposed)
    stride_out_i, stride_out_j,         # out: (i, j) strides
    hidden_dim: tl.int32,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # Tile identifiers
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)   # tile over i
    pid_n = tl.program_id(1)   # tile over j
    pid_h = tl.program_id(2)   # combined batch*hidden index

    # --------------------------------------------------------------
    # Decode batch and hidden indices
    # --------------------------------------------------------------
    batch_id = pid_h // hidden_dim
    hid = pid_h - batch_id * hidden_dim   # same as pid_h % hidden_dim

    # --------------------------------------------------------------
    # Base pointers for this (batch, hidden) slice
    # --------------------------------------------------------------
    batch_stride = M * K * hidden_dim          # = N * N * H
    left_base = left_ptr + batch_id * batch_stride + hid
    right_base = right_ptr + batch_id * batch_stride + hid
    out_base = out_ptr + batch_id * batch_stride + hid

    # --------------------------------------------------------------
    # Offsets within the tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Accumulator
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k_iter in range(num_k_blocks):
        cur_k = k_iter * BLOCK_K
        k_valid = (cur_k + offs_k) < K

        # Load A tile (left) – shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + (offs_m[:, None] * stride_left_i
                              + (cur_k + offs_k)[None, :] * stride_left_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_valid[None, :],
                    other=0.0)

        # Load B tile (right, transposed) – shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + ((cur_k + offs_k)[:, None] * stride_right_k
                               + offs_n[None, :] * stride_right_j)
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (M,K) @ (K,N) → (M,N)

    # --------------------------------------------------------------
    # Write back result (cast to fp16)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = out_base + (offs_m[:, None] * stride_out_i
                         + offs_n[None, :] * stride_out_j)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
```
