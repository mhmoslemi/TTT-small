def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16
    x = x.to(torch.float16)

    B, N, _, _ = x.shape
    M = B * N * N

    # Flatten x
    x_flat = x.reshape(M, dim).contiguous()

    # Load weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16).contiguous()
    left_gate_w = weights["left_gate.weight"].to(torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(torch.float16).contiguous()
    out_gate_w = weights["out_gate.weight"].to(torch.float16).contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Prepare mask
    if mask is not None and not nomask:
        # mask shape [B,N,N]; flatten
        mask_flat = mask.to(torch.float16).reshape(M).contiguous()
    else:
        mask_flat = None

    # Compute left and right via fused kernel
    left = _proj_gate_mask_fused(x_flat, left_proj_w, left_gate_w, mask_flat, M, dim, hidden_dim)
    right = _proj_gate_mask_fused(x_flat, right_proj_w, right_gate_w, mask_flat, M, dim, hidden_dim)

def proj_gate_mask_kernel(
    x_ptr, proj_w_ptr, gate_w_ptr, mask_ptr, out_ptr,
    M, dim, hidden_dim,
    stride_x_m, stride_x_d,
    stride_proj_w_d, stride_proj_w_h,
    stride_gate_w_d, stride_gate_w_h,
    stride_mask_m,
    stride_out_m, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_M
    offs_m = block_start + tl.arange(0, BLOCK_M)
    m_valid = offs_m < M

    offs_h = tl.arange(0, BLOCK_H)
    h_valid = offs_h < hidden_dim

    acc_proj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    num_d_blocks = tl.cdiv(dim, BLOCK_D)
    for d in range(num_d_blocks):
        d_start = d * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        d_valid = offs_d < dim

        x = tl.load(x_ptr + offs_m[:, None] * stride_x_m + offs_d[None, :] * stride_x_d,
                    mask=m_valid[:, None] & d_valid[None, :],
                    other=0.0)

        proj_w = tl.load(proj_w_ptr + offs_d[:, None] * stride_proj_w_d + offs_h[None, :] * stride_proj_w_h,
                         mask=d_valid[:, None] & h_valid[None, :],
                         other=0.0)

        gate_w = tl.load(gate_w_ptr + offs_d[:, None] * stride_gate_w_d + offs_h[None, :] * stride_gate_w_h,
                         mask=d_valid[:, None] & h_valid[None, :],
                         other=0.0)

        acc_proj += tl.dot(x, proj_w)  # (BLOCK_M, BLOCK_H)
        acc_gate += tl.dot(x, gate_w)

"""
TriMul (outgoing) forward pass – Triton‑accelerated elementwise mask/gate fusion
and a single fused linear projection to drastically reduce the number of GEMMs.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for compute‑heavy work.
3. Fuse the five linear projections (left_proj, right_proj,
   left_gate, right_gate, out_gate) into ONE GEMM by concatenating the
   weight matrices (shape (5*hidden_dim, dim)).  Split the result and
   apply sigmoid to the three gate tensors.
4. Fuse projection, gate and optional mask in a Triton kernel that computes
   `proj * sigmoid(gate) * mask` (mask broadcast over the hidden dimension).
5. Core reduction: for each head `h`,
   `out_h = left_h @ right_h.T`   (batched GEMM using torch.bmm).
6. Output layer‑norm, multiply by `out_gate`, and a final linear projection
   back to `dim`.
7. Cast back to the original dtype (usually float32).

Only the elementwise mask/gate fusion is performed with Triton; the heavy
linear and matrix‑multiplication work leverages cuBLAS/Tensor‑cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: elementwise (proj * sigmoid(gate) * mask)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (total,)
    gate_ptr,          # fp16 * (total,)
    mask_ptr,          # fp16 * (mask_len,)
    out_ptr,           # fp16 * (total,)
    total_elems,       # int
    hidden_dim,        # int
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # guard
    valid = offs < total_elems

    # which mask element corresponds to this offset (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # sigmoid
    gate = 1.0 / (1.0 + tl.exp(-gate))

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 2048) -> torch.Tensor:
    """
    Fuse projection, sigmoid gate and optional mask using a Triton kernel.
    All tensors must be fp16 and contiguous.
    proj, gate : (M, hidden_dim)
    mask       : (M,)  (binary 0/1, broadcast over hidden_dim)
    Returns proj * sigmoid(gate) * mask   (shape (M, hidden_dim)).
    """
    assert proj.is_contiguous() and gate.is_contiguous()
    M, hidden_dim = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        hidden_dim,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C]
            - mask         : torch.Tensor of shape [B, N, N] (or None)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys
                  * "dim" – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy part
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    B, N, _, _ = x.shape
    M = B * N * N                     # flattened batch‑seq‑seq dimension

    # --------------------------------------------------------------
    # 3. Fuse all five linear projections into a single GEMM
    # --------------------------------------------------------------
    # flatten to (M, dim)
    x_flat = x.reshape(M, dim).contiguous()

    # collect weight matrices (shape: out_features x in_features)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16).contiguous()
    right_proj_w  = weights["right_proj.weight"].to(torch.float16).contiguous()
    left_gate_w   = weights["left_gate.weight"].to(torch.float16).contiguous()
    right_gate_w  = weights["right_gate.weight"].to(torch.float16).contiguous()
    out_gate_w    = weights["out_gate.weight"].to(torch.float16).contiguous()

    # concatenate on the out_features dimension (first dim)
    combined_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    # single linear projection
    combined = F.linear(x_flat, combined_weight)          # (M, 5*hidden_dim)

    # split back
    splits = torch.split(combined, hidden_dim, dim=1)
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = splits

    # sigmoid gates
    left_gate  = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate   = out_gate_pre.sigmoid()                   # (M, hidden_dim)

    # --------------------------------------------------------------
    # 4. Apply optional mask and fuse with gating (Triton kernel)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous().reshape(M)
    else:
        # dummy mask of ones – the kernel multiplies by 1.0
        mask_fp16 = torch.ones((M,), dtype=torch.float16, device=device)

    # left = left_proj * sigmoid(left_gate_pre) * mask
    left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
    # right = right_proj * sigmoid(right_gate_pre) * mask
    right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)

    # reshape back to [B, N, N, hidden_dim]
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # For each head h we need left_h @ right_h.T
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)                     # (B,N,N,dim)

    # --------------------------------------------------------------
    # 7. Cast back to the original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
