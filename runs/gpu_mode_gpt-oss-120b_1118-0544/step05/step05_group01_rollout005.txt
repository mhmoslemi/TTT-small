@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bn, stride_bk,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # 1D grid: (batch, grid_m, grid_n) flattened
    # compute total number of blocks for M,N
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    # Determine block indices
    batch = pid // (num_m * num_n)
    block_id = pid % (num_m * num_n)
    pid_m = block_id // num_n
    pid_n = block_id % num_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_ptrs = a_ptr + batch * batch_stride_a + offs_m[:, None] * stride_am + 0 * stride_ak  # placeholder
    # We'll compute inside loop.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for valid rows/cols
        a_mask = (offs_m < M)[:, None] & (k_offset < K)[None, :]
        b_mask = (k_offset[:, None] < K) & (offs_n[None, :] < N)

        # load A block
        a = tl.load(a_ptr + batch * batch_stride_a + offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak,
                    mask=a_mask, other=0.0)

        # load B block (transposed)
        b = tl.load(b_ptr + batch * batch_stride_b + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn,
                    mask=b_mask, other=0.0)

        # dot
        acc += tl.dot(a, b)

    # write back
    c = acc.to(tl.float16)
    # mask output
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptr + batch * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
             c, mask=c_mask)

def custom_kernel(data):
   input_tensor, mask, weights, config = data

   # get dims
   dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

   # layernorm
   x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

   # combine mask if present
   if mask is not None and not nomask:
       # mask shape (B,N,N) -> unsqueeze(-1) broadcast to hidden_dim later
       # We'll apply mask after projection: left = left * mask[..., None]
   else:
       # ignore
       pass

   # cast to fp16
   x = x.to(torch.float16)

   # convert weight to fp16
   left_proj_w = weights["left_proj.weight"].to(torch.float16)
   right_proj_w = weights["right_proj.weight"].to(torch.float16)
   left_gate_w = weights["left_gate.weight"].to(torch.float16)
   right_gate_w = weights["right_gate.weight"].to(torch.float16)
   out_gate_w = weights["out_gate.weight"].to(torch.float16)
   to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
   to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
   to_out_w = weights["to_out.weight"].to(torch.float16)

   # linear projections
   left_proj = torch.nn.functional.linear(x, left_proj_w) # shape (B,N,N,hidden_dim)
   right_proj = torch.nn.functional.linear(x, right_proj_w)
   left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
   right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
   out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

   if mask is not None and not nomask:
       mask_ = mask.unsqueeze(-1).to(torch.float16)  # shape (B,N,N,1)
       left = left_proj * left_gate * mask_
       right = right_proj * right_gate * mask_
   else:
       left = left_proj * left_gate
       right = right_proj * right_gate

   # reshape for kernel
   B, N, _, H = left.shape
   # permute to bring hidden dim before N,N
   left_ = left.permute(0,3,1,2).contiguous()   # shape (B, H, N, N)
   right_ = right.permute(0,3,1,2).contiguous()
   # flatten first two dims
   left_flat = left_.view(B*H, N, N)
   right_flat = right_.view(B*H, N, N)

   # compute using Triton kernel
   out_flat = _tri_mul_matmul(left_flat, right_flat, N, BLOCK_M=128, BLOCK_N=128, BLOCK_K=32)  # choose block sizes

   # reshape back
   out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()  # shape (B,N,N,H)

   # output layer norm
   out = torch.nn.functional.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

   # multiply by out_gate
   out = out * out_gate

   # final linear projection to dim
   out = torch.nn.functional.linear(out, to_out_w)  # shape (B,N,N,dim)

   # cast back to original dtype
   out = out.to(input_tensor.dtype)

   return out

"""
TriMul (outgoing) forward implemented with Triton.
-------------------------------------------------
1. Layer‑norm the input (dim‑wise).
2. FP16 linear projections → left/right tensors (shape B,N,N,H).
3. Compute gating (sigmoid) and optionally apply a binary mask.
4. Fuse left/right via a batched GEMM kernel:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to a batch of matrix multiplies
   left_h @ right_h.T for each hidden slice h.
   The kernel is a tiled GEMM written in Triton, using fp16 loads,
   fp32 accumulation and storing the result in fp16.
5. Layer‑norm over the hidden dimension, multiply by the output gate,
   and final linear projection back to the original channel size.
6. Return the result in the input dtype (usually float32).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------
# Triton kernel: batched GEMM for left @ right.T
# --------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # strides (in elements)
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bn, stride_bk,
    batch_stride_c, stride_cm, stride_cn,
    # sizes
    M, N, K,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of blocks along M and N
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    # decode batch, block‑row and block‑col from 1‑D grid
    batch = pid // (num_m * num_n)
    block_id = pid % (num_m * num_n)
    pid_m = block_id // num_n
    pid_n = block_id % num_n

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulate in fp32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (k_offset[None, :] < K)
        b_mask = (k_offset[:, None] < K) & (offs_n[None, :] < N)

        # A block: left[i,k]
        a = tl.load(
            a_ptr
            + batch * batch_stride_a
            + offs_m[:, None] * stride_am
            + k_offset[None, :] * stride_ak,
            mask=a_mask,
            other=0.0,
        )

        # B block (transposed): right_T[k,j] == right[j,k]
        b = tl.load(
            b_ptr
            + batch * batch_stride_b
            + k_offset[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=b_mask,
            other=0.0,
        )

        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # write back (convert to fp16)
    c = acc.to(tl.float16)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(
        c_ptr
        + batch * batch_stride_c
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=c_mask,
    )


def _batched_matmul_tiled(
    left: torch.Tensor,
    right: torch.Tensor,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 32,
) -> torch.Tensor:
    """
    Compute C = left @ right.T for a batch of matrices using the Triton kernel above.
    All tensors must be contiguous fp16 with shape (B, N, N).
    """
    assert left.shape == right.shape
    B, N, _ = left.shape
    C = torch.empty_like(left)

    # strides in elements (not bytes)
    batch_stride_a = left.stride(0)
    stride_am = left.stride(1)
    stride_ak = left.stride(2)

    batch_stride_b = right.stride(0)
    stride_bn = right.stride(1)  # original N dimension (j)
    stride_bk = right.stride(2)  # original K dimension (k)

    batch_stride_c = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    # launch configuration
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_blocks = B * grid_m * grid_n
    grid = (total_blocks,)

    _tri_mul_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=C,
        batch_stride_a=batch_stride_a,
        stride_am=stride_am,
        stride_ak=stride_ak,
        batch_stride_b=batch_stride_b,
        stride_bn=stride_bn,
        stride_bk=stride_bk,
        batch_stride_c=batch_stride_c,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        M=N,
        N=N,
        K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (same dtype as input_tensor)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all linear weights to fp16 (no bias)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections and gates
    # --------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 5. Apply mask (if any) and fuse with gating
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B,N,N) -> broadcast over hidden dim
        mask_ = mask.unsqueeze(-1).to(torch.float16)
        left = left_proj * left_gate * mask_
        right = right_proj * right_gate * mask_
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6. Core reduction using the Triton batched GEMM kernel
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # reshape to (B*H, N, N) for the kernel
    left_ = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_.view(B * H, N, N)
    right_flat = right_.view(B * H, N, N)

    # Triton GEMM (left @ right.T)
    out_flat = _batched_matmul_tiled(left_flat, right_flat)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 8. Cast back to the original dtype (usually float32)
    # --------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
