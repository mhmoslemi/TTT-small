@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Compute batch index and block indices
    pid = tl.program_id(0)
    batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_id = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    # Now compute block_m and block_n
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    block_m = block_id // tl.cdiv(N, BLOCK_N)
    block_n = block_id % tl.cdiv(N, BLOCK_N)

    # Compute start indices
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute pointers for A and B; we need to offset batch
    a_ptr = a_ptr + batch * batch_stride_a
    b_ptr = b_ptr + batch * batch_stride_b
    c_ptr = c_ptr + batch * batch_stride_c

    # init acc
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # Load tiles
        a = tl.load(a_ptr + (offs_m[:, None] * stride_am + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak), mask=(offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K), other=0.0)
        # For B transpose, we need to load B[b, n, k] but we want B_t[n,k] = B[b, n, k]. To compute A @ B^T, we need B values at (k,n) pairs.
        # So we load B as B_T? Let's compute B_T_ptr = b_ptr + (k_offset + tl.arange(0, BLOCK_K)) * stride_bk + offs_n[None, :] * stride_bn. Actually our original B shape is (batch,N,K) (since we treat B as (batch,N,K) but we need B_T as (batch,K,N). To compute A (M,K) * B_T (K,N), we need B in transposed layout: we can compute pointer offsets accordingly.
        b = tl.load(b_ptr + ((k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk + offs_n[None, :] * stride_bn), mask=(k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N), other=0.0)
        # dot
        acc += tl.dot(a, b)
    # Write back
    c = acc.to(tl.float16) # maybe output dtype
    tl.store(c_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn), c, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).
    The heavy O(N³·H) reduction is performed by a custom batched GEMM
    kernel that computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
    Masking and gating are done with native PyTorch element‑wise ops.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1) LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32)
    )

    # Cast to fp16 for heavy ops
    x = x.to(torch.float16)

    # 2) Cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # 3) Linear projections to hidden dim
    left = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w)  # (B,N,N,H)

    # 4) Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # 5) Apply mask and gates (element‑wise)
    if mask is not None and not nomask:
        # mask: [B,N,N] -> expand to [B,N,N,1]
        mask_e = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_e * left_gate
        right = right * mask_e * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # 6) Prepare tensors for batched GEMM:
    #   shape (B, H, N, N) -> (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, N, _, H = left.shape
    # Flatten batch*head dimension
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Allocate output tensor
    out_flat = torch.empty_like(left_flat)

    # 7) Launch Triton kernel for batched GEMM (A @ B^T)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    M = N
    K = N
    # compute strides
    batch_stride_a = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    batch_stride_b = right_flat.stride(0)
    stride_bn = right_flat.stride(1)
    stride_bk = right_flat.stride(2)

    batch_stride_c = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    total_tiles_m = (M + BLOCK_M - 1) // BLOCK_M
    total_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = (B * H) * total_tiles_m * total_tiles_n
    grid = (total_programs,)

    batched_gemm_triton[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_stride_a, batch_stride_b, batch_stride_c,
        stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
        M, N, K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # 8) Reshape back to (B,N,N,H)

    # 9) Output LayerNorm over hidden dim
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 10) Apply out_gate
    out = out * out_gate

    # 11) Final projection back to original dim
    out = F.linear(out, to_out_w)

    # Cast back to original dtype if needed
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out

"""
TriMul (outgoing) – Triton‑accelerated implementation
----------------------------------------------------
The heavy O(N³·H) reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed by a custom batched GEMM kernel that computes
A @ Bᵀ for each head (hidden dimension) in parallel.
Masking and gating are fused with native PyTorch element‑wise
operations; the remaining linear layers and LayerNorms use
torch's highly‑optimized kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_transpose_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak, stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,                                 # matrix sizes (all == seq_len)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute, for each batch index `b` and head `h`,
        C[b,h,i,j] = Σ_k A[b,h,i,k] * B[b,h,j,k]
    where A and B are stored as (batch, M, K) and (batch, N, K) respectively.
    The kernel reads B transposed on‑the‑fly, i.e. accesses B[b,h,k,j].
    """
    pid = tl.program_id(0)

    # --- tile indices ---------------------------------------------------------
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_m * num_n

    batch_idx = pid // tiles_per_batch
    tile_idx = pid % tiles_per_batch
    pid_m = tile_idx // num_n
    pid_n = tile_idx % num_n

    # --- offsets for this tile ------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for the current batch (head)
    a_base = a_ptr + batch_idx * batch_stride_a
    b_base = b_ptr + batch_idx * batch_stride_b
    c_base = c_ptr + batch_idx * batch_stride_c

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --- main K‑loop ---------------------------------------------------------
    for kb in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kb * BLOCK_K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_base + (offs_m[:, None] * stride_am
                      + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak),
            mask=(offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K),
            other=0.0,
        )
        # B tile (transposed on‑the‑fly): shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_base + ((k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk
                      + offs_n[None, :] * stride_bn),
            mask=(k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        acc += tl.dot(a, b)

    # --- write back -----------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_base + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] or None (binary)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (opt) – if True, ignore mask

    Returns:
        torch.Tensor [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ----------------------------------------------------------------------
    # Unpack
    # ----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # ----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # ----------------------------------------------------------------------
    # 2) Cast to FP16 for heavy arithmetic
    # ----------------------------------------------------------------------
    x = x.to(torch.float16)

    # ----------------------------------------------------------------------
    # 3) Cast all model weights to FP16
    # ----------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ----------------------------------------------------------------------
    # 4) Linear projections to hidden dimension H
    # ----------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ----------------------------------------------------------------------
    # 5) Compute gating values (sigmoid)
    # ----------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ----------------------------------------------------------------------
    # 6) Apply optional mask and gates (element‑wise, fully fused by PyTorch)
    # ----------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_exp = mask.to(torch.float16).unsqueeze(-1)            # (B,N,N,1)
        left  = left  * mask_exp * left_gate
        right = right * mask_exp * right_gate
    else:
        left  = left  * left_gate
        right = right * right_gate

    # ----------------------------------------------------------------------
    # 7) Prepare tensors for the batched GEMM: (B, H, N, N) → (B*H, N, N)
    # ----------------------------------------------------------------------
    B, N, _, H = left.shape
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat  = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # ----------------------------------------------------------------------
    # 8) Allocate output buffer
    # ----------------------------------------------------------------------
    out_flat = torch.empty_like(left_flat)

    # ----------------------------------------------------------------------
    # 9) Launch the Triton batched GEMM kernel (computes A @ Bᵀ)
    # ----------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    M = N
    K = N

    # Strides in elements (not bytes)
    batch_stride_a = left_flat.stride(0)
    stride_am      = left_flat.stride(1)
    stride_ak      = left_flat.stride(2)

    batch_stride_b = right_flat.stride(0)
    stride_bn      = right_flat.stride(1)   # N
    stride_bk      = right_flat.stride(2)   # 1

    batch_stride_c = out_flat.stride(0)
    stride_cm      = out_flat.stride(1)     # N
    stride_cn      = out_flat.stride(2)     # 1

    tiles_m = (M + BLOCK_M - 1) // BLOCK_M
    tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = (B * H) * tiles_m * tiles_n
    grid = (total_programs,)

    _batched_gemm_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        M,
        N,
        K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ----------------------------------------------------------------------
    # 10) Reshape back to (B, N, N, H)
    # ----------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ----------------------------------------------------------------------
    # 11) Second LayerNorm over hidden dimension
    # ----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----------------------------------------------------------------------
    # 12) Apply output gate
    # ----------------------------------------------------------------------
    out = out * out_gate

    # ----------------------------------------------------------------------
    # 13) Final linear projection back to original dimension
    # ----------------------------------------------------------------------
    out = F.linear(out, to_out_w)  # (B,N,N,dim)

    # ----------------------------------------------------------------------
    # 14) Cast back to the original dtype if needed
    # ----------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
