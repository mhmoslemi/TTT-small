    pid_j = pid % num_tiles_n

    # offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for oob
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # a block (M,N_i,K)
        a_ptrs = a_ptr + batch_idx * stride_am + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # b block; we need (K,N_j) layout
        b_ptrs = b_ptr + batch_idx * stride_bm + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)   # yields fp32

    # store result
    c_ptrs = c_ptr + batch_idx * stride_cm + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_m = tl.program_id(2)  # M index

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

"""
TriMul (outgoing) forward pass with a Triton kernel for the
core O(N³·H) contraction.

Algorithm
---------
1. Layer‑norm the input (float32 for stability).
2. Cast to fp16 and apply linear projections + sigmoidal gates.
3. Optionally mask the projected tensors.
4. Fuse the contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a custom Triton kernel that treats (b,d) as a batched
   dimension and performs a tiled matrix‑multiply (left @ rightᵀ).
5. Layer‑norm over the hidden dimension, apply the output gate and a
   final linear projection back to the original dimension.
6. Cast the result back to the input dtype.

At least one Triton kernel (the contraction) is used, all other
operations are expressed in PyTorch for clarity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,                     # fp16 tensors
    N, K,                                     # sequence length (N) and inner dim (K=N)
    stride_am, stride_ai, stride_ak,          # strides for a (M,N_i,K)
    stride_bm, stride_bk, stride_bj,          # strides for b (M,K,N_j) – note order for transposed access
    stride_cm, stride_ci, stride_cj,          # strides for c (M,N_i,N_j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a batch of matrix multiplications:
        c[b,i,j] = Σ_k a[b,i,k] * b[b,j,k]
    where a and b have shape (M, N, K) with K = N.
    """
    pid_i = tl.program_id(0)   # tile row index (i)
    pid_j = tl.program_id(1)   # tile column index (j)
    pid_m = tl.program_id(2)   # batch*hidden index

    # Offsets for the i/j tiles
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # a block: (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + pid_m * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # b block: we need (BLOCK_K, BLOCK_N) i.e. k as rows, j as columns
        b_ptrs = (
            b_ptr
            + pid_m * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # Accumulate
        acc += tl.dot(a, b)   # fp32 accumulation

    # Store result back in fp16
    c_ptrs = (
        c_ptr
        + pid_m * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing "dim" (C) and "hidden_dim" (H)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------ #
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # ------------------------------------------------------------------ #
    # 2. Cast to fp16 for the heavy path
    # ------------------------------------------------------------------ #
    x = x.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 3. Linear projections (no bias) – all in fp16
    # ------------------------------------------------------------------ #
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left  = F.linear(x, left_proj_w)    # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------ #
    # 4. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------ #
    if mask is not None and not nomask:
        m = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------ #
    # 5. Gating
    # ------------------------------------------------------------------ #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,hidden_dim)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 6. Core O(N³·H) contraction via Triton kernel
    # ------------------------------------------------------------------ #
    B, N, _, H = left.shape
    M = B * H  # fused batch*hidden dimension

    # Bring hidden dimension to the leading axis and flatten batch*hidden
    left_perm  = left.permute(0, 3, 1, 2).contiguous().view(M, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous().view(M, N, N)

    # Output buffer
    out_mat = torch.empty((M, N, N), dtype=torch.float16, device=device)

    # Strides (all tensors are contiguous after the view)
    stride_am = left_perm.stride(0)
    stride_ai = left_perm.stride(1)
    stride_ak = left_perm.stride(2)

    stride_bm = right_perm.stride(0)
    stride_bk = right_perm.stride(2)   # stride over K dimension
    stride_bj = right_perm.stride(1)   # stride over J (row) dimension

    stride_cm = out_mat.stride(0)
    stride_ci = out_mat.stride(1)
    stride_cj = out_mat.stride(2)

    # Block sizes – chosen to fit in shared memory and expose tensor cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M ),
        ( (N + BLOCK_N - 1) // BLOCK_N ),
        M,
    )

    _tri_mul_kernel[grid](
        left_perm,
        right_perm,
        out_mat,
        N,
        N,                     # K == N
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B,N,N,H) in fp16

    # ------------------------------------------------------------------ #
    # 7. Output LayerNorm, gate, and final linear projection
    # ------------------------------------------------------------------ #
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply output gate (already fp16)
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------ #
    # 8. Cast back to the original dtype and return
    # ------------------------------------------------------------------ #
    out = out.to(dtype_in)
    return out
