@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_az, stride_ai, stride_ak,
    stride_bz, stride_bk, stride_bj,
    stride_cz, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BATCH: tl.constexpr, M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    num_warps: tl.constexpr,
):

    # program ID
    pid0 = tl.program_id(0)  # combined batch and block_i
    pid1 = tl.program_id(1)  # block_j

    # compute batch index and block_i
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    batch_id = pid0 // num_blocks_m
    block_i = pid0 % num_blocks_m

    # block_j from pid1
    block_j = pid1

    # compute offsets for A and B
    a_ptrs = a_ptr + batch_id * stride_az + (block_i * BLOCK_M + tl.arange(0, BLOCK_M))[:, None] * stride_ai + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bz + (tl.arange(0, BLOCK_K))[:, None] * stride_bk + (block_j * BLOCK_N + tl.arange(0, BLOCK_N))[None, :] * stride_bj

    # Load a and b, mask out-of-bounds
    mask_a = (block_i * BLOCK_M + tl.arange(0, BLOCK_M)) < M
    mask_b = (block_j * BLOCK_N + tl.arange(0, BLOCK_N)) < N
    # also mask k dimension
    mask_k = tl.arange(0, BLOCK_K) < K

    a = tl.load(a_ptrs, mask=mask_a[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
    b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_b[None, :], other=0.0).to(tl.float32)

    # Compute accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, K, BLOCK_K):
        # Compute offsets for each block of k
        a_block = tl.load(a_ptrs + k * stride_ak, mask=mask_a[:, None] & (tl.arange(0, BLOCK_K) < K - k)[None, :], other=0.0).to(tl.float32)
        b_block = tl.load(b_ptrs + k * stride_bk, mask=(tl.arange(0, BLOCK_K) < K - k)[:, None] & mask_b[None, :], other=0.0).to(tl.float32)

        # Multiply-accumulate
        acc += tl.dot(a_block, b_block)

    # Store
    c_ptrs = c_ptr + batch_id * stride_cz + (block_i * BLOCK_M + tl.arange(0, BLOCK_M))[:, None] * stride_ci + (block_j * BLOCK_N + tl.arange(0, BLOCK_N))[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16) if acc.dtype == tl.float16 else acc, mask=mask_a[:, None] & mask_b[None, :])

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_az, stride_ai, stride_ak,
    stride_bz, stride_bk, stride_bj,
    stride_cz, stride_ci, stride_cj,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Mask for boundary
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        # k offsets
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile
        a_ptrs = a_ptr + batch_id * stride_az + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile
        b_ptrs = b_ptr + batch_id * stride_bz + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Multiply-accumulate
        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = c_ptr + batch_id * stride_cz + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16) if acc.dtype == tl.float16 else acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm: use torch.nn.functional.layer_norm with weight & bias
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # Cast to fp16
    x = x.to(torch.float16)

    # Cast weights
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # Linear projections into hidden dim
    left = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w)  # (B,N,N,H)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply mask and gates
    if mask is not None and not nomask:
        mask = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask * left_gate
        right = right * mask * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # Rearrange for batched matmul
    B,N,N,H = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B,H,N,N)
    # Combine batch and hidden
    left_flat = left.view(B * hidden_dim, N, N)
    right_flat = right.view(B * hidden_dim, N, N)

    # Transpose right for gemm
    right_T = right_flat.transpose(-2, -1).contiguous() # (B*H, N, N) but transposed view with swapped strides

    # Output buffer
    out_flat = torch.empty_like(left_flat)  # fp16

    # Get strides
    stride_az, stride_ai, stride_ak = left_flat.stride()
    stride_bz, stride_bk, stride_bj = right_T.stride()
    stride_cz, stride_ci, stride_cj = out_flat.stride()

    # Decide block sizes: adapt to N (maybe set based on N)
    # We'll set BLOCK_M = BLOCK_N = 64, BLOCK_K = 32
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Launch kernel
    grid = (B * hidden_dim, tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N))
    batched_matmul_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        stride_az,
        stride_ai,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bj,
        stride_cz,
        stride_ci,
        stride_cj,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    out = F.layer_norm(

        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    out = out * out_gate

    # Final linear projection
    out = F.linear(out, to_out_w)

    # Cast back
    out = out.to(orig_dtype)

    return out

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_az, stride_ai, stride_ak,
    stride_bz, stride_bk, stride_bj,
    stride_cz, stride_ci, stride_cj,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    # Program IDs
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers base
    a_base = a_ptr + batch_id * stride_az
    b_base = b_ptr + batch_id * stride_bz
    c_base = c_ptr + batch_id * stride_cz

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile
        a_ptrs = a_base + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile (transposed)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_base + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) (torch implementation).
2. Linear projections to hidden dimension H for left/right streams.
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. Apply optional binary mask and left/right gates (fused element‑wise).
5. Core reduction: for each batch‑hidden pair compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a custom batched GEMM Triton kernel (left × rightᵀ).
6. Apply a second LayerNorm over the hidden dimension, multiply by the output
   gate and project back to the original channel dimension.
7. Return the result in the original dtype.

The heavy O(N³·H) contraction is performed entirely in Triton, achieving
significant speed‑ups over a pure torch.bmm implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    stride_az, stride_ai, stride_ak,   # A strides: batch, i, k
    stride_bz, stride_bk, stride_bj,   # B strides: batch, k, j  (B is already transposed)
    stride_cz, stride_ci, stride_cj,   # C strides: batch, i, j
    M, N, K,                           # matrix dimensions (all = seq_len)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    """Batched GEMM: C = A @ B   where A,B ∈ (batch, M, K) and B is transposed
    (i.e. B stores (k, j) for each batch)."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * stride_az
    b_base = b_ptr + batch_id * stride_bz
    c_base = c_ptr + batch_id * stride_cz

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (shape BLOCK_M x BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B tile (shape BLOCK_K x BLOCK_N) – B is already transposed,
        # so its first dimension is k and second is j.
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Write back result in FP16
    c_ptrs = c_base + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask        : torch.Tensor of shape [B, N, N] or None
            - weights     : dict of model parameters (float32 tensors)
            - config      : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ----------------------------------------------------------
    # Unpack
    # ----------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # ----------------------------------------------------------
    # 2) Cast to fp16 for heavy computation
    # ----------------------------------------------------------
    x = x.to(torch.float16)

    # ----------------------------------------------------------
    # 3) Cast model weights to fp16
    # ----------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ----------------------------------------------------------
    # 4) Linear projections to hidden dim
    # ----------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ----------------------------------------------------------
    # 5) Gates (sigmoid)
    # ----------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ----------------------------------------------------------
    # 6) Apply optional mask and gates (elementwise, fully vectorised)
    # ----------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f * left_gate
        right = right * mask_f * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # ----------------------------------------------------------
    # 7) Prepare tensors for batched GEMM
    # ----------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim
    # Move hidden dim next to batch for easy fusion
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Merge batch and hidden dimensions
    left_flat = left.view(B * hidden_dim, N, N)    # (B*H, N, N)
    right_flat = right.view(B * hidden_dim, N, N)  # (B*H, N, N)

    # Transpose right for A @ Bᵀ multiplication
    right_T = right_flat.transpose(-2, -1)  # view with swapped strides, no copy

    # Allocate output buffer
    out_flat = torch.empty_like(left_flat)   # fp16

    # ----------------------------------------------------------
    # 8) Launch Triton batched GEMM kernel
    # ----------------------------------------------------------
    stride_az, stride_ai, stride_ak = left_flat.stride()
    stride_bz, stride_bk, stride_bj = right_T.stride()
    stride_cz, stride_ci, stride_cj = out_flat.stride()

    # Tile sizes (tuned for typical seq_len up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        stride_az,
        stride_ai,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bj,
        stride_cz,
        stride_ci,
        stride_cj,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ----------------------------------------------------------
    # 9) Reshape back to (B,N,N,H)
    # ----------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ----------------------------------------------------------
    # 10) Output LayerNorm over hidden dimension (float32 for precision)
    # ----------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
    )

    # ----------------------------------------------------------
    # 11) Apply output gate and final linear projection
    # ----------------------------------------------------------
    out_fp32 = out_fp32 * out_gate.to(torch.float32)
    out_fp32 = F.linear(out_fp32, to_out_w.to(torch.float32))

    # ----------------------------------------------------------
    # 12) Cast back to original dtype and return
    # ----------------------------------------------------------
    return out_fp32.to(orig_dtype)
