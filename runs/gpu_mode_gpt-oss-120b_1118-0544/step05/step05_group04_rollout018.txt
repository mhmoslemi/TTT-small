@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start offsets
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Load pointers for this batch
    a_ptrs = A_ptr + batch_id * stride_Ab + off_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptrs = B_ptr + batch_id * stride_Bb + tl.arange(0, BLOCK_K)[:, None] * stride_Bk + off_n[None, :] * stride_Bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load block of A and B
        a = tl.load(a_ptrs, mask= (off_m < M)[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask= (tl.arange(0, BLOCK_K) < K)[:, None] & (off_n < N)[None, :], other=0.0)
        # accumulate
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # Write back
    c_ptrs = C_ptr + batch_id * stride_Cb + off_m[:, None] * stride_Cm + off_n[None, :] * stride_Cn
    mask_c = (off_m < M)[:, None] & (off_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)
```

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C).
2. Project to hidden dimension H with two linear layers (left/right).
3. Compute three sigmoid gates (left, right, out).
4. Fuse mask (if present) and left/right gates in a single Triton kernel.
5. Perform the core contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM (B*H independent N×N matmuls) using a custom Triton
   kernel that tiles over M,N and loops over K.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate and project back to the original channel dimension.
7. Return the result in the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Kernel 1: fuse mask and left/right gates
# --------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,               # const float* left   (B,N,N,H) flattened
    right_ptr,              # const float* right
    mask_ptr,               # const float* mask   (B,N,N) flattened
    left_gate_ptr,          # const float* left_gate
    right_gate_ptr,         # const float* right_gate
    left_out_ptr,           # float* left_out
    right_out_ptr,          # float* right_out
    H: tl.constexpr,        # hidden dimension
    TOTAL: tl.constexpr,    # total number of elements (B*N*N*H)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL

    # load raw tensors
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # mask index (broadcast over hidden dim)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # fused multiply‑add
    tl.store(left_out_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


# --------------------------------------------------------------
# Kernel 2: batched GEMM (A @ Bᵀ) for many (B*H) independent N×N matmuls
# --------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr,          # const float* A  (batch, M, K) flattened
    B_ptr,          # const float* B  (batch, K, N) flattened (already transposed)
    C_ptr,          # float* C        (batch, M, N) flattened
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # predicates for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of this batch
    a_ptrs = A_ptr + batch_id * stride_Ab + offs_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptrs = B_ptr + batch_id * stride_Bb + tl.arange(0, BLOCK_K)[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # accumulate in fp32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
        # promote to fp32 before dot
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # write result back (cast to fp16)
    c_ptrs = C_ptr + batch_id * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    store_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=store_mask)


# --------------------------------------------------------------
# Entrypoint
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask        : torch.Tensor of shape [B, N, N] or None (binary)
            - weights     : dict of model parameters (float32 tensors)
            - config      : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) LayerNorm (float32 for stable statistics)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # Work in fp16 for the heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden dimension (fp16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w) # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B,N,N,hidden_dim)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4) Fuse mask and left/right gates (Triton)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is None or nomask:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(torch.float16)

    total_elements = left.numel()                     # B * N * N * hidden_dim
    BLOCK = 32768                                     # tweakable; 32K elements per program

    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = (triton.cdiv(total_elements, BLOCK),)
    _apply_mask_gate_kernel[grid](
        left,
        right,
        mask_tensor.view(-1),
        left_gate,
        right_gate,
        left_out,
        right_out,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 5) Core contraction: batched GEMM left @ rightᵀ
    #    Shapes -> (B*hidden_dim, N, N)
    # ------------------------------------------------------------------
    # reshape to (B, hidden_dim, N, N) then collapse leading two dims
    left_flat = left_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # transpose right for matmul (K,N) layout required by our kernel
    right_T = right_flat.transpose(1, 2).contiguous()

    out_flat = torch.empty_like(left_flat)

    # strides (in elements)
    stride_Ab, stride_Am, stride_Ak = left_flat.stride()
    stride_Bb, stride_Bk, stride_Bn = right_T.stride()
    stride_Cb, stride_Cm, stride_Cn = out_flat.stride()

    # tiling parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_gemm_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        stride_Ab,
        stride_Am,
        stride_Ak,
        stride_Bb,
        stride_Bk,
        stride_Bn,
        stride_Cb,
        stride_Cm,
        stride_Cn,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm over hidden dimension (fp16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)
    return out
