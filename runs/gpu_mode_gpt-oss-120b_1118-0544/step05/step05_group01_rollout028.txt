@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_z = tl.program_id(0)  # combines batch and head
    pid_x = tl.program_id(1)  # i tile
    pid_y = tl.program_id(2)  # j tile

    # compute batch and head indices from pid_z
    b = pid_z // H
    h = pid_z % H

    # base pointers for this batch/head
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # tile offsets
    i0 = pid_x * BLOCK_M
    j0 = pid_y * BLOCK_N

    # Range of i and j
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    # Masks for i, j within N
    mask_i = i < N
    mask_j = j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, N, BLOCK_K):
        k0 = k
        k_idx = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # Load left tile (M, K)
        left_ptrs = left_base + i[:, None] * stride_left_i + k_idx[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for the heavy compute (projections, gates, reduction).
3. Linear projections (no bias) to a hidden dimension `hidden_dim`:
      left_proj  = x @ W_left
      right_proj = x @ W_right
4. Compute sigmoid gates and apply them (and optional binary mask) element‑wise.
5. Core reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton kernel that performs a batched GEMM
   (left @ rightᵀ) for every pair (batch, head).
6. Layer‑norm over the hidden dimension, multiply by the output gate,
   and a final linear projection back to `dim`.
7. Cast back to the original dtype (usually fp32) and return.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the TriMul reduction.
# Computes: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # fp16 * (B, N, N, H)
    right_ptr,         # fp16 * (B, N, N, H)
    out_ptr,           # fp16 * (B, N, N, H)
    B, N, H,           # int32 scalar
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)            # (batch * head)
    pid_x = tl.program_id(1)            # i tile
    pid_y = tl.program_id(2)            # j tile

    # ------- decode batch and head ------------------------------------
    b = pid_z // H
    h = pid_z % H

    # base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # ------- tile offsets ---------------------------------------------
    i0 = pid_x * BLOCK_M
    j0 = pid_y * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        k0 = k

        # left[i, k]  -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + i[:, None] * stride_left_i + k_idx[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right[k, j] (right transposed) -> shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + k_idx[:, None] * stride_right_k + j[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # accumulate dot product
        acc += tl.dot(left, right)

    # ------- write out -------------------------------------------------
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : Tensor[B, N, N, C]   (float32)
        - mask         : Tensor[B, N, N] or None (binary 0/1)
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    Tensor[B, N, N, C]  (same dtype as input)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over channel dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all linear weights to fp16 (no bias)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = torch.nn.functional.linear(x, right_proj_w) # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5. Gating projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask and gating (mask broadcast over hidden dim)
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    if mask is not None and not nomask:
        # mask shape: (B, N, N) -> (B, N, N, 1) for broadcasting
        mask_fp = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 7. Core reduction (batched GEMM) using Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)   # (B, N, N, H), fp16

    # strides for the tensors (in elements, not bytes)
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_j, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    # block sizes (tuned for modern GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # grid: (batch*head, i-tiles, j-tiles)
    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        s_l_b, s_l_i, s_l_k, s_l_h,
        s_r_b, s_r_j, s_r_k, s_r_h,
        s_o_b, s_o_i, s_o_j, s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
