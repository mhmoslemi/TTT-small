@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    mask_ptr,
    left_gate_ptr, right_gate_ptr,
    B, N, H,
    stride_la_batch, stride_la_i, stride_la_k, stride_la_h,  # left stride along batch, i, k, h
    stride_ra_batch, stride_ra_j, stride_ra_k, stride_ra_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    stride_mask_batch, stride_mask_i, stride_mask_j,
    stride_lgate_batch, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_batch, stride_rgate_j, stride_rgate_k, stride_rgate_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)  # batch_h
    pid1 = tl.program_id(1)  # block_i
    pid2 = tl.program_id(2)  # block_j

    # Decode batch and hidden idx
    batch_idx = pid0 // H
    h_idx = pid0 % H

    # Compute offsets for i and j
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = torch.nn.functional.layer_norm(input_tensor,
                (dim,),
                weight=weights["norm.weight"],
                bias=weights["norm.bias"])
    # cast to fp16
    x = x.to(torch.float16)

    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16))
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)))

    # mask
    if mask is None or nomask:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(torch.float16)

    # allocate output
    out = torch.empty_like(left)  # shape B,N,N,H

    # compute strides
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_lgate_batch, stride_lgate_i, stride_lgate_k, stride_lgate_h = left_gate.stride()
    stride_rgate_batch, stride_rgate_j, stride_rgate_k, stride_rgate_h = right_gate.stride()
    stride_mask_batch, stride_mask_i, stride_mask_j = mask_tensor.stride()
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * hidden_dim,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tri_mul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_tensor,
        B=B,
        N=N,
        H=hidden_dim,
        stride_la_batch=stride_left_batch,
        stride_la_i=stride_left_i,
        stride_la_k=stride_left_k,
        stride_la_h=stride_left_h,
        stride_ra_batch=stride_right_batch,
        stride_ra_j=stride_right_j,
        stride_ra_k=stride_right_k,
        stride_ra_h=stride_right_h,
        stride_lgate_batch=stride_lgate_batch,
        stride_lgate_i=stride_lgate_i,
        stride_lgate_k=stride_lgate_k,
        stride_lgate_h=stride_lgate_h,
        stride_rgate_batch=stride_rgate_batch,
        stride_rgate_i=stride_rgate_j,  # using same naming for j
        stride_rgate_k=stride_rgate_k,
        stride_rgate_h=stride_rgate_h,
        stride_mask_batch=stride_mask_batch,
        stride_mask_i=stride_mask_i,
        stride_mask_j=stride_mask_j,
        stride_out_batch=stride_out_batch,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # apply to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    # apply out_gate
    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))
    # cast back
    if input_tensor.dtype != out.dtype:
        out = out.to(input_tensor.dtype)
    return out

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    B, N, H,
    # strides...
    stride_la_batch, stride_la_i, stride_la_k, stride_la_h,
    stride_ra_batch, stride_ra_j, stride_ra_k, stride_ra_h,
    stride_lgate_batch, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_batch, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    stride_mask_batch, stride_mask_i, stride_mask_j,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)  # batch*H
    pid1 = tl.program_id(1)  # block i
    pid2 = tl.program_id(2)  # block j

    batch_idx = pid0 // H
    h_idx = pid0 % H

    # compute global i,j offsets
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Base pointers for left, right, gates, mask, out
    off_batch_h = batch_idx * stride_la_batch + h_idx * stride_la_h
    left_base = left_ptr + off_batch_h
    right_base = right_ptr + batch_idx * stride_ra_batch + h_idx * stride_ra_h
    lgate_base = left_gate_ptr + batch_idx * stride_lgate_batch + h_idx * stride_lgate_h
    rgate_base = right_gate_ptr + batch_idx * stride_rgate_batch + h_idx * stride_rgate_h
    mask_base = mask_ptr + batch_idx * stride_mask_batch
    out_base = out_ptr + batch_idx * stride_out_batch + h_idx * stride_out_h

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left (i,k)
        left_ptrs = left_base + offs_i[:, None] * stride_la_i + offs_k[None, :] * stride_la_k
        lgate_ptrs = lgate_base + offs_i[:, None] * stride_lgate_i + offs_k[None, :] * stride_lgate_k
        mask_left_ptrs = mask_base + offs_i[:, None] * stride_mask_i + offs_k[None, :] * stride_mask_j

        a = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        ag = tl.load(lgate_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        m_left = tl.load(mask_left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        a = a * ag * m_left

        # right (j,k) -> we need B_T (k,j)
        right_ptrs = right_base + offs_k[:, None] * stride_ra_k + offs_j[None, :] * stride_ra_j
        rgate_ptrs = rgate_base + offs_k[:, None] * stride_rgate_k + offs_j[None, :] * stride_rgate_j
        mask_right_ptrs = mask_base + offs_k[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j

        b = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)
        bg = tl.load(rgate_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)
        m_right = tl.load(mask_right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)
        b = b * bg * m_right

        acc += tl.dot(a, b)

    # store output
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) using the provided LN weights.
2. Cast to fp16 and apply two linear projections → left/right  (hidden dim H).
3. Compute three sigmoid gates (left, right, out) from the normalized tensor.
4. Fuse mask, gate and the core N³ reduction in a single Triton kernel:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * mask[b,i,k] * left_gate[b,i,k,h] *
                 right[b,j,k,h] * mask[b,k,j] * right_gate[b,j,k,h]
   The kernel tiles the (i,j) plane (BLOCK_M×BLOCK_N) and iterates over k
   (BLOCK_K). Accumulation is performed in fp32 and the result is stored as
   fp16.
5. Apply a second LayerNorm over the hidden dimension, multiply by the output
   gate and project back to the original channel dimension.
6. Cast the result back to the original dtype.

The kernel implements the heavy O(N³·H) work on the GPU, while the surrounding
operations (LayerNorm, linear layers, sigmoid) are performed with PyTorch
operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    # tensors
    left_ptr, right_ptr, out_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    # runtime parameters
    B, N, H,
    # strides for left (B,N,N,H)
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    # strides for right (B,N,N,H)
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    # strides for left_gate (same layout as left)
    stride_lg_batch, stride_lg_i, stride_lg_k, stride_lg_h,
    # strides for right_gate (same layout as right)
    stride_rg_batch, stride_rg_j, stride_rg_k, stride_rg_h,
    # strides for mask (B,N,N)
    stride_m_batch, stride_m_i, stride_m_j,
    # strides for output (B,N,N,H)
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = Σ_k (left * mask * left_gate) @ (right * mask * right_gate)^T"""
    pid0 = tl.program_id(0)  # combined batch * hidden
    pid1 = tl.program_id(1)  # i‑tile
    pid2 = tl.program_id(2)  # j‑tile

    # decode batch and hidden indices
    batch_idx = pid0 // H
    h_idx = pid0 % H

    # i and j offsets within the tile
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (batch, h) pair
    left_base   = left_ptr   + batch_idx * stride_l_batch   + h_idx * stride_l_h
    right_base  = right_ptr  + batch_idx * stride_r_batch   + h_idx * stride_r_h
    lg_base     = left_gate_ptr  + batch_idx * stride_lg_batch  + h_idx * stride_lg_h
    rg_base     = right_gate_ptr + batch_idx * stride_rg_batch  + h_idx * stride_rg_h
    mask_base   = mask_ptr   + batch_idx * stride_m_batch
    out_base    = out_ptr    + batch_idx * stride_o_batch    + h_idx * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---------- left side (i,k) ----------
        left_ptrs   = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        lg_ptrs     = lg_base   + offs_i[:, None] * stride_lg_i + offs_k[None, :] * stride_lg_k
        mask_l_ptrs = mask_base + offs_i[:, None] * stride_m_i + offs_k[None, :] * stride_m_j

        a  = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        ag = tl.load(lg_ptrs,   mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        ml = tl.load(mask_l_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        a = a * ag * ml

        # ---------- right side (j,k)  (loaded as (k,j) for dot) ----------
        right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        rg_ptrs    = rg_base    + offs_k[:, None] * stride_rg_k + offs_j[None, :] * stride_rg_j
        mask_r_ptrs = mask_base + offs_k[:, None] * stride_m_i + offs_j[None, :] * stride_m_j

        b  = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)
        bg = tl.load(rg_ptrs,   mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)
        mr = tl.load(mask_r_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)
        b = b * bg * mr

        # accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # store the result
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (all float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ----------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------
    # 1) Input LayerNorm
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to fp16 for the heavy computation
    x = x.to(torch.float16)

    # ----------------------------------------------------------
    # 2) Linear projections to hidden dimension (no bias)
    # ----------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ----------------------------------------------------------
    # 3) Compute gates (sigmoid of linear projections)
    # ----------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ----------------------------------------------------------
    # 4) Prepare mask (broadcast over hidden dim)
    # ----------------------------------------------------------
    if mask is None or nomask:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(torch.float16)

    # ----------------------------------------------------------
    # 5) Core reduction: fused mask, gates and N³ multiply via Triton
    # ----------------------------------------------------------
    out = torch.empty_like(left)  # will hold (B, N, N, H)

    # strides (all tensors are contiguous)
    stride_l = left.stride()          # (B,N,N,H)
    stride_r = right.stride()
    stride_lg = left_gate.stride()
    stride_rg = right_gate.stride()
    stride_m = mask_tensor.stride()
    stride_o = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                     # axis 0: batch * hidden
        triton.cdiv(N, BLOCK_M),            # axis 1: i‑tiles
        triton.cdiv(N, BLOCK_N),            # axis 2: j‑tiles
    )

    _tri_mul_gemm_kernel[grid](
        left, right, out,
        left_gate, right_gate,
        mask_tensor,
        B, N, hidden_dim,
        # left strides
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        # right strides
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        # left_gate strides
        stride_lg[0], stride_lg[1], stride_lg[2], stride_lg[3],
        # right_gate strides
        stride_rg[0], stride_rg[1], stride_rg[2], stride_rg[3],
        # mask strides
        stride_m[0], stride_m[1], stride_m[2],
        # out strides
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ----------------------------------------------------------
    # 6) Output LayerNorm over hidden dimension
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float16),
    )

    # ----------------------------------------------------------
    # 7) Apply output gate
    # ----------------------------------------------------------
    out = out * out_gate

    # ----------------------------------------------------------
    # 8) Final linear projection back to original dim
    # ----------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ----------------------------------------------------------
    # 9) Cast back to the original dtype
    # ----------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
