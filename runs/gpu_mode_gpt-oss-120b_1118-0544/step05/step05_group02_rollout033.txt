import torch
import torch.nn.functional as F
import math
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_la, stride_li, stride_lk,
    stride_ra, stride_rk, stride_rn,
    stride_oa, stride_oi, stride_on,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, NUM_K: tl.constexpr,
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this batch (head)
    left_batch_ptr = left_ptr + batch * stride_la
    right_batch_ptr = right_ptr + batch * stride_ra
    out_batch_ptr = out_ptr + batch * stride_oa

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for kk in range(NUM_K):
        k_offset = kk * BLOCK_K
        k_range = tl.arange(0, BLOCK_K)

        # Global k indices
        k_global = k_offset + k_range

        # Masks for out‑of‑bounds k
        k_valid = k_global < N

        # Pointers to the current K‑tile of A and B
        a_ptrs = left_batch_ptr + (offs_m[:, None] * stride_li) + (k_global[None, :] * stride_lk)
        b_ptrs = right_batch_ptr + (k_global[:, None] * stride_rk) + (offs_n[None, :] * stride_rn)

        # Load with masking
        a = tl.load(a_ptrs, mask=mask_m[:, None] & k_valid[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=k_valid[:, None] & mask_n[None, :], other=0.0)

        # Accumulate dot product
        acc += tl.dot(a, b)

    # Write out the tile
    c = acc.to(tl.float16)
    out_ptrs = out_batch_ptr + (offs_m[:, None] * stride_oi) + (offs_n[None, :] * stride_on)
    tl.store(out_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

    Implements the forward pass of the AlphaFold3 TriMul module:
      1. Layer‑norm the 4‑D input.
      2. Linear projections -> left/right, gate projections -> sigmoid.
      3. Optional mask multiplication.
      4. Fuse gates with projected tensors.
      5. Core O(N³·H) reduction using a fused Triton batched GEMM
         (left @ rightᵀ) for each hidden head.
      6. Output layer‑norm, output‑gate and final linear projection.
    All compute‑intensive steps run in fp16, accumulating in fp32.
    The final result is returned in the input dtype (usually fp32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for compute‑heavy stages
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Cast weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (B,N,N,dim) -> (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5. Apply optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left_proj * mask_fp16
        right = right_proj * mask_fp16
    else:
        left = left_proj
        right = right_proj

    # ------------------------------------------------------------------
    # 6. Fuse gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction: out = left @ rightᵀ  (batched)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # reshape to (B*H, N, N) for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)

    # strides in element units
    stride_la = left_flat.stride(0)
    stride_li = left_flat.stride(1)
    stride_lk = left_flat.stride(2)

    stride_ra = right_flat.stride(0)
    stride_rk = right_flat.stride(1)
    stride_rn = right_flat.stride(2)

    stride_oa = out_flat.stride(0)
    stride_oi = out_flat.stride(1)
    stride_on = out_flat.stride(2)

    # Triton block configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    NUM_K = math.ceil(N / BLOCK_K)

    grid = (
        B * H,
        math.ceil(N / BLOCK_M),
        math.ceil(N / BLOCK_N),
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_la,
        stride_li,
        stride_lk,
        stride_ra,
        stride_rk,
        stride_rn,
        stride_oa,
        stride_oi,
        stride_on,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        NUM_K=NUM_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to input dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)

    return out

    stride_la, stride_li, stride_lk,   # strides of left   (batch, i, k)
    stride_ra, stride_rk, stride_rn,   # strides of right  (batch, k, n) – note the transpose
    stride_oa, stride_oi, stride_on,   # strides of out    (batch, i, n)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_K: tl.constexpr,                # = ceil(N / BLOCK_K)
):
    """Batched GEMM: out = left @ rightᵀ for each head."""
    batch = tl.program_id(0)          # combined (batch * head) dimension
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile column

    # offsets inside the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch/head
    left_batch_ptr = left_ptr + batch * stride_la
    right_batch_ptr = right_ptr + batch * stride_ra
    out_batch_ptr = out_ptr + batch * stride_oa

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for kk in range(NUM_K):
        k_offset = kk * BLOCK_K
        k_range = tl.arange(0, BLOCK_K)
        k_global = k_offset + k_range                      # actual k indices
        k_valid = k_global < N

        # pointers to the current K‑tile
        a_ptrs = left_batch_ptr + (offs_m[:, None] * stride_li) + (k_global[None, :] * stride_lk)
        b_ptrs = right_batch_ptr + (k_global[:, None] * stride_rk) + (offs_n[None, :] * stride_rn)

        # load with masking
        a = tl.load(a_ptrs, mask=mask_m[:, None] & k_valid[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=k_valid[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)     # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # store the computed tile
    c = acc.to(tl.float16)
    out_ptrs = out_batch_ptr + (offs_m[:, None] * stride_oi) + (offs_n[None, :] * stride_on)
    tl.store(out_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Steps
    -----
    1) Layer‑norm the input (B, N, N, C).
    2) fp16 linear projections: left/right and their gates.
    3) Optional mask multiplication (broadcast over hidden dim).
    4) Fuse gates with the projections.
    5) Core O(N³·H) reduction performed by a custom Triton GEMM:
       out_h = left_h @ right_hᵀ for each head h.
    6) Output layer‑norm, output‑gate and final linear projection.
    7) Cast back to the original dtype.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Cast model weights to fp16
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (B,N,N,dim) -> (B,N,N,hidden_dim)
    # -----------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 4. Gate projections + sigmoid
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 5. Optional mask (broadcast over hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left_proj * mask_fp16
        right = right_proj * mask_fp16
    else:
        left = left_proj
        right = right_proj

    # -----------------------------------------------------------------
    # 6. Fuse gates
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 7. Core reduction: out = left @ rightᵀ   (batched GEMM)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape               # (B, N, N, H)

    # reshape to (B*H, N, N) for a plain batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_flat = torch.empty_like(left_flat)   # fp16 output

    # strides in element units
    stride_la = left_flat.stride(0)
    stride_li = left_flat.stride(1)
    stride_lk = left_flat.stride(2)

    stride_ra = right_flat.stride(0)
    stride_rk = right_flat.stride(1)
    stride_rn = right_flat.stride(2)

    stride_oa = out_flat.stride(0)
    stride_oi = out_flat.stride(1)
    stride_on = out_flat.stride(2)

    # Triton block sizes – tuned for fp16 on A100/B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    NUM_K = math.ceil(N / BLOCK_K)

    grid = (
        B * H,                                 # batch * head dimension
        math.ceil(N / BLOCK_M),                # rows
        math.ceil(N / BLOCK_N),                # cols
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_la,
        stride_li,
        stride_lk,
        stride_ra,
        stride_rk,
        stride_rn,
        stride_oa,
        stride_oi,
        stride_on,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        NUM_K=NUM_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # -----------------------------------------------------------------
    # 9. Cast back to the original dtype
    # -----------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
