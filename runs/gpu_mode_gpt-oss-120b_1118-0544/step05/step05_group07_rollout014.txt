"""
TriMul (outgoing) forward pass – Triton fused mask/gate and batched GEMM.

Algorithm
---------
1. Input LayerNorm (float32) → fp16.
2. Linear projections (left/right) and gates (left/right/out) in fp16.
3. Fuse projection, gate and optional mask with a tiny Triton kernel
   (`_gate_mask_kernel`).  This avoids an extra memory pass.
4. Core reduction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
   Implemented as a batched GEMM (B*H independent N×N matrix
   multiplications) using a custom Triton kernel (`_batched_gemm_kernel`).
5. Second LayerNorm on the hidden dimension, multiply by out_gate, and
   project back to the original channel dimension.
6. Cast the result back to the input dtype (float32).

Both kernels run on the GPU (B200) and use fp16 compute with fp32
accumulation for numeric stability.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) on a flattened view
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32   – total number of elements in proj/gate
    hidden_dim,        # i32   – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask index (broadcast over hidden dimension)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using Triton.
    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],           # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ B) where A: (Z, M, K), B: (Z, K, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_ay, stride_ax,   # batch, row, col strides for A (Z, M, K)
    stride_bz, stride_by, stride_bx,   # batch, row, col strides for B (Z, K, N)
    stride_cz, stride_cy, stride_cx,   # batch, row, col strides for C (Z, M, N)
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)   # batch dimension (Z = B * hidden_dim)
    pid_m = tl.program_id(1)   # output row block
    pid_n = tl.program_id(2)   # output col block

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in float32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = A_ptr + pid_z * stride_az \
                 + offs_m[:, None] * stride_ay \
                 + offs_k[None, :] * stride_ax
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)

        # Load B tile (K x N)
        b_ptrs = B_ptr + pid_z * stride_bz \
                 + offs_k[:, None] * stride_by \
                 + offs_n[None, :] * stride_bx
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        # Matrix multiplication for the tile
        acc += tl.dot(a, b)

    # Write back the result tile
    c_ptrs = C_ptr + pid_z * stride_cz \
             + offs_m[:, None] * stride_cy \
             + offs_n[None, :] * stride_cx
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def _batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.
    A: (Z, M, K) fp16
    B: (Z, K, N) fp16  (already transposed as needed)
    Returns C: (Z, M, N) fp16
    """
    assert A.is_contiguous() and B.is_contiguous()
    Z, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Inner dimensions must match."

    C = torch.empty((Z, M, N), dtype=torch.float16, device=A.device)

    # Strides as Python ints
    stride_az, stride_ay, stride_ax = A.stride()
    stride_bz, stride_by, stride_bx = B.stride()
    stride_cz, stride_cy, stride_cx = C.stride()

    # Tile sizes (tunable)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        Z,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_gemm_kernel[grid](
        A,
        B,
        C,
        stride_az,
        stride_ay,
        stride_ax,
        stride_bz,
        stride_by,
        stride_bx,
        stride_cz,
        stride_cy,
        stride_cx,
        M,
        N,
        K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------- LayerNorm (input) -------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)  # compute in fp16

    # ----------------------- cast weights to fp16 -----------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ----------------------- linear projections ---------------------------
    # shape: (B, N, N, H)
    left_proj = torch.nn.functional.linear(x, left_proj_w)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------- gates -----------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()  # (B,N,N,H)

    # ------------------- mask & gate fusion (Triton) --------------------
    if mask is not None and not nomask:
        # mask is float (0/1); broadcast over hidden_dim
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ----------------- core O(N³·H) reduction via custom batched GEMM ----------
    # Rearrange to (B, H, N, N) and then to (B*H, N, N)
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left_perm.view(-1, N, N)   # [B*H, N, N]
    right_mat = right_perm.view(-1, N, N) # [B*H, N, N]

    # Transpose the second matrix for GEMM: (Z, K, N)
    right_mat_T = right_mat.transpose(-2, -1).contiguous()

    # Perform batched GEMM: C = left_mat @ right_mat_T
    out_mat = _batched_gemm(left_mat, right_mat_T)   # [B*H, N, N] fp16

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim)

    # Cast back to original dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
