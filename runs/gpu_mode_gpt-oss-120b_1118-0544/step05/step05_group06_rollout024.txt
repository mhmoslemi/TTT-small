@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_aba, stride_abm, stride_abk,
    stride_bba, stride_bbk, stride_bbn,
    stride_cca, stride_ccm, stride_ccn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    # Starting offsets for this block
    offs_m = pid_m*BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n*BLOCK_N + tl.arange(0, BLOCK_N)
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute pointer offsets for batch dimension
    a_batch_ptr = a_ptr + batch_id * stride_aba
    b_batch_ptr = b_ptr + batch_id * stride_bba
    c_batch_ptr = c_ptr + batch_id * stride_cca

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k*BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # mask for bounds
        a_mask = (offs_m < M) & (offs_k < K)
        b_mask = (offs_n < N) & (offs_k < K)
        # Load blocks
        a = tl.load(a_batch_ptr + offs_m[:, None] * stride_abm + offs_k[None, :] * stride_abk,
                    mask=a_mask, other=0.0)
        b = tl.load(b_batch_ptr + offs_k[:, None] * stride_bbk + offs_n[None, :] * stride_bbn,
                    mask=b_mask, other=0.0)
        # MK dot NK -> MN
        acc += tl.dot(a, b)
    # Store result
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_batch_ptr + offs_m[:, None] * stride_ccm + offs_n[None, :] * stride_ccn,
             acc.to(tl.float16), mask=c_mask)

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_ab, stride_am, stride_ak,
    stride_bb, stride_bk, stride_bn,
    stride_cb, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_batch_ptr = a_ptr + batch * stride_ab
    b_batch_ptr = b_ptr + batch * stride_bb
    c_batch_ptr = c_ptr + batch * stride_cb

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K loop
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # bounds
        a_mask = (offs_m < M) & (offs_k < K)
        b_mask = (offs_n < N) & (offs_k < K)

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a = tl.load(a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
                    mask=a_mask, other=0.0)
        # Load B block transposed: B is originally shape (N,K); we need B^T (K,N).
        # Since we want to multiply A (M,K) with B^T (K,N).
        # View B's memory as (K,N) using stride for k and n.
        b = tl.load(b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
                    mask=b_mask, other=0.0)
        # compute multiply-accumulate
        acc += tl.dot(a, b)

    # write back
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
             acc.to(tl.float16), mask=c_mask)

"""
TriMul (outgoing) forward pass with Triton fusion.

Algorithm
---------
1. Layer‐normalize the input tensor (B,N,N,C) (C = dim).
2. Cast to FP16 for compute‑heavy stages.
3. Linear projections to hidden dimension H (left_proj, right_proj) and
   three gating projections (left_gate, right_gate, out_gate).
4. Fuse mask (if present), projection, and gate with a small Triton kernel:
   out = proj * gate * mask   (mask is broadcast over H).
5. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batch of H independent matrix multiplications
   (left @ rightᵀ). Implemented with a custom Triton batched GEMM kernel.
6. Layer‑normalize over the hidden dimension, apply out_gate, and a final
   linear projection back to the original dim.
7. Cast the result back to the input dtype (usually float32).

Only the mask‑gate fusion and the batched GEMM are executed in Triton;
all other operations use PyTorch’s highly‑optimized kernels.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused proj * gate * mask (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # *[total_elems]   fp16
    gate_ptr,          # *[total_elems]   fp16
    mask_ptr,          # *[mask_elems]    fp16
    out_ptr,           # *[total_elems]   fp16
    total_elems,       # i64
    hidden_dim,        # i64 (size of last dim H)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # mask for out‑of‑bounds elements
    valid = offs < total_elems

    # each H consecutive elements share the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(out_ptr + offs, proj * gate * m, mask=valid)


def _apply_gate_and_mask(proj, gate, mask, block_size: int = 1024):
    """
    Element‑wise ``proj * gate * mask`` where ``mask`` has shape (B,N,N)
    and is broadcast over the hidden dimension H.
    All tensors must be FP16 and contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)
    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)

    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],   # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for left @ rightᵀ
#   A: (B, M, K)  (here M = N, K = N)
#   B: (B, N, K)  (we treat it as transposed inside the kernel)
#   C: (B, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_ab, stride_am, stride_ak,   # A: batch, M, K strides
    stride_bb, stride_bk, stride_bn,   # B: batch, K, N strides (note: K stride = stride for last dim)
    stride_cb, stride_cm, stride_cn,   # C: batch, M, N strides
    M, N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m   = tl.program_id(1)
    pid_n   = tl.program_id(2)

    # compute start offsets of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch = a_ptr + batch_id * stride_ab
    b_batch = b_ptr + batch_id * stride_bb
    c_batch = c_ptr + batch_id * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # masks for out‑of‑bounds
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # A block (M x K)
        a = tl.load(
            a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
            mask=a_mask, other=0.0
        )
        # B block transposed (K x N) -> we read B as (K,N) using stride_bk for K and stride_bn for N
        b = tl.load(
            b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=b_mask, other=0.0
        )
        acc += tl.dot(a, b)   # (M,N) accumulator in fp32

    # write back
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(
        c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
        acc.to(tl.float16), mask=c_mask
    )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict containing all linear and LayerNorm parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy stages
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Fuse mask, projection and gate (Triton) if required
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # left shape (B, N, N, H)

    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # output buffer
    out_flat = torch.empty_like(left_flat)

    # strides for the three tensors (batch‑first, contiguous)
    stride_ab = left_flat.stride(0)   # batch stride
    stride_am = left_flat.stride(1)   # M stride (N)
    stride_ak = left_flat.stride(2)   # K stride (N)

    stride_bb = right_flat.stride(0)  # batch stride
    stride_bk = right_flat.stride(2)  # K stride (last dim) = 1
    stride_bn = right_flat.stride(1)  # N stride = K

    stride_cb = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    M = N
    N_dim = N
    K = N

    # Blocking parameters (tuned for typical B200 GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        math.ceil(N / BLOCK_M),
        math.ceil(N / BLOCK_N),
    )

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        stride_ab,
        stride_am,
        stride_ak,
        stride_bb,
        stride_bk,
        stride_bn,
        stride_cb,
        stride_cm,
        stride_cn,
        M,
        N_dim,
        K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)                      # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
