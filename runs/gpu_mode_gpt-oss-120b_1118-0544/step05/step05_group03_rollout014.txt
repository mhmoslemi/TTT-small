@triton.jit
def apply_gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, total_elements, hidden_dim, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs // hidden_dim
    valid = offs < total_elements
    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask, mask=valid, other=1.0)
    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)
```

Now define function to call it:

```python
def apply_gate_and_mask(proj, gate, mask, block_size=32768):
    B,N,_,H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    apply_gate_mask_kernel[grid](proj, gate, mask, out, total, H, BLOCK_SIZE=block_size)
    return out
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # cast to fp16
    x = x.to(torch.float16)

    # combine projection weights
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    combined_weight = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=1)  # shape (dim, 5*hidden_dim)
    combined_weight = combined_weight.to(torch.float16)

    # linear projection
    combined = F.linear(x, combined_weight)  # shape (B,N,N,5*hidden_dim)
    # split
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.chunk(combined, 5, dim=-1)
    # apply sigmoid to gates
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # mask handling
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # core reduction via batched GEMM (half precision)
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_flat = right.permute(0,3,1,2).contiguous().view(B*H, N, N)

    # matmul
    out_flat = torch.bmm(left_flat, right_flat.transpose(1,2))
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()

    # output layer norm
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate

    # final linear
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)

    # cast back to original dtype (float32)
    out = out.to(input_tensor.dtype)

    return out
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to FP16 for compute‑heavy ops.
3. Fuse the five linear projections (left, right, left_gate, right_gate,
   out_gate) into a single GEMM by concatenating the weight matrices.
4. Split the fused result, apply sigmoid to the three gate tensors.
5. Fuse element‑wise mask, gate and projection using a Triton kernel
   (proj * gate * mask).  If no mask is provided the kernel is skipped.
6. Core reduction: for each batch and hidden index compute
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM (torch.bmm) on the reshaped tensors.
7. Output LayerNorm, multiply by out_gate, final linear projection.
8. Cast back to the original input dtype.

Only the mask/gate fusion is performed in Triton; all other heavy
computations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fusion_kernel(
    proj_ptr,     # fp16*  (B*N*N*H,)
    gate_ptr,     # fp16*  (B*N*N*H,)
    mask_ptr,     # fp16*  (B*N*N,)
    out_ptr,      # fp16*  (B*N*N*H,)
    total_elems,  # i32    total number of elements in proj/gate (B*N*N*H)
    hidden_dim,   # i32    size H
    BLOCK_SIZE: tl.constexpr,
):
    """Compute out = proj * gate * mask (mask broadcast over H)."""
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds
    mask = offs < total_elems

    # Index of the mask element (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 32768) -> torch.Tensor:
    """
    Fuse projection, gate and mask (if present) with a Triton kernel.
    All tensors must be fp16, contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N) broadcasted over H
    Returns proj * gate * mask with shape (B, N, N, H).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    _gate_mask_fusion_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor of shape [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"         – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask"      – bool, if True ignore the mask
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the heavy compute path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Fuse the five linear projections into a single GEMM
    #    (left_proj, right_proj, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    # Gather weight matrices (each of shape [hidden_dim, dim] except gates)
    left_proj_w   = weights["left_proj.weight"]   # (H, dim)
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Concatenate along the output dimension -> shape (5*H, dim)
    combined_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    ).to(torch.float16)  # (5*H, dim)

    # Single linear projection
    combined = F.linear(x, combined_weight)  # (B, N, N, 5*H)

    # Split the fused tensor
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        combined, hidden_dim, dim=-1
    )  # each (B, N, N, H)

    # ------------------------------------------------------------------
    # 4. Apply sigmoid to gate tensors
    # ------------------------------------------------------------------
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse mask and gate (if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16, contiguous, and broadcastable
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        # Simple element‑wise multiplication when no mask
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Perform batched matrix multiplication (FP16)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout (B, N, N, H)
    out = (
        out_flat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 8. Cast back to the original dtype of the input
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
