"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor (B, N, N, C).
2. Linear projection to a hidden dimension H for left and right streams.
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. Fuse mask (if present) and the left/right gates in a single Triton kernel.
5. Perform the core O(N³·H) reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication
   where each (b,h) pair is a separate matrix and we multiply
   left @ rightᵀ using a custom Triton kernel.
6. Layer‑normalize the hidden result, apply the output gate and a final
   linear projection back to the original channel dimension.
7. Cast the output back to the input dtype.

Only the mask‑gate fusion and the heavy reduction are written in Triton;
all other ops use PyTorch (which is highly optimized on the target GPU).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. Fuse mask and left/right gates (element‑wise)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # const fp16* left   (B,N,N,H) flattened
    right_ptr,         # const fp16* right
    mask_ptr,          # const fp16* mask   (B,N,N) flattened
    left_gate_ptr,     # const fp16* left_gate
    right_gate_ptr,    # const fp16* right_gate
    left_out_ptr,      # fp16* left_out
    right_out_ptr,     # fp16* right_out
    H: tl.constexpr,       # hidden dimension
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # load tensors
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # mask is per (B,N,N) element -> broadcast over H
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # apply
    tl.store(left_out_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


# ----------------------------------------------------------------------
# 2. Batched matrix multiplication: C = A @ Bᵀ
#    Shapes: (batch, M, K) @ (batch, N, K).T -> (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (batch, M, K)
    stride_bk, stride_bn,          # B: (batch, N, K)  (we will read B as (batch, K, N))
    stride_cm, stride_cn,          # C: (batch, M, N)
    stride_az, stride_bz, stride_cz,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_z = tl.program_id(2)   # batch index (B*H)

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator in fp32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers to the current tiles of A and B
        a_ptrs = (
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + k_off[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_z * stride_bz
            + k_off[:, None] * stride_bk   # B is read as (batch, K, N)
            + offs_n[None, :] * stride_bn
        )

        # Masks for out‑of‑bound loads
        a_mask = (offs_m[:, None] < M) & (k_off[None, :] < K)
        b_mask = (k_off[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        # Accumulate dot product
        acc += tl.dot(a, b)

    # Write the result back (cast to fp16)
    c_ptrs = (
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


# ----------------------------------------------------------------------
# 3. Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask        : torch.Tensor of shape [B, N, N] or None
            - weights     : dict of model parameters (float32 tensors)
            - config      : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)      # use FP16 for the heavy part

    # ------------------------------------------------------------------
    # 2) Cast linear weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Fuse mask and left/right gates
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_fp = mask.to(torch.float16)
    else:
        mask_fp = torch.ones((B, N, N), dtype=torch.float16, device=device)

    total_elements = left.numel()          # B * N * N * H
    BLOCK = 65536                           # reasonable block size for 1‑D launch

    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = (triton.cdiv(total_elements, BLOCK),)
    _mask_gate_fuse_kernel[grid](
        left,
        right,
        mask_fp.view(-1),          # flatten mask
        left_gate,
        right_gate,
        left_out,
        right_out,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6) Core reduction: batched matmul (left @ rightᵀ)
    #    Shape after view: (B*H, N, N)
    # ------------------------------------------------------------------
    # bring hidden dim to batch dimension
    left_perm = left_out.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_out.permute(0, 3, 1, 2).contiguous()
    # transpose the second operand
    right_T = right_perm.transpose(-2, -1).contiguous()

    batch_h = left_perm.shape[0] * left_perm.shape[1]      # B*H
    left_flat = left_perm.view(-1, N, N)                  # (B*H, N, N)
    right_flat = right_T.view(-1, N, N)                    # (B*H, N, N)

    out_flat = torch.empty_like(left_flat)

    # strides (in elements)
    stride_a_batch = left_flat.stride(0)
    stride_a_row   = left_flat.stride(1)
    stride_a_col   = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_row   = right_flat.stride(1)   # will be used as stride_bk
    stride_b_col   = right_flat.stride(2)   # will be used as stride_bn

    stride_c_batch = out_flat.stride(0)
    stride_c_row   = out_flat.stride(1)
    stride_c_col   = out_flat.stride(2)

    # Block sizes – tune for the target GPU
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_h,
    )
    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_a_row, stride_a_col,
        stride_b_row, stride_b_col,
        stride_c_row, stride_c_col,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 7) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10) Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 11) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
