    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
    M, N, K,

def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute base pointers for this batch
    a_ptrs = a_ptr + batch_idx * batch_stride_a
    b_ptrs = b_ptr + batch_idx * batch_stride_b
    c_ptrs = c_ptr + batch_idx * batch_stride_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # So loop over K
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)

        # mask for valid loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak), mask=a_mask, other=0.0)

        # dot

    right_proj = torch.nn.functional.linear(x, right_proj_w)

    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()

        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,

    )

    out = out * out_gate

    out = torch.nn.functional.linear(out, to_out_w)

    # cast back to original dtype
    out = out.to(dtype_in)

```
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_ptrs = a_ptr + batch_idx * batch_stride_a
    b_ptrs = b_ptr + batch_idx * batch_stride_b
    c_ptrs = c_ptr + batch_idx * batch_stride_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)

        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
                    mask=a_mask, other=0.0)
        b = tl.load(b_ptrs + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
                    mask=b_mask, other=0.0)

        acc += tl.dot(a, b)

    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c = acc.to(tl.float16)
    tl.store(c_ptrs + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn), c, mask=c_mask)
```

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to FP16 for compute‑heavy stages.
3. Linear projections (no bias) to a hidden dimension H.
4. Compute sigmoid gates.
5. Fuse mask, projection and gate (if a mask is present) with a small Triton kernel.
6. Core reduction:
     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM using a custom Triton kernel that processes
   each (batch,head) pair as an independent matrix multiply.
7. Layer‑norm over the hidden dimension, multiply by the output gate and a final
   linear projection back to the original channel dimension.
8. Cast the result back to the input dtype.

The implementation mixes Torch primitives (layer‑norm, linear, sigmoid) with
two Triton kernels: one for mask‑gate fusion and one for the O(N³) reduction.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Kernel 1: fuse projection, gate and mask (mask is broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B, N, N, H)
    gate_ptr,          # fp16 * (B, N, N, H)
    mask_ptr,          # fp16 * (B, N, N)
    out_ptr,           # fp16 * (B, N, N, H)
    total_elems,       # B*N*N*H
    hidden_dim,        # H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # mask is repeated hidden_dim times – obtain corresponding mask index
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse ``proj * gate * mask`` (mask broadcast over the hidden dimension).
    All tensors must be ``torch.float16`` and contiguous.
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the core TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    batch_stride_a, batch_stride_b, batch_stride_c,   # strides between batches (in elements)
    stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,  # matrix strides
    M, N, K,                      # dimensions
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets within the M/N tiles
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch = a_ptr + batch_idx * batch_stride_a
    b_batch = b_ptr + batch_idx * batch_stride_b
    c_batch = c_ptr + batch_idx * batch_stride_c

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)

        # mask out‑of‑bounds elements
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=a_mask,
            other=0.0,
        )
        b = tl.load(
            b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=b_mask,
            other=0.0,
        )
        acc += tl.dot(a, b)   # fp16×fp16 → fp32 accumulation

    # write back
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=c_mask,
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None (binary)
            - weights: dict of model parameters (float32)
            - config: dict with keys
                * "dim" (C)
                * "hidden_dim" (H)
                * "nomask" (optional, bool)

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C]  (float32)

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to FP16 (no bias terms)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (to hidden dimension)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, H]
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask & gates (fused via Triton when a mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (B*H independent matrix multiplies)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Bring the hidden dimension to the batch axis and transpose right for GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)  (k,j)

    left_mat   = left_perm.view(-1, N, N)   # (B*H, N, N)  -> (batch, M, K)
    right_mat  = right_perm.view(-1, N, N)  # (B*H, N, N)  -> (batch, K, N)
    batch_cnt  = left_mat.shape[0]

    out_mat = torch.empty_like(left_mat)

    # Strides in element units
    stride_am = left_mat.stride(1)   # stride over M
    stride_ak = left_mat.stride(2)   # stride over K
    stride_bk = right_mat.stride(1)  # stride over K
    stride_bn = right_mat.stride(2)  # stride over N
    stride_cm = out_mat.stride(1)    # stride over M
    stride_cn = out_mat.stride(2)    # stride over N

    batch_stride_a = left_mat.stride(0)
    batch_stride_b = right_mat.stride(0)
    batch_stride_c = out_mat.stride(0)

    # Tiling parameters – tuned for typical GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_cnt,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        N,  # M
        N,  # N
        N,  # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output layer‑norm, output gate and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype (typically float32)
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)

    return out
