"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate.
Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) with the provided LN weights.
2. Cast to fp16 for compute‑intensive ops.
3. Perform four linear projections (left/right + their gates) in fp16.
4. Fuse each projection with its gate and the optional mask using a tiny
   Triton kernel (`_gate_mask_kernel`).  The kernel multiplies
   `proj * gate * mask` element‑wise, broadcasting the mask over the hidden
   dimension.
5. Compute the core O(N³·H) reduction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is equivalent to `H` independent matrix multiplications of size
   N×N.  We reshape to (B*H, N, N) and call `torch.bmm`, which uses the
   highly‑optimized cuBLAS batched GEMM.
6. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate, and project back to the original channel dimension.
7. Cast the result back to the input dtype (float32).

Only the mask‑gate fusion is performed in Triton, satisfying the requirement
to have a custom kernel while keeping the overall runtime fast.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _gate_mask_kernel(
    proj_ptr,        # fp16 * (B*N*N*H)
    gate_ptr,        # fp16 * (B*N*N*H)
    mask_ptr,        # fp16 * (B*N*N)
    out_ptr,         # fp16 * (B*N*N*H)
    total_elems,     # i32  – total number of elements in proj/gate
    hidden_dim,      # i32  – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    """Element‑wise multiply proj * gate * mask (mask broadcast over H)."""
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # validity mask for the tail block
    valid = offs < total_elems

    # each hidden_dim consecutive elements share the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using Triton.

    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],   # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dictionary of model parameters (float32)
            - config: dict containing at least "dim" and "hidden_dim".
                      Optional key "nomask" (bool) to ignore the mask.

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype  # keep track to cast back at the end

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for the rest of the pipeline
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Cast weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,H]
    right_proj = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5. Fuse projection, gate and optional mask using Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous; broadcast later inside kernel
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core O(N³·H) reduction via batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Transform to shape (B*H, N, N) for batched matrix multiplication
    # left: [B, N, N, H] -> [B, H, N, N] -> (B*H, N, N)
    left_reshaped = left.permute(0, 3, 1, 2).contiguous().view(-1, left.shape[1], left.shape[2])
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(-1, right.shape[1], right.shape[2])

    # Perform batched GEMM: (B*H, N, N) @ (B*H, N, N)^T
    out_reshaped = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))

    # Reshape back to [B, N, N, H]
    out = out_reshaped.view(left.shape[0], hidden_dim, left.shape[1], left.shape[2])
    out = out.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7. Second LayerNorm + output gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
