"""
Optimized outgoing TriMul forward pass.

Algorithm
---------
1. Layer‑norm the input (float32 for stability).
2. Cast to fp16 and compute left / right projections and their sigmoid gates.
3. Fuse mask (if present) with the gates using plain PyTorch elementwise ops.
4. Perform the heavy O(N³·H) contraction via a custom Triton batched‑GEMM
   kernel that computes, for each head h,   out_h = left_h @ right_hᵀ .
5. Apply the output layer‑norm, output gate and final linear projection.
6. Cast the result back to the original dtype.

Only the contraction (step 4) is implemented in Triton, satisfying the
requirement to use a kernel while keeping the overall runtime low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to fp16 tensors
    M, N, K,                           # matrix dimensions
    stride_aa, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_bb, stride_bk, stride_bn,   # strides for B (batch, K, N)
    stride_cc, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ Bᵀ.
    A, B, C shapes: (batch, M, K), (batch, N, K) (note B is transposed in the
    multiplication), (batch, M, N).  All tensors are stored in row‑major order.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / columns
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the beginning of the current batch
    a_batch_ptr = a_ptr + batch_id * stride_aa
    b_batch_ptr = b_ptr + batch_id * stride_bb
    c_batch_ptr = c_ptr + batch_id * stride_cc

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load block of A (M × K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am
                                 + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load block of B (N × K) – note we need Bᵀ, so we load as (K × N)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk
                                 + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Convert to fp32 and accumulate
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)

    # Write back the result in fp16
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm
                             + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for the heavy compute
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float32)  # keep in fp32 for final projection

    # -----------------------------------------------------------------
    # 3. Linear projections and gates (fp16)
    # -----------------------------------------------------------------
    left  = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)         # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, H)

    # -----------------------------------------------------------------
    # 4. Apply mask and gate (pure PyTorch – cheap)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * left_gate * mask_fp
        right = right * right_gate * mask_fp
    else:
        left = left * left_gate
        right = right * right_gate

    # Release temporary gate tensors early
    del left_gate, right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ rightᵀ) per head
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # Rearrange to (batch, M, K) where batch = B*H, M = N, K = N
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch = B * H
    left_mat = left_mat.view(batch, N, N)
    right_mat = right_mat.view(batch, N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat)

    # Triton block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,  # M
        N,  # N
        N,  # K
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),   # A strides
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),# B strides
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),     # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output processing
    # -----------------------------------------------------------------
    # Compute out_gate (still in fp16, then cast to fp32)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w)).to(torch.float32)

    # Output LayerNorm (float32)
    out = out.to(torch.float32)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to original dim
    out = F.linear(out, to_out_w)      # (B, N, N, dim), fp32

    # Cast back to the original dtype of the input
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
