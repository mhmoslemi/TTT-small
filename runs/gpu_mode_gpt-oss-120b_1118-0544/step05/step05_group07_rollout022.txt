def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor,
        (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # cast to fp16
    x = x.to(torch.float16)

    # flatten
    B, N, _, _ = x.shape   # shape (B,N,N,dim)
    total_positions = B * N * N
    x_flat = x.reshape(total_positions, dim)   # (B*N*N, dim)

    # Concatenate weights for proj and gate
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    proj_w_cat = torch.cat([left_proj_w, right_proj_w], dim=0)  # (2*H, dim)

    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    gate_w_cat = torch.cat([left_gate_w, right_gate_w], dim=0)  # (2*H, dim)

    # Compute projections and gates
    proj_cat = torch.nn.functional.linear(x_flat, proj_w_cat)   # (total_positions, 2*H)
    gate_cat = torch.nn.functional.linear(x_flat, gate_w_cat)   # (total_positions, 2*H)
    gate_cat = torch.sigmoid(gate_cat)

    # split
    left_proj = proj_cat[:, :hidden_dim]
    right_proj = proj_cat[:, hidden_dim:]
    left_gate = gate_cat[:, :hidden_dim]
    right_gate = gate_cat[:, hidden_dim:]

    # Apply mask & gate using Triton (fuse)
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).reshape(total_positions)  # (total_positions)
        # left = left_proj * left_gate * mask_fp16
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16, block_size=1024)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16, block_size=1024)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # reshape left/right to (B, N, N, H)
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # compute out via batched gemm
    # permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # reshape to (B*H, N, N)
    left_mat = left_perm.view(-1, N, N)
    right_mat = right_perm.view(-1, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape back to (B, H, N, N)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # second layernorm
    out = torch.nn.functional.layer_norm(out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"])

    # compute out_gate
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.nn.functional.linear(x_flat, out_gate_w)   # (total_positions, hidden_dim)
    out_gate = torch.sigmoid(out_gate)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    out = out * out_gate

    # final linear
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # cast back
    out = out.to(dtype_in)

    return out
```

@triton.jit
def _gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                      total_elems, hidden_dim,
                      BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems
    mask_idx = offs // hidden_dim
    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)
    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)

def _apply_gate_and_mask(proj, gate, mask, block_size=1024):
    assert proj.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out

"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) using the provided
   weight/bias (fp16).
2. Flatten the tensor to (B*N*N, C) and perform **one** large linear
   projection that yields all required intermediate tensors:
   left_proj, right_proj, left_gate, right_gate, out_gate.
3. Apply sigmoid to the gate tensors.
4. Fuse element‑wise mask, projection and gating for ``left`` and
   ``right`` with a tiny Triton kernel (avoids extra memory passes).
5. Reshape to (B,N,N,H) and compute the core reduction
   ``out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]``
   as a batched GEMM: for each hidden channel perform
   ``L @ Rᵀ`` using ``torch.bmm``.
6. Second LayerNorm over the hidden dimension, multiply by ``out_gate``,
   and project back to the original channel dimension.
7. Cast the result back to the input dtype (float32).

Only the mask‑gate fusion is implemented in Triton; all other
operations use PyTorch's highly‑optimized kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask)
# --------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (total_elems,)
    gate_ptr,          # fp16 * (total_elems,)
    mask_ptr,          # fp16 * (total_positions,)
    out_ptr,           # fp16 * (total_elems,)
    total_elems,       # i32
    hidden_dim,        # i32
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # mask is broadcasted over the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse ``proj * gate * mask`` with Triton.

    Arguments:
        proj: (total_positions, hidden_dim) fp16 tensor (contiguous)
        gate: same shape as ``proj`` (fp16)
        mask: (total_positions,) fp16 tensor (contiguous)
        block_size: Triton block size

    Returns:
        Tensor of shape (total_positions, hidden_dim) (fp16)
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],        # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of FP32 model parameters
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------- LayerNorm (input) -------------------------------
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )
    x = x.to(torch.float16)                           # [B, N, N, C]

    B, N, _, _ = x.shape
    total_positions = B * N * N                         # B·N·N
    x_flat = x.reshape(total_positions, dim)            # [B·N·N, C]

    # ------------------- Combined linear projection -----------------------
    # Concatenate all required weight matrices:
    #   left_proj, right_proj, left_gate, right_gate, out_gate
    left_proj_w = weights["left_proj.weight"].to(torch.float16)    # (H, C)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)  # (H, C)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)    # (H, C)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)  # (H, C)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)      # (H, C)

    combined_w = torch.cat(
        [left_proj_w, right_proj_w,
         left_gate_w, right_gate_w,
         out_gate_w],
        dim=0,
    )                                                    # (5*H, C)

    # One huge matmul
    all_feat = torch.nn.functional.linear(x_flat, combined_w)  # (B·N·N, 5·H)

    H = hidden_dim
    left_proj   = all_feat[:, 0:H]
    right_proj  = all_feat[:, H:2*H]
    left_gate_raw  = all_feat[:, 2*H:3*H]
    right_gate_raw = all_feat[:, 3*H:4*H]
    out_gate_raw   = all_feat[:, 4*H:5*H]

    left_gate   = torch.sigmoid(left_gate_raw)
    right_gate  = torch.sigmoid(right_gate_raw)
    out_gate    = torch.sigmoid(out_gate_raw)   # (B·N·N, H)

    # ------------------- mask & gate fusion (Triton) --------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).reshape(total_positions)  # (B·N·N,)
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # Reshape back to (B, N, N, H)
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # ----------------- core O(N³·H) reduction via batched GEMM ----------
    # Permute to (B, H, N, N) and flatten batch*H
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_mat = left_perm.view(-1, N, N)                  # (B*H, N, N)
    right_mat = right_perm.view(-1, N, N)                # (B*H, N, N)

    # GEMM: L @ Rᵀ  → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- second LayerNorm & output gate -----------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # reshape out_gate to (B, N, N, H) and apply
    out_gate = out_gate.view(B, N, N, H)
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (C, H)
    out = torch.nn.functional.linear(out, to_out_w)        # (B, N, N, C)

    # ------------------------ cast back to input dtype -----------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
