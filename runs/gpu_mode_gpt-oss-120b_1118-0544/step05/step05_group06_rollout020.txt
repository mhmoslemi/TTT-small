    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_head, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_head, stride_r_j, stride_r_k,

):

    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    m_j = j < N

    o_ptrs = out_ptr + batch * stride_ob + head * stride_oh

    # mask for bounds
    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # load left: shape (BLOCK_M, BLOCK_K)

        a = tl.load(left_offset, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right: shape (BLOCK_N, BLOCK_K)
        # for right we want right[batch, j, k, head]

        # dot: a (M,K) dot b (N,K) trans_b=True => (M,N)
        acc += tl.dot(a, b, trans_b=True)

    # store
    out_offset = o_ptrs + i[:, None] * stride_oi + j[None, :] * stride_ok
    tl.store(out_offset, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
Optimized Triton implementation of the outgoing Triangle Multiplicative Update
(TriMul) operator (AlphaFold3 style).

Algorithm
---------
1. Layer‑normalize the input tensor (B, N, N, C) with the provided LN weights.
2. Cast to FP16 for the heavy compute.
3. Linear projections → (B, N, N, H) for “left” and “right”.
4. Compute gate values (sigmoid) and optionally apply a binary mask.
   The element‑wise `proj * gate * mask` is fused in a small Triton kernel.
5. Core reduction:
   out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   implemented as a batched GEMM in Triton.  Each program computes a tile
   (BLOCK_M × BLOCK_N) of the output for a specific (batch, head) pair,
   accumulating over K with fp32 precision and storing fp16 results.
6. Apply a second LayerNorm over the hidden dimension H, multiply by the
   output gate, and project back to the original dimension C.
7. Cast the result back to the input dtype (usually float32).

The implementation uses only one custom Triton kernel for the masked
projection/gate fusion and a second kernel for the O(N³·H) reduction.
All other ops are performed with torch.nn.functional for simplicity and
speed.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Kernel 1: fused projection * gate * mask (broadcast mask over H)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,      # *[total_elems]   fp16
    gate_ptr,      # *[total_elems]   fp16
    mask_ptr,      # *[mask_elems]    fp16 (B,N,N)
    out_ptr,       # *[total_elems]   fp16
    total_elems,   # i64
    hidden_dim,    # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    in_bounds = offs < total_elems

    # mask is broadcast over the hidden dimension
    mask_idx = offs // hidden_dim

    p = tl.load(proj_ptr + offs, mask=in_bounds, other=0.0)
    g = tl.load(gate_ptr + offs, mask=in_bounds, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=in_bounds, other=1.0)

    tl.store(out_ptr + offs, p * g * m, mask=in_bounds)


def _apply_gate_and_mask(proj, gate, mask, block_size=1024):
    """Element‑wise proj * gate * mask (mask broadcast over hidden dim)."""
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the core TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for left (B,N,N,H)
    stride_lb, stride_li, stride_lk, stride_lh,
    # strides for right (same layout)
    stride_rb, stride_ri, stride_rk, stride_rh,
    # strides for out   (same layout)
    stride_ob, stride_oi, stride_ok, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # 0) identify which (batch, head) this program works on
    # --------------------------------------------------------------
    pid = tl.program_id(0)          # combined batch*head dimension
    batch = pid // H
    head = pid % H

    pid_m = tl.program_id(1)        # row tile index
    pid_n = tl.program_id(2)        # col tile index

    # --------------------------------------------------------------
    # 1) compute tile coordinates
    # --------------------------------------------------------------
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # base pointers for this (batch, head)
    left_base  = left_ptr  + batch * stride_lb + head * stride_lh
    right_base = right_ptr + batch * stride_rb + head * stride_rh
    out_base   = out_ptr   + batch * stride_ob + head * stride_oh

    # accumulator in fp32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 2) loop over K dimension in blocks
    # --------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left: shape (BLOCK_M, BLOCK_K) -> left[batch, i, k, head]
        left_off = left_base + i[:, None] * stride_li + k[None, :] * stride_lk
        a = tl.load(left_off, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right: shape (BLOCK_N, BLOCK_K) -> right[batch, j, k, head]
        right_off = right_base + j[:, None] * stride_ri + k[None, :] * stride_rk
        b = tl.load(right_off, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # dot product over K (a @ bᵀ)
        acc += tl.dot(a, b, trans_b=True)

    # --------------------------------------------------------------
    # 3) write result back (fp16)
    # --------------------------------------------------------------
    out_off = out_base + i[:, None] * stride_oi + j[None, :] * stride_ok
    tl.store(out_off, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32) → cast to fp16
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2) Cast weights to fp16 once
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 3) Linear projections (B,N,N,hidden_dim) in fp16
    # --------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 4) Gate values (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # --------------------------------------------------------------
    # 5) Apply mask + gate (fused Triton kernel) if needed
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6) Core TriMul reduction via a custom Triton GEMM
    # --------------------------------------------------------------
    B, N, _, H = left.shape        # left shape (B, N, N, H)

    out = torch.empty_like(left)   # allocate (B, N, N, H) fp16

    # strides for the (B, N, N, H) layout (contiguous)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_ok, stride_oh = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                     # batch * head dimension
        math.ceil(N / BLOCK_M),    # number of row tiles
        math.ceil(N / BLOCK_N),    # number of col tiles
    )

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_ri,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_ok,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 7) Output LayerNorm over hidden dimension, gating, final proj
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)      # (B, N, N, C)

    # --------------------------------------------------------------
    # 8) Cast back to original dtype (usually float32) and return
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
