@triton.jit
def gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    out_gate_ptr,
    mask_ptr,
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)
    out_gate = tl.load(out_gate_ptr + offs, mask=mask)

    left_gate = tl.sigmoid(left_gate)
    right_gate = tl.sigmoid(right_gate)
    out_gate = tl.sigmoid(out_gate)

    if HAS_MASK:
        # Each hidden slice shares the same mask entry: mask index = offs // hidden_dim
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
    # store out_gate (already sigmoid) back
    tl.store(out_gate_ptr + offs, out_gate, mask=mask)

    norm_b = weights["norm.bias"].to(device).to(torch.float32)

    # Cast to fp16
    x = x.to(torch.float16)

    B, N, _, _ = x.shape

    # Combine projection weights
    # each weight shape (hidden_dim, dim)
    w_left = weights["left_proj.weight"]
    w_right = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]

    # Stack along dim=0
    w_concat = torch.cat([w_left, w_right, w_left_gate, w_right_gate, w_out_gate], dim=0).to(device).to(torch.float16) # shape (5*hidden_dim, dim)

    # Flatten input

    # Linear projection (single matmul)
    proj = torch.nn.functional.linear(x_flat, w_concat)  # (B*N*N, 5*hidden_dim), half

    # Split
    splits = torch.split(proj, hidden_dim, dim=1)
    left_raw = splits[0].reshape(B, N, N, hidden_dim).contiguous()
    right_raw = splits[1].reshape(B, N, N, hidden_dim).contiguous()
    left_gate_raw = splits[2].reshape(B, N, N, hidden_dim).contiguous()
    right_gate_raw = splits[3].reshape(B, N, N, hidden_dim).contiguous()
    out_gate_raw = splits[4].reshape(B, N, N, hidden_dim).contiguous()

    # Flatten for kernel
    total = left_raw.numel()
    BLOCK = 1024
    left_flat = left_raw.view(-1)
    right_flat = right_raw.view(-1)
    left_gate_flat = left_gate_raw.view(-1)
    right_gate_flat = right_gate_raw.view(-1)
    out_gate_flat = out_gate_raw.view(-1)

    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        mask_flat = mask_fp.view(-1)  # (B*N*N)

    )

    left = left_flat.view_as(left_raw)      # already gated
    right = right_flat.view_as(right_raw)
    out_gate = out_gate_flat.view_as(out_gate_raw)  # sigmoid applied

    # Contraction
    B, N, _, H = left.shape

    # (B, H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layer norm
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    # Apply out_gate
    out = out * out_gate

    # Final linear projection
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input (float32) and cast to fp16.
2. Fuse all five linear projections (left, right, left_gate, right_gate, out_gate)
   into a single matrix multiplication by concatenating the weight matrices.
3. In‑place Triton kernel:
   * apply sigmoid to the three gate tensors,
   * multiply left/right by their gates,
   * optionally multiply by the mask (broadcast over the hidden dimension).
   The kernel also stores the sigmoid‑ed out_gate for later use.
4. Perform the core O(N³·H) contraction as a batched GEMM:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as torch.bmm on tensors of shape (B·H, N, N).
5. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate and project back to the original channel dimension.
6. Cast the result back to the input dtype.

Only the gate‑mask fusion is done in Triton; the heavy contraction uses
cuBLAS (torch.bmm) which is already optimal on the target GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    out_gate_ptr,
    mask_ptr,
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of sigmoid gates, mask and element‑wise multiplication."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=active)
    right = tl.load(right_ptr + offs, mask=active)
    left_gate = tl.load(left_gate_ptr + offs, mask=active)
    right_gate = tl.load(right_gate_ptr + offs, mask=active)
    out_gate = tl.load(out_gate_ptr + offs, mask=active)

    # sigmoid(x) = 1 / (1 + exp(-x))
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate))
    out_gate = 1.0 / (1.0 + tl.exp(-out_gate))

    if HAS_MASK:
        # mask is shared across the hidden dimension
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)
    tl.store(out_gate_ptr + offs, out_gate, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)   (float32 or fp16)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy path
    # -----------------------------------------------------------------
    x = x.to(torch.float16)
    B, N, _, _ = x.shape

    # -----------------------------------------------------------------
    # 3. Fuse all five linear projections into one matmul
    # -----------------------------------------------------------------
    w_left = weights["left_proj.weight"]
    w_right = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]

    # Concatenate along the output dimension: (5*hidden_dim, dim)
    w_concat = torch.cat(
        [w_left, w_right, w_left_gate, w_right_gate, w_out_gate], dim=0
    ).to(device).to(torch.float16)

    # Flatten input to (B*N*N, dim) and apply the single linear layer
    x_flat = x.reshape(-1, dim)                     # (B*N*N, dim)
    proj = F.linear(x_flat, w_concat)               # (B*N*N, 5*hidden_dim)

    # Split results back to their respective tensors
    split_tensors = torch.split(proj, hidden_dim, dim=1)
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = [
        t.reshape(B, N, N, hidden_dim).contiguous() for t in split_tensors
    ]

    # -----------------------------------------------------------------
    # 4. Triton kernel: sigmoid gates, mask fusion, and in‑place multiplication
    # -----------------------------------------------------------------
    total = left_raw.numel()            # B * N * N * hidden_dim
    BLOCK = 1024

    left_flat = left_raw.view(-1)
    right_flat = right_raw.view(-1)
    left_gate_flat = left_gate_raw.view(-1)
    right_gate_flat = right_gate_raw.view(-1)
    out_gate_flat = out_gate_raw.view(-1)

    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        mask_flat = mask_fp.view(-1)          # (B*N*N,)
        has_mask = True
    else:
        mask_flat = torch.empty(1, device=device, dtype=torch.float16)
        has_mask = False

    grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
    _gate_mask_kernel[grid](
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        out_gate_flat,
        mask_flat,
        total,
        hidden_dim,
        HAS_MASK=has_mask,
        BLOCK_SIZE=BLOCK,
    )

    # Reshape after kernel (in‑place modification)
    left = left_flat.view_as(left_raw)
    right = right_flat.view_as(right_raw)
    out_gate = out_gate_flat.view_as(out_gate_raw)   # sigmoid already applied

    # -----------------------------------------------------------------
    # 5. Core O(N³·H) contraction via batched GEMM (L @ Rᵀ)
    # -----------------------------------------------------------------
    # Shape to (B*H, N, N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Restore (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate and final projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate                         # apply output gate (sigmoid)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)                # (B,N,N,dim) in fp16

    # -----------------------------------------------------------------
    # 7. Cast back to the input dtype and return
    # -----------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
