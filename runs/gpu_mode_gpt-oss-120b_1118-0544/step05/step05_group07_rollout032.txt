First, understand the operation:

Given x: [B, N, N, C] (C=dim). Apply layer norm over dim C. Then linear transformations to hidden_dim H:

left = x @ left_proj_weight (C->H) = shape [B, N, N, H]
right = x @ right_proj_weight (C->H) = shape [B, N, N, H]

Compute gates:

left_gate = sigmoid(x @ left_gate_weight) -> shape [B, N, N, H]
right_gate = sigmoid(x @ right_gate_weight) -> shape [B, N, N, H]
out_gate = sigmoid(x @ out_gate_weight) -> shape [B, N, N, H]

Mask: shape [B, N, N], broadcast to H.

Then:

left = left * left_gate * mask (if mask)
right = right * right_gate * mask (if mask)

Core = einsum('... i k d, ... j k d -> ... i j d', left, right). This is for outgoing version (i->j, k is intermediate). It loops over k dimension (the second N dimension in left and right). That is left is shape (B, N, N, H), with indices (b,i,k,d). Right is (b,j,k,d). Summation over k.

So core operation is for each b,d: we have left[b,:, :, d], right[b,:, :, d]. left's 'i,k' slices, right's 'j,k' slices (note same k dimension). For each fixed d, we need to compute dot product across k: left[b,i,: , d] dot right[b,j,: , d] => output[b,i,j,d]. So this is essentially a matrix multiplication: left_prof[b,:, :, d] * right_prof[b,:, :, d]^T (since right[b,j,k,d] is similar to row j, column k). Actually we can think of left matrix L (N x N) where rows = i, columns = k; right matrix R (N x N) where rows = j, columns = k. Then we need L * R^T, because for output[i,j], we sum over k: L[i,k] * R[j,k] => L(i,k) * R(j,k) = L[i,k]*R^T[k,j] = (L * R^T)[i,j].

Thus for each b,d we compute O_d = L_d @ R_d^T. That's standard batched matrix multiplication with shape (N,N) each.

Thus a simple approach: we reshape left and right to shape (B*H, N, N) and perform batch GEMM: out = torch.bmm(L, R.transpose(1,2)). This uses cuBLAS and is likely quite fast. import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_i, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # get program IDs
    pid_i = tl.program_id(0)  # block row index
    pid_j = tl.program_id(1)  # block col index
    pid_batch = tl.program_id(2)  # combined batch and head

    # derive batch and head
    b = pid_batch // H
    h = pid_batch % H

    # compute start offsets
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # pointers to the beginning of the batch/head slice
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, N, BLOCK_K):
        k0 = k

        # compute valid masks
        i_valid = i0 + tl.arange(0, BLOCK_M) < N
        j_valid = j0 + tl.arange(0, BLOCK_N) < N
        k_valid = k0 + tl.arange(0, BLOCK_K) < N

        # pointers for A tile (M x K)
        a_ptrs = left_base + (i0 + tl.arange(0, BLOCK_M)[:, None]) * stride_left_i \
                               + (k0 + tl.arange(0, BLOCK_K)[None, :]) * stride_left_k
        # load A
        a = tl.load(a_ptrs, mask=i_valid[:, None] & k_valid[None, :], other=0.0)
        a = a.to(tl.float32)

        # pointers for B tile (K x N) after transposition
        b_ptrs = right_base + (k0 + tl.arange(0, BLOCK_K)[:, None]) * stride_right_k \
                                 + (j0 + tl.arange(0, BLOCK_N)[None, :]) * stride_right_i
        b = tl.load(b_ptrs, mask=k_valid[:, None] & j_valid[None, :], other=0.0)
        b = b.to(tl.float32)

        # dot product and accumulate
        acc += tl.dot(a, b)

    # store result
    c_ptrs = out_base + (i0 + tl.arange(0, BLOCK_M)[:, None]) * stride_out_i \
                           + (j0 + tl.arange(0, BLOCK_N)[None, :]) * stride_out_j
    mask_c = i_valid[:, None] & j_valid[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)

"""
Implementation of outgoing TriMul forward pass using fused Triton kernel for the central
reduction (sum over the intermediate sequence dimension). The kernel performs a batched
matrix multiplication for each head and batch, computing:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
All other operations (layer‑norm, linear projections, gating, mask) are performed with
PyTorch in fp16. The result is cast back to the input dtype (float32).
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_i, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_i = tl.program_id(0)   # block row
    pid_j = tl.program_id(1)   # block column
    pid_batch = tl.program_id(2)  # combined batch and head index

    # decode batch and head
    b = pid_batch // H
    h = pid_batch % H

    # start offsets
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # base pointers for this (b, h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, N, BLOCK_K):
        k0 = k

        # validity masks
        i_valid = i0 + tl.arange(0, BLOCK_M) < N
        j_valid = j0 + tl.arange(0, BLOCK_N) < N
        k_valid = k0 + tl.arange(0, BLOCK_K) < N

        # pointers for A tile (M x K)
        a_ptrs = left_base + (i0 + tl.arange(0, BLOCK_M)[:, None]) * stride_left_i \
                               + (k0 + tl.arange(0, BLOCK_K)[None, :]) * stride_left_k
        a = tl.load(a_ptrs, mask=i_valid[:, None] & k_valid[None, :], other=0.0)
        a = a.to(tl.float32)

        # pointers for B tile (K x N) -> we load transposed version
        b_ptrs = right_base + (k0 + tl.arange(0, BLOCK_K)[:, None]) * stride_right_k \
                                 + (j0 + tl.arange(0, BLOCK_N)[None, :]) * stride_right_i
        b = tl.load(b_ptrs, mask=k_valid[:, None] & j_valid[None, :], other=0.0)
        b = b.to(tl.float32)

        # accumulate dot product
        acc += tl.dot(a, b)

    # store result
    c_ptrs = out_base + (i0 + tl.arange(0, BLOCK_M)[:, None]) * stride_out_i \
                           + (j0 + tl.arange(0, BLOCK_N)[None, :]) * stride_out_j
    mask_c = i_valid[:, None] & j_valid[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: tensors of shape (B, H, N, N) in fp16, contiguous.
    Returns out of same shape.
    """
    B, H, N, _ = left.shape
    out = torch.empty((B, H, N, N), dtype=left.dtype, device=left.device)

    # strides
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)
    _tri_mul_outgoing_kernel[grid](
        left, right, out,
        B, H, N,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.
    Arguments:
        data: (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] with same dtype as input.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # cast to fp16 for the heavy part
    x = x.to(torch.float16)

    # 2. Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    left_proj = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, H]
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Gates (fp16)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: [B, N, N] -> [B, N, N, 1]
        mask_fp = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_fp
        right_proj = right_proj * mask_fp

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate   # [B, N, N, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core reduction via Triton (batched GEMM for each head)
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) and make contiguous for the kernel
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    # kernel produces (B, H, N, N) in fp16
    out = _batched_tri_mul(left, right)

    # ------------------------------------------------------------------
    # 7. Back to (B, N, N, H) layout
    # ------------------------------------------------------------------
    out = out.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Second LayerNorm on hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # ------------------------------------------------------------------
    # 9. Output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # 10. Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)

    # ------------------------------------------------------------------
    if out.dtype != dtype_in:

"""
Outgoing TriMul forward pass (AlphaFold3).

The heavy‑weight reduction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
is implemented as a fused Triton kernel that performs a batched matrix
multiplication for every (batch, head) pair. All other stages (layer‑norm,
linear projections, gating, optional mask) are done in PyTorch using fp16.
The final result is cast back to the input dtype (typically float32).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_i, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM for the TriMul core:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    left/right shape  : (B, H, N, N)  -- row‑major, dtype fp16
    out shape         : (B, H, N, N)  -- same layout
    """
    pid_i = tl.program_id(0)          # block row   (i dimension)
    pid_j = tl.program_id(1)          # block col   (j dimension)
    pid_batch = tl.program_id(2)      # combined batch & head index

    # decode batch and head
    b = pid_batch // H
    h = pid_batch % H

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # base pointers for this (b, h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator in float32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k
    for k in range(0, N, BLOCK_K):
        k0 = k

        # masks for boundary handling
        i_valid = i0 + tl.arange(0, BLOCK_M) < N
        j_valid = j0 + tl.arange(0, BLOCK_N) < N
        k_valid = k0 + tl.arange(0, BLOCK_K) < N

        # ---- load A tile (M x K) ----
        a_ptrs = left_base + (i0 + tl.arange(0, BLOCK_M)[:, None]) * stride_left_i \
                               + (k0 + tl.arange(0, BLOCK_K)[None, :]) * stride_left_k
        a = tl.load(a_ptrs,
                    mask=i_valid[:, None] & k_valid[None, :],
                    other=0.0)
        a = a.to(tl.float32)

        # ---- load B tile transposed (K x N) ----
        # we load right[j, k] as (k, j) to match GEMM semantics
        b_ptrs = right_base + (k0 + tl.arange(0, BLOCK_K)[:, None]) * stride_right_k \
                                 + (j0 + tl.arange(0, BLOCK_N)[None, :]) * stride_right_i
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & j_valid[None, :],
                    other=0.0)
        b = b.to(tl.float32)

        # ---- accumulate dot product ----
        acc += tl.dot(a, b)      # (BLOCK_M, BLOCK_N)

    # store the computed tile
    c_ptrs = out_base + (i0 + tl.arange(0, BLOCK_M)[:, None]) * stride_out_i \
                           + (j0 + tl.arange(0, BLOCK_N)[None, :]) * stride_out_j
    mask_c = i_valid[:, None] & j_valid[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Launches the Triton kernel for the core TriMul reduction.
    left / right: tensors of shape (B, H, N, N), dtype fp16, contiguous.
    Returns a tensor of the same shape and dtype.
    """
    B, H, N, _ = left.shape
    out = torch.empty((B, H, N, N), dtype=left.dtype, device=left.device)

    # strides for each dimension
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)
    _tri_mul_outgoing_kernel[grid](
        left, right, out,
        B, H, N,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, C]   (float32 or float16)
            - mask: [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, C] with the same dtype as the input.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # -----------------------------------------------------------------
    # 1. Input layer‑norm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)   # fp16 for the compute‑heavy part

    # -----------------------------------------------------------------
    # 2. Linear projections (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    left_proj = torch.nn.functional.linear(x, left_proj_w)    # [B, N, N, H]
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 3. Gates (fp16)
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # -----------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)      # [B, N, N, 1]
        left_proj = left_proj * mask_fp
        right_proj = right_proj * mask_fp

    # -----------------------------------------------------------------
    # 5. Apply gates
    # -----------------------------------------------------------------
    left = left_proj * left_gate      # [B, N, N, H]
    right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 6. Core reduction via Triton (batched GEMM per head)
    # -----------------------------------------------------------------
    # bring head dimension forward for the kernel
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()
    out = _batched_tri_mul(left, right)            # [B, H, N, N]

    # -----------------------------------------------------------------
    # 7. Restore original layout
    # -----------------------------------------------------------------
    out = out.permute(0, 2, 3, 1).contiguous()     # [B, N, N, H]

    # -----------------------------------------------------------------
    # 8. Second layer‑norm on hidden dimension
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9. Output gate
    # -----------------------------------------------------------------
    out = out * out_gate

    # -----------------------------------------------------------------
    # 10. Final linear projection back to original dim
    # -----------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, C]

    # -----------------------------------------------------------------
    # 11. Cast back to input dtype
    # -----------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
