    B, M, N, K,

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

def triton_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,

):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for valid m,n
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to start of block for a and b
    a_ptrs = a_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + 0 * stride_ak

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device

    # layernorm 1
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # cast to fp16
    x = x.to(torch.float16)

    # cast weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # linear projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x, right_proj_w)
    # gates
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # mask and gate fusion using Triton kernel
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    B, N, _, _ = left.shape  # B,N,N,hidden

    # allocate out tensor
    out = torch.empty_like(left)  # same shape and dtype fp16

    # For each hidden dimension, compute matmul
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    for h in range(hidden_dim):
        # slice data for hidden dimension
        left_h = left[..., h]   # shape [B,N,N]
        right_h = right[..., h] # shape [B,N,N]
        out_h = out[..., h]     # shape [B,N,N]

        # Launch kernel
        grid = (B,
                triton.cdiv(N, BLOCK_M),
                triton.cdiv(N, BLOCK_N))
        triton_matmul_kernel[grid](
            left_h, right_h, out_h,
            B, N, N, N,
            left_h.stride(1), left_h.stride(2), left_h.stride(0),
            right_h.stride(2), right_h.stride(1), right_h.stride(0),
            out_h.stride(1), out_h.stride(2), out_h.stride(0),
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        )
    # now out shape [B,N,N,hidden_dim] (fp16)

    # second layernorm over hidden dim
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # multiply by out_gate
    out = out * out_gate

    # final linear projection
    out = torch.nn.functional.linear(out, to_out_w)

    # cast to original dtype
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

"""
TriMul (outgoing) – Triton‑fused mask/gate and batched matmul.

Algorithm
---------
1. Layer‑normalize the input (float32) → fp16.
2. Linear projections (left/right) and gating projections (sigmoid) in fp16.
3. Fuse mask, projection and gate with a tiny Triton kernel
   (`_gate_mask_kernel`).  This saves a memory pass.
4. Core O(N³·H) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   Implemented as a custom batched GEMM kernel (`tri_mul_kernel`) that
   processes all hidden dimensions in a single launch (batch × hidden as
   the first program dimension).  The kernel uses tiling (64×64×64)
   and accumulates in fp32, then stores fp16.
5. Second LayerNorm over the hidden dimension, multiply by the output gate.
6. Final linear projection back to the original channel dimension.
7. Cast back to the input dtype.

The implementation keeps all heavy compute in fp16/Triton while using
torch’s highly‑optimized layer‑norms and linear layers for the remaining
operations.
"""

import math
from typing import Tuple, Dict, Any

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) on a flattened view
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask is broadcasted over the hidden dimension
    mask_index = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_index, mask=valid, other=1.0)

    tl.store(out_ptr + offs, proj * gate * mask, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """Fuse `proj * gate * mask` using Triton."""
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],                 # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the core TriMul reduction
# Computes out = left @ right^T  (all tensors are fp16)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, hidden_dim, N,
    # strides for left tensor
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right tensor
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides for out tensor
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_comb = tl.program_id(0)          # combined batch*hidden index
    pid_m = tl.program_id(1)             # block row i
    pid_n = tl.program_id(2)             # block col j

    # decode combined index into (batch, hidden)
    b = pid_comb // hidden_dim
    h = pid_comb % hidden_dim

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this (b, h)
    a_base = left_ptr + b * stride_left_b + h * stride_left_h
    b_base = right_ptr + b * stride_right_b + h * stride_right_h
    c_base = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K dimension (= N)
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: (M, K)
        a = tl.load(
            a_base
            + offs_m[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right block (transposed): (K, N)
        b = tl.load(
            b_base
            + offs_k[:, None] * stride_right_k
            + offs_n[None, :] * stride_right_j,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)  # fp16 for the heavy part

    # ------------------------------------------------------------------
    # 2. Cast weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)    # [B, N, N, H]
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5. Fuse mask & gate (Triton) or simple element‑wise mul
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    B, N, _, _ = left.shape  # shape = [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Core O(N³·H) reduction via custom batched GEMM
    # ------------------------------------------------------------------
    out = torch.empty_like(left)   # [B, N, N, H], fp16

    # Tiling parameters (tuned for A100/B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides for the original 4‑d tensors
    # left: (B, N, N, H)
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    # right: (B, N, N, H)
    stride_right_b = right.stride(0)
    stride_right_j = right.stride(1)   # second N   (j)
    stride_right_k = right.stride(2)   # third  N   (k)
    stride_right_h = right.stride(3)

    # out: (B, N, N, H)
    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    grid = (
        B * hidden_dim,                     # combined batch*hidden
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        hidden_dim,
        N,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # 8. Apply output gate
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 10. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
