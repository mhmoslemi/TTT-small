def _gate_mask_inplace_kernel(proj_ptr, gate_ptr, mask_ptr, total_elems, hidden_dim, BLOCK_SIZE):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # broadcast mask across hidden_dim: element offset // hidden_dim
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(proj_ptr + offs, out, mask=mask)

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_inplace_kernel(
    proj_ptr,  # *[total_elems] fp16
    gate_ptr,  # *[total_elems] fp16
    mask_ptr,  # *[mask_elems] fp16
    total_elems,
    hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # compute mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(proj_ptr + offs, out, mask=mask)

def _apply_gate_and_mask_inplace(proj: torch.Tensor, gate: torch.Tensor,
                                 mask: torch.Tensor,
                                 block_size: int = 1024):
    """
    In-place elementwise `proj = proj * gate * mask`, where mask is broadcast
    over the hidden dimension. All inputs must be fp16 and contiguous.
    """
    total = proj.numel()

    mask_flat = mask.contiguous().view(-1)  # shape (B*N*N,)
    _gate_mask_inplace_kernel[grid](
        proj,
        gate,
        mask_flat,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    # No return; proj modified in-place.

    Forward pass of the outgoing Triangle Multiplicative Update (TriMul)
    operator (AlphaFold3 style). All heavy arithmetic is performed in FP16.
    A Triton kernel fuses the element‑wise gate·mask multiplication.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],

    # 3. Cast model weights to FP16 (no bias terms)

    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # 4. Linear projections (no bias)
    left_proj = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # 5. Gate projections (sigmoid)
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # 6. Fuse mask and gate (in‑place) if needed
    if mask is not None and not nomask:

        _apply_gate_and_mask_inplace(right_proj, right_gate, mask_fp16)

        right = right_proj

"""
Highly‑optimized forward pass for the outgoing TriMul operator (AlphaFold3).

Algorithm
---------
1. Layer‑normalize the 4‑D input.
2. Cast everything to FP16 for the compute‑heavy part.
3. Linear projections and gate projections (no bias).
4. Fuse element‑wise “proj * gate * mask” with a Triton kernel
   (in‑place on the projection tensor).  If there is no mask we simply
   multiply by the gate in‑place.
5. Core O(N³·H) reduction:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   realized with a single torch.einsum, avoiding an explicit
   reshape+contiguous copy.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate and project back to the original channel dimension.
7. Cast the result back to the input dtype.

Only the gate‑mask fusion is implemented in Triton; the rest leverages
high‑performance PyTorch kernels (layer‑norm, linear and einsum).
"""

import torch.nn.functional as F
import triton
import triton.language as tl
