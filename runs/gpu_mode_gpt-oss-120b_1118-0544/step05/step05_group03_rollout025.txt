import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,  # *fp16
    gate_ptr,  # *fp16
    mask_ptr,  # *fp16
    out_ptr,   # *fp16
    total_elems,  # i64
    hidden_dim,   # i32
    BLOCK_SIZE: tl.constexpr,
):

    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Index into mask after broadcasting over hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor, block_size: int = 2048) -> torch.Tensor:
    """
    Fuse projection, gate and (optional) mask using a Triton kernel.
    proj, gate : (B, N, N, H)  fp16
    mask       : (B, N, N)      fp16   (0 or 1)
    Returns: (B, N, N, H) fp16

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

    single batched `torch.matmul` (FP16) for the N³ reduction.
    The final result is cast back to the input dtype (typically float32).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor of shape [B, N, N, C]
            mask: torch.Tensor of shape [B, N, N] (or None)
            weights: dict of model parameters (float32)
            config: dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm (FP32 for numerical stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for heavy compute
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Prepare FP16 weight tensors (concatenate where possible)
    # --------------------------------------------------------------
    # Linear projections (left + right)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    proj_concat_w = torch.cat([left_proj_w, right_proj_w], dim=0)  # (2*hidden_dim, dim)

    # Gate projections (left_gate + right_gate)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    gate_concat_w = torch.cat([left_gate_w, right_gate_w], dim=0)  # (2*hidden_dim, dim)

    # Out gate
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # Output layer‑norm and final linear
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Combined linear projections
    # --------------------------------------------------------------
    proj_concat = F.linear(x, proj_concat_w)  # (B, N, N, 2*hidden_dim)
    left_proj = proj_concat[..., :hidden_dim]
    right_proj = proj_concat[..., hidden_dim:]

    # --------------------------------------------------------------
    # 5. Combined gate projections + sigmoid
    # --------------------------------------------------------------
    gate_concat = F.linear(x, gate_concat_w)
    left_gate = gate_concat[..., :hidden_dim].sigmoid()
    right_gate = gate_concat[..., hidden_dim:].sigmoid()

    # --------------------------------------------------------------
    # 6. Out gate
    # --------------------------------------------------------------
    out_gate = F.linear(x, out_gate_w).sigmoid()  # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 7. Fuse mask, gating and projection (Triton kernel)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:

        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 8. Core N³ reduction via batched matmul
    #    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    # --------------------------------------------------------------
    # Shape transformations: (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2)
    right_perm_T = right.permute(0, 3, 2, 1)  # transpose K dimension
    out_perm = torch.matmul(left_perm, right_perm_T)  # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 9. Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(

        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 10. Cast back to input dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

"""
TriMul (outgoing) forward pass – Triton‑accelerated mask & gate fusion.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (float32).
2. Cast to FP16 for compute‑heavy parts.
3. Concatenate left/right linear projections and left/right gates to halve the
   number of GEMM calls.
4. Fuse projection, gate and optional mask with a small Triton kernel
   (proj * gate * mask).
5. Perform the N³ reduction with a single batched `torch.matmul`:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   (implemented as a batched matrix multiply after permuting axes).
6. Apply output LayerNorm, multiply by the out‑gate and a final linear.
7. Cast the result back to the original dtype.

Only the mask‑gate fusion is implemented in Triton; the remaining heavy
operations use highly‑optimized PyTorch kernels (FP16 GEMM / Tensor‑cores).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,   # fp16 *
    gate_ptr,   # fp16 *
    mask_ptr,   # fp16 *
    out_ptr,    # fp16 *
    total_elems,  # i64
    hidden_dim,   # i32
    BLOCK_SIZE: tl.constexpr,
):
    """
    Compute ``proj * gate * mask`` where ``mask`` is broadcast over the last
    hidden dimension. All tensors are flattened 1‑D pointers.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # Because ``mask`` lacks the hidden dimension, each hidden element maps to
    # the same mask entry.
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)
