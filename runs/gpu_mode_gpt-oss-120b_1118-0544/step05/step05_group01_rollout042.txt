@triton.jit
def triton_matmul_transpose_kernel(
    left_ptr, right_ptr, out_ptr,
    B, M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_n, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Compute batch id
    pid = tl.program_id(0)
    total_batch = B
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    # each program processes one (batch, block_m, block_n)
    batch = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    block_m = block_idx // num_blocks_n
    block_n = block_idx % num_blocks_n

    # Compute offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Load a tile from left: (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + batch * stride_l_batch + offs_m[:, None] * stride_l_m + k_offset[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (k_offset[None, :] < K), other=0.0)
        # Load a tile from right: we need (BLOCK_N, BLOCK_K) because we compute right^T
        # right layout is (batch, N, K). We need element right[batch, j, k]
        b_ptrs = right_ptr + batch * stride_r_batch + offs_n[:, None] * stride_r_n + k_offset[None, :] * stride_r_k
        b = tl.load(b_ptrs, mask=(offs_n[:, None] < N) & (k_offset[None, :] < K), other=0.0)
        # compute dot product
        acc += tl.dot(a, b)
        # increment pointers not needed as we compute via offsets each iteration
    # Write back
    out_ptrs = out_ptr + batch * stride_o_batch + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n
    tl.store(out_ptrs, acc.to(tl.float16), mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _triton_matmul_transpose_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_n, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch_idx = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    block_m = block_idx // num_blocks_n
    block_n = block_idx % num_blocks_n

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile (M x K)
        a_ptrs = left_ptr + batch_idx * stride_l_batch + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right tile (K x N) – note transposed layout
        b_ptrs = right_ptr + batch_idx * stride_r_batch + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    out_ptrs = out_ptr + batch_idx * stride_o_batch + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated outgoing TriMul forward.

    Steps
    -----
    1. Layer‑norm the input (dim).
    2. Cast to fp16 for the heavy part of the computation.
    3. Linear projections and gates (no bias).  Gating uses sigmoid.
    4. Optional mask is fused by simple element‑wise multiplication.
    5. Core reduction:
         out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
       Implemented as a batched GEMM (left @ rightᵀ) in a custom Triton kernel.
    6. Layer‑norm over the hidden dimension, multiply by the output gate.
    7. Final linear projection back to ``dim``.
    8. Cast back to the original dtype.
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # 1. Input layer norm (float32)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim) float32

    # 2. Cast to fp16 for compute‑heavy operations
    x = x.to(torch.float16)

    # 3. Linear projections (no bias)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N, H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # 4. Apply mask (if present) and gating
    if mask is not None and not nomask:
        # mask shape: (B,N,N) -> broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 5. Core reduction via Triton: out = left @ rightᵀ for each (b,h)
    B, N, _, H = left.shape

    # reshape to (B*H, N, N) where the last dim is the reduction axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)

    # strides for the contiguous tensors
    stride_l_batch = left_flat.stride(0)
    stride_l_m = left_flat.stride(1)
    stride_l_k = left_flat.stride(2)

    stride_r_batch = right_flat.stride(0)
    stride_r_n = right_flat.stride(1)   # N dimension
    stride_r_k = right_flat.stride(2)

    stride_o_batch = out_flat.stride(0)
    stride_o_m = out_flat.stride(1)
    stride_o_n = out_flat.stride(2)

    # block sizes (tuned for fp16 on Hopper)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = (B * H) * num_blocks_m * num_blocks_n

    grid = (total_tiles,)

    _triton_matmul_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        N,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_n,
        stride_r_k,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Output layer norm over hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 7. Apply output gate
    out = out * out_gate

    # 8. Final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N, dim) fp16

    # 9. Cast back to original dtype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

"""
custom_kernel: Triton‑accelerated outgoing TriMul forward.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (`dim`).
2. Linear projections (no bias) for left/right and three gates;
   apply sigmoid to the gates.
3. Optional binary mask is fused by simple element‑wise multiplication.
4. Core reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication (left @ rightᵀ) for each
   hidden slice.  This reduction is performed by a custom Triton kernel.
5. Layer‑norm over the hidden dimension, multiply by the output gate,
   and a final linear projection back to `dim`.
6. Cast the result back to the original dtype (usually float32).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _triton_matmul_transpose_kernel(
    left_ptr,                     # fp32 * (batch, M, K)
    right_ptr,                    # fp32 * (batch, N, K)
    out_ptr,                      # fp32 * (batch, M, N)
    batch, M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_n, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out = left @ rightᵀ for each batch entry."""
    pid = tl.program_id(0)

    # number of tiles per dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # decode program id into (batch, block_i, block_j)
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    block_i = block_idx // num_blocks_n
    block_j = block_idx % num_blocks_n

    # tile offsets
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension
    for kb in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile (M × K)
        a_ptrs = (left_ptr
                  + batch_idx * stride_l_batch
                  + offs_i[:, None] * stride_l_m
                  + offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right tile transposed (K × N)  – we read as (k, n)
        b_ptrs = (right_ptr
                  + batch_idx * stride_r_batch
                  + offs_k[:, None] * stride_r_k
                  + offs_j[None, :] * stride_r_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # write back
    out_ptrs = (out_ptr
                + batch_idx * stride_o_batch
                + offs_i[:, None] * stride_o_m
                + offs_j[None, :] * stride_o_n)
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask         : torch.Tensor [B, N, N] or None (binary 0/1)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim` (float32)
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim) float32

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) – keep FP32 for accuracy
    # --------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # --------------------------------------------------------------
    # 3. Apply optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(dtype=x.dtype).unsqueeze(-1)            # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp
        right = right_proj * right_gate * mask_fp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 4. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (left @ rightᵀ) with Triton.
    # --------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B*H, N, N) where the last dim is the reduction axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)  # fp32

    # strides for contiguous tensors (batch, M, K) / (batch, N, K)
    stride_l_batch = left_flat.stride(0)
    stride_l_m = left_flat.stride(1)
    stride_l_k = left_flat.stride(2)

    stride_r_batch = right_flat.stride(0)
    stride_r_n = right_flat.stride(1)   # N dimension
    stride_r_k = right_flat.stride(2)

    stride_o_batch = out_flat.stride(0)
    stride_o_m = out_flat.stride(1)
    stride_o_n = out_flat.stride(2)

    # Tuned block sizes for Hopper (fp32)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = (B * H) * num_blocks_m * num_blocks_n
    grid = (total_tiles,)

    _triton_matmul_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        N,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_n,
        stride_r_k,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5. Output LayerNorm over the hidden dimension,
    #    apply output gate and final linear projection.
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # --------------------------------------------------------------
    # 6. Cast back to original dtype (usually float32)
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
