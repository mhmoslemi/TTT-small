@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_head, N,
    stride_l_bh, stride_l_i, stride_l_k,
    stride_r_bh, stride_r_j, stride_r_k,
    stride_o_bh, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # program ids
    bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    # compute offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for i,j
    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left tile
        # left ptr: base + bh*stride_l_bh + i*stride_l_i + k*stride_l_k
        left = tl.load(left_ptr + bh * stride_l_bh + i[:, None] * stride_l_i + k_offset[None, :] * stride_l_k,
                       mask = mask_i[:, None] & mask_k[None, :],
                       other = 0.0)

        # load right tile transposed: shape (BLOCK_K, BLOCK_N)
        # right ptr: base + bh*stride_r_bh + k*stride_r_k + j*stride_r_j
        right = tl.load(right_ptr + bh * stride_r_bh + k_offset[:, None] * stride_r_k + j[None, :] * stride_r_j,
                        mask = mask_k[:, None] & mask_j[None, :],
                        other = 0.0)

        acc += tl.dot(left, right)  # (M,K)*(K,N) -> (M,N)
    # store
    out = acc.to(tl.float16)
    out_ptrs = out_ptr + bh * stride_o_bh + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_ptrs, out, mask = mask_i[:, None] & mask_j[None, :])
```

    batch_head, N,

    stride_r_bh, stride_r_j, stride_r_k,

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to float16 for compute‑heavy parts.
3. Linear projections (no bias) → left_proj, right_proj.
4. Compute three gates (sigmoid of linear projections).
5. Fuse mask, projection and gate (when a mask is present) with a tiny
   Triton kernel: proj * gate * mask  (mask broadcast over the hidden dim).
6. Core reduction:
     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM in Triton (one GEMM per head).
7. Layer‑norm over the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original channel dimension.
8. Cast the result back to the input dtype (usually float32).

The kernel uses float16 arithmetic with float32 accumulation and exploits
Tensor‑Core instructions (via `tl.dot`).  All heavy work is performed on
the GPU; only light‑weight element‑wise operations use PyTorch.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Kernel: proj * gate * mask  (mask broadcast over hidden_dim)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B,N,N,H)
    gate_ptr,          # fp16 * (B,N,N,H)
    mask_ptr,          # fp16 * (B,N,N)
    out_ptr,           # fp16 * (B,N,N,H)
    total_elems,       # B*N*N*H
    hidden_dim,        # H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # each hidden_dim consecutive entries share the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(out_ptr + offs, proj * gate * m, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse projection, gate and mask using Triton.
    Shapes:
        proj, gate : (B, N, N, H)
        mask       : (B, N, N)   (binary 0/1)
    Returns:
        proj * gate * mask  (broadcast over H)   dtype = proj.dtype
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel: batched GEMM for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,   # fp16 * (BH, N, N)
    right_ptr,  # fp16 * (BH, N, N)
    out_ptr,    # fp16 * (BH, N, N)
    BH: tl.int32,   # B * hidden_dim
    N: tl.int32,
    # strides for left
    stride_l_bh, stride_l_i, stride_l_k,
    # strides for right (note: we load the transpose)
    stride_r_bh, stride_r_j, stride_r_k,
    # strides for output
    stride_o_bh, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    bh = tl.program_id(0)          # batch * head index
    pid_i = tl.program_id(1)       # tile row
    pid_j = tl.program_id(2)       # tile column

    # Starting coordinates of the tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left tile: shape (M, K)
        left = tl.load(
            left_ptr
            + bh * stride_l_bh
            + i[:, None] * stride_l_i
            + k_offset[None, :] * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load right tile transposed: shape (K, N)
        # right is stored as (row=j, col=k); we need (k, j)
        right = tl.load(
            right_ptr
            + bh * stride_r_bh
            + k_offset[:, None] * stride_r_k
            + j[None, :] * stride_r_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        acc += tl.dot(left, right)

    # Write back the result (cast to fp16)
    out = acc.to(tl.float16)
    out_ptrs = out_ptr + bh * stride_o_bh + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C]
            - mask         : torch.Tensor [B, N, N]  or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (no bias) – all weights cast to FP16
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    left_proj = torch.nn.functional.linear(x, left_proj_w)          # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)        # (B,N,N,H)

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # --------------------------------------------------------------
    # 4. Apply mask and gating (fused via Triton if mask is present)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5. Core reduction (batched GEMM) using Triton
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # left shape: (B, N, N, H)

    # Rearrange to (B, H, N, N) and flatten the first two dims
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left_perm.view(B * H, N, N)   # (BH, N, N)
    right_flat = right_perm.view(B * H, N, N) # (BH, N, N)

    out_flat = torch.empty_like(left_flat)

    # Strides (in elements) for the flattened tensors
    stride_l_bh = left_flat.stride(0)  # N*N
    stride_l_i  = left_flat.stride(1)  # N
    stride_l_k  = left_flat.stride(2)  # 1

    stride_r_bh = right_flat.stride(0)  # N*N
    stride_r_j  = right_flat.stride(1)  # N   (row index j)
    stride_r_k  = right_flat.stride(2)  # 1   (col index k)

    stride_o_bh = out_flat.stride(0)    # N*N
    stride_o_i  = out_flat.stride(1)    # N
    stride_o_j  = out_flat.stride(2)    # 1

    # Block sizes – chosen to fit typical GPU register budget
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        stride_l_bh,
        stride_l_i,
        stride_l_k,
        stride_r_bh,
        stride_r_j,
        stride_r_k,
        stride_o_bh,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, C)

    # --------------------------------------------------------------
    # 7. Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
