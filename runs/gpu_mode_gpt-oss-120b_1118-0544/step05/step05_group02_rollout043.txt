Pseudo:

@triton.jit
def matmul_tile_kernel(
    left_ptr,   # *fp16
    right_ptr,  # *fp16
    out_ptr,    # *fp16
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch_h = num_tiles_m * num_tiles_n
    batch_h = pid // tiles_per_batch_h
    tile_idx = pid % tiles_per_batch_h
    pid_m = tile_idx // num_tiles_n
    pid_n = tile_idx % num_tiles_n

    b = batch_h // H
    h = batch_h % H

    # Compute base offsets for the given batch and hidden channel.
    # For left: start at offset for (b,i=pid_m*BLOCK_M, k=0, h)
    # Actually we need offset for row start i and column start k.
    # row offset: i_offset = pid_m*BLOCK_M
    # col offset: k_offset varies in loop.

    # Offsets for left tile for the block rows
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Compute base pointers for left and right
    # left base: b * stride_left_b + h * stride_left_h
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    # Right base: same (b, h)
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    # out base
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_iter in range(0, N, BLOCK_K):
        # Determine actual K block size: need to bound by N
        cur_K = tl.minimum(BLOCK_K, N - k_iter)
        # for varying K we need to adjust offsets
        # We'll load with appropriate mask.

        # Offsets for k
        k = k_iter + offs_k
        mask_k = k < N

        # Compute pointers
        left_ptrs = left_base + offs_m[:, None] * stride_left_i + k[None, :] * stride_left_k
        right_ptrs = right_base + offs_n[:, None] * stride_right_i + k[None, :] * stride_right_k
        # left shape (BLOCK_M, cur_K)
        # right shape (BLOCK_N, cur_K) but we need (cur_K, BLOCK_N) for dot.
        # So we will load right as (BLOCK_N, cur_K) then transpose using tl.view? Actually dot expects A of shape (M,K) and B of shape (K,N). We can load right as (cur_K, BLOCK_N) via transposed pointer.

        # To load right transposed, we can compute pointers with row index as k and column as j.
        # but we loaded using (BLOCK_N, cur_K) where first index is j (offs_n) and second is k. That's shape (N, K). We need shape (K,N). We can load and then do tl.load... but tl.dot expects B shape (K,N). We can pass transposed view: we can use tl.load to get B of shape (cur_K, BLOCK_N) by swapping offsets in pointer, e.g., right_ptrs_T = right_base + k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i.

        # Let's use that.
        right_ptrs_T = right_base + k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i
        # Now load
        a = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(right_ptrs_T, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        # Multiply-accumulate
        acc += tl.dot(a, b)

    # Store result
    out_ptrs = out_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

        rp_str[0], rp_str[1], rp_str[2], rp_str[3],
        # right_gate strides
        rg_str[0], rg_str[1], rg_str[2], rg_str[3],
        # mask strides
        mask_str[0] if has_mask else 0,
        mask_str[1] if has_mask else 0,
        mask_str[2] if has_mask else 0,
        # output strides
        out_str[0], out_str[1], out_str[2], out_str[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HAS_MASK=int(has_mask),
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gate and final projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 7. Cast back to the original dtype and return
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)
    return out
```
