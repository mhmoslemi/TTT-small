@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_aba, stride_ama, stride_ak,
    stride_bba, stride_bna, stride_bk,
    stride_cca, stride_cma, stride_cna,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # bound checks
    m_mask = offs_m < M
    n_mask = offs_n < N

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # a and b loading
        a_ptrs = a_ptr + pid_b * stride_aba + offs_m[:, None] * stride_ama + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_b * stride_bba + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bna

        a = tl.load(a_ptrs, mask=m_mask[:, None] & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & n_mask[None, :], other=0.0)

    # store result

"""
TriMul (outgoing) forward pass – Triton‑accelerated core reduction.

Algorithm
---------
1. Layer‑norm the 4‑D input (B, N, N, C) in FP32.
2. Cast to FP16 for the compute‑intensive part.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) are performed with torch.nn.functional.linear
   (cuBLAS) in FP16.
4. Apply mask (if any) and element‑wise gating.
5. Core O(N³·H) reduction:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM kernel in Triton.  The tensors are
   reshaped to (B·H, N, N) and the kernel computes
      C = A @ Bᵀ   (A,B shape (batch, N, K) with K = N)
   using tiled fp16 loads, fp32 accumulation and tensor‑core friendly
   block sizes.
6. Apply a second LayerNorm, multiply by the output gate and linearly
   project back to the original channel dimension.
7. Cast the result back to the original dtype (float32).

The implementation keeps all heavy matrix‑multiplication work on the GPU
via Triton while the surrounding linear / norm / sigmoid operations use
high‑performance PyTorch kernels.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes, for each batch index b:
        C[b, i, j] = Σₖ A[b, i, k] * B[b, j, k]
    where A and B have shape (batch, M, K) and (batch, N, K) respectively.
    The result C has shape (batch, M, N).
    """
    pid_b = tl.program_id(0)            # batch dimension
    pid_m = tl.program_id(1)            # block row of C
    pid_n = tl.program_id(2)            # block column of C

    # Offsets inside the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    m_mask = offs_m < M
    n_mask = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=m_mask[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile: (BLOCK_K, BLOCK_N)  (note B is stored as (N, K))
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & n_mask[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # Write back result in FP16
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=m_mask[:, None] & n_mask[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model weights (see keys in implementation)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑intensive stages
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all model weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6. Apply mask (if present) and fuse with gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction – batched GEMM in Triton
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                # (B, N, N, H)

    # Rearrange to (B, H, N, N) and flatten batch*H
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left.view(B * H, N, N)             # (B*H, N, N)
    right_flat = right.view(B * H, N, N)

    # Allocate output buffer
    out_flat = torch.empty_like(left_flat)

    # Strides for the three tensors (all contiguous)
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)   # K dimension (inner)
    stride_b_n = right_flat.stride(1)   # N dimension (outer)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Tile sizes – chosen for good occupancy on B200/H200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                                 # batch dimension
        (N + BLOCK_M - 1) // BLOCK_M,          # rows
        (N + BLOCK_N - 1) // BLOCK_N,          # cols
    )

    batched_gemm_kernel[grid](
        a_ptr=left_flat.data_ptr(),
        b_ptr=right_flat.data_ptr(),
        c_ptr=out_flat.data_ptr(),
        batch=B * H,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
