@triton.jit
def tri_mul_outgoing_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_j,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)  # batch * hidden
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for boundaries
    mask_m = offs_m < N
    mask_n = offs_n < N

    # compute base pointers for this batch
    left_batch_ptr = left_ptr + pid_b * stride_left_batch
    right_batch_ptr = right_ptr + pid_b * stride_right_batch

    # pointers for each k block will be updated in loop
    a_ptrs = left_batch_ptr + offs_m[:, None] * stride_left_i + tl.arange(0, BLOCK_K)[None, :] * stride_left_k
    b_ptrs = right_batch_ptr + tl.arange(0, BLOCK_K)[:, None] * stride_right_k + offs_n[None, :] * stride_right_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = cur_k < N

        a = tl.load(a_ptrs, mask=mask_m[:, None] & k_mask[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=k_mask[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)

        # advance pointers to next k block
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # write output
    out_ptrs = out_ptr + pid_b * stride_out_batch + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel for mask and gate fusion
@triton.jit
def _gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # broadcast mask across hidden dimension
    mask_idx = offs // hidden_dim
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """Fuse proj * gate * mask using Triton."""
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

# Triton kernel for batched GEMM (left @ right^T) for each hidden channel
@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_j,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)   # batch * hidden index
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block col

    # compute absolute coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch
    left_batch = left_ptr + pid_b * stride_left_batch
    right_batch = right_ptr + pid_b * stride_right_batch

    # pointers for the current k‑tile
    a_ptrs = left_batch + offs_m[:, None] * stride_left_i + tl.arange(0, BLOCK_K)[None, :] * stride_left_k
    b_ptrs = right_batch + tl.arange(0, BLOCK_K)[:, None] * stride_right_k + offs_n[None, :] * stride_right_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = cur_k < N

        a = tl.load(a_ptrs, mask=mask_m[:, None] & k_mask[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=k_mask[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # write result
    out_ptrs = out_ptr + pid_b * stride_out_batch + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    The heavy part (mask+gate fusion and the N³ reduction) is implemented in Triton.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ----------- LayerNorm on input (float32) -------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for compute‑intensive path
    x = x.to(torch.float16)

    # ----------- Cast weights to fp16 --------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ----------- Linear projections -----------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,H]
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ----------- Gates -----------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # [B,N,N,H]

    # ----------- Mask & gate fusion (Triton) -------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ----------- Core reduction (batched GEMM) ----------
    B, N, _, H = left.shape  # left shape [B,N,N,H]

    # reshape to (B*H, N, N) and make sure memory is contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_dim = B * H
    left_flat = left_perm.view(batch_dim, N, N)
    right_flat = right_perm.view(batch_dim, N, N)

    out_flat = torch.empty_like(left_flat)   # fp16 output

    # strides (in elements)
    s_batch_l, s_i_l, s_k_l = left_flat.stride()
    s_batch_r, s_i_r, s_k_r = right_flat.stride()
    s_batch_o, s_i_o, s_k_o = out_flat.stride()

    # Triton block sizes (tuned for Hopper)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        s_batch_l, s_i_l, s_k_l,
        s_batch_r, s_k_r, s_i_r,   # note: k stride = original col stride, j stride = original row stride
        s_batch_o, s_i_o, s_k_o,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B,N,N,H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------- Second LayerNorm -----------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ----------- Final linear projection ---------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]

    # Cast back to original dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

"""
TriMul (outgoing) – Triton‑fused mask+gate and batched GEMM.

Algorithm
---------
1. Layer‑norm the input tensor (float32) and cast to fp16.
2. Linear projections (left/right) and their gates (all fp16).
3. Fuse projection × gate × optional mask with a tiny Triton kernel
   (`_gate_mask_kernel`).  This avoids extra memory passes.
4. Reshape to (B·H, N, N) and compute the core reduction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batch of H independent GEMMs (left @ rightᵀ) using a custom
   Triton kernel (`_tri_mul_gemm_kernel`).  Computation is in fp16 with
   fp32 accumulation.
5. Reshape back to (B,N,N,H), apply a second LayerNorm, multiply by the
   output gate and a final linear projection to the original channel
   dimension.
6. Cast the result back to the input dtype (usually float32).

The kernel implements the N³·H reduction efficiently on the B200,
meeting the target runtime.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) on a flattened view
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask is broadcast over the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    msk  = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * msk
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse ``proj * gate * mask`` using Triton.

    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],               # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the core TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_j,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)   # batch * hidden index
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block column

    # absolute coordinates of the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this (batch, hidden) pair
    left_batch  = left_ptr  + pid_b * stride_left_batch
    right_batch = right_ptr + pid_b * stride_right_batch

    # pointers for the k‑tiles
    a_ptrs = left_batch + offs_m[:, None] * stride_left_i + tl.arange(0, BLOCK_K)[None, :] * stride_left_k
    b_ptrs = right_batch + tl.arange(0, BLOCK_K)[:, None] * stride_right_k + offs_n[None, :] * stride_right_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over the sequence dimension k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = cur_k < N

        a = tl.load(a_ptrs, mask=mask_m[:, None] & k_mask[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=k_mask[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # write the result
    out_ptrs = out_ptr + pid_b * stride_out_batch + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

            - config: dict with keys:

                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    """

    hidden_dim = config["hidden_dim"]

    dtype_in = input_tensor.dtype

        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    x = x.to(torch.float16)

    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # [B,N,N,H]

    # ------------------- mask & gate fusion (Triton) --------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ----------------- core O(N³·H) reduction via batched GEMM ----------
    B, N, _, H = left.shape                     # left: [B,N,N,H]

    # bring hidden dimension to leading axis and make memory contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_dim = B * H
    left_flat  = left_perm.view(batch_dim, N, N)   # [B*H, N, N]
    right_flat = right_perm.view(batch_dim, N, N)

    out_flat = torch.empty_like(left_flat)          # fp16 output buffer

    # strides (in elements)
    s_batch_l, s_i_l, s_k_l = left_flat.stride()
    s_batch_r, s_i_r, s_k_r = right_flat.stride()
    s_batch_o, s_i_o, s_k_o = out_flat.stride()

    # Triton block sizes (tuned for Hopper B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        s_batch_l, s_i_l, s_k_l,            # left strides
        s_batch_r, s_k_r, s_i_r,            # right: k‑stride = col‑stride, j‑stride = row‑stride
        s_batch_o, s_i_o, s_k_o,            # out strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B,N,N,H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]

    # ------------------------ cast back to input dtype ----------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
