"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate.

Algorithm
---------
1. Apply LayerNorm on the input tensor (B,N,N,C).
2. Cast to fp16 for the compute‑intensive path.
3. Compute four linear projections (left/right and their gates) in fp16.
4. Fuse the element‑wise multiplication of projection, gate and optional
   mask using a tiny Triton kernel (`_gate_mask_fuse`). This removes an
   extra memory pass.
5. Perform the core O(N³·H) reduction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
   This is implemented as a batch of H independent GEMMs of size N×N
   via `torch.bmm` on a reshaped view (B*H, N, N).
6. Apply a second LayerNorm on the hidden dimension, multiply by the
   output gate, and project back to the original channel dimension.
7. Cast the result back to the original input dtype (usually float32).

The kernel is required only for the mask‑gate fusion; the heavy reduction
uses cuBLAS through `torch.bmm`, which is highly optimized on the target
B200 GPU.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) with mask broadcasted
# over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse(
    proj_ptr,          # fp16 * (total_elems,)
    gate_ptr,          # fp16 * (total_elems,)
    mask_ptr,          # fp16 * (mask_elems,)
    out_ptr,           # fp16 * (total_elems,)
    total_elems,       # i32  – total number of elements in proj/gate
    hidden_dim,        # i32  – H (size of gate dimension)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    off = pid * BLOCK + tl.arange(0, BLOCK)
    mask = off < total_elems

    # each hidden_dim entries correspond to the same mask location
    mask_idx = off // hidden_dim

    proj = tl.load(proj_ptr + off, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + off, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + off, out, mask=mask)


def _apply_gate_mask(proj: torch.Tensor,
                     gate: torch.Tensor,
                     mask: torch.Tensor,
                     block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using Triton.
    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_fuse[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],        # hidden_dim
        BLOCK=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (fp16)
    # ------------------------------------------------------------------
    # Shape: [B, N, N, H]
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gates (fp16) + sigmoid
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Fuse mask, projection and gate
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the hidden dimension
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Rearrange to (B*H, N, N) for torch.bmm
    B, N, _, _ = left.shape
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # GEMM: (B*H, N, N) x (B*H, N, N)^T  -> (B*H, N, N)
    out_ = torch.bmm(left_, right_.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Second LayerNorm on hidden dimension (fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9. Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 10. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
