"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Input LayerNorm (float32 for stability).
2. Fuse left/right projections and their gates into two concatenated Linear
   operations, reducing the number of matrix multiplies from 5 to 3.
3. Fuse mask multiplication together with the element‑wise gate
   multiplication in a small Triton kernel (zero extra Python overhead).
4. Core contraction:  Einstein‑summation
   `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`,
   implemented with torch.einsum (batched GEMM internally).
5. Output LayerNorm, output gate, and final Linear back to the original
   dimension.
All heavy arithmetic is performed in fp16; the final result is cast back
to the input dtype (float32 by default).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr, total_elems, hidden_dim,
    HAS_MASK: tl.constexpr, BLOCK_SIZE: tl.constexpr
):
    """
    In‑place fusion of:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    Mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load data
    left      = tl.load(left_ptr      + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden entry shares the same mask value
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, dim)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, dim) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for compute‑heavy parts
    x = x.to(torch.float16)

    B, N, _, _ = x.shape
    positions = B * N * N
    x_flat = x.reshape(positions, dim)          # (B·N·N, dim)

    # -----------------------------------------------------------------
    # 2. Combined Linear projections (3 matmuls instead of 5)
    # -----------------------------------------------------------------
    # left_proj & left_gate
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device).to(torch.float16)                     # (2·H, dim)
    # right_proj & right_gate
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device).to(torch.float16)                     # (2·H, dim)
    # out_gate
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_comb = F.linear(x_flat, left_comb_w)          # (pos, 2·H)
    right_comb = F.linear(x_flat, right_comb_w)        # (pos, 2·H)
    out_gate_pre = F.linear(x_flat, out_gate_w)        # (pos, H)

    # split & activations
    left_proj = left_comb[:, :hidden_dim]
    left_gate = torch.sigmoid(left_comb[:, hidden_dim:])
    right_proj = right_comb[:, :hidden_dim]
    right_gate = torch.sigmoid(right_comb[:, hidden_dim:])
    out_gate = torch.sigmoid(out_gate_pre)

    # reshape to 4‑D tensors
    left_proj = left_proj.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_proj = right_proj.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 3. Fuse mask and gate multiplication (Triton)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16)
        mask_flat = mask_fp.view(-1)                     # (pos,)

        # flatten tensors for the kernel
        left_proj_f = left_proj.reshape(-1)
        left_gate_f = left_gate.reshape(-1)
        right_proj_f = right_proj.reshape(-1)
        right_gate_f = right_gate.reshape(-1)

        total = left_proj_f.numel()                      # pos * H
        BLOCK = 8192
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_proj_f,
            left_gate_f,
            right_proj_f,
            right_gate_f,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left = left_proj_f.view_as(left_proj)
        right = right_proj_f.view_as(right_proj)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 4. Core contraction (Einstein summation)
    # -----------------------------------------------------------------
    #   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)

    # -----------------------------------------------------------------
    # 5. Output LayerNorm, output gate and final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)  # (dim, H)
    out = F.linear(out, to_out_w)                                     # (B,N,N,dim)

    # Cast back to original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
