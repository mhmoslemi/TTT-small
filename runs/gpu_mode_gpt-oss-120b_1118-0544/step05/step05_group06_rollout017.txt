def _gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, total_elems, hidden_dim, BLOCK_SIZE: tl.constexpr):

"""
Highly‑optimized forward pass for the outgoing Triangle Multiplicative Update
(TriMul) operator (AlphaFold3 style).

Algorithm
---------
1. Layer‑normalize the input tensor (float32) and cast to FP16.
2. Fuse left/right linear projections into a single matmul (2·H output
   channels) and similarly fuse left/right gates.  This halves the number of
   dense GEMM calls.
3. Apply the optional mask together with the gates using a tiny Triton kernel
   that performs element‑wise `proj * gate * mask` (mask is broadcast over the
   hidden dimension).
4. Core reduction: for each head `h` compute
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched matrix multiplication
        (B, H, N, N) @ (B, H, N, N)ᵀ via `torch.matmul`.
5. Apply a second LayerNorm over the hidden dimension, multiply by the
   `out_gate`, and finally linearly project back to the original channel size.
6. Cast the result back to the original dtype (typically float32).

A single Triton kernel is used (step 3) to satisfy the requirement of a
custom kernel while the compute‑intensive dense operations leverage
cuBLAS/Tensor‑cores through PyTorch’s fused kernels.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,            # *[total_elems]   fp16
    gate_ptr,            # *[total_elems]   fp16
    mask_ptr,            # *[mask_elems]    fp16
    out_ptr,             # *[total_elems]   fp16
    total_elems,         # i64
    hidden_dim,          # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # Index in the mask (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Element‑wise ``proj * gate * mask`` where ``mask`` has shape (B,N,N)
    and is broadcast over the hidden dimension ``H``.
    All tensors must be ``torch.float16`` and contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],          # hidden_dim (H)
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict containing all linear and LayerNorm parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    orig_dtype = x.dtype
    device = x.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) and cast to FP16 for heavy work
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Cast weights to FP16 (once)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Fuse left/right projections into a single matmul
    #    Shape after linear: (B, N, N, 2*H) -> split into left/right
    # ------------------------------------------------------------------
    proj_w = torch.cat([left_proj_w, right_proj_w], dim=0)  # (2*H, C)
    proj = F.linear(x, proj_w)                               # (B, N, N, 2*H)
    left_proj, right_proj = proj.chunk(2, dim=-1)

    # ------------------------------------------------------------------
    # 4. Fuse left/right gates (with sigmoid) into a single matmul
    # ------------------------------------------------------------------
    gate_w = torch.cat([left_gate_w, right_gate_w], dim=0)   # (2*H, C)
    gate = torch.sigmoid(F.linear(x, gate_w))                # (B, N, N, 2*H)
    left_gate, right_gate = gate.chunk(2, dim=-1)

    # ------------------------------------------------------------------
    # 5. Out gate (separate)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))        # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Apply mask (if present) together with gating using Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched matmul over the sequence dimension
    #    left:  (B, N, N, H) -> (B, H, N, N)
    #    right: (B, N, N, H) -> (B, H, N, N)
    #    out  = left @ rightᵀ   => (B, H, N, N)
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2)    # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2)   # (B, H, N, N)
    out_t   = torch.matmul(left_t, right_t.transpose(-2, -1))  # (B, H, N, N)
    out = out_t.permute(0, 2, 3, 1).contiguous()               # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)        # (B, N, N, C)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)
    return out
