def gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, total_elements, hidden_dim, BLOCK_SIZE: tl.constexpr):

    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)

    # compute mask  index: each element index corresponds to (b, i, k). We need to map each element to mask position.
    # For flatten order (B,N,N,hidden_dim), offset // hidden_dim corresponds to element index in mask flatten (B,N,N). So mask_idx = offs // hidden_dim.

    tl.store(out_ptr + offs, out, mask=mask)

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C).
2. Cast to FP16 and apply three linear projections:
   * left_proj, right_proj  → (B,N,N,H)
   * left_gate, right_gate, out_gate → (B,N,N,H) (sigmoid)
3. Fuse element‑wise mask (if present) with the projection and gate
   using a tiny Triton kernel.
4. Core O(N³·H) reduction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication in Triton
   (one program id for the batch‑head dimension, two for the i/j tiles).
5. Apply a second LayerNorm, multiply by the output gate and linearly
   project back to the original channel dimension.
6. Cast back to the input dtype.

The heavy O(N³·H) operation is performed entirely in Triton, while
all linear layers use Torch’s cuBLAS‑backed `F.linear`.  This reduces
memory reshapes and fuses the reduction across the hidden dimension,
improving throughput on B200 GPUs.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Element‑wise mask * gate * projection fusion
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_fuse_kernel(
    proj_ptr,          # *[total_elems]   fp16
    gate_ptr,          # *[total_elems]   fp16
    mask_ptr,          # *[mask_elems]    fp16
    out_ptr,           # *[total_elems]   fp16
    total_elems,       # int
    hidden_dim,        # int
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # mask index = element index // hidden_dim  (since mask lacks the hidden dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # default 1.0 → no‑mask effect

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    proj, gate : (B, N, N, H)  fp16 contiguous
    mask       : (B, N, N)    fp16 contiguous
    Returns    : proj * gate * mask   (fp16)
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.view(-1).contiguous()
    gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Batched matrix multiplication for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def trmul_outgoing_kernel(
    left_ptr,          # *[B,N,N,H] fp16
    right_ptr,         # *[B,N,N,H] fp16
    out_ptr,           # *[B,N,N,H] fp16
    B, N, H,           # runtime sizes
    stride_b, stride_i, stride_k, stride_h,   # strides in elements
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)        # tile row   → i
    pid_j = tl.program_id(1)        # tile col   → j
    pid_bn = tl.program_id(2)       # batch * hidden

    # decode batch and hidden‑dim index
    b = pid_bn // H
    h = pid_bn % H

    # offsets within the tile
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_off < N
    j_mask = j_off < N

    # base pointer for this (b, h) slice
    base = b * stride_b + h * stride_h

    # accumulator in FP32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, N, BLOCK_K):
        k_off = k + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # left tile: shape (M, K)   → left[b,i,k,h]
        left = tl.load(
            left_ptr + base
            + i_off[:, None] * stride_i
            + k_off[None, :] * stride_k,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0,
        )

        # right tile transposed: shape (K, N)   → right[b,j,k,h] accessed as (k, j)
        right = tl.load(
            right_ptr + base
            + k_off[:, None] * stride_k
            + j_off[None, :] * stride_i,
            mask=k_mask[:, None] & j_mask[None, :],
            other=0.0,
        )

        # dot product (M×K)·(K×N) → (M×N)
        acc += tl.dot(left, right)

    # write back the result (cast to fp16)
    out_ptrs = out_ptr + base \
        + i_off[:, None] * stride_i \
        + j_off[None, :] * stride_k
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])


def trmul_batched_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : (B, N, N, H)  fp16 contiguous
    Returns      : (B, N, N, H)  fp16  (out = left @ rightᵀ over the second dim)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # contiguous strides (elements, not bytes)
    stride_b = left.stride(0)
    stride_i = left.stride(1)   # stride for the first N dimension (i / j)
    stride_k = left.stride(2)   # stride for the second N dimension (k)
    stride_h = left.stride(3)   # should be 1

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    trmul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_b,
        stride_i,
        stride_k,
        stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    data = (input_tensor, mask, weights, config)
    Returns a tensor of shape [bs, seqlen, seqlen, dim] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the compute‑intensive part
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Cast all model weights to FP16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣ Linear projections: (B,N,N,dim) → (B,N,N,hidden_dim)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)   # (B,N,N,H)

    # --------------------------------------------------------------
    # 5️⃣ Gate projections + sigmoid
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,H)

    # --------------------------------------------------------------
    # 6️⃣ Fuse mask, projection and gate (Triton) – or simple mul if no mask
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = apply_gate_and_mask(left_proj,  left_gate,  mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7️⃣ Core O(N³·H) reduction via a Triton batched matmul
    # --------------------------------------------------------------
    out = trmul_batched_matmul(left, right)   # (B,N,N,H)

    # --------------------------------------------------------------
    # 8️⃣ Output LayerNorm, output‑gate and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 9️⃣ Cast back to the original dtype
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
