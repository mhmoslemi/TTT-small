"""
Optimized Triton‑accelerated forward pass for the outgoing TriMul operator
(AlphaFold3 style).

Algorithm
---------
1. Layer‑normalize the 4‑D input tensor (B, N, N, C) (float32).
2. Cast to FP16 for compute‑heavy stages.
3. Linear projections (left, right) and gate projections (left_gate,
   right_gate, out_gate) are performed with FP16 weights.
4. Fuse element‑wise multiplication of projection, gate and optional mask
   with a small Triton kernel (mask is broadcast over the hidden dim H).
5. Core reduction:
      out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   implemented with a single `torch.einsum`, avoiding costly permutes /
   copies.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   out‑gate and project back to the original channel dimension.
7. Cast the result back to the input dtype (usually float32).

Only the element‑wise gate‑mask fusion is done in Triton; the rest relies on
highly‑optimised PyTorch kernels (layer‑norm, linear, and einsum) which
together meet the required runtime on modern GPUs.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,                # *[total_elems]   fp16
    gate_ptr,                # *[total_elems]   fp16
    mask_ptr,                # *[mask_elems]    fp16
    out_ptr,                 # *[total_elems]   fp16
    total_elems,             # i64  (B*N*N*H)
    hidden_dim,              # i64  (H)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # Map flat index -> mask index (broadcast H)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Element‑wise ``proj * gate * mask`` where ``mask`` has shape (B, N, N)
    and is broadcast over the hidden dimension H.
    All tensors must be FP16 and contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],   # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to half for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all linear weights to half (biases are None for these)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B, N, N, H) – FP16
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (sigmoid) – FP16
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Apply mask + gate (Triton kernel) or just gate
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with torch.einsum to avoid extra permutes / copies.
    # ------------------------------------------------------------------
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm over hidden dimension, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)          # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype (usually float32) and return
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
