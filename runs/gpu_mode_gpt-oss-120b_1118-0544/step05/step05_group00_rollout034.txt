def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)
    
    # LayerNorm on last dim
    norm_weight = weights["norm.weight"].to(device).float()
    norm_bias = weights["norm.bias"].to(device).float()
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)  # float32
    x = x.to(torch.float16)
    
    # Linear layers (bias-free)
    left_proj_weight = weights["left_proj.weight"].to(device).half()
    right_proj_weight = weights["right_proj.weight"].to(device).half()
    left_gate_weight = weights["left_gate.weight"].to(device).half()
    right_gate_weight = weights["right_gate.weight"].to(device).half()
    out_gate_weight = weights["out_gate.weight"].to(device).half()
    
    # compute raw projections
    left_raw = F.linear(x, left_proj_weight)          # (B,N,N,H)
    right_raw = F.linear(x, right_proj_weight)       # (B,N,N,H)
    left_gate_raw = F.linear(x, left_gate_weight)    # (B,N,N,H)
    right_gate_raw = F.linear(x, right_gate_weight)  # (B,N,N,H)
    out_gate_raw = F.linear(x, out_gate_weight)      # (B,N,N,H)
    
    # Apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)
    
    # Apply mask if needed
    if mask is not None and not nomask:
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_raw * left_gate * mask_f
        right = right_raw * right_gate * mask_f
    else:
        left = left_raw * left_gate
        right = right_raw * right_gate
    
    # Core contraction: for each batch and channel compute bmm
    B, N, _, H = left.shape
    # reshape to (B*H,N,N)
    left_mat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_mat = right.permute(0,3,1,2).contiguous().view(B*H, N, N)
    
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))  # (B*H,N,N)
    
    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()  # (B,N,N,H)
    
    # Output layernorm and linear
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).half()
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).half()
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    
    out = out * out_gate   # out_gate shape same.
    
    to_out_weight = weights["to_out.weight"].to(device).half()
    out = F.linear(out, to_out_weight)   # (B,N,N,dim)
    
    # Cast back to original dtype
    out = out.to(dtype)
    return out

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_apply_kernel(
    out_ptr, raw_ptr, gate_ptr, mask_ptr,
    total_elements, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elements

    # load raw and gate
    raw = tl.load(raw_ptr + offs, mask=active, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=active, other=0.0)
    # sigmoid
    sig = 1.0 / (1.0 + tl.exp(-gate))

    if HAS_MASK:
        # mask index = offset // hidden_dim (mask has no hidden dim)
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        out = raw * sig * m
    else:
        out = raw * sig

    tl.store(out_ptr + offs, out, mask=active)

def _apply_gate_and_mask(tensor_raw, gate_tensor, mask_tensor, has_mask):
    """
    In-place: tensor_raw = tensor_raw * sigmoid(gate_tensor) * mask
    tensor_raw, gate_tensor, mask_tensor are flat contiguous tensors of same dtype (fp16).
    mask_tensor has shape (B,N,N) flattened; has_mask indicates whether to use it.
    """
    total = tensor_raw.numel()
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

    _gate_apply_kernel[grid](
        tensor_raw,
        tensor_raw,
        gate_tensor,
        mask_tensor,
        total,
        tensor_raw.shape[-1],   # hidden_dim
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )
    return tensor_raw

def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul update.
    Performs:
      1. LayerNorm on input.
      2. Linear projections to hidden dim.
      3. In‑place gating + optional mask via a tiny Triton kernel.
      4. Batched GEMM (L @ Rᵀ) for the O(N³) contraction.
      5. Output LayerNorm, output‑gate, and final linear projection.
    Returns a tensor of shape (B, N, N, dim) with the same dtype as the input.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------ #
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # Cast to fp16 for the compute‑intensive path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------ #
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left_raw   = F.linear(x, left_proj_w)        # (B,N,N,hidden_dim)
    right_raw  = F.linear(x, right_proj_w)       # (B,N,N,hidden_dim)

    left_gate_raw  = F.linear(x, left_gate_w)    # (B,N,N,hidden_dim)
    right_gate_raw = F.linear(x, right_gate_w)   # (B,N,N,hidden_dim)
    out_gate_raw   = F.linear(x, out_gate_w)     # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------ #
    # 3. Fuse sigmoid gating and optional mask using a Triton kernel
    # ------------------------------------------------------------------ #
    has_mask = (mask is not None) and (not nomask)
    if has_mask:
        # mask shape (B,N,N) -> flatten
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        mask_flat = mask_fp.view(-1)
    else:
        # dummy placeholder (won't be accessed)
        mask_flat = torch.empty(0, device=device, dtype=torch.float16)

    # Apply in-place to left_raw and right_raw
    _apply_gate_and_mask(left_raw, left_gate_raw, mask_flat, has_mask)
    _apply_gate_and_mask(right_raw, right_gate_raw, mask_flat, has_mask)

    # Compute out_gate (sigmoid) – this one is cheap and can stay in PyTorch
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------ #
    # 4. Core contraction: batched matrix multiplication L @ Rᵀ
    # ------------------------------------------------------------------ #
    B, N, _, H = left_raw.shape
    # reshape to (B*H, N, N)
    left_mat  = left_raw.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_raw.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched GEMM using torch (cuBLAS / Tensor cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 5. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------ #
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (half‑precision)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. LayerNorm on the input (float32 for stability).
2. Cast to fp16 for the compute‑intensive path.
3. Bias‑less linear projections to the hidden dimension.
4. Fuse sigmoid gating and optional binary mask with a tiny Triton kernel
   (in‑place on the projected tensors).
5. Core O(N³·H) contraction performed as a batched GEMM:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   i.e. left @ rightᵀ for each hidden head.
6. Apply a second LayerNorm, multiply by the output gate and a final linear
   projection back to the original dimension.
7. Cast back to the input dtype.

A single Triton kernel is used for the element‑wise gating+mask step;
the heavy N³ contraction leverages cuBLAS via torch.bmm.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_apply_kernel(
    out_ptr, raw_ptr, gate_ptr, mask_ptr,
    total_elements, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """In‑place: out = raw * sigmoid(gate) * mask (if present)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elements

    raw = tl.load(raw_ptr + offs, mask=active, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=active, other=0.0)

    # sigmoid
    sig = 1.0 / (1.0 + tl.exp(-gate))

    if HAS_MASK:
        # mask has no hidden dimension → collapse that axis
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        out = raw * sig * m
    else:
        out = raw * sig

    tl.store(out_ptr + offs, out, mask=active)


def _apply_gate_and_mask(tensor_raw, gate_tensor, mask_flat, has_mask):
    """Apply sigmoid(gate) and optional mask in‑place using Triton."""
    total = tensor_raw.numel()
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    _gate_apply_kernel[grid](
        tensor_raw,               # out_ptr (in‑place)
        tensor_raw,               # raw_ptr
        gate_tensor,              # gate_ptr
        mask_flat,                # mask_ptr (may be empty when HAS_MASK=False)
        total,
        tensor_raw.shape[-1],     # hidden_dim
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )
    return tensor_raw


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul update.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: (B, N, N, dim) torch.Tensor
            - mask: (B, N, N) torch.Tensor or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, dim) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    ln_w = weights["norm.weight"].to(device).to(torch.float32)
    ln_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=ln_w, bias=ln_b)  # (B,N,N,dim)

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for the heavy path
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (bias‑less)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left_raw   = F.linear(x, left_proj_w)     # (B,N,N,hidden_dim)
    right_raw  = F.linear(x, right_proj_w)    # (B,N,N,hidden_dim)

    left_gate_raw  = F.linear(x, left_gate_w)   # (B,N,N,hidden_dim)
    right_gate_raw = F.linear(x, right_gate_w)  # (B,N,N,hidden_dim)
    out_gate_raw   = F.linear(x, out_gate_w)    # (B,N,N,hidden_dim)

    # -----------------------------------------------------------------
    # 4. Fuse sigmoid gating and optional mask (Triton kernel)
    # -----------------------------------------------------------------
    has_mask = (mask is not None) and (not nomask)
    if has_mask:
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        mask_flat = mask_fp.view(-1)                 # (B*N*N,)
    else:
        mask_flat = torch.empty(0, device=device, dtype=torch.float16)

    # In‑place gating + mask
    _apply_gate_and_mask(left_raw, left_gate_raw, mask_flat, has_mask)
    _apply_gate_and_mask(right_raw, right_gate_raw, mask_flat, has_mask)

    # Output gate (sigmoid) – cheap, keep in PyTorch
    out_gate = torch.sigmoid(out_gate_raw)   # (B,N,N,hidden_dim)

    # -----------------------------------------------------------------
    # 5. Core contraction via batched GEMM (L @ Rᵀ)
    # -----------------------------------------------------------------
    B, N, _, H = left_raw.shape
    left_mat  = left_raw.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_raw.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden_dim)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output‑gate, final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate                         # apply output gate (fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)               # (B,N,N,dim) in fp16

    # -----------------------------------------------------------------
    # 7. Cast back to the input dtype and return
    # -----------------------------------------------------------------
    out = out.to(dtype_in)
    return out
