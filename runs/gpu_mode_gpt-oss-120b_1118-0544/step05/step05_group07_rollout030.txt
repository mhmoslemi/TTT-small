import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Kernel to fuse projection, gate and mask
@triton.jit
def _gate_mask_kernel(
    proj_ptr,   # fp16 *
    gate_ptr,   # fp16 *
    mask_ptr,   # fp16 *
    out_ptr,    # fp16 *
    total_elems,   # i32
    hidden_dim,    # i32
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # mask is broadcasted over hidden dimension
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 4096) -> torch.Tensor:
    """Fuse proj * gate * mask using a Triton kernel."""
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out

# Batched GEMM kernel for the core O(N³·H) reduction
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,  # all equal to N in our case
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)   # batch index (B * H)
    pid_m = tl.program_id(0)   # row tile
    pid_n = tl.program_id(1)   # column tile

    # Starting offsets of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Batch offsets
    a_batch_offset = pid_b * stride_a_batch
    b_batch_offset = pid_b * stride_b_batch
    c_batch_offset = pid_b * stride_c_batch

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A block [M, K]
        a_ptrs = a_ptr + a_batch_offset \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0)

        # Load B block [K, N] (note: B is stored as (K, N) because we pre‑transposed it)
        b_ptrs = b_ptr + b_batch_offset \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back the result in FP16
    c_ptrs = c_ptr + c_batch_offset \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    c = acc.to(tl.float16)
    tl.store(c_ptrs,
             c,
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])

def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    The function follows the PyTorch reference implementation but
    fuses the projection‑gate‑mask step and the N³ reduction into
    custom Triton kernels for better performance.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm (keep in float32 for numerical stability)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj   = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj  = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16))
    left_gate   = torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate  = torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate    = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4) Fuse mask and gate (if a mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Core reduction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM with a Triton kernel.
    # ------------------------------------------------------------------
    batch, seq_len, _, _ = left.shape  # batch = B, seq_len = N

    # Bring the hidden dimension to the batch axis
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(-1, seq_len, seq_len)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(-1, seq_len, seq_len)  # (B*H, N, N)

    # Transpose the second operand so that we can perform a standard matmul
    right_t = right_mat.transpose(1, 2).contiguous()  # (B*H, N, N)

    # Prepare output tensor
    out_mat = torch.empty_like(left_mat)

    # Strides (in elements) for the three tensors
    stride_a_batch = left_mat.stride(0)
    stride_a_m     = left_mat.stride(1)
    stride_a_k     = left_mat.stride(2)

    stride_b_batch = right_t.stride(0)
    stride_b_k     = right_t.stride(1)   # K dimension (originally rows after transpose)
    stride_b_n     = right_t.stride(2)   # N dimension (columns after transpose)

    stride_c_batch = out_mat.stride(0)
    stride_c_m     = out_mat.stride(1)
    stride_c_n     = out_mat.stride(2)

    # Tile sizes – tuned for the B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = triton.cdiv(seq_len, BLOCK_M)
    num_blocks_n = triton.cdiv(seq_len, BLOCK_N)
    batch_cnt    = left_mat.shape[0]   # B * hidden_dim

    grid = (num_blocks_m, num_blocks_n, batch_cnt)
    _batched_matmul_kernel[grid](
        left_mat,
        right_t,
        out_mat,
        seq_len,
        seq_len,
        seq_len,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_mat.view(batch, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Second layer‑norm, output gate and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate

    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
    )

    # ------------------------------------------------------------------
    # 7) Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    return out.to(input_tensor.dtype)

    total_elems,   # i32   – total number of elements in proj/gate
    hidden_dim,    # i32   – size of the hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # load projection & gate
    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # mask is broadcasted over the hidden dimension
    mask_idx = offs // hidden_dim
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 4096) -> torch.Tensor:
    """Element‑wise proj * gate * mask fused in a Triton kernel."""
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],       # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the N³ reduction
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                     # matrix sizes (all equal to seq_len)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)   # index over (batch * hidden_dim)
    pid_m = tl.program_id(0)   # row tile
    pid_n = tl.program_id(1)   # column tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # batch offsets
    a_batch = pid_b * stride_a_batch
    b_batch = pid_b * stride_b_batch
    c_batch = pid_b * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        a = tl.load(
            a_ptr + a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k,
            mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptr + b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n,
            mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    c = acc.to(tl.float16)
    tl.store(
        c_ptr + c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n,
        c,
        mask=(offs_m < M)[:, None] & (offs_n < N)[None, :],
    )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    The implementation mirrors the reference PyTorch code but
    fuses the projection‑gate‑mask step and the N³ reduction
    into two Triton kernels for higher performance.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (keep in float32 for stability)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(
        x, weights["left_proj.weight"].to(torch.float16)
    )
    right_proj = torch.nn.functional.linear(
        x, weights["right_proj.weight"].to(torch.float16)
    )
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4) Fuse mask and gate (if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Core reduction via batched GEMM (TriMul)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B = batch, N = seq_len

    # reshape to (B*H, N, N) – each hidden channel becomes an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N) # (B*H, N, N)

    # transposed operand for the GEMM (so we can do A @ B_T)
    right_t = right_mat.transpose(1, 2).contiguous()  # (B*H, N, N)

    out_mat = torch.empty_like(left_mat)

    # strides in elements
    stride_a_batch = left_mat.stride(0)
    stride_a_m     = left_mat.stride(1)
    stride_a_k     = left_mat.stride(2)

    stride_b_batch = right_t.stride(0)
    stride_b_k     = right_t.stride(1)   # K dimension after transpose
    stride_b_n     = right_t.stride(2)   # N dimension after transpose

    stride_c_batch = out_mat.stride(0)
    stride_c_m     = out_mat.stride(1)
    stride_c_n     = out_mat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)
    batch_cnt = left_mat.shape[0]  # B * hidden_dim

    grid = (num_blocks_m, num_blocks_n, batch_cnt)
    _batched_matmul_kernel[grid](
        left_mat,
        right_t,
        out_mat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Second LayerNorm, output gate and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate

    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float16)
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 7) Cast back to the original dtype
    # ------------------------------------------------------------------
    return out.to(input_tensor.dtype)
