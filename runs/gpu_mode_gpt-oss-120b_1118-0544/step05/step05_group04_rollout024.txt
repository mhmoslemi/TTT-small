Define kernel:

def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0) # block row index
    pid_j = tl.program_id(1) # block col index
    pid_batch = tl.program_id(2) # batch*hidden index

    # compute offsets for this batch
    a_ptr = a_ptr + pid_batch * stride_ab
    b_ptr = b_ptr + pid_batch * stride_bb
    c_ptr = c_ptr + pid_batch * stride_cb

    # compute block start indices
    block_i = pid_i * BLOCK_M
    block_j = pid_j * BLOCK_N

    # compute range of i and j within block
    i = block_i + tl.arange(0, BLOCK_M)
    j = block_j + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_i = i < M
    mask_j = j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # load tiles
        a = tl.load(a_ptr + (i[:, None] * stride_am + k_offset[None, :] * stride_ak), mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptr + (k_offset[:, None] * stride_bk + j[None, :] * stride_bn), mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # compute dot product (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

    # write back
    c = acc.to(tl.float16)
    tl.store(c_ptr + (i[:, None] * stride_cm + j[None, :] * stride_cn), c, mask=mask_i[:, None] & mask_j[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) using the provided LN weights.
2. Project to hidden dimension H with two linear layers (left/right).
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. Fuse mask (if present) and left/right gates in a Triton kernel:
       left_out = left  * mask * left_gate
       right_out = right * mask * right_gate
5. Perform the core reduction
       out[b,i,j,h] = Σ_k left_out[b,i,k,h] * right_out[b,j,k,h]
   with a custom Triton batched‑matmul kernel (no cuBLAS).
6. Reshape back to (B,N,N,H), apply a second LayerNorm over H,
   multiply by the output gate and linearly project back to the original
   channel dimension C.
7. Cast the result back to the input dtype (typically float32).

The heavy O(N³·H) work is carried out by the Triton kernel, while
element‑wise work (masking, gating, layer‑norms, final projection) uses
native PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,          # const float* left   (B,N,N,H)
    right_ptr,         # const float* right
    mask_ptr,          # const float* mask   (B,N,N) flattened
    left_gate_ptr,     # const float* left_gate
    right_gate_ptr,    # const float* right_gate
    left_out_ptr,      # float* left_out
    right_out_ptr,     # float* right_out
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,   # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask and gates:
       left_out  = left  * mask * left_gate
       right_out = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # load values
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # mask is broadcast over the hidden dimension
    mask_idx = offs // H            # (B,N,N) index
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # apply fused mask & gates
    tl.store(left_out_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,        # a: (batch, M, K)
    stride_bk, stride_bn, stride_bb,        # b: (batch, K, N)  (accessed as K,N)
    stride_cm, stride_cn, stride_cb,        # c: (batch, M, N)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication:
       for each batch index p:
           C[p] = A[p] @ B[p].T   (A: MxK, B: NxK  =>  C: MxN)
    """
    pid_i = tl.program_id(0)   # tile row index
    pid_j = tl.program_id(1)   # tile col index
    pid_p = tl.program_id(2)   # batch*hidden index

    # pointers for the current batch
    a_batch = a_ptr + pid_p * stride_ab
    b_batch = b_ptr + pid_p * stride_bb
    c_batch = c_ptr + pid_p * stride_cb

    # compute tile start indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # range within the tile
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < M
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch + (i[:, None] * stride_am + k[None, :] * stride_ak),
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile (as K x N): (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_batch + (k[:, None] * stride_bk + j[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # write back
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (i[:, None] * stride_cm + j[None, :] * stride_cn),
        c,
        mask=mask_i[:, None] & mask_j[None, :],
    )

    stride_ak = 1               # col stride of A
    stride_ab = N * N           # batch stride of A

    stride_bk = N               # row stride of B (K dimension)
    stride_bn = 1               # col stride of B
    stride_bb = N * N           # batch stride of B

    stride_cm = N               # row stride of C
    stride_cn = 1               # col stride of C
    stride_cb = N * N           # batch stride of C

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_mat = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_h,
    )
    _batched_matmul_kernel[grid_mat](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6) Reshape back to (B,N,N,H)
    # ------------------------------------------------------------------
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
