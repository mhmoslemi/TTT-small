def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets for each block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers offset to this batch
    a_batch_ptr = a_ptr + batch_id * stride_a_batch
    b_batch_ptr = b_ptr + batch_id * stride_b_batch
    c_batch_ptr = c_ptr + batch_id * stride_c_batch

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # use fp32

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # tiles
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load a tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(a_batch_ptr + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k), mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load b tile: shape (BLOCK_K, BLOCK_N)

        # compute dot product
        acc += tl.dot(a, b)

    # Store result
    c = acc.to(tl.float16)  # convert to fp16
    tl.store(c_batch_ptr + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n), c, mask=mask_m[:, None] & mask_n[None, :])

def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # offsets for batch strides
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load a tile
        a = tl.load(a_batch + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k),
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        # load b tile
        b = tl.load(b_batch + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n),
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # accumulate

    # store result

"""
custom_kernel: Triton‑accelerated outgoing TriMul (Triangle Multiplicative Update)

Algorithm
---------
1. Layer‑norm the input (float32 for numeric stability) and cast to FP16.
2. Linear projections (bias‑less) produce `left`, `right` and three gate tensors.
3. Fuse element‑wise mask (if present) and gates with a tiny Triton kernel
   (`_gate_mask_kernel`).  The result is written back in‑place, avoiding extra passes.
4. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication L @ Rᵀ for each hidden channel `d`.
   A custom Triton kernel (`_batched_matmul_kernel`) performs the batched GEMM
   with FP16 inputs and FP32 accumulation.
5. Layer‑norm over the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original channel dimension.
6. Cast the result back to the input dtype.

Only the mask‑gate fusion and the massive O(N³·H) contraction are
implemented in Triton; everything else uses PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------------- #
# 1. Fuse mask and gates (in‑place)                                            #
# ----------------------------------------------------------------------------- #
@triton.jit
def _gate_mask_kernel(
    left_ptr,               # fp16* [B,N,N,H] flattened
    right_ptr,              # fp16* [B,N,N,H] flattened
    left_gate_ptr,          # fp16* same shape
    right_gate_ptr,         # fp16* same shape
    mask_ptr,               # fp16* [B,N,N] flattened (broadcast over H)
    total_elems,            # int64 – number of elements in left/right
    hidden_dim,             # int64 – H
    HAS_MASK: tl.constexpr, # compile‑time bool
    BLOCK_SIZE: tl.constexpr,  # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # load tensors
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden slice shares the same mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # store back in‑place
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


# ----------------------------------------------------------------------------- #
# 2. Batched matrix multiplication (L @ Rᵀ)                                    #
# ----------------------------------------------------------------------------- #
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # fp16* tensors
    M, N, K,                                 # matrix sizes (all = seq_len)
    stride_a_batch, stride_a_m, stride_a_k,  # A strides
    stride_b_batch, stride_b_k, stride_b_n,  # B strides (B is already transposed)
    stride_c_batch, stride_c_m, stride_c_n,  # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # batch pointers
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # accumulator (FP32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M x K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B tile (K x N) – note B is already transposed
        b = tl.load(
            b_batch + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # FP16 × FP16 → FP32 accumulation

    # write back result (cast to FP16)
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------------- #
# 3. Entry point                                                               #
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C) (float32 or float16)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – True ⇒ ignore mask even if provided
    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as the input.
    """
    # ------------------------------------------------- #
    # unpack inputs
    # ------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------- #
    # 1. Input LayerNorm (float32) + cast to fp16
    # ------------------------------------------------- #
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)
    x = x.to(torch.float16)

    # ------------------------------------------------- #
    # 2. Linear projections (bias‑less)
    # ------------------------------------------------- #
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)                     # (B,N,N,H)
    right = F.linear(x, right_proj_w)                   # (B,N,N,H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B,N,N,H)

    # ------------------------------------------------- #
    # 3. Fuse mask and gates (if required)
    # ------------------------------------------------- #
    use_mask = (mask is not None) and (not nomask)
    if use_mask:
        # ensure mask is fp16 and contiguous
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten all tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_fp.contiguous().view(-1)  # (B*N*N,)

        total = left_f.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # simple element‑wise fusion without mask
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------- #
    # 4. Core contraction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------- #
    B, N, _, H = left.shape
    # Move hidden dimension to the batch axis and make Right transposed
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # (B*H, N, N) = Rᵀ

    batch = B * H
    M = N
    Ndim = N
    K = N

    # allocate output buffer
    out_mat = torch.empty((batch, M, Ndim), dtype=torch.float16, device=device)

    # strides (in elements)
    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_k = right_mat.stride(1)   # rows correspond to K after transpose
    stride_b_n = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_m = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    GROUP_M = 8

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M,
        Ndim,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_M=GROUP_M,
    )

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------- #
    # 5. Output layer‑norm, gate, final linear
    # ------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (FP16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in FP16

    # ------------------------------------------------- #
    # 6. Cast back to input dtype
    # ------------------------------------------------- #
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
