@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    batch,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_j,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # number of blocks in i and j direction
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    # compute batch index and tile indices
    pid_batch = pid // (num_blocks_m * num_blocks_n)
    pid_in_batch = pid % (num_blocks_m * num_blocks_n)
    pid_m = pid_in_batch // num_blocks_n
    pid_n = pid_in_batch % num_blocks_n

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers offsets
        a_ptrs = left_ptr + pid_batch * stride_left_batch + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        b_ptrs = right_ptr + pid_batch * stride_right_batch + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)  # accumulate in fp32

    # store result
    c_ptrs = out_ptr + pid_batch * stride_out_batch + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    # cast to fp16 before store
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)
    device = x.device
    dtype = x.dtype

    # LayerNorm
    norm_weight = weights["norm.weight"].to(device).to(torch.float32)
    norm_bias = weights["norm.bias"].to(device).to(torch.float32)
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # Cast to fp16 for compute
    x = x.to(torch.float16)

    # Gather all projection weights
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)  # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

                              right_proj_weight, right_gate_weight,
                              out_gate_weight], dim=0)  # (5*hidden_dim, dim)

    # Split
    H = hidden_dim

    left_gate = proj[:, H:2*H]

    right_gate = proj[:, 3*H:4*H]
    out_gate = proj[:, 4*H:5*H]

    # Apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Apply mask to gates if any
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
    else:
        mask_fp = None

    # Compute left and right after gating
    left = left_proj * left_gate
    right = right_proj * right_gate

    if mask_fp is not None:
        left = left * mask_fp
        right = right * mask_fp

    # Reshape to (B,N,N,H)
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # Core contraction: outgoing tri mul
    # Permute to (B,H,N,N) and make contiguous for batching
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Flatten batch dimension
    batch = B * H
    left_flat = left_perm.view(batch, N, N)
    right_flat = right_perm.view(batch, N, N)

    out_flat = torch.empty_like(left_flat)

    # Get strides
    left_strides = left_flat.stride()
    right_strides = right_flat.stride()
    out_strides = out_flat.stride()

    # Set block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Compute grid size
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    # Launch kernel
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_strides[0],
        left_strides[1],
        left_strides[2],
        right_strides[0],
        right_strides[2],
        right_strides[1],
        out_strides[0],
        out_strides[1],
        out_strides[2],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,H)

    # Output LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out = torch.nn.functional.layer_norm(out, (H,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply output gate
    out_gate = out_gate.view(B, N, N, H)  # already in that shape
    out = out * out_gate

    # Final linear projection
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)
    # convert to fp32 for final linear? Use fp16 then cast back.
    out = torch.nn.functional.linear(out.to(torch.float16), to_out_weight)  # shape (B,N,N,dim)

    # Cast to original dtype
    out = out.to(dtype)

    return out

"""
Optimized outgoing TriMul forward pass using Triton.

Algorithm:
1. Apply LayerNorm on the input tensor (float32).
2. Fuse the five linear projections (left_proj, left_gate,
   right_proj, right_gate, out_gate) into a single GEMM.
3. Apply sigmoid gates, element‑wise mask and compute left/right tensors.
4. Perform the O(N³·H) contraction
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   with a custom batched GEMM kernel written in Triton.
5. Apply the output LayerNorm, output gate and final linear projection.
All heavy arithmetic is done in fp16; the final result is returned in the
original dtype (float32 by default).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr,               # *fp16   [batch, N, N]
    right_ptr,              # *fp16   [batch, N, N]
    out_ptr,                # *fp16   [batch, N, N]
    N: tl.constexpr,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_j,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: out = left @ rightᵀ
    left, right: (batch, N, N) float16
    out: (batch, N, N) float16
    """
    pid = tl.program_id(0)

    # number of blocks per output matrix
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    # batch index and tile indices inside the batch
    batch_idx = pid // blocks_per_batch
    pid_in_batch = pid % blocks_per_batch
    pid_m = pid_in_batch // num_blocks_n
    pid_n = pid_in_batch % num_blocks_n

    # offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers to the current blocks of A and Bᵀ
        a_ptrs = left_ptr + batch_idx * stride_left_batch \
                 + offs_m[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k
        b_ptrs = right_ptr + batch_idx * stride_right_batch \
                 + offs_k[:, None] * stride_right_k \
                 + offs_n[None, :] * stride_right_j

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)  # accumulate in fp32

    # write back the result
    c_ptrs = out_ptr + batch_idx * stride_out_batch \
             + offs_m[:, None] * stride_out_i \
             + offs_n[None, :] * stride_out_j
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) (same dtype as input_tensor).
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device).to(torch.float32)
    norm_bias = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # -----------------------------------------------------------------
    # 2. Fuse all linear projections (left/right proj + gates + out_gate)
    #    using a single GEMM:   Y = X @ W_fusedᵀ
    # -----------------------------------------------------------------
    # Gather and cat the five weight matrices (no bias)
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float32)      # (H, C)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float32)    # (H, C)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float32)      # (H, C)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float32)    # (H, C)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float32)        # (H, C)

    fused_weight = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0
    )  # shape (5*H, C)

    B, N, _, _ = x.shape
    M = B * N * N
    x_flat = x.view(M, dim)                     # (M, C)

    # Single GEMM (float32)
    proj = F.linear(x_flat, fused_weight)       # (M, 5*H)

    H = hidden_dim
    left_proj = proj[:, :H]
    left_gate = proj[:, H:2*H]
    right_proj = proj[:, 2*H:3*H]
    right_gate = proj[:, 3*H:4*H]
    out_gate = proj[:, 4*H:5*H]

    # Sigmoid gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Apply mask if present
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float32).unsqueeze(-1)   # (B,N,N,1)
    else:
        mask_fp = None

    # Compute left and right tensors (float32)
    left = left_proj * left_gate
    right = right_proj * right_gate
    if mask_fp is not None:
        left = left * mask_fp
        right = right * mask_fp

    # Reshape back to (B, N, N, H)
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # -----------------------------------------------------------------
    # 3. Core contraction: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM in fp16.
    # -----------------------------------------------------------------
    # Cast to fp16 for the heavy O(N³) part
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Permute to (B, H, N, N) and merge batch & head dimensions
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()

    batch = B * H
    left_flat = left_perm.view(batch, N, N)
    right_flat = right_perm.view(batch, N, N)

    out_flat = torch.empty_like(left_flat)  # fp16

    # Strides (in elements)
    l_str0, l_str1, l_str2 = left_flat.stride()
    r_str0, r_str1, r_str2 = right_flat.stride()
    o_str0, o_str1, o_str2 = out_flat.stride()

    # Kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        l_str0,
        l_str1,
        l_str2,
        r_str0,
        r_str2,   # stride over k (original N)
        r_str1,   # stride over j (original K)
        o_str0,
        o_str1,
        o_str2,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # -----------------------------------------------------------------
    # 4. Output LayerNorm, output gate and final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # out_gate was computed in float32 earlier; reshape to match out
    out_gate = out_gate.view(B, N, N, H)
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)   # (C, H)
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the input dtype if necessary
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```

"""
Optimized outgoing TriMul forward pass using Triton.

Algorithm
---------
1. Input LayerNorm (float32) for numerical stability.
2. Fuse the five linear projections (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a single GEMM: Y = X @ W_fusedᵀ.
   Split Y, apply sigmoid to the three gates and compute
   left = left_proj * left_gate, right = right_proj * right_gate.
   Optionally multiply by the pair‑wise mask.
3. Cast left/right to fp16 and perform the O(N³·H) contraction
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] with a custom
   batched GEMM kernel written in Triton (uses Tensor Cores).
4. Apply the output LayerNorm, the output gate and the final linear
   projection back to the original dimensionality.
All heavy arithmetic (the cubic contraction) is executed in fp16;
the final result is returned in the input dtype (float32 by default).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr,               # *fp16  [batch, N, N]
    right_ptr,              # *fp16  [batch, N, N]
    out_ptr,                # *fp16  [batch, N, N]
    N: tl.int32,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_j,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM that computes:
        out = left @ rightᵀ
    where left, right, out are (batch, N, N) tensors of type fp16.
    """
    pid = tl.program_id(0)

    # number of 2‑D blocks per output matrix
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    # derive batch index and tile coordinates inside the batch
    batch_idx = pid // blocks_per_batch
    pid_in_batch = pid % blocks_per_batch
    pid_m = pid_in_batch // num_blocks_n
    pid_n = pid_in_batch % num_blocks_n

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A block (batch, i, k)
        a_ptrs = left_ptr + batch_idx * stride_left_batch \
                 + offs_m[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k
        # Bᵀ block (batch, k, j)
        b_ptrs = right_ptr + batch_idx * stride_right_batch \
                 + offs_k[:, None] * stride_right_k \
                 + offs_n[None, :] * stride_right_j

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)          # fp32 accumulation

    # write the result
    c_ptrs = out_ptr + batch_idx * stride_out_batch \
             + offs_m[:, None] * stride_out_i \
             + offs_n[None, :] * stride_out_j
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    # -----------------------------------------------------------------
    # Unpack
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,C)

    # -----------------------------------------------------------------
    # 2. Fuse all five linear projections into a single GEMM
    # -----------------------------------------------------------------
    # Gather weight matrices (float32)
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float32)   # (H, C)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float32)  # (H, C)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float32)   # (H, C)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float32)  # (H, C)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float32)    # (H, C)

    fused_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0
    )  # (5*H, C)

    B, N, _, _ = x.shape
    M = B * N * N
    x_flat = x.view(M, dim)                     # (M, C)

    # Single GEMM (float32)
    proj = F.linear(x_flat, fused_w)            # (M, 5*H)

    H = hidden_dim
    left_proj = proj[:, :H]
    left_gate = proj[:, H:2*H]
    right_proj = proj[:, 2*H:3*H]
    right_gate = proj[:, 3*H:4*H]
    out_gate = proj[:, 4*H:5*H]

    # Free unused large tensors
    del x_flat, fused_w, proj

    # Gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Mask (float32) – broadcast over the hidden dimension
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float32).unsqueeze(-1)   # (B,N,N,1)
    else:
        mask_fp = None

    # Compute left/right (float32)
    left = left_proj * left_gate
    right = right_proj * right_gate
    if mask_fp is not None:
        left = left * mask_fp
        right = right * mask_fp

    # Reshape to (B,N,N,H)
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # -----------------------------------------------------------------
    # 3. O(N³·H) contraction via a custom Triton GEMM (fp16)
    # -----------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # (B, H, N, N) -> (B*H, N, N)
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()
    batch = B * H
    left_flat = left_perm.view(batch, N, N)
    right_flat = right_perm.view(batch, N, N)

    out_flat = torch.empty_like(left_flat)      # fp16 output buffer

    # Strides (in elements)
    l0, l1, l2 = left_flat.stride()
    r0, r1, r2 = right_flat.stride()
    o0, o1, o2 = out_flat.stride()

    # Kernel launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    nb_m = (N + BLOCK_M - 1) // BLOCK_M
    nb_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch * nb_m * nb_n
    grid = (total_blocks,)

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        l0, l1, l2,
        r0, r2, r1,               # note: r2 is stride over k, r1 over j (transposed)
        o0, o1, o2,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 4. Output LayerNorm, output gate and final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # out_gate was produced in float32; reshape to match out
    out_gate = out_gate.view(B, N, N, H)
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)   # (C, H)
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Return to original dtype if necessary
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
