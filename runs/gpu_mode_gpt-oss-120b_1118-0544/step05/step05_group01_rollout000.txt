"""
custom_kernel: Triton‑accelerated outgoing TriMul forward pass.

Algorithm
---------
1. LayerNorm over the channel dimension `dim`.
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (no bias):
     left  = x @ W_left      , right = x @ W_right
   Gate projections:
     left_gate  = σ(x @ W_left_gate)
     right_gate = σ(x @ W_right_gate)
     out_gate   = σ(x @ W_out_gate)
4. Fuse mask, projection and gate in one Triton kernel (if a mask is present):
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
5. Core reduction (batched matrix multiplication for each hidden slice):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented via torch.bmm on a view (B*H, N, N).
6. LayerNorm over the hidden dimension, element‑wise multiply by `out_gate`,
   and a final linear projection back to `dim`.
7. Cast the result back to the original dtype (usually float32).

Only the mask‑gate fusion (step 4) is written in Triton; the rest
leverages highly‑optimized PyTorch kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# ------------------------------------------------------------------
# Triton kernel: (proj * gate * mask) – mask broadcast over hidden dim
# ------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,            # fp16 * (B*N*N*H,)
    gate_ptr,            # fp16 * (B*N*N*H,)
    mask_ptr,            # fp16 * (B*N*N,)
    out_ptr,             # fp16 * (B*N*N*H,)
    total_elems,         # total number of elements = B*N*N*H
    hidden_dim,          # H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # Each contiguous block of `hidden_dim` entries shares the same mask value
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse linear projection, gating and optional mask using Triton.

    All tensors must be fp16, contiguous and have shapes:
        proj, gate : (B, N, N, H)
        mask       : (B, N, N)   (binary 0/1, broadcast over H)

    Returns
    -------
    torch.Tensor
        Tensor of shape (B, N, N, H) = proj * gate * mask.
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
        - mask         : torch.Tensor [B, N, N] (binary) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys:
            * "dim"          – input channel dimension C
            * "hidden_dim"   – hidden dimension H
            * "nomask" (optional) – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. LayerNorm over the channel dimension `dim`
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy sections
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all linear weights to fp16 (no bias)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B, N, N, hidden_dim)
    # --------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w) # (B,N,N,H)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # --------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Apply mask and gating (fused via Triton if mask present)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16, contiguous and broadcastable
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM across hidden dimension.
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T --> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
