    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # linear projections:
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"]) # shape (B,N,N,hidden_dim)
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # gate projections with sigmoid
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    if mask is not None and not nomask:

        mask_fp = mask.to(x.dtype).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp)

    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)  # shape (B,N,N,hidden_dim)

def gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, total_elems, hidden_dim, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < total_elems

    # compute mask index
    mask_idx = offsets // hidden_dim

    # load
    proj = tl.load(proj_ptr + offsets, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offsets, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # if mask missing? we always pass mask.

    out = proj * gate * m

    tl.store(out_ptr + offsets, out, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1: layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast to half
    x = x.to(torch.float16)

    # Cast weights to half
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # 2: linear projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)  # shape (B,N,N,hidden_dim)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # 3: gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # 4: apply mask and gate
    if mask is not None and not nomask:
        mask_half = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_half)
        right = apply_gate_and_mask(right_proj, right_gate, mask_half)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 5: core reduction using batched GEMM
    B, N, _, H = left.shape  # left shape (B,N,N,H)

    # reshape to (B*H,N,N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched multiplication: left_flat @ right_flat.transpose(1,2)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 6: all out layernorm, multiplication with out_gate, and final linear
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)

    # 7: cast back to original dtype
    out = out.to(input_tensor.dtype)

    return out

def apply_gate_and_mask(proj, gate, mask, block_size=65536):
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = ( (total + block_size - 1) // block_size, )

    mask_flat = mask.view(-1)
    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑normalize the 4‑D input (B,N,N,C) using the provided
   `norm.weight` / `norm.bias`.
2. Cast to float16 for the compute‑intensive part.
3. Linear projections (`left_proj`, `right_proj`) and gate projections
   (`left_gate`, `right_gate`, `out_gate`) are performed with
   `torch.nn.functional.linear`. Gate values are passed through a sigmoid.
4. If a mask is present, a tiny Triton kernel fuses the element‑wise
   multiplication `proj * gate * mask` (broadcasting the mask over the
   hidden dimension H).  This avoids extra kernel launches and memory traffic.
5. Core reduction:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is exactly a batched matrix multiplication for each head `h`.
   We reshape the tensors to (B*H, N, N) and use `torch.bmm`,
   letting cuBLAS handle the heavy work efficiently.
6. Apply a second LayerNorm (`to_out_norm`), multiply by the output gate,
   and linearly project back to the original channel dimension.
7. Cast the result back to the original dtype (usually float32).

Only the mask‑gate fusion is written in Triton; the bulk of the work
leverages cuBLAS via `torch.bmm`, which is optimal on modern GPUs.
"""

import torch

import triton
import triton.language as tl


@triton.jit
def gate_mask_kernel(
    proj_ptr,          # *[total_elems]   fp16 / fp32
    gate_ptr,          # *[total_elems]   fp16 / fp32
    mask_ptr,          # *[mask_elems]    fp16 / fp32
    out_ptr,           # *[total_elems]   fp16 / fp32
    total_elems,       # int
    hidden_dim,        # int
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse element‑wise projection, gate and mask:
        out = proj * gate * mask
    `proj` and `gate` have shape (B,N,N,H) flattened,
    `mask` has shape (B,N,N) flattened.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # each consecutive `hidden_dim` elements share the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 65536) -> torch.Tensor:
    """
    Fuse projection, gate and mask using Triton.

    Parameters
    ----------
    proj : torch.Tensor
        Tensor of shape (B, N, N, H), dtype float16/float32, contiguous.
    gate : torch.Tensor
        Same shape as `proj`.
    mask : torch.Tensor
        Tensor of shape (B, N, N), same dtype as `proj`, contiguous.
    block_size : int
        Number of elements processed per kernel launch.

    Returns
    -------
    torch.Tensor
        Tensor of shape (B, N, N, H) = proj * gate * mask.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    # flatten mask (B,N,N) -> (B*N*N,)
    mask_flat = mask.view(-1)

    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape (B, N, N, C)
        - mask         : torch.Tensor of shape (B, N, N) or None
        - weights      : dict of torch.Tensors with the model parameters
        - config       : dict containing at least "dim", "hidden_dim", and optionally "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape (B, N, N, C) (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to float16 for compute‑intensive stages
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all weights to float16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B,N,N,dim) -> (B,N,N,hidden_dim)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 6. Fuse mask and gate (Triton) or simple elementwise multiply
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask shares dtype and is contiguous
        mask_fp = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core O(N³·H) reduction – batched GEMM using torch.bmm
    #    out_h = left_h @ right_hᵀ   for each (b, h)
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # reshape to (B*H, N, N) for batched matrix multiplication
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matmul: (B*H, N, N) @ (B*H, N, N)ᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
