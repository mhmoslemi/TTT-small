"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input (B,N,N,C) → FP16.
2. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) are performed with torch.nn.functional.linear
   (cuBLAS) on FP16 tensors.
3. Apply mask (if present) and element‑wise gates (simple torch ops).
4. Core O(N³·H) reduction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each (b,h) pair:
       out_{bh} = left_{bh} @ right_{bh}ᵀ
   Implemented with a custom Triton kernel that tiles the matrices
   (BLOCK_M×BLOCK_N tiles, BLOCK_K inner dimension) and accumulates
   in FP32, storing the result in FP16.
5. Output layer‑norm, output‑gate and final linear projection.
6. Cast back to the original dtype (float32).

The Triton kernel fuses the entire batched GEMM and eliminates the
intermediate torch.bmm, yielding a substantial speed‑up on the
tested configurations.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul of (B*H, N, N) @ (B*H, N, N).T
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    left_ptr,          # *[BH, N, N]  fp16
    right_ptr,         # *[BH, N, N]  fp16
    out_ptr,           # *[BH, N, N]  fp16
    N,                 # int: sequence length
    stride_l_bh, stride_l_i, stride_l_k,
    stride_r_bh, stride_r_i, stride_r_k,
    stride_o_bh, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(2)        # batch*head index
    pid_m = tl.program_id(0)         # tile row
    pid_n = tl.program_id(1)         # tile col

    # tile start offsets
    i0 = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j0 = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out-of-bounds rows / columns
    mask_i = i0 < N
    mask_j = j0 < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k0 < N

        # pointers for left (i,k) and right (j,k)
        left_offs = (
            pid_bh * stride_l_bh
            + i0[:, None] * stride_l_i
            + k0[None, :] * stride_l_k
        )
        right_offs = (
            pid_bh * stride_r_bh
            + j0[:, None] * stride_r_i
            + k0[None, :] * stride_r_k
        )

        # load tiles with masking
        left = tl.load(
            left_ptr + left_offs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        right = tl.load(
            right_ptr + right_offs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # dot product, note that right is transposed (B,N,K) -> (K,N)
        acc += tl.dot(left, right, trans_b=True)

    # store result
    out_offs = (
        pid_bh * stride_o_bh
        + i0[:, None] * stride_o_i
        + j0[None, :] * stride_o_j
    )
    mask_ij = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptr + offs := out_offs, acc.to(tl.float16), mask=mask_ij)


def triton_batch_matmul(left: torch.Tensor, right: torch.Tensor,
                       BLOCK_M=64, BLOCK_N=64, BLOCK_K=32) -> torch.Tensor:
    """
    Batched matrix multiplication using Triton:
        out = left @ right.T   (both shapes (B,H,N,N))
    Returns tensor of same dtype (fp16) and shape (B,H,N,N).
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, N, _ = left.shape
    BH = B * H

    # reshape to (BH, N, N)
    left_flat = left.view(BH, N, N)
    right_flat = right.view(BH, N, N)
    out_flat = torch.empty_like(left_flat)

    # strides
    stride_l_bh = left_flat.stride(0)
    stride_l_i = left_flat.stride(1)
    stride_l_k = left_flat.stride(2)

    stride_r_bh = right_flat.stride(0)
    stride_r_i = right_flat.stride(1)
    stride_r_k = right_flat.stride(2)

    stride_o_bh = out_flat.stride(0)
    stride_o_i = out_flat.stride(1)
    stride_o_j = out_flat.stride(2)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        BH,
    )
    batch_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_l_bh, stride_l_i, stride_l_k,
        stride_r_bh, stride_r_i, stride_r_k,
        stride_o_bh, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, H, N, N)
    out = out_flat.view(B, H, N, N)
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    data = (input_tensor, mask, weights, config)
    Returns: Tensor [bs, seq_len, seq_len, dim] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute intensive stages
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all relevant weights to FP16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B,N,N,dim) -> (B,N,N,hidden_dim)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gates + optional mask
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * left_gate * mask_fp16
        right = right * right_gate * mask_fp16
    else:
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 6. Core O(N³·H) reduction using Triton batched GEMM
    #    left, right: (B,N,N,H) → permute to (B,H,N,N)
    # --------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    out = triton_batch_matmul(left_perm, right_perm)   # (B, H, N, N)

    # permute back to (B, N, N, H)
    out = out.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Output LayerNorm, output‑gate and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 8. Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
