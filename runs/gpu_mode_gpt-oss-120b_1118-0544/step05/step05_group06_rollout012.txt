def gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, total_elems, hidden_dim, BLOCK_SIZE: tl.constexpr):

    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # compute mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=mask)

Highly‑optimized Triton‑accelerated forward pass for the outgoing TriMul
operator (AlphaFold3 style).

Algorithm
---------
1. Layer‑normalize the 4‑D input (B, N, N, C).
2. Cast to FP16 for the compute‑intensive stages.
3. Fuse the five linear projections (left_proj, right_proj, left_gate,
   right_gate, out_gate) into two matrix multiplications by concatenating
   the corresponding weight matrices.
4. Split the results, apply sigmoid to the gate tensors.
5. Fuse element‑wise multiplication of projection, gate and (optional)
   mask using a small Triton kernel.
6. Core reduction:
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   which is a batched GEMM (left @ rightᵀ) for each hidden channel.
   We reshape to (B·H, N, N) and use torch.bmm (cuBLAS) on FP16.
7. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate and project back to the original channel dimension.
8. Cast the result back to the input dtype (usually FP32).

The Triton kernel is used for the element‑wise proj * gate * mask
fusion; all heavy linear algebra is performed in FP16 for speed.
"""

    total_elems,             # i64
    hidden_dim,              # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # each hidden dimension shares the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 32768) -> torch.Tensor:
    """
    Element‑wise ``proj * gate * mask`` where ``mask`` has shape (B, N, N) and
    is broadcast over the hidden dimension H.
    All tensors must be FP16 and contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],   # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights: dict containing all model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Prepare concatenated weight matrices (FP16)
    # ------------------------------------------------------------------
    # Linear projections (left + right)
    proj_weight = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(torch.float16).contiguous()                # (2*H, dim)

    # Gate projections (left_gate, right_gate, out_gate)
    gate_weight = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()               # (3*H, dim)

    # Output projection & norms
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Fuse linear projections (B*N*N, dim) -> (B*N*N, 2*H) and gates
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)

    # Projections
    proj = F.linear(x_flat, proj_weight)         # (B*N*N, 2*H)
    left_proj = proj[:, :hidden_dim].reshape(B, N, N, hidden_dim)
    right_proj = proj[:, hidden_dim:].reshape(B, N, N, hidden_dim)

    # Gates (sigmoid)
    gates = F.linear(x_flat, gate_weight)        # (B*N*N, 3*H)
    left_gate = torch.sigmoid(gates[:, :hidden_dim]).reshape(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(gates[:, hidden_dim:2*hidden_dim]).reshape(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(gates[:, 2*hidden_dim:]).reshape(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5. Apply mask & gate (Triton kernel) or just gate
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16 and contiguous, shape (B, N, N)
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core O(N³·H) reduction: batched GEMM (left @ rightᵀ)
    #    Reshape to (B*H, N, N) and use torch.bmm (cuBLAS) on FP16.
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8. Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    out = out.to(dtype_in)
    return out
