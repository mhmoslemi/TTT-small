def mask_apply_kernel(
    x_ptr, mask_ptr, out_ptr,
    dim, total_elems,
    BLOCK_SIZE: tl.constexpr

    pid = tl.program_id(0)

    out = x * m
    tl.store(out_ptr + offs, out, mask=mask)

"""TriMul (outgoing) forward pass – Triton‑fused mask.

The kernel applies a per‑pair mask to the half‑precision input tensor
before the linear projections.  The heavy O(N³·H) reduction is performed
as a strided‑batched GEMM (torch.bmm) on FP16 tensors, which uses
Tensor‑cores on the B200.  All other operations (LayerNorm, linear
projections, gates) are executed with PyTorch but in FP16.  The final
output is cast back to the input dtype (float32)."""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: apply mask (broadcast over the last dim) to a 4‑D tensor
# ----------------------------------------------------------------------
@triton.jit
def _mask_apply_kernel(
    x_ptr,            # fp16 * (B*N*N*C,)
    mask_ptr,         # fp16 * (B*N*N,)
    out_ptr,          # fp16 * (B*N*N*C,)
    dim,              # i32  – size of the last dimension C
    total_elems,      # i32  – total number of elements in x
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # each group of `dim` consecutive elements shares the same mask entry
    mask_idx = offs // dim

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # 1.0 = no masking
    out = x * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_mask(x: torch.Tensor, mask: torch.Tensor, block_size: int = 8192) -> torch.Tensor:
    """
    Fuse ``x * mask[...,None]`` using Triton.

    Args:
        x:    Tensor of shape [B, N, N, C], dtype float16, contiguous.
        mask: Tensor of shape [B, N, N], dtype float16, contiguous.
        block_size: number of elements processed per program (must be <= 2**15).

    Returns:
        Tensor with the same shape & dtype as ``x``.
    """
    assert x.is_contiguous() and mask.is_contiguous()
    total = x.numel()
    out = torch.empty_like(x)
    grid = (triton.cdiv(total, block_size),)
    _mask_apply_kernel[grid](
        x,
        mask,
        out,
        x.shape[-1],
        total,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict containing:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # Input LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Apply mask before the two projection linear layers (Triton)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        x_masked = _apply_mask(x, mask_fp16)  # shape [B,N,N,dim]
    else:
        x_masked = x

    # ------------------------------------------------------------------
    # Cast model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x_masked, left_proj_w)   # [B,N,N,H]
    right_proj = torch.nn.functional.linear(x_masked, right_proj_w)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # Gates (no mask)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # Apply gates
    # ------------------------------------------------------------------
    left  = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·H) reduction – batched GEMM over the hidden dimension
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # (B,H,N,N) → view as (B*H,N,N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_reshape  = left_perm.view(B * H, N, N)
    right_reshape = right_perm.view(B * H, N, N)

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    out_reshape = torch.bmm(left_reshape, right_reshape.transpose(1, 2))

    # Reshape back to (B,N,N,H)
    out = out_reshape.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Second LayerNorm + output gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # Cast to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
