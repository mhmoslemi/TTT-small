    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # each program instance processes a specific (b,h) pair and block tile (i0,j0)
    pid = tl.program_id(0)  # combined batch*head
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block column

    # compute indices
    b = pid // H
    h = pid % H

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

"""
TriMul (outgoing) forward pass – Triton implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor.
2. Cast to fp16 for compute‑intensive stages.
3. Linear projections (left / right) and gates (left_gate, right_gate, out_gate)
   are performed with torch.nn.functional.linear (cuBLAS) on fp16 tensors.
4. If a mask is present, a tiny Triton kernel fuses:
       proj * gate * mask   (broadcasting the mask over the hidden dimension).
5. Core O(N³·H) reduction:
   For each batch and hidden head we compute
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication left @ rightᵀ.
   This is realised with a custom Triton kernel that tiles the
   (i, j) output plane (BLOCK_M×BLOCK_N) and iterates over K
   in blocks of size BLOCK_K, accumulating in fp32.
6. Output LayerNorm, multiplicative out‑gate and final linear projection.
7. Cast back to the original dtype (float32).

The kernel works for any batch size ≤2, any hidden dimension,
and sequence lengths up to 1024 as required by the test suite.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. Gate‑mask fusion (optional)
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    proj_ptr,          # *[total_elems]   fp16
    gate_ptr,          # *[total_elems]   fp16
    mask_ptr,          # *[mask_elems]    fp16
    out_ptr,           # *[total_elems]   fp16
    total_elems,       # int
    hidden_dim,        # int
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # each hidden element repeats `hidden_dim` times in the flattened view
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(out_ptr + offs, proj * gate * m, mask=mask)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Element‑wise: proj * gate * mask   (all fp16)
    proj, gate : (B, N, N, H)  contiguous
    mask       : (B, N, N)    contiguous
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.view(-1)  # (B*N*N,)
    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# 2. Core TriMul kernel (left @ rightᵀ) – batched GEMM in Triton
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for left  (B, i, k, h)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right (B, j, k, h) – we will treat it as (K, N) inside the kernel
    stride_right_b, stride_right_k, stride_right_j, stride_right_h,
    # strides for out   (B, i, j, h)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)      # combined batch * head index
    pid_m = tl.program_id(1)    # tile row index (i)
    pid_n = tl.program_id(2)    # tile col index (j)

    # --------------------------------------------------------------
    # 0. Decode batch and head from pid
    # --------------------------------------------------------------
    b = pid // H
    h = pid % H

    # --------------------------------------------------------------
    # 1. Tile offsets
    # --------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # --------------------------------------------------------------
    # 2. Accumulator (fp32 for numerical stability)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3. Loop over K dimension in blocks
    # --------------------------------------------------------------
    for k_offset in range(0, N, BLOCK_K):
        # ----------- load A (left) tile: shape (BLOCK_M, BLOCK_K) ----------
        a_ptrs = left_ptr + \
                 b * stride_left_b + h * stride_left_h + \
                 i0 * stride_left_i + k_offset * stride_left_k + \
                 tl.arange(0, BLOCK_M)[:, None] * stride_left_i + \
                 tl.arange(0, BLOCK_K)[None, :] * stride_left_k
        a_mask = (i0 + tl.arange(0, BLOCK_M)[:, None] < N) & \
                 (k_offset + tl.arange(0, BLOCK_K)[None, :] < N)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)

        # ----------- load B (right) tile transposed:
        #           we need right[j, k] → (K, N) layout for dot ----------
        b_ptrs = right_ptr + \
                 b * stride_right_b + h * stride_right_h + \
                 k_offset * stride_right_k + j0 * stride_right_j + \
                 tl.arange(0, BLOCK_K)[:, None] * stride_right_k + \
                 tl.arange(0, BLOCK_N)[None, :] * stride_right_j
        b_mask = (k_offset + tl.arange(0, BLOCK_K)[:, None] < N) & \
                 (j0 + tl.arange(0, BLOCK_N)[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        # ----------- accumulate partial result ----------
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # --------------------------------------------------------------
    # 4. Store the result (cast back to fp16)
    # --------------------------------------------------------------
    out_ptrs = out_ptr + \
               b * stride_out_b + h * stride_out_h + \
               i0 * stride_out_i + j0 * stride_out_j + \
               tl.arange(0, BLOCK_M)[:, None] * stride_out_i + \
               tl.arange(0, BLOCK_N)[None, :] * stride_out_j
    out_mask = (i0 + tl.arange(0, BLOCK_M)[:, None] < N) & \
               (j0 + tl.arange(0, BLOCK_N)[None, :] < N)
    tl.store(out_ptrs, acc.to(tl.float16), mask=out_mask)


# ----------------------------------------------------------------------
# 3. Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    data = (input_tensor, mask, weights, config)
    Returns: Tensor [bs, seq_len, seq_len, dim] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the compute‑intensive part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (dim -> hidden_dim)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6. Fuse mask and gate (if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = apply_gate_and_mask(left, left_gate, mask_fp16)
        right = apply_gate_and_mask(right, right_gate, mask_fp16)
    else:
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction via Triton (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # allocate output tensor (fp16)
    out = torch.empty((B, N, N, H), dtype=torch.float16, device=left.device)

    # strides for each tensor (contiguous layout)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tile sizes – chosen to fit comfortably in shared memory on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    trimul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_k, stride_right_j, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)

    return out
