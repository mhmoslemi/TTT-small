"""
TriMul (outgoing) – Triton‑accelerated core reduction.

Algorithm
---------
1. Layer‑norm the input tensor (B, N, N, C) in FP32.
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections and gates (left/right/out) are performed with
   torch.nn.functional.linear (cuBLAS) on FP16.
4. Optional mask is applied element‑wise.
5. Core O(N³·H) reduction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication LEFT @ RIGHTᵀ.
   The reduction is carried out by a custom Triton kernel that
   processes a (BLOCK_M × BLOCK_N) tile of the output for each
   (batch * head) pair, looping over the K dimension in tiles.
6. Output LayerNorm, multiplication by the output gate and a final
   linear projection back to the original channel dimension.
7. Cast the result back to the input dtype (typically FP32).

Only the reduction (step 5) is implemented in Triton; the rest uses
high‑performance PyTorch primitives.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the TriMul reduction
# ----------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    left_ptr,          # *[M, N, N]   fp16
    right_ptr,         # *[M, N, N]   fp16 (already transposed on the last two dims)
    out_ptr,           # *[M, N, N]   fp16
    M, N,              # int
    stride_lm, stride_li, stride_lk,
    stride_rm, stride_rk, stride_rj,
    stride_om, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_batch = tl.program_id(0)          # which (batch * head) we are processing
    pid_m = tl.program_id(1)              # tile row
    pid_n = tl.program_id(2)              # tile col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for the current tile
        left_tile_ptr = left_ptr + pid_batch * stride_lm \
                                   + offs_m[:, None] * stride_li \
                                   + offs_k[None, :] * stride_lk
        right_tile_ptr = right_ptr + pid_batch * stride_rm \
                                     + offs_k[:, None] * stride_rk \
                                     + offs_n[None, :] * stride_rj

        # load with masking
        left = tl.load(left_tile_ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)
        right = tl.load(right_tile_ptr,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        # dot product accumulates in FP32
        acc += tl.dot(left, right)

    # write out (store as FP16)
    out = acc.to(tl.float16)
    out_tile_ptr = out_ptr + pid_batch * stride_om \
                               + offs_m[:, None] * stride_oi \
                               + offs_n[None, :] * stride_oj
    tl.store(out_tile_ptr,
             out,
             mask=mask_m[:, None] & mask_n[None, :])

            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask        : torch.Tensor [B, N, N] (bool/float) or None
            - weights     : dict of model parameters
            - config      : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------- #
    # 1. Input LayerNorm (FP32)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------------- #
    # 2. Cast to FP16 for compute‑intensive stages
    # --------------------------------------------------------------------- #
    x = x.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 3. Cast all relevant weights to FP16
    # --------------------------------------------------------------------- #
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------------- #
    # 4. Linear projections (B, N, N, hidden_dim)
    # --------------------------------------------------------------------- #
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)  # (B, N, N, H)

    # --------------------------------------------------------------------- #
    # 5. Gate projections + sigmoid
    # --------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # --------------------------------------------------------------------- #
    # 6. Apply mask (if any) and combine with gates
    # --------------------------------------------------------------------- #
    if mask is not None and not nomask:
        # mask: (B, N, N) -> (B, N, N, 1) for broadcasting
        mask_fp = mask.to(torch.float16).unsqueeze(-1)
        left  = left_proj * mask_fp * left_gate
        right = right_proj * mask_fp * right_gate
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------------- #
    # 7. Core reduction: batched GEMM (left @ rightᵀ) via Triton
    # --------------------------------------------------------------------- #
    B, N, _, H = left.shape  # (B, N, N, H)

    # left  : (B, N, N, H) -> (B, H, N, N)   (row = i, col = k)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    # right : (B, N, N, H) -> (B, H, N, N)   (row = k, col = j) -> transpose on the last two dims
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # flatten batch and head dimensions
    M = B * H
    left_flat  = left_perm.view(M, N, N)   # (M, N, N)
    right_flat = right_perm.view(M, N, N)  # (M, N, N) – already transposed for GEMM
    out_flat   = torch.empty_like(left_flat)  # (M, N, N)

    # strides (in elements)
    stride_lm, stride_li, stride_lk = left_flat.stride()
    stride_rm, stride_rk, stride_rj = right_flat.stride()
    stride_om, stride_oi, stride_oj = out_flat.stride()

    # block sizes – tuned for typical N (<= 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        M,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M,
        N,
        stride_lm,
        stride_li,
        stride_lk,
        stride_rm,
        stride_rk,
        stride_rj,
        stride_om,
        stride_oi,
        stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 8. Output LayerNorm, output‑gate and final linear projection
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------------- #
    # 9. Cast back to the original dtype
    # --------------------------------------------------------------------- #
    out = out.to(input_tensor.dtype)
    return out
