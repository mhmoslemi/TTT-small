def _apply_gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elts, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elts
    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)

    if HAS_MASK:
        # compute mask index per hidden dim group
        # each element off = index in flattened (B * N * N * H)
        # offset // hidden_dim gives index into mask dimension flattened (B*N*N)
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype
    # retrieve config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    # layer norm
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    # cast to fp16 for speed
    x = x.to(torch.float16)
    # linear projections
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    # compute left and right
    left = torch.nn.functional.linear(x, left_proj_w)  # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_w)

    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))
    # apply mask & gate on left/right using Triton kernel if mask and not nomask
    if mask is not None and not nomask:
        # ensure mask float16 contiguous
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_fp.view(-1)  # shape B*N*N

            BLOCK_SIZE=BLOCK_SIZE,

        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
        # free gates? not needed
        # do not need left_gate, right_gate any longer; can delete
    else:

        right = right * right_gate

    HAS_MASK: tl.constexpr,

    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input (float32 for stability).
2. Cast to fp16 for the compute‑heavy path.
3. Linear projections:
   * left  = x @ left_proj_weight
   * right = x @ right_proj_weight
   * left_gate  = sigmoid(x @ left_gate_weight)
   * right_gate = sigmoid(x @ right_gate_weight)
   * out_gate   = sigmoid(x @ out_gate_weight)
4. Fuse element‑wise mask (if present) and gate multiplications with a small
   Triton kernel that updates ``left`` and ``right`` in‑place.
5. Core contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM (L @ Rᵀ) for each hidden channel.
6. Output layer‑norm, apply ``out_gate`` and a final linear projection back
   to the original dimension.
7. Cast back to the input dtype and return.

Only the mask‑gate fusion is written in Triton; the rest leverages
high‑performance PyTorch kernels (layer‑norm, linear, batched GEMM).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fusion_kernel(
    left_ptr,               # [B*N*N*H] left projection (fp16)
    right_ptr,              # [B*N*N*H] right projection (fp16)
    left_gate_ptr,          # [B*N*N*H] left gate (fp16)
    right_gate_ptr,         # [B*N*N*H] right gate (fp16)
    mask_ptr,               # [B*N*N]   mask (fp16), broadcast over H
    total_elems,            # scalar: number of elements in the flattened tensors
    hidden_dim,             # scalar: H
    HAS_MASK: tl.constexpr, # compile‑time bool
    BLOCK_SIZE: tl.constexpr # compile‑time int
):
    """In‑place fusion of:
           left  = left  * left_gate  * mask
           right = right * right_gate * mask
       (mask is broadcast over the hidden dimension).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load values (zero for out‑of‑bounds)
    left      = tl.load(left_ptr + offs,      mask=mask, other=tl.float16(0.0))
    right     = tl.load(right_ptr + offs,     mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate= tl.load(right_gate_ptr + offs,mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        mask_idx = offs // hidden_dim               # index into mask (B*N*N)
        mask_val = tl.load(mask_ptr + mask_idx,
                           mask=mask,
                           other=tl.float16(1.0))
        left  = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs,  left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, mask is ignored

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------ #
    # Unpack arguments
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------ #
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # ------------------------------------------------------------------ #
    # 2. Cast to fp16 for the heavy path
    # ------------------------------------------------------------------ #
    x = x.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 3. Linear projections (bias‑less) – all in fp16
    # ------------------------------------------------------------------ #
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left  = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)  # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------ #
    # 4. Fuse mask (if present) and gate multiplications via Triton kernel
    # ------------------------------------------------------------------ #
    if mask is not None and not nomask:
        # mask -> fp16, contiguous
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten everything to 1‑D
        left_flat       = left.contiguous().view(-1)
        right_flat      = right.contiguous().view(-1)
        left_gate_flat  = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat       = mask_fp.view(-1)                     # (B*N*N,)

        total = left_flat.numel()
        BLOCK_SIZE = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fusion_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # reshape back
        left  = left_flat.view_as(left)
        right = right_flat.view_as(right)
        # gates are no longer needed – allow GC
        del left_gate, right_gate
    else:
        # Simple element‑wise multiplication when no mask is applied
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------ #
    # 5. Core O(N³·H) contraction via batched GEMM (L @ Rᵀ)
    # ------------------------------------------------------------------ #
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched matmul
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (uses cuBLAS tensor cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 6. Output LayerNorm, gate, and final linear projection
    # ------------------------------------------------------------------ #
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (still fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------ #
    # 7. Cast back to original dtype
    # ------------------------------------------------------------------ #
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
