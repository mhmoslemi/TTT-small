@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH, M, N, K,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_j, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    
    # Compute start indices
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N
    
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # Pointers base for this batch
    left_batch_offset = batch_id * stride_left_batch
    right_batch_offset = batch_id * stride_right_batch
    
    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        # compute valid range for k block
        mask_k = (k_start + tl.arange(0, BLOCK_K)) < K
        
        # Load left tile
        left_ptrs = left_ptr + left_batch_offset + offs_i[:, None] * stride_left_i + (k_start + tl.arange(0, BLOCK_K))[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        
        # Load right tile transposed (k as rows, j as cols)
        # right originally shape (batch, N, N) with indices (j,k)
        # We want right_T shape (k,j)
        right_ptrs = right_ptr + right_batch_offset + (k_start + tl.arange(0, BLOCK_K))[:, None] * stride_right_i + offs_j[None, :] * stride_right_k
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        
        # dot
        acc += tl.dot(left, right)  # left: (M,K), right: (K,N) -> (M,N)
    
    # Store result
    out_ptrs = out_ptr + batch_id * stride_out_batch + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)

    # cast weights to half
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (B, N, N, C) (FP16).
2. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) are computed with torch.nn.functional.linear
   on FP16 tensors.
3. Optional mask * gate fusion is performed element‑wise (torch).
4. Core reduction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication of shape (B·H, N, N) ×
   (B·H, N, N)ᵀ.  A custom Triton kernel tiles the i/j dimensions
   (BLOCK_M×BLOCK_N) and loops over k in BLOCK_K tiles, using
   `tl.dot` (FP16×FP16→FP32) to exploit tensor cores.
5. Output layer‑norm, multiplication by the output gate and a final
   linear projection back to the original channel dimension.
6. Cast the result back to the input dtype (typically float32).

The implementation keeps all heavy compute in FP16 while using
float32 accumulation for numerical stability.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    BATCH, M, N, K,                    # sizes
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ Bᵀ
    A : (batch, M, K)   B : (batch, N, K)   C : (batch, M, N)
    A is indexed as (i,k) and B as (j,k); we load B transposed.
    """
    batch_id = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Offsets for the output tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # Base offsets for this batch
    a_batch_off = batch_id * stride_a_batch
    b_batch_off = batch_id * stride_b_batch
    c_batch_off = batch_id * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        k_range = cur_k + tl.arange(0, BLOCK_K)
        mask_k = k_range < K

        # Load A tile (M x K)
        a_ptrs = A_ptr + a_batch_off + offs_i[:, None] * stride_a_i + k_range[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile transposed (K x N)
        # B is stored as (j, k); we need (k, j)
        b_ptrs = B_ptr + b_batch_off + k_range[:, None] * stride_b_i + offs_j[None, :] * stride_b_k
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Perform the dot product for this K‑tile
        acc += tl.dot(a, b)          # (M,K)·(K,N) → (M,N)

    # Write the result back
    c_ptrs = C_ptr + c_batch_off + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of model weight tensors
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (FP16)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Cast weights to FP16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 4. Gate projections + sigmoid
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 5. Apply mask and gates
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6. Prepare tensors for batched GEMM
    # --------------------------------------------------------------
    B, N, _, H = left.shape                     # (B, N, N, H)
    # Move hidden dim to batch axis for a single GEMM batch
    left = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Collapse batch & hidden into one axis
    left_flat = left.view(B * H, N, N)
    right_flat = right.view(B * H, N, N)

    # Output placeholder
    out_flat = torch.empty_like(left_flat)

    # --------------------------------------------------------------
    # 7. Launch Triton kernel for the O(N³·H) reduction
    # --------------------------------------------------------------
    # Strides in number of elements (not bytes)
    stride_a_batch, stride_a_i, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_i, stride_b_k = right_flat.stride()
    stride_c_batch, stride_c_i, stride_c_j = out_flat.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                 # batch * hidden
        math.ceil(N / BLOCK_M),
        math.ceil(N / BLOCK_N),
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H, N, N, N,
        stride_a_batch, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_i, stride_b_k,
        stride_c_batch, stride_c_i, stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 8. Reshape back to (B, N, N, H)
    # --------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 9. Output LayerNorm, out‑gate and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)

    # --------------------------------------------------------------
    # 10. Cast back to original dtype (usually float32)
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
