def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])
    # cast to half
    x = x.to(torch.float16)

    # cast weights to half
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # linear projections
    left_proj = F.linear(x, left_proj_w)   # shape B,N,N,hidden_dim
    right_proj = F.linear(x, right_proj_w)

    # gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # mask handling
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # core reduction
    B, N, _, H = left.shape
    # reshape for batched matmul
    left_flat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_flat = right.permute(0,3,1,2).contiguous().view(B*H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1,2))
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()

    # output layernorm, gating and linear
    out = F.layer_norm(out,
                       (H,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)

    # cast back to input dtype (likely float32)
    out = out.to(input_tensor.dtype)
    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (B, N, N, C).
2. Cast to FP16 for the compute‑intensive stage.
3. Linear projections (left/right) and corresponding gates are computed with
   `torch.nn.functional.linear`.  The gates are passed through `sigmoid`.
4. A small Triton kernel fuses the element‑wise multiplication of the
   projection, gate and optional mask (broadcasted over the hidden dimension).
5. Core O(N³·H) reduction:
       out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   which is a batched matrix multiplication performed with `torch.bmm`.
6. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate and linearly project back to the original channel dimension.
7. Cast the result back to the input dtype (float32).

Only the element‑wise gate‑mask fusion is written in Triton; all large
matrix multiplications rely on cuBLAS (via `torch.bmm`) for maximal
throughput on the B200 GPU.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def proj_gate_mask_kernel(
    proj_ptr,          # *[total_elems]   fp16
    gate_ptr,          # *[total_elems]   fp16
    mask_ptr,          # *[mask_elems]    fp16
    out_ptr,           # *[total_elems]   fp16
    total_elems,       # i32
    hidden_dim,        # i32
    BLOCK_SIZE: tl.constexpr,
):
    """
    out = proj * gate * mask   (mask broadcast over hidden_dim)
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # map linear index to mask index (mask has no hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def apply_gate_and_mask(proj: torch.Tensor,
                        gate: torch.Tensor,
                        mask: torch.Tensor,
                        block_size: int = 8192) -> torch.Tensor:
    """
    Element‑wise multiplication of projection, gate and mask (broadcast over
    hidden dimension).  All tensors must be FP16, contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    # flatten mask to 1‑D (B·N·N)
    mask_flat = mask.view(-1)

    grid = (math.ceil(total / block_size),)
    proj_gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],          # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out

        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [bs, seq_len, seq_len, dim]
            - mask:        torch.Tensor [bs, seq_len, seq_len] (may be None)
            - weights:     dict of model weights (see code)
            - config:      dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Cast to FP16 for compute‑intensive part
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Cast all relevant weights to FP16
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4. Linear projections (B,N,N,dim) -> (B,N,N,hidden_dim)
    # -----------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 6. Fuse mask and gate (Triton) or simple element‑wise multiply
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 7. Core reduction – batched GEMM: out_h = left_h @ right_hᵀ
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # reshape to (B*H, N, N) for batched matmul
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiplication
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # -----------------------------------------------------------------
    # 9. Cast back to original dtype
    # -----------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
