@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr
):
    pid = tl.program_id(axis=0)  # combined batch * grid_M
    pid_n = tl.program_id(axis=1)  # block columns

    grid_m = tl.cdiv(M, BLOCK_M)
    total_blocks = batch * grid_m
    batch_idx = pid // grid_m
    block_i = pid % grid_m

    # Compute offsets
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute k offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + batch_idx * stride_Ab + offs_i[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=(offs_i[:, None] < M) & (offs_k[None, :] < K), other=0.0)  # shape (BLOCK_M, BLOCK_K)

        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = B_ptr + batch_idx * stride_Bb + offs_k[:, None] * stride_Bk + offs_j[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_j[None, :] < N), other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N) accumulation in float32

    # Store result
    c_ptrs = C_ptr + batch_idx * stride_Cb + offs_i[:, None] * stride_Cm + offs_j[None, :] * stride_Cn
    # Write only valid entries
    mask = (offs_i[:, None] < M) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16) if tl.dtype(C_ptr) == tl.float16 else acc, mask=mask)

                     total_elems, hidden_dim, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # Mask index: each hidden_dim elements share same mask element
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)

def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj, gate, mask, out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_n = tl.program_id(1)

    # Compute grid dimensions
    grid_m = tl.cdiv(M, BLOCK_M)

    # Determine batch index and block row index
    batch_idx = pid // grid_m
    block_i = pid % grid_m

    # Offsets for the block
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out-of-bounds
    mask_i = offs_i < M
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile
        a_ptrs = A_ptr + batch_idx * stride_Ab + offs_i[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (B is (batch, K, N))
        b_ptrs = B_ptr + batch_idx * stride_Bb + offs_k[:, None] * stride_Bk + offs_j[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = C_ptr + batch_idx * stride_Cb + offs_i[:, None] * stride_Cm + offs_j[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    The implementation follows the reference PyTorch code but fuses
    the mask‑gate multiplication and the batch of matrix‑multiplications
    (the O(N³·hidden_dim) core) into custom Triton kernels.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for the compute‑intensive portion
    x = x.to(torch.float16)

    # Cast weights to fp16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # 2. Linear projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, H]
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # 3. Gates
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # 4. Fuse mask and gate (Triton kernel)
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 5. Core reduction: batched GEMM (left @ right^T)
    # Prepare tensors
    # left: (B, N, N, H) -> (B*H, N, N)
    left_reshaped = left.permute(0, 3, 1, 2).contiguous().view(-1, left.shape[1], left.shape[2])
    # right: transpose last two dims before reshaping
    right_t = right.permute(0, 3, 2, 1).contiguous()
    right_reshaped = right_t.view(-1, right_t.shape[2], right_t.shape[3])

    # Allocate output
    out_reshaped = torch.empty_like(left_reshaped)

    batch = left_reshaped.shape[0]
    M = left_reshaped.shape[1]
    K = left_reshaped.shape[2]  # = N
    N = right_reshaped.shape[2]  # = N

    # strides (in elements)
    stride_Ab = left_reshaped.stride(0)
    stride_Am = left_reshaped.stride(1)
    stride_Ak = left_reshaped.stride(2)

    stride_Bb = right_reshaped.stride(0)
    stride_Bk = right_reshaped.stride(1)
    stride_Bn = right_reshaped.stride(2)

    stride_Cb = out_reshaped.stride(0)
    stride_Cm = out_reshaped.stride(1)
    stride_Cn = out_reshaped.stride(2)

    # Block sizes – tuned for 16‑bit matmul on A100/B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = triton.cdiv(M, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)

    # Axis 0 encodes batch * M‑blocks, axis 1 encodes N‑blocks
    grid = (batch * grid_m, grid_n)

    _batched_gemm_kernel[grid](
        left_reshaped,
        right_reshaped,
        out_reshaped,
        batch, M, N, K,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to (B, N, N, H)
    out = out_reshaped.view(left.shape[0], hidden_dim, left.shape[1], left.shape[2])
    out = out.permute(0, 2, 3, 1).contiguous()

    # 6. Second LayerNorm + output gate
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # 7. Final projection back to dim
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to original dtype
    out = out.to(input_tensor.dtype)
    return out

The implementation follows the reference PyTorch code but fuses:
1) element‑wise (proj * gate * mask) using a small Triton kernel,
2) the core O(N³·hidden_dim) reduction (batch of N×N matrix multiplications)
   via a custom batched‑GEMM Triton kernel.

# Kernel: proj * gate * mask   (mask broadcasted over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(out_ptr + offs, proj * gate * mask, mask=valid)

        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to (B, N, N, H)
    out = out_reshaped.view(left.shape[0], hidden_dim, left.shape[1], left.shape[2])
    out = out.permute(0, 2, 3, 1).contiguous()

    # ---------- 8. Second LayerNorm + output gate ----------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ---------- 9. Final linear projection ----------
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, C]

    # ---------- 10. Cast back to original dtype ----------
    out = out.to(input_tensor.dtype)
    return out
```
