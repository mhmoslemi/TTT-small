    B, N, H,

    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Compute program ids
    pid0 = tl.program_id(0)   # combined batch * hidden tile
    pid1 = tl.program_id(1)   # i tile
    pid2 = tl.program_id(2)   # j tile

    # compute number of hidden tiles
    num_hidden_tiles = tl.cdiv(H, BLOCK_H)
    batch_idx = pid0 // num_hidden_tiles
    hidden_tile_idx = pid0 % num_hidden_tiles

    # start offsets for i, j, h
    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N
    h0 = hidden_tile_idx * BLOCK_H

    # define ranges
    i_idx = i0 + tl.arange(0, BLOCK_M)
    j_idx = j0 + tl.arange(0, BLOCK_N)
    h_idx = h0 + tl.arange(0, BLOCK_H)

    # mask for valid i,j,h
    mask_i = i_idx < N
    mask_j = j_idx < N
    mask_h = h_idx < H

    # compute base offsets
    # compute batch offset
    batch_offset = batch_idx * stride_lb

    # compute pointers for left: base offset = batch_offset
    # left[i,k,h] offset: i*stride_li + k*stride_lk + h*stride_lh
    # similarly right[j,k,h] offset: j*stride_rj + k*stride_rk + h*stride_rh

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # loop over k dimension
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k0 = k_tile * BLOCK_K
        k_idx = k0 + tl.arange(0, BLOCK_K)

        # valid k mask
        mask_k = k_idx < N

        # combine masks
        # left load for each i,k,h
        # compute pointer offsets for left
        # We can load using broadcasting:
        # left_ptr_offset = batch_offset + (i_idx[:, None, None] * stride_li) + (k_idx[None, :, None] * stride_lk) + (h_idx[None, None, :] * stride_lh)
        # Similarly for right

        # Load left slice (M, K, H)
        left = tl.load(
            left_ptr + batch_offset

            + k_idx[None, :, None] * stride_lk
            + h_idx[None, None, :] * stride_lh,

            other=0.0
        )

        # Load right slice (N, K, H)
        right = tl.load(
            right_ptr + batch_offset
            + j_idx[:, None, None] * stride_rj
            + k_idx[None, :, None] * stride_rk
            + h_idx[None, None, :] * stride_rh,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],

        # we can loop over kk within BLOCK_K using unroll to reduce overhead.

        # Let's compute using `tl.sum` across K dimension.
        # Multiply left and right elementwise across K and H:
        # left shape (M,K,H), right shape (N,K,H)
        # Compute product = left[:, :, None, :] * right[None, :, :, :] ??? Actually we need shape M,N,K,H.
        # We can broadcast using tl.reshape? Might be easier to loop across K.

        # Loop across k in tile:
        for kk in range(BLOCK_K):
            # check if this kk is valid (k0+kk < N)
            k_valid = mask_k[kk]
            # load slice
            l_slice = left[:, kk, :]   # (M, H)
            r_slice = right[:, kk, :]  # (N, H)
            # convert to float32
            l_slice_f = l_slice.to(tl.float32)
            r_slice_f = r_slice.to(tl.float32)

            # compute outer product: (M,1,H) * (1,N,H) -> (M,N,H)
            # use broadcasting:
            prod = l_slice_f[:, None, :] * r_slice_f[None, :, :]   # shape (M,N,H)
            # accumulate
            acc += prod

    # store accumulator
    # cast to desired dtype (fp16)
    out = acc.to(tl.float16)

    # compute output pointer offsets
    out_ptrs = out_ptr + batch_offset \
        + i_idx[:, None, None] * stride_oi \
        + j_idx[None, :, None] * stride_oj \
        + h_idx[None, None, :] * stride_oh

    tl.store(out_ptrs, out, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device
    dtype = input_tensor.dtype

    # layernorm
    x = torch.nn.functional.layer_norm(input_tensor,
                                      (dim,),
                                      weight=weights["norm.weight"],
                                      bias=weights["norm.bias"])
    x = x.to(torch.float16)

    # cast weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # linear projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)  # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # gates
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # mask & gate fusion via Triton
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguity for kernel
    left = left.contiguous()
    right = right.contiguous()

    # Allocate output buffer
    out = torch.empty_like(left)  # same shape and dtype (fp16)

    # Compute core reduction via Triton kernel
    B, N, _, H = left.shape
    # strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()

    BLOCK_N = 32

    BLOCK_H = 8   # hidden tile

        BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H

    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):

    pid1 = tl.program_id(1)
    pid2 = tl.program_id(2)

    # hidden tile handling
    num_hidden_tiles = tl.cdiv(H, BLOCK_H)
    batch_idx = pid0 // num_hidden_tiles
    hidden_tile = pid0 % num_hidden_tiles

    # compute start indices
    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N
    h0 = hidden_tile * BLOCK_H

    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)
    h = h0 + tl.arange(0, BLOCK_H)

    mask_j = j < N
    mask_h = h < H

"""
TriMul (outgoing) forward pass – Triton‑fused mask‑gate and core reduction.

Algorithm
---------
1. Layer‑normalize the input tensor.
2. Cast to fp16 for the compute‑intensive part.
3. Linear projections (left/right) and their gates are performed in fp16.
4. Fuse projection, gate and optional mask with a tiny Triton kernel
   (`_gate_mask_kernel`). This avoids an extra memory pass.
5. Core O(N³·H) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton kernel (`tri_mul_reduction_kernel`) that
   processes a tile of (i,j) per block and a tile of the hidden dimension
   (h) per program.  The kernel loads vectors of length `BLOCK_H` for each
   (i,k) and (j,k) and accumulates the outer‑product across the reduction
   dimension k.
6. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate, and project back to the original channel dimension.
7. Cast back to the input dtype (float32).

The implementation respects the required interface and uses at least one
Triton kernel, while keeping all heavy compute in fp16 for speed.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (total_proj,)
    gate_ptr,          # fp16 * (total_proj,)
    mask_ptr,          # fp16 * (total_mask,)
    out_ptr,           # fp16 * (total_proj,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – size of hidden dimension (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask is broadcasted over hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(out_ptr + offs, proj * gate * mask, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """Fuse `proj * gate * mask` using Triton."""
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: core TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_reduction_kernel(
    left_ptr,   # fp16 * (B, N, N, H)
    right_ptr,  # fp16 * (B, N, N, H)
    out_ptr,    # fp16 * (B, N, N, H)
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid0 = tl.program_id(0)  # combined batch * hidden‑tile
    pid1 = tl.program_id(1)  # i‑tile
    pid2 = tl.program_id(2)  # j‑tile

    # ----- decode batch & hidden‑tile -----
    num_hidden_tiles = tl.cdiv(H, BLOCK_H)
    batch_idx = pid0 // num_hidden_tiles
    hidden_tile = pid0 % num_hidden_tiles

    # ----- start indices for this tile -----
    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N
    h0 = hidden_tile * BLOCK_H

    # ----- range vectors -----
    i = i0 + tl.arange(0, BLOCK_M)          # (M)
    j = j0 + tl.arange(0, BLOCK_N)          # (N)
    h = h0 + tl.arange(0, BLOCK_H)          # (H)

    mask_i = i < N
    mask_j = j < N
    mask_h = h < H

    # ----- batch offset (same for left/right/out) -----
    batch_offset = batch_idx * stride_l_b

    # ----- accumulator (per (i,j) we keep a vector of length BLOCK_H) -----
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ----- reduction over k -----
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        for kk in range(BLOCK_K):
            k = k_start + kk
            mask_k = k < N

            # Load left vector L[i,k,:] -> shape (M, H)
            left_vec = tl.load(
                left_ptr
                + batch_offset
                + i[:, None] * stride_l_i
                + k * stride_l_k
                + h[None, :] * stride_l_h,
                mask=mask_i[:, None] & mask_h[None, :] & mask_k,
                other=0.0,
            )
            # Load right vector R[j,k,:] -> shape (N, H)
            right_vec = tl.load(
                right_ptr
                + batch_offset
                + j[:, None] * stride_r_j
                + k * stride_r_k
                + h[None, :] * stride_r_h,
                mask=mask_j[:, None] & mask_h[None, :] & mask_k,
                other=0.0,
            )
            # Outer‑product accumulation across hidden dimension
            # left_vec: (M, H) -> (M,1,H)
            # right_vec: (N, H) -> (1,N,H)
            acc += left_vec[:, None, :] * right_vec[None, :, :]

    # ----- store result -----
    out_ptrs = (
        out_ptr
        + batch_offset
        + i[:, None, None] * stride_o_i
        + j[None, :, None] * stride_o_j
        + h[None, None, :] * stride_o_h
    )
    mask_store = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_store)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------- LayerNorm (input) -------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ----------------------- cast weights to fp16 -----------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ----------------------- linear projections ---------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------- gates -----------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------- mask & gate fusion (Triton) --------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguity for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------ core reduction ----------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # fp16 output buffer

    # strides (B,N,N,H) layout
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tuning parameters – keep product BLOCK_M*BLOCK_N <= 1024
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 8   # reduction tile
    BLOCK_H = 8   # hidden‑dimension tile (hidden_dim is a multiple of 8)

    num_hidden_tiles = (H + BLOCK_H - 1) // BLOCK_H
    grid = (
        B * num_hidden_tiles,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    tri_mul_reduction_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_j,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------ cast back to input dtype ----------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
