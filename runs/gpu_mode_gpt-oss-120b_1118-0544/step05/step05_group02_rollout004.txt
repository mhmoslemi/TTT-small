@triton.jit
def matmul_triton_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
    stride_abatch, stride_bbatch, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offset
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute base pointers offset for this batch
    a_batch_ptr = a_ptr + batch_id * stride_abatch
    b_batch_ptr = b_ptr + batch_id * stride_bbatch
    c_batch_ptr = c_ptr + batch_id * stride_cbatch

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # load A and B tiles
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs, mask=offs_m < M and offs_k < K, other=0.0)
        b = tl.load(b_ptrs, mask=offs_k < K and offs_n < N, other=0.0)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m < M) & (offs_n < N))
```

@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid positions
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M x K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B tile (K x N)
        b_ptrs = b_batch + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

Note: Access patterns: Strides for A: stride_a_m = stride between rows (M dimension). stride_a_k = stride between columns (K dimension). For contiguous layout row major: stride_a_m = K, stride_a_k = 1.

For B: shape (batch, K, N) after transposition. stride_b_k = stride between rows (K dimension) = N, stride_b_n = stride between columns (N dimension) = 1.

For C: shape (batch, M, N). stride_c_m = N, stride_c_n = 1.

def proj_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr,
    out_ptr,
    hidden_dim,
    total_elements,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # compute indices:
    # Each element corresponds to (batch * N * N * hidden_dim) linear index.
    # mask index = (linear_index // hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input (B, N, N, C) using the provided LN weights.
2. Cast to fp16 for the compute‑intensive part.
3. Perform linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) with torch.nn.functional.linear (cuBLAS) on fp16.
4. Fuse projection * gate * optional mask via a tiny Triton kernel.
5. Core O(N³·H) reduction:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (B·H independent matmuls
   of size N×N).  Implemented with a custom Triton matmul kernel that
   loops over the K dimension in shared memory and accumulates in fp32.
6. Apply a second LayerNorm, multiply by the output gate and a final
   linear projection back to the original channel dimension.
7. Cast the result back to the input dtype.

Only the gate‑mask fusion and the batched matmul are executed in Triton,
the rest uses highly‑optimized PyTorch primitives.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------
# Triton kernel: elementwise proj * gate * mask
# -------------------------------------------------
@triton.jit
def proj_gate_mask_kernel(
    proj_ptr,   # *[total_elems]  fp16
    gate_ptr,   # *[total_elems]  fp16
    mask_ptr,   # *[mask_elems]   fp16 (broadcasted over hidden_dim)
    out_ptr,    # *[total_elems]  fp16
    total_elems,  # int
    hidden_dim,   # int
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # mask is broadcasted over hidden_dim
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def fuse_proj_gate_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 32768) -> torch.Tensor:
    """
    proj, gate : (B, N, N, H)  fp16 contiguous
    mask       : (B, N, N)     fp16 contiguous
    Returns     proj * gate * mask    (fp16)
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.view(-1)  # (B*N*N,)
    proj_gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# -------------------------------------------------
# Triton kernel: batched matmul A @ B.T  (A,B shapes: (batch, N, N))
# -------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets within the M/N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_batch + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Dot product for the tile
        acc += tl.dot(a, b)

    # Write back result in fp16
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------
# Entry point
# -------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    data = (input_tensor, mask, weights, config)
    Returns: Tensor [bs, seq_len, seq_len, dim] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------
    # 1. Input LayerNorm
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive stages
    # -------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 3. Cast all relevant weights to fp16
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 4. Linear projections (B,N,N,dim) -> (B,N,N,hidden_dim)
    # -------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # -------------------------------------------------
    # 5. Gate projections + sigmoid
    # -------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # -------------------------------------------------
    # 6. Fuse mask and gate (Triton) or simple elementwise multiply
    # -------------------------------------------------
    if mask is not None and not nomask:
        # mask shape (B, N, N) -> broadcast over hidden_dim
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = fuse_proj_gate_mask(left_proj, left_gate, mask_fp16)
        right = fuse_proj_gate_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------------------------------
    # 7. Core O(N³·H) reduction via batched Triton matmul
    # -------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # Rearrange to (batch, H, N, N) and flatten batch*H dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Transpose right on the last two dimensions (equivalent to j,k -> k,j)
    right_t = right_perm.permute(0, 1, 3, 2).contiguous()  # (B, H, N, N)

    # Merge batch and hidden dim
    batch_h = B * H
    left_flat = left_perm.view(batch_h, N, N)
    right_flat = right_t.view(batch_h, N, N)

    # Prepare output buffer
    out_flat = torch.empty_like(left_flat)

    # Strides for the flattened tensors
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(1)  # now K dimension
    stride_b_n = right_flat.stride(2)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Blocking parameters (tuned for typical GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)

    # -------------------------------------------------
    # 9. Cast back to original dtype
    # -------------------------------------------------
    out = out.to(dtype)
    return out
