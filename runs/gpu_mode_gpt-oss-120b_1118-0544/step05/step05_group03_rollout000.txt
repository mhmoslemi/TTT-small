"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1. **LayerNorm** over the channel dimension `dim` (FP32 for stability).
2. Cast to **FP16** for compute‑heavy parts.
3. **Linear projections** (left, right) and **gate projections** (left_gate,
   right_gate, out_gate).  All linear weights are stored in FP16.
4. **Fuse** projection × gate × optional mask with a single Triton kernel.
   The mask (B,N,N) is broadcast over the hidden dimension `H`.
5. **Core reduction**:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented via a batched GEMM: reshape to (B*H, N, N) and call
   `torch.bmm(left, right.T)`.
6. **Output LayerNorm**, multiply by `out_gate`, and final linear projection
   back to the original dimension `dim`.
7. Cast the result back to the input dtype (usually float32).

Only steps 4 (gate‑mask fusion) are performed in Triton; the rest use
high‑performance PyTorch primitives (layer_norm, linear, bmm).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) with mask broadcast
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,          # fp16 * (total_elements,)
    gate_ptr,          # fp16 * (total_elements,)
    mask_ptr,          # fp16 * (mask_elements,)  # B*N*N
    out_ptr,           # fp16 * (total_elements,)
    total_elems,       # int
    hidden_dim,        # int  = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard: only compute within total_elems
    mask_valid = offs < total_elems

    # Compute mask index (each hidden_dim values share same mask)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask_valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask_valid, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=mask_valid)


def _apply_gate_mask(proj: torch.Tensor,
                     gate: torch.Tensor,
                     mask: torch.Tensor,
                     block_size: int = 2048) -> torch.Tensor:
    """
    Fuse proj * gate * mask (mask broadcast over hidden dim) using Triton.

    All tensors must be `torch.float16` and contiguous.
    Shapes:
        proj, gate : (B, N, N, H)
        mask       : (B, N, N)
    Returns:
        tensor of shape (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask        : torch.Tensor of shape [B, N, N] (or None)
            - weights     : dict with the following keys (all float32):
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over channel dimension `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑intensive part
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all linear weights to FP16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B, N, N, H)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # --------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # --------------------------------------------------------------
    # 6. Fuse projection, gating and optional mask
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16, contiguous, values 0/1
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_mask(right_proj, right_gate, mask_fp16)
    else:
        # Simple element‑wise multiplication (mask = 1)
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) where dim0 = batch*hidden
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N).T  -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
