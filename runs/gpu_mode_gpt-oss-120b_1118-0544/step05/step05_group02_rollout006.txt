@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,  # a dimensions: (batch_h, M, K)
    batch_stride_b, stride_bk, stride_bn,  # b dimensions: (batch_h, K, N)
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Compute block indices
    pid_batch = tl.program_id(0)  # batch_hidden index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        a = tl.load(
            a_ptr + pid_batch * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
            mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptr + pid_batch * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
            other=0.0,
        )
        # Multiply-accumulate
        acc += tl.dot(a, b)  # returns (BLOCK_M, BLOCK_N)
    # Store result
    c = acc.to(tl.float16)  # cast back to half
    tl.store(
        c_ptr + pid_batch * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
        c,
        mask=(offs_m < M)[:, None] & (offs_n < N)[None, :],
    )

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,},
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # 2. Cast to FP16
    x = x.to(torch.float16)

    # 3. Load weights to FP16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)

    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # 4. Linear projections and gates
    left_proj = F.linear(x, left_proj_w)    # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # 5. Apply mask if present and not nomask
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        # broadcast mask over hidden dim
        # shape of left_proj/right_proj: (B,N,N,H)
        # Use kernel to compute left = left_proj * left_gate * mask
        left = elementwise_mul_mask(left_proj, left_gate, mask_fp16)
        right = elementwise_mul_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 6. Prepare for batched matmul
    # left shape: (B,N,N,H)
    # We'll permute to (B,H,N,N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    # Transpose right's last two dims for multiplication
    right_T = right.permute(0, 1, 3, 2).contiguous()  # shape (B,H,N,N)

    B, H, N, _ = left.shape

    # flatten batch*H
    batch = B * H
    left_flat = left.view(batch, N, N)
    right_flat = right_T.view(batch, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # launch Triton kernel
    # Determine grid dimensions
    # program_id(0): batch index
    # program_id(1): block row
    # program_id(2): block col
    # number of blocks per dimension
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch, num_blocks_m, num_blocks_n)

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        # strides
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Reshape back to (B, H, N, N)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,H)

    # 7. Output LayerNorm + gate + final linear
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)

    # 8. Cast back to original dtype
    out = out.to(input_tensor.dtype)
    return out

@triton.jit
def mul_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Compute mask index: each group of hidden_dim entries shares same mask element
    idx = offs // hidden_dim
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + idx, mask=mask, other=1.0)
    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start indices of block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Create accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dim
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # load A and B tiles, applying masks for boundaries
        a = tl.load(
            a_ptr + pid_batch * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_ptr + pid_batch * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        # accumulate
        acc += tl.dot(a, b)
    # Write out
    c = acc.to(tl.float16)
    tl.store(
        c_ptr + pid_batch * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
        c,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )

"""
TriMul (outgoing) forward pass – Triton‑accelerated core.
Algorithm
---------
1. Layer‑norm the 4‑D input (B,N,N,C).
2. Cast to FP16 for compute‑intensive parts.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) are performed with torch.nn.functional.linear.
4. If a mask is present, a tiny Triton kernel fuses:
       proj * gate * mask  (broadcast mask over hidden dim).
5. Core O(N³·H) reduction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (left @ rightᵀ) for each
   (b,h) pair, implemented as a fully‑fused Triton GEMM kernel.
6. Apply a second LayerNorm, multiply by the output gate and linearly
   project back to the original channel dimension.
7. Cast the result back to the original dtype (float32).

Only the element‑wise mask‑gate fusion and the batched GEMM are
implemented in Triton; the remaining operations use PyTorch (highly
optimized cuBLAS/cuDNN).
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: elementwise proj * gate * mask (mask broadcasted over hidden)
# ----------------------------------------------------------------------
@triton.jit
def mul_mask_kernel(
    proj_ptr,               # *[total_elems] fp16
    gate_ptr,               # *[total_elems] fp16
    mask_ptr,               # *[mask_elems] fp16
    out_ptr,                # *[total_elems] fp16
    total_elems,            # int
    hidden_dim,             # int
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # each hidden_dim entries share the same mask element
    mask_idx = offs // hidden_dim

    p = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(out_ptr + offs, p * g * m, mask=mask)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 8192) -> torch.Tensor:
    """
    Fuse projection, gate and optional mask:
        out = proj * gate * mask   (mask broadcasted over hidden_dim)
    All tensors are FP16 and contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    mask_flat = mask.view(-1)  # (B*N*N,)

    grid = ((total + block_size - 1) // block_size,)

    mul_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication
#   C[b] = A[b] @ B[b]   where A,B have shape (M,K) and (K,N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # batch * hidden index
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block col

    # (M, N) tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        a = tl.load(
            a_ptr
            + pid_batch * batch_stride_a
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_ptr
            + pid_batch * batch_stride_b
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        acc += tl.dot(a, b)

    # write back in FP16
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_batch * batch_stride_c
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    data = (input_tensor, mask, weights, config)
    Returns: Tensor [bs, seq_len, seq_len, dim] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑intensive stages
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast model weights to FP16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections and gate activations (FP16)
    # --------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)      # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 5. Apply mask and gates (fused via Triton if mask present)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6. Core reduction: batched GEMM left @ rightᵀ for each (b, h)
    # --------------------------------------------------------------
    # Move hidden dimension to the second axis for easy batching
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Transpose the last two dimensions of right for the matmul
    right_t = right.permute(0, 1, 3, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left.shape
    batch = B * H

    # Reshape to (batch, N, N)
    left_flat = left.view(batch, N, N)
    right_flat = right_t.view(batch, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch, num_blocks_m, num_blocks_n)

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Output LayerNorm, output‑gate and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 8. Cast back to the original dtype (typically float32)
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
