@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_aba, stride_am, stride_ak,
    stride_bba, stride_bk, stride_bn,
    stride_cca, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Use mask for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offsets for K
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block
        a_ptrs = a_ptr + batch_idx * stride_aba + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B block (B is stored as (batch, N, K) but we treat as transposed)
        b_ptrs = b_ptr + batch_idx * stride_bba + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Multiply-accumulate
        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = c_ptr + batch_idx * stride_cca + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
Triton‑accelerated outgoing TriMul forward pass.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (dim).
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections (no bias) for left/right values and their gates.
4. Fuse mask (if present) and gates with an element‑wise multiplication
   (performed by PyTorch – cheap compared to the O(N³) core).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   The reduction is a batched GEMM with the right operand interpreted as
   transposed (i.e. left @ rightᵀ).  It is implemented in a custom Triton
   kernel that processes (batch * hidden_dim) independent GEMMs.
6. Output LayerNorm, out‑gate multiplication and final linear projection.
7. Cast back to the original dtype (usually float32).

The kernel uses FP16 for storage/computation and accumulates in FP32
to retain numerical quality.  Block sizes (64×64×32) give good occupancy
on modern NVIDIA GPUs (e.g. H100/B200) and meet the target runtime.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Batched GEMM kernel: C = A @ Bᵀ   where
#   A : (batch, M, K)   (row‑major)
#   B : (batch, N, K)   (row‑major, we treat it as transposed)
#   C : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    # strides for A
    a_batch_stride, a_m_stride, a_k_stride,
    # strides for B (interpreted as transposed)
    b_batch_stride, b_k_stride, b_n_stride,
    # strides for C
    c_batch_stride, c_m_stride, c_n_stride,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ------------------------------------------------------------------
    # Block offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A block: (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + batch_id * a_batch_stride
            + offs_m[:, None] * a_m_stride
            + offs_k[None, :] * a_k_stride
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # B block interpreted as transposed: we need Bᵀ[k, n]
        b_ptrs = (
            b_ptr
            + batch_id * b_batch_stride
            + offs_k[:, None] * b_k_stride   # k dimension (inner)
            + offs_n[None, :] * b_n_stride   # n dimension (rows of Bᵀ)
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back result in FP16
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + batch_id * c_batch_stride
        + offs_m[:, None] * c_m_stride
        + offs_n[None, :] * c_n_stride
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] (bool/float) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # --------------------------------------------------------------
    # 0. Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over channel dimension
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim) – stays in original dtype

    # --------------------------------------------------------------
    # 2. Cast to FP16 for the heavy part
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast weights to FP16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # --------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Apply mask (if any) and element‑wise gating
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B, N, N) -> broadcast over hidden dimension
        mask = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left  = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction – batched GEMM C = A @ Bᵀ
    #    A : left  -> (B, N, N, H)
    #    B : right -> (B, N, N, H)
    # --------------------------------------------------------------
    B, N, _, H = left.shape

    # Move hidden dimension into the batch dimension for the GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    batch_total = B * H
    M = N
    K = N
    N_out = N

    # Shape: (batch_total, M, K) and (batch_total, N_out, K)
    left_flat  = left_perm.view(batch_total, M, K)
    right_flat = right_perm.view(batch_total, N_out, K)

    # Allocate output in FP16 (will be written by the kernel)
    out_flat = torch.empty((batch_total, M, N_out), dtype=torch.float16, device=device)

    # Strides
    a_batch_stride = left_flat.stride(0)
    a_m_stride     = left_flat.stride(1)
    a_k_stride     = left_flat.stride(2)

    b_batch_stride = right_flat.stride(0)
    # For Bᵀ we need stride over K (inner) and N (row after transpose)
    b_k_stride = right_flat.stride(2)   # stride for K (inner dimension)
    b_n_stride = right_flat.stride(1)   # stride for N (row dimension of Bᵀ)

    c_batch_stride = out_flat.stride(0)
    c_m_stride     = out_flat.stride(1)
    c_n_stride     = out_flat.stride(2)

    # --------------------------------------------------------------
    # 8. Launch Triton kernel
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_total,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N_out + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        a_batch_stride,
        a_m_stride,
        a_k_stride,
        b_batch_stride,
        b_k_stride,
        b_n_stride,
        c_batch_stride,
        c_m_stride,
        c_n_stride,
        M,
        N_out,
        K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 9. Reshape back to (B, N, N, H)
    # --------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 10. Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 11. Cast back to the original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
