"""
TriMul (outgoing) forward pass – Triton‑accelerated batched GEMM.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for heavy compute.
3. Linear projections:
   - left  = x @ W_left   (B,N,N,H)
   - right = x @ W_right
4. Gate projections + sigmoid:
   - left_gate, right_gate, out_gate  (B,N,N,H)
5. Fuse projection, gate and optional mask:
   left  = left  * left_gate  * mask
   right = right * right_gate * mask      (mask broadcast over H)
6. Core reduction (batched GEMM):
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented with a custom Triton kernel that performs
   B*H independent matrix multiplications of size N×N.
7. Output LayerNorm over the hidden dimension H, multiply by `out_gate`,
   final linear projection back to `dim`.
8. Cast back to the original dtype.

Only the GEMM (step 6) is written in Triton; all other operations use
torch.nn.functional for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched GEMM (left @ rightᵀ) for many tiny batches.
# Input tensors are assumed to be contiguous fp16 of shape
#   a: (batch, M, K)   – left matrix
#   b: (batch, M, K)   – right matrix (will be treated as transposed)
# Output:
#   c: (batch, M, N)   – result matrix (M=N, K=N)
# --------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    batch, M, N, K,                     # problem size
    stride_abatch, stride_am, stride_ak,  # strides for a
    stride_bbatch, stride_bk, stride_bn,  # strides for b (b is transposed)
    stride_cbatch, stride_cm, stride_cn,  # strides for c
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)      # batch index
    pid_m = tl.program_id(1)    # block row
    pid_n = tl.program_id(2)    # block col

    # ------------------------------------------------------------------
    # Compute offsets for this tile.
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer to the beginning of the batch.
    a_batch_ptr = a_ptr + pid * stride_abatch
    b_batch_ptr = b_ptr + pid * stride_bbatch
    c_batch_ptr = c_ptr + pid * stride_cbatch

    # Accumulator in fp32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles.
    num_blocks_k = tl.cdiv(K, BLOCK_K)

    for k in range(num_blocks_k):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load a tile of shape (BLOCK_K, BLOCK_N) from the transposed view of b
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute block multiplication (fp16×fp16 → fp32)
        acc += tl.dot(a, b)

    # Write back result in fp16.
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy part
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all linear weights to fp16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B,N,N,H)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_w)          # (B,N,N,H)
    right = F.linear(x, right_proj_w)        # (B,N,N,H)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid (same shape)
    # --------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Apply mask and gating (mask broadcast over hidden_dim)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape: (B,N,N) – broadcast over H
        mask_fp16 = mask.to(torch.float16).contiguous()
        # expand mask to (B,N,N,1) for broadcasting
        mask_fp16 = mask_fp16.unsqueeze(-1)
        left = left * left_gate * mask_fp16
        right = right * right_gate * mask_fp16
    else:
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction: batched GEMM left @ rightᵀ
    #    Treat (B, H) as a single batch dimension.
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # left shape [B,N,N,H]

    # Move hidden dimension to batch for the GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    batch = B * H
    # reshape to (batch, N, N)
    left_mat  = left_perm.view(batch, N, N)
    right_mat = right_perm.view(batch, N, N)

    # output placeholder
    out_mat = torch.empty_like(left_mat)

    # strides for the three‑dimensional tensors (batch, M, K) etc.
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_m, stride_b_k = right_mat.stride()
    stride_c_batch, stride_c_m, stride_c_k = out_mat.stride()

    # For the transposed multiplication we swap the last two strides of B
    stride_bk = stride_b_k           # stride of inner dimension (k)
    stride_bn = stride_b_m           # stride of outer dimension (j after transpose)

    # ------------------------------------------------------------------
    # Launch the batched GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_bk, stride_bn,
        stride_c_batch, stride_c_m, stride_c_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm over hidden dimension, apply out_gate,
    #    final linear projection back to `dim`.
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B,N,N,dim)

    # --------------------------------------------------------------
    # 9. Cast back to the original dtype (typically float32)
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
