"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the provided weight/bias.
2. Fuse the two projection linear layers (proj + gate) for the *left* and *right*
   sides by concatenating their weight matrices, then split the result.
   The output‑gate linear layer is computed separately.
3. All heavy tensors are cast to float16 for speed.
4. Apply the optional binary mask and the left/right sigmoid gates *in‑place*
   with a tiny Triton kernel (no extra allocations).
5. Perform the core reduction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched matrix multiplication:
        (B*H, N, N) × (B*H, N, N)^T  using torch.bmm (cuBLAS).
6. Layer‑norm over the hidden dimension, multiply by the output gate and
   project back to the original channel dimension.
7. Cast the result back to the original dtype of the input.

Only the mask+gate fusion is implemented in Triton; all GEMM/Linear ops use
torch’s highly‑optimized kernels. This reduces the number of GEMM calls from
six to three (by concatenating projection+gate weights) and removes temporary
buffers for the masked tensors, achieving a fast forward pass.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_inplace_kernel(
    left_ptr,            # float16*  left tensor,  shape [B,N,N,H] flattened
    right_ptr,           # float16*  right tensor, same shape
    mask_ptr,            # float16*  mask tensor, shape [B,N,N] flattened
    left_gate_ptr,       # float16*  left gate tensor, same shape as left
    right_gate_ptr,      # float16*  right gate tensor, same shape as right
    total_elements: tl.constexpr,   # B*N*N*H
    H: tl.constexpr,                # hidden dimension
    BLOCK_SIZE: tl.constexpr,       # launch block size
):
    """In‑place fusing of (mask * left_gate) and (mask * right_gate)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # load
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # mask is broadcast over the hidden dimension H
    mask_idx = offs // H          # index into flattened mask (B,N,N)
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # apply
    l = l * m * lg
    r = r * m * rg

    # store back in‑place
    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask        : torch.Tensor of shape [B, N, N] or None
            - weights     : dict of model parameters (float32)
            - config      : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack and basic shapes
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    B, N, _, _ = input_tensor.shape  # input shape [B, N, N, C]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Work in float16 for the heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Concatenate projection + gate weights for left and right sides
    # ------------------------------------------------------------------
    # left: (proj + gate)
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)          # (2*H, C)

    # right: (proj + gate)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)          # (2*H, C)

    # out_gate weight (no concatenation)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear transformations (flattened for efficiency)
    # ------------------------------------------------------------------
    x_flat = x.reshape(-1, dim)  # (B*N*N, C)

    left_concat = F.linear(x_flat, left_concat_w)   # (B*N*N, 2*H)
    right_concat = F.linear(x_flat, right_concat_w)  # (B*N*N, 2*H)

    # Split proj and gate
    left = left_concat[:, :hidden_dim]
    left_gate = left_concat[:, hidden_dim:]
    right = right_concat[:, :hidden_dim]
    right_gate = right_concat[:, hidden_dim:]

    # Output gate (needs sigmoid)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w))  # (B*N*N, H)

    # Reshape back to [B, N, N, H]
    left = left.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4) Prepare mask (broadcast over hidden dim) and fuse mask+gates
    # ------------------------------------------------------------------
    if mask is None or nomask:
        # No mask -> use ones – skip the Triton kernel entirely
        left = left * left_gate
        right = right * right_gate
    else:
        # Ensure mask is half precision and contiguous
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        # In‑place fusion kernel
        total_elements = left.numel()          # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)
        _mask_gate_inplace_kernel[grid](
            left.data_ptr(),
            right.data_ptr(),
            mask_fp16.view(-1).data_ptr(),
            left_gate.data_ptr(),
            right_gate.data_ptr(),
            total_elements,
            hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )

    # ------------------------------------------------------------------
    # 5) Core reduction: batched GEMM (left @ rightᵀ)
    #    Shape after view: (B*H, N, N)
    # ------------------------------------------------------------------
    # Bring hidden dimension into the batch axis
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication (cuBLAS)
    out_flat = torch.bmm(left_flat, right_flat.transpose(-2, -1))   # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
