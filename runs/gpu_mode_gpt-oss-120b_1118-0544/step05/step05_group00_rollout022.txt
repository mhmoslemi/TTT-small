    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BATCH, M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offsets
    block_m = pid_m * BLOCK_M
    block_n = pid_n * BLOCK_N

    # pointers to start of batch
    A = A_ptr + pid_batch * stride_Ab
    B = B_ptr + pid_batch * stride_Bb
    C = C_ptr + pid_batch * stride_Cb

    # load block of A (block_m: block_m+BLOCK_M, 0:K)
    offs_m = block_m + tl.arange(0, BLOCK_M)
    offs_n = block_n + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Input LayerNorm (float32 for stability).
2. Cast to fp16 for the compute‑heavy path.
3. Linear projections for left/right, their gates and the output gate.
4. Fuse element‑wise mask (if present) with the left/right gates via a tiny
   Triton kernel (in‑place, no extra memory).
5. Core contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM `left @ rightᵀ`. Implemented with a custom
   Triton kernel that processes the batch dimension and tiles the N×N
   matrices (block sizes 64×64×32). Accumulation is in float32, output in fp16.
6. Output LayerNorm over the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original dimension.
7. Cast back to the input dtype (usually float32) and return.

The kernel runs efficiently for N up to 1024 and hidden_dim = 128 on an
NVIDIA B200 (Triton 3.3.1).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,          # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,
    mask_ptr,                     # [B,N,N] flattened
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place multiplication:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
       mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # each H‑wide slice shares the same mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_ab, stride_am, stride_ak,   # A: batch, M, K
    stride_bb, stride_bk, stride_bn,   # B: batch, K, N  (B is logically transposed)
    stride_cb, stride_cm, stride_cn,   # C: batch, M, N
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM computing C = A @ Bᵀ where
       A,B shape = (batch, M, K) with B accessed as (K, N) via swapped strides.
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = A_ptr + pid_batch * stride_ab
    b_batch = B_ptr + pid_batch * stride_bb
    c_batch = C_ptr + pid_batch * stride_cb

    # Pointers for the current block of A and B
    a_ptrs = a_batch + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] + k * BLOCK_K < K)),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=((tl.arange(0, BLOCK_K)[:, None] + k * BLOCK_K < K) & mask_n[None, :]),
                    other=0.0)
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N)
        # Advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back result
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------ #
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # ------------------------------------------------------------------ #
    # 2. Cast to fp16 for the compute‑heavy path
    # ------------------------------------------------------------------ #
    x = x.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 3. Linear projections (bias‑less) – all in fp16
    # ------------------------------------------------------------------ #
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)    # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)   # (B,N,N,hidden_dim)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------ #
    # 4. Fuse mask and gates (if mask is present)
    # ------------------------------------------------------------------ #
    if mask is not None and not nomask:
        # Prepare contiguous tensors for the kernel
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_fp.view(-1)          # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024

            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Restore shapes
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Core O(N³·H) contraction via a custom Triton GEMM
    #    Compute out = left @ rightᵀ for each hidden channel.
    # ------------------------------------------------------------------ #
    # Reshape to (B*H, N, N) where the hidden dimension is merged into batch.
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    out_mat = torch.empty_like(left_mat)                # (B*H, N, N), fp16

    stride_am = left_mat.stride(1)   # row stride (M) = N
    stride_ak = left_mat.stride(2)   # col stride (K) = 1

    stride_bb = right_mat.stride(0)
    # For Bᵀ we swap K/N strides: K stride = original col stride = 1, N stride = original row stride = N
    stride_bk = right_mat.stride(2)   # = 1

    stride_cb = out_mat.stride(0)
    stride_cm = out_mat.stride(1)     # = N
    stride_cn = out_mat.stride(2)     # = 1

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # ------------------------------------------------------------------ #
