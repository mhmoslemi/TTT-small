"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for the compute‑heavy portion.
3. Linear projections (no bias):
       left  = x @ W_left   ,   right = x @ W_right
   Gate projections + sigmoid:
       left_gate  = σ(x @ W_left_gate)
       right_gate = σ(x @ W_right_gate)
       out_gate   = σ(x @ W_out_gate)
4. Apply optional binary mask (broadcast over the hidden dimension) and fuse the
   gate by simple elementwise multiplication (done in PyTorch).
5. Core reduction (the O(N³) step):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM using a custom Triton kernel that processes
   all hidden slices (batch * hidden_dim) in parallel.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.
7. Cast the result back to the original dtype.

Only the reduction in step 5 is written in Triton; all other operations use
high‑performance PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# -------------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ B) for many batches.
# A : (batch, M, K)  – fp16
# B : (batch, K, N)  – fp16
# C : (batch, M, N)  – fp16 (accumulated in fp32)
# -------------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                 # pointers
    batch_stride_a, batch_stride_b, batch_stride_c,  # stride to next batch
    M, N, K,                             # matrix sizes
    stride_am, stride_ak,                # A row/col strides
    stride_bk, stride_bn,                # B row/col strides
    stride_cm, stride_cn,                # C row/col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Decode program id into (batch, tile_m, tile_n)
    # --------------------------------------------------------------
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_pid_m * num_pid_n)
    pid_in_batch = pid % (num_pid_m * num_pid_n)
    pid_m = pid_in_batch // num_pid_n
    pid_n = pid_in_batch % num_pid_n

    # --------------------------------------------------------------
    # Offsets for the output tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Base pointers for this batch
    # --------------------------------------------------------------
    a_batch_ptr = a_ptr + batch * batch_stride_a
    b_batch_ptr = b_ptr + batch * batch_stride_b
    c_batch_ptr = c_ptr + batch * batch_stride_c

    # --------------------------------------------------------------
    # Accumulator (fp32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles of size BLOCK_K
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = tl.arange(0, BLOCK_K)
        k_idx = k_offset + k_range

        # Load A tile [BLOCK_M, BLOCK_K]
        a = tl.load(
            a_batch_ptr
            + offs_m[:, None] * stride_am
            + k_idx[None, :] * stride_ak,
            mask=mask_m[:, None] & (k_idx[None, :] < K),
            other=0.0,
        )
        # Load B tile [BLOCK_K, BLOCK_N]
        b = tl.load(
            b_batch_ptr
            + k_idx[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=(k_idx[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        # Dot product (fp32 accumulation)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write back
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _triton_batched_gemm(a: torch.Tensor, b: torch.Tensor,
                         BLOCK_M: int = 64, BLOCK_N: int = 64, BLOCK_K: int = 32) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.
    a: (batch, M, K)   fp16, contiguous
    b: (batch, K, N)   fp16, contiguous
    returns c: (batch, M, N)   fp16, contiguous
    """
    assert a.is_contiguous() and b.is_contiguous()
    assert a.dtype == torch.float16 and b.dtype == torch.float16
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)

    # strides (in elements)
    stride_am = a.stride(1)
    stride_ak = a.stride(2)
    stride_bk = b.stride(1)
    stride_bn = b.stride(2)
    stride_cm = c.stride(1)
    stride_cn = c.stride(2)

    batch_stride_a = a.stride(0)
    batch_stride_b = b.stride(0)
    batch_stride_c = c.stride(0)

    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    _tri_mul_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        batch_stride_a=batch_stride_a,
        batch_stride_b=batch_stride_b,
        batch_stride_c=batch_stride_c,
        M=M,
        N=N,
        K=K,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return c


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] (bool/float) or None
            - weights      : dict with weight tensors (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast model weights to FP16 (biases are None for linear layers)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B, N, N, hidden_dim)
    # --------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)  # (B,N,N,H)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # --------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Apply mask (if present) and combine with gates
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    left = left_proj * left_gate          # (B,N,N,H)
    right = right_proj * right_gate       # (B,N,N,H)

    # --------------------------------------------------------------
    # 7. Core reduction using the Triton batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # Move hidden dimension to the second axis and make tensors contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N)  (k,j) order

    left_flat = left_perm.view(B * H, N, N)    # (batch, M, K) where M=K=N
    right_flat = right_perm.view(B * H, N, N)  # (batch, K, N)

    # Triton batched GEMM
    out_flat = _triton_batched_gemm(left_flat, right_flat,
                                    BLOCK_M=64, BLOCK_N=64, BLOCK_K=32)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
