Define kernel similar to earlier:

```
@triton.jit
def _apply_gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * lg * m
        right = right * rg * m
    else:
        left = left * lg
        right = right * rg

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Key ideas
---------
1. Fuse the five linear projections/gates into a single GEMM by stacking the
   weights.  This reduces memory traffic because the large input tensor
   (B·N·N·dim) is read only once.
2. Apply the mask and sigmoid gates with a tiny Triton kernel that works in‑place
   on the projected tensors (left/right) to keep the operation memory‑bound.
3. Perform the core O(N³·H) contraction with a single batched GEMM
   (torch.bmm) on tensors of shape (B·H, N, N).  This is the same computation
   as the reference `einsum('... i k d, ... j k d -> ... i j d')` but
   leverages cuBLAS for maximum throughput.
4. Use FP16 for all heavy arithmetic while keeping LayerNorms in FP32 for
   numerical stability.  The final result is cast back to the input dtype
   (usually float32).

The function `custom_kernel` follows the required signature:
    data = (input_tensor, mask, weights, config)
and returns a tensor of shape (B, N, N, dim).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton kernel: in‑place mask & gate fusion for left/right tensors
# ---------------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,                # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,      # same shape as left/right
    mask_ptr,                           # [B,N,N] flattened, broadcast over H
    total_elems, hidden_dim,            # runtime scalars
    HAS_MASK: tl.constexpr,             # compile‑time bool
    BLOCK_SIZE: tl.constexpr,           # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load tensors
    left      = tl.load(left_ptr      + offs, mask=active, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden_dim entries share a single mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)

# ---------------------------------------------------------------------------
# Main entry point
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    # -----------------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for the compute‑heavy parts
    x = x.to(torch.float16)

    # -----------------------------------------------------------------------
    # 2. Fuse the five linear projections/gates into one GEMM
    # -----------------------------------------------------------------------
    # Gather and concatenate the five weight matrices (each hidden_dim x dim)
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Shape after concat: (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    ).to(device).to(torch.float16)

    # Perform a single linear transformation
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)
    combined = F.linear(x_flat, combined_weight) # (B*N*N, 5*hidden_dim)
    combined = combined.view(B, N, N, 5 * hidden_dim)

    # Split into the five tensors
    splits = torch.split(combined, hidden_dim, dim=-1)
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = splits
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)

    # -----------------------------------------------------------------------
    # 3. Fuse mask and gate multiplication (in‑place) via Triton
    # -----------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask is broadcast over the hidden dimension
        mask_fp = mask.to(device).to(torch.float16)
        total = left_proj.numel()
        BLOCK = 65536  # tuneable; must be a multiple of 32 for best occupancy
        grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

        # Flatten tensors for the kernel
        left_flat      = left_proj.contiguous().view(-1)
        right_flat     = right_proj.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # The views are updated in‑place; reshape back to original layout
        left = left_proj
        right = right_proj
    else:
        # No mask: simple element‑wise multiplication with gates
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------------
    # 4. Core contraction: batched GEMM (left @ rightᵀ) over the sequence dimension
    # -----------------------------------------------------------------------
    # Rearrange to (B*H, N, N) for a single torch.bmm call
    B, N, _, H = left.shape
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------------
    # 5. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply the output gate
    out = out * out_gate

    # Final linear projection back to the original feature dimension
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)  # (dim, hidden_dim)
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # Cast back to the input dtype (typically float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
