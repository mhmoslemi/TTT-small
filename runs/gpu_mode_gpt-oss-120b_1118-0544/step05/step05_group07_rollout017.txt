"""
TriMul (outgoing) – Triton‑fused mask & gate + fused linear projections.

Algorithm
---------
1. Layer‑norm the input (B,N,N,C) using the provided LN weight/bias.
2. Cast to fp16 for all heavy compute.
3. Fuse each pair of linear projection + its gate (weight shapes (H,C) each)
   into a single matmul of shape (B·N·N, 2·H) and split the result.
4. Apply sigmoid to the gate halves.
5. Fuse element‑wise (proj * gate * mask) with a tiny Triton kernel
   (mask broadcasted over the hidden dimension).  If `nomask=True` the mask
   is ignored.
6. Core reduction: for every hidden feature d, compute
      out[...,i,j,d] = Σ_k left[...,i,k,d] * right[...,j,k,d]
   which is a batch of (B·H) matrix multiplies (N×N)·(N×N)ᵀ.  Implemented via
   `torch.bmm` on a reshaped view (highly‑optimised cuBLAS/Tensor‑core).
7. Second LayerNorm over the hidden dimension, multiply by the output gate.
8. Final linear projection back to the original channel dimension.
9. Cast result back to the input dtype (float32).

Only the mask‑gate fusion is performed in Triton, satisfying the
requirement of using a custom kernel while the bulk of the work is done
with highly‑optimised GEMM kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) where mask is broadcasted
# over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16*  (B*N*N*H,)
    gate_ptr,          # fp16*  (B*N*N*H,)
    mask_ptr,          # fp16*  (B*N*N,)
    out_ptr,           # fp16*  (B*N*N*H,)
    total_elems,       # i32   total number of elements in proj/gate
    hidden_dim,        # i32   H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Each mask element repeats hidden_dim times along the last dim
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using a Triton kernel.
    All inputs must be fp16, contiguous and on the same CUDA device.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],   # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------- LayerNorm (input) -------------------------------
    # Use float32 for LN to keep numerical stability, then cast to fp16.
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------- cast weights to fp16 -------------------------
    # Fuse left projection & its gate, same for right.
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ------------------- linear projections + gates ----------------------
    # Combined matmul: (B*N*N, 2*H)
    left_combined = torch.nn.functional.linear(x, left_combined_w)   # [B,N,N,2H]
    right_combined = torch.nn.functional.linear(x, right_combined_w) # [B,N,N,2H]

    # Split into proj and gate halves
    left_proj, left_gate_raw = left_combined.split([hidden_dim, hidden_dim], dim=-1)
    right_proj, right_gate_raw = right_combined.split([hidden_dim, hidden_dim], dim=-1)

    left_gate = left_gate_raw.sigmoid()
    right_gate = right_gate_raw.sigmoid()

    # Output gate (no mask)
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # [B,N,N,H]

    # ------------------- mask & gate fusion (Triton) --------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ----------------- core O(N³·H) reduction via batched GEMM ----------
    B, N, _, _ = left.shape  # N == seq_len
    # Rearrange to (B*H, N, N) for torch.bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left_perm.view(-1, N, N)                 # (B*H, N, N)
    right_mat = right_perm.view(-1, N, N)               # (B*H, N, N)

    # Compute left_mat @ right_mat^T for each (B*H) batch
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------ cast back to input dtype -----------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
