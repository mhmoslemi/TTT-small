        BLOCK_N=block_n,

"""
TriMul (outgoing) forward – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for the heavy compute part.
3. Linear projections for left/right and the three gates (no bias).
4. Fuse projection, gate (sigmoid) and optional mask in a Triton kernel:
      out = proj * gate * mask   (mask broadcast over the hidden dim).
5. Batched matrix multiplication:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton GEMM that computes A @ Bᵀ where
   A and B are reshaped to (B*H, N, N).  This avoids the Python‑level
   reshapes and the torch.bmm call.
6. Cast back to FP32, apply output LayerNorm, multiply by `out_gate`,
   and a final linear projection to the original `dim`.
7. Return the result in the original dtype (usually float32).

Both the mask‑gate fusion and the batched GEMM are written in Triton,
providing a substantial speed‑up over a pure PyTorch reference.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask
# --------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # int64: total number of elements in proj/gate/out
    hidden_dim,        # int64: H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # mask index = floor(offs / H) because mask is broadcast over H
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 2048) -> torch.Tensor:
    """
    Element‑wise multiplication proj * gate * mask (mask broadcast over the hidden dim).
    All tensors must be fp16 and contiguous.
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# --------------------------------------------------------------
# Triton kernel: batched GEMM with B‑dim as batch (computes A @ Bᵀ)
# --------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,          # A: (batch, M, K)
    stride_bk, stride_bn, stride_bb,          # B (will be accessed as Bᵀ): (batch, K, N)
    stride_cm, stride_cn, stride_cb,          # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch index

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for this batch
    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bb
    c_batch = c_ptr + pid_b * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load Bᵀ tile: we need Bᵀ[k, n] = B[n, k]
        b = tl.load(
            b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # Write back C
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_gemm_transpose(a: torch.Tensor,
                            b: torch.Tensor,
                            block_m: int = 64,
                            block_n: int = 64,
                            block_k: int = 32) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for tensors A and B of shape (B, N, N) and dtype fp16.
    The function treats the first dimension as batch and performs a
    separate GEMM per batch element.
    """
    assert a.shape == b.shape
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

    # Strides (in elements)
    a_stride_ab, a_stride_am, a_stride_ak = a.stride()
    b_stride_bb, b_stride_bj, b_stride_bk = b.stride()     # (batch, row=j, col=k)
    # For Bᵀ we need stride over K (original col) and N (original row)
    stride_bk = b_stride_bk      # stride for original column dimension (k)
    stride_bn = b_stride_bj      # stride for original row dimension (j)

    c_stride_cb, c_stride_cm, c_stride_cn = c.stride()

    grid = (
        triton.cdiv(M, block_m),
        triton.cdiv(N, block_n),
        batch,
    )

    _batched_gemm_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        a_stride_am,
        a_stride_ak,
        a_stride_ab,
        stride_bk,
        stride_bn,
        b_stride_bb,
        c_stride_cm,
        c_stride_cn,
        c_stride_cb,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return c

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys
                  * "dim" – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all linear weights to FP16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B,N,N,H)  (no bias)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # --------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # 6. Fuse mask, projection and gate (if mask is present)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous (0/1)
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the sequence dimension
    #    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B, H, N, N) → (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM with transposed right operand
    out_flat = _batched_gemm_transpose(left_flat, right_flat)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    # Cast to FP32 for the remaining ops (better stability)
    out = out.to(torch.float32)

    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, to_out_w.to(torch.float32))   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 9. Cast back to the original dtype (usually float32)
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
