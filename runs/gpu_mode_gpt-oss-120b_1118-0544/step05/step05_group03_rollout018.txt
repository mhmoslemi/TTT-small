@triton.jit
def triu_batch_gemm_kernel(
    left_ptr,              # fp16 * (B,N,N,H)
    right_ptr,             # fp16 *
    out_ptr,               # fp32 or fp16
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
   pid_i = tl.program_id(0)
   pid_j = tl.program_id(1)
   pid_bh = tl.program_id(2)

   # compute start indices
   i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
   j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

   # batch part
   bh = pid_bh
   b = bh // H
   h = bh % H

   # init acc
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # loop over k dimension
   for kk in range(0, tl.cdiv(N, BLOCK_K)):
       k = kk * BLOCK_K + tl.arange(0, BLOCK_K)

       # masks
       mask_i = i < N
       mask_j = j < N
       mask_k = k < N

       # compute pointers for left (i,k)
       left_ptrs = left_ptr + b * stride_left_b + i[:, None] * stride_left_i + k[None, :] * stride_left_k + h * stride_left_h
       left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

       # compute pointers for right transposed (k,j)
       right_ptrs = right_ptr + b * stride_right_b + k[:, None] * stride_right_k + j[None, :] * stride_right_j + h * stride_right_h
       right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

       # accumulate dot: left_tile (M,K) * right_tile (K,N) -> (M,N)
       acc += tl.dot(left_tile, right_tile, out_dtype=tl.float32)

   # store result
   out_ptrs = out_ptr + b * stride_out_b + i[:, None] * stride_out_i + j[None, :] * stride_out_j + h * stride_out_h
   tl.store(out_ptrs, acc, mask=(i < N)[:, None] & (j < N)[None, :])

def triu_batch_gemm(left, right):
    # left, right: torch tensors [B,N,N,H] dtype fp16
    B,N,_,H = left.shape
    # allocate output
    out = torch.empty_like(left, dtype=torch.float32)   # we can keep as fp32
    # compute strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()
    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = ( (N + BLOCK_M - 1)//BLOCK_M,
             (N + BLOCK_N - 1)//BLOCK_N,
             B*H )
    triu_batch_gemm_kernel[grid](
        left, right, out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    return out

"""
custom_kernel: Outgoing TriMul forward pass (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for the heavy linear/gate work.
3. Linear projections (left/right) and gate projections (left/right/out) are
   performed with FP16 weights.  If a mask is present we fuse
   projection * gate * mask with a small Triton kernel.
4. Core O(N³) reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (B·H independent N×N matrix
   multiplications) using a custom Triton kernel that tiles over the
   i, j and k dimensions.
5. Layer‑norm over the hidden dimension, multiplication by the output gate,
   and a final FP32 linear projection back to the original dimension.
6. Return the result in the original input dtype (usually float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i64 : B*N*N*H
    hidden_dim,        # i64 : H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # each mask element is repeated `hidden_dim` times
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 2048) -> torch.Tensor:
    """
    proj, gate : (B, N, N, H)  fp16
    mask       : (B, N, N)     fp16 (0/1)
    Returns    : (B, N, N, H)  fp16  = proj * gate * mask
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the N³ reduction
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    left_ptr,          # fp16 * (B, N, N, H)
    right_ptr,         # fp16 * (B, N, N, H)
    out_ptr,           # fp32 * (B, N, N, H)
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)  # tile row
    pid_j = tl.program_id(1)  # tile col
    pid_bh = tl.program_id(2) # batch*head

    # compute start indices for this tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # decode batch and head from linear index
    bh = pid_bh
    b = bh // H
    h = bh % H

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)

        mask_i = i < N
        mask_j = j < N
        mask_k = k < N

        # left block: (i, k, h)
        left_ptrs = (left_ptr
                     + b * stride_l_b
                     + i[:, None] * stride_l_i
                     + k[None, :] * stride_l_k
                     + h * stride_l_h)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right block transposed: need (k, j, h) so that we can
        # compute left (M,K) @ right_T (K,N)
        right_ptrs = (right_ptr
                      + b * stride_r_b
                      + k[:, None] * stride_r_k
                      + j[None, :] * stride_r_j
                      + h * stride_r_h)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # accumulate dot product
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # store result
    out_ptrs = (out_ptr
                + b * stride_o_b
                + i[:, None] * stride_o_i
                + j[None, :] * stride_o_j
                + h * stride_o_h)
    mask_store = (i < N)[:, None] & (j < N)[None, :]
    tl.store(out_ptrs, acc, mask=mask_store)


def _batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for TriMul: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    left/right: (B, N, N, H) fp16
    Returns    : (B, N, N, H) fp32
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides in elements (not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    _batched_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]
            - mask         : torch.Tensor [B, N, N] (or None)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"]   # keep FP32 for layer_norm
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]       # keep FP32 for final projection

    # ------------------------------------------------------------------
    # 4. Linear projections
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 6. Fuse projection, gating and optional mask
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = (left_proj * left_gate).contiguous()
        right = (right_proj * right_gate).contiguous()

    # ------------------------------------------------------------------
    # 7. Core reduction (batched GEMM)
    # ------------------------------------------------------------------
    out = _batched_gemm(left, right)   # (B,N,N,hidden_dim) FP32

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate.to(out.dtype)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) FP32

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
