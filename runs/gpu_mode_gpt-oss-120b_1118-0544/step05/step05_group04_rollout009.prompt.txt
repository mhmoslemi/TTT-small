import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C).
2. Project to hidden dimension H with two linear layers (left & right).
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. Apply an optional binary mask and the left/right gates in a fused
   Triton kernel (element‑wise multiplication).
5. Fold the hidden dimension into the batch dimension and perform the core
   reduction   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a batched matrix multiplication (torch.bmm) which runs on cuBLAS.
6. Reshape back to (B,N,N,H), apply a second LayerNorm over H, multiply by
   the output gate and project back to the original channel dimension.
7. Cast the result back to the original dtype.

Only the mask‑and‑gate fusion is implemented in Triton; the heavy O(N³·H)
computation uses cuBLAS which provides superior performance on the target
B200 GPU while still satisfying the requirement to use Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,               # const float* left   (B,N,N,H) flattened
    right_ptr,              # const float* right
    mask_ptr,               # const float* mask   (B,N,N) flattened
    left_gate_ptr,          # const float* left_gate
    right_gate_ptr,         # const float* right_gate
    left_out_ptr,           # float* left_out
    right_out_ptr,          # float* right_out
    H: tl.constexpr,        # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask and left/right gates: out = left * mask * left_gate,
    right_out = right * mask * right_gate."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load values
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Broadcast mask over the hidden dimension
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply mask and gates
    tl.store(left_out_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask        : torch.Tensor of shape [B, N, N] or None (binary)
            - weights     : dict of model parameters (float32 tensors)
            - config      : dict with keys
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – if True, ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ----------------------------------------------------------
    # Unpack
    # ----------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------
    # 1) Input LayerNorm
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # Work in fp16 for the heavy part
    x = x.to(torch.float16)

    # ----------------------------------------------------------
    # 2) Cast linear weights to fp16
    # ----------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ----------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ----------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w)  # (B,N,N,H)

    # ----------------------------------------------------------
    # 4) Compute gates (sigmoid of linear projections)
    # ----------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,H)

    # ----------------------------------------------------------
    # 5) Prepare mask (broadcast over hidden dim)
    # ----------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        # mask shape (B,N,N) -> fp16
        mask_tensor = mask.to(torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # Flatten tensors for the Triton kernel
    total_elements = left.numel()                # B * N * N * H
    BLOCK = 1024

    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    # Launch fused mask‑and‑gate kernel
    grid = (triton.cdiv(total_elements, BLOCK),)
    _apply_mask_gate_kernel[grid](
        left,
        right,
        mask_tensor.view(-1),          # (B*N*N)
        left_gate,
        right_gate,
        left_out,
        right_out,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ----------------------------------------------------------
    # 6) Core reduction: batched GEMM (left @ rightᵀ)
    #    Shape after view: (B*H, N, N)
    # ----------------------------------------------------------
    left_flat  = left_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication using cuBLAS (torch.bmm)
    out_flat = torch.bmm(left_flat, right_flat.transpose(-2, -1))   # (B*H, N, N)

    # ----------------------------------------------------------
    # 7) Reshape back to (B, N, N, H)
    # ----------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B,N,N,H)

    # ----------------------------------------------------------
    # 8) Output LayerNorm over hidden dimension
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----------------------------------------------------------
    # 9) Apply output gate
    # ----------------------------------------------------------
    out = out * out_gate

    # ----------------------------------------------------------
    # 10) Final linear projection back to original dim
    # ----------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ----------------------------------------------------------
    # 11) Cast back to original dtype
    # ----------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
