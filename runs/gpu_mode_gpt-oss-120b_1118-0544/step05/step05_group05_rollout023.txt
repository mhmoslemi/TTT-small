"""
Optimized outgoing TriMul forward pass.

Algorithm
---------
1. Apply LayerNorm to the 4‑D input (float32 for stability).
2. Cast to fp16 and compute linear projections + sigmoid gates.
3. Fuse element‑wise mask (if present) and gate multiplication with a Triton
   kernel to avoid extra memory traffic.
4. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   implemented with a single torch.einsum call (fp16) which internally
   performs a highly‑optimized tiled reduction.
5. Apply output LayerNorm, the output gate and a final linear projection.
6. Cast back to the original dtype.

At least one operation (mask‑gate fusion) is executed on Triton, while the
heavy O(N³·H) contraction uses a fused einsum for maximum throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape as left/right
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    """In‑place fusion of left/right with their gates and (optional) mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # load
    left      = tl.load(left_ptr      + offs, mask=active, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden_dim entries share one mask element
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    # store back
    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (LayerNorms, Linear weights)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, torch.float32)
    norm_b = weights["norm.bias"].to(device, torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)   # cast to fp16 for the heavy part

    # -----------------------------------------------------------------
    # 2. Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gates (all fp16)
    # -----------------------------------------------------------------
    left      = F.linear(x, left_proj_w)          # (B, N, N, H)
    right     = F.linear(x, right_proj_w)         # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 4. Fuse mask and gate multiplication with Triton (if needed)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # mask broadcasted over hidden dimension, fp16 for speed
        mask_fp = mask.to(device, torch.float16)

        # flatten tensors for the kernel
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left  = left_flat.view_as(left)
        right = right_flat.view_as(right)
        # free intermediate tensors
        del left_gate_flat, right_gate_flat, mask_flat, left_flat, right_flat
    else:
        # No mask → direct element‑wise multiplication with gates
        left  = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction via a fused einsum (fp16)
    # -----------------------------------------------------------------
    # out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the original dtype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
