@triton.jit
def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # batch
    pid_m = tl.program_id(axis=1)
    pid_n = tl.program_id(axis=2)

    # Compute block offset
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute offsets for K block
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A block [M, K]
        a_ptrs = a_ptr + pid * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=(offs_m < M) & (offs_k < K), other=0.0)

        # Load B block [K, N]
        b_ptrs = b_ptr + pid * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=(offs_k < K) & (offs_n < N), other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = c_ptr + pid * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=(offs_m < M) & (offs_n < N))
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype
    # retrieve config etc
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    # Stage 1: LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Convert to fp16 for subsequent compute
    x = x.to(torch.float16)
    # Stage 2: Linear projections
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    left = F.linear(x, left_proj_w)   # shape [B,N,N,hidden]
    right = F.linear(x, right_proj_w)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Mask
    if mask is not None and not nomask:
        mask_tensor = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
    else:
        mask_tensor = torch.ones_like(left[..., :1])  # correct dtype

    left = left * mask_tensor * left_gate
    right = right * mask_tensor * right_gate

    # Prepare for matmul: reshape and transpose
    B, N, _, H = left.shape
    left_flat = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_flat = right.permute(0, 3, 1, 2).contiguous()

    # Merge batch and hidden
    left_mat = left_flat.view(B * H, N, N)
    right_mat = right_flat.view(B * H, N, N).transpose(1, 2).contiguous() # [B*H, N, N] as transposed

    # Allocate output
    out_mat = torch.empty_like(left_mat)  # same shape, dtype fp16

    # Launch Triton GEMM kernel
    _triton_matmul(
        left_mat,
        right_mat,
        out_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=32,
    )

    # Reshape result back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # Output layernorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Apply out_gate
    out = out * out_gate

    # Final linear projection
    out = F.linear(out, to_out_w)

    # Cast back to original dtype
    if out.dtype != orig_dtype:

    batch, M, K = a.shape

def _matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak, stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize ACC
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for K
        k_valid = k_offset < K

        # load A
        a_ptrs = a_ptr + pid * batch_stride_a + offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak

"""
custom_kernel: Triton‑accelerated outgoing TriMul (AlphaFold3).

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C).
2. Linear projections to a hidden dimension H producing `left` and `right`.
3. Compute three sigmoid gates from the normalized input.
4. Apply optional binary mask and left/right gates (elementwise).
5. Core O(N³·H) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (left @ rightᵀ) using a custom Triton
   kernel that processes B·H independent matrix multiplications in parallel.
6. Layer‑norm over the hidden dimension, multiply by the output gate and a
   final linear projection back to the original channel dimension.
7. Cast the result back to the input dtype.

Only the heavy reduction (step 5) is performed in Triton; all other
operations use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _triton_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak, stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   (A: [M,K], B: [K,N]) for each batch."""
    pid = tl.program_id(0)          # batch index  (0 .. B*H-1)
    pid_m = tl.program_id(1)        # block row
    pid_n = tl.program_id(2)        # block col

    # Offsets of the block within the output matrix C
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for boundary handling
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to A and B tiles
        a_ptrs = a_ptr + pid * batch_stride_a \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid * batch_stride_b \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # Load tiles with masks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate the dot product
        acc += tl.dot(a, b)

    # Write the result back to C
    c_ptrs = c_ptr + pid * batch_stride_c \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _triton_gemm(a: torch.Tensor, b: torch.Tensor, out: torch.Tensor,
                 BLOCK_M=128, BLOCK_N=128, BLOCK_K=32):
    """
    Batched GEMM wrapper calling the Triton kernel.
    a: [B*H, M, K]   (float16, contiguous)
    b: [B*H, K, N]   (float16, contiguous)
    out: [B*H, M, N] (float16, contiguous)
    """
    assert a.is_contiguous() and b.is_contiguous() and out.is_contiguous()
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "Incompatible inner dimensions"

    # Strides (in elements)
    batch_stride_a = a.stride(0)
    stride_am = a.stride(1)
    stride_ak = a.stride(2)

    batch_stride_b = b.stride(0)
    stride_bk = b.stride(1)
    stride_bn = b.stride(2)

    batch_stride_c = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _triton_gemm_kernel[grid](
        a,
        b,
        out,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        M,
        N,
        K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask        : torch.Tensor of shape [B, N, N] or None
            - weights     : dict of model parameters (float32 tensors)
            - config      : dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – if True, ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ----------------------------------------------------------
    # Unpack
    # ----------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to fp16 for the heavy computation
    x = x.to(torch.float16)

    # ----------------------------------------------------------
    # 2) Load weights and cast to fp16
    # ----------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ----------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ----------------------------------------------------------
    left = F.linear(x, left_proj_w)    # [B, N, N, H]
    right = F.linear(x, right_proj_w)  # [B, N, N, H]

    # ----------------------------------------------------------
    # 4) Compute gates (sigmoid of linear projections)
    # ----------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # [B, N, N, H]

    # ----------------------------------------------------------
    # 5) Apply optional mask and the left/right gates
    # ----------------------------------------------------------
    if mask is not None and not nomask:
        mask_tensor = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
    else:
        mask_tensor = torch.ones_like(left[..., :1], dtype=torch.float16)

    left = left * mask_tensor * left_gate
    right = right * mask_tensor * right_gate

    # ----------------------------------------------------------
    # 6) Core reduction: batched GEMM (left @ rightᵀ)
    #    Shape after view: (B*H, N, N)
    # ----------------------------------------------------------
    B, N, _, H = left.shape
    # bring H into batch dimension for the matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_flat = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat = left_flat.view(B * H, N, N)                     # [B*H, N, N]
    right_mat = right_flat.view(B * H, N, N).transpose(1, 2).contiguous()  # [B*H, N, N] (K,N)

    out_mat = torch.empty_like(left_mat)  # accumulator in fp16
    _triton_gemm(left_mat, right_mat, out_mat,
                 BLOCK_M=128, BLOCK_N=128, BLOCK_K=32)

    # ----------------------------------------------------------
    # 7) Reshape back to (B, N, N, H)
    # ----------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ----------------------------------------------------------
    # 8) Output LayerNorm over hidden dimension (float32 for stability)
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
    )

    # ----------------------------------------------------------
    # 9) Apply output gate
    # ----------------------------------------------------------
    out = out * out_gate

    # ----------------------------------------------------------
    # 10) Final linear projection back to original dim
    # ----------------------------------------------------------
    out = F.linear(out, to_out_w)   # [B, N, N, C]

    # ----------------------------------------------------------
    # 11) Cast back to original dtype if needed
    # ----------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
