```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B,N,N_dim, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Cast to fp16
    x = x.to(torch.float16)

    # Linear weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)  # B,N,N,hidden_dim
    right_proj = torch.nn.functional.linear(x, right_proj_w)
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # Apply gates and mask
    if mask is not None and not nomask:
        mask = mask.to(torch.float16).unsqueeze(-1)  # B,N,N,1
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Core reduction using Triton
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_flat = right.permute(0,3,1,2).contiguous().view(B*H, N, N)

    out_flat = _batched_gemm_triton(left_flat, right_flat, N, BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)

    # Reshape back
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()

    # Output layer norm + gating + final linear
    out = torch.nn.functional.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # dims: (B,N,N,dim)

    # Cast back to original dtype
    out = out.to(input_tensor.dtype)
    return out
```

```
def _batched_gemm_triton(a: torch.Tensor, b: torch.Tensor, N: int, BLOCK_M: int=64, BLOCK_N: int=64, BLOCK_K: int=64):
    assert a.is_contiguous()
    assert b.is_contiguous()
    M = a.shape[0]  # B*H
    # allocate output
    c = torch.empty_like(a)  # same shape (M,N,N)
    # strides
    stride_am = a.stride(0)
    stride_ai = a.stride(1)
    stride_ak = a.stride(2)
    stride_bm = b.stride(0)
    stride_bj = b.stride(1)
    stride_bk = b.stride(2)
    stride_cm = c.stride(0)
    stride_ci = c.stride(1)
    stride_cj = c.stride(2)
    grid = (M, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    _batched_gemm_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        N,  # K same as N (since square)
        stride_am, stride_ai, stride_ak,
        stride_bk, stride_bj, stride_bm,  # Actually need to pass order correctly
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K

    return c
```

"""
custom_kernel: Triton‑accelerated outgoing TriMul forward.

Algorithm
---------
1. Layer‑norm the 4‑D input over the channel dimension `dim`.
2. Cast to fp16 for the compute‑heavy path.
3. Linear projections (no bias) to obtain `left`, `right` and the three
   gating tensors (`left_gate`, `right_gate`, `out_gate`).  Apply sigmoid
   to the gates.
4. Multiply projections by their gates and (optionally) by the pairwise
   mask (broadcast over the hidden dimension).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication `C = A @ Bᵀ` for every
   (batch, hidden) slice.  It is implemented in Triton with a
   three‑dimensional grid (batch‑slice, block‑row, block‑col) and
   blocked GEMM (BLOCK_M×BLOCK_N×BLOCK_K).  Accumulation is performed
   in fp32 and the result is stored as fp16.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to the original `dim`.
7. Cast back to the input dtype (normally float32) and return.

Only the reduction (step 5) is Triton‑coded; all other ops use
high‑performance PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
# --------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr,                # fp16 * (M, N, N)
    b_ptr,                # fp16 * (M, N, N)
    c_ptr,                # fp16 * (M, N, N)
    N,                    # matrix dimension (square)
    stride_am, stride_ai, stride_ak,   # strides for A (batch, row, col)
    stride_bm, stride_bk, stride_bj,   # strides for B (batch, row=k, col=j)
    stride_cm, stride_ci, stride_cj,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)          # batch‑slice index (M = B*H)
    pid_i = tl.program_id(1)        # block row
    pid_j = tl.program_id(2)        # block col

    # ----- block start coordinates -----
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # ----- offsets inside the block -----
    i_off = i_base + tl.arange(0, BLOCK_M)
    j_off = j_base + tl.arange(0, BLOCK_N)

    # ----- validity masks for i/j (handle tail) -----
    i_mask = i_off < N
    j_mask = j_off < N

    # ----- accumulator in fp32 -----
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over k dimension in blocks -----
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_base = k * BLOCK_K
        k_off = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # A block: (i,k)
        a_ptrs = (
            a_ptr
            + pid * stride_am
            + (i_off[:, None] * stride_ai)
            + (k_off[None, :] * stride_ak)
        )
        a = tl.load(
            a_ptrs,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0,
        ).to(tl.float32)

        # B block (transposed): we need Bᵀ → (k,j)
        b_ptrs = (
            b_ptr
            + pid * stride_bm
            + (k_off[:, None] * stride_bk)   # row = k
            + (j_off[None, :] * stride_bj)   # col = j
        )
        b = tl.load(
            b_ptrs,
            mask=k_mask[:, None] & j_mask[None, :],
            other=0.0,
        ).to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # ----- write back result (store as fp16) -----
    c_ptrs = (
        c_ptr
        + pid * stride_cm
        + (i_off[:, None] * stride_ci)
        + (j_off[None, :] * stride_cj)
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=i_mask[:, None] & j_mask[None, :],
    )


def _batched_gemm_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    block_m: int = 64,
    block_n: int = 64,
    block_k: int = 64,
) -> torch.Tensor:
    """
    Wrapper around the Triton batched GEMM kernel.
    A, B: fp16 tensors of shape (M, N, N) – contiguous.
    Returns C = A @ Bᵀ with shape (M, N, N) (fp16).
    """
    assert A.is_contiguous() and B.is_contiguous()
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    M, N, K = A.shape
    assert N == K, "Matrices must be square."

    C = torch.empty_like(A)

    # Strides (in elements)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bk, stride_bj = B.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    grid = (
        M,
        (N + block_m - 1) // block_m,
        (N + block_n - 1) // block_n,
    )
    _batched_gemm_kernel[grid](
        A,
        B,
        C,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bj,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return C


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] (binary) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Load linear weights (fp16, no bias)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections
    # --------------------------------------------------------------
    left_proj   = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj  = torch.nn.functional.linear(x, right_proj_w)

    left_gate   = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate  = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate    = torch.nn.functional.linear(x, out_gate_w).sigmoid()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 5. Apply gates and optional mask
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape (B,N,N) -> broadcast over hidden dim
        mask = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left  = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM (A @ Bᵀ) on (B*H) matrices.
    # --------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B*H, N, N)  – contiguous layout required by the kernel
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Triton batched GEMM
    out_flat = _batched_gemm_triton(
        left_flat, right_flat, block_m=64, block_n=64, block_k=64
    )  # (B*H, N, N), fp16

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Output LayerNorm, gating, final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 8. Cast back to original dtype and return
    # --------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)
    return out
