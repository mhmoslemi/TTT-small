def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Norm
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)

    # Compute normalized input (using float32 then cast)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # Load weight tensors and concatenate
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    # all shape (hidden_dim, dim), ensure dtype half
    w_list = [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w]
    # We will reorder to match earlier splitting: left_proj, left_gate, right_proj, right_gate, out_gate
    # Actually we need left_proj, left_gate, right_proj, right_gate, out_gate in that order
    combined_weight = torch.cat([w.to(device).to(torch.float16) for w in w_list], dim=0)  # shape (5*hidden_dim, dim)

    # Linear projection (combined)
    combined = torch.nn.functional.linear(x, combined_weight)  # shape (B,N,N,5*H)

    H = hidden_dim
    left = combined[..., 0:H]
    left_gate = combined[..., H:2*H]
    right = combined[..., 2*H:3*H]
    right_gate = combined[..., 3*H:4*H]
    out_gate = torch.sigmoid(combined[..., 4*H:5*H])

    # Apply mask and gate
    if mask is not None and not nomask:
        # Launch triton kernel for left/right gating with mask
        # Preprocess mask to half and flatten
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # Flatten all
        total = left.numel()
        BLOCK = 1024

        # Flatten views
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_fp.view(-1)  # shape (B*N*N,)
        # Define kernel
        @triton.jit
        def _gate_mask_kernel(left_ptr, right_ptr,
                              left_gate_ptr, right_gate_ptr,
                              mask_ptr,
                              total_elems, hidden_dim,
                              HAS_MASK: tl.constexpr,
                              BLOCK_SIZE: tl.constexpr):
            pid = tl.program_id(0)
            offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
            mask = offs < total_elems
            # Load
            left = tl.load(left_ptr + offs, mask=mask, other=0.0)
            right = tl.load(right_ptr + offs, mask=mask, other=0.0)
            left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
            right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
            if HAS_MASK:
                idx = offs // hidden_dim
                mask_val = tl.load(mask_ptr + idx, mask=mask, other=1.0)
                left = left * left_gate * mask_val
                right = right * right_gate * mask_val
            else:
                left = left * left_gate
                right = right * right_gate
            tl.store(left_ptr + offs, left, mask=mask)
            tl.store(right_ptr + offs, right, mask=mask)

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
        _gate_mask_kernel[grid](
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            total, hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
    else:
        left = left * left_gate
        right = right * right_gate

    # Prepare for batched matrix multiplication
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))  # (B*H,N,N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,H)

    # Output layer norm
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply output gate and final linear
    out = out * out_gate  # elementwise half

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # Cast back to original dtype (likely float32)
    out = out.to(input_tensor.dtype)

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    if HAS_MASK:
        idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    This implementation fuses the five linear projections (two projections,
    two gate projections and the output gate) into a single matrix
    multiplication, applies a Triton kernel to fuse element‑wise mask and
    gate multiplications, and uses a batched GEMM for the core O(N³·H)
    contraction.  All heavy compute is performed in fp16; the final result
    is returned in the input dtype.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)  # half precision for the compute‑heavy path

    # -----------------------------------------------------------------
    # 2. Fuse all linear projections (left, right, gates, out_gate)
    # -----------------------------------------------------------------
    # Gather weights, convert to fp16, and concatenate.
    w_left_proj = weights["left_proj.weight"].to(device).to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(device).to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(device).to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(device).to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(device).to(torch.float16)

    # Order: left_proj, left_gate, right_proj, right_gate, out_gate
    combined_weight = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate], dim=0
    )  # shape (5*hidden_dim, dim)

    # Single linear call
    combined = F.linear(x, combined_weight)  # (B, N, N, 5*H)

    H = hidden_dim
    left = combined[..., 0:H]
    left_gate = combined[..., H:2*H]
    right = combined[..., 2*H:3*H]
    right_gate = combined[..., 3*H:4*H]
    out_gate = torch.sigmoid(combined[..., 4*H:5*H])

    # -----------------------------------------------------------------
    # 3. Fuse mask and gate multiplications with a Triton kernel
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        total = left.numel()
        BLOCK = 1024
        # Flatten tensors for the kernel
        left_f = left.view(-1)
        right_f = right.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)
        mask_f = mask_fp.view(-1)  # (B*N*N,)

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
        _gate_mask_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            H,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back (already in-place)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 4. Core O(N³·H) contraction: batched GEMM (left @ rightᵀ)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 5. Output LayerNorm, output gate and final projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply learned output gate (half precision)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # back to original dim

    # -----------------------------------------------------------------
    # 6. Cast back to the original dtype
    # -----------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
