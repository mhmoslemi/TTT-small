@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_ah, stride_ak, stride_aa,
    stride_bk, stride_bn, stride_bb,
    stride_ch, stride_cm, stride_cn,
    BATCH, M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0) # index for batch dimension
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer offsets for A and B
    a_ptrs = a_ptr + batch * stride_aa + (offs_m[:, None] * stride_ah) + (tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_ptr + batch * stride_bb + (tl.arange(0, BLOCK_K)[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offsets for this block
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K - k * BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] < K - k * BLOCK_K) & mask_n[None, :], other=0.0)
        # dot
        acc += tl.dot(a, b)

        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c_ptrs = c_ptr + batch * stride_cc + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections (no bias) → left/right (hidden_dim) and three gating
   projections.  Gating values are passed through a sigmoid.
4. Fuse mask, projection and gate in a small Triton kernel
   (only when a mask is present).  The kernel computes
        proj * gate * mask   (mask broadcast over the hidden dim).
5. Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM in Triton where each batch corresponds to a
   (batch, hidden) pair.
6. Layer‑norm over the hidden dimension, multiply by the output gate and
   finally linearly project back to `dim`.
7. Cast the result back to the original dtype (usually float32).

At least one operation (mask‑gate fusion) and the main reduction are performed
with Triton kernels for speed.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: element‑wise proj * gate * mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B,N,N,H)
    gate_ptr,          # fp16 * (B,N,N,H)
    mask_ptr,          # fp16 * (B,N,N)
    out_ptr,           # fp16 * (B,N,N,H)
    total_elems,       # B*N*N*H
    hidden_dim,        # H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # each hidden_dim consecutive entries share the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse projection, gate and mask using Triton.
    All tensors must be fp16, contiguous and have shapes:
        proj, gate : (B, N, N, H)
        mask       : (B, N, N)   (binary 0/1)
    Returns: proj * gate * mask  (broadcast over H).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B) where each batch is (b, hidden)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column
    batch_id = tl.program_id(2)       # which (batch, hidden) pair

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        k_offs = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offs < K

        # A: (M, K)
        a_ptrs = a_ptr \
                 + batch_id * stride_a_batch \
                 + (offs_m[:, None] * stride_a_m) \
                 + (k_offs[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B: (K, N)
        b_ptrs = b_ptr \
                 + batch_id * stride_b_batch \
                 + (k_offs[:, None] * stride_b_k) \
                 + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)   # FP32 accumulation

    # write back
    c_ptrs = c_ptr \
             + batch_id * stride_c_batch \
             + (offs_m[:, None] * stride_c_m) \
             + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm(
    a: torch.Tensor,
    b: torch.Tensor,
    block_m: int = 64,
    block_n: int = 64,
    block_k: int = 64,
) -> torch.Tensor:
    """
    Batched GEMM using Triton.
    a : (B_batch, M, K)   fp16, contiguous
    b : (B_batch, K, N)   fp16, contiguous
    Returns c : (B_batch, M, N)   fp16, contiguous
    """
    B_batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "inner dimensions must match"

    c = torch.empty((B_batch, M, N), dtype=a.dtype, device=a.device)

    grid = (
        triton.cdiv(M, block_m),
        triton.cdiv(N, block_n),
        B_batch,
    )

    _batched_gemm_kernel[grid](
        a,
        b,
        c,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        c.stride(0),
        c.stride(1),
        c.stride(2),
        M,
        N,
        K,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return c

    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast linear weights to FP16 (no bias)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections
    # --------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gating projections + sigmoid
    # --------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Apply mask and gating (fused via Triton if mask present)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Free intermediate tensors that are no longer needed
    left_proj = left_gate = None
    right_proj = right_gate = None

    # --------------------------------------------------------------
    # 7. Core reduction: batched GEMM (left @ right^T) via Triton
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) where A is (i,k) and B is (k,j)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # (B*H, M=N, K=N)
    b = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)         # (B*H, K=N, N=j)

    # Triton batched GEMM
    out_flat = _batched_gemm(a, b,
                             block_m=64,
                             block_n=64,
                             block_k=64)                                 # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
