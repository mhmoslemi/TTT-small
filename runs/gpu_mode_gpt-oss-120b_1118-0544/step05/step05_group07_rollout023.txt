"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) with the provided LN parameters.
2. Cast to fp16 for the compute‑intensive part.
3. Fuse the projection and its gate for the left and right streams:
   * Linear(x, [W_proj; W_gate]) → (B,N,N,2·H)  (out‑features stacked).
   * Split the result into projection and gate tensors and apply sigmoid to the gates.
   * The mask (if present) and the gate are fused element‑wise by a tiny Triton kernel.
4. Core O(N³·H) reduction:
   * Reshape the masked left/right tensors to (B·H, N, N) and perform a batched GEMM
     using torch.bmm: out = left @ rightᵀ.
   * Reshape back to (B,N,N,H).
5. Second LayerNorm on the hidden dimension, multiplication by the output gate,
   and a final linear projection back to the original channel dimension.
6. Cast the result back to the input dtype (normally float32).

Only the element‑wise mask‑gate fusion is done in Triton; the large linear
projections and the batched GEMM use PyTorch’s highly‑optimized kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: (proj * gate * mask) – fused element‑wise operation
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (total_elems,)
    gate_ptr,          # fp16 * (total_elems,)
    mask_ptr,          # fp16 * (mask_len = total_elems // hidden_dim,)
    out_ptr,           # fp16 * (total_elems,)
    total_elems,       # i32 – total number of elements in proj/gate (B*N*N*H)
    hidden_dim,        # i32 – H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # Index of the corresponding mask element (broadcast over hidden dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=tl.float16(0))
    gate = tl.load(gate_ptr + offs, mask=mask, other=tl.float16(0))
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1))

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using Triton.

    All tensors must be on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------- LayerNorm (input) -------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for compute‑heavy parts
    x = x.to(torch.float16)

    # --------------------- cast weights to fp16 -------------------------
    # Stack proj & gate weights to fuse two linear layers into one
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # shape (2*H, C)
    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # shape (2*H, C)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)          # (H, C)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)    # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)        # (H,)
    to_out_weight = weights["to_out.weight"].to(torch.float16)              # (C, H)

    # ------------------------ linear + gate ------------------------------
    # Left stream: projection + gate (stacked)
    left_comb = torch.nn.functional.linear(x, left_comb_weight)  # (B,N,N,2*H)
    left_proj = left_comb[..., :hidden_dim]                     # (B,N,N,H)
    left_gate = left_comb[..., hidden_dim:].sigmoid()           # (B,N,N,H)

    # Right stream: projection + gate (stacked)
    right_comb = torch.nn.functional.linear(x, right_comb_weight)  # (B,N,N,2*H)
    right_proj = right_comb[..., :hidden_dim]                     # (B,N,N,H)
    right_gate = right_comb[..., hidden_dim:].sigmoid()           # (B,N,N,H)

    # Output gate
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()  # (B,N,N,H)

    # ------------------- mask & gate fusion (Triton) --------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ----------------- core O(N³·H) reduction via batched GEMM ----------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_reshaped = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N)

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    out_reshaped = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_reshaped.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_weight)  # (B,N,N,dim)

    # ------------------------ cast back to input dtype -----------------
    out = out.to(dtype_in)

    return out
