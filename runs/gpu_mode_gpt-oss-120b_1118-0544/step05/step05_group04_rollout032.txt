    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Abatch, stride_Arow, stride_Acol,
    stride_Bbatch, stride_Brow, stride_Bcol,
    stride_Cbatch, stride_Crow, stride_Ccol,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(2)  # one per batch matrix
    pid_m = tl.program_id(0)      # row tile
    pid_n = tl.program_id(1)      # col tile

    # Compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Masks for bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to start of A and B for this batch
    a_ptrs = A_ptr + batch_idx * stride_Abatch
    b_ptrs = B_ptr + batch_idx * stride_Bbatch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A tile (BLOCK_M x BLOCK_K)
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(a_ptrs + offs_m[:, None] * stride_Arow + offs_k[None, :] * stride_Acol,
                    mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (BLOCK_N x BLOCK_K) of B^T (i.e., row j, col k)
        b = tl.load(b_ptrs + offs_n[:, None] * stride_Brow + offs_k[None, :] * stride_Bcol,
                    mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

        # Advance pointers to next k block
        a_ptrs += BLOCK_K * stride_Acol
        b_ptrs += BLOCK_K * stride_Bcol

    # Write back C
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + batch_idx * stride_Cbatch + offs_m[:, None] * stride_Crow + offs_n[None, :] * stride_Ccol
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1️⃣ Layer‑normalize the [B,N,N,C] input (C = dim) in FP32.
2️⃣ Cast to FP16 and apply two linear projections → left/right (shape B,N,N,H).
3️⃣ Compute three sigmoid gates (left, right, out) from the same normalized tensor.
4️⃣ Fuse mask (if any) with left/right gates in a tiny Triton kernel.
5️⃣ Perform the core reduction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a *batched* matrix‑multiply (B*H independent GEMMs of size N×N) using a
   custom Triton kernel (accumulation in FP32, result in FP16).
6️⃣ Reshape back to (B,N,N,H), apply a second LayerNorm over H, multiply by the
   output gate and project back to the original channel dimension.
7️⃣ Cast the result back to the original dtype.

Only the mask‑gate fusion and the batched GEMM are written in Triton; all other
operations use PyTorch’s highly‑optimized kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. Mask + gate fusion (left/right)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # const fp16* left   (B,N,N,H) flattened
    right_ptr,         # const fp16* right
    mask_ptr,          # const fp16* mask   (B,N,N) flattened
    left_gate_ptr,     # const fp16* left_gate
    right_gate_ptr,    # const fp16* right_gate
    left_out_ptr,      # fp16* left_out
    right_out_ptr,     # fp16* right_out
    H: tl.constexpr,   # hidden dimension
    total: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """out_left = left * mask * left_gate ; out_right = right * mask * right_gate"""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # load
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # mask is per (B,N,N), broadcast over H
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    tl.store(left_out_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute C[m,i,j] = Σ_k A[m,i,k] * B[m,j,k]  (B is NOT transposed, we dot over k)."""
    batch_id = tl.program_id(2)               # 0 .. M-1
    pid_m   = tl.program_id(0)                # row tile of output
    pid_n   = tl.program_id(1)                # col tile of output

    # ---------- tile start indices ----------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = A_ptr + batch_id * stride_Abatch
    b_batch = B_ptr + batch_id * stride_Bbatch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------- loop over K dimension ----------
    for k_iter in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile : (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch + offs_m[:, None] * stride_Arow + offs_k[None, :] * stride_Acol,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile : (BLOCK_N, BLOCK_K)   (note: we dot over k, no transpose needed)
        b = tl.load(
            b_batch + offs_n[:, None] * stride_Brow + offs_k[None, :] * stride_Bcol,
            mask=mask_n[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Dot product and accumulate (cast to FP32 for better precision)
        acc += tl.dot(a, b).to(tl.float32)

    # ---------- write back ----------
    c = acc.to(tl.float16)
    c_ptrs = (
        C_ptr
        + batch_id * stride_Cbatch
        + offs_m[:, None] * stride_Crow
        + offs_n[None, :] * stride_Ccol
    )
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 3. Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights      : dict of model parameters (float32 tensors)
            - config       : dict with keys
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – if True, ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ----------------------- unpack -----------------------
    inp, mask, weights, cfg = data
    device = inp.device
    orig_dtype = inp.dtype

    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    # ----------------------- 1. LayerNorm -----------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # work in FP16 for the heavy part
    x = x.to(torch.float16)

    # ----------------------- 2. Linear projections -----------------------
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    left = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w) # (B,N,N,H)

    # ----------------------- 3. Gates -----------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ----------------------- 4. Mask & gate fusion -----------------------
    B, N, _, _ = left.shape
    if mask is None or nomask:
        mask_fp = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp = mask.to(torch.float16)

    total = left.numel()                     # B * N * N * H
    BLOCK = 1024

    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = (triton.cdiv(total, BLOCK),)
    _mask_gate_fusion_kernel[grid](
        left,
        right,
        mask_fp.view(-1),
        left_gate,
        right_gate,
        left_out,
        right_out,
        hidden,
        total,
        BLOCK,
        num_warps=4,
    )

    # ----------------------- 5. Core reduction (batched GEMM) -----------------------
    # reshape to (M, N, N) where M = B * hidden
    left_flat = left_out.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_flat = right_out.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # allocate output buffer
    out_flat = torch.empty_like(left_flat)

    # strides (in elements)
    sA_batch, sA_row, sA_col = left_flat.stride()
    sB_batch, sB_row, sB_col = right_flat.stride()
    sC_batch, sC_row, sC_col = out_flat.stride()

    # block sizes (tuned for Hopper/RTX etc.)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),     # rows
        triton.cdiv(N, BLOCK_N),     # cols
        B * hidden,                  # batch * hidden
    )
    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * hidden,
        N,
        N,
        sA_batch,
        sA_row,
        sA_col,
        sB_batch,
        sB_row,
        sB_col,
        sC_batch,
        sC_row,
        sC_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # reshape back to (B,N,N,H)
    out = (
        out_flat.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)      # (B,N,N,H)
        .contiguous()
    )

    # ----------------------- 6. Output LayerNorm over hidden dim -----------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----------------------- 7. Apply output gate -----------------------
    out = out * out_gate

    # ----------------------- 8. Final linear projection back to dim -----------------------
    out = F.linear(out, to_out_w)  # (B,N,N,dim)

    # ----------------------- 9. Cast back to original dtype -----------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
