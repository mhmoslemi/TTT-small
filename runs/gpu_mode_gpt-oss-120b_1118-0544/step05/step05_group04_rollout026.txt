    stride_b_k = right_flat.stride(1)
    stride_b_n = right_flat.stride(2)

    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    BLOCK_K = 32

        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply out_gate
    out = out * out_gate

    # Final projection to dim
    out = F.linear(out, to_out_w)

    # Cast back to original dtype if needed
    out = out.to(dtype)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C).
2. Linear projections to a hidden dimension H (left & right) and three sigmoid gates.
3. Fuse mask (if any) and left/right gates in a Triton kernel:
   left_out  = left  * mask * left_gate
   right_out = right * mask * right_gate
4. Compute the core O(N³·H) reduction:
   out[b,i,j,h] = Σ_k left_out[b,i,k,h] * right_out[b,j,k,h]
   Implemented as a batched GEMM (B*H independent N×N matrix multiplies) in Triton.
5. Reshape back to (B,N,N,H), apply a second LayerNorm over H, multiply by
   the output gate, and project back to the original channel dimension C.
6. Cast the result back to the input dtype.

Only the mask‑gate fusion and the batched GEMM are written in Triton;
all other ops use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------


@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,          # const * left  (B,N,N,H) flattened
    right_ptr,         # const * right (B,N,N,H) flattened
    mask_ptr,          # const * mask  (B,N,N) flattened
    left_gate_ptr,     # const * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # const * right_gate (B,N,N,H) flattened
    left_out_ptr,      # * left_out
    right_out_ptr,     # * right_out
    H: tl.constexpr,   # hidden dimension
    TOTAL_ELEMS: tl.constexpr,  # total number of elements in left/right (B*N*N*H)
    BLOCK_SIZE: tl.constexpr,   # threads per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL_ELEMS

    # Load tensors (fp16)
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Derive mask index (mask is without H dimension)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Fuse mask and gates
    tl.store(left_out_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_az, stride_am, stride_ak,   # A: (batch, M, K)
    stride_bz, stride_bk, stride_bn,   # B: (batch, K, N)
    stride_cz, stride_cm, stride_cn,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B, where A is (batch, M, K) and B is (batch, K, N)."""
    pid_z = tl.program_id(2)           # batch index
    pid_m = tl.program_id(0)           # block row index
    pid_n = tl.program_id(1)           # block col index

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers with batch offset
    a_batch = a_ptr + pid_z * stride_az
    b_batch = b_ptr + pid_z * stride_bz
    c_batch = c_ptr + pid_z * stride_cz

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a = tl.load(
            a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (K x N)
        b = tl.load(
            b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Compute partial dot product
        acc += tl.dot(a, b)

    # Write result (fp16)
    c = acc.to(tl.float16)
    tl.store(
        c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask        : torch.Tensor of shape [B, N, N] or None (binary)
            - weights     : dict of model parameters (float32 tensors)
            - config      : dict with keys
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask"       – if True, ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ----------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------
    # 1) LayerNorm on the input (float32 for stability)
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # ----------------------------------------------------------
    # 2) Cast to fp16 for the heavy computation
    # ----------------------------------------------------------
    x = x.to(torch.float16)

    # ----------------------------------------------------------
    # 3) Load weights (fp16)
    # ----------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ----------------------------------------------------------
    # 4) Linear projections to hidden dimension H
    # ----------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ----------------------------------------------------------
    # 5) Compute sigmoid gates
    # ----------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ----------------------------------------------------------
    # 6) Fuse mask and gates (Triton)
    # ----------------------------------------------------------
    B, N, _, H = left.shape
    total_elems = left.numel()                     # B * N * N * H

    if mask is not None and not nomask:
        mask_tensor = mask.to(torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    BLOCK = 1024  # threads per program
    grid = (triton.cdiv(total_elems, BLOCK),)
    _apply_mask_gate_kernel[grid](
        left,                     # left_ptr
        right,                    # right_ptr
        mask_tensor.view(-1),     # mask_ptr (flattened)
        left_gate,                # left_gate_ptr
        right_gate,               # right_gate_ptr
        left_out,                 # left_out_ptr
        right_out,                # right_out_ptr
        hidden_dim,               # H
        total_elems,              # TOTAL_ELEMS
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ----------------------------------------------------------
    # 7) Core reduction: batched GEMM (left @ rightᵀ)
    #    Shape after permute: (B, H, N, N)
    # ----------------------------------------------------------
    left_perm = left_out.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N) -> (i,k)
    right_perm = right_out.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N) -> (j,k)
    right_perm_T = right_perm.transpose(2, 3).contiguous()  # (B, H, N, N) -> (k,j)

    batch = B * hidden_dim
    left_flat = left_perm.view(batch, N, N)      # (batch, M=N, K=N)
    right_flat = right_perm_T.view(batch, N, N)  # (batch, K=N, N=N)
    out_flat = torch.empty_like(left_flat)       # (batch, N, N)

    # Strides for the kernel (int64)
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(1)
    stride_b_n = right_flat.stride(2)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Tiling parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # dim M
        triton.cdiv(N, BLOCK_N),   # dim N
        batch,                     # batch dimension
    )
    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,               # M
        N,               # N
        N,               # K
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ----------------------------------------------------------
    # 8) Reshape back to (B, N, N, H)
    # ----------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ----------------------------------------------------------
    # 9) Output LayerNorm over hidden dimension
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----------------------------------------------------------
    # 10) Apply output gate
    # ----------------------------------------------------------
    out = out * out_gate

    # ----------------------------------------------------------
    # 11) Final projection back to original dim
    # ----------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ----------------------------------------------------------
    # 12) Cast back to original dtype if needed
    # ----------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
