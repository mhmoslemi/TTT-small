@triton.jit
def apply_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    idx = offs
    # compute mask index: each mask element repeats hidden_dim times
    mask_idx = offs // hidden_dim
    proj = tl.load(proj_ptr + idx, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + idx, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    out = proj * gate * m
    tl.store(out_ptr + idx, out, mask=mask)

def apply_gate_and_mask(proj, gate, mask, block_size=1024):
    B,N,N,H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    apply_gate_mask_kernel[grid](
        proj, gate, mask, out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated mask/gate fusion +
heavy linear‑projection fusion.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for compute‑heavy ops.
3. Fuse the four linear projections (left/right *proj* and *gate*) into a
   single matrix multiplication:
       [left_proj , left_gate , right_proj , right_gate] =
           x_flat @ W_comb   (W_comb = concat of the four weight matrices)
   Split the result and apply sigmoid to the two gate parts.
4. Fuse element‑wise mask, gating and optional masking into a single
   Triton kernel (one launch per side).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM with torch.bmm on the (B·H, N, N) tensors.
6. Output LayerNorm, multiply by `out_gate`, and a final linear projection
   back to the original dimension.
7. Cast back to the input dtype.

Only the mask‑gate fusion is performed in Triton; all other heavy
linear algebra uses cuBLAS via `torch.nn.functional.linear`/`torch.bmm`,
which is optimal on the B200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: fused (proj * gate * mask) – mask broadcast over H
# --------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (total,)
    gate_ptr,          # fp16 * (total,)
    mask_ptr,          # fp16 * (mask_total,)
    out_ptr,           # fp16 * (total,)
    total_elems,       # int: total number of proj elements (B*N*N*H)
    hidden_dim,        # int: H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # each mask element repeats `hidden_dim` times in the proj/gate tensors
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 65536,
) -> torch.Tensor:
    """
    proj, gate : (B, N, N, H)   fp16, contiguous
    mask       : (B, N, N)      fp16, contiguous
    Returns proj * gate * mask  (shape (B,N,N,H)).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32/float16)
            - mask         : torch.Tensor [B, N, N]  (or None)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor [B, N, N, C] (same dtype as input).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy part
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Fuse the four linear projections (left/right *proj* and *gate*)
    # --------------------------------------------------------------
    # Reshape for a single matmul
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)

    # Concatenate the four weight matrices (each shape (hidden, dim))
    w_left_proj  = weights["left_proj.weight"].to(torch.float16)
    w_left_gate  = weights["left_gate.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)

    weight_comb = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate], dim=0
    )                                             # (hidden*4, dim)

    # Single linear pass
    proj_gate = F.linear(x_flat, weight_comb)    # (B*N*N, hidden*4)

    # Split results
    left_proj  = proj_gate[:, 0 * hidden : 1 * hidden]
    left_gate  = proj_gate[:, 1 * hidden : 2 * hidden]
    right_proj = proj_gate[:, 2 * hidden : 3 * hidden]
    right_gate = proj_gate[:, 3 * hidden : 4 * hidden]

    # Apply sigmoid to gate tensors
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # --------------------------------------------------------------
    # 4. Optional mask & gate fusion (Triton)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(
            left_proj.view(B, N, N, hidden),
            left_gate.view(B, N, N, hidden),
            mask_fp,
        )
        right = _apply_gate_and_mask(
            right_proj.view(B, N, N, hidden),
            right_gate.view(B, N, N, hidden),
            mask_fp,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        left = left.view(B, N, N, hidden)
        right = right.view(B, N, N, hidden)

    # --------------------------------------------------------------
    # 5. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM with torch.bmm.
    # --------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, hidden)
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output gate
    # --------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(torch.float16)  # (hidden, dim)
    out_gate_flat = F.linear(x_flat, w_out_gate)               # (B*N*N, hidden)
    out_gate = torch.sigmoid(out_gate_flat).view(B, N, N, hidden)

    # --------------------------------------------------------------
    # 7. Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # Final linear back to `dim`
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)
    out = F.linear(out, to_out_w)                          # (B, N, N, dim)

    # --------------------------------------------------------------
    # 8. Cast back to the original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
