```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for projecting + gating + optional mask
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,      # fp16 *
    gate_ptr,      # fp16 *
    mask_ptr,      # fp16 *
    out_ptr,       # fp16 *
    total_elems,   # int
    hidden_dim,    # int
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask_valid = offs < total_elems

    # mask index is broadcast over hidden_dim
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask_valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask_valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=mask_valid)

def _apply_gate_and_mask(proj, gate, mask, block_size=2048):
    """
    Fuse elementwise projection, gate and mask.
    All tensors must be contiguous fp16.
    proj, gate: (B,N,N,H)
    mask: (B,N,N) broadcast over H.
    Returns tensor of shape (B,N,N,H).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out

# Triton kernel for the batched GEMM of the outgoing TriMul:
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr,    # fp16 *
    right_ptr,   # fp16 *
    out_ptr,     # fp16 *
    N: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)   # batch * hidden
    pid_m = tl.program_id(0)   # row block
    pid_n = tl.program_id(1)   # col block

    # Offsets for rows and cols within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out-of-bounds rows/cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Compute batch offset (each batch is N*N elements)
    batch_offset = pid_b * N * N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, N, BLOCK_K):
        cur_K = tl.minimum(BLOCK_K, N - k)
        offs_k = k + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = left_ptr + batch_offset + offs_m[:, None] * N + offs_k[None, :]
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < N), other=0.0)

        # Load B_T tile (K x N) where B_T[k, n] = right[n, k]
        b_ptrs = right_ptr + batch_offset + offs_k[:, None] * 1 + offs_n[None, :] * N
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < N) & mask_n[None, :], other=0.0)

        # Cast to float32 for accumulation
        a_f32 = a.to(tl.float32)
        b_f32 = b.to(tl.float32)

        acc += tl.dot(a_f32, b_f32)   # (BLOCK_M, BLOCK_N)

    # Write result
    c_ptrs = out_ptr + batch_offset + offs_m[:, None] * N + offs_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def _tri_mul_outgoing(left, right, block_m=64, block_n=64, block_k=32):
    """
    Batched GEMM for outgoing TriMul.
    left, right: tensors of shape (B, N, N, H) in fp16, contiguous.
    Returns tensor of same shape.
    """
    B, N, _, H = left.shape
    # Permute to (B, H, N, N) and flatten batch*H
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch_h = B * H
    left_flat = left_perm.view(batch_h, N, N)
    right_flat = right_perm.view(batch_h, N, N)
    out_flat = torch.empty_like(left_flat)

    grid = (
        triton.cdiv(N, block_m),
        triton.cdiv(N, block_n),
        batch_h,
    )
    _tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    # Restore shape (B, H, N, N) -> (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
        - mask         : torch.Tensor [B, N, N] (binary) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # 1. LayerNorm on channel dimension
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for compute‑heavy part
    x = x.to(torch.float16)

    # 2. Cast weights to fp16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # 3. Linear projections
    left_proj = F.linear(x, left_proj_w)      # (B,N,N,hidden)
    right_proj = F.linear(x, right_proj_w)

    # 4. Gates (sigmoid)
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # 5. Fuse projection, gate and optional mask
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 6. Core reduction (batched GEMM)
    out = _tri_mul_outgoing(left, right)

    # 7. Output LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 8. Apply out_gate and final linear projection
    out = out * out_gate
    out = F.linear(out, to_out_w)

    # 9. Cast back to original dtype
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated core reduction.

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to fp16 for compute‑heavy ops.
3. Linear projections → left/right (B,N,N,H) and gate projections → sigmoid.
4. Fuse projection × gate × optional mask with a small Triton kernel.
5. Batched GEMM:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton kernel that processes each (b,h) pair as
   an independent matrix multiply (A @ Bᵀ) with fp16 inputs and fp32
   accumulation.
6. Layer‑norm over the hidden dimension, multiply by out‑gate and a final
   linear layer back to the original channel dimension.
7. Cast the result back to the input dtype.

Only the element‑wise mask/gate fusion and the large O(N³) reduction are
executed on Triton; everything else uses PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse projection, gate and (optional) mask
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,      # fp16 *
    gate_ptr,      # fp16 *
    mask_ptr,      # fp16 *
    out_ptr,       # fp16 *
    total_elems,   # int
    hidden_dim,    # int
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask is broadcast over the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 2048) -> torch.Tensor:
    """
    Fuse (proj * gate * mask) using Triton.
    All tensors must be fp16, contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N) – broadcasted over H
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _apply_gate_mask_kernel[grid](
        proj, gate, mask, out,
        total, proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the outgoing TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr,    # fp16 *
    right_ptr,   # fp16 *
    out_ptr,     # fp16 *
    N: tl.constexpr,          # sequence length (compile‑time constant)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)          # batch * hidden index
    pid_m = tl.program_id(0)          # row block index
    pid_n = tl.program_id(1)          # col block index

    # row / col indices of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # each (b, h) sub‑matrix occupies N*N contiguous elements
    batch_offset = pid_b * N * N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction dimension k
    for k in range(0, N, BLOCK_K):
        cur_k = tl.minimum(BLOCK_K, N - k)
        offs_k = k + tl.arange(0, BLOCK_K)

        # ------------------- load A tile (M × K) -------------------
        a_ptrs = left_ptr + batch_offset + offs_m[:, None] * N + offs_k[None, :]
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < N),
            other=0.0,
        )

        # ------------------- load Bᵀ tile (K × N) -------------------
        # Bᵀ[k, n] = right[n, k]
        b_ptrs = right_ptr + batch_offset + offs_k[:, None] * 1 + offs_n[None, :] * N
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < N) & mask_n[None, :],
            other=0.0,
        )

        # fp16 → fp32 for accurate accumulation
        a_f32 = a.to(tl.float32)
        b_f32 = b.to(tl.float32)

        acc += tl.dot(a_f32, b_f32)   # (BLOCK_M, BLOCK_N)

    # ------------------- write result -------------------
    c_ptrs = out_ptr + batch_offset + offs_m[:, None] * N + offs_n[None, :]
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _tri_mul_outgoing(left: torch.Tensor, right: torch.Tensor,
                      block_m: int = 64, block_n: int = 64, block_k: int = 32) -> torch.Tensor:
    """
    Perform the core TriMul reduction using the Triton kernel above.
    left, right : (B, N, N, H) – fp16, contiguous.
    Returns tensor of the same shape.
    """
    B, N, _, H = left.shape

    # reshape to (B*H, N, N) for a simple batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    batch_h = B * H
    left_flat = left_perm.view(batch_h, N, N)
    right_flat = right_perm.view(batch_h, N, N)
    out_flat = torch.empty_like(left_flat)

    grid = (
        triton.cdiv(N, block_m),   # blocks along M
        triton.cdiv(N, block_n),   # blocks along N
        batch_h,                   # batch*hidden
    )
    _tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys
              * "dim"         – input channel dimension C
              * "hidden_dim" – hidden dimension H
              * "nomask"      – if True, mask is ignored (optional)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    orig_dtype = x.dtype
    x = x.to(torch.float16)  # fp16 for the heavy part

    # ------------------------------------------------------------------
    # 2. Cast weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)      # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse projection, gate and optional mask
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core reduction (batched GEMM)
    # ------------------------------------------------------------------
    out = _tri_mul_outgoing(left, right)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8. Multiply by out_gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
