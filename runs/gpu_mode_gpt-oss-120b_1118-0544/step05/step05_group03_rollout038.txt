@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,  # note b row stride is N (since original shape batch,N,N)
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
   # Compute program IDs
   pid = tl.program_id(0)
   total_m = tl.cdiv(M, BLOCK_M)
   total_n = tl.cdiv(N, BLOCK_N)
   batch = pid // (total_m * total_n)
   pid_bn = pid % (total_m * total_n)
   block_m = pid_bn // total_n
   block_n = pid_bn % total_n

   # Offsets
   offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
   offs_k = tl.arange(0, BLOCK_K)

   # Compute masks for bounds
   mask_m = offs_m < M
   mask_n = offs_n < N

   # Pointers base for this batch
   a_batch_ptr = a_ptr + batch * stride_a_batch
   b_batch_ptr = b_ptr + batch * stride_b_batch
   c_batch_ptr = c_ptr + batch * stride_c_batch

   # Initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # Loop over K dimension
   for k in range(0, tl.cdiv(K, BLOCK_K)):
       k_start = k * BLOCK_K
       cur_k = k_start + offs_k
       mask_k = cur_k < K

       # Pointers for tile A (M x K)
       a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_row) + (cur_k[None, :] * stride_a_col)
       # Pointers for tile B transposed: we need tile shape (K x N) where rows=k, cols=n
       # Original B shape (batch,N,K). So we want B[batch, n, k].
       b_ptrs = b_batch_ptr + (offs_n[None, :] * stride_b_row) + (cur_k[:, None] * stride_b_col)
       # Load with mask
       a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
       b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

       acc += tl.dot(a, b)

   # Cast to output dtype (fp16) and store

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # cast to fp16
    x = x.to(torch.float16)

    # cast weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # linear projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # shape (B,N,N,hidden_dim)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # gates
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # apply mask if present
    if mask is not None and not nomask:
        # broadcast mask to hidden_dim: just multiply
        mask = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Permute to (B, H, N, N)
    B, N, _, H = left.shape

    left_flat = left.permute(0, 3, 1, 2).contiguous()
    right_flat = right.permute(0, 3, 1, 2).contiguous()
    # reshape to (B*H, N, N)
    left_flat = left_flat.view(B * H, N, N)
    right_flat = right_flat.view(B * H, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # launch custom batched gemm kernel
    total_batches = B * H
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # compute grid size: number of blocks to cover M,N per batch
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = total_batches * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    # compute strides
    stride_a_batch = N * N  # because each batch matrix has N*N elements, each element is fp16
    stride_a_row = N  # row stride
    stride_a_col = 1
    stride_b_batch = N * N  # each batch of B also N*N
    stride_b_row = N
    stride_b_col = 1
    stride_c_batch = N * N
    stride_c_row = N
    stride_c_col = 1

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape out_flat to (B, H, N, N)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # output layernorm (dim hidden_dim)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # shape (B,N,N,dim)

    # Cast back to original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    total_m = tl.cdiv(M, BLOCK_M)
    total_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (total_m * total_n)
    pid_in_batch = pid % (total_m * total_n)
    block_m = pid_in_batch // total_n
    block_n = pid_in_batch % total_n

    # compute offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # create masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch_ptr = a_ptr + batch * stride_a_batch
    b_batch_ptr = b_ptr + batch * stride_b_batch
    c_batch_ptr = c_ptr + batch * stride_c_batch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loops over K
    num_k_iters = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_iters):
        k_start = k * BLOCK_K
        cur_k = k_start + offs_k
        mask_k = cur_k < K

        # pointers for A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_row) + (cur_k[None, :] * stride_a_col)
        # pointers for B (transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch_ptr + (offs_n[None, :] * stride_b_row) + (cur_k[:, None] * stride_b_col)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # write out
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_row) + (offs_n[None, :] * stride_c_col)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections (left / right) and three gate projections (left, right,
   out) – all in FP16.
4. Apply optional mask and fuse it with the gate (simple elementwise mul).
5. Core reduction: for each batch * hidden_dim compute
      out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]
   This is a batched matrix multiplication C = A @ Bᵀ where A and B have
   shape (batch, N, N).  Implemented with a custom Triton kernel that
   iterates over K‑tiles and accumulates in FP32.
6. Output LayerNorm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.
7. Cast back to the original dtype (normally FP32).

All heavy matrix‑multiplications are performed in FP16 with FP32 accumulation,
giving near‑peak Tensor‑core performance on the B200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B shape: (B, N, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to FP16 tensors
    M, N, K,                           # matrix sizes (M=N=K=seq_len)
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    total_m = tl.cdiv(M, BLOCK_M)
    total_n = tl.cdiv(N, BLOCK_N)

    # batch index and block coordinates within the batch
    batch_id = pid // (total_m * total_n)
    pid_in_batch = pid % (total_m * total_n)
    block_m = pid_in_batch // total_n
    block_n = pid_in_batch % total_n

    # offsets for the current tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_offset = (offs_m[:, None] * stride_a_row) + (cur_k[None, :] * stride_a_col)
        a = tl.load(a_batch + a_offset,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Bᵀ tile: we need B[j, k] where j = column index, k = row index.
        # Original B layout: (batch, N, N) with row‑stride = N, col‑stride = 1.
        # Bᵀ[k, j] = B[j, k].
        b_offset = (offs_n[None, :] * stride_b_row) + (cur_k[:, None] * stride_b_col)
        b = tl.load(b_batch + b_offset,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # write result
    c = acc.to(tl.float16)
    c_offset = (offs_m[:, None] * stride_c_row) + (offs_n[None, :] * stride_c_col)
    tl.store(c_batch + c_offset,
             c,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper to launch the batched GEMM kernel
# ----------------------------------------------------------------------
def _batched_gemm(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a, b: tensors of shape (B, N, N) and dtype float16, contiguous.
    Returns c of the same shape and dtype.
    """
    assert a.shape == b.shape
    B, N, _ = a.shape
    c = torch.empty_like(a)

    # strides (in elements, not bytes)
    stride_a_batch = N * N
    stride_a_row = N
    stride_a_col = 1
    stride_b_batch = N * N
    stride_b_row = N
    stride_b_col = 1
    stride_c_batch = N * N
    stride_c_row = N
    stride_c_col = 1

    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32  # good trade‑off for B200

    total_m = (N + BLOCK_M - 1) // BLOCK_M
    total_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B * total_m * total_n
    grid = (total_blocks,)

    _batched_gemm_kernel[grid](
        a,
        b,
        c,
        N, N, N,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return c

# ----------------------------------------------------------------------
# Entry point: custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C]
            - mask         : torch.Tensor of shape [B, N, N] (or None)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys
                  * "dim" – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    dtype_in = input_tensor.dtype
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy ops
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all linear weights to FP16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B, N, N, H)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Apply mask and combine projection + gate
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B, N, N) -> broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16)
        mask_fp16 = mask_fp16.unsqueeze(-1)          # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction: batched GEMM C = A @ Bᵀ
    #    Shapes:
    #        left, right : (B, N, N, H)
    #        -> (B*H, N, N) after permute + view
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left = left.view(B * H, N, N)
    right = right.view(B * H, N, N)

    # batched GEMM in FP16, accumulation in FP32
    out = _batched_gemm(left, right)               # shape (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype if necessary
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
