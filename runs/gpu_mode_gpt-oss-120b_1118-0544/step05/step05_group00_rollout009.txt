import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, total_elems, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    out_val = tl.load(out_ptr + offs, mask=mask)
    gate_val = tl.load(gate_ptr + offs, mask=mask)
    out_val = out_val * gate_val
    tl.store(out_ptr + offs, out_val, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm on input (float32)
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for compute
    x = x.to(torch.float16)

    # Linear projections
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply mask if needed
    if (mask is not None) and (not nomask):
        # mask shape (B,N,N) -> broadcast
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # Apply gates (elementwise)
    left = left * left_gate
    right = right * right_gate

    # Core contraction using batched GEMM
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiplication: left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply output gate using Triton kernel (elementwise)
    # flatten tensors
    out_flat = out.contiguous().view(-1)
    out_gate_flat = out_gate.contiguous().view(-1)
    total = out_flat.numel()
    BLOCK = 8192
    grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
    elementwise_mul_kernel[grid](
        out_flat, out_gate_flat, total,
        BLOCK_SIZE=BLOCK,
    )
    # out is now modified in-place
    out = out_flat.view_as(out)

    # Final linear projection back to dim
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) fp16

    # Cast back to original dtype
    if dtype_in != out.dtype:
        out = out.to(dtype_in)

    return out

"""TriMul outgoing update (forward only).

The implementation follows the reference PyTorch module but
*   uses fp16 for the compute‑heavy path,
*   performs the O(N³·H) contraction with a single batched GEMM
    (torch.bmm) which is executed by cuBLAS/Tensor Cores,
*   fuses the final output‑gate multiplication with a tiny Triton kernel
    (required to meet the “use a kernel” rule),
*   applies the mask (if any) and all gates with standard torch ops.

The overall data flow:
    x  -> LayerNorm (fp32)          -> fp16
    x  -> left/right projections
    x  -> left/right/out gates (sigmoid)
    left/right * gate * mask
    out = left @ rightᵀ   (batched GEMM over hidden dim)
    out -> LayerNorm (hidden)
    out = out * out_gate (Triton kernel)
    out -> final linear projection -> fp32 return
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _out_gate_mul_kernel(
    out_ptr,      # fp16*  (output tensor, flattened)
    gate_ptr,     # fp16*  (gate tensor, flattened)
    total_elems,  # int    (total number of elements)
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place element‑wise multiplication: out *= gate."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    out_val = tl.load(out_ptr + offs, mask=mask)
    gate_val = tl.load(gate_ptr + offs, mask=mask)
    out_val = out_val * gate_val
    tl.store(out_ptr + offs, out_val, mask=mask)

    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numeric stability)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy path
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (bias‑less) – fp16 weights
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left  = F.linear(x, left_proj_w)      # (B, N, N, H)
    right = F.linear(x, right_proj_w)     # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 4. Apply mask (if present) and gates
    # --------------------------------------------------------------
    if (mask is not None) and (not nomask):
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp
        right = right * mask_fp

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core O(N³·H) contraction via batched GEMM (L @ Rᵀ)
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, H, N, N) -> (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (uses Tensor Cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm (hidden dim) and gating
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Triton kernel: out = out * out_gate (in‑place)
    out_flat = out.contiguous().view(-1)
    out_gate_flat = out_gate.contiguous().view(-1)
    total = out_flat.numel()
    BLOCK = 8192
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)
    _out_gate_mul_kernel[grid](
        out_flat,
        out_gate_flat,
        total,
        BLOCK_SIZE=BLOCK,
    )
    out = out_flat.view_as(out)

    # --------------------------------------------------------------
    # 7. Final linear projection back to original dimensionality
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, C) in fp16

    # --------------------------------------------------------------
    # 8. Cast back to input dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
