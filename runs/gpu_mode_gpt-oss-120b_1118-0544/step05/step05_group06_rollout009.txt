"""
TriMul (outgoing) forward pass using mixed‑precision and a Triton kernel.
The implementation follows the AlphaFold3 reference:

1. Layer‑norm the 4‑D input (B, N, N, C).
2. Cast to FP16 for compute‑heavy stages.
3. Linear projections (left/right) and corresponding gates are computed with
   torch.nn.functional.linear (FP16 weights).
4. If a mask is present, a Triton kernel fuses element‑wise
   `proj * gate * mask` (mask broadcast over the hidden dimension H).
5. Core reduction:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is realized as a batched GEMM (`left @ rightᵀ`) after reshaping to
   (B·H, N, N).
6. Output LayerNorm → gate multiplication → final linear projection back to C.
7. Cast the result back to the input dtype (normally float32).

Only the element‑wise gate‑mask multiplication is done with Triton; all
large matrix operations leverage highly‑optimized PyTorch kernels.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,            # *[total_elems]   fp16
    gate_ptr,            # *[total_elems]   fp16
    mask_ptr,            # *[mask_elems]    fp16
    out_ptr,             # *[total_elems]   fp16
    total_elems,         # i64
    hidden_dim,          # i64  (size of the last dimension H)
    mask_elems,          # i64  (B*N*N)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # Broadcast mask over the hidden dimension
    mask_idx = (offs // hidden_dim) % mask_elems

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Element‑wise ``proj * gate * mask`` where ``mask`` has shape (B, N, N)
    and is broadcast over the hidden dimension H.
    All tensors must be FP16 and contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)

    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],          # hidden_dim
        mask_flat.shape[0],      # mask_elems (= B*N*N)
        BLOCK_SIZE=block_size,
    )
    return out

# Entry point
# ----------------------------------------------------------------------

    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)

            - mask: torch.Tensor of shape [B, N, N] (float32/16) or None
            - weights: dict containing all linear and LayerNorm parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for heavy compute
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all linear weights to FP16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B,N,N,hidden_dim)
    # --------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # --------------------------------------------------------------
    # 6. Apply mask (if any) and fuse gate
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction: batched GEMM (left @ rightᵀ)
    #    Shape: (B, N, N, H) -> (B*H, N, N)
    # --------------------------------------------------------------
    B, N, _, H = left.shape

    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, gating, and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
```
