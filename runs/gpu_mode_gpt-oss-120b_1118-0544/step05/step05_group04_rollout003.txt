"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the (B,N,N,C) input.
2. Linear projection to hidden dimension H → tensors *left* and *right*.
3. Compute three sigmoid gates from the normalized input:
   left_gate, right_gate and out_gate (all (B,N,N,H)).
4. Fuse mask (if present) and gate application in a single Triton kernel:
   left_out  = left  * mask * left_gate
   right_out = right * mask * right_gate
5. Perform the core O(N³·H) reduction using a batched matrix multiply:
   out[b,h,i,j] = Σ_k left_out[b,i,k,h] * right_out[b,j,k,h]
   implemented as torch.bmm over (B·H) batches.
6. Layer‑normalize the hidden result, multiply by out_gate and linearly
   project back to the original channel dimension C.
7. Cast the final result to float32.

Only the element‑wise mask‑gate fusion is written in Triton; the heavy
reduction uses cuBLAS via torch.bmm, which is the fastest path on modern
GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # const fp16*  (B,N,N,H) flattened
    right_ptr,         # const fp16*  (B,N,N,H)
    mask_ptr,          # const fp16*  (B,N,N)
    left_gate_ptr,     # const fp16*  (B,N,N,H)
    right_gate_ptr,    # const fp16*  (B,N,N,H)
    left_out_ptr,      # fp16*        (B,N,N,H)
    right_out_ptr,     # fp16*        (B,N,N,H)
    H: tl.constexpr,   # hidden dimension
    TOTAL: tl.constexpr,  # B*N*N*H
    BLOCK: tl.constexpr,  # block size
):
    """Fuse mask loading + gate application."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < TOTAL

    # Load raw tensors
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Compute index into mask (broadcasted over H)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply mask and gates
    tl.store(left_out_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (fp32 or fp16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (fp32 tensors)
            - config: dict with keys
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask"       – bool, if True ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] in float32.
    """
    # ----------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # ----------------------------------------------------------
    # Cast to fp16 for the heavy part
    # ----------------------------------------------------------
    x = x.to(torch.float16)

    # ----------------------------------------------------------
    # 2) Load linear weights (fp16)
    # ----------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, dtype=torch.float16)

    # ----------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ----------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ----------------------------------------------------------
    # 4) Compute gates (sigmoid of linear projections)
    # ----------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,H)

    # ----------------------------------------------------------
    # 5) Prepare mask (broadcast over hidden dim)
    # ----------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(dtype=torch.float16, device=device)
    else:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # ----------------------------------------------------------
    # 6) Fuse mask & gates with Triton
    # ----------------------------------------------------------
    total_elements = left.numel()               # B * N * N * H
    BLOCK = 1024

    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = (triton.cdiv(total_elements, BLOCK),)
    _mask_gate_fuse_kernel[grid](
        left,
        right,
        mask_fp16.view(-1),
        left_gate,
        right_gate,
        left_out,
        right_out,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ----------------------------------------------------------
    # 7) Core O(N³·H) reduction via batched matrix multiply
    #    left_out, right_out : (B,N,N,H)
    #    reshape to (B*H, N, N) and compute (B*H, N, N) = left @ right^T
    # ----------------------------------------------------------
    left_flat  = left_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(-2, -1))   # (B*H, N, N)

    # ----------------------------------------------------------
    # 8) Reshape back to (B, N, N, H)
    # ----------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B,N,N,H)

    # ----------------------------------------------------------
    # 9) Output LayerNorm over hidden dimension
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----------------------------------------------------------
    # 10) Apply output gate
    # ----------------------------------------------------------
    out = out * out_gate

    # ----------------------------------------------------------
    # 11) Final linear projection back to original dim C
    # ----------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,C)

    # ----------------------------------------------------------
    # 12) Cast to float32 as required
    # ----------------------------------------------------------
    out = out.to(torch.float32)

    return out
