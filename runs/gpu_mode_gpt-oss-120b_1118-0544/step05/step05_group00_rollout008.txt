"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input (float32 for numeric stability).
2. Cast to fp16 for the heavy compute path.
3. Linear projections (left, right, three gates) using bias‑less half‑precision
   weights.
4. Fuse mask (if present) and gate multiplications with a tiny Triton kernel
   (`_gate_mask_kernel`).  The kernel works in‑place on the flattened tensors:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
   For the common “nomask” case we simply use PyTorch element‑wise multiplies.
5. Core O(N³·H) contraction:
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM L @ Rᵀ for each hidden channel d.
   We reshape to (B·H, N, N) and call `torch.bmm`, which uses cuBLAS tensor‑cores.
6. Output LayerNorm over the hidden dimension, element‑wise multiplication by
   the output gate, and a final linear projection back to the original dimension.
7. Cast back to the input dtype (float32) and return.

Only the mask‑gate fusion is executed in Triton; all other heavy
operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr,                # *float16  [B,N,N,H] flattened
    right_ptr,               # *float16  same shape
    left_gate_ptr,           # *float16  same shape
    right_gate_ptr,          # *float16  same shape
    mask_ptr,                # *float16  [B,N,N] flattened
    total_elems, hidden_dim, # scalars
    HAS_MASK: tl.constexpr,  # compile‑time bool
    BLOCK_SIZE: tl.constexpr # compile‑time int
):
    """In‑place fusion of mask and gate multiplications.
    Each program processes BLOCK_SIZE elements of the flattened tensors.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # mask is common for all hidden channels of a (b,i,j) entry
        mask_idx = offs // hidden_dim                # integer division
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)

    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    right = F.linear(x, right_proj_w)  # (B,N,N,hidden_dim)

            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # Reshape back to original layout
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # Simple element‑wise fusion (no mask)
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Core O(N³·H) contraction via batched GEMM (L @ Rᵀ)
    # --------------------------------------------------------------------- #
    B, N, _, H = left.shape
    # (B, N, N, H) -> (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (uses cuBLAS tensor cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 6. Output LayerNorm, output gate, final linear projection
    # --------------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate   # apply output gate (half‑precision)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # --------------------------------------------------------------------- #
    # 7. Cast back to input dtype and return
    # --------------------------------------------------------------------- #
    out = out.to(dtype_in)
    return out
```
