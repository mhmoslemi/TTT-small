Algorithm
---------
1. Layer‑normalize the [B, N, N, C] input (float32 → float16).
2. Linear projections to hidden dimension H (left, right) and three sigmoid gates.

   is performed by a batched GEMM kernel written entirely in Triton,
   avoiding the costly permute‑contiguous copy required by torch.bmm.
5. Apply a second LayerNorm over the hidden dimension, an output gate and a final
   linear projection back to C.
6. Cast the result back to the original dtype (float32).

The heavy reduction (step 4) is the only part done in pure Triton; all other
operations use fused PyTorch kernels for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    left_out_ptr, right_out_ptr,
    H: tl.constexpr, TOTAL_ELEMS: tl.constexpr, BLOCK: tl.constexpr
):
    """Element‑wise fusion: left_out = left * mask * left_gate,
    right_out = right * mask * right_gate."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < TOTAL_ELEMS

    # Load data
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Broadcast mask over hidden dimension
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply
    tl.store(left_out_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


@triton.jit
def _trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_i = tl.program_id(0)          # block index in i dimension
    pid_j = tl.program_id(1)          # block index in j dimension
    pid_bh = tl.program_id(2)         # combined batch*hidden index

    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N
    bh = pid_bh
    b = bh // H
    h = bh % H

    # Pointers to the beginning of the i‑tile and j‑tile
    left_base = left_ptr + b * stride_l_b + h * stride_l_h + i_offset * stride_l_i
    right_base = right_ptr + b * stride_r_b + h * stride_r_h + j_offset * stride_r_j
    out_base = out_ptr + b * stride_o_b + h * stride_o_h + i_offset * stride_o_i + j_offset * stride_o_j

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K

        # Offsets for the current tile
        offs_i = tl.arange(0, BLOCK_M)
        offs_j = tl.arange(0, BLOCK_N)
        offs_k = tl.arange(0, BLOCK_K)

        # Bounds for i/k/j
        i_valid = i_offset + offs_i < N
        j_valid = j_offset + offs_j < N
        k_valid = k_offset + offs_k < N

        # Load left tile  [M, K]
        left = tl.load(
            left_base
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k,
            mask=i_valid[:, None] & k_valid[None, :],
            other=0.0,
        )
        # Load right tile [K, N]  (right is indexed as (k, j))
        right = tl.load(
            right_base
            + offs_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_j,
            mask=k_valid[:, None] & j_valid[None, :],
            other=0.0,
        )

        # Accumulate (dot: (M,K) @ (K,N) -> (M,N))
        acc += tl.dot(left, right)

    # Store result back, cast to fp16
    mask_store = i_valid[:, None] & j_valid[None, :]
    tl.store(
        out_base
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j,
        acc.to(tl.float16),
        mask=mask_store,
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask        : torch.Tensor of shape [B, N, N] or None
            - weights     : dict of model parameters (float32 tensors)
            - config      : dict with keys
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, ignore mask if True (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float16 compute)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections and gates (still in fp16)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)          # (B,N,N,H)
    right = F.linear(x, right_proj_w)        # (B,N,N,H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B,N,N,H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4) Mask handling (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_tensor = mask.to(device=device, dtype=torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 5) Fuse mask and left/right gates (Triton)
    # ------------------------------------------------------------------
    total_elems = left.numel()
    BLOCK = 8192  # tuneable; must be multiple of 128 for good occupancy
    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = (triton.cdiv(total_elems, BLOCK),)
    _mask_gate_fuse_kernel[grid](
        left,
        right,
        mask_tensor.view(-1),
        left_gate,
        right_gate,
        left_out,
        right_out,
        hidden_dim,
        total_elems,
        BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction – batched GEMM in Triton
    # ------------------------------------------------------------------
    out = torch.empty_like(left_out)  # shape (B,N,N,H)

    # Strides (in elements, not bytes)
    s_l_b, s_l_i, s_l_k, s_l_h = left_out.stride()
    s_r_b, s_r_i, s_r_k, s_r_h = right_out.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    # For the right tensor we need stride for j (third dim) and k (second dim)
    stride_r_j = s_r_k   # original third dim (j) stride
    stride_r_k = s_r_i   # original second dim (i) stride, used as k

    # Choose block sizes; 64 works well for N up to 1024 on Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    grid_bh = B * hidden_dim
    grid = (grid_i, grid_j, grid_bh)

    _trimul_outgoing_kernel[grid](
        left_out,
        right_out,
        out,
        B,
        N,
        hidden_dim,
        s_l_b,
        s_l_i,
        s_l_k,
        s_l_h,
        s_r_b,
        stride_r_j,
        stride_r_k,
        s_r_h,
        s_o_b,
        s_o_i,
        s_o_j,
        s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 7) Output LayerNorm over hidden dimension (fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
