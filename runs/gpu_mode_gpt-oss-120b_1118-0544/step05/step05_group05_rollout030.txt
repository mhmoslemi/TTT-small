"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32 for stability) and cast to fp16.
2. Linear projections (left/right) and their sigmoid gates are computed with fp16.
3. A Triton kernel fuses mask (if present), gate multiplication and
   layout transformation from [B,N,N,H] → [B*H,N,N] required by the
   batched GEMM.  This eliminates the extra copy performed by a
   separate permute+contiguous step.
4. Core contraction is a batched matrix multiply:
        out[b*h] = left_buf[b*h] @ right_buf[b*h].T
   using Torch’s highly‑optimized fp16 GEMM (cuBLAS).
5. The result is reshaped back to [B,N,N,H], followed by an output
   LayerNorm, output‑gate multiplication and a final linear projection
   back to the original feature dimension.
All heavy arithmetic stays in fp16; the final tensor is returned in the
input dtype (normally float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_layout_fuse_kernel(
    left_ptr, right_ptr,                # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,      # same shape
    mask_ptr,                           # [B,N,N] flattened (optional)
    out_left_ptr, out_right_ptr,        # [B*H,N,N] flattened
    B, N, H,                            # runtime scalars
    total_elems,                        # B*N*N*H
    HAS_MASK: tl.constexpr,             # compile‑time bool
    BLOCK_SIZE: tl.constexpr,           # compile‑time int
):
    """Fuse mask & gates and reshape from (B,N,N,H) to (B*H,N,N)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load raw tensors
    left_val      = tl.load(left_ptr      + offs, mask=active, other=0.0)
    right_val     = tl.load(right_ptr     + offs, mask=active, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate_val= tl.load(right_gate_ptr+ offs, mask=active, other=0.0)

    # De‑linearize the flat index
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    k = rem2 // H
    h = rem2 % H

    # Apply gates
    left_val  = left_val  * left_gate_val
    right_val = right_val * right_gate_val

    if HAS_MASK:
        mask_idx = b * (N * N) + i * N + k               # index into mask
        mask_val = tl.load(mask_ptr + mask_idx,
                           mask=active, other=1.0)
        left_val  = left_val  * mask_val
        right_val = right_val * mask_val

    # Compute target batch index for the reshaped layout
    out_batch = b * H + h                         # range [0, B*H)
    out_offset = out_batch * (N * N) + i * N + k   # flatten (B*H,N,N)

    tl.store(out_left_ptr  + out_offset, left_val,  mask=active)
    tl.store(out_right_ptr + out_offset, right_val, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor of shape (B, N, N, C)   (C == dim)
            mask:        torch.Tensor of shape (B, N, N) or None
            weights:     dict of model parameters (see code)
            config:      dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numeric stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for the heavy part
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gates (all fp16)
    # -----------------------------------------------------------------
    left      = F.linear(x, left_proj_w)                 # (B,N,N,hidden_dim)
    right     = F.linear(x, right_proj_w)                # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))  # (B,N,N,hidden_dim)

    # -----------------------------------------------------------------
    # 4. Fuse mask, gates and reshape for GEMM via a Triton kernel
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    total = B * N * N * hidden_dim
    BLOCK = 1024

    # Flatten tensors for the kernel
    left_flat      = left.contiguous().view(-1)
    right_flat     = right.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_gate_flat= right_gate.contiguous().view(-1)

    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16)
        mask_flat = mask_fp.contiguous().view(-1)  # (B*N*N,)
    else:
        mask_fp = torch.empty(0, dtype=torch.float16, device=device)
        mask_flat = mask_fp

    # Buffers in the layout required for the batched GEMM: (B*H, N, N)
    left_buf  = torch.empty((B * hidden_dim, N, N), dtype=torch.float16,
                            device=device)
    right_buf = torch.empty_like(left_buf)

    grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

    _gate_mask_layout_fuse_kernel[grid](
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        left_buf.view(-1),
        right_buf.view(-1),
        B,
        N,
        hidden_dim,
        total,
        HAS_MASK= (mask is not None and not nomask),
        BLOCK_SIZE=BLOCK,
    )

    # -----------------------------------------------------------------
    # 5. Core contraction via batched GEMM: (B*H, N, N) @ (B*H, N, N)^T
    # -----------------------------------------------------------------
    # right_buf needs to be transposed on its last two dimensions
    out_buf = torch.bmm(left_buf, right_buf.transpose(1, 2))   # (B*H, N, N)

    # Reshape back to (B, N, N, hidden_dim)
    out = out_buf.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w,
                       bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # Cast back to the original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
