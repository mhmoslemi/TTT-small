def custom_kernel(data):
    input_tensor, mask, weights, config = data

    # Retrieve dims
    B,N,N,dim = input_tensor.shape  # N repeated
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm: using torch.nn.functional.layer_norm with weights
    x = torch.nn.functional.layer_norm(input_tensor,
                                       (dim,),
                                       weight=weights["norm.weight"],
                                       bias=weights["norm.bias"])
    # Cast to fp16
    x = x.to(torch.float16)

    # Linear projections using torch.nn.functional.linear: left_proj = x @ left_proj_weight.T (weight shape out_features, in_features)
    # weights dict have shape (hidden_dim, dim). Let’s assume standard linear: weight shape (out_features, in_features). So we should do linear(x, weight). For torch.nn.functional.linear expects weight shape (out_features, in_features), bias optional. So we will use that.

    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16))
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Apply mask and gating
    if mask is not None and not nomask:
        mask = mask.to(torch.float16).unsqueeze(-1) # shape (B,N,N,1)
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Now compute reduction via Triton.
    # Ensure left and right are contiguous and shape (B,N,N,H). We'll have hidden dimension H=hidden_dim.

    # Permute to (B, H, N, N) then reshape to (B*H, N, N)
    left_pre = left.permute(0, 3, 1, 2).contiguous()
    right_pre = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_pre.shape  # Actually left_pre shape (B, H, N, N)
    # reshape
    left_flat = left_pre.view(B * H, N, N)
    right_flat = right_pre.view(B * H, N, N)

    # Prepare output tensor of same shape (B*H, N, N) FP16
    out_flat = torch.empty_like(left_flat)

    # Compute strides
    stride_a_batch, stride_a_row, stride_a_col = left_flat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_flat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_flat.stride()

    # Here we want B_T = right_flat transposed. So for B_T we need row stride = stride_b_col (original col stride) and col stride = stride_b_row (original row stride).

    # Setup block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32  # typical for fp16

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)

    # Launch kernel
    _triton_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,   # M,N,K are all N
        stride_a_row, stride_a_col,
        stride_b_col, stride_b_row,  # transposed
        stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm (layernorm over hidden dim H) and gating
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    # final linear projection to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast to original dtype
    out = out.to(input_tensor.dtype)

    return out

"""
TriMul (outgoing) forward implementation with a Triton kernel for the
core reduction:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

Algorithm
---------
1. Layer‑norm the input.
2. FP16 linear projections (left/right) and their sigmoid gates.
3. Apply mask (if provided) and gate element‑wise (pure PyTorch, cheap).
4. Reshape tensors to (B*H, N, N) and perform a batched GEMM with the
   transpose of the right operand using a custom Triton kernel.
5. Reshape back to (B, N, N, H), apply a second LayerNorm, multiply by the
   output gate and project back to the original channel dimension.
All heavy arithmetic (step 4) is done in FP16 with FP32 accumulation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _triton_matmul_kernel(
    a_ptr,                 # fp16*  (B*H, N, N)
    b_ptr,                 # fp16*  (B*H, N, N)  (will be accessed transposed)
    c_ptr,                 # fp16*  (B*H, N, N)
    M, N, K,               # matrix dimensions (all = seq_len)
    stride_ab, stride_am, stride_ak,   # batch, row, col strides for A
    stride_bb, stride_bk, stride_bn,   # batch, row, col strides for B (transposed)
    stride_cb, stride_cm, stride_cn,   # batch, row, col strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_b = tl.program_id(2)  # batch index (B*H)

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the beginning of the current batch
    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bb
    c_batch = c_ptr + pid_b * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Main reduction loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (cur_k[None, :] < K)
        b_mask = (cur_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=a_mask,
            other=0.0,
        )
        # B is accessed transposed: Bᵀ[k, n] = B[n, k]
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=b_mask,
            other=0.0,
        )

        acc += tl.dot(a, b)          # FP16 × FP16 → FP32 accumulation

    # Write the result back (FP32 → FP16)
    c = acc.to(tl.float16)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=c_mask,
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [B, N, N, dim] (same dtype as input_tensor).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (no bias) and gates
    # --------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16))

    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --------------------------------------------------------------
    # 4. Apply mask (if any) and gate
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)          # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5. Core reduction using Triton (batched GEMM with transpose)
    # --------------------------------------------------------------
    # Rearrange to (B, H, N, N) → (B*H, N, N)
    left_pre = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_pre = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_pre.shape
    left_flat = left_pre.view(B * H, N, N)
    right_flat = right_pre.view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)  # FP16 output

    # Strides
    stride_ab, stride_am, stride_ak = left_flat.stride()
    stride_bb, stride_bk_raw, stride_bn_raw = right_flat.stride()
    stride_cb, stride_cm, stride_cn = out_flat.stride()

    # For transposed access of B we swap its row/col strides
    stride_bk = stride_bn_raw   # row stride after transpose (original col stride)
    stride_bn = stride_bk_raw   # col stride after transpose (original row stride)

    # Blocking parameters (tuned for typical GPUs)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H,                     # batch dimension (B * hidden_dim)
    )

    _triton_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cb, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # free intermediate tensors early
    del left, right, left_proj, right_proj, left_gate, right_gate, left_pre, right_pre, left_flat, right_flat, out_flat

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 7. Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(dtype_in)

    return out
