"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input (float32 for numerical stability).
2. Cast to float16 for the heavy compute path.
3. Linear projections → left, right, left_gate, right_gate, out_gate (no bias).
4. Fuse mask (if present) and the element‑wise gate multiplications with a
   tiny Triton kernel (`_gate_mask_kernel`).  This writes the gated tensors
   back in‑place, avoiding extra memory passes.
5. Core O(N³·H) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication `L @ Rᵀ` for each hidden
   channel `h`.  We reshape to (B*H, N, N) and use `torch.bmm` (cuBLAS).
6. Apply the output layer‑norm, multiply by the output gate and a final
   linear projection back to the original dimension.
7. Cast the result back to the input dtype.

Only the mask‑gate fusion is performed in a custom Triton kernel; the
remaining operations use highly‑optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr,               # [B,N,N,H] flattened
    right_ptr,              # same shape
    left_gate_ptr,          # same shape
    right_gate_ptr,         # same shape
    mask_ptr,               # [B,N,N] flattened
    total_elems,            # scalar total number of elems in left/right
    hidden_dim,             # scalar H
    HAS_MASK: tl.constexpr, # compile‑time bool
    BLOCK_SIZE: tl.constexpr # compile‑time int
):
    """In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    `mask` is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load data
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # map each element to its mask entry (one mask per [b,i,k])
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store the fused results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int)        – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------- #
    # 1. Input LayerNorm (float32 for stability)
    # --------------------------------------------------------------------- #
    norm_w = weights["norm.weight"].to(device=device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device=device, dtype=torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # --------------------------------------------------------------------- #
    # 2. Cast to fp16 for the compute‑heavy path
    # --------------------------------------------------------------------- #
    x = x.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 3. Linear projections (no bias) – all in fp16
    # --------------------------------------------------------------------- #
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # (B,N,N,hidden_dim)
    left = F.linear(x, left_proj_w)
    right = F.linear(x, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------------- #
    # 4. Fuse mask and gates (if a mask is present and not disabled)
    # --------------------------------------------------------------------- #
    if mask is not None and not nomask:
        # mask: (B,N,N) → fp16 and flatten
        mask_fp = mask.to(device=device, dtype=torch.float16).contiguous()
        mask_flat = mask_fp.view(-1)                     # (B*N*N,)

        # Flatten tensors for the kernel
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)

        total = left_flat.numel()
        BLOCK_SIZE = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
        # Release temporary tensors
        del left_gate, right_gate, left_gate_flat, right_gate_flat, mask_fp, mask_flat, left_flat, right_flat
        torch.cuda.empty_cache()
    else:
        # Simple element‑wise fusion without a mask
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate
        torch.cuda.empty_cache()

    # --------------------------------------------------------------------- #
    # 5. Core O(N³·H) contraction via batched GEMM (L @ Rᵀ)
    # --------------------------------------------------------------------- #
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (uses cuBLAS tensor‑cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left_mat, right_mat, out_mat, left, right
    torch.cuda.empty_cache()

    # --------------------------------------------------------------------- #
    # 6. Output LayerNorm, gate, and final projection
    # --------------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (half‑precision)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # --------------------------------------------------------------------- #
    # 7. Cast back to original dtype and return
    # --------------------------------------------------------------------- #
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
