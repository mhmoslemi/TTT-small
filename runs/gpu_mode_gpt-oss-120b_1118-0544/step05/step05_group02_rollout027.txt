def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Cast to fp16
    x = x.to(torch.float16)

    # Cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Projections
    left_proj = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply mask & gates
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguous for kernel
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape

    # Allocate output
    out = torch.empty((B, N, N, H), dtype=torch.float16, device=left.device)

    # Strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Grid dimensions
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid_z = B * H

    triu_mul_kernel[(grid_m, grid_n, grid_z)](

        out,
        B,
        N,
        H,

        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,

    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    out = out.to(input_tensor.dtype)

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑normalize the 4‑D input tensor (B, N, N, C).
2. Cast to fp16 and apply linear projections (left, right) and
   three gate projections (left_gate, right_gate, out_gate).
3. Fuse mask (if present) with the projected tensors via an
   element‑wise product.
4. Core O(N³·H) reduction:
       out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   This is a batched matrix multiplication (left @ rightᵀ) performed
   by a custom Triton kernel that tiles the i, j and k dimensions.
5. Apply a second LayerNorm, multiply by the output gate and finally
   linearly project back to the original channel dimension.
6. Cast the result back to the original dtype.

The heavy reduction (step 4) is the only part implemented in Triton;
all other operations use PyTorch/Libtorch (cuBLAS, etc.).
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,        # *[B, N, N, H]   fp16
    right_ptr,       # *[B, N, N, H]   fp16
    out_ptr,         # *[B, N, N, H]   fp16
    B, N, H,
    # strides for left  (batch, i, k, h)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right (batch, j, k, h)
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides for out   (batch, i, j, h)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)                     # block index over i
    pid_n = tl.program_id(1)                     # block index over j
    pid_z = tl.program_id(2)                     # combined batch * head

    # i and j coordinates of the block
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # decode batch index b and head index h from pid_z
    batch_head = pid_z
    b = batch_head // H
    h = batch_head % H

    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # left tile: shape (BLOCK_M, BLOCK_K)   left[b, i, k, h]
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            i[:, None] * stride_left_i + \
            k_off[None, :] * stride_left_k + \
            h * stride_left_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile: shape (BLOCK_K, BLOCK_N)   right[b, j, k, h]
        # note that we load with k as the first dimension to match tl.dot
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            k_off[:, None] * stride_right_k + \
            j[None, :] * stride_right_j + \
            h * stride_right_h
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # accumulate – left (M×K) dot right (K×N) → (M×N)
        acc += tl.dot(left, right)   # fp16 dot, accumulation promoted to fp32

    # store the result
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        i[:, None] * stride_out_i + \
        j[None, :] * stride_out_j + \
        h * stride_out_h
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

    # 4. Linear projections
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6. Apply mask (if any) and element‑wise gate
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape (B, N, N, 1) broadcast over hidden_dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguous tensors for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) multiplication via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    out = torch.empty((B, N, N, H), dtype=torch.float16, device=left.device)

    # strides (in elements) of the tensors
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # block sizes – tuned for fp16 Tensor‑Core usage
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid_z = B * H

    tri_mul_outgoing_kernel[(grid_m, grid_n, grid_z)](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
