    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):

   pid_n = tl.program_id(1) # block j
   pid_bh = tl.program_id(2) # batch*hidden index

   # compute batch index and hidden index:

   h = pid_bh % H

   # We'll offset base = left_ptr + b*stride_left_b + h*stride_left_h

def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile col
    pid_bh = tl.program_id(2) # batch * hidden

    # compute batch index and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # compute base pointer offsets for this batch and hidden
    base_off = b * stride_left_b + h * stride_left_h

    # offsets of row/col inside N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute masks for left and right loads
        # left tile: M x K
        left = tl.load(
            left_ptr + base_off + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # right tile transposed: K x N
        right = tl.load(
            right_ptr + base_off + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )

        acc += tl.dot(left, right)

    # Write out
    out = acc.to(tl.float16)

    tl.store(
        out_ptr + b * stride_out_b + h * stride_out_h + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j,
        out,
        mask=mask_m[:, None] & mask_n[None, :]
    )

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Apply LayerNorm on the 4‑D input `[B,N,N,D]`.
2. Compute left/right projections and their sigmoid gates (all in fp16).
3. Fuse mask (if present) with the gates using simple pointwise ops.
4. Compute the core O(N³·H) contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton batched GEMM that works on the
   original `[B,N,N,H]` layout, thus avoiding the costly
   `permute(...).contiguous()` copy required by `torch.bmm`.
5. Apply output LayerNorm, the output gate, and a final linear projection
   back to the original feature dimension.
All heavy arithmetic is carried out in fp16; the final result is cast
back to the input dtype (usually fp32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for the outgoing triangle multiplicative update.
    Computes, for each batch b and hidden channel h:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    """
    pid_m = tl.program_id(0)          # tile row (i)
    pid_n = tl.program_id(1)          # tile col (j)
    pid_bh = tl.program_id(2)         # combined batch*hidden index

    # decode batch and hidden channel
    b = pid_bh // H
    h = pid_bh % H

    # base offset for this (b,h) slice
    base_off_left  = b * stride_left_b  + h * stride_left_h
    base_off_right = b * stride_right_b + h * stride_right_h
    base_off_out   = b * stride_out_b    + h * stride_out_h

    # tile coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block:   [M, K]  (i, k)
        left = tl.load(
            left_ptr
            + base_off_left
            + offs_m[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right block transposed: [K, N]  (k, j)
        right = tl.load(
            right_ptr
            + base_off_right
            + offs_k[:, None] * stride_right_k
            + offs_n[None, :] * stride_right_i,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(left, right)

    # store result in fp16
    out = acc.to(tl.float16)
    tl.store(
        out_ptr
        + base_off_out
        + offs_m[:, None] * stride_out_i
        + offs_n[None, :] * stride_out_j,
        out,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, D)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, D) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, torch.float32)
    norm_b = weights["norm.bias"].to(device, torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy parts
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, torch.float16)

    # -----------------------------------------------------------------
    # 4. Linear projections and gates (fp16)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 5. Apply mask and gates (element‑wise, stays in fp16)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * left_gate * mask
        right = right * right_gate * mask
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Core O(N³·H) contraction via a custom Triton batched GEMM
    # -----------------------------------------------------------------
    B, N, _, H = left.shape

    out = torch.empty_like(left)  # allocate (B,N,N,H) fp16 buffer

    # strides for the kernel
    l_str = left.stride()
    r_str = right.stride()
    o_str = out.stride()

    # kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        l_str[0],
        l_str[1],
        l_str[2],
        l_str[3],
        r_str[0],
        r_str[1],
        r_str[2],
        r_str[3],
        o_str[0],
        o_str[1],
        o_str[2],
        o_str[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 7. Output LayerNorm, output gate, final projection
    # -----------------------------------------------------------------
    out = out.to(torch.float32)          # convert for stable LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, to_out_w.to(torch.float32))   # (B, N, N, D)

    # Cast back to original dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
