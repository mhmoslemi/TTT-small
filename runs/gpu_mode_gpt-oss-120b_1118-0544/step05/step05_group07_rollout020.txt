@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Compute total number of programs:
    # grid = (B*H * ceil(N/BLOCK_M) * ceil(N/BLOCK_N),)
    # decode ids
    pid = tl.program_id(0)
    num_blocks_M = tl.cdiv(N, BLOCK_M)
    num_blocks_N = tl.cdiv(N, BLOCK_N)
    batch_size = B * H
    # get batch index
    batch_idx = pid // (num_blocks_M * num_blocks_N)
    rem = pid % (num_blocks_M * num_blocks_N)
    block_i = rem // num_blocks_N
    block_j = rem % num_blocks_N

    # derive b and h from batch idx
    b = batch_idx // H
    h = batch_idx % H

    # offsets for i and j
    i_start = block_i * BLOCK_M
    j_start = block_j * BLOCK_N
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # mask
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this batch (b,h)
    left_base = left_ptr + b * stride_left_batch + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base = out_ptr + b * stride_out_batch + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape [BLOCK_M, BLOCK_K]
        left_tile = tl.load(
            left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )

        # Load right tile: shape [BLOCK_K, BLOCK_N] (k as rows, j as cols)
        right_tile = tl.load(
            right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )

        # Compute dot product
        acc += tl.dot(left_tile, right_tile)

    # Store result
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j

"""
TriMul (outgoing) forward pass – Triton‑fused core reduction.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C).
2. Cast to fp16 and apply four linear projections (left/right and their gates)
   using torch.nn.functional.linear (batched GEMM on the flattened (B·N·N)×C).
3. Fuse gate activations (sigmoid) and optional mask with simple element‑wise
   multiplication (PyTorch is sufficient; the main requirement of using Triton
   is satisfied by the core reduction kernel).
4. Core O(N³·H) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   Implemented as a custom Triton kernel that performs a batch of H independent
   GEMMs of size N×N for each batch element, using block tiling (BLOCK_M×BLOCK_N×BLOCK_K).
5. Second LayerNorm over the hidden dimension, multiply by the output gate,
   and project back to the original channel dimension.
6. Cast the result back to the input dtype (float32).

The kernel processes each (batch, head) pair as a separate GEMM, loading tiles
of the left and right tensors, accumulating in fp32, and storing the result in fp16.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # fp16 * (B, N, N, H)
    right_ptr,         # fp16 * (B, N, N, H)
    out_ptr,           # fp16 * (B, N, N, H)
    B, N, H,           # tensor dimensions (int)
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""

    pid = tl.program_id(0)

    # Number of output blocks per dimension
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch_cnt = B * H

    # Decode program id into (batch, block_i, block_j)
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    rem = pid % (num_blocks_m * num_blocks_n)
    block_i = rem // num_blocks_n
    block_j = rem % num_blocks_n

    # Decompose batch index into original batch and head
    b = batch_idx // H
    h = batch_idx % H

    # Compute base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_left_batch  + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_batch   + h * stride_out_h

    # Offsets for the output tile
    i_start = block_i * BLOCK_M
    j_start = block_j * BLOCK_N
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load right block transposed: shape (BLOCK_K, BLOCK_N)
        right = tl.load(
            right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # Accumulate
        acc += tl.dot(left, right)

    # Store the result in fp16
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to fp16 where appropriate
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (batched GEMM on flattened view)
    # ------------------------------------------------------------------
    # Shape after projection: (B, N, N, H)
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,H]
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6. Fuse projection, gate and optional mask (in‑place)
    # ------------------------------------------------------------------
    left_proj.mul_(left_gate)
    right_proj.mul_(right_gate)

    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj.mul_(mask_fp16)
        right_proj.mul_(mask_fp16)

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction via custom Triton kernel
    # ------------------------------------------------------------------
    B, N, _, H = left_proj.shape
    out = torch.empty_like(left_proj)  # fp16 output tensor

    # Strides (in elements) for the tensors
    stride_left  = left_proj.stride()   # (B_dim, i_dim, k_dim, h_dim)
    stride_right = right_proj.stride()
    stride_out   = out.stride()

    # Define block sizes; these work well for typical N (256–1024) on B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * H * num_blocks_m * num_blocks_n

    grid = (total_programs,)

    trimul_outgoing_kernel[grid](
        left_proj,
        right_proj,
        out,
        B, N, H,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8. Second LayerNorm over hidden dimension (float32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9. Output gate and final projection back to original dim
    # ------------------------------------------------------------------
    out_gated = out_norm * out_gate.to(torch.float32)
    out_final = torch.nn.functional.linear(out_gated, to_out_w)   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 10. Cast back to input dtype if necessary and return
    # ------------------------------------------------------------------
    if out_final.dtype != dtype_in:
        out_final = out_final.to(dtype_in)
    return out_final
