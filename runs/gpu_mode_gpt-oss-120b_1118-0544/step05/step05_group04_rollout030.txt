def matmul_batch_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_a_batch,
    stride_bk, stride_bn, stride_b_batch,
    stride_cm, stride_cn, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr, 
):
    pid = tl.program_id(0)
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    num_pid_b = tl.cdiv(tl.num_programs(0), num_pid_m * num_pid_n)  # number of batch*head
    pid_b = pid // (num_pid_m * num_pid_n)
    pid_mn = pid % (num_pid_m * num_pid_n)
    pid_m = pid_mn // num_pid_n
    pid_n = pid_mn % num_pid_n

    # offsets in M,N dims
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # offset in K
    offs_k = tl.arange(0, BLOCK_K)

    # pointers for batch b
    a_ptrs = a_ptr + pid_b * stride_a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_b_batch + offs_n[None, :] * stride_bn + offs_k[:, None] * stride_bk  # need to treat b as (K,N) after transposition: k is row, n is column.
    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=offs_m[:, None] < M and offs_k[None, :] < K, other=0.0)
        b = tl.load(b_ptrs, mask=offs_k[:, None] < K and offs_n[None, :] < N, other=0.0)
        acc += tl.dot(a, b)  # a: (BLOCK_M, BLOCK_K) ; b: (BLOCK_K, BLOCK_N)
        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
    # write back
    c_ptrs = c_ptr + pid_b * stride_c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))
```

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', False)

    # 1) LayerNorm on input
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights['norm.weight'].to(device=device, dtype=torch.float32),
        bias=weights['norm.bias'].to(device=device, dtype=torch.float32),
    )
    # Cast to fp16
    x = x.to(torch.float16)

    # 2) Linear projections
    left = F.linear(x, weights["left_proj.weight"].to(device, dtype=torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(device, dtype=torch.float16))

    # 3) Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(device, dtype=torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(device, dtype=torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device, dtype=torch.float16)))

    # 4) Mask application
    if mask is not None and not nomask:
        mask = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
    else:
        mask = torch.ones_like(left, dtype=torch.float16)

    left = left * mask * left_gate
    right = right * mask * right_gate

    # 5) Prepare for batch GEMM
    B, N, _, H = left.shape
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Flatten batch*head
    left_flat = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)
    # Transpose right
    right_T = right_flat.transpose(-2, -1).contiguous()

    # 6) Triton batched GEMM
    out_flat = torch.empty_like(left_flat)  # same dtype

    # Get strides
    a_ptr = left_flat
    b_ptr = right_T
    c_ptr = out_flat

    stride_a_batch, stride_a_m, stride_a_k = a_ptr.stride()
    stride_b_batch, stride_b_k, stride_b_n = b_ptr.stride()
    stride_c_batch, stride_c_m, stride_c_n = c_ptr.stride()

    M = N   # rows
    K = N   # inner dimension
    N_out = N   # columns (same)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

        b_ptr,
        c_ptr,
        M,
        N_out,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # 7) Reshape back to (B,N,N,H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 8) Output layernorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device, dtype=torch.float16),
    )

    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"].to(device, dtype=torch.float16))

    out = out.to(orig_dtype)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The forward pass follows the reference PyTorch implementation:
1. Layer‑normalize the input tensor (B,N,N,C).
2. Linear projections to a hidden dimension H (left/right).
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. Apply an optional binary mask and the left/right gates (element‑wise, done in PyTorch).
5. Core O(N³·H) reduction:
   For each batch element and each hidden head h,
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (left @ rightᵀ) implemented with a
   custom Triton kernel that fuses all batch heads into a single persistent kernel.
6. Layer‑normalize the hidden output, multiply by the output gate, and project back
   to the original channel dimension C.
All heavy arithmetic is performed in float16 with float32 accumulation.
"""

import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr,                     # left  (batch, M, K)   float16
    b_ptr,                     # right (batch, K, N)   float16 (already transposed)
    c_ptr,                     # out   (batch, M, N)   float16
    M, N, K,                   # matrix sizes (constexpr)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # grid layout: one dimensional grid, each program processes a (block_m, block_n) tile
    # for a specific batch element.
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    batch_id = pid // blocks_per_batch
    pid_in_batch = pid % blocks_per_batch
    pid_m = pid_in_batch // num_blocks_n
    pid_n = pid_in_batch % num_blocks_n

    # Offsets for the current tile.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for the current batch.
    a_ptrs = a_ptr + batch_id * stride_a_batch \
                + offs_m[:, None] * stride_a_m \
                + offs_k[None, :] * stride_a_k

    b_ptrs = b_ptr + batch_id * stride_b_batch \
                + offs_k[:, None] * stride_b_k \
                + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension.
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N)
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Write back.
    c_ptrs = c_ptr + batch_id * stride_c_batch \
                + offs_m[:, None] * stride_c_m \
                + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] (binary) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 4) Optional mask + gate fusion (plain PyTorch – low overhead)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_ = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
    else:
        mask_ = torch.ones_like(left, dtype=torch.float16)

    left = left * mask_ * left_gate
    right = right * mask_ * right_gate

    # ------------------------------------------------------------------
    # 5) Prepare tensors for the batched GEMM
    #     left: (B, H, N, N) -> (B*H, N, N)
    #     rightᵀ: (B, H, N, N) -> (B*H, N, N) and then transpose to (B*H, K, N)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_flat = left_perm.view(B * hidden_dim, N, N)          # (B*H, M, K) with M=K=N
    right_flat = right_perm.view(B * hidden_dim, N, N)        # (B*H, K, N) after transpose
    right_T = right_flat.transpose(-2, -1).contiguous()      # (B*H, K, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # 6) Triton batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    a_ptr = left_flat
    b_ptr = right_T
    c_ptr = out_flat

    stride_a_batch, stride_a_m, stride_a_k = a_ptr.stride()
    stride_b_batch, stride_b_k, stride_b_n = b_ptr.stride()
    stride_c_batch, stride_c_m, stride_c_n = c_ptr.stride()

    M = N
    K = N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_batches = B * hidden_dim
    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = num_batches * num_blocks_m * num_blocks_n

    grid = (total_blocks,)

    _batched_gemm_kernel[grid](
        a_ptr,
        b_ptr,
        c_ptr,
        M,
        N,
        K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9) Apply output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10) Final linear projection back to original dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 11) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
