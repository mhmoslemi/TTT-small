"""
TriMul outgoing kernel (AlphaFold3 style)
---------------------------------------
Implements the forward pass of the Triangle Multiplicative Update (TriMul)
operator using mixed‑precision (FP16) for the compute‑heavy parts and a
single Triton kernel to fuse the element‑wise mask, projection and gate
multiplications for the left & right branches.  The core O(N³·H) reduction
is performed with a batched GEMM (torch.bmm) on FP16 data, which is
highly optimized on modern NVIDIA GPUs (B200).  The final output is
returned in the original dtype (usually FP32).
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse (proj * gate * mask) for both left and right branches
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_two(
    proj1_ptr,  # *[total]  fp16  left projection
    gate1_ptr,  # *[total]  fp16  left gate
    proj2_ptr,  # *[total]  fp16  right projection
    gate2_ptr,  # *[total]  fp16  right gate
    mask_ptr,   # *[mask]   fp16  mask (flattened B*N*N)
    out1_ptr,   # *[total]  fp16  fused left output
    out2_ptr,   # *[total]  fp16  fused right output
    total_elems,    # i64   total number of elements in a proj tensor
    hidden_dim,     # i64   size of the last dimension H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # each contiguous block of `hidden_dim` elements corresponds to a single mask entry
    mask_idx = offs // hidden_dim

    p1 = tl.load(proj1_ptr + offs, mask=mask, other=0.0)
    g1 = tl.load(gate1_ptr + offs, mask=mask, other=0.0)
    p2 = tl.load(proj2_ptr + offs, mask=mask, other=0.0)
    g2 = tl.load(gate2_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out1 = p1 * g1 * m
    out2 = p2 * g2 * m

    tl.store(out1_ptr + offs, out1, mask=mask)
    tl.store(out2_ptr + offs, out2, mask=mask)


def _apply_gate_and_mask_two(left_proj, right_proj,
                             left_gate, right_gate,
                             mask, block_size: int = 2048):
    """
    Fuse left/right projection, gate and mask using a single Triton kernel.
    All tensors must be FP16, contiguous and have shape (B, N, N, H) for
    projections/gates and (B, N, N) for mask.
    Returns:
        left_fused, right_fused   tensors of shape (B, N, N, H) in FP16.
    """
    total = left_proj.numel()
    out_left = torch.empty_like(left_proj)
    out_right = torch.empty_like(right_proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)

    _gate_mask_fuse_two[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask_flat,
        out_left,
        out_right,
        total,
        left_proj.shape[-1],      # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out_left, out_right


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (C = dim)
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict mapping weight names to torch.Tensors
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for heavy compute
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all linear/gate weights to FP16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B,N,N,hidden_dim)
    # --------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # --------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # --------------------------------------------------------------
    # 6. Apply mask, projection and gate in one fused Triton kernel
    #    (if a mask is present and nomask=False)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_gate_and_mask_two(
            left_proj, right_proj,
            left_gate, right_gate,
            mask_fp16,
            block_size=2048,
        )
    else:
        # Simple element‑wise multiplication when no mask is used
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core O(N³·H) reduction: batched GEMM (left @ rightᵀ) per hidden dim
    #    Shape manipulation: (B, N, N, H) -> (B*H, N, N)
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, H, N, N) -> (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: left * rightᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm over hidden dimension, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype (usually float32)
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
```
