@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          BATCH, M, N, K,
                          stride_a_batch, stride_a_m, stride_a_k,
                          stride_b_batch, stride_b_k, stride_b_n,
                          stride_c_batch, stride_c_m, stride_c_n,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    # compute batch, m block, n block
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_tiles_m * num_tiles_n)
    block_index = pid % (num_tiles_m * num_tiles_n)
    pid_m = block_index // num_tiles_n
    pid_n = block_index % num_tiles_n

    # compute offsets for the block
    # start indices in A and B
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute mask for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute start of K dimension
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for A and B block
        a_ptrs = A_ptr + batch * stride_a_batch + (offs_m[:, None] * stride_a_m) + (offs_k[None, :] * stride_a_k)
        b_ptrs = B_ptr + batch * stride_b_batch + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)

        # load tiles, using mask for out-of-bounds
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

    # store result
    c_ptrs = C_ptr + batch * stride_c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangular Multiplicative Update (TriMul)

Algorithm
---------
1. Input LayerNorm (float32 for numerical stability).
2. Cast to fp16 for the compute‑intensive path.
3. Linear projections → left, right (no bias) and three sigmoid gates
   (left_gate, right_gate, out_gate).
4. Apply mask (if present) and fuse it with the gates using regular
   PyTorch element‑wise ops (cheap compared to the O(N³) contraction).
5. Core contraction: for each hidden head h,
       out_h = left_h @ right_hᵀ
   implemented as a fused batched GEMM in Triton
   (input tensors are reshaped to (B*H, N, N) and multiplied with
   the transposed right tensor).
6. Output LayerNorm over the hidden dimension, multiply by out_gate,
   and final linear projection back to the original channel dimension.
7. Cast back to the input dtype (float32) and return.

The heavy O(N³·H) work is performed by a custom Triton kernel,
while all other lightweight operations use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B   where
        A: (batch, M, K)   (fp16)
        B: (batch, K, N)   (fp16)   – already transposed if needed
        C: (batch, M, N)   (fp16)

    Accumulates in fp32 for precision.
    """
    pid = tl.program_id(0)

    # number of tiles in each dimension
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)

    # decode program id into batch / tile indices
    batch = pid // (num_tiles_m * num_tiles_n)
    pid_in_batch = pid % (num_tiles_m * num_tiles_n)
    pid_m = pid_in_batch // num_tiles_n
    pid_n = pid_in_batch % num_tiles_n

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for the current tile
        a_ptrs = (
            A_ptr
            + batch * stride_a_batch
            + (offs_m[:, None] * stride_a_m)
            + (offs_k[None, :] * stride_a_k)
        )
        b_ptrs = (
            B_ptr
            + batch * stride_b_batch
            + (offs_k[:, None] * stride_b_k)
            + (offs_n[None, :] * stride_b_n)
        )

        # load with masks, promote to fp32
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # block‑wise dot product
        acc += tl.dot(a, b)

    # store result in fp16
    c_ptrs = (
        C_ptr
        + batch * stride_c_batch
        + (offs_m[:, None] * stride_c_m)
        + (offs_n[None, :] * stride_c_n)
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with dtype matching the input.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Load weights (move to device) and cast to appropriate dtypes
    # -----------------------------------------------------------------
    # LayerNorm (input)
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)

    # Linear projections (no bias) – fp16 for compute
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    # Output LayerNorm and final projection – fp16
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 3. Cast to fp16 for heavy compute path
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 4. Linear projections and gates
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 5. Apply mask (if present) and gates – cheap elementwise ops
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # mask is broadcast over the hidden dimension
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left  = left * left_gate * mask_fp
        right = right * right_gate * mask_fp
    else:
        left  = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Core O(N³·H) contraction via fused batched GEMM (Triton)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # left.shape == (B, N, N, H)

    # reshape to (B*H, N, N) for batched matmul
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B*H, N, N)

    # we need rightᵀ for the einsum; transpose then make contiguous
    right_t = right_mat.transpose(1, 2).contiguous()  # (B*H, N, N)

    # output buffer
    out_mat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # strides for the kernel (int64)
    stride_a_batch = left_mat.stride(0)
    stride_a_m     = left_mat.stride(1)
    stride_a_k     = left_mat.stride(2)

    stride_b_batch = right_t.stride(0)
    stride_b_k     = right_t.stride(1)
    stride_b_n     = right_t.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_m     = out_mat.stride(1)
    stride_c_n     = out_mat.stride(2)

    # tuning parameters – chosen for good occupancy on B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B * H * num_tiles_m * num_tiles_n

    grid = (total_blocks,)

    _batched_matmul_kernel[grid](
        left_mat,
        right_t,
        out_mat,
        B * H,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # -----------------------------------------------------------------
    # 7. Output LayerNorm, gate, and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim) in fp16

    # -----------------------------------------------------------------
    # 8. Cast back to the original dtype and return
    # -----------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)
    return out
