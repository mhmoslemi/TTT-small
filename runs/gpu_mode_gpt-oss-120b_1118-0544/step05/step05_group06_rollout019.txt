@triton.jit
def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # K loop
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32) # accumulate in FP32

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # a block: [BLOCK_M, BLOCK_K] per batch
        a_ptrs = a_ptr + pid_batch * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_batch * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)

        # cast to float32
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # Write results
    c_ptrs = c_ptr + pid_batch * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

"""
TriMul (outgoing) forward pass – highly‑optimized Triton implementation.

Algorithm
---------
1. Layer‑normalize the 4‑D input (B,N,N,C) with the provided LN parameters.
2. Cast to FP16 for compute‑intensive stages.
3. Linear projections → left/right (B,N,N,H) and gate projections → left_gate,
   right_gate, out_gate (all FP16).
4. Fuse element‑wise mask (if present), gate and projection using a small
   Triton kernel:  out = proj * gate * mask   (broadcast mask over H).
5. Core reduction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden channel.  We
   flatten the (B,H) dimensions → (B*H, N, N) and run a custom Triton GEMM
   kernel that tiles the output (i,j) and streams over the reduction axis k.
   Accumulation is performed in FP32 and the result written back as FP16.
6. Reshape back to (B,N,N,H), apply a second LayerNorm, multiply by out_gate,
   and project back to the original channel dimension with a final linear.
7. Cast the result to the input dtype (usually float32) and return.

The implementation uses Triton for two fused kernels (gate‑mask fusion and
batched GEMM) while delegating the cheap linear/LayerNorm ops to PyTorch.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise proj * gate * mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,            # *[total_elems]  fp16
    gate_ptr,            # *[total_elems]  fp16
    mask_ptr,            # *[mask_elems]   fp16
    out_ptr,             # *[total_elems]  fp16
    total_elems,         # i64
    hidden_dim,          # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    # broadcast mask over hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(out_ptr + offs, proj * gate * m, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse ``proj * gate * mask`` where ``mask`` is (B,N,N) and is broadcast
    over the hidden dimension ``H``. All tensors must be FP16 and contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],          # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for (batch, M, K) @ (batch, K, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr,                # *[batch, M, K]   fp16
    b_ptr,                # *[batch, K, N]   fp16
    c_ptr,                # *[batch, M, N]   fp16
    M, N, K,
    stride_am, stride_ak,       # strides for A
    stride_bk, stride_bn,       # strides for B
    stride_cm, stride_cn,       # strides for C
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # masks for out‑of‑bounds loads
        mask_a = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        mask_b = (offs_k < K)[:, None] & (offs_n < N)[None, :]

        # pointers with batch offset
        a = tl.load(
            a_ptr
            + pid_batch * batch_stride_a
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_a,
            other=0.0,
        )
        b = tl.load(
            b_ptr
            + pid_batch * batch_stride_b
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_b,
            other=0.0,
        )

        # cast to FP32 for accumulation
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # write back
    mask_c = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(
        c_ptr
        + pid_batch * batch_stride_c
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        acc.to(tl.float16),
        mask=mask_c,
    )


def _batched_gemm(a: torch.Tensor,
                  b: torch.Tensor,
                  block_m: int = 128,
                  block_n: int = 128,
                  block_k: int = 32) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    a: (batch, M, K)   fp16
    b: (batch, K, N)   fp16
    Returns c: (batch, M, N)  fp16
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    # output allocation
    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)

    # strides
    stride_am, stride_ak = a.stride(1), a.stride(2)
    stride_bk, stride_bn = b.stride(1), b.stride(2)
    stride_cm, stride_cn = c.stride(1), c.stride(2)

    batch_stride_a = a.stride(0)
    batch_stride_b = b.stride(0)
    batch_stride_c = c.stride(0)

    grid = (
        batch,
        math.ceil(M / block_m),
        math.ceil(N / block_n),
    )

    _batched_gemm_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return c


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3 style).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict with all linear/LayerNorm parameters
            - config: dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Fuse mask, projection and gate
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction via batched GEMM (Triton)
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # reshape to (B*H, N, N) for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched GEMM: left @ rightᵀ
    out_flat = _batched_gemm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
