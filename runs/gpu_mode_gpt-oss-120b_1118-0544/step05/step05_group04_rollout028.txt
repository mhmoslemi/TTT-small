@triton.jit
def gemm_kernel(
    A, B, C,
    M, N, K,
    stride_Ab, stride_Ai, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Ci, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block start coordinates.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for output coordinates
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Compute masks
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A + pid_batch * stride_Ab + offs_m[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = B + pid_batch * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = C + pid_batch * stride_Cb + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cn
    c = acc.to(tl.float16)

"""
TriMul (outgoing) – fast Triton implementation
------------------------------------------------
1. Layer‑norm the input (float32 weights, fp16 accumulation).
2. Linear projections to hidden dimension H (left/right) and three sigmoid gates
   (left_gate, right_gate, out_gate).
3. Fuse mask (if present) and the two gates in a Triton kernel:
   left_out = left   * mask * left_gate
   right_out = right * mask * right_gate
4. Reshape to (B·H, N, N) and transpose the right operand so that the
   reduction   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   becomes a batched GEMM:   out = left @ rightᵀ.
   The GEMM is performed by a custom Triton kernel operating on fp16
   inputs and accumulating in fp32.
5. Reshape back to (B, N, N, H) and apply a second LayerNorm over H.
6. Multiply by the output gate and project back to the original channel
   dimension dim.
7. Cast the result back to the original dtype.

All heavy element‑wise work (mask & gates, batched GEMM) is done in Triton,
while the remaining linear / norm operations use PyTorch (cuBLAS) for
maximum throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Fuse mask and left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,          # const fp16*  (B,N,N,H) flattened
    right_ptr,         # const fp16*
    mask_ptr,          # const fp16*  (B,N,N) flattened
    left_gate_ptr,     # const fp16*
    right_gate_ptr,    # const fp16*
    left_out_ptr,      #      fp16*
    right_out_ptr,     #      fp16*
    H: tl.constexpr,   # hidden dimension
    total_elems: tl.constexpr,   # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # load values (masked)
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # mask is broadcast over the hidden dimension
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # apply mask and gates
    tl.store(left_out_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


# ----------------------------------------------------------------------
# 2) Batched GEMM kernel (out = left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A,            # fp16*  (B*H, N, N)  – left operand
    B,            # fp16*  (B*H, N, N)  – rightᵀ operand (already transposed)
    C,            # fp16*  (B*H, N, N)  – output
    M: tl.constexpr,   # = N (rows of A)
    N: tl.constexpr,   # = N (cols of Bᵀ)
    K: tl.constexpr,   # = N (reduction dim)
    stride_Ab, stride_Ai, stride_Ak,   # strides for A
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (k, n)
    stride_Cb, stride_Ci, stride_Cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # index over B*H
    pid_i = tl.program_id(1)              # tile over M dimension
    pid_j = tl.program_id(2)              # tile over N dimension

    # Block start positions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds
    mask_i = offs_i < M
    mask_j = offs_j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = (A
                  + pid_batch * stride_Ab
                  + offs_i[:, None] * stride_Ai
                  + offs_k[None, :] * stride_Ak)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile (K x N) – note B is already transposed
        b_ptrs = (B
                  + pid_batch * stride_Bb
                  + offs_k[:, None] * stride_Bk
                  + offs_j[None, :] * stride_Bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Multiply‑add
        acc += tl.dot(a, b).to(tl.float32)

    # Write result
    c_ptrs = (C
              + pid_batch * stride_Cb
              + offs_i[:, None] * stride_Ci
              + offs_j[None, :] * stride_Cn)
    c = acc.to(tl.float16)
    tl.store(c_ptrs,
             c,
             mask=mask_i[:, None] & mask_j[None, :])

    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None (binary)
            - weights      : dict of model parameters (float32 tensors)
            - config       : dict with keys
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"     – bool, ignore mask if True (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """

    # Unpack
    # --------------------------------------------------------------
    inp, mask, weights, config = data
    device = inp.device
    orig_dtype = inp.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) LayerNorm over channel dimension
    # --------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # Work in fp16 for the heavy part
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2) Cast linear weights to fp16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------
    # 3) Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_w)      # (B, N, N, H)
    right = F.linear(x, right_proj_w)    # (B, N, N, H)

    # --------------------------------------------------------------
    # 4) Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # --------------------------------------------------------------
    # 5) Apply mask and gates (fused Triton kernel)
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_tensor = mask.to(device=device, dtype=torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    total_elements = left.numel()                # B * N * N * H
    BLOCK = 1024

    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = (triton.cdiv(total_elements, BLOCK),)
    _apply_mask_gate_kernel[grid](
        left, right,
        mask_tensor.view(-1),
        left_gate, right_gate,
        left_out, right_out,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6) Prepare for batched GEMM (outgoing TriMul reduction)
    # --------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N) and make contiguous
    left_perm = left_out.permute(0, 3, 1, 2).contiguous()
    right_perm = right_out.permute(0, 3, 1, 2).contiguous()
    # Transpose each (N,N) matrix of right so that we get (K,N) for GEMM
    right_t = right_perm.transpose(-2, -1).contiguous()

    # Flatten batch and hidden dimension -> (B*H, N, N)
    left_flat = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_t.view(B * hidden_dim, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # --------------------------------------------------------------
    # 7) Batched GEMM via Triton
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,               # M
        N,               # N
        N,               # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A strides
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2), # B strides (K,N)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),       # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 8) Reshape back to (B, N, N, H)
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 9) Second LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 10) Apply output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 11) Final linear projection back to original dim
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 12) Cast back to original dtype if needed
    # --------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
