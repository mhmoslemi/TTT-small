   back to the original channel dimension.
6. Cast the result to the original input dtype.

Only the mask‑gate fusion is Triton‑implemented; the heavy N³ reduction uses
high‑performance cuBLAS through torch.matmul.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,            # const float* left  (B,N,N,H) flattened
    right_ptr,           # const float* right (B,N,N,H)
    mask_ptr,            # const float* mask  (B,N,N) flattened
    left_gate_ptr,       # const float* left_gate  (B,N,N,H)
    right_gate_ptr,      # const float* right_gate (B,N,N,H)
    left_out_ptr,        # float* left_out
    right_out_ptr,       # float* right_out
    H: tl.constexpr,     # hidden dimension
    total_elements: tl.int64,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask and gate: left_out = left * mask * left_gate, same for right."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    in_bounds = offs < total_elements

    # Load tensors
    l = tl.load(left_ptr + offs, mask=in_bounds).to(tl.float16)
    r = tl.load(right_ptr + offs, mask=in_bounds).to(tl.float16)
    lg = tl.load(left_gate_ptr + offs, mask=in_bounds).to(tl.float16)
    rg = tl.load(right_gate_ptr + offs, mask=in_bounds).to(tl.float16)

    # Compute mask index: each H elements share one mask element
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=in_bounds).to(tl.float16)

    # Apply mask and gates
    tl.store(left_out_ptr + offs, l * m * lg, mask=in_bounds)
    tl.store(right_out_ptr + offs, r * m * rg, mask=in_bounds)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask        : torch.Tensor of shape [B, N, N] or None (binary)
            - weights     : dict of model parameters (float32 tensors)
            - config      : dict with keys
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – if True, ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ----------------------------------------------------------
    # Unpack
    # ----------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # ----------------------------------------------------------
    # 2) Cast to fp16 for the heavy computation
    # ----------------------------------------------------------
    x = x.to(torch.float16)

    # ----------------------------------------------------------
    # 3) Cast all linear weights to fp16
    # ----------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(device, torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(device, torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(device, torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(device, torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(device, torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(device, torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(device, torch.float16)
    w_to_out      = weights["to_out.weight"].to(device, torch.float16)

    # ----------------------------------------------------------
    # 4) Linear projections to hidden dimension H
    # ----------------------------------------------------------
    left  = F.linear(x, w_left_proj)   # (B, N, N, H)
    right = F.linear(x, w_right_proj)  # (B, N, N, H)

    # ----------------------------------------------------------
    # 5) Compute gates (sigmoid of linear projections)
    # ----------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w_left_gate))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, w_right_gate))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, w_out_gate))    # (B, N, N, H)

    # ----------------------------------------------------------
    # 6) Prepare mask (broadcast over hidden dim)
    # ----------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_tensor = mask.to(torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # ----------------------------------------------------------
    # 7) Fuse mask and left/right gates with Triton
    # ----------------------------------------------------------
    total_elements = left.numel()                     # B * N * N * H
    BLOCK = 1024

    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = (triton.cdiv(total_elements, BLOCK),)
    _mask_gate_fusion_kernel[grid](
        left,
        right,
        mask_tensor.view(-1),
        left_gate,
        right_gate,
        left_out,
        right_out,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ----------------------------------------------------------
    # 8) Core reduction: batched GEMM over the k dimension
    #    Compute out[b,i,j,h] = Σ_k left_out[b,i,k,h] * right_out[b,j,k,h]
    # ----------------------------------------------------------
    # Bring hidden dim to batch dimension: (B, H, N, N)
    left_perm  = left_out.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right_out.permute(0, 3, 1, 2)  # (B, H, N, N)

    # Batched matrix multiplication (cuBLAS via torch.matmul)
    out = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # (B, H, N, N)

    # Restore original layout (B, N, N, H)
    out = out.permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------
    # 9) Output LayerNorm over hidden dimension H
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
    )

    # ----------------------------------------------------------
    # 10) Apply output gate
    # ----------------------------------------------------------
    out = out * out_gate

    # ----------------------------------------------------------
    # 11) Final linear projection back to original dimension C
    # ----------------------------------------------------------
    out = F.linear(out, w_to_out)   # (B, N, N, C)

    # ----------------------------------------------------------
    # 12) Cast back to original dtype
    # ----------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
