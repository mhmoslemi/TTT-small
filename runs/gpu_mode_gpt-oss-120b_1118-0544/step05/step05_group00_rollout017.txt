   stride_bk, stride_bn,  # B: (K,N) proper ordering (since we want B^T)
   stride_cm, stride_cn,  # C: (M,N)

   batch_stride_a, batch_stride_b, batch_stride_c,  # per batch offset

   batch_id = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   # Compute offsets for this block:

   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   offs_k = tl.arange(0, BLOCK_K)

   m_mask = offs_m < M
   n_mask = offs_n < N

   # Compute base pointers for this batch
   a_ptrs = a_ptr + batch_id * batch_stride_a + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input (float32 for stability) and cast to fp16.
2. Linear projections (bias‑free) to a hidden dimension H:
   left, right, left_gate, right_gate, out_gate.
3. Apply optional mask (broadcast over H) and sigmoid gates.
4. Core contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   implemented as a batched matrix multiplication C = A @ Bᵀ using a
   custom Triton kernel (batch = B·H, each block computes a tile of C).
5. Output LayerNorm over the hidden dimension, apply the output gate and
   a final linear projection back to the original dimension.
6. Cast the result back to the input dtype (float32).

The Triton kernel fuses the whole batched GEMM, which is the dominant
O(N³·H) work, and therefore yields a much lower runtime than the
reference implementation that relies on torch.bmm.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B shape: (B, M, K), C: (B, M, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_transpose_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to tensors A, B, C
    M, N, K,                           # matrix dimensions (all int)
    stride_am, stride_ak,              # A: (batch, M, K) strides
    stride_bk, stride_bn,              # B (to be read as transposed): (batch, K, N) strides
    stride_cm, stride_cn,              # C: (batch, M, N) strides
    batch_stride_a, batch_stride_b, batch_stride_c,   # stride between batches
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the current batch
    a_batch_ptr = a_ptr + batch_id * batch_stride_a
    b_batch_ptr = b_ptr + batch_id * batch_stride_b
    c_batch_ptr = c_ptr + batch_id * batch_stride_c

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K
        cur_k = k_offset + offs_k
        k_mask = cur_k < K

        # Load A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am
                                    + cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & k_mask[None, :], other=0.0)

        # Load B tile (transposed): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch_ptr + (cur_k[:, None] * stride_bk
                                + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=k_mask[:, None] & mask_n[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

        # Advance K pointers (implicit via cur_k on next iteration)
    # End of K loop

    # Store C tile
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm
                           + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape (B, N, N, C)
        - mask         : torch.Tensor of shape (B, N, N) or None
        - weights      : dict of model parameters (float32 tensors)
        - config       : dict with keys "dim" (C) and "hidden_dim" (H)

    Returns
    -------
    torch.Tensor of shape (B, N, N, C) (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) and cast to fp16
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"].to(device, torch.float32)
    ln_b = weights["norm.bias"].to(device, torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=ln_w, bias=ln_b)
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free) to hidden dim
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)

    left = F.linear(x, left_proj_w)                      # (B,N,N,H)
    right = F.linear(x, right_proj_w)                    # (B,N,N,H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))     # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3. Apply mask (if any) and elementwise gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B,N,N) -> (B,N,N,1) -> fp16
        mask_fp = mask.to(device, torch.float16).unsqueeze(-1)
        left = left * mask_fp
        right = right * mask_fp

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core O(N³·H) contraction using Triton batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Reshape to (B*H, N, N) for batched GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_mat = torch.empty_like(left_mat)  # (B*H, N, N)

    # Strides (in elements)
    a_batch_stride, a_m_stride, a_k_stride = left_mat.stride()
    b_batch_stride, b_k_stride, b_n_stride = right_mat.stride()   # note: k and n swapped for transpose
    c_batch_stride, c_m_stride, c_n_stride = out_mat.stride()

    # Choose tile sizes (tuned for typical B200 FP16 performance)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                                 # batch dimension
        (N + BLOCK_M - 1) // BLOCK_M,          # blocks in M
        (N + BLOCK_N - 1) // BLOCK_N,          # blocks in N
    )

    _batched_gemm_transpose_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                               # M, N, K
        a_m_stride, a_k_stride,
        b_k_stride, b_n_stride,
        c_m_stride, c_n_stride,
        a_batch_stride, b_batch_stride, c_batch_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (half‑precision)

    to_out_w = weights["to_out.weight"].to(device, torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------
    # 6. Cast back to original dtype and return
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)
    return out
