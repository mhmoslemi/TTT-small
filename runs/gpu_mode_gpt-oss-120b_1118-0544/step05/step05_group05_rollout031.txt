@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_lk, stride_li, stride_lh,
    stride_rb, stride_rk, stride_rj, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program_id indexes over (b, h, block_m, block_n). For simplicity, flatten: pid = program_id(0)
    # total blocks = B * H * (ceil(N/BLOCK_M) * ceil(N/BLOCK_N))
    pid = tl.program_id(0)
    # compute indexes: first compute total blocks per batch channel.
    num_block_m = tl.cdiv(N, BLOCK_M)
    num_block_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_block_m * num_block_n
    batch_idx = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    pid_m = block_idx // num_block_n
    pid_n = block_idx % num_block_n
    # compute offsets for i and j
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute batch and head index
    b = batch_idx // H
    h = batch_idx % H
    # pointers
    left_ptrs = left_ptr + b*stride_lb + h*stride_lh + offs_m[:,None]*stride_li + tl.arange(0, BLOCK_K)[None,:]*stride_lk
    right_ptrs = right_ptr + b*stride_rb + h*stride_rh + offs_n[:,None]*stride_rj + tl.arange(0, BLOCK_K)[None,:]*stride_rk
    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load tiles
        left_tile = tl.load(left_ptrs, mask=(offs_m[:,None] < N) & (tl.arange(0, BLOCK_K)[None,:] < N), other=0.0)
        right_tile = tl.load(right_ptrs, mask=(offs_n[:,None] < N) & (tl.arange(0, BLOCK_K)[None,:] < N), other=0.0)
        # dot product: left: MxK, right: NxK (need transposed? we have right tile across K but want right^T, which is KxN)
        # Since we loaded right as shape K x N? Actually right_ptrs: we set dimension semantics: we want right[j,k] used as right[j,k] i.e., we need right tile shaped K x N? We have right_ptrs defined with k dimension as stride_rk and j as stride_rj. For each k block we need to load right tile for all j values.
        # We loaded right_tile as shape (BLOCK_N, BLOCK_K). For matmul left (M x K) * right^T (K x N). right_tile should be transposed: right_tile_T = tl.trans(right_tile) # shape (BLOCK_K, BLOCK_N)
        # Then acc += tl.dot(left_tile, right_tile_T)
        acc += tl.dot(left_tile, tl.trans(right_tile))
        # increment pointers
        left_ptrs += BLOCK_K * stride_lk
        right_ptrs += BLOCK_K * stride_rk
    # store result
    out_ptrs = out_ptr + b*stride_ob + h*stride_oh + offs_m[:,None]*stride_oi + offs_n[None,:]*stride_oj
    mask = (offs_m[:,None] < N) & (offs_n[None,:] < N)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# gate+mask fuse kernel
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

# core tri-mul kernel
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute block grid sizes
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n
    total_batches = B * H
    batch_idx = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # compute batch and head indices
    b = batch_idx // H
    h = batch_idx % H

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for this (b,h)
    left_base = left_ptr + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base = out_ptr + b * stride_ob + h * stride_oh

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        # compute mask for each tile
        mask_m = offs_m < N
        mask_n = offs_n < N
        mask_k = (k_offset + offs_k) < N
        # load left tile (M x K)
        left_tile = tl.load(
            left_base + (offs_m[:, None] * stride_li) + ((k_offset + offs_k)[None, :] * stride_lk),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load right tile (K x N)
        right_tile = tl.load(
            right_base + ((k_offset + offs_k)[:, None] * stride_rk) + (offs_n[None, :] * stride_rj),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # accumulate
        acc += tl.dot(left_tile, right_tile)
        # continue loop
    # store result
    out_ptrs = out_base + (offs_m[:, None] * stride_oi) + (offs_n[None, :] * stride_oj)
    mask_out = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)

def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    The implementation follows the reference PyTorch code but fuses
    mask‑gate multiplication and the O(N³·H) contraction with Triton
    kernels. All compute‑heavy steps use fp16, while the final output
    is returned in the input dtype (usually fp32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, torch.float32)
    norm_b = weights["norm.bias"].to(device, torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for all subsequent ops
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gates
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w) # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,H)

    # -----------------------------------------------------------------
    # 4. Fuse mask and gate multiplication using Triton
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape (B,N,N) -> broadcast over hidden dim
        mask_fp = mask.to(device).to(torch.float16)
        # Flatten tensors for kernel
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)   # (B*N*N,)
        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask → direct element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: left @ rightᵀ per channel (batched GEMM) via Triton
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Compute strides for the kernel
    stride_lb = left.stride(0)
    stride_li = left.stride(1)
    stride_lk = left.stride(2)
    stride_lh = left.stride(3)

    stride_rb = right.stride(0)
    stride_rj = right.stride(1)   # j dimension
    stride_rk = right.stride(2)
    stride_rh = right.stride(3)

    stride_ob = out_gate.stride(0)  # will allocate new out tensor later
    # allocate output tensor for contraction
    out = torch.empty((bs, N, N, H), dtype=torch.float16, device=device)
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_oh = out.stride(3)

    # Triton grid configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = bs * H * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        bs,
        N,
        H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # Cast back to original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

  * element‑wise mask·gate multiplication,
  * the O(N³·H) contraction (left @ rightᵀ) per hidden channel,
using Triton kernels. All heavy arithmetic is performed in fp16 while the
final result is returned in the input dtype (usually fp32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask, left‑gate and right‑gate (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape as left/right
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left      = tl.load(left_ptr      + offs, mask=active, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden_dim entries share the same mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


# ----------------------------------------------------------------------
# Kernel 2: core O(N³·H) contraction (left @ rightᵀ) per hidden channel
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,   # [B,N,N,H] fp16
    right_ptr,  # [B,N,N,H] fp16
    out_ptr,    # [B,N,N,H] fp16
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # ---------------------------------------
    # 1) Determine which (b, h) and tile we handle
    # ---------------------------------------
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n                 # tiles per (b,h)
    batch_h_idx   = pid // blocks_per_batch                         # 0 .. B*H-1
    tile_idx      = pid %  blocks_per_batch
    pid_m = tile_idx // num_blocks_n
    pid_n = tile_idx %  num_blocks_n

    b = batch_h_idx // H
    h = batch_h_idx % H

    # ---------------------------------------
    # 2) Offsets for this tile
    # ---------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i indices
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j indices
    offs_k = tl.arange(0, BLOCK_K)                    # k indices

    # Base pointers that already include batch and head offsets
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # Accumulator in float32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------
    # 3) Loop over K dimension
    # ---------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K

        # Masks for out‑of‑bounds handling
        mask_m = offs_m < N
        mask_n = offs_n < N
        mask_k = (k_offset + offs_k) < N

        # Load left tile: shape (M, K)
        left_tile = tl.load(
            left_base
            + (offs_m[:, None] * stride_li)
            + ((k_offset + offs_k)[None, :] * stride_lk),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load right tile (transposed already): shape (K, N)
        right_tile = tl.load(
            right_base
            + ((k_offset + offs_k)[:, None] * stride_rk)
            + (offs_n[None, :] * stride_rj),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate multiplication
        acc += tl.dot(left_tile, right_tile)

    # ---------------------------------------
    # 4) Write out the result
    # ---------------------------------------
    out_ptrs = out_base \
        + (offs_m[:, None] * stride_oi) \
        + (offs_n[None, :] * stride_oj)
    mask_out = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, dim)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (see code)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, dim) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, torch.float32)
    norm_b = weights["norm.bias"].to(device, torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy parts
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, torch.float16)

    # -----------------------------------------------------------------
    # 4. Linear projections and gates
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_w)       # (B, N, N, H)
    right = F.linear(x, right_proj_w)     # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 5. Fuse mask & gate multiplications (Triton)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device, torch.float16)

        # Flatten tensors for the kernel (no copy, just view)
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)           # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Restore original shapes
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask → simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Core contraction: left @ rightᵀ per hidden channel (Triton)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Allocate output tensor for the contraction
    out = torch.empty((bs, N, N, H), dtype=torch.float16, device=device)

    # Compute strides needed by the kernel
    stride_lb = left.stride(0)
    stride_li = left.stride(1)
    stride_lk = left.stride(2)
    stride_lh = left.stride(3)

    stride_rb = right.stride(0)
    stride_rj = right.stride(1)   # j dimension
    stride_rk = right.stride(2)
    stride_rh = right.stride(3)

    stride_ob = out.stride(0)
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_oh = out.stride(3)

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = bs * H * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        bs,
        N,
        H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 7. Output LayerNorm, output gate, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # Cast back to original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
