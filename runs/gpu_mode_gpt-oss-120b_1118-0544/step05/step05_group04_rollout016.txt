def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
                     bias=weights["norm.bias"].to(device=device, dtype=torch.float32))
    # Cast to fp16 for compute
    x = x.to(torch.float16)

    # Linear weights to fp16
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # Projections
    left = F.linear(x, left_proj_w)  # (B,N,N,H)
    right = F.linear(x, right_proj_w)  # (B,N,N,H)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Mask handling
    if mask is not None and not nomask:
        mask = mask.to(x.dtype).unsqueeze(-1) # (B,N,N,1)
        left = left * mask * left_gate
        right = right * mask * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # Core reduction using batched matmul
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2)  # (B,H,N,N)
    right_perm = right.permute(0, 3, 2, 1)  # (B,H,N,N) transposed on last two dims
    # torch.matmul works on batch dims of left_perm (B*H) * (B*H)
    out_mat = torch.matmul(left_perm, right_perm)  # (B,H,N,N)
    out = out_mat.permute(0, 2, 3, 1)  # (B,N,N,H)

    # To_out layernorm
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # Apply out_gate
    out = out * out_gate

    # Final linear projection
    out = F.linear(out, to_out_w)  # (B,N,N,dim)

    # Cast back to original dtype
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
```

"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) in float32.
2. Cast to float16 and compute three linear projections:
   * left  = x @ left_proj_weight          → (B,N,N,H)
   * right = x @ right_proj_weight         → (B,N,N,H)
   * left_gate  = sigmoid(x @ left_gate_weight)   → (B,N,N,H)
   * right_gate = sigmoid(x @ right_gate_weight) → (B,N,N,H)
   * out_gate   = sigmoid(x @ out_gate_weight)   → (B,N,N,H)
3. Fuse mask, left‑gate and right‑gate using a tiny Triton kernel.
4. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication
   (left permuted to (B,H,N,N) multiplied by transposed right).
5. Layer‑normalize over the hidden dimension, multiply by out_gate,
   and project back to the original channel dimension.
All heavy arithmetic is performed in float16; the final result is cast
back to the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask and gate (left/right) element‑wise
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,        # const * left  (B,N,N,H) flattened
    right_ptr,       # const * right (B,N,N,H)
    mask_ptr,        # const * mask  (B,N,N) flattened
    left_gate_ptr,   # const * left_gate  (B,N,N,H)
    right_gate_ptr,  # const * right_gate (B,N,N,H)
    left_out_ptr,    # * left_out  (B,N,N,H)
    right_out_ptr,   # * right_out (B,N,N,H)
    H: tl.constexpr,               # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load tensors
    l  = tl.load(left_ptr + offs, mask=mask)
    r  = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Broadcast mask over hidden dimension
    mask_idx = offs // H                     # index in (B,N,N) flatten
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply mask and gates
    tl.store(left_out_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C]
            - mask         : torch.Tensor [B, N, N] or None (binary)
            - weights      : dict of model parameters (float32 tensors)
            - config       : dict with keys
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – if True, ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # Cast to fp16 for the heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B,N,N,H)
    right = F.linear(x, right_proj_w)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4) Compute gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5) Prepare mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_tensor = mask.to(dtype=x.dtype, device=device)   # (B,N,N)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=x.dtype, device=device)

    # ------------------------------------------------------------------
    # 6) Fuse mask and gates with Triton
    # ------------------------------------------------------------------
    total_elements = left.numel()                      # B * N * N * H
    BLOCK = 1024                                        # threads per program

    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = (triton.cdiv(total_elements, BLOCK),)
    _mask_gate_fuse_kernel[grid](
        left,
        right,
        mask_tensor.view(-1),          # flatten mask
        left_gate,
        right_gate,
        left_out,
        right_out,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) so that batch dimension = B*H
    left_mat  = left_out.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_mat = right_out.permute(0, 3, 2, 1)  # (B, H, N, N)  (transpose on last two dims)

    # Batched matrix multiplication (uses cuBLAS / tensor cores)
    out_mat = torch.matmul(left_mat, right_mat)   # (B, H, N, N)

    # Restore original ordering (B, N, N, H)
    out = out_mat.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10) Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 11) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
