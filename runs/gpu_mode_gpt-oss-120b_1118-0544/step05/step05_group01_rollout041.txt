@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_batch, stride_i, stride_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)  # combined batch*hidden
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute b, h
    b = pid // H
    h = pid % H
    base = b * stride_batch + h  # scalar offset

    # offsets for i,j,k
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    mask_i = i < N
    mask_j = j < N
    mask_k = k < N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K + k
        mask_k = k_offset < N

        a_ptrs = left_ptr + base + i[:, None] * stride_i + k_offset[None, :] * stride_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        b_ptrs = right_ptr + base + k_offset[:, None] * stride_i + j[None, :] * stride_k
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)
    # store
    out_ptrs = out_ptr + base + i[:, None] * stride_i + j[None, :] * stride_k
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm 1
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # cast to fp16
    x = x.to(torch.float16)

    # load weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # linear projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)  # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # gates
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # core reduction
    out = triton_core_mul(left, right)

    # output layernorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)

    # cast back to original dtype
    out = out.to(input_tensor.dtype)
    return out
```

"""
custom_kernel: Triton‑accelerated outgoing TriMul forward pass.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (`dim`).
2. Cast to FP16 for compute‑heavy parts.
3. Linear projections (no bias) → left/right tensors.
4. Compute gating logits, apply sigmoid, and (optionally) fuse mask
   with projection + gate using a small Triton kernel.
5. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM with the K‑dimension summed in a
   custom Triton kernel (no extra memory copies or permutes).
6. Layer‑norm over the hidden dimension (`hidden_dim`), multiply by
   the output gate and project back to the original dimensionality.
7. Cast back to the original dtype (usually float32).

The implementation keeps most heavy work in FP16 while accumulating
in FP32 for numeric stability. Both the mask‑gate fusion and the
core O(N³) reduction are performed on the GPU via Triton.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask (broadcast over H)
# --------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # total number of elements in proj/gate (=B*N*N*H)
    hidden_dim,        # H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # each consecutive hidden_dim entries share the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse projection, gate and mask using Triton.
    All tensors must be fp16, contiguous and have shapes:
        proj, gate : (B, N, N, H)
        mask       : (B, N, N)   (binary 0/1)
    Returns: proj * gate * mask  (broadcast over H).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# --------------------------------------------------------------
# Triton kernel: core TriMul reduction (batched GEMM)
# --------------------------------------------------------------
@triton.jit
def _tri_mul_core_kernel(
    left_ptr,   # fp16 * (B*N*N*H,)
    right_ptr,  # fp16 * (B*N*N*H,)
    out_ptr,    # fp16 * (B*N*N*H,)
    B, N, H,
    stride_batch,    # N*N*H
    stride_i,        # N*H   (row stride)
    stride_k,        # H     (col stride)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid = tl.program_id(0)      # combined batch * hidden index
    pid_m = tl.program_id(1)    # block row
    pid_n = tl.program_id(2)    # block col

    # Decompose pid into batch and hidden index
    b = pid // H                # batch index
    h = pid % H                 # hidden index
    base = b * stride_batch + h  # start offset for this (b,h)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: left[b,i,k,h]
        a_ptrs = left_ptr + base + offs_m[:, None] * stride_i + offs_k[None, :] * stride_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile transposed: right[b,k,j,h] (we need (k,j) order)
        b_ptrs = right_ptr + base + offs_k[:, None] * stride_i + offs_n[None, :] * stride_k
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Write back
    out_ptrs = out_ptr + base + offs_m[:, None] * stride_i + offs_n[None, :] * stride_k
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _tri_mul_core(left: torch.Tensor, right: torch.Tensor,
                  block_m: int = 64, block_n: int = 64, block_k: int = 32) -> torch.Tensor:
    """
    Perform the core TriMul reduction using Triton.
    left/right: (B, N, N, H) fp16 tensors.
    Returns: out tensor of same shape and dtype.
    """
    B, N, _, H = left.shape
    # Ensure contiguous memory for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    stride_batch = N * N * H
    stride_i = N * H
    stride_k = H

    grid = (
        B * H,
        (N + block_m - 1) // block_m,
        (N + block_n - 1) // block_n,
    )
    _tri_mul_core_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_batch,
        stride_i,
        stride_k,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return out


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C]
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim), dtype = input dtype

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all linear weights to FP16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B, N, N, hidden_dim)
    # --------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # --------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Apply mask and gating (fused via Triton if mask present)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction (batched GEMM across hidden dimension)
    # --------------------------------------------------------------
    out = _tri_mul_core(left, right)

    # --------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(dtype_in)
    return out
