import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offset < total_elems

    left = tl.load(left_ptr + offset, mask=mask, other=0.0)
    right = tl.load(right_ptr + offset, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offset, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offset, mask=mask, other=0.0)

    if HAS_MASK:
        # mask index = offset // hidden_dim
        mask_idx = offset // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offset, left, mask=mask)
    tl.store(right_ptr + offset, right, mask=mask)

    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.
    Implements:
    1. LayerNorm over last dimension.
    2. Linear projections + sigmoid gates.
    3. Fused mask & gate multiplication via a Triton kernel.
    4. Core contraction via batched GEMM (left @ rightᵀ).
    5. Output LayerNorm, output gate, final linear projection.
    Input/Output dtype matches input tensor (float32 by default).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input LayerNorm (FP32 for stability)
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for compute intensive parts
    x = x.to(torch.float16)

    # 2. Load linear weights (fp16)
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)

    # 3. Linear projections and gates (fp16)
    left = F.linear(x, left_proj_w)   # [B,N,N,H]
    right = F.linear(x, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # 4. Fuse mask and gates with Triton kernel (in-place)
    if mask is not None and not nomask:
        # mask shape (B,N,N). broadcast to (B,N,N,1)
        mask_fp = mask.to(device).to(torch.float16)
        # flatten everything

        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)  # length B*N*N

        total = left_flat.numel()

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # 5. Core contraction: for each channel d compute left @ rightᵀ
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))  # (B*H,N,N)

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Output layernorm, output gate, final projection
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)

    # Cast to original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Input LayerNorm (float32 for numeric stability).
2. Linear projections `left` and `right` and three sigmoid gates.
3. Fuse mask, left‑gate and right‑gate multiplications in a small Triton
   kernel (in‑place, reduces memory traffic).
4. Core O(N³·H) contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication `left @ rightᵀ` for every
   channel `d`. Implemented with `torch.bmm` (cuBLAS) on fp16 tensors.
5. Output LayerNorm, output‑gate multiplication and final linear projection
   back to the original feature dimension.
All heavy arithmetic is performed in fp16; the result is cast back to the
input dtype (usually float32).

The kernel `_gate_mask_fuse_kernel` implements step 3 and must be compiled
with Triton 3.3.1.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # *fp16  [B,N,N,H] flattened
    right_ptr,              # *fp16  same shape
    left_gate_ptr,          # *fp16  same shape
    right_gate_ptr,         # *fp16  same shape
    mask_ptr,               # *fp16  [B,N,N] flattened
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,  # compile‑time bool
    BLOCK_SIZE: tl.constexpr,  # compile‑time int
):
    """In‑place multiplication:
           left  = left  * left_gate  * mask
           right = right * right_gate * mask
       `mask` is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load data (default 0.0 for out‑of‑bounds)
    left  = tl.load(left_ptr  + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    lgate = tl.load(left_gate_ptr  + offs, mask=active, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # One mask value per (b,i,k) entry, i.e. per hidden_dim block
        mask_idx = offs // hidden_dim
        mval = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * lgate * mval
        right = right * rgate * mval
    else:
        left  = left  * lgate
        right = right * rgate

    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)

    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)

    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)

    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)

    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Batched matrix multiplication: left @ rightᵀ
