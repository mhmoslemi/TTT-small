"""
custom_kernel: Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C).
2. Fuse the five linear projections (left, right, left_gate, right_gate,
   out_gate) into a single GEMM by concatenating their weight matrices.
3. Split the combined projection, apply sigmoid to the three gate tensors.
4. If a mask is provided, fuse mask, left‑gate and right‑gate multiplication
   in a Triton kernel:
   left  = left * mask * left_gate
   right = right * mask * right_gate
   Otherwise perform the two element‑wise multiplications in‑place.
5. Perform the core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is realized as a batched GEMM (B*H independent matrix
   multiplications) using torch.bmm.
6. Layer‑normalize over the hidden dimension, multiply by the output gate,
   and project back to the original channel dimension.
7. Cast the result to the original input dtype.

The kernel in step 4 provides the required Triton component.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,          # const float* left   (B,N,N,H) flattened
    right_ptr,         # const float* right
    mask_ptr,          # const float* mask   (B,N,N) flattened
    left_gate_ptr,     # const float* left_gate
    right_gate_ptr,    # const float* right_gate
    left_out_ptr,      # float* left_out (in‑place)
    right_out_ptr,     # float* right_out (in‑place)
    H: tl.constexpr,   # hidden dimension
    total_elements,    # total number of elements = B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load values
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Compute mask index (broadcast over H)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply mask and gates, write back in‑place
    tl.store(left_out_ptr + offs, l * lg * m, mask=mask)
    tl.store(right_out_ptr + offs, r * rg * m, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of model parameters (float32 tensors)
        - config       : dict with keys
            * "dim"         – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask"      – bool, True ⇒ ignore mask

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------------------------------------------------------
    # 1) Input LayerNorm (keep in original dtype for stability)
    # ---------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=orig_dtype)
    norm_bias = weights["norm.bias"].to(device=device, dtype=orig_dtype)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # Cast to fp16 for the heavy compute
    x = x.to(torch.float16)

    # ---------------------------------------------------------
    # 2) Fuse five linear projections into one GEMM
    # ---------------------------------------------------------
    # Convert projection weights to fp16 and concatenate
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    combined_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape (5*H, C)

    # Single linear projection
    combined = F.linear(x, combined_weight)  # (B, N, N, 5*H)

    # Split into the five tensors
    left, right, left_gate, right_gate, out_gate = torch.split(
        combined, hidden_dim, dim=-1
    )  # each (B,N,N,H)

    # ---------------------------------------------------------
    # 3) Gates: sigmoid
    # ---------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ---------------------------------------------------------
    # 4) Apply mask and gate multiplication (fused via Triton)
    # ---------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_tensor = mask.to(device=device, dtype=torch.float16)
        # flatten tensors for kernel launch
        total_elements = left.numel()  # B * N * N * H
        BLOCK = 65536

        # In‑place fusion: overwrite left and right
        grid = (triton.cdiv(total_elements, BLOCK),)
        _apply_mask_gate_kernel[grid](
            left,
            right,
            mask_tensor.view(-1),
            left_gate,
            right_gate,
            left,          # write result back into left
            right,         # write result back into right
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → simple in‑place gating
        left.mul_(left_gate)
        right.mul_(right_gate)
        if mask is not None:
            # mask exists but nomask=True → treat as all‑ones (no effect)
            pass

    # ---------------------------------------------------------
    # 5) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as B*H independent matrix multiplications.
    # ---------------------------------------------------------
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Batched GEMM
    out_flat = torch.bmm(left_flat, right_flat.transpose(-2, -1))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, H)

    # ---------------------------------------------------------
    # 6) Output LayerNorm over hidden dimension
    # ---------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ---------------------------------------------------------
    # 7) Apply output gate
    # ---------------------------------------------------------
    out = out * out_gate

    # ---------------------------------------------------------
    # 8) Final linear projection back to original dim
    # ---------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, C)

    # ---------------------------------------------------------
    # 9) Cast back to original dtype
    # ---------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
