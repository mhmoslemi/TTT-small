Pseudo:

```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_h, stride_left_i, stride_left_j,
    stride_right_b, stride_right_h, stride_right_i, stride_right_j,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(0) # batch*hidden index
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute b, h
    b = pid_bh // H
    h = pid_bh % H

    # offsets
    i_offsets = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offsets = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for i,j
    i_mask = i_offsets < N
    j_mask = j_offsets < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, N, BLOCK_K):
        k_offsets = k + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < N

        # load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + i_offsets[:, None] * stride_left_i + k_offsets[None, :] * stride_left_j
        left = tl.load(left_ptrs, mask = i_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

        # load right tile transposed: shape (BLOCK_K, BLOCK_N)
        # right is indexed as right[b, h, j, k] because we want transpose of right (j,k) for dot
        right_ptrs = right_ptr + b*stride_right_b + h*stride_right_h + k_offsets[:, None] * stride_right_i + j_offsets[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask = k_mask[:, None] & j_mask[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_ptr + b*stride_out_b + h*stride_out_h + i_offsets[:, None] * stride_out_i + j_offsets[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask = i_mask[:, None] & j_mask[None, :])
```

   back to the original feature dimension.
All heavy arithmetic is performed in fp16 with fp32 accumulation; the
final result is cast back to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
#  Kernel 1: fuse gates and optional mask (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,                # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,      # same shape
    mask_ptr,                           # [B,N,N] flattened
    total_elems, hidden_dim,            # runtime scalars
    HAS_MASK: tl.constexpr,             # compile‑time bool
    BLOCK_SIZE: tl.constexpr,           # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # load tensors
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # mask is broadcast over the hidden dimension
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


# ----------------------------------------------------------------------
#  Kernel 2: batched matrix multiply left @ rightᵀ for each channel
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,                     # [B,H,N,N] contiguous
    right_ptr,                    # same layout as left
    out_ptr,                      # output buffer
    B, N, H,                      # dimensions
    stride_left_b, stride_left_h, stride_left_i, stride_left_j,
    stride_right_b, stride_right_h, stride_right_i, stride_right_j,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr,        # tile size for i dimension
    BLOCK_N: tl.constexpr,        # tile size for j dimension
    BLOCK_K: tl.constexpr,        # tile size for reduction dimension k
):
    # program ids
    pid_bh = tl.program_id(0)           # combined batch * hidden index
    pid_i = tl.program_id(1)            # i‑tile
    pid_j = tl.program_id(2)            # j‑tile

    # decode batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # offsets for the i and j tiles
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_offset < N
    j_mask = j_offset < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_base in range(0, N, BLOCK_K):
        k_offset = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = (left_ptr
                     + b * stride_left_b
                     + h * stride_left_h
                     + i_offset[:, None] * stride_left_i
                     + k_offset[None, :] * stride_left_j)
        left = tl.load(left_ptrs,
                       mask=(i_mask[:, None] & k_mask[None, :]),
                       other=0.0).to(tl.float32)

        # right tile transposed (rightᵀ): we need right[j,k] → treat as (k,j)
        # by swapping strides i/j when addressing memory
        right_ptrs = (right_ptr
                      + b * stride_right_b
                      + h * stride_right_h
                      + k_offset[:, None] * stride_right_j   # note swapped stride
                      + j_offset[None, :] * stride_right_i)
        right = tl.load(right_ptrs,
                        mask=(k_mask[:, None] & j_mask[None, :]),
                        other=0.0).to(tl.float32)

        # matrix multiply‑accumulate
        acc += tl.dot(left, right)   # (M,K)·(K,N) → (M,N)

    # store result (cast back to fp16)
    out_ptrs = (out_ptr
                + b * stride_out_b
                + h * stride_out_h
                + i_offset[:, None] * stride_out_i
                + j_offset[None, :] * stride_out_j)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=(i_mask[:, None] & j_mask[None, :]))


# ----------------------------------------------------------------------
#  Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape (B, N, N, C)
        - mask         : torch.Tensor of shape (B, N, N) or None
        - weights      : dict of model parameters
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape (B, N, N, C) with the same dtype as ``input_tensor``.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]            # original feature dimension C
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Load linear weights (fp16)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections and gates
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)                     # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)                   # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 5. Fuse mask & gates (Triton) if a mask is present
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask -> fp16 and flatten
        mask_fp = mask.to(device).to(torch.float16)
        # flatten tensors
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_f    = left_gate.contiguous().view(-1)
        right_gate_f   = right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_f,
            right_gate_f,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask → element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core O(N³·H) contraction via a Triton kernel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B,N,N,hidden_dim

    # bring channel dimension forward for contiguous layout
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B,H,N,N)

    out_perm = torch.empty_like(left_perm)  # (B,H,N,N)

    # strides for the contiguous tensors
    s_lb, s_lh, s_li, s_lj = left_perm.stride()
    s_rb, s_rh, s_ri, s_rj = right_perm.stride()
    s_ob, s_oh, s_oi, s_oj = out_perm.stride()

    # tile sizes (tuned for typical B200 / Ampere/Hopper)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _tri_mul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B, N, H,
        s_lb, s_lh, s_li, s_lj,
        s_rb, s_rh, s_ri, s_rj,
        s_ob, s_oh, s_oi, s_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # permute back to original layout (B,N,N,H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, output gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    out = out * out_gate
    out = F.linear(out, to_out_w)        # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8. Cast back to input dtype
    # ------------------------------------------------------------------
    out = out.to(dtype_in)
    return out
```
