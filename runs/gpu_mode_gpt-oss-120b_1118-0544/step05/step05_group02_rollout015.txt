@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    b = pid_bh // H
    h = pid_bh % H

    # compute offsets
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    i_offsets = i_base + tl.arange(0, BLOCK_I)
    j_offsets = j_base + tl.arange(0, BLOCK_J)

    # compute mask for i and j
    i_mask = i_offsets < N
    j_mask = j_offsets < N

    # pointers base for this batch and head
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k
    for k_offset in range(0, N, BLOCK_K):
        # compute actual k length for this block
        cur_k = tl.minimum(N - k_offset, BLOCK_K)
        # offsets for k within block
        k_offsets = k_offset + tl.arange(0, BLOCK_K)
        # mask for k within bounds
        k_mask = k_offsets < N

        # set masks for loads
        left_mask = i_mask[:, None] & k_mask[None, :]
        right_mask = k_mask[:, None] & j_mask[None, :]

        # compute pointers
        left_ptrs = left_base + i_offsets[:, None] * stride_l_i + k_offsets[None, :] * stride_l_k
        right_ptrs = right_base + k_offsets[:, None] * stride_r_k + j_offsets[None, :] * stride_r_i

        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        acc += tl.dot(left_tile, right_tile)

    # store result
    out_ptrs = out_base + i_offsets[:, None] * stride_o_i + j_offsets[None, :] * stride_o_j
    out_mask = i_mask[:, None] & j_mask[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=out_mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', False)

    # 1. LayerNorm input
    x = torch.nn.functional.layer_norm(input_tensor,
                                        normalized_shape=(dim,),
                                        weight=weights['norm.weight'],
                                        bias=weights['norm.bias'])
    # cast to fp16
    x = x.to(torch.float16)

    # cast weight matrices to fp16
    left_proj_w = weights['left_proj.weight'].to(torch.float16)
    right_proj_w = weights['right_proj.weight'].to(torch.float16)
    left_gate_w = weights['left_gate.weight'].to(torch.float16)
    right_gate_w = weights['right_gate.weight'].to(torch.float16)
    out_gate_w = weights['out_gate.weight'].to(torch.float16)
    to_out_norm_w = weights['to_out_norm.weight'].to(torch.float16)
    to_out_norm_b = weights['to_out_norm.bias'].to(torch.float16)
    to_out_w = weights['to_out.weight'].to(torch.float16)

    # 2. Linear projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # 3. Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # 4. Apply mask and gates
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * left_gate * mask_f
        right = right_proj * right_gate * mask_f
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguity
    left = left.contiguous()
    right = right.contiguous()

    # 5. Core TriMul via Triton
    B,N,_,H = left.shape
    out = torch.empty_like(left)  # shape (B,N,N,H)

    # launch kernel
    trimul_kernel_launcher(left, right, out, BLOCK_I=64, BLOCK_J=64, BLOCK_K=32)

    # 6. Output LayerNorm
    out = torch.nn.functional.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # 7. Apply out gate and final linear
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)

    # 8. Cast back to original dtype
    out = out.to(input_tensor.dtype)

    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated core reduction.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (B, N, N, C).
2. Cast to FP16 for compute‑intensive stages.
3. Linear projections and gate projections (no bias) are performed with
   torch.nn.functional.linear on FP16 tensors.
4. Optional mask is fused with the projections and gates using simple
   element‑wise multiplication (torch).
5. Core O(N³·H) reduction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is implemented as a custom Triton kernel that performs a batched
   matrix multiplication L·Rᵀ for each head `h`.  The kernel tiles the
   i‑ and j‑dimensions (BLOCK_I × BLOCK_J) and loops over k in blocks
   (BLOCK_K), accumulating in FP32 and storing the result in FP16.
6. Second LayerNorm, output‑gate multiplication and final linear
   projection back to the original channel dimension.
7. Cast the result back to the input dtype (float32).

Only the heavy reduction (step 5) is implemented in Triton; all other
operations use PyTorch for simplicity.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_kernel(
    left_ptr,                # *[B,N,N,H]  fp16
    right_ptr,               # *[B,N,N,H]  fp16
    out_ptr,                 # *[B,N,N,H]  fp16
    B, N, H,                 # runtime ints
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_i = tl.program_id(0)          # tile row   (i)
    pid_j = tl.program_id(1)          # tile col   (j)
    pid_bh = tl.program_id(2)         # combined batch * head

    # decode batch and head from linear id
    b = pid_bh // H
    h = pid_bh % H

    # base offsets for this (b, h)
    left_base  = left_ptr  + b * stride_l_b  + h * stride_l_h
    right_base = right_ptr + b * stride_r_b  + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b  + h * stride_o_h

    # tile coordinates
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    i_offsets = i_base + tl.arange(0, BLOCK_I)
    j_offsets = j_base + tl.arange(0, BLOCK_J)

    i_mask = i_offsets < N
    j_mask = j_offsets < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k dimension
    for k_offset in range(0, N, BLOCK_K):
        k_offsets = k_offset + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < N

        # left tile: (i, k)
        left_ptrs = left_base + i_offsets[:, None] * stride_l_i + k_offsets[None, :] * stride_l_k
        left_mask = i_mask[:, None] & k_mask[None, :]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # right tile (transposed): (k, j)
        right_ptrs = right_base + k_offsets[:, None] * stride_r_k + j_offsets[None, :] * stride_r_i
        right_mask = k_mask[:, None] & j_mask[None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # dot: (i,k) · (k,j) -> (i,j)
        acc += tl.dot(left_tile, right_tile)

    # write result
    out_ptrs = out_base + i_offsets[:, None] * stride_o_i + j_offsets[None, :] * stride_o_j
    out_mask = i_mask[:, None] & j_mask[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=out_mask)


def _launch_trimul(left: torch.Tensor, right: torch.Tensor, out: torch.Tensor,
                  BLOCK_I=64, BLOCK_J=64, BLOCK_K=32):
    """Launch the Triton kernel for the core TriMul reduction."""
    B, N, _, H = left.shape
    grid = (math.ceil(N / BLOCK_I), math.ceil(N / BLOCK_J), B * H)

    # strides (in elements)
    lsb, lsi, lsk, lsh = left.stride()
    rsb, rsi, rsk, rsh = right.stride()
    osb, osi, osj, osh = out.stride()

    _trimul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        lsb, lsi, lsk, lsh,
        rsb, rsi, rsk, rsh,
        osb, osi, osj, osh,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, C] (float32)
            mask:        torch.Tensor [B, N, N] (float32 or bool)
            weights:     dict of model parameters (torch.Tensor)
            config:      dict containing 'dim' and 'hidden_dim' and optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6. Apply mask (if any) and gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 7. Core TriMul reduction using Triton
    # ------------------------------------------------------------------
    out = torch.empty_like(left)  # shape (B,N,N,H), FP16
    _launch_trimul(left, right, out, BLOCK_I=64, BLOCK_J=64, BLOCK_K=32)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)      # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype (float32)
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
