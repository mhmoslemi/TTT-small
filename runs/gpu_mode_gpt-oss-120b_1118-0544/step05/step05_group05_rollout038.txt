@triton.jit
def proj_gate_kernel(
    x_ptr, mask_ptr,
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    out_gate_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    B, N, dim, hidden_dim,
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lp_h, stride_lp_d,
    stride_lg_h, stride_lg_d,
    stride_rp_h, stride_rp_d,
    stride_rg_h, stride_rg_d,
    stride_og_h, stride_og_d,
    stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_h,
    stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_h,
    stride_og_out_b, stride_og_out_i, stride_og_out_j, stride_og_out_h,
    HAS_MASK: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    # Compute (b,i,j) coordinate
    pid = tl.program_id(0)  # flatten over B*N*N
    total_ij = B * N * N
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # hidden block offset
    hid = tl.program_id(1)
    offs_h = hid * BLOCK_H + tl.arange(0, BLOCK_H)
    mask_h = offs_h < hidden_dim

    # Initialize accumulators
    acc_left = tl.zeros((BLOCK_H,), dtype=tl.float32)
    acc_left_gate = tl.zeros((BLOCK_H,), dtype=tl.float32)
    acc_right = tl.zeros((BLOCK_H,), dtype=tl.float32)
    acc_right_gate = tl.zeros((BLOCK_H,), dtype=tl.float32)
    acc_out_gate = tl.zeros((BLOCK_H,), dtype=tl.float32)

    # Mask value (scalar per (b,i,j))
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + b*stride_mask_b + i*stride_mask_i + j*stride_mask_j, mask=tl.broadcast_to(True, (1,)), other=1.0)
        # mask_val broadcast to hidden later.
    else:
        mask_val = 1.0

    # Loop over dim
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # Load x segment
        x = tl.load(x_ptr + b*stride_xb + i*stride_xi + j*stride_xj + offs_d * stride_xd, mask=mask_d, other=0.0)

        # Load weight blocks
        w_left = tl.load(left_proj_ptr + offs_h[:, None] * stride_lp_h + offs_d[None, :] * stride_lp_d, mask=mask_h[:, None] & mask_d[None, :], other=0.0)
        w_left_gate = tl.load(left_gate_ptr + offs_h[:, None] * stride_lg_h + offs_d[None, :] * stride_lg_d, mask=mask_h[:, None] & mask_d[None, :], other=0.0)
        w_right = tl.load(right_proj_ptr + offs_h[:, None] * stride_rp_h + offs_d[None, :] * stride_rp_d, mask=mask_h[:, None] & mask_d[None, :], other=0.0)
        w_right_gate = tl.load(right_gate_ptr + offs_h[:, None] * stride_rg_h + offs_d[None, :] * stride_rg_d, mask=mask_h[:, None] & mask_d[None, :], other=0.0)
        w_out_gate = tl.load(out_gate_ptr + offs_h[:, None] * stride_og_h + offs_d[None, :] * stride_og_d, mask=mask_h[:, None] & mask_d[None, :], other=0.0)

        # Accumulate dot products (float32)
        acc_left += tl.dot(w_left, x).to(tl.float32)
        acc_left_gate += tl.dot(w_left_gate, x).to(tl.float32)
        acc_right += tl.dot(w_right, x).to(tl.float32)
        acc_right_gate += tl.dot(w_right_gate, x).to(tl.float32)
        acc_out_gate += tl.dot(w_out_gate, x).to(tl.float32)

    # Apply sigmoid to gates
    left_gate_sig = tl.sigmoid(acc_left_gate)
    right_gate_sig = tl.sigmoid(acc_right_gate)
    out_gate_sig = tl.sigmoid(acc_out_gate)

    # Apply mask and gates to projections
    left_out = acc_left * left_gate_sig * mask_val
    right_out = acc_right * right_gate_sig * mask_val

    # Store results (cast to fp16)
    # compute output pointer offsets
    off_out = b*stride_lo_b + i*stride_lo_i + j*stride_lo_j + offs_h * stride_lo_h
    tl.store(left_out_ptr + off_out, left_out.to(tl.float16), mask=mask_h)
    tl.store(right_out_ptr + off_out, right_out.to(tl.float16), mask=mask_h)
    tl.store(out_gate_out_ptr + off_out, out_gate_sig.to(tl.float16), mask=mask_h)

def custom_kernel(data):

    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Extract weights and convert to fp16
    # Norm weights: float32
    # rest: fp16

    # Input LayerNorm (float32)
    norm_w = weights["norm.weight"].to(device).float()
    norm_b = weights["norm.bias"].to(device).float()
    x = torch.nn.functional.layer_norm(input_tensor.float(), (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # Allocate output tensors for left,right,out_gate
    B, N, _, _ = x.shape
    left = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)
    right = torch.empty_like(left)
    out_gate = torch.empty_like(left)

    # Determine mask handling
    HAS_MASK = (mask is not None) and (not nomask)
    if HAS_MASK:
        mask_fp = mask.to(device).to(torch.float16)
    else:
        mask_fp = torch.empty((1,), dtype=torch.float16, device=device)  # dummy

    # get strides
    stride_xb, stride_xi, stride_xj, stride_xd = x.stride()
    stride_mask_b, stride_mask_i, stride_mask_j = mask_fp.stride() if HAS_MASK else (0,0,0)

    # Left projection weight etc.
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    # weight strides
    stride_lp_h, stride_lp_d = left_proj_w.stride()
    stride_rg_h, stride_rg_d = right_gate_w.stride()
    stride_lg_h, stride_lg_d = left_gate_w.stride()
    stride_rp_h, stride_rp_d = right_proj_w.stride()
    stride_og_h, stride_og_d = out_gate_w.stride()

    # Output strides
    stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_h = left.stride()
    stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_h = right.stride()
    stride_ogout_b, stride_ogout_i, stride_ogout_j, stride_ogout_h = out_gate.stride()

    # Define block sizes
    BLOCK_H = 32  # hidden dim block
    BLOCK_D = 32  # input dim block

    num_h_blocks = (hidden_dim + BLOCK_H - 1) // BLOCK_H

        x_ptr = x,

        left_proj_ptr = left_proj_w,
        left_gate_ptr = left_gate_w,
        right_proj_ptr = right_proj_w,
        right_gate_ptr = right_gate_w,
        out_gate_ptr = out_gate_w,
        left_out_ptr = left,
        right_out_ptr = right,
        out_gate_out_ptr = out_gate,
        B = B,
        N = N,

"""
Outgoing triangle multiplicative update (TriMul) forward pass.

The kernel fuses:
  * Linear projections (left/right) and their sigmoid gates,
  * Optional binary mask broadcasting,
  * Computation of the out‑gate (sigmoid).

All element‑wise work runs in a single Triton kernel, reducing
memory traffic.  The heavy O(N³·H) contraction is performed with a
batched GEMM (torch.bmm) on the masked/gated tensors.
Mixed‑precision (fp16) is used for the bulk of the computation; the
final result is returned in the original dtype.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _proj_gate_kernel(
    # tensors
    x_ptr, mask_ptr,                     # [B,N,N,D]
    left_proj_ptr, left_gate_ptr,        # [H,D]
    right_proj_ptr, right_gate_ptr,      # [H,D]
    out_gate_ptr,                        # [H,D]
    left_out_ptr, right_out_ptr, out_gate_out_ptr,  # [B,N,N,H]

    # shape / strides
    B, N, D, H,
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lp_h, stride_lp_d,
    stride_lg_h, stride_lg_d,
    stride_rp_h, stride_rp_d,
    stride_rg_h, stride_rg_d,
    stride_og_h, stride_og_d,
    stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_h,
    stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_h,
    stride_ogout_b, stride_ogout_i, stride_ogout_j, stride_ogout_h,

    # compile‑time constants
    HAS_MASK: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Fuse projections, gates and optional mask."""
    pid = tl.program_id(0)                # index over (b,i,j)
    hid = tl.program_id(1)                # block over hidden dimension

    # -----------------------------------------------------------------
    # decode (b,i,j) from pid
    # -----------------------------------------------------------------
    b = pid // (N * N)
    tmp = pid % (N * N)
    i = tmp // N
    j = tmp % N

    # hidden block offsets
    offs_h = hid * BLOCK_H + tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # -----------------------------------------------------------------
    # accumulators (float32 for stability)
    # -----------------------------------------------------------------
    acc_left = tl.zeros((BLOCK_H,), dtype=tl.float32)
    acc_left_gate = tl.zeros((BLOCK_H,), dtype=tl.float32)
    acc_right = tl.zeros((BLOCK_H,), dtype=tl.float32)
    acc_right_gate = tl.zeros((BLOCK_H,), dtype=tl.float32)
    acc_out_gate = tl.zeros((BLOCK_H,), dtype=tl.float32)

    # mask value (scalar per (b,i,j))
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j)
    else:
        mask_val = 1.0

    # -----------------------------------------------------------------
    # loop over the input channel dimension D
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        # load a slice of the input vector x[b,i,j,:]
        x = tl.load(
            x_ptr
            + b * stride_xb
            + i * stride_xi
            + j * stride_xj
            + offs_d * stride_xd,
            mask=mask_d,
            other=0.0,
        )

        # load weight blocks (H x D) for all five linear maps
        w_left = tl.load(
            left_proj_ptr
            + offs_h[:, None] * stride_lp_h
            + offs_d[None, :] * stride_lp_d,
            mask=mask_h[:, None] & mask_d[None, :],
            other=0.0,
        )
        w_left_gate = tl.load(
            left_gate_ptr
            + offs_h[:, None] * stride_lg_h
            + offs_d[None, :] * stride_lg_d,
            mask=mask_h[:, None] & mask_d[None, :],
            other=0.0,
        )
        w_right = tl.load(
            right_proj_ptr
            + offs_h[:, None] * stride_rp_h
            + offs_d[None, :] * stride_rp_d,
            mask=mask_h[:, None] & mask_d[None, :],
            other=0.0,
        )
        w_right_gate = tl.load(
            right_gate_ptr
            + offs_h[:, None] * stride_rg_h
            + offs_d[None, :] * stride_rg_d,
            mask=mask_h[:, None] & mask_d[None, :],
            other=0.0,
        )
        w_out_gate = tl.load(
            out_gate_ptr
            + offs_h[:, None] * stride_og_h
            + offs_d[None, :] * stride_og_d,
            mask=mask_h[:, None] & mask_d[None, :],
            other=0.0,
        )

        # dot products: (H,D) · (D,) -> (H,)
        acc_left += tl.dot(w_left, x).to(tl.float32)
        acc_left_gate += tl.dot(w_left_gate, x).to(tl.float32)
        acc_right += tl.dot(w_right, x).to(tl.float32)
        acc_right_gate += tl.dot(w_right_gate, x).to(tl.float32)
        acc_out_gate += tl.dot(w_out_gate, x).to(tl.float32)

    # -----------------------------------------------------------------
    # apply sigmoid gates and mask
    # -----------------------------------------------------------------
    left_gate_sig = tl.sigmoid(acc_left_gate)
    right_gate_sig = tl.sigmoid(acc_right_gate)
    out_gate_sig = tl.sigmoid(acc_out_gate)

    left_out = acc_left * left_gate_sig * mask_val
    right_out = acc_right * right_gate_sig * mask_val

    # -----------------------------------------------------------------
    # write results (cast back to fp16)
    # -----------------------------------------------------------------
    off = (
        b * stride_lo_b
        + i * stride_lo_i
        + j * stride_lo_j
        + offs_h * stride_lo_h
    )
    tl.store(left_out_ptr + off, left_out.to(tl.float16), mask=mask_h)
    tl.store(right_out_ptr + off, right_out.to(tl.float16), mask=mask_h)
    tl.store(out_gate_out_ptr + off, out_gate_sig.to(tl.float16), mask=mask_h)

    )

    # --------------------------------------------------------------
    # 6. Core contraction: batched GEMM over the sequence dimension
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)          # (B, N, N, dim)

    # --------------------------------------------------------------
    # 8. Cast back to the input dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
