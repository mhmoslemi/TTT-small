@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_stride_a, stride_a_row, stride_a_col,
    batch_stride_b, stride_b_row, stride_b_col,
    batch_stride_c, stride_c_row, stride_c_col,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute offsets
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Initialize accum
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        off_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Load A tile
        a_ptrs = A_ptr + batch_id * batch_stride_a + off_m[:, None] * stride_a_row + off_k[None, :] * stride_a_col
        b_ptrs = B_ptr + batch_id * batch_stride_b + off_k[:, None] * stride_b_row + off_n[None, :] * stride_b_col
        a = tl.load(a_ptrs, mask=off_m < M and off_k < K, other=0.0)
        b = tl.load(b_ptrs, mask=off_k < K and off_n < N, other=0.0)
        acc += tl.dot(a, b)
    # Write back
    c_ptrs = C_ptr + batch_id * batch_stride_c + off_m[:, None] * stride_c_row + off_n[None, :] * stride_c_col
    # cast to half
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=(off_m < M) & (off_n < N))

"""
Highly‑optimized Triton‑accelerated forward pass for the outgoing TriMul
operator (AlphaFold3 style).

Algorithm
---------
1. Layer‑normalize the input tensor (B, N, N, C) and cast to FP16.
2. Perform three linear projections (left, right, gates) in FP16 using
   torch.nn.functional.linear (which maps to cuBLAS GEMM).
3. Fuse element‑wise multiplication of projection, gate and optional mask
   with a tiny Triton kernel.
4. Core reduction:
      out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   is a batched matrix multiplication (left @ rightᵀ) over the hidden
   dimension h.  This is implemented with a custom Triton kernel that
   processes all (B·H) matrices in a single launch.
5. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate and project back to the original channel dimension.
6. Cast the result back to the original dtype (usually float32).

The implementation keeps the heaviest operations (gate·mask fusion and the
batched GEMM) on the GPU via Triton while re‑using PyTorch's highly‑tuned
linear and LayerNorm kernels for the remaining work.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse (proj * gate * mask) where mask is broadcast over H
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,            # *[total_elems]   fp16
    gate_ptr,            # *[total_elems]   fp16
    mask_ptr,            # *[mask_elems]    fp16
    out_ptr,             # *[total_elems]   fp16
    total_elems,         # i64
    hidden_dim,          # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # bound check
    mask = offs < total_elems

    # each mask element is reused hidden_dim times
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """Element‑wise ``proj * gate * mask`` with the mask broadcast over H."""
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],   # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ Bᵀ) for many small
# matrices.  A, B, C are all in row‑major layout.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_stride_a, stride_a_row, stride_a_col,
    batch_stride_b, stride_b_row, stride_b_col,
    batch_stride_c, stride_c_row, stride_c_col,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m    = tl.program_id(1)
    pid_n    = tl.program_id(2)

    # ------------------------------------------------------------------
    # Offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = (A_ptr + batch_id * batch_stride_a
                  + offs_m[:, None] * stride_a_row
                  + offs_k[None, :] * stride_a_col)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile (already transposed): (BLOCK_K, BLOCK_N)
        b_ptrs = (B_ptr + batch_id * batch_stride_b
                  + offs_k[:, None] * stride_b_row
                  + offs_n[None, :] * stride_b_col)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)      # fp16·fp16 → fp32 accumulation

    # ------------------------------------------------------------------
    # Write back
    # ------------------------------------------------------------------
    c_ptrs = (C_ptr + batch_id * batch_stride_c
              + offs_m[:, None] * stride_c_row
              + offs_n[None, :] * stride_c_col)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict containing all linear and LayerNorm parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → float16 later)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the compute‑heavy stages
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all linear weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,hidden_dim) in FP16
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask + gate (Triton kernel) or just gate
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched matmul (left @ rightᵀ)
    #    left, right : (B, N, N, H) → (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # bring hidden dim to 2nd axis and collapse batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # Strides (all tensors are contiguous, row‑major)
    batch_stride_a = left_flat.stride(0)   # N*N
    stride_a_row   = left_flat.stride(1)   # N
    stride_a_col   = left_flat.stride(2)   # 1

    batch_stride_b = right_flat.stride(0)
    # We need B transposed: swap row/col strides
    stride_b_row   = right_flat.stride(2)   # 1   (original col stride)
    stride_b_col   = right_flat.stride(1)   # N   (original row stride)

    batch_stride_c = out_flat.stride(0)
    stride_c_row   = out_flat.stride(1)    # N
    stride_c_col   = out_flat.stride(2)    # 1

    # Block sizes – chosen to suit typical H100/B200 matrix‑core dimensions
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    total_batches = B * H
    grid = (
        total_batches,
        math.ceil(N / BLOCK_M),
        math.ceil(N / BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_stride_a, stride_a_row, stride_a_col,
        batch_stride_b, stride_b_row, stride_b_col,
        batch_stride_c, stride_c_row, stride_c_col,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    out = out.to(dtype)
    return out
