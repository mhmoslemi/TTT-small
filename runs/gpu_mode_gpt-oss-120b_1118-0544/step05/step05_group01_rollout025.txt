@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute base pointers for this batch
    a_base = a_ptr + pid_batch * stride_a_batch
    b_base = b_ptr + pid_batch * stride_b_batch
    c_base = c_ptr + pid_batch * stride_c_batch

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for boundary
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + offs_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Compute dot product
        acc += tl.dot(a, b)

    # Write out
    c_ptrs = c_base + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (no bias):
      left  = x @ W_left      , right = x @ W_right
      left_gate  = σ(x @ W_left_gate)
      right_gate = σ(x @ W_right_gate)
      out_gate   = σ(x @ W_out_gate)
4. Fuse mask, projection and gate (if a mask is present) in a Triton
   kernel that computes  proj * gate * mask  (mask broadcast over the hidden dim).
5. Core reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM in Triton (one GEMM per head).
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.
7. Cast the result back to the original dtype (usually float32).

The implementation mixes PyTorch primitives (layer‑norm, linear) with two
Triton kernels:
   * `_apply_gate_mask_kernel` – fused mask & gate.
   * `batched_gemm_kernel`      – batched matrix multiplication for the
                                 O(N³) reduction.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Kernel 1: fuse projection * gate * mask (mask is broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B,N,N,H)
    gate_ptr,          # fp16 * (B,N,N,H)
    mask_ptr,          # fp16 * (B,N,N)
    out_ptr,           # fp16 * (B,N,N,H)
    total_elems,       # B*N*N*H
    hidden_dim,        # H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # every contiguous `hidden_dim` entries share the same mask element
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(out_ptr + offs, proj * gate * m, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    proj, gate : (B,N,N,H) fp16 tensors (contiguous)
    mask       : (B,N,N) fp16 tensor (contiguous, 0/1)
    Returns proj * gate * mask (mask broadcast over H) as a new fp16 tensor.
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the O(N³) reduction
# Computes C = A @ B   where
#   A shape = (batch, M, K)   (M=N, K=N)
#   B shape = (batch, K, N)   (K=N, N=N)  (already transposed)
#   C shape = (batch, M, N)   (M=N, N=N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # batch index (B * H)
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block col

    # Base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile (K x N)
        b_ptrs = b_batch + offs_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back C tile (M x N) in fp16
    c_ptrs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C]
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all linear weights to fp16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (result: [B, N, N, H])
    # --------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid (still fp16)
    # --------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Apply mask & gating (fused via Triton if mask present)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM using Triton.
    # --------------------------------------------------------------
    B, N, _, H = left.shape

    # Bring batch and head together and make matrices contiguous
    # left: (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # We need right transposed on the last two axes (k,j) for GEMM
    right_mat_t = right_mat.transpose(1, 2).contiguous()

    # Allocate output
    out_mat = torch.empty_like(left_mat)  # fp16

    # Strides
    stride_a_batch = left_mat.stride(0)
    stride_a_row   = left_mat.stride(1)
    stride_a_col   = left_mat.stride(2)

    stride_b_batch = right_mat_t.stride(0)
    stride_b_row   = right_mat_t.stride(1)
    stride_b_col   = right_mat_t.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_row   = out_mat.stride(1)
    stride_c_col   = out_mat.stride(2)

    # Block sizes – these values work well on A100 / B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H,                    # batch * head dimension
            triton.cdiv(N, BLOCK_M),  # row blocks
            triton.cdiv(N, BLOCK_N))  # col blocks

    batched_gemm_kernel[grid](
        left_mat,
        right_mat_t,
        out_mat,
        N, N, N,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype (usually float32)
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
