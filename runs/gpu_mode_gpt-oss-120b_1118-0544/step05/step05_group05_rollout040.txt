def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # batch_hidden
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch and hidden
    bh = pid
    b = bh // H
    h = bh % H

    # calculate offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)

    for kk in range(num_k_blocks):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for A and B
        a_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + offs_i[:, None]*stride_left_i + offs_k[None, :]*stride_left_k
        b_ptrs = right_ptr + b*stride_right_b + h*stride_right_h + offs_j[:, None]*stride_right_i + offs_k[None, :]*stride_right_k

        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # accumulate
        acc += tl.dot(a, b)   # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)

    # store result
    c_ptrs = out_ptr + b*stride_out_b + h*stride_out_h + offs_i[:, None]*stride_out_i + offs_j[None, :]*stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
Highly optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Input LayerNorm (float32 for stability).
2. Linear projections (left/right) and three sigmoid gates (left, right, out) in fp16.
3. Optional mask application and element‑wise gate multiplication (broadcasted).
4. Core O(N³·H) contraction computed by a custom Triton kernel that performs
   a batched matrix multiplication over the sequence dimension for all hidden
   channels simultaneously, avoiding expensive permute/view copies.
5. Output LayerNorm, multiplication by the out‑gate and final linear projection
   back to the original feature dimension.
All heavy arithmetic is done in fp16 with fp32 accumulation inside the kernel.
The final tensor is returned in the original dtype of the input.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,                     # [B,N,N,H] tensors
    B, N, H,                                          # runtime scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    using a tiled GEMM style algorithm.  Each program instance processes a
    (i,j) tile for a specific (b,h) pair.
    """
    pid = tl.program_id(0)   # combines batch and hidden dimension
    pid_i = tl.program_id(1) # tile row (i)
    pid_j = tl.program_id(2) # tile column (j)

    # decode batch index and hidden index from pid
    b = pid // H
    h = pid % H

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 for numerical quality
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_blocks):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load tiles of left and right
        a_ptrs = (left_ptr + b * stride_left_b + h * stride_left_h
                  + offs_i[:, None] * stride_left_i
                  + offs_k[None, :] * stride_left_k)
        b_ptrs = (right_ptr + b * stride_right_b + h * stride_right_h
                  + offs_j[:, None] * stride_right_i
                  + offs_k[None, :] * stride_right_k)

        a = tl.load(a_ptrs,
                    mask=(mask_i[:, None] & mask_k[None, :]),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(mask_j[:, None] & mask_k[None, :]),
                    other=0.0)

        # Accumulate the block product
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Store result back to the output tensor
    c_ptrs = (out_ptr + b * stride_out_b + h * stride_out_h
              + offs_i[:, None] * stride_out_i
              + offs_j[None, :] * stride_out_j)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=(mask_i[:, None] & mask_j[None, :]))


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (see code)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for compute‑heavy parts
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections and gates (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    left  = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)         # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 3. Apply mask and gate multiplication (broadcasted)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask = mask.to(device).to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * left_gate * mask
        right = right * right_gate * mask
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 4. Core contraction – custom Triton kernel
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # (B, N, N, H)

    # Strides (in elements)
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 5. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the original dtype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
