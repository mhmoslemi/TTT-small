@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_a_row, stride_a_col,
    stride_b_row, stride_b_col,
    stride_c_row, stride_c_col,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Determine batch and tile indices.
    pid = tl.program_id(0)   # total number of blocks = batch * ceil(M/BM) * ceil(N/BN)
    batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_index = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = block_index // tl.cdiv(N, BLOCK_N)
    pid_n = block_index % tl.cdiv(N, BLOCK_N)

    # Use same logic as typical matmul with batch offset.
    # Compute offsets:
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute pointers
    a_ptrs = A_ptr + batch * batch_stride_a + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
    b_ptrs = B_ptr + batch * batch_stride_b + offs_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col
    c_ptrs = C_ptr + batch * batch_stride_c + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)
        # cast to fp16 for matmul then to fp32?
        acc += tl.dot(a, b)
        # Increment pointers
        a_ptrs += BLOCK_K * stride_a_col
        b_ptrs += BLOCK_K * stride_b_row

    # Store output
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated outgoing TriMul forward.
    Performs layer‑norm, linear projections, gating, optional mask,
    a batched matrix multiplication in Triton, output layer‑norm,
    gating and final linear projection.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)
    # Cast to fp16 for compute heavy parts
    x = x.to(torch.float16)

    # Convert weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections (no bias)
    left_proj = torch.nn.functional.linear(x, left_proj_w)          # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)       # (B,N,N,H)

    # Gate projections
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()   # (B,N,N,H)
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid() # (B,N,N,H)
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()     # (B,N,N,H)

    # Apply mask and gate
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Core reduction: batched GEMM across hidden dimension
    # Rearrange to (B*H, N, N) and (B*H, N, N) (right transposed)
    B, N, _, H = left.shape
    # left: (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: (B, N, N, H) -> (B, H, N, N) with K,J ordering
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    # Flatten batch and head dimension
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)
    # Batched matmul using Triton
    out_flat = _batched_matmul_triton(left_flat, right_flat)
    # Restore shape (B, H, N, N) -> (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layer norm
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # Multiply by out gate
    out = out * out_gate

    # Final linear projection back to original dim
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to original dtype if needed
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
```

```python
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    M, N, K,
    batch,
    num_blocks_m, num_blocks_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    tiles_per_batch = num_blocks_m * num_blocks_n
    batch_idx = pid // tiles_per_batch
    tile_idx = pid % tiles_per_batch
    pid_m = tile_idx // num_blocks_n
    pid_n = tile_idx % num_blocks_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # compute pointers for this batch
    a_ptr = A_ptr + batch_idx * stride_a_batch + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
    b_ptr = B_ptr + batch_idx * stride_b_batch + offs_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col
    c_ptr = C_ptr + batch_idx * stride_c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        a_ptr += BLOCK_K * stride_a_col
        b_ptr += BLOCK_K * stride_b_row

    # write back
    c = acc.to(tl.float16)
    tl.store(c_ptr, c, mask=mask_m[:, None] & mask_n[None, :])
```

```python
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems
    mask_idx = offs // hidden_dim
    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)
    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)
```

"""
custom_kernel: Triton‑accelerated outgoing TriMul forward.

The implementation follows the reference PyTorch TriMul module:
1. Layer‑norm over the channel dimension.
2. FP16 cast for compute‑intensive parts.
3. Linear projections (left/right) and gating projections (left_gate,
   right_gate, out_gate) using the provided weights (no bias).
4. Optional mask‑gate fusion in a small Triton kernel.
5. Core reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM across the hidden dimension with a
   custom Triton kernel (one GEMM per head per batch).
6. Output layer‑norm over the hidden dimension, multiplication by out_gate,
   and final linear projection back to the original channel dimension.
7. Cast the result back to the original dtype (typically float32).

All heavy‑weight tensor operations are performed in FP16 to exploit Tensor
Cores, while accumulation is in FP32 for numerical stability.
"""

import math
from typing import Tuple, Dict

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel: proj * gate * mask  (mask broadcast over the hidden dimension)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B,N,N,H)
    gate_ptr,          # fp16 * (B,N,N,H)
    mask_ptr,          # fp16 * (B,N,N)
    out_ptr,           # fp16 * (B,N,N,H)
    total_elems,       # int: B*N*N*H
    hidden_dim,        # int: H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # each hidden_dim consecutive entries share the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """Fuse projection, gate and mask using Triton."""
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = ((total + block_size - 1) // block_size,)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel: batched matrix multiplication (batch * H independent GEMMs)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # fp16 pointers
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    M, N, K,                                 # runtime dims (ints)
    batch,                                   # total number of batches (B*H)
    num_blocks_m, num_blocks_n,              # tiles per batch (ints)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    tiles_per_batch = num_blocks_m * num_blocks_n
    batch_idx = pid // tiles_per_batch
    tile_idx = pid % tiles_per_batch
    pid_m = tile_idx // num_blocks_n
    pid_n = tile_idx % num_blocks_n

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers for this batch
    a_ptr = (
        A_ptr
        + batch_idx * stride_a_batch
        + offs_m[:, None] * stride_a_row
        + offs_k[None, :] * stride_a_col
    )
    b_ptr = (
        B_ptr
        + batch_idx * stride_b_batch
        + offs_k[:, None] * stride_b_row
        + offs_n[None, :] * stride_b_col
    )
    c_ptr = (
        C_ptr
        + batch_idx * stride_c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for _ in range(0, K, BLOCK_K):
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)                     # FP16·FP16 → FP32 accumulation
        a_ptr += BLOCK_K * stride_a_col
        b_ptr += BLOCK_K * stride_b_row

    c = acc.to(tl.float16)
    tl.store(c_ptr, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using Triton.
    A: (batch, M, K)   fp16, contiguous
    B: (batch, K, N)   fp16, contiguous
    Returns C: (batch, M, N) fp16, contiguous
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Strides in element counts
    stride_a_batch = A.stride(0)
    stride_a_row   = A.stride(1)
    stride_a_col   = A.stride(2)
    stride_b_batch = B.stride(0)
    stride_b_row   = B.stride(1)
    stride_b_col   = B.stride(2)
    stride_c_batch = C.stride(0)
    stride_c_row   = C.stride(1)
    stride_c_col   = C.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (batch * num_blocks_m * num_blocks_n,)

    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        M,
        N,
        K,
        batch,
        num_blocks_m,
        num_blocks_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] (binary 0/1) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B, N, N, H)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask and gating (fused via Triton if mask present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM across hidden dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # Rearrange to (B, H, N, N) where the reduction dimension is the middle one
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N) -> (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()    # (B, H, N, N) -> (k,j)

    # Collapse batch & head dimensions
    left_flat = left_perm.view(B * H, N, N)    # (B*H, M, K)
    right_flat = right_perm.view(B * H, N, N)  # (B*H, K, N)

    # Batched matrix multiplication via Triton
    out_flat = _batched_matmul_triton(left_flat, right_flat)  # (B*H, N, N)

    # Restore (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
