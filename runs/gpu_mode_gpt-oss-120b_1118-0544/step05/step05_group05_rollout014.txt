"""
Optimized outgoing TriMul forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32 for stability).
2. Cast to fp16 and apply two linear projections (left/right) and their
   sigmoid gates.
3. Fuse mask, gates and projection multiplication in a custom Triton kernel
   (in‑place on `left` and `right`).
4. Perform the core O(N³·H) contraction as a batched GEMM:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using `torch.bmm` on the reshaped tensors (fast cuBLAS implementation).
5. Apply the output LayerNorm, an output sigmoid gate and a final linear
   projection back to the original feature dimension.
All heavy arithmetic is carried out in fp16, the final result is cast back to
the input dtype (float32 by default).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # [B,N,N,H] flattened
    right_ptr,              # same as left
    left_gate_ptr,          # same shape
    right_gate_ptr,         # same shape
    mask_ptr,               # [B,N,N] flattened
    total_elems: tl.int64,  # number of elements in left/right tensors
    hidden_dim: tl.int64,   # H
    HAS_MASK: tl.constexpr, # compile‑time bool
    BLOCK_SIZE: tl.constexpr,
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    (mask is broadcast over the hidden dimension).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load values (masked)
    left_val      = tl.load(left_ptr      + offs, mask=active, other=0.0)
    right_val     = tl.load(right_ptr     + offs, mask=active, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate_val= tl.load(right_gate_ptr+ offs, mask=active, other=0.0)

    if HAS_MASK:
        # Each hidden entry shares the same mask entry.
        mask_idx = offs // hidden_dim            # integer division
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
        left_val  = left_val  * left_gate_val  * mask_val
        right_val = right_val * right_gate_val * mask_val
    else:
        left_val  = left_val  * left_gate_val
        right_val = right_val * right_gate_val

    # Store back in‑place
    tl.store(left_ptr  + offs, left_val,  mask=active)
    tl.store(right_ptr + offs, right_val, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for compute‑heavy parts
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gates (fp16)
    # -----------------------------------------------------------------
    left      = F.linear(x, left_proj_w)          # (B, N, N, H)
    right     = F.linear(x, right_proj_w)         # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,H)

    # -----------------------------------------------------------------
    # 4. Fuse mask and gate multiplication with Triton (in‑place)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # Cast mask to fp16 and flatten
        mask_fp = mask.to(device).to(torch.float16)
        mask_flat = mask_fp.contiguous().view(-1)   # (B*N*N,)

        # Flatten tensors for kernel launch
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)

        total = left_flat.numel()                 # B*N*N*H
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left  = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask: simple elementwise multiplication with gates
        left  = left  * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction via batched GEMM (left @ rightᵀ)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiply: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to original dtype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
