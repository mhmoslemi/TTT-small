"""
TriMul (outgoing) forward pass – fused linear + mask/gate Triton kernel.
Algorithm
---------
1. Layer‑norm `x` over the channel dimension `dim` (float32).
2. Cast to FP16 for compute‑heavy ops.
3. Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
   into a single matrix multiplication:
   y = x_flat @ W_concat      where W_concat stacks the 5 weight matrices.
4. Split `y` into the five tensors, apply sigmoid to the three gate tensors.
5. Fuse mask (if present) with the projections and gates via a Triton kernel:
        left  = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask
   (If no mask, simple element‑wise multiplication is used.)
6. Core reduction – batched GEMM over the sequence dimension:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented with a single `torch.bmm` on reshaped tensors.
7. Output LayerNorm over the hidden dimension, multiply by `out_gate`,
   and final linear projection back to `dim` (all in FP16).
8. Cast the result back to the input dtype (usually FP32) and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (total_elems,)
    gate_ptr,          # fp16 * (total_elems,)
    mask_ptr,          # fp16 * (mask_elems,)
    out_ptr,           # fp16 * (total_elems,)
    total_elems,       # int: B*N*N*hidden
    hidden_dim,        # int: hidden
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # guard
    mask = offs < total_elems

    # mask index (broadcast over hidden)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    proj, gate: (B, N, N, H) fp16 contiguous
    mask: (B, N, N) fp16 contiguous
    Returns proj * gate * mask (broadcasted over H) as fp16 tensor.
    """
    B, N, _, H = proj.shape
    total = B * N * N * H
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (or None)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim" (int) – input channel dimension C
                  * "hidden_dim" (int) – hidden dimension H
                  * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim` (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑intensive part
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Fuse the 5 linear projections into a single matmul
    # --------------------------------------------------------------
    # Flatten to (M, dim) where M = B * N * N
    B, N, _, _ = x.shape
    M = B * N * N
    x_flat = x.view(M, dim)                     # (M, dim)

    # Stack weight matrices along the output-feature dimension:
    # each weight has shape (hidden, dim)
    w_left      = weights["left_proj.weight"].to(torch.float16)
    w_right     = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate= weights["right_gate.weight"].to(torch.float16)
    w_out_gate  = weights["out_gate.weight"].to(torch.float16)

    weight_concat = torch.cat(
        [w_left, w_right, w_left_gate, w_right_gate, w_out_gate],
        dim=0,
    )  # (5*hidden, dim)

    # Single linear projection (M, 5*hidden)
    proj_all = F.linear(x_flat, weight_concat)   # no bias

    # --------------------------------------------------------------
    # 4. Split into individual components and apply sigmoid gates
    # --------------------------------------------------------------
    H = hidden
    left_proj   = proj_all[:, 0:H]                     # (M, H)
    right_proj  = proj_all[:, H:2*H]                  # (M, H)
    left_gate_r = proj_all[:, 2*H:3*H]                # (M, H) raw
    right_gate_r= proj_all[:, 3*H:4*H]                # (M, H) raw
    out_gate_r  = proj_all[:, 4*H:5*H]                # (M, H) raw

    left_gate   = left_gate_r.sigmoid()
    right_gate  = right_gate_r.sigmoid()
    out_gate    = out_gate_r.sigmoid()

    # Reshape back to (B, N, N, H)
    left_proj   = left_proj.view(B, N, N, H)
    right_proj  = right_proj.view(B, N, N, H)
    left_gate   = left_gate.view(B, N, N, H)
    right_gate  = right_gate.view(B, N, N, H)
    out_gate    = out_gate.view(B, N, N, H)

    # --------------------------------------------------------------
    # 5. Apply mask and gating (fused via Triton if mask present)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Apply out_gate
    out = out * out_gate

    # Final linear projection back to `dim`
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)
    # flatten, apply linear, reshape
    out_flat2 = out.view(B * N * N, H)                     # (M, H)
    out_proj = F.linear(out_flat2, to_out_w)               # (M, dim)
    out = out_proj.view(B, N, N, dim)

    # --------------------------------------------------------------
    # 8. Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
