"""
TriMul (outgoing) forward pass – Triton‑accelerated gating & mask.

Algorithm
---------
1. Layer‑normalize the input tensor (B, N, N, C) with provided LN parameters.
2. Cast to FP16 for the compute‑heavy stages.
3. Linear projections (left/right) and gating projections (left/right/out)
   are performed with FP16 weights.
4. If a mask is supplied, a small Triton kernel fuses the element‑wise
   multiplication:  proj * gate * mask  (mask is broadcast over the hidden dim).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (left @ rightᵀ) for each
   hidden channel h.  We reshape to (B·H, N, N) and use torch.bmm,
   which is highly optimized on NVIDIA GPUs.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate, and project back to the original channel dimension.
7. Cast the result back to the input dtype (usually float32) and return.

The Triton kernel is only used for the gating‑mask fusion; the rest
leverages PyTorch's fast cuBLAS kernels.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # *[total_elems]   fp16
    gate_ptr,          # *[total_elems]   fp16
    mask_ptr,          # *[mask_elems]    fp16
    out_ptr,           # *[total_elems]   fp16
    total_elems,       # i64
    hidden_dim,        # i64  (size of the last dimension H)
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse (proj * gate * mask) where mask is broadcast over hidden_dim."""
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # boundary check
    active = offs < total_elems

    # map flat index to mask index (broadcast over H)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=active, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=active, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=active)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Element‑wise ``proj * gate * mask`` where ``mask`` has shape (B, N, N)
    and is broadcast over the hidden dimension H.
    All tensors must be FP16 and contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    # launch one‑dimensional grid
    grid = (math.ceil(total / block_size),)

    # flatten mask for fast indexing
    mask_flat = mask.contiguous().view(-1)

    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],   # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out

    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (all float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all linear weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)   # (B, N, N, H)

    # 5. Gate projections (sigmoid)
    # ------------------------------------------------------------------

    right_gate = F.linear(x, right_gate_w).sigmoid()

    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction: batch GEMM (left @ rightᵀ)
    #    Reshape to (B*H, N, N) and use torch.bmm (cuBLAS).
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # left shape (B, N, N, H)

    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)        # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype (usually float32) and return
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
