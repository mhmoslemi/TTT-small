    a_ptr, b_ptr, c_ptr,

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets for m and n dimensions inside the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)

    mask_n = offs_n < N

"""
Triton‑accelerated implementation of the outgoing TriMul operator.

Algorithm
---------
1. Layer‑norm the 4‑D input (B, N, N, C).
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections and gate predictions (left/right, left/right gate,
   out gate) are performed with torch.nn.functional.linear.
4. Apply the optional mask and element‑wise gates (simple PyTorch
   broadcasting; the heavy work is in the GEMM).
5. Core reduction:
       out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   This is a batched matrix multiplication (left @ rightᵀ) over the
   (B·H) batches.  It is implemented in Triton as a tiled GEMM with a
   float32 accumulator and fp16 inputs/outputs.
6. Second LayerNorm, out‑gate multiplication and final linear projection
   back to the original channel dimension.
7. Cast the result back to the original dtype.

Only the N³ reduction is written in Triton; all other ops use PyTorch
for simplicity.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batch_gemm_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    total_batches, M, N, K,            # dimensions (int)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_m, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ Bᵀ for many independent (M×K)·(N×K) products."""
    pid_batch = tl.program_id(0)   # batch index
    pid_m = tl.program_id(1)       # row tile index
    pid_n = tl.program_id(2)       # col tile index

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K dimension
    for k in range(0, K, BLOCK_K):
        k_offsets = tl.arange(0, BLOCK_K)
        k_valid = (k + k_offsets) < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + (k + k_offsets)[None, :] * stride_a_k,
            mask=mask_m[:, None] & k_valid[None, :],
            other=0.0,
        )
        # B tile (remember we need B[j, k], so we load (BLOCK_N, BLOCK_K))
        b = tl.load(
            b_ptr
            + pid_batch * stride_b_batch
            + offs_n[:, None] * stride_b_m
            + (k + k_offsets)[None, :] * stride_b_k,
            mask=mask_n[:, None] & k_valid[None, :],
            other=0.0,
        )
        # dot product with B transposed (B is (N, K), we need (K, N))
        acc += tl.dot(a, b, trans_b=True)

    # store result as fp16
    c = acc.to(tl.float16)
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, c, mask=mask)

            - weights: dict of model parameters (torch.Tensors)
            - config: dict with keys "dim", "hidden_dim", "nomask" (optional)

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive stages
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6. Apply mask (if any) and element‑wise gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core N³ reduction: batched GEMM (left @ rightᵀ) via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # batch size, sequence length
    # reshape to (B*H, N, N) for a simple batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # strides (all tensors are contiguous)
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_m, stride_b_k = right_flat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    total_batches = B * hidden_dim
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        total_batches,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batch_gemm_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        total_batches,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_m,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
