"""
custom_kernel
--------------

Fast forward pass for the outgoing TriMul operator (AlphaFold3 style).

Algorithm
---------
1. Layer‑normalize the 4‑D input (B, N, N, C) using the supplied LN parameters.
2. Cast to FP16 for the compute‑intensive stages.
3. Fuse each pair of projection + gate linear layers:
   * left  = Linear(x, [left_proj; left_gate]) → split → proj, gate → sigmoid(gate)
   * right = Linear(x, [right_proj; right_gate]) → split → proj, gate → sigmoid(gate)
   This reduces the number of matrix‑multiplications by half.
4. If a mask is present (and ``nomask`` is False) apply a Triton kernel that
   fuses ``proj * gate * mask`` for both left and right tensors.
   Otherwise perform the element‑wise multiplication in PyTorch.
5. Core reduction: for every hidden channel h compute
      out[..., h] = left[..., h] @ right[..., h].T
   Implemented as a single batched GEMM using ``torch.bmm`` on a view of shape
   (B·H, N, N) → (B·H, N, N).  This is the most expensive O(N³·H) step but
   leverages cuBLAS which is highly optimized for the target B200 GPU.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   ``out_gate`` (computed with a separate linear + sigmoid) and project back
   to the original channel dimension.
7. Cast the final result back to the input dtype (usually float32).

At least one operation (proj × gate × mask) is realised in a Triton kernel
as required.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused projection * gate * mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    proj_ptr,        # *[total_elems]  fp16
    gate_ptr,        # *[total_elems]  fp16
    mask_ptr,        # *[mask_elems]   fp16
    out_ptr,         # *[total_elems]  fp16
    total_elems,     # i64
    hidden_dim,      # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # mask is broadcasted over the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_proj_gate_mask(proj: torch.Tensor, gate: torch.Tensor,
                          mask: torch.Tensor,
                          block_size: int = 1024) -> torch.Tensor:
    """
    Apply ``proj * gate * mask`` where ``mask`` has shape (B, N, N) and is
    broadcast over the hidden dimension H.  All tensors must be FP16 and
    contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)
    _proj_gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],   # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict with all linear and LayerNorm parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32 -> FP16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Prepare combined projection+gate weights (FP16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]

    # Concatenate along out_features dimension: (2*H, dim)
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear pass producing [proj, gate] for left and right
    # ------------------------------------------------------------------
    left_combined = F.linear(x, left_combined_w)   # (B, N, N, 2*H)
    right_combined = F.linear(x, right_combined_w) # (B, N, N, 2*H)

    left_proj = left_combined[..., :hidden_dim]
    left_gate = left_combined[..., hidden_dim:].sigmoid()

    right_proj = right_combined[..., :hidden_dim]
    right_gate = right_combined[..., hidden_dim:].sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask + gate (using Triton if needed)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16 contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_proj_gate_mask(left_proj, left_gate, mask_fp16)
        right = _apply_proj_gate_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. out_gate (separate linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = F.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Core reduction: batched GEMM over k dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matmul: left @ right^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8. Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
