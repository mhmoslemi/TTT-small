@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,
    right_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,
    total_elems,
    hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    if HAS_MASK:
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', False)

    # Get weight tensors
    norm_w = weights['norm.weight'].to(device)
    norm_b = weights['norm.bias'].to(device)
    left_proj_w = weights['left_proj.weight'].to(device)
    right_proj_w = weights['right_proj.weight'].to(device)
    left_gate_w = weights['left_gate.weight'].to(device)
    right_gate_w = weights['right_gate.weight'].to(device)
    out_gate_w = weights['out_gate.weight'].to(device)
    to_out_norm_w = weights['to_out_norm.weight'].to(device)
    to_out_norm_b = weights['to_out_norm.bias'].to(device)
    to_out_w = weights['to_out.weight'].to(device)

    # LN input in float32 for stability
    x_fp32 = torch.nn.functional.layer_norm(input_tensor.to(torch.float32), (dim,), weight=norm_w.to(torch.float32), bias=norm_b.to(torch.float32))
    x = x_fp32.to(torch.float16)

    # Concatenate left proj and left gate weights
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)

    # Linear transforms
    left_concat = torch.nn.functional.linear(x, left_concat_w)  # shape B,N,N,2*hidden_dim
    right_concat = torch.nn.functional.linear(x, right_concat_w)

    # Split
    left_proj = left_concat[..., :hidden_dim]  # B,N,N,hidden_dim
    left_gate_pre = left_concat[..., hidden_dim:]  # B,N,N,hidden_dim
    right_proj = right_concat[..., :hidden_dim]
    right_gate_pre = right_concat[..., hidden_dim:]

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w.to(torch.float16)))

    # Apply mask and gates using Triton kernel (or simple multiplication if no mask)
    if mask is not None and not nomask:
        # Convert mask to fp16 and flatten without extra dim
        mask_fp = mask.to(device).to(torch.float16)
        # flatten
        left_flat = left_proj.contiguous().view(-1)
        right_flat = right_proj.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)  # shape B*N*N
        total = left_flat.numel()
        BLOCK_SIZE = 8192  # can be tuned

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _mask_gate_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # reshape back
        left = left_flat.view_as(left_proj)
        right = right_flat.view_as(right_proj)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # contraction
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matmul: left @ right.T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # output layernorm
    out = torch.nn.functional.layer_norm(out, (H,), weight=to_out_norm_w.to(torch.float16), bias=to_out_norm_b.to(torch.float16))
    # apply out gate
    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(out, to_out_w.to(torch.float16))

    # Cast back to original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

"""
Optimized outgoing TriMul forward pass.

Algorithm
---------
1. LayerNorm on the input (float32 for stability).
2. Fuse the left/right projections with their respective gates by a single
   matmul each (concatenated weight matrices), followed by a sigmoid.
3. Fuse mask, left‑gate and right‑gate multiplications into one Triton kernel
   to cut memory traffic.
4. Compute the core O(N³·H) contraction via a batched GEMM
   (left @ rightᵀ) using torch.bmm (highly tuned cuBLAS).
5. Apply output LayerNorm, the output gate and a final linear projection.
All heavy arithmetic is performed in fp16; the final result is cast back to
the original dtype (usually float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape as left/right
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    """In‑place fuse:
         left  = left  * left_gate  * mask
         right = right * right_gate * mask
    Mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left      = tl.load(left_ptr      + offs, mask=active, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden_dim entries share the same mask element
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as the input.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numeric stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor.to(torch.float32), (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)  # switch to fp16 for the compute‑heavy path

    # -----------------------------------------------------------------
    # 2. Fuse projection + gate linear layers
    # -----------------------------------------------------------------
    # concatenate projection and gate weight matrices (biases are None)
    left_concat_w = torch.cat([weights["left_proj.weight"],
                               weights["left_gate.weight"]], dim=0).to(device).to(torch.float16)
    right_concat_w = torch.cat([weights["right_proj.weight"],
                                weights["right_gate.weight"]], dim=0).to(device).to(torch.float16)

    left_concat = F.linear(x, left_concat_w)   # (B, N, N, 2*H)
    right_concat = F.linear(x, right_concat_w) # (B, N, N, 2*H)

    # split into projection and gate parts
    left_proj = left_concat[..., :hidden_dim]            # (B, N, N, H)
    left_gate_pre = left_concat[..., hidden_dim:]        # (B, N, N, H)
    right_proj = right_concat[..., :hidden_dim]
    right_gate_pre = right_concat[..., hidden_dim:]

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device).to(torch.float16)))  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 3. Fuse mask and gate multiplication (Triton kernel)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # mask broadcasted over hidden dimension, fp16
        mask_fp = mask.to(device).to(torch.float16)

        # Flatten tensors for the kernel
        left_flat = left_proj.contiguous().view(-1)
        right_flat = right_proj.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 8192  # can be tuned per GPU
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _mask_gate_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left = left_flat.view_as(left_proj)
        right = right_flat.view_as(right_proj)
    else:
        # No mask → simple elementwise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 4. Core contraction: batched GEMM (left @ rightᵀ) over the sequence dim
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiply: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
