"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The kernel fuses the element‑wise mask and gate multiplications into a single
Triton kernel.  The heavy O(N³·H) reduction is performed with a single
torch.einsum (which internally uses an optimized CUDA kernel).  All linear
layers and LayerNorms are carried out in fp16 for speed, and the final
output is cast back to the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate into the projected tensors.
# ---------------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # const * left   (B,N,N,H) flattened
    right_ptr,         # const * right
    mask_ptr,          # const * mask   (B,N,N) flattened
    left_gate_ptr,     # const * left_gate
    right_gate_ptr,    # const * right_gate
    left_out_ptr,      # * left_out
    right_out_ptr,     # * right_out
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Index of the corresponding (b,i,j) for the mask (broadcast over H)
    idx_mask = offs // H
    m = tl.load(mask_ptr + idx_mask, mask=mask)

    # Apply mask and gates
    tl.store(left_out_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask        : torch.Tensor [B, N, N] (binary) or None
            - weights     : dict of model parameters
            - config      : dict with keys
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask"       – if True, ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # -----------------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------------
    # LayerNorm is performed in float32 for numerical stability,
    # then cast to fp16 for the compute‑heavy path.
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # -----------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # -----------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # -----------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # -----------------------------------------------------------------------
    # 4) Compute gates (sigmoid of linear projections)
    # -----------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,H)

    # -----------------------------------------------------------------------
    # 5) Prepare mask (broadcast over hidden dimension)
    # -----------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_fp = mask.to(torch.float16)
    else:
        mask_fp = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # -----------------------------------------------------------------------
    # 6) Fuse mask and gate multiplications with Triton
    # -----------------------------------------------------------------------
    total_elems = left.numel()                     # B * N * N * H
    BLOCK = 1024                                   # typical good block size
    grid = (triton.cdiv(total_elems, BLOCK),)

    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    _mask_gate_fuse_kernel[grid](
        left,                      # left_ptr
        right,                     # right_ptr
        mask_fp.view(-1),          # mask_ptr (flattened)
        left_gate,                 # left_gate_ptr
        right_gate,                # right_gate_ptr
        left_out,                  # left_out_ptr
        right_out,                 # right_out_ptr
        hidden_dim,
        total_elems,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # -----------------------------------------------------------------------
    # 7) Core reduction: sum over k dimension
    #    Using torch.einsum which is backed by an optimized CUDA kernel.
    # -----------------------------------------------------------------------
    # left_out: (B,N,N,H)   right_out: (B,N,N,H)
    # out[b,i,j,h] = Σ_k left_out[b,i,k,h] * right_out[b,j,k,h]
    out = torch.einsum("b i k h, b j k h -> b i j h", left_out, right_out)

    # -----------------------------------------------------------------------
    # 8) Output LayerNorm over hidden dimension H
    # -----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------------
    # 9) Apply output gate
    # -----------------------------------------------------------------------
    out = out * out_gate

    # -----------------------------------------------------------------------
    # 10) Final linear projection back to original channel dimension C
    # -----------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # -----------------------------------------------------------------------
    # 11) Cast back to original dtype
    # -----------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
