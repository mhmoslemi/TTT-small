@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)
    # compute batch and head indices
    b = pid_bh // H
    h = pid_bh % H

    # compute start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # create range indices
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    # mask for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointers
        left_ptrs = left_ptr + \
            b * stride_lb + h * stride_lh + \
            offs_i[:, None] * stride_li + \
            offs_k[None, :] * stride_lk

        right_ptrs = right_ptr + \
            b * stride_rb + h * stride_rh + \
            offs_k[:, None] * stride_rk + \
            offs_j[None, :] * stride_rj

        # load with mask
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product
        acc = acc + tl.dot(left, right)  # both fp16 -> fp32?

    # Store result
    out_ptrs = out_ptr + \
        b * stride_ob + h * stride_oh + \
        offs_i[:, None] * stride_oi + \
        offs_j[None, :] * stride_oj

    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)  # block row
    pid_j = tl.program_id(1)  # block col
    pid_bh = tl.program_id(2) # batch*head

    b = pid_bh // H
    h = pid_bh % H

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers for left (B, i, k, H)
        left_ptrs = left_ptr + \
            b * stride_lb + h * stride_lh + \
            offs_i[:, None] * stride_li + \
            offs_k[None, :] * stride_lk

        # Pointers for right (B, j, k, H) -> we need (k, j) order for dot
        right_ptrs = right_ptr + \
            b * stride_rb + h * stride_rh + \
            offs_k[:, None] * stride_rk + \
            offs_j[None, :] * stride_rj

        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        acc += tl.dot(left, right)

    # Store the result
    out_ptrs = out_ptr + \
        b * stride_ob + h * stride_oh + \
        offs_i[:, None] * stride_oi + \
        offs_j[None, :] * stride_oj

    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

def tri_mul_forward(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Batched triangular multiplicative update using a Triton kernel.

    Args:
        left, right: tensors of shape (B, N, N, H) in fp16, contiguous.

    Returns:
        Tensor of shape (B, N, N, H) in fp16.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = ((N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            B * H)

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out

def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to FP16 for the heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Cast weights to FP16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # Gate projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # Core O(N³·H) reduction via Triton
    # ------------------------------------------------------------------
    out = tri_mul_forward(left, right)   # (B, N, N, H) fp16

    # ------------------------------------------------------------------
    # Output LayerNorm, output‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # Cast back to original dtype (float32)
    out = out.to(input_tensor.dtype)
    return out

Highly‑optimized outgoing TriMul implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input (B, N, N, C) in FP32.
2. Cast to FP16 for compute‑intensive stages.
3. Linear projections (left/right) and three gate projections are performed
   with torch.nn.functional.linear (cuBLAS) in FP16.
4. Optional mask is applied (broadcast over the hidden dimension) and the
   gates are multiplied.
5. Core reduction:
       out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   is executed by a custom Triton kernel that processes a tile of i‑rows
   and j‑columns for each (batch, head) pair, accumulating in FP32 and
   storing the result in FP16.
6. A second LayerNorm, multiplication by the output gate and a final linear
   projection bring the tensor back to the original channel dimension.
7. The result is cast back to the input dtype (usually float32).

The kernel fuses the O(N³·H) reduction without resorting to cuBLAS,
yielding a large speed‑up for the target configurations.

        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: (i, k)  shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            b * stride_lb + h * stride_lh + \
            offs_i[:, None] * stride_li + \
            offs_k[None, :] * stride_lk

        # right block: (k, j)  shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + \
            b * stride_rb + h * stride_rh + \
            offs_k[:, None] * stride_rk + \
            offs_j[None, :] * stride_rj

        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(left, right)

    # write result back, casting to FP16
    out_ptrs = out_ptr + \
        b * stride_ob + h * stride_oh + \
        offs_i[:, None] * stride_oi + \
        offs_j[None, :] * stride_oj

    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def tri_mul_forward(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel.
    left, right : tensors of shape (B, N, N, H) in fp16, contiguous.
    Returns a tensor of the same shape and dtype (fp16).
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i blocks
        (N + BLOCK_N - 1) // BLOCK_N,   # j blocks
        B * H                           # batch * hidden
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask        : torch.Tensor [B, N, N] or None
            - weights     : dict of model parameters
            - config      : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to FP16 for the heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Cast model weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6. Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguity for pointer arithmetic inside the kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 7. Core reduction via Triton
    # ------------------------------------------------------------------
    out = tri_mul_forward(left, right)   # (B, N, N, H) fp16

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)        # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype (float32) and return
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
