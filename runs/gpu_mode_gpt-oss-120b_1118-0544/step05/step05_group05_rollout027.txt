"""
Optimized outgoing Triangular Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑normalize the 4‑D input (float32 for stability).
2. Cast to fp16 and compute left/right projections together with their
   respective gates in two fused linear layers (project+gate concatenated).
3. Fuse gate, optional mask and element‑wise multiplication of the
   projected tensors with a small Triton kernel (in‑place, no extra memory).
4. Perform the core O(N³·H) contraction with a single torch.einsum
   (equivalent to batched GEMM across the hidden dimension).
5. Apply the output LayerNorm, output gate and final linear projection.
   The result is cast back to the original dtype (usually float32).

Only part of the work (gate/mask fusion) is performed in Triton as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,          # [B,N,N,H] flattened (in‑place output)
    left_gate_ptr, right_gate_ptr,
    mask_ptr,                     # [B,N,N] flattened (broadcasted over H)
    total_elems: tl.constexpr,   # number of elements in left/right
    hidden_dim: tl.constexpr,
    HAS_MASK: tl.constexpr,      # compile‑time bool
    BLOCK_SIZE: tl.constexpr,
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    Mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left_val = tl.load(left_ptr + offs, mask=active, other=0.0)
    right_val = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate_val = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden_dim entries share a mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left_val = left_val * left_gate_val * mask_val
        right_val = right_val * right_gate_val * mask_val
    else:
        left_val = left_val * left_gate_val
        right_val = right_val * right_gate_val

    tl.store(left_ptr + offs, left_val, mask=active)
    tl.store(right_ptr + offs, right_val, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Fused projection + gate for left/right streams
    # --------------------------------------------------------------
    # concatenate proj and gate weights to halve the number of matmuls
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device).to(torch.float16)   # (2*H, dim)
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device).to(torch.float16)   # (2*H, dim)

    # combined linear -> shape (B,N,N,2*H)
    left_combined = F.linear(x, left_combined_w)
    right_combined = F.linear(x, right_combined_w)

    # split into proj and gate parts
    left_proj, left_gate_raw = torch.split(left_combined, hidden, dim=-1)
    right_proj, right_gate_raw = torch.split(right_combined, hidden, dim=-1)

    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # --------------------------------------------------------------
    # 4. Fuse gate, optional mask and element‑wise multiplication
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # broadcast mask over hidden dim
        mask_exp = mask_fp.view(-1)  # (B*N*N,)
        mask_tensor = mask_exp
        has_mask = True
    else:
        # dummy tensor (will not be accessed)
        mask_tensor = left_proj  # any tensor with same dtype
        has_mask = False

    # in‑place fusion: left_proj <- left_proj * left_gate * mask,
    #                  right_proj <- right_proj * right_gate * mask
    total = left_proj.numel()
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

    _gate_mask_fuse_kernel[grid](
        left_proj,               # output pointer (in‑place)
        right_proj,
        left_gate,
        right_gate,
        mask_tensor,
        total,
        hidden,
        HAS_MASK=has_mask,
        BLOCK_SIZE=BLOCK,
    )
    # After the kernel, left_proj and right_proj already contain the
    # gated (and masked) values.

    # --------------------------------------------------------------
    # 5. Core O(N³·H) contraction via einsum
    # --------------------------------------------------------------
    out = torch.einsum('... i k d, ... j k d -> ... i j d', left_proj, right_proj)

    # --------------------------------------------------------------
    # 6. Output LayerNorm, output gate and final linear projection
    # --------------------------------------------------------------
    # output LayerNorm on hidden dimension
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b)

    # output gate
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))
    out = out * out_gate

    # final linear projection back to original dim
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # --------------------------------------------------------------
    # 7. Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
