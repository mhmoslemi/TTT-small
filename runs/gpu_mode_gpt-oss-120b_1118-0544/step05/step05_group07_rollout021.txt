def matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Am, stride_Ak,  # batch_stride maybe?
    stride_Bk, stride_Bn,
    stride_Cm, stride_Cn,
    BATCH_STRIDE_A: tl.constexpr,
    BATCH_STRIDE_B: tl.constexpr,
    BATCH_STRIDE_C: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask to handle out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers with batch offset
    a_ptrs = A_ptr + batch_id * BATCH_STRIDE_A + (offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak)
    b_ptrs = B_ptr + batch_id * BATCH_STRIDE_B + (offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn)
    # load a and b for loop over K
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K - k * BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K - k * BLOCK_K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # store result to C
    c_ptrs = C_ptr + batch_id * BATCH_STRIDE_C + (offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass – Triton‐fused mask & gate and a custom
batched GEMM.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) using provided weight/bias.
2. Cast to fp16 for compute‑intensive parts.
3. Linear projections (left/right) and their gates are computed with
   torch.nn.functional.linear (cuBLAS).
4. Fuse element‑wise mask, gate and projection using a small Triton kernel.
5. Core contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batch of H independent GEMMs of size N×N.
   We implement this batched GEMM with a custom Triton kernel
   (no extra kernel launches as in torch.bmm).
6. Apply a second LayerNorm, multiply by the output gate and project back
   to the original channel dimension.
7. Cast the result back to the input dtype (float32).

Only the mask‑gate fusion and the batched GEMM are performed in Triton,
fulfilling the requirement of using a Triton kernel while keeping the
implementation correct and reasonably fast.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask (mask broadcast over hidden)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32   – total number of elements in proj/gate
    hidden_dim,        # i32   – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # mask index after broadcasting hidden dim
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """Fuse `proj * gate * mask` using Triton."""
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],               # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the core O(N³·H) reduction
# Computes: C[b] = A[b] @ B[b]ᵀ   with A,B shape (batch, M, K) and (batch, K, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers with batch offset
        a_ptrs = A_ptr + batch_id * batch_stride_a + (
            offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        )
        b_ptrs = B_ptr + batch_id * batch_stride_b + (
            offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)  # accumulate in FP32

    # Write back result
    c_ptrs = C_ptr + batch_id * batch_stride_c + (
        offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    device = inp.device
    dtype_in = inp.dtype

    # ------------------- LayerNorm (input) -------------------------------
    x = torch.nn.functional.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for compute‑heavy parts
    x = x.to(torch.float16)

    # ----------------------- cast weights to fp16 -----------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ----------------------- linear projections ---------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------- gates -----------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()  # (B,N,N,H)

    # ------------------- mask & gate fusion (Triton) --------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ----------------- core O(N³·H) reduction via custom batched GEMM ----------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) and prepare rightᵀ
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_mat   = left_perm.view(-1, N, N)                # (B*H, N, N)
    right_mat_T = right_perm.transpose(2, 3).contiguous().view(-1, N, N)  # (B*H, N, N)

    out_mat = torch.empty_like(left_mat)

    # strides
    stride_am, stride_ak = left_mat.stride(1), left_mat.stride(2)
    batch_stride_a = left_mat.stride(0)

    stride_bk, stride_bn = right_mat_T.stride(1), right_mat_T.stride(2)
    batch_stride_b = right_mat_T.stride(0)

    stride_cm, stride_cn = out_mat.stride(1), out_mat.stride(2)
    batch_stride_c = out_mat.stride(0)

    M = N
    N_dim = N
    K = N

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (out_mat.shape[0],                     # batch dimension
            triton.cdiv(M, BLOCK_M),
            triton.cdiv(N_dim, BLOCK_N))

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        M, N_dim, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # ------------------------ cast back to input dtype ------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
