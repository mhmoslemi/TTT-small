@triton.jit
def fused_proj_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr,
    out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    # mask for valid indices
    mask = offs < total_elems
    # Compute mask index: because mask has shape (B*N*N, )? Actually mask shape after unsqueeze: [B,N,N,1] => total_mask = B*N*N.
    # We compute mask index as offs // hidden_dim
    mask_idx = offs // hidden_dim
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)
    
    # LayerNorm on input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    
    # Cast to fp16
    x = x.to(torch.float16)
    
    # Cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)
    
    # Combine weight for first 5 linear ops
    combined_weight = torch.cat([
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w,
        out_gate_w
    ], dim=0)   # shape (5*hidden_dim, dim)
    
    # Linear: (B,N,N,5*H)
    combined = torch.nn.functional.linear(x, combined_weight)  # bias None
    
    # Split
    splits = torch.split(combined, hidden_dim, dim=-1)
    left_proj, right_proj, left_gate, right_gate, out_gate = splits
    
    # Apply sigmoid to gate tensors
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()
    
    if mask is not None and not nomask:
        # Ensure mask dtype / shape
        mask_fp16 = mask.to(torch.float16)
        # expand mask to shape (B,N,N,1)
        mask_fp16 = mask_fp16.unsqueeze(-1)  # shape (B,N,N,1)
        # Flatten pointers for kernel
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
    
    # Compute batched GEMM
    B, N, _, _ = left.shape # left shape (B,N,N,H)
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Reshape to (B*H, N, N)
    left_reshaped = left_perm.view(-1, N, N)
    right_reshaped = right_perm.view(-1, N, N)
    # Compute solution
    out_reshaped = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))
    # Reshape back
    out = out_reshaped.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    
    # LayerNorm on hidden dim
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    
    # Multiply by out_gate (already fp16)
    out = out * out_gate
    
    # Final linear projection
    out = torch.nn.functional.linear(out, to_out_w)
    
    # Cast back to original dtype (float32)
    return out.to(input_tensor.dtype)
```

def _apply_gate_and_mask(proj, gate, mask, block_size=1024):
    # proj, gate: shape (B, N, N, H)
    # mask: shape (B, N, N, 1) or (B, N, N)
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _fused_proj_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )

"""
TriMul (outgoing) forward pass – Triton fused mask & gate.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) (float32 → float16).
2. Fuse the five linear layers (left_proj, right_proj, left_gate, right_gate,
   out_gate) into a single matmul with a concatenated weight matrix
   (shape (5*H, C)).  Split the result to obtain the individual tensors.
3. Apply sigmoid to the three gate tensors.
4. Fuse element‑wise mask, projection and gate for *left* and *right*
   using a tiny Triton kernel (`_fused_proj_gate_mask_kernel`).
   This computes `proj * gate * mask` in one pass.
5. Compute the core O(N³·H) reduction as a batch of GEMMs:
   for each batch b and hidden channel h,
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   Implemented via `torch.bmm` on tensors reshaped to (B*H, N, N).
6. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate, and project back to the original channel dimension with a final
   linear layer.
7. Cast the result back to the input dtype (float32).

Only the element‑wise mask & gate fusion is performed in Triton; the heavy
batched GEMM leverages cuBLAS/Tensor‑cores.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused (proj * gate * mask) for a flattened view
# ----------------------------------------------------------------------
@triton.jit
def _fused_proj_gate_mask_kernel(
    proj_ptr,     # fp16 * (B*N*N*H,)
    gate_ptr,     # fp16 * (B*N*N*H,)
    mask_ptr,     # fp16 * (B*N*N,)
    out_ptr,      # fp16 * (B*N*N*H,)
    total_elems,  # i32 – total number of elements in proj/gate
    hidden_dim,   # i32 – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # mask is broadcasted over the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using Triton.

    All inputs must be contiguous fp16 tensors.
    `proj` and `gate` have shape (B, N, N, H),
    `mask` has shape (B, N, N, 1) (broadcastable to H).
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _fused_proj_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],          # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------- LayerNorm (input) -------------------------------
    # LN over the channel dimension C
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for compute‑intensive part
    x = x.to(torch.float16)

    # ------------------- Cast weights to fp16 ---------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------- Fuse five linear layers ------------------------
    # Concatenate the five weight matrices (each hidden_dim x dim)
    combined_weight = torch.cat([
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
    ], dim=0)                     # shape (5*H, C)

    # Single matmul: (B,N,N,C) @ (C,5*H)^T → (B,N,N,5*H)
    combined = torch.nn.functional.linear(x, combined_weight)

    # Split into individual tensors
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        combined, hidden_dim, dim=-1
    )

    # ------------------- Gates (sigmoid) --------------------------------
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # ------------------- Mask & gate fusion (Triton) --------------------
    if mask is not None and not nomask:
        # mask: [B, N, N] → fp16 → unsqueeze to broadcast over H
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------- Core O(N³·H) reduction via batched GEMM -------
    B, N, _, _ = left.shape  # left shape (B,N,N,H)

    # Rearrange to (B, H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Merge batch and hidden dimensions → (B*H, N, N)
    left_mat = left_perm.view(-1, N, N)
    right_mat = right_perm.view(-1, N, N)

    # Batched GEMM: left @ right^T   (each batch corresponds to a hidden channel)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- Second LayerNorm -------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Multiply by output gate (broadcasted over hidden dim)
    out = out * out_gate

    # ------------------- Final linear projection ------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,C)

    # Cast back to the original dtype (typically float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
