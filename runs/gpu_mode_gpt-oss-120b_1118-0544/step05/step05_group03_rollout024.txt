"""
TriMul (outgoing) forward pass – Triton‑accelerated elementwise gate/mask fusion
and a custom batched GEMM kernel.

Algorithm
---------
1. LayerNorm the input over the channel dimension `dim`.
2. Cast to fp16 for all heavy compute.
3. Linear projections:
      left  = x @ W_left      (B,N,N,H)
      right = x @ W_right
   Gate projections (sigmoid):
      left_gate  = σ(x @ W_left_gate)
      right_gate = σ(x @ W_right_gate)
      out_gate   = σ(x @ W_out_gate)    # (B,N,N,H)
4. Fuse projection * gate * (optional) mask with a Triton kernel
   (broadcast mask over the hidden dimension H).
5. Core tensor update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (left @ rightᵀ) in a dedicated
   Triton kernel; the batch dimension is B·H.
6. Output LayerNorm over hidden dimension H, multiply by `out_gate`,
   and a final linear projection back to `dim`.
7. Cast the result back to the original dtype.

The implementation mixes Triton kernels (steps 4 & 5) with
torch.nn.functional for the remaining ops.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------------
# Kernel 1: fuse projection, gate and mask (broadcast over hidden dim)
# ------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i64: total number of elements in proj/gate (B*N*N*H)
    hidden_dim,        # i64: H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # each hidden_dim consecutive elements share the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 4096) -> torch.Tensor:
    """Elementwise `proj * gate * mask` (mask broadcast over last dim)."""
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ------------------------------------------------------------------
# Kernel 2: batched matrix multiplication (left @ rightᵀ)
# ------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    left_ptr,          # fp16 * (P, N, N)
    right_ptr,         # fp16 * (P, N, N)
    out_ptr,           # fp16 * (P, N, N)
    N,                 # i32
    stride_pl, stride_il, stride_kl,   # left strides
    stride_pr, stride_ir, stride_kr,   # right strides
    stride_po, stride_io, stride_jo,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)      # 0 .. P-1
    pid_m = tl.program_id(0)          # row tile
    pid_n = tl.program_id(1)          # col tile

    # compute start indices of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulators in fp32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile: (M, K) -----
        left_ptrs = left_ptr + \
            pid_batch * stride_pl + \
            offs_m[:, None] * stride_il + \
            offs_k[None, :] * stride_kl
        a = tl.load(left_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # ----- load right tile (transposed on the fly): (K, N) -----
        # we need right[batch, k, j] -> treat k as "row", j as "col"
        right_ptrs = right_ptr + \
            pid_batch * stride_pr + \
            offs_k[:, None] * stride_ir + \
            offs_n[None, :] * stride_kr
        b = tl.load(right_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # accumulate
        acc += tl.dot(a, b)   # a:(M,K) , b:(K,N) -> (M,N)

    # ----- store result tile -----
    out_ptrs = out_ptr + \
        pid_batch * stride_po + \
        offs_m[:, None] * stride_io + \
        offs_n[None, :] * stride_jo
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ------------------------------------------------------------------
# Entry point
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] (binary) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy part
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all linear weights to fp16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B, N, N, H)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid (same shape)
    # --------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Apply mask and gating (fused via Triton if mask is present)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16, contiguous and broadcastable
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction: batched GEMM (left @ rightᵀ) via Triton
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    P = B * H  # total batch for the GEMM

    # reshape to (P, N, N) – contiguous layout
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(P, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(P, N, N)
    out_flat   = torch.empty_like(left_flat)

    # strides for the contiguous (P,N,N) tensors
    stride_pl, stride_il, stride_kl = left_flat.stride()
    stride_pr, stride_ir, stride_kr = right_flat.stride()
    stride_po, stride_io, stride_jo = out_flat.stride()

    # Choose block sizes (tuned for typical B200 GPU)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        P,                         # batch (B*H)
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_pl, stride_il, stride_kl,
        stride_pr, stride_ir, stride_kr,
        stride_po, stride_io, stride_jo,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, apply out_gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 9. Cast back to the original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
