"""
Optimized outgoing TriMul forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32 for numerical stability).
2. Project the normalized tensor to a hidden dimension with three linear layers:
   - left_proj, right_proj for the two operands of the contraction.
   - left_gate, right_gate, out_gate for sigmoid‑gated modulation.
   * All linear ops are performed in fp16.
3. Fuse element‑wise mask (if present) and gate multiplication for the left and
   right tensors in a single Triton kernel, drastically reducing memory traffic.
4. Perform the core O(N³·H) contraction using a batched GEMM:
   out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   realised as a batched matrix multiply (bmm) over the flattened (h, N, N)
   tensors – this is the same as the einsum in the reference implementation.
5. Apply the output LayerNorm, multiply by the output gate, and a final linear
   projection back to the original feature dimension.
6. Cast the result back to the input dtype (typically float32).

Only the mask‑gate fusion step is implemented in Triton; the remaining
operations rely on highly‑optimised PyTorch kernels (cuBLAS for the batched
matrix multiply, fused linear ops, etc.).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,               # [total] flattened fp16
    left_gate_ptr, right_gate_ptr,     # same shape as left/right
    mask_ptr,                          # [B*N*N] flattened fp16 (0 or 1)
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    Mask is broadcast over the hidden dimension. When HAS_MASK=False the mask
    load is omitted and the operation reduces to element‑wise gate multiplication.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load data
    left      = tl.load(left_ptr      + offs, mask=active, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=active, other=0.0)

    if HAS_MASK:
        # Each hidden_dim contiguous elements share a mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple):
            - input_tensor: torch.Tensor of shape (B, N, N, dim)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (all weight tensors are
                       2‑D with shape [out_dim, in_dim] and bias tensors are 1‑D)
            - config: dict with keys
                * "dim"         – input feature dimension (C)
                * "hidden_dim"  – hidden dimension (H)
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape (B, N, N, dim) with the same dtype as the input.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # Convert to fp16 for the heavy compute
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gates (fp16)
    # -----------------------------------------------------------------
    # Linear without bias (bias=False in the original module)
    left      = F.linear(x, left_proj_w)          # (B, N, N, H)
    right     = F.linear(x, right_proj_w)         # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 4. Fuse mask and gate multiplication in Triton
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is float16 (0.0/1.0) on the same device
        mask_fp = mask.to(device).to(torch.float16)
        # Flatten tensors for kernel launch
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _mask_gate_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask → only gate multiplication (still fused in‑place)
        left  = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ rightᵀ) over the sequence dim
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # Shape to (B*H, N, N) for bmm
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batch matrix multiplication: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    # Apply LayerNorm in float32 for numerical safety, then cast back to fp16
    out_fp32 = out.to(torch.float32)
    out_fp32 = F.layer_norm(out_fp32, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out_fp32 = out_fp32 * out_gate.to(torch.float32)
    out_fp32 = F.linear(out_fp32, to_out_w.to(torch.float32))   # (B,N,N,dim)

    # Cast back to original input dtype (usually float32)
    if out_fp32.dtype != dtype_in:
        out_fp32 = out_fp32.to(dtype_in)

    return out_fp32
