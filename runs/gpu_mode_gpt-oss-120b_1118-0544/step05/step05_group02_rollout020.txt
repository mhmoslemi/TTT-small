def batched_gemm(a, b):
    # a: (B*H, M, K) float16 contiguous
    # b: (B*H, K, N) float16 contiguous
    batch = a.shape[0]
    M, K = a.shape[1], a.shape[2]
    _, Kb, N = b.shape
    assert K == Kb
    # strides
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    # allocate output
    out = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)
    stride_c_batch, stride_c_m, stride_c_n = out.stride()
    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (batch, (M + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)
    batched_gemm_kernel[grid](
        a, b, out,
        batch, M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out
```

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to batch start
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)

        # Load B tile
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        # Dot product
        acc += tl.dot(a, b)

    # Write back
    c = acc.to(tl.float16)
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             c,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

def batched_gemm(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Compute batched matrix multiplication C = A @ B, where
    A shape (batch, M, K), B shape (batch, K, N). All tensors must be
    contiguous and of dtype torch.float16.
    """
    batch, M, K = a.shape
    _, Kb, N = b.shape
    assert K == Kb
    # strides in elements
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    # allocate output
    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)
    stride_c_batch, stride_c_m, stride_c_n = c.stride()
    # block sizes (tuned)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)
    batched_gemm_kernel[grid](
        a, b, c,
        batch, M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return c

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the input tensor over its channel dimension.
2. Cast to FP16 for compute‑intensive stages.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) are performed with torch.nn.functional.linear
   (cuBLAS) on FP16 tensors.
4. If a mask is present, a tiny Triton kernel fuses
       proj * gate * mask
   broadcasting the mask over the hidden dimension.
5. Core O(N³·H) reduction:
       out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   is implemented as a batched GEMM (left @ rightᵀ) using a custom
   Triton kernel, processing the (batch * hidden_dim) matrices in
   parallel.
6. Apply a second LayerNorm, multiply by the output gate and linearly
   project back to the original channel dimension.
7. Cast the result back to the input dtype.

Only the element‑wise gate‑mask fusion and the batched GEMM are written
in Triton; all other operations use PyTorch for simplicity and
correctness.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    proj_ptr,               # *[total_elems]  fp16
    gate_ptr,               # *[total_elems]  fp16
    mask_ptr,               # *[mask_len]    fp16
    out_ptr,                # *[total_elems] fp16
    total_elems,            # int
    hidden_dim,             # int
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse proj * gate * mask (mask broadcast over hidden_dim)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # each mask element repeats `hidden_dim` times
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM kernel: C = A @ B  where
        A: (batch, M, K)
        B: (batch, K, N)
        C: (batch, M, N)
    Accumulation is in FP32, result stored as FP16.
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        acc += tl.dot(a, b)   # FP32 accumulation

    c = acc.to(tl.float16)
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             c,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    proj, gate : (B, N, N, H)  fp16 contiguous
    mask       : (B, N, N)    fp16 contiguous
    Returns    proj * gate * mask  (fp16)
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (math.ceil(total / block_size),)

    mask_flat = mask.view(-1)  # (B*N*N,)
    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


def batched_gemm(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Compute batched matrix multiplication C = A @ B where
        A: (batch, M, K)   fp16
        B: (batch, K, N)   fp16
    Returns C of shape (batch, M, N) fp16.
    """
    batch, M, K = a.shape
    _, Kb, N = b.shape
    assert K == Kb, "Inner dimensions must match"

    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()

    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_gemm_kernel[grid](
        a, b, c,
        batch, M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return c


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    data = (input_tensor, mask, weights, config)
    Returns: Tensor [bs, seq_len, seq_len, dim] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> float16 later)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑intensive part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all model weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6. Fuse mask and gate (Triton) or simple elementwise multiply
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM (left @ rightᵀ) via Triton
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) and then merge batch & head dimensions
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    batch_total = B * H

    left_flat = left_perm.view(batch_total, N, N)      # (B*H, N, N)
    right_flat = right_perm.view(batch_total, N, N)    # (B*H, N, N)

    # Transpose the second operand for the dot product
    right_T = right_flat.transpose(1, 2).contiguous()  # (B*H, N, N)

    out_flat = batched_gemm(left_flat, right_T)        # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)

    return out
