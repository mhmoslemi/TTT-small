@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    BH, N,
    stride_l_b, stride_l_m, stride_l_k,
    stride_r_b, stride_r_k, stride_r_n,
    stride_out_b, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    batch_pid = tl.program_id(1)
    # Compute block coordinates
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_blocks_n
    pid_n = pid % num_blocks_n
    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # masks for bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator in float32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block (M x K)
        a_ptrs = left_ptr + batch_pid * stride_l_b + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B block (K x N) - note using right as transposed (k,n)
        b_ptrs = right_ptr + batch_pid * stride_r_b + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Compute dot product
        acc += tl.dot(a, b)

    # Write back output
    out_ptrs = out_ptr + batch_pid * stride_out_b + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass – Triton‑accelerated core reduction.

Algorithm
---------
1. Input LayerNorm (torch.nn.functional.layer_norm).
2. Cast to FP16 for compute‑intensive stages.
3. Linear projections `left_proj` / `right_proj` and gate projections.
4. Fuse `proj * gate * mask` with a small Triton kernel
   (mask is broadcast over the hidden dimension).
5. Core O(N³·H) reduction:
   For each hidden slice h we need
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication `left @ rightᵀ`.
   We implement a custom Triton matmul kernel that processes the
   flattened batch (B·H) dimension, accumulating in FP32 and storing FP16.
6. Output LayerNorm → gating → final linear projection.
7. Cast back to the original dtype (usually float32).

All heavy compute (steps 4‑5) runs on the GPU via Triton;
the remaining operations use highly‑optimized PyTorch kernels.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise proj * gate * mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # *[total_elems]   fp16
    gate_ptr,          # *[total_elems]   fp16
    mask_ptr,          # *[mask_elems]    fp16
    out_ptr,           # *[total_elems]   fp16
    total_elems,       # i64
    hidden_dim,        # i64  (size of the last dimension H)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # map flat index -> mask index (broadcast over H)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(out_ptr + offs, proj * gate * m, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * gate * mask`.  `mask` has shape (B,N,N) and is broadcast
    over the hidden dimension H.
    All inputs must be fp16 and contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched matmul (A @ Bᵀ) for tensors of shape (BH, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    BH, N,                              # int64
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for each batch in BH using FP16 inputs,
    accumulating in FP32 and storing FP16."""
    pid = tl.program_id(0)          # combined (m,n) block index
    batch = tl.program_id(1)        # batch index (0 .. BH-1)

    # 2‑D grid layout
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_blocks_n
    pid_n = pid % num_blocks_n

    # Offsets for the output block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block (M x K)
        a_ptrs = a_ptr + batch * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B block (K x N) – note we use Bᵀ, so rows = k, cols = n
        b_ptrs = b_ptr + batch * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Dot product (FP16×FP16 → FP32 accumulation)
        acc += tl.dot(a, b)

    # Write result
    c_ptrs = c_ptr + batch * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3 style).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict of layer‑norm and linear parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for the compute‑heavy stages
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. FP16 weight tensors
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections
    # --------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Fuse projection * gate * mask (optional)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core O(N³·H) reduction via a custom Triton batched GEMM
    # --------------------------------------------------------------
    B, N, _, H = left.shape                     # left shape: (B, N, N, H)

    # Rearrange to (B, H, N, N) → (BH, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * H
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    # Prepare output tensor (FP16)
    out_flat = torch.empty_like(left_flat)

    # Block sizes – tuned for typical B200 memory / compute
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (M‑blocks * N‑blocks, batch‑dimension)
    grid = (
        (math.ceil(N / BLOCK_M) * math.ceil(N / BLOCK_N),),
        (BH,),
    )

    # Strides for the flattened tensors (they are contiguous)
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_flat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)          # (B, N, N, C)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
