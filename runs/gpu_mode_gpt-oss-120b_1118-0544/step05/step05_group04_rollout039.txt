custom_kernel: Triton‑accelerated outgoing TriMul.

Implements the forward pass of the TriMul module described in AlphaFold3.
...
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _apply_mask_gate_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    left_out_ptr, right_out_ptr,
    H: tl.constexpr,
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    tl.store(left_out_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute the batch offset (in elements)
    a_batch = pid_z * stride_ab
    b_batch = pid_z * stride_bb
    c_batch = pid_z * stride_cb

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile
        a_ptrs = a_ptr + a_batch + offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (k_offset[None, :] < K),
                    other=0.0)

        # Load B tile (note B is already transposed: shape (K, N))
        b_ptrs = b_ptr + b_batch + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=(k_offset[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict containing model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """

    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # Work in float16 for the heavy arithmetic
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Cast all linear weights to float16
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections to hidden dimension
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_w)     # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 4. Gates (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # -----------------------------------------------------------------
    # 5. Apply optional mask and fuse with left/right gates (Triton)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_tensor = mask.to(device=device, dtype=torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    total_elements = left.numel()  # B * N * N * H
    BLOCK = 1024

    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = (triton.cdiv(total_elements, BLOCK),)
    _apply_mask_gate_kernel[grid](
        left,
        right,
        mask_tensor.view(-1),
        left_gate,
        right_gate,
        left_out,
        right_out,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 6. Core reduction: batch GEMM using Triton (A @ B^T)
    #    left_out  : (B, N, N, H) -> (B, H, N, N) -> (BH, N, N)
    #    right_out : (B, N, N, H) -> (B, H, N, N) -> (BH, N, N) and transposed
    # -----------------------------------------------------------------
    # Rearrange to (B, H, N, N)
    left_perm = left_out.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N)
    right_perm = right_out.permute(0, 3, 2, 1).contiguous()    # (B, H, N, N) == (B, H, K, N) where K=N

    # Flatten batch*hidden
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)          # (BH, M=N, K=N)
    right_flat = right_perm.view(BH, N, N)        # (BH, K=N, N)

    out_flat = torch.empty_like(left_flat)        # (BH, M, N)

    # Strides (in elements)
    stride_am = left_flat.stride(1)   # row stride = N
    stride_ak = left_flat.stride(2)   # col stride = 1
    stride_bk = right_flat.stride(1)  # row stride = N
    stride_bn = right_flat.stride(2)  # col stride = 1
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    stride_ab = left_flat.stride(0)   # batch stride = N * N
    stride_bb = right_flat.stride(0)
    stride_cb = out_flat.stride(0)

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        BH,
    )
    _triton_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                     # M, N, K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ab,
        stride_bb,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # 7. Reshape back to (B, N, N, H)
    # -----------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # -----------------------------------------------------------------
    # 8. Output LayerNorm over hidden dimension
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 9. Apply output gate
    # -----------------------------------------------------------------
    out = out * out_gate

    # -----------------------------------------------------------------
    # 10. Final linear projection back to original dim
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # -----------------------------------------------------------------
    # 11. Cast back to original dtype if needed
    # -----------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Implements the forward pass of the TriMul module from AlphaFold3.
The heavy O(N³·H) reduction is performed with a custom Triton batched GEMM,
while mask‑gate fusion is also realized in Triton. All other linear
operations use torch.nn.functional for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,            # const float* left   (B,N,N,H) flattened
    right_ptr,           # const float* right
    mask_ptr,            # const float* mask   (B,N,N) flattened
    left_gate_ptr,       # const float* left_gate
    right_gate_ptr,      # const float* right_gate
    left_out_ptr,        # float* left_out
    right_out_ptr,       # float* right_out
    H: tl.constexpr,     # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask and sigmoid gates:
       left_out  = left  * mask * left_gate
       right_out = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    within = offs < total_elements

    l = tl.load(left_ptr + offs,     mask=within)
    r = tl.load(right_ptr + offs,    mask=within)
    lg = tl.load(left_gate_ptr + offs,  mask=within)
    rg = tl.load(right_gate_ptr + offs, mask=within)

    # broadcast mask over the hidden dimension
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=within)

    tl.store(left_out_ptr + offs,  l * m * lg, mask=within)
    tl.store(right_out_ptr + offs, r * m * rg, mask=within)


@triton.jit
def _triton_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr = 4,
):
    """Batched GEMM: for each batch (pid_z) compute C = A @ B,
       where A shape (M,K), B shape (K,N), C shape (M,N).
       All tensors are in row‑major layout.
    """
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_z = tl.program_id(2)   # batch * hidden

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # batch offsets (in elements)
    a_batch = pid_z * stride_ab
    b_batch = pid_z * stride_bb
    c_batch = pid_z * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_ptr + a_batch + offs_m[:, None] * stride_am + k_off[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (k_off[None, :] < K),
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_ptr + b_batch + k_off[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=(k_off[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)
