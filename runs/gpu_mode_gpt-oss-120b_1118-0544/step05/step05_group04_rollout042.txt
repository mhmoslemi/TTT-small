                     B, N, dim, hidden_dim,
                     stride_xb, stride_xi, stride_xj, stride_xd,
                     stride_outb, stride_outi, stride_outj, stride_outh,
                     stride_wi, stride_wo,
                     BLOCK_H: tl.constexpr, BLOCK_K: tl.constexpr):
   pid0 = tl.program_id(0) # tile for (b,i,j)
   pid1 = tl.program_id(1) # hidden block index

   b = pid0 // (N * N)

   i = tmp // N

def custom_kernel(data):
   input_tensor, mask, weights, config = data
   device = input_tensor.device
   orig_dtype = input_tensor.dtype
   dim = config["dim"]
   hidden_dim = config["hidden_dim"]
   nomask = config.get("nomask", False)

   # LayerNorm
   x = torch.nn.functional.layer_norm(input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32)
   )
   # Cast to fp16
   x = x.to(torch.float16)

   # Prepare mask
   B, N, _, _ = x.shape
   if mask is None or nomask:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)
   else:
        mask_tensor = mask.to(torch.float16)

   # Allocate output tensors
   left_out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)
   right_out = torch.empty_like(left_out)
   out_gate = torch.empty_like(left_out)

   # Get weight tensors and cast to fp16
   w_left = weights["left_proj.weight"].to(device=device, dtype=torch.float16) # shape (hidden_dim, dim)
   w_right = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
   w_left_gate = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
   w_right_gate = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
   w_out_gate = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

   # Compute strides
   stride_xb, stride_xi, stride_xj, stride_xd = x.stride()
   stride_outb, stride_outi, stride_outj, stride_outh = left_out.stride()
   stride_maskb, stride_maski, stride_maskj = mask_tensor.stride()
   # weight strides
   stride_wk = w_left.stride(1)  # stride for dim (inner)
   stride_wh = w_left.stride(0)  # stride for hidden dimension

   # Determine block sizes
   BLOCK_H = 64  # may adjust
   BLOCK_K = 64  # choose optimum for dim <= 384

   # Determine grid sizes
   grid0 = B * N  # (b,i) combos
   grid1 = N      # j dimension
   grid2 = (hidden_dim + BLOCK_H - 1) // BLOCK_H

   grid = (grid0, grid1, grid2)

   # Launch kernel
   _proj_gate_kernel[grid](
        x, left_out, right_out, out_gate,
        w_left, w_right, w_left_gate, w_right_gate, w_out_gate,
        mask_tensor,
        B, N, dim, hidden_dim,
        stride_xb, stride_xi, stride_xj, stride_xd,
        stride_outb, stride_outi, stride_outj, stride_outh,
        stride_maskb, stride_maski, stride_maskj,
        stride_wk, stride_wh,
        BLOCK_H=BLOCK_H, BLOCK_K=BLOCK_K,
        num_warps=4,
   )

   # Now compute core reduction with torch.bmm
   # reshape left_out and right_out to (B*hidden_dim, N, N)
   left_flat = left_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
   right_flat = right_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

   # Compute out via batched matmul
   # multiply left_flat (B*H, N, N) with right_flat.transpose(-2, -1) (B*H, N, N)
   out_flat = torch.bmm(left_flat, right_flat.transpose(-2, -1))  # shape (B*H, N, N)

   # Reshape back to (B, N, N, hidden_dim)
   out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

   # Apply to_out_norm (layernorm)
   out = torch.nn.functional.layer_norm(
       out,
       (hidden_dim,),
       weight=weights["to_out_norm.weight"].to(device=device, dtype=torch.float32),
       bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float32),
   )

   # Multiply by out_gate
   out = out * out_gate.to(out.dtype)  # ensure same dtype

   # Final linear projection to original dim
   to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
   out = torch.nn.functional.linear(out, to_out_w)  # returns float16

   # Cast back to original dtype
   out = out.to(orig_dtype)

   return out

@triton.jit
def _proj_gate_kernel(
    x_ptr, left_out_ptr, right_out_ptr, out_gate_ptr,
    w_left_ptr, w_right_ptr, w_lgate_ptr, w_rgate_ptr, w_outgate_ptr,
    mask_ptr,
    B, N, dim, hidden_dim,
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_outb, stride_outi, stride_outj, stride_outh,
    stride_maskb, stride_maski, stride_maskj,
    stride_wk, stride_wh,
    BLOCK_H: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Projection + gating kernel.
    Computes left = (x @ W_left) * mask * sigmoid(x @ W_left_gate)
    similarly for right. Also computes out_gate = sigmoid(x @ W_out_gate).
    """
    pid0 = tl.program_id(0)  # b_i index
    pid1 = tl.program_id(1)  # j index
    pid2 = tl.program_id(2)  # hidden block

    # decode b,i
    b = pid0 // N
    i = pid0 % N
    j = pid1
    h_start = pid2 * BLOCK_H
    offs_h = h_start + tl.arange(0, BLOCK_H)
    h_mask = offs_h < hidden_dim

    # load mask scalar
    mask_offset = b * stride_maskb + i * stride_maski + j * stride_maskj
    m = tl.load(mask_ptr + mask_offset, mask=tl.full([1], True, tl.int1), other=0.0)
    m_f = tl.float32(m)  # cast

    # initialize accumulators
    acc_l = tl.zeros((BLOCK_H,), dtype=tl.float32)
    acc_r = tl.zeros((BLOCK_H,), dtype=tl.float32)
    acc_lg = tl.zeros((BLOCK_H,), dtype=tl.float32)
    acc_rg = tl.zeros((BLOCK_H,), dtype=tl.float32)
    acc_og = tl.zeros((BLOCK_H,), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, dim, BLOCK_K):
        k_end = k + tl.arange(0, BLOCK_K)
        k_mask = k_end < dim

        # load x slice
        x = tl.load(x_ptr + b * stride_xb + i * stride_xi + j * stride_xj + k_end * stride_xd,
                    mask=k_mask, other=0.0)
        x_f = tl.float32(x)

        # load weight blocks
        w_l = tl.load(w_left_ptr + k_end[:, None] * stride_wk + offs_h[None, :] * stride_wh,
                      mask=k_mask[:, None] & h_mask[None, :], other=0.0)

"""
custom_kernel: Triton‑accelerated outgoing TriMul (AlphaFold3).

Algorithm
---------
1. Layer‑normalize the 4‑D input (B,N,N,C).
2. Fuse the five linear projections (left, right, left‑gate, right‑gate,
   out‑gate) together with the optional binary mask and sigmoid gates
   in a single Triton kernel.
   - left  = (x @ W_left)  * mask * sigmoid(x @ W_left_gate)
   - right = (x @ W_right) * mask * sigmoid(x @ W_right_gate)
   - out_gate = sigmoid(x @ W_out_gate)
3. Compute the core contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched matrix multiplication (left @ rightᵀ) using cuBLAS.
4. Apply a second LayerNorm over the hidden dimension, multiply by
   out_gate and project back to the original channel dimension.
5. Return the result in the original dtype.

The heavy memory traffic of the five separate linear layers is reduced
to a single pass over the normalized input, yielding a large speed‑up.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _proj_gate_kernel(
    x_ptr,               # const *float16, shape (B,N,N,C)
    left_out_ptr,        # *float16, shape (B,N,N,H)
    right_out_ptr,       # *float16, shape (B,N,N,H)
    out_gate_ptr,        # *float16, shape (B,N,N,H)
    w_left_ptr,          # *float16, shape (H,C)
    w_right_ptr,         # *float16, shape (H,C)
    w_lgate_ptr,         # *float16, shape (H,C)
    w_rgate_ptr,         # *float16, shape (H,C)
    w_outgate_ptr,       # *float16, shape (H,C)
    mask_ptr,            # *float16, shape (B,N,N) (binary, 0/1)
    B, N, C, H,          # tensor dimensions
    stride_xb, stride_xi, stride_xj, stride_xc,   # strides of x
    stride_outb, stride_outi, stride_outj, stride_outh, # strides of outputs
    stride_maskb, stride_maski, stride_maskj,    # strides of mask
    stride_wk, stride_wh,                        # weight strides (k dim, h dim)
    BLOCK_H: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Fused projection + gating + mask.

    Computes:
        left  = (x @ W_left)  * mask * sigmoid(x @ W_left_gate)
        right = (x @ W_right) * mask * sigmoid(x @ W_right_gate)
        out_gate = sigmoid(x @ W_out_gate)
    The result tensors have shape (B,N,N,H).
    """
    # program ids:
    pid0 = tl.program_id(0)   # combines batch and i dimension
    pid1 = tl.program_id(1)   # j dimension
    pid2 = tl.program_id(2)   # hidden block

    # decode batch, i, j
    b = pid0 // N
    i = pid0 % N
    j = pid1

    # hidden‑block offsets
    h_start = pid2 * BLOCK_H
    offs_h = h_start + tl.arange(0, BLOCK_H)
    h_mask = offs_h < H

    # ------------------------------------------------------------
    # Load mask scalar for this (b,i,j)
    # ------------------------------------------------------------
    mask_off = b * stride_maskb + i * stride_maski + j * stride_maskj
    m = tl.load(mask_ptr + mask_off, mask=tl.full([1], True, tl.int1), other=0.0)
    m_f = tl.float32(m)               # cast to fp32 for arithmetic

    # ------------------------------------------------------------
    # Accumulators (fp32)
    # ------------------------------------------------------------
    acc_l = tl.zeros((BLOCK_H,), dtype=tl.float32)
    acc_r = tl.zeros((BLOCK_H,), dtype=tl.float32)
    acc_lg = tl.zeros((BLOCK_H,), dtype=tl.float32)
    acc_rg = tl.zeros((BLOCK_H,), dtype=tl.float32)
    acc_og = tl.zeros((BLOCK_H,), dtype=tl.float32)

    # ------------------------------------------------------------
    # Loop over the input channel dimension C
    # ------------------------------------------------------------
    for k in range(0, C, BLOCK_K):
        k_off = k + tl.arange(0, BLOCK_K)
        k_mask = k_off < C

        # x slice: shape (BLOCK_K,)
        x = tl.load(
            x_ptr
            + b * stride_xb
            + i * stride_xi
            + j * stride_xj
            + k_off * stride_xc,
            mask=k_mask,
            other=0.0,
        )
        x_f = tl.float32(x)  # fp32 for dot

        # weight slices: (BLOCK_K, BLOCK_H)
        w_l = tl.load(
            w_left_ptr
            + k_off[:, None] * stride_wk
            + offs_h[None, :] * stride_wh,
            mask=k_mask[:, None] & h_mask[None, :],
            other=0.0,
        )
        w_r = tl.load(
            w_right_ptr
            + k_off[:, None] * stride_wk
            + offs_h[None, :] * stride_wh,
            mask=k_mask[:, None] & h_mask[None, :],
            other=0.0,
        )
        w_lg = tl.load(
            w_lgate_ptr
            + k_off[:, None] * stride_wk
            + offs_h[None, :] * stride_wh,
            mask=k_mask[:, None] & h_mask[None, :],
            other=0.0,
        )
        w_rg = tl.load(
            w_rgate_ptr
            + k_off[:, None] * stride_wk
            + offs_h[None, :] * stride_wh,
            mask=k_mask[:, None] & h_mask[None, :],
            other=0.0,
        )
        w_og = tl.load(
            w_outgate_ptr
            + k_off[:, None] * stride_wk
            + offs_h[None, :] * stride_wh,
            mask=k_mask[:, None] & h_mask[None, :],
            other=0.0,
        )

        # cast weights to fp32
        w_l_f = tl.float32(w_l)
        w_r_f = tl.float32(w_r)
        w_lg_f = tl.float32(w_lg)
        w_rg_f = tl.float32(w_rg)
        w_og_f = tl.float32(w_og)

        # dot products (result shape (1, BLOCK_H), take row 0)
        acc_l += tl.dot(x_f[:, None], w_l_f)[0]
        acc_r += tl.dot(x_f[:, None], w_r_f)[0]
        acc_lg += tl.dot(x_f[:, None], w_lg_f)[0]
        acc_rg += tl.dot(x_f[:, None], w_rg_f)[0]
        acc_og += tl.dot(x_f[:, None], w_og_f)[0]

    # ------------------------------------------------------------
    # Apply sigmoid gates
    # ------------------------------------------------------------
    sigmoid_lg = 1.0 / (1.0 + tl.exp(-acc_lg))
    sigmoid_rg = 1.0 / (1.0 + tl.exp(-acc_rg))
    sigmoid_og = 1.0 / (1.0 + tl.exp(-acc_og))

    # ------------------------------------------------------------
    # Apply mask and write results (cast back to fp16)
    # ------------------------------------------------------------
    left_val  = acc_l * m_f * sigmoid_lg
    right_val = acc_r * m_f * sigmoid_rg

    left_fp16   = tl.float16(left_val)
    right_fp16  = tl.float16(right_val)
    outgate_fp16 = tl.float16(sigmoid_og)

    out_off = (
        b * stride_outb
        + i * stride_outi
        + j * stride_outj
        + h_start * stride_outh
    )
    tl.store(left_out_ptr + out_off, left_fp16, mask=h_mask)
    tl.store(right_out_ptr + out_off, right_fp16, mask=h_mask)
    tl.store(out_gate_ptr + out_off, outgate_fp16, mask=h_mask)

    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask        : torch.Tensor of shape [B, N, N] or None (binary)
            - weights     : dict of model parameters (float32 tensors)
            - config      : dict with keys
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – if True, ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)  # work in fp16 for the heavy part

    batch_size, seq_len, _, _ = x.shape
    B, N = batch_size, seq_len
    C = dim
    H = hidden_dim

    # ------------------------------------------------------------------
    # 2) Prepare mask (binary 0/1). If no mask or nomask flag, use all‑ones.
    # ------------------------------------------------------------------
    if mask is None or nomask:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Allocate output buffers for the fused projection kernel
    # ------------------------------------------------------------------
    left_out   = torch.empty((B, N, N, H), dtype=torch.float16, device=device)
    right_out  = torch.empty_like(left_out)
    out_gate   = torch.empty_like(left_out)

    # ------------------------------------------------------------------
    # 4) Cast linear weights to fp16
    # ------------------------------------------------------------------
    w_left   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)   # (H, C)
    w_right  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    w_lgate  = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    w_rgate  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    w_outgate = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 5) Strides needed by the Triton kernel
    # ------------------------------------------------------------------
    stride_xb, stride_xi, stride_xj, stride_xc = x.stride()
    stride_outb, stride_outi, stride_outj, stride_outh = left_out.stride()
    stride_maskb, stride_maski, stride_maskj = mask_tensor.stride()
    stride_wk = w_left.stride(1)          # stride over C (inner dim) -> 1
    stride_wh = w_left.stride(0)          # stride over H (outer dim) -> C

    # ------------------------------------------------------------------
    # 6) Launch the fused projection + gating kernel
    # ------------------------------------------------------------------
    BLOCK_H = 64   # hidden‑dim tile
    BLOCK_K = 64   # channel‑dim tile (C ≤ 384, so ≤ 6 iterations)

    grid = (
        B * N,          # (b,i) combined dimension
        N,              # j dimension
        (H + BLOCK_H - 1) // BLOCK_H,  # hidden blocks
    )
    _proj_gate_kernel[grid](
        x,
        left_out,
        right_out,
        out_gate,
        w_left,
        w_right,
        w_lgate,
        w_rgate,
        w_outgate,
        mask_tensor,
        B, N, C, H,
        stride_xb, stride_xi, stride_xj, stride_xc,
        stride_outb, stride_outi, stride_outj, stride_outh,
        stride_maskb, stride_maski, stride_maskj,
        stride_wk, stride_wh,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # Ensure kernel completion before next ops
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 7) Core reduction: batched GEMM (left @ rightᵀ)
    #    left_out/right_out : (B,N,N,H) -> (B*H, N, N)
    # ------------------------------------------------------------------
    left_flat  = left_out.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right_out.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(-2, -1))  # (B*H, N, N)

    # ------------------------------------------------------------------
    # 8) Reshape back to (B,N,N,H)
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float16),
    )

    # ------------------------------------------------------------------
    # 10) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate.to(out.dtype)

    # ------------------------------------------------------------------
    # 11) Final linear projection back to original dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 12) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
