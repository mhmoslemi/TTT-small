"""
TriMul outgoing operator (AlphaFold3 style) – Triton‑accelerated forward pass.

Algorithm
---------
1. Input LayerNorm (dim) → FP16.
2. Linear projections: left_proj, right_proj (to hidden_dim).
3. Gate projections: left_gate, right_gate, out_gate (sigmoid).
4. Fuse mask (if present) with projection & gate using a tiny Triton kernel.
5. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (left @ rightᵀ) with batch = B·hidden_dim.
   Implemented via torch.bmm which leverages cuBLAS/Tensor‑cores.
6. Output LayerNorm (hidden_dim) → multiply by out_gate.
7. Final linear projection back to original dim.
8. Cast result back to the input dtype (usually float32).

Only the mask‑gate fusion is implemented in Triton; the heavy O(N³) reduction
uses cuBLAS for maximal throughput on an NVIDIA B200.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) with mask broadcast over H
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # *fp16  total_elems
    gate_ptr,          # *fp16  total_elems
    mask_ptr,          # *fp16  mask_elems = B*N*N
    out_ptr,           # *fp16  total_elems
    total_elems,       # i64
    hidden_dim,        # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # mask index = floor(offs / hidden_dim) because mask is (B,N,N)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Fuse projection, gate and mask (broadcast over hidden dim) with Triton.

    All tensors must be fp16, contiguous and on the same CUDA device.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    # flatten mask (B,N,N) → 1‑D, the kernel will broadcast it
    mask_flat = mask.contiguous().view(-1)

    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],   # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (LayerNorm, Linear weights)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over channel dim C)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the compute‑heavy stages
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Load all linear weights as fp16 (biases are None in the model)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections to hidden_dim
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Fuse mask (if any) with projection & gate via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 on the same device
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM (left @ rightᵀ) over hidden_dim
    #    Shape transformation:
    #        left, right : (B, N, N, H)
    #        → (B*H, N, N) for torch.bmm
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # Bring hidden_dim next to batch and flatten
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiply: (B*H, N, N) @ (B*H, N, N)ᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm over hidden_dim, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)          # back to (B, N, N, C)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
