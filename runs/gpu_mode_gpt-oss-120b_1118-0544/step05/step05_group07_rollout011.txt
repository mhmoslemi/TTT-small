```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Compute mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    # proj, gate: fp16 contiguous of same shape (B,N,N,H)
    # mask: fp16 contiguous shape (B,N,N)
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out
```

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.
    Implements:
      1. LayerNorm on input.
      2. Single fused linear projection for left/right projections
         and three gates.
      3. Sigmoid on gates.
      4. Optional mask & gate fusion using a Triton kernel.
      5. Core reduction via batched GEMM (left @ right^T).
      6. Second LayerNorm, output gate, final linear projection.
    All compute‑intensive kernels run in fp16; result is cast back to input dtype.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast to fp16 for performance
    x = x.to(torch.float16)

    # Prepare combined weight for five linear transforms
    combined_weight = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)  # shape (5*H, dim)

    # Single linear projection (B, N, N, 5*H)
    combined = torch.nn.functional.linear(x, combined_weight)  # fp16

    # Split into individual tensors
    H = hidden_dim
    left_proj = combined[..., 0:H]
    right_proj = combined[..., H:2*H]
    left_gate_raw = combined[..., 2*H:3*H]
    right_gate_raw = combined[..., 3*H:4*H]
    out_gate_raw = combined[..., 4*H:5*H]

    # Sigmoid activations for gates
    left_gate = left_gate_raw.sigmoid()
    right_gate = right_gate_raw.sigmoid()
    out_gate = out_gate_raw.sigmoid()

    # Apply mask and gate fusion using Triton if needed
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Core reduction: batched GEMM over hidden dimension
    # reshape to (B*H, N, N)
    batch, N, _, _ = left.shape
    left_reshaped = left.permute(0, 3, 1, 2).contiguous().view(batch * hidden_dim, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(batch * hidden_dim, N, N)

    # Compute left @ right^T for each (b, d)
    out_reshaped = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_reshaped.view(batch, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Second LayerNorm on hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast back to input dtype (float32)
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) – forward pass.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) using the provided
   weight/bias.
2. Cast to fp16 for the compute‑intensive part.
3. Fuse the five linear transformations (left/right projections and three
   gates) into a single matmul by concatenating their weight matrices.
4. Split the result, apply sigmoid to the three gate tensors.
5. If a mask is present, fuse mask, gate and projection using a tiny
   Triton kernel (`_gate_mask_kernel`) that computes
        out = proj * sigmoid(gate) * mask
   for the left and right branches.
6. Core O(N³·H) reduction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
   This is performed as a batched GEMM of shape (B*H, N, N) via
   `torch.bmm`, which leverages cuBLAS/Tensor‑core.
7. Apply a second LayerNorm on the hidden dimension, multiply by the
   output gate, and finally project back to the original channel dimension.
8. Cast the result back to the input dtype (float32).

All heavy element‑wise work (mask & gate fusion) is executed in Triton,
while the large matrix multiplications use PyTorch's highly‑optimized
kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _gate_mask_kernel(
    proj_ptr,   # fp16 * (B*N*N*H,)
    gate_ptr,   # fp16 * (B*N*N*H,)
    mask_ptr,   # fp16 * (B*N*N,)
    out_ptr,    # fp16 * (B*N*N*H,)
    total_elems,    # i32 total number of elements in proj/gate
    hidden_dim,     # i32 H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask is broadcasted over the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using Triton.

    All tensors must be fp16, contiguous and on the same CUDA device.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for all heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Fuse five linear transforms into a single matmul
    # ------------------------------------------------------------------
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)  # (5*H, C)

    combined = torch.nn.functional.linear(x, combined_weight)  # (B,N,N,5*H)

    # ------------------------------------------------------------------
    # 4) Split and apply sigmoid to gate tensors
    # ------------------------------------------------------------------
    H = hidden_dim
    left_proj   = combined[..., 0:H]
    right_proj  = combined[..., H:2 * H]
    left_gate_r = combined[..., 2 * H:3 * H]
    right_gate_r = combined[..., 3 * H:4 * H]
    out_gate_r  = combined[..., 4 * H:5 * H]

    left_gate   = left_gate_r.sigmoid()
    right_gate  = right_gate_r.sigmoid()
    out_gate    = out_gate_r.sigmoid()

    # ------------------------------------------------------------------
    # 5) Optional mask & gate fusion via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core O(N³·H) reduction using batched GEMM
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*H, N, N)
    left_reshaped = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    out_reshaped = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = (
        out_reshaped.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 7) Second LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float16),
    )
    out = out * out_gate

    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(device=device, dtype=torch.float16),
    )  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
