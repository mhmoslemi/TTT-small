@triton.jit
def triton_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr, # maybe for prefetch
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # pointers for this batch
    a_ptrs = A_ptr + pid_batch * batch_stride_a + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B_ptr + pid_batch * batch_stride_b + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=offs_m < M and tl.arange(0, BLOCK_K) < K, other=0.0)
        b = tl.load(b_ptrs, mask=tl.arange(0, BLOCK_K) < K and offs_n < N, other=0.0)
        acc += tl.dot(a, b)  # dot: shape (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c_ptrs = C_ptr + pid_batch * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)
```

def triton_matmul(A, B, BLOCK_M=64, BLOCK_N=64, BLOCK_K=32):
    # A: (batch, M, K)
    # B: (batch, K, N)
    batch, M, K = A.shape
    _, _, N = B.shape
    out = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)
    grid = (batch, (M + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    # compute strides
    stride_a_batch = A.stride(0)
    stride_a_m = A.stride(1)
    stride_a_k = A.stride(2)
    stride_b_batch = B.stride(0)
    stride_b_k = B.stride(1)
    stride_b_n = B.stride(2)
    stride_out_batch = out.stride(0)
    stride_out_m = out.stride(1)
    stride_out_n = out.stride(2)
    matmul_kernel[grid](
        A, B, out,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_out_batch, stride_out_m, stride_out_n,
        M, N, K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out
```

@triton.jit
def matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # create mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for A and B
    a_ptrs = A_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_ptrs = B_ptr + pid_batch * stride_b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    c_ptrs = C_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = mask_m[:, None] & mask_n[None, :]

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])
    # Cast to fp16
    x = x.to(torch.float16)

    # Cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Projections
    left = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply mask and gate
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * left_gate * mask_fp16
        right = right * right_gate * mask_fp16
    else:
        left = left * left_gate
        right = right * right_gate

    B, N, _, H = left.shape

    # reshape for batched GEMM
    # combine batch and head dimensions
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B*H,N,N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # transpose right for gemm
    right_t = right_flat.permute(0, 2, 1).contiguous()

    # perform batched matmul using Triton
    out_flat = triton_matmul(left_flat, right_t, BLOCK_M=64, BLOCK_N=64, BLOCK_K=32)

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # Output LayerNorm
    out = F.layer_norm(out,
                       (H,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # Apply out_gate (broadcast)
    out = out * out_gate

    # Final linear projection
    out = F.linear(out, to_out_w)

    # Cast back to original dtype (usually float32)
    out = out.to(input_tensor.dtype)

    return out

def triton_matmul(A, B, BLOCK_M=64, BLOCK_N=64, BLOCK_K=32):
    # expects A: (batch, M, K) ; B: (batch, K, N)
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, _, N = B.shape
    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    grid = (batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # compute strides in elements
    stride_a_batch = A.stride(0)
    stride_a_m = A.stride(1)
    stride_a_k = A.stride(2)
    stride_b_batch = B.stride(0)
    stride_b_k = B.stride(1)
    stride_b_n = B.stride(2)
    stride_c_batch = C.stride(0)
    stride_c_m = C.stride(1)
    stride_c_n = C.stride(2)

    matmul_kernel[grid](
        A, B, C,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        M, N, K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C

@triton.jit
def matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute pointer offsets for A and B
    a_ptrs = A_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_ptrs = B_ptr + pid_batch * stride_b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_valid = tl.arange(0, BLOCK_K) < K
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_valid[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # write back results
    c_ptrs = C_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated core reduction.

The implementation follows the reference PyTorch TriMul but replaces the
O(N³·H) reduction with a custom Triton kernel that performs a batched
matrix multiplication:
    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]

The kernel tiles the output (i, j) plane (BLOCK_M×BLOCK_N) and iterates
over the reduction dimension k in blocks of size BLOCK_K.  Accumulation is
performed in FP32 for numerical stability and the result is stored in
FP16.  All remaining operations (LayerNorm, linear projections, gating,
and final projection) are done with PyTorch (using FP16 where possible).

The `custom_kernel` entry point expects:
    data = (input_tensor, mask, weights, config)
and returns a tensor of shape [B, N, N, dim] in the original dtype.
"""

import torch.nn.functional as F
import triton
import triton.language as tl

# Triton batched GEMM kernel (A @ Bᵀ) for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _triton_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  where
       A: (batch, M, K),  B: (batch, N, K) (i.e. Bᵀ has shape (batch, K, N)).
    """
    pid_batch = tl.program_id(0)      # batch index
    pid_m = tl.program_id(1)          # row tile
    pid_n = tl.program_id(2)          # col tile

    # Offsets within the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current tile in A and B
    a_ptrs = (
        A_ptr
        + pid_batch * stride_a_batch
        + offs_m[:, None] * stride_a_m
        + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    )
    b_ptrs = (
        B_ptr
        + pid_batch * stride_b_batch
        + tl.arange(0, BLOCK_K)[:, None] * stride_b_k
        + offs_n[None, :] * stride_b_n
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        k_valid = tl.arange(0, BLOCK_K) < K

        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & k_valid[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=k_valid[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Write the result back to C
    c_ptrs = (
        C_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


def _triton_batched_gemm(A: torch.Tensor, B: torch.Tensor,
                         BLOCK_M=128, BLOCK_N=128, BLOCK_K=32) -> torch.Tensor:
    """
    Batched GEMM using the above Triton kernel.
    A: (batch, M, K)   fp16, contiguous
    B: (batch, N, K)   fp16, contiguous (will be interpreted as Bᵀ)
    Returns C: (batch, M, N) fp16
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, _ = B.shape
    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Strides in element (not byte) units
    stride_a_batch = A.stride(0)
    stride_a_m = A.stride(1)
    stride_a_k = A.stride(2)

    stride_b_batch = B.stride(0)
    stride_b_k = B.stride(1)      # because B is (batch, N, K)
    stride_b_n = B.stride(2)

    stride_c_batch = C.stride(0)
    stride_c_m = C.stride(1)
    stride_c_n = C.stride(2)

    _triton_gemm_kernel[grid](
        A, B, C,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        M, N, K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict of model parameters
            - config: dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (FP32 for stability)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑intensive parts
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all trainable weights to FP16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections
    # --------------------------------------------------------------
    left = F.linear(x_fp16, left_proj_w)   # (B, N, N, H)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    # --------------------------------------------------------------
    # 6. Apply mask (if any) and gates
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * left_gate * mask_fp16
        right = right * right_gate * mask_fp16
    else:
        left = left * left_gate
        right = right * right_gate

    B, N, _, H = left.shape  # (B, N, N, H)

    # --------------------------------------------------------------
    # 7. Prepare tensors for batched GEMM: (B*H, N, N)
    # --------------------------------------------------------------
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (batch, M, K)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (batch, N, K)

    # Transpose the second operand so that we compute A @ Bᵀ
    right_t = right_flat.permute(0, 2, 1).contiguous()  # (batch, K, N)

    # --------------------------------------------------------------
    # 8. Core O(N³·H) reduction via Triton batched GEMM
    # --------------------------------------------------------------
    out_flat = _triton_batched_gemm(left_flat, right_t)  # (B*H, N, N)

    # --------------------------------------------------------------
    # 9. Reshape back to (B, N, N, H)
    # --------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 10. Output LayerNorm, output‑gate and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 11. Cast back to original dtype (usually float32)
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
