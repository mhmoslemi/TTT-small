# Triton kernel
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid = tl.program_id(0)  # combined batch and hidden dimension
    pid_i = tl.program_id(1)  # block row index (i)
    pid_j = tl.program_id(2)  # block col index (j)

    # decode batch and hidden index
    b = pid // H
    h = pid % H

    # compute global row and col offsets
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # range vectors
    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)

    # masks for boundary
    i_mask = i < N
    j_mask = j < N

    # base pointers for this (b,h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # load left block: shape (BLOCK_M, BLOCK_K)
        left_offset = i[:, None] * stride_left_i + k[None, :] * stride_left_k
        left = tl.load(left_base + left_offset,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0)
        left = left.to(tl.float32)

        # load right block: shape (BLOCK_N, BLOCK_K)
        right_offset = j[:, None] * stride_right_j + k[None, :] * stride_right_k
        right = tl.load(right_base + right_offset,
                        mask=j_mask[:, None] & k_mask[None, :],
                        other=0.0)
        right = right.to(tl.float32)

        # accumulate product: left (M,K) dot right.T (K,N) -> (M,N)
        acc += tl.dot(left, tl.trans(right))

    # store result
    out_offset = i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_base + out_offset,
             acc.to(tl.float16),
             mask=i_mask[:, None] & j_mask[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # Cast to float16 for compute (optional)
    x_fp16 = x.to(torch.float16)

    # Cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Projections
    left_proj = F.linear(x_fp16, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x_fp16, right_proj_w)

    # Gate values
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    # Apply gate and optional mask
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()

    B,N,N_check,H = left.shape
    assert N == N_check, "non-square input not supported"
    # Allocate output
    out = torch.empty_like(left)   # same dtype (fp16)

    # Determine block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute strides
    left_stride_b, left_stride_i, left_stride_k, left_stride_h = left.stride()
    right_stride_b, right_stride_j, right_stride_k, right_stride_h = right.stride()
    out_stride_b, out_stride_i, out_stride_j, out_stride_h = out.stride()

    # Grid dimensions
    grid = (B * H,
            math.ceil(N / BLOCK_M),
            math.ceil(N / BLOCK_N))

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        left_stride_b, left_stride_i, left_stride_k, left_stride_h,
        right_stride_b, right_stride_j, right_stride_k, right_stride_h,
        out_stride_b, out_stride_i, out_stride_j, out_stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # out currently shape (B,N,N,H)
    # Apply output layer norm (on hidden dim)
    out = F.layer_norm(out,
                       (H,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # Cast back to original dtype
    out = out.to(dtype)

    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Apply LayerNorm on the 4‑D input (B, N, N, C).
2. Cast to FP16 for the compute‑intensive part.
3. Linear projections (left/right) and three gating projections are performed
   with torch.nn.functional.linear (FP16 weights) and a sigmoid.
4. Optional mask is broadcasted and fused with the projections and gates.
5. Core reduction:
       out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   is implemented as a custom Triton kernel that performs a batched GEMM
   over the (i, j) plane for each (batch, head) pair, accumulating in FP32.
6. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate and project back to the original channel dimension.
7. Cast the result back to the original dtype (usually float32).

The heavy O(N³·H) reduction is the only part written in Triton; all other
operations use PyTorch for simplicity.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h].

    Grid:
        pid0  – combined batch and hidden dimension (b * H + h)
        pid1  – block row index i
        pid2  – block column index j
    """
    pid = tl.program_id(0)       # combined (b, h)
    pid_i = tl.program_id(1)     # block row
    pid_j = tl.program_id(2)     # block column

    # Decode batch and hidden indices
    b = pid // H
    h = pid % H

    # Base offsets for this (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # Global i / j indices for this block
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    # Boundary masks
    i_mask = i < N
    j_mask = j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = kk * BLOCK_K
        k = k0 + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # Load left block: shape (BLOCK_M, BLOCK_K)
        left_offset = i[:, None] * stride_left_i + k[None, :] * stride_left_k
        left = tl.load(left_base + left_offset,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0)
        left = left.to(tl.float32)

        # Load right block: shape (BLOCK_N, BLOCK_K)
        right_offset = j[:, None] * stride_right_j + k[None, :] * stride_right_k
        right = tl.load(right_base + right_offset,
                        mask=j_mask[:, None] & k_mask[None, :],
                        other=0.0)
        right = right.to(tl.float32)

        # Accumulate dot product (M×K)·(K×N) → (M×N)
        acc += tl.dot(left, tl.trans(right))

    # Store the result (cast back to FP16)
    out_offset = i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_base + out_offset,
             acc.to(tl.float16),
             mask=i_mask[:, None] & j_mask[None, :])

    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B, N, N, C) -> (B, N, N, H)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6. Fuse mask, projection and gate (mask is optional)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B, N, N) -> (B, N, N, 1) for broadcasting
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguity for Triton
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction using Triton
    # ------------------------------------------------------------------
    out = torch.empty_like(left)  # (B, N, N, H), fp16

    # Block sizes – tuned for typical GPU architectures
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides for pointer arithmetic
    lsb, lsi, lsk, lsh = left.stride()
    rsb, rsj, rsk, rsh = right.stride()
    osb, osi, osj, osh = out.stride()

    # Grid: (B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        B * H,
        math.ceil(N / BLOCK_M),
        math.ceil(N / BLOCK_N),
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        lsb,
        lsi,
        lsk,
        lsh,
        rsb,
        rsj,
        rsk,
        rsh,
        osb,
        osi,
        osj,
        osh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)      # (B, N, N, C)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)
    return out
