```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Cast to fp16
    x = x.to(torch.float16)

    # combine weights
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    # shape (out_features, in_features). We'll concat along out_features.
    combined_weight = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0)  # (5*hidden_dim, dim)
    combined_weight = combined_weight.to(torch.float16)

    # Linear projection: flatten input
    B, N, _, _ = x.shape
    x_flat = x.view(B * N * N, dim)  # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_weight)  # (B*N*N, 5*hidden_dim)
    # reshape
    combined_out = combined_out.view(B, N, N, 5 * hidden_dim)

    # split
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(combined_out, hidden_dim, dim=3)

    # gating
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # Apply mask and gating for left and right
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        # Use Triton kernel to compute left = left_proj * left_gate * mask; similarly for right
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # core reduction: (B*H, N, N) matmul
    B, N, _, H = left.shape
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out * out_gate
    # final linear
    to_out_w = weights["to_out.weight"].to(torch.float16)
    # Convert out to fp16 for linear
    out = out.to(torch.float16)
    out = F.linear(out, to_out_w)  # (B,N,N,dim)
    # cast back to original dtype
    out = out.to(dtype)

    return out
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated elementwise mask/gate fusion
and a single‑call fused linear projection for all five weight matrices.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Cast to fp16 for compute‑heavy kernels.
3. Concatenate the five weight matrices (left_proj, right_proj,
   left_gate, right_gate, out_gate) along the output‑feature axis and
   perform *one* torch.nn.functional.linear call → shape (B,N,N,5*H).
   Split the result into the five tensors.
4. Apply sigmoid to the three gate tensors.
5. Fuse (proj * gate * mask) for `left` and `right` with a tiny Triton kernel
   (mask is broadcast over the hidden dimension).  If no mask is given the
   multiplication is performed with pure PyTorch.
6. Batched matrix multiplication per head:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented via torch.bmm on tensors reshaped to (B*H,N,N).
7. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to the original channel dimension.
8. Cast the result back to the input dtype (normally float32).

The kernel in step 5 satisfies the requirement to use Triton code.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: proj * gate * mask  (mask broadcast over hidden_dim)
# --------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,      # fp16*   (B*N*N*H,)
    gate_ptr,      # fp16*   (B*N*N*H,)
    mask_ptr,      # fp16*   (B*N*N,)
    out_ptr,       # fp16*   (B*N*N*H,)
    total_elems,   # i32     total number of elements = B*N*N*H
    hidden_dim,    # i32     H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # index of the corresponding mask element (broadcast over H)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using Triton. All tensors must be fp16 and
    contiguous.  `mask` has shape (B,N,N) and is broadcast over the hidden
    dimension of `proj`/`gate`.
    """
    assert proj.shape == gate.shape
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict containing all model parameters (float32)
            - config: dict with keys "dim" (C) and "hidden_dim" (H);
                      optional key "nomask" (bool).

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Fuse the five linear projections into a single matmul
    # --------------------------------------------------------------
    # Gather and cast weights (out_features, in_features) -> fp16
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Concatenate along the output‑feature axis: (5*H, dim)
    combined_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    ).to(torch.float16)

    B, N, _, _ = x.shape
    x_flat = x.view(B * N * N, dim)                     # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_weight)   # (B*N*N, 5*H)
    combined_out = combined_out.view(B, N, N, 5 * hidden_dim)

    # --------------------------------------------------------------
    # 4. Split the combined result into its five components
    # --------------------------------------------------------------
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        combined_out, hidden_dim, dim=3
    )

    # --------------------------------------------------------------
    # 5. Apply sigmoid to the three gate tensors
    # --------------------------------------------------------------
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # --------------------------------------------------------------
    # 6. Fuse mask (if any) and gating for `left` and `right`
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16, contiguous and broadcastable
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # final projection back to `dim`
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = out.to(torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9. Cast back to the original dtype
    # --------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
