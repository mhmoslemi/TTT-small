@triton.jit
def batch_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, N,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute number of blocks along M and N
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    # derive indices
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    block_i = block_idx // num_blocks_n
    block_j = block_idx % num_blocks_n

    # bounds
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # mask for valid rows/cols
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # compute linear offsets
        a_ptr = left_ptr + batch_idx * N * N + offs_i[:, None] * N + offs_k[None, :]
        b_ptr = right_ptr + batch_idx * N * N + offs_k[:, None] * N + offs_j[None, :]

        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)  # fp16 dot -> fp32 accumulation

    # store result
    c_ptr = out_ptr + batch_idx * N * N + offs_i[:, None] * N + offs_j[None, :]
    mask_store = (offs_i < N)[:, None] & (offs_j < N)[None, :]
    tl.store(c_ptr, acc.to(tl.float16), mask=mask_store)

@triton.jit
def gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # mask index: include broadcasting: each hidden dim element corresponds to same mask element across hidden_dim per position
    mask_idx = offs // hidden_dim
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑normalize the input tensor.
2. Cast to fp16 for compute‑intensive work.
3. Perform linear projections (left/right) and their gates.
4. Fuse projection, gate and optional mask with a tiny Triton kernel.
5. Compute the core O(N³·H) reduction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
   Implemented as a batched GEMM (B·H independent N×N matmuls) in Triton.
6. Apply a second LayerNorm, multiply by the output gate and project back.
7. Return the result in the original dtype (float32).

Both the element‑wise gate‑mask fusion and the batched matrix multiplication
are performed with Triton kernels, satisfying the requirement of having
part of the work in custom kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask (element‑wise)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (total_elems,)
    gate_ptr,          # fp16 * (total_elems,)
    mask_ptr,          # fp16 * (num_positions,)
    out_ptr,           # fp16 * (total_elems,)
    total_elems,       # i32  – total number of elements in proj/gate
    hidden_dim,        # i32  – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # each hidden_dim entries share the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse proj * gate * mask using Triton.
    All tensors must be on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],               # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _batch_matmul_kernel(
    left_ptr,   # fp16 * (batch, N, N)
    right_ptr,  # fp16 * (batch, N, N)
    out_ptr,    # fp16 * (batch, N, N)
    batch,      # i32 – total number of batches (B * H)
    N,          # i32 – sequence length
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # grid layout: (batch, num_blocks_m, num_blocks_n) flattened
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch_idx = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    block_i = block_idx // num_blocks_n
    block_j = block_idx % num_blocks_n

    # offsets within the tile
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointer for this batch (contiguous layout)
    base = batch_idx * N * N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # masks for out‑of‑bounds loads
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # left tile: (M, K)
        a_ptr = left_ptr + base + offs_i[:, None] * N + offs_k[None, :]
        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile transposed: (K, N) -> we load (K, N) directly
        b_ptr = right_ptr + base + offs_k[:, None] * N + offs_j[None, :]
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product (fp16 inputs produce fp32 accumulation)
        acc += tl.dot(a, b)

    # store result back (cast to fp16)
    c_ptr = out_ptr + base + offs_i[:, None] * N + offs_j[None, :]
    store_mask = (offs_i < N)[:, None] & (offs_j < N)[None, :]
    tl.store(c_ptr, acc.to(tl.float16), mask=store_mask)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> fp16)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(
        x, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, H]
    right_proj = torch.nn.functional.linear(
        x, weights["right_proj.weight"].to(torch.float16)
    )

    # ------------------------------------------------------------------
    # 3. Gates (fp16, sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4. Fuse projection, gate and optional mask with Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³·H) reduction via batched GEMM in Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Move hidden dim to batch axis: (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch_h = B * H
    left_flat = left_perm.view(batch_h, N, N)
    right_flat = right_perm.view(batch_h, N, N)

    out_flat = torch.empty_like(left_flat)

    # Choose tile sizes (product <= 1024 threads per program)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch_h * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    _batch_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_h,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Second LayerNorm, output gate and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gate

    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float16)
    )  # (B, N, N, C)

    # ------------------------------------------------------------------
    # 7. Cast back to original dtype (float32)
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
