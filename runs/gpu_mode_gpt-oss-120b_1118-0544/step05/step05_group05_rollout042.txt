def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Input layernorm in fp32 for stability
    norm_weight = weights["norm.weight"].to(device).to(torch.float32)
    norm_bias = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    x = x.to(torch.float16)

    # Linear projections
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left = F.linear(x, left_proj_w)   # shape (B,N,N,hidden)
    right = F.linear(x, right_proj_w)

    # Gates
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Optional mask
    has_mask = mask is not None and not config.get("nomask", False)
    if has_mask:
        mask = mask.to(device).to(torch.float16)
    # fuse mask+gate multiplication for left/right via Triton
    _gate_mask_fuse_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask if has_mask else None,
            total_elems=left.numel(),
            hidden_dim=hidden_dim,
            HAS_MASK=has_mask,
            BLOCK_SIZE=BLOCK,
    )
    # final left,right ready
    # free gate tensors (optional)
    del left_gate, right_gate

    # Core contraction (batched GEMM)
    # reshape to (B*H,N,N) for both left and right
    left_mat = left.permute(0,3,1,2).contiguous().view(B*hidden_dim, N, N)
    right_mat = right.permute(0,3,1,2).contiguous().view(B*hidden_dim, N, N)

    # Perform batched matmul: left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # Apply output LayerNorm + out_gate (fused)
    # We will use a kernel that does layernorm + multiply by out_gate in place
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)

    # define grid for fused kernel
    total_ijk = B * N * N
    BLOCK_H = 128 if hidden_dim <= 128 else 384 # but we need compile-time const; set 384 (max)
    grid = lambda meta: (total_ijk,)

    _layernorm_gate_fuse_kernel[grid](
         out_ptr=out,
         out_gate_ptr=out_gate,
         gamma_ptr=to_out_norm_weight,
         beta_ptr=to_out_norm_bias,
         stride_out_b0=out.stride(0),
         stride_out_b1=out.stride(1),
         stride_out_b2=out.stride(2),
         stride_out_b3=out.stride(3),
         stride_gate_b0=out_gate.stride(0),
         stride_gate_b1=out_gate.stride(1),
         stride_gate_b2=out_gate.stride(2),
         stride_gate_b3=out_gate.stride(3),
         B=B, N=N, H=hidden_dim,
         EPS=1e-5,
         BLOCK_H=BLOCK_H,
    )
    # Now out is normalized and gated.

    # Final linear
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    # flatten for linear
    out_flat = out.view(B * N * N, hidden_dim)
    result_flat = F.linear(out_flat, to_out_w)  # bias None
    result = result_flat.view(B, N, N, -1)

    # Cast back to original dtype
    if result.dtype != dtype:
        result = result.to(dtype)
    return result

def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr
):

    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # mask index: element index // hidden_dim (since mask broadcast over hidden dim)
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

The kernel performs:
1. Input LayerNorm (float32 for stability, then fp16 for compute).
2. Linear projections (left/right) and gating projections.
3. Fused Triton kernel that applies mask (if any) and multiplies by sigmoid gates,
   in‑place on left and right tensors.
4. Core O(N³·H) contraction via a strided batched GEMM (left @ rightᵀ) in fp16.
5. Fused Triton kernel that applies the output LayerNorm, the output gate,
   and writes the result back in‑place.
6. Final linear projection back to the original feature dimension.

All heavy arithmetic is performed in fp16; the result is cast back to the
input dtype (usually float32). Two Triton kernels are used to reduce memory
traffic and improve runtime.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape as left/right
    mask_ptr,                          # [B,N,N] flattened (broadcasted)
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left      = tl.load(left_ptr      + offs, mask=active, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=active, other=0.0)

    if HAS_MASK:
        mask_idx = offs // hidden_dim                # each hidden_dim entries share a mask entry
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)

@triton.jit
def _layernorm_gate_fuse_kernel(
    out_ptr, out_gate_ptr,          # [B,N,N,H] flattened
    gamma_ptr, beta_ptr,            # [H]
    stride_out_b0, stride_out_b1, stride_out_b2, stride_out_b3,
    stride_gate_b0, stride_gate_b1, stride_gate_b2, stride_gate_b3,
    B, N, H,
    EPS: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # Decode (b,i,j) from linear pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # Thread‑local hidden index
    h = tl.arange(0, BLOCK_H)
    mask = h < H

    # Compute base pointers
    out_base   = out_ptr   + b * stride_out_b0   + i * stride_out_b1   + j * stride_out_b2
    out_off    = out_base   + h * stride_out_b3

    gate_base  = out_gate_ptr + b * stride_gate_b0 + i * stride_gate_b1 + j * stride_gate_b2
    gate_off   = gate_base + h * stride_gate_b3

    # Load data
    val   = tl.load(out_off,  mask=mask, other=0.0)
    gate  = tl.load(gate_off, mask=mask, other=0.0)
    gamma = tl.load(gamma_ptr + h, mask=mask, other=0.0)
    beta  = tl.load(beta_ptr  + h, mask=mask, other=0.0)

    # --- LayerNorm over the hidden dimension ---
    sum_val = tl.sum(val,  axis=0)
    sum_sq  = tl.sum(val * val, axis=0)
    mean    = sum_val / H
    var     = sum_sq / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    normed = (val - mean) * inv_std
    # Apply scale+bias and output gate
    out = (normed * gamma + beta) * gate

    # Store result in‑place
    tl.store(out_off, out, mask=mask)

    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)

            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    import math
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32) → fp16 for compute
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections and gates (all fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left      = F.linear(x, left_proj_w)          # (B, N, N, H)
    right     = F.linear(x, right_proj_w)         # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 3. Fuse mask and gate multiplication (in‑place) via Triton
    # -----------------------------------------------------------------
    has_mask = (mask is not None) and (not nomask)
    if has_mask:
        mask_fp = mask.to(device).to(torch.float16)          # (B,N,N)
    else:
        mask_fp = torch.empty((1,), device=device, dtype=torch.float16)  # dummy

    total = left.numel()
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

    _gate_mask_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_fp,
        total_elems=total,
        hidden_dim=hidden_dim,
        HAS_MASK=has_mask,
        BLOCK_SIZE=BLOCK,
    )
    # left/right now contain masked & gated values
    # -----------------------------------------------------------------
    # 4. Core contraction: batched GEMM (left @ rightᵀ)
    # -----------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matmul (left @ rightᵀ)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5. Output LayerNorm + output gate (fused)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    total_ijk = B * N * N
    grid2 = (total_ijk,)

    _layernorm_gate_fuse_kernel[grid2](
        out_ptr=out,
        out_gate_ptr=out_gate,
        gamma_ptr=to_out_norm_w,
        beta_ptr=to_out_norm_b,
        stride_out_b0=out.stride(0),
        stride_out_b1=out.stride(1),
        stride_out_b2=out.stride(2),
        stride_out_b3=out.stride(3),
        stride_gate_b0=out_gate.stride(0),
        stride_gate_b1=out_gate.stride(1),
        stride_gate_b2=out_gate.stride(2),
        stride_gate_b3=out_gate.stride(3),
        B=B, N=N, H=hidden_dim,
        EPS=1e-5,
        BLOCK_H=384,   # compile‑time max hidden dimension
    )

    # -----------------------------------------------------------------
    # 6. Final linear projection back to original dimension
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)  # (C, H)
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)
    return out

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

The algorithm:
1. Input LayerNorm (float32 for numerical stability) → cast to fp16.
2. Linear projections (left/right) and gating projections (sigmoid) in fp16.
3. Triton kernel fuses mask (if present) and gate multiplication, updating
   left and right in‑place.
4. Core O(N³·H) contraction using a strided batched GEMM:
   out_{b,i,j,:} = Σ_k left_{b,i,k,:} * right_{b,j,k,:}.
5. Triton kernel applies the output LayerNorm, multiplies by the output gate,
   and writes the result back in‑place.
6. Final linear projection back to the original feature dimension.
All heavy arithmetic runs in fp16; the result is cast back to the input dtype.
Two Triton kernels are used to reduce memory traffic and improve runtime.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask (optional) and sigmoid gates into left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape as left/right
    mask_ptr,                          # [B,N,N] flattened (broadcasted)
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left      = tl.load(left_ptr      + offs, mask=active, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden_dim entries share one mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


# ----------------------------------------------------------------------
# Kernel 2: fused output LayerNorm + output gate (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_gate_fuse_kernel(
    out_ptr, out_gate_ptr,          # [B,N,N,H] flattened
    gamma_ptr, beta_ptr,            # [H]
    stride_out_b0, stride_out_b1, stride_out_b2, stride_out_b3,
    stride_gate_b0, stride_gate_b1, stride_gate_b2, stride_gate_b3,
    B, N, H,
    EPS: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # Decode (b,i,j) from linear pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # Thread‑local hidden index
    h = tl.arange(0, BLOCK_H)
    mask = h < H

    # Base pointers for the current (b,i,j) slice
    out_base   = out_ptr   + b * stride_out_b0   + i * stride_out_b1   + j * stride_out_b2
    out_off    = out_base   + h * stride_out_b3

    gate_base  = out_gate_ptr + b * stride_gate_b0 + i * stride_gate_b1 + j * stride_gate_b2
    gate_off   = gate_base + h * stride_gate_b3

    # Load values
    val   = tl.load(out_off,  mask=mask, other=0.0)          # fp16
    gate  = tl.load(gate_off, mask=mask, other=0.0)          # fp16
    gamma = tl.load(gamma_ptr + h, mask=mask, other=0.0)    # fp16
    beta  = tl.load(beta_ptr  + h, mask=mask, other=0.0)    # fp16

    # ---- LayerNorm over the hidden dimension (computed in fp32) ----
    val_f32 = val.to(tl.float32)
    sum_val = tl.sum(val_f32, axis=0)
    sum_sq  = tl.sum(val_f32 * val_f32, axis=0)
    mean    = sum_val / H
    var     = sum_sq / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    norm_f32 = (val_f32 - mean) * inv_std          # fp32 normalized
    norm_fp16 = norm_f32.to(tl.float16)            # back to fp16

    # Apply scale, bias and output gate
    out = (norm_fp16 * gamma + beta) * gate

    # Store the result in‑place
    tl.store(out_off, out, mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32) → fp16
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections and gates (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left      = F.linear(x, left_proj_w)          # (B, N, N, H)
    right     = F.linear(x, right_proj_w)         # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # Ensure contiguity for the kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # -----------------------------------------------------------------
    # 3. Fuse mask and gate multiplication (in‑place) via Triton
    # -----------------------------------------------------------------
    has_mask = (mask is not None) and (not nomask)
    if has_mask:
        mask_fp = mask.to(device).to(torch.float16)   # (B, N, N)
    else:
        # Dummy tensor – never accessed when HAS_MASK=False
        mask_fp = torch.empty((1,), device=device, dtype=torch.float16)

    total = left.numel()
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

    _gate_mask_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_fp,
        total_elems=total,
        hidden_dim=hidden_dim,
        HAS_MASK=has_mask,
        BLOCK_SIZE=BLOCK,
    )
    # -----------------------------------------------------------------
    # 4. Core contraction: batched GEMM (left @ rightᵀ)
    # -----------------------------------------------------------------
    # reshape to (B*H, N, N) for both tensors
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore (B, N, N, H) layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5. Output LayerNorm + output gate (fused)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    total_ijk = B * N * N
    grid2 = (total_ijk,)

    _layernorm_gate_fuse_kernel[grid2](
        out_ptr=out,
        out_gate_ptr=out_gate,
        gamma_ptr=to_out_norm_w,
        beta_ptr=to_out_norm_b,
        stride_out_b0=out.stride(0), stride_out_b1=out.stride(1),
        stride_out_b2=out.stride(2), stride_out_b3=out.stride(3),
        stride_gate_b0=out_gate.stride(0), stride_gate_b1=out_gate.stride(1),
        stride_gate_b2=out_gate.stride(2), stride_gate_b3=out_gate.stride(3),
        B=B, N=N, H=hidden_dim,
        EPS=1e-5,
        BLOCK_H=384,   # max supported hidden dimension (compile‑time)
    )

    # -----------------------------------------------------------------
    # 6. Final linear projection back to original dimension
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)  # (C, H)
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the original dtype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)
    return out
