"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm Overview
------------------
1. Layer‑norm the input (float32) using learned weight/bias.
2. Cast to fp16 for compute‑heavy path.
3. Linear projections to hidden dimension for left/right tensors and three
   sigmoid gates (left, right, out).
4. Fuse mask (if present) and element‑wise gate multiplication into a single
   Triton kernel that updates ``left`` and ``right`` in‑place.
5. Compute the core contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   by reshaping to (B*H, N, N) and calling ``torch.bmm`` (batched GEMM).
6. Apply a second LayerNorm over the hidden dimension, multiply by the output
   gate and project back to the original channel dimension.
7. Cast back to the input dtype and return.

The kernel in step 4 is necessary for the task; the remaining heavy
computation is handled by cuBLAS via ``torch.bmm`` which is already
highly optimized on the target GPU (NVIDIA H100/B200).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    """
    In‑place fusion of
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    where ``mask`` is broadcast over the hidden dimension ``H``.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load tensors (use 0.0 for masked‑out elements – they will be ignored)
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # mask index = element // hidden_dim (broadcast over H)
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as the input.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device=device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device=device, dtype=torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for the compute‑heavy path
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (bias‑less) – all in fp16
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    left  = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)  # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,hidden_dim)

    # -----------------------------------------------------------------
    # 4. Fuse mask and gates (if needed) via a Triton kernel
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp = mask.to(device=device, dtype=torch.float16).contiguous()
        # Flatten tensors for the kernel
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back to original layout
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # Simple element‑wise fusion without mask
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core O(N³·H) contraction via batched GEMM (L @ Rᵀ)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Merge batch and hidden dimensions to use torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B*H, N, N)

    # Batched matrix multiplication (leverages cuBLAS tensor‑cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, apply output gate, final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (half‑precision)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # -----------------------------------------------------------------
    # 7. Cast back to the input dtype (usually float32) and return
    # -----------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,hidden_dim)

    # -----------------------------------------------------------------
    # 4. Fuse mask and gates via a Triton kernel (if needed)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device=device, dtype=torch.float16).contiguous()
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core O(N³·H) contraction via batched GEMM (L @ Rᵀ)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, apply output gate, final projection back to dim
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (half‑precision)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # -----------------------------------------------------------------
    # 7. Cast back to the original dtype and return
    # -----------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
