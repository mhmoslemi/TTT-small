"""
TriMul (outgoing) forward pass – Triton‑accelerated version.

Algorithm
---------
1. Layer‑norm the input 4‑D tensor (B, N, N, C).
2. Cast to FP16 for compute‑intensive stages.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) are performed with torch.nn.functional.linear
   (cuBLAS) on FP16.
4. Apply mask (if present) and element‑wise gating.
5. Core O(N³·H) reduction:
   out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   implemented as a custom Triton batched GEMM kernel that directly
   reads the (B,N,N,H) layout without any extra copies.
6. Output LayerNorm, output‑gate and final projection back to original
   channel dimension.
7. Cast the result back to the input dtype (typically FP32).

Only the heavy reduction (step 5) is written in Triton; all other
operations use highly‑optimized PyTorch kernels.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def triton_gemm_forward(
    a_ptr, b_ptr, c_ptr,               # pointers to left, right, output
    B, N, H,                           # batch, seq_len, hidden_dim
    stride_ab, stride_ai, stride_ak, stride_ah,   # strides for a (left)
    stride_bb, stride_bi, stride_bk, stride_bh,   # strides for b (right)
    stride_cb, stride_ci, stride_cj, stride_ch,   # strides for c (out)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for tensors of shape (B, N, N, H).

    Computes, for each (b, h):
        C[b, i, j, h] = Σₖ A[b, i, k, h] * B[b, j, k, h]

    A, B, C are stored in memory as (B, N, N, H) with the strides
    given above.
    """
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_bh = tl.program_id(2)         # combined batch*hidden index

    # -------------------------------------------------------------
    # 1) Compute concrete indices
    # -------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    bh = pid_bh

    b = bh // H                       # batch index
    h = bh - b * H                     # hidden index (equiv. bh % H)

    # base offsets for this (b, h)
    a_batch = a_ptr + b * stride_ab + h * stride_ah
    b_batch = b_ptr + b * stride_bb + h * stride_bh
    c_batch = c_ptr + b * stride_cb + h * stride_ch

    # -------------------------------------------------------------
    # 2) Offsets inside the block
    # -------------------------------------------------------------
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for better numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----------- load A tile (BLOCK_M x BLOCK_K) -------------
        a_ptrs = a_batch + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # ----------- load B tile (BLOCK_K x BLOCK_N) -------------
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bi
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # ----------- matrix multiply & accumulate -----------------
        acc += tl.dot(a, b)   # a: (M,K), b: (K,N) → (M,N)

    # -------------------------------------------------------------
    # 3) Write back the result (cast to FP16)
    # -------------------------------------------------------------
    c_ptrs = c_batch + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

    # unpack inputs
    # -------------------------------------------------------------
    input_tensor, mask, weights, config = data

    dtype = input_tensor.dtype

    # -------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive stages
    # -------------------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------------------
    # 3) Cast all relevant weights to FP16
    # -------------------------------------------------------------
    def hf(t): return t.to(torch.float16)
    left_proj_w   = hf(weights["left_proj.weight"])
    right_proj_w  = hf(weights["right_proj.weight"])
    left_gate_w   = hf(weights["left_gate.weight"])
    right_gate_w  = hf(weights["right_gate.weight"])
    out_gate_w    = hf(weights["out_gate.weight"])
    to_out_norm_w = hf(weights["to_out_norm.weight"])
    to_out_norm_b = hf(weights["to_out_norm.bias"])
    to_out_w      = hf(weights["to_out.weight"])

    # -------------------------------------------------------------
    # 4) Linear projections (B,N,N,dim) -> (B,N,N,hidden_dim)
    # -------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # -------------------------------------------------------------
    # 5) Gate projections + sigmoid
    # -------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # -------------------------------------------------------------
    # 6) Apply mask (if any) and element‑wise gate
    # -------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape (B, N, N) → broadcast over hidden dim
        mask = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguous layout for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # -------------------------------------------------------------
    # 7) Core O(N³·H) reduction via a custom Triton batched GEMM
    # -------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)   # (B, N, N, H), fp16

    # stride information (in elements)
    stride_a = left.stride()
    stride_b = right.stride()
    stride_c = out.stride()

    # block sizes – chosen to fit well on typical tensor‑core hardware
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        math.ceil(N / BLOCK_M),
        math.ceil(N / BLOCK_N),
        B * H,
    )

    triton_gemm_forward[grid](
        left, right, out,
        B, N, H,
        stride_a[0], stride_a[1], stride_a[2], stride_a[3],
        stride_b[0], stride_b[1], stride_b[2], stride_b[3],
        stride_c[0], stride_c[1], stride_c[2], stride_c[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # -------------------------------------------------------------
    # 8) Output LayerNorm, output‑gate and final projection
    # -------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim), fp16

    # -------------------------------------------------------------
    # 9) Cast back to original dtype
    # -------------------------------------------------------------
    out = out.to(dtype)
    return out
