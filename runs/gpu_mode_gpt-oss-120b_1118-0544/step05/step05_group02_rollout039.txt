    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    The kernel fuses the O(N³·H) reduction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using a blocked matrix‑multiply implementation.
    All other linear & layer‑norm operations are performed with
    torch.nn.functional for simplicity.
    """

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # 2. Cast to fp16 for compute‑intensive parts
    x = x.to(torch.float16)

    # 3. Cast model weights to fp16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_w      = weights["to_out.weight"].to(torch.float16)

    left_proj  = F.linear(x, left_proj_w)    # (B,N,N,H)

"""
TriMul (outgoing) forward pass – Triton‑accelerated O(N³·H) reduction.

The kernel computes
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
for every batch b and hidden channel h using a blocked matrix‑multiply
implementation (M×K)·(K×N) → (M×N).  All other operations (layer‑norm,
linear projections, gating) are performed with torch.nn.functional for
simplicity.  The final output is returned in the original dtype
(e.g. float32).
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k, stride_h,
    out_stride_b, out_stride_i, out_stride_j, out_stride_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM for the TriMul reduction.

    left  : (B, N, N, H) interpreted as (batch, i, k, h)
    right : (B, N, N, H) interpreted as (batch, j, k, h)
    out   : (B, N, N, H) interpreted as (batch, i, j, h)
    """
    pid_i = tl.program_id(0)          # tile over i
    pid_j = tl.program_id(1)          # tile over j
    pid_bh = tl.program_id(2)         # combined batch * hidden index

    # Offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Decode batch and hidden indices
    batch_idx  = pid_bh // H
    hidden_idx = pid_bh % H

    # Base pointers for this (batch, hidden) slice
    left_base  = left_ptr  + batch_idx * stride_b + hidden_idx * stride_h
    right_base = right_ptr + batch_idx * stride_b + hidden_idx * stride_h

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile   (M, K)  -> rows i, cols k
        left_ptrs = left_base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # Load right tile  (K, N)  -> rows k, cols j
        # Note: stride_i is the stride of the first N dimension (j) in the right tensor
        right_ptrs = right_base + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(left, right)   # (M,K)·(K,N) → (M,N)

    # Store the result (convert back to FP16)
    out_ptrs = out_ptr + batch_idx * out_stride_b + hidden_idx * out_stride_h \
               + offs_i[:, None] * out_stride_i + offs_j[None, :] * out_stride_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, C]  (float32)
        - mask          : [B, N, N]    (float32 or bool, may be None)
        - weights       : dict of model parameters
        - config        : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑intensive parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast model weights to FP16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B,N,N,dim) → (B,N,N,hidden_dim)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 6. Apply mask (if any) and elementwise gate
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguous layout for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape

    # --------------------------------------------------------------
    # 7. Core O(N³·H) reduction using a Triton GEMM kernel
    # --------------------------------------------------------------
    out = torch.empty_like(left)  # (B, N, N, H), FP16

    # Strides (in element counts)
    stride_b, stride_i, stride_k, stride_h = left.stride()
    out_stride_b, out_stride_i, out_stride_j, out_stride_h = out.stride()

    # Block sizes – tuned for the target GPU
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (tiles over i, tiles over j, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_b,
        stride_i,
        stride_k,
        stride_h,
        out_stride_b,
        out_stride_i,
        out_stride_j,
        out_stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 8. Output LayerNorm, output gate, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)      # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9. Cast back to the original dtype
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
