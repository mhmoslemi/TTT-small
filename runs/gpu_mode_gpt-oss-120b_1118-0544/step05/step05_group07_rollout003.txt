    # Gates (sigmoid)
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    if mask is not None and not nomask:

        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)

        right = right_proj * right_gate

    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()

"""
TriMul (outgoing) forward pass – Triton‑fused mask & gating.

Algorithm
---------
1. **Input LayerNorm** – Normalize the (B,N,N,C) tensor.
2. **Linear projections** – Compute `left_proj`, `right_proj` and the three
   gate tensors (`left_gate`, `right_gate`, `out_gate`). All heavy work is
   performed in FP16.
3. **Mask & gate fusion** – If a mask is present, a tiny Triton kernel
   multiplies `proj * gate * mask` element‑wise, avoiding an extra memory
   pass. Otherwise we simply multiply `proj * gate`.
4. **Core O(N³·H) reduction** – For each hidden dimension `h`,
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   This is equivalent to a batch of H independent GEMMs of size N×N.
   By reshaping to (B*H, N, N) we can use `torch.bmm`, which runs on the
   highly‑optimized cuBLAS/Tensor‑Core path.
5. **Second LayerNorm, output gate & final projection** – Normalize the
   hidden tensor, apply the output gate and project back to the original
   channel dimension.
6. **Cast back** – Return the result in the original input dtype.

Only the mask‑gate fusion is written in Triton; the rest uses PyTorch
operations for maximum performance on the B200.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _gate_mask_kernel(
    proj_ptr,        # fp16 * (B*N*N*H,)
    gate_ptr,        # fp16 * (B*N*N*H,)
    mask_ptr,        # fp16 * (B*N*N,)
    out_ptr,         # fp16 * (B*N*N*H,)
    total_elems,     # i32 – total number of elements in proj/gate
    hidden_dim,      # i32 – H (for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    """Element‑wise multiplication: out = proj * gate * mask."""
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    # each hidden dim repeats the same mask value
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(out_ptr + offs, proj * gate * m, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using Triton.

    All tensors must be on the same CUDA device, dtype = torch.float16,
    and contiguous.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],          # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict containing all linear and layer‑norm parameters
            - config: dict with keys:
                * "dim" (int)          – input channel dimension C
                * "hidden_dim" (int)   – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (float32 -> FP16 for compute)
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2) Linear projections (FP16)
    # -----------------------------------------------------------------
    left_proj = torch.nn.functional.linear(
        x, weights["left_proj.weight"].to(torch.float16)
    )               # (B, N, N, H)
    right_proj = torch.nn.functional.linear(
        x, weights["right_proj.weight"].to(torch.float16)
    )               # (B, N, N, H)

    # -----------------------------------------------------------------
    # 3) Gates (sigmoid) – still FP16
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()    # (B, N, N, H)

    # -----------------------------------------------------------------
    # 4) Fuse mask & gate with Triton (if needed)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5) Core reduction: batch of H GEMMs (N×N @ N×Nᵀ)
    # -----------------------------------------------------------------
    # Rearrange to (B, H, N, N) → (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape

    left_mat = left_perm.view(B * H, N, N)                # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)              # (B*H, N, N)

    # GEMM on each (B*H) slice
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6) Second LayerNorm, output gate and final projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate

    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float16)
    )   # (B, N, N, dim)

    # -----------------------------------------------------------------
    # 7) Cast back to the original dtype
    # -----------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
