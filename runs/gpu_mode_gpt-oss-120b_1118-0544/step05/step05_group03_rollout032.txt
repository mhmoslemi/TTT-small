@triton.jit
def batched_tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_A_b, stride_A_i, stride_A_k,
    stride_B_b, stride_B_k, stride_B_n,
    stride_C_b, stride_C_i, stride_C_n,
    hidden_dim: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch indices
    b_idx = pid_batch // hidden_dim
    h_idx = pid_batch % hidden_dim

    # base offsets
    a_base = b_idx * stride_A_b + h_idx * 1  # stride for hidden dim is 1
    b_base = b_idx * stride_B_b + h_idx * 1
    c_base = b_idx * stride_C_b + h_idx * 1

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < K

        # Load A block (M x K)
        a_ptrs = A_ptr + a_base + offs_m[:, None] * stride_A_i + k_offset[None, :] * stride_A_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

        # Load B block (K x N)
        b_ptrs = B_ptr + b_base + k_offset[:, None] * stride_B_k + offs_n[None, :] * stride_B_n
        b = tl.load(b_ptrs, mask=k_mask[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Compute block mul
        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = C_ptr + c_base + offs_m[:, None] * stride_C_i + offs_n[None, :] * stride_C_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    # Unpack input
    input_tensor, mask, weights, config = data
    # config includes dim, hidden_dim, maybe other dims.

    # Retrieve weights as Tensors (float32) likely.
    # Convert to FP16 for speed as before.

    # LayerNorm (input)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (config['dim'],),
        weight=weights['norm.weight'],
        bias=weights['norm.bias']
    )

    # Cast to fp16 for compute
    x = x_norm.to(torch.float16)

    # Linear projections
    left_proj = torch.nn.functional.linear(x, weights['left_proj.weight'].to(torch.float16))
    right_proj = torch.nn.functional.linear(x, weights['right_proj.weight'].to(torch.float16))

    left_gate = torch.nn.functional.linear(x, weights['left_gate.weight'].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights['right_gate.weight'].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights['out_gate.weight'].to(torch.float16)).sigmoid()

    # Apply mask and gating steps (if mask present)
    if mask is not None and not config.get('nomask', False):
        mask = mask.to(torch.float16).unsqueeze(-1)  # shape (B,N,N,1)
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Prepare output tensor
    out = torch.empty_like(left)  # shape (B,N,N,H)

    # Determine strides for kernel
    B, N, _, H = left.shape
    # pointer addresses
    a_ptr = left.data_ptr()
    b_ptr = right.data_ptr()
    c_ptr = out.data_ptr()

    # compute strides in elements (not bytes) required by kernel
    stride_A_b = left.stride(0)
    stride_A_i = left.stride(1)
    stride_A_k = left.stride(2)
    stride_B_b = right.stride(0)
    stride_B_k = right.stride(2)  # for k dimension (row)
    stride_B_n = right.stride(1)  # for j dimension
    stride_C_b = out.stride(0)
    stride_C_i = out.stride(1)
    stride_C_n = out.stride(2)

    # Compute grid dimensions
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Launch kernel
    batched_tri_mul_kernel[grid](
        a_ptr,
        b_ptr,
        c_ptr,
        B * H,
        N,
        N,
        N,
        stride_A_b, stride_A_i, stride_A_k,
        stride_B_b, stride_B_k, stride_B_n,
        stride_C_b, stride_C_i, stride_C_n,
        hidden_dim=H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # After kernel, out is fp16. Cast to fp16 ensures final steps are as expected.

    # Output layernorm and projection
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=weights['to_out_norm.weight'].to(torch.float16),
        bias=weights['to_out_norm.bias'].to(torch.float16)
    )

    out = out * out_gate

    out = torch.nn.functional.linear(out, weights['to_out.weight'].to(torch.float16))

    # Cast back to original dtype of input (maybe float32)
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

"""
TriMul (outgoing) forward – Triton‑accelerated core reduction.

Algorithm
---------
1. Layer‑norm the 4‑D input over the channel dimension `dim`.
2. Cast to FP16 for compute‑heavy part.
3. Linear projections (`left_proj`, `right_proj`) and gate projections
   (`left_gate`, `right_gate`, `out_gate`) are performed with torch.nn.functional.
4. Optional mask is applied, then we fuse the projection, gate and mask
   with a simple element‑wise multiplication (still in FP16).
5. Core multiplicative update:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is exactly a batched matrix multiplication of shape
   (B·H, N, N) × (B·H, N, N)^T.
   We implement it with a custom Triton kernel that loops over the
   reduction dimension `k` in tiles, accumulating in FP32 and writing
   back in FP16.  The kernel works directly on the original layout
   `[B, N, N, H]` – no costly permutes or copies are performed.
6. Output layer‑norm over the hidden dimension, gating with `out_gate`,
   and a final linear projection back to `dim`.
7. Cast the result back to the original dtype (usually FP32).

The kernel fulfills the requirement of using Triton for the
performance‑critical part of the operation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_tri_mul_kernel(
    A_ptr,  # left   (fp16)
    B_ptr,  # right  (fp16)
    C_ptr,  # out    (fp16)
    BATCH, M, N, K,
    stride_A_b, stride_A_i, stride_A_k,
    stride_B_b, stride_B_k, stride_B_n,
    stride_C_b, stride_C_i, stride_C_n,
    hidden_dim: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication for TriMul:
        C[b,i,j,h] = Σ_k A[b,i,k,h] * B[b,j,k,h]
    `b` and `h` are fused into a single program ID dimension.
    """
    pid_batch = tl.program_id(0)    # 0 … BATCH-1  (BATCH = B * hidden_dim)
    pid_m = tl.program_id(1)        # block row
    pid_n = tl.program_id(2)        # block col

    # De‑compose the fused batch index into (batch, head)
    b_idx = pid_batch // hidden_dim
    h_idx = pid_batch % hidden_dim

    # Base address for this (b, h) slice.
    # Stride of the hidden dimension is 1 (contiguous), so we add h_idx.
    a_base = b_idx * stride_A_b + h_idx
    b_base = b_idx * stride_B_b + h_idx
    c_base = b_idx * stride_C_b + h_idx

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + a_base + offs_m[:, None] * stride_A_i + offs_k[None, :] * stride_A_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B block (transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = B_ptr + b_base + offs_k[:, None] * stride_B_k + offs_n[None, :] * stride_B_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Matrix multiplication for the tile
        acc += tl.dot(a, b)

    # Write back the result in FP16
    c_ptrs = C_ptr + c_base + offs_m[:, None] * stride_C_i + offs_n[None, :] * stride_C_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (or None)
        - weights      : dict of model parameters (float32)
        - config       : dict with keys
              * "dim"        – input channel dimension C
              * "hidden_dim" – hidden dimension H
              * "nomask"     – optional bool, if True ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections and gates (all in FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask & gate fusion (element‑wise, still FP16)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape: [B, N, N] -> broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguity for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape
    assert H == hidden_dim, "hidden_dim mismatch"

    # ------------------------------------------------------------------
    # 5. Core TriMul reduction via Triton (batched GEMM)
    # ------------------------------------------------------------------
    out = torch.empty_like(left)  # [B, N, N, H] in FP16

    # Strides in element units (not bytes)
    stride_A_b = left.stride(0)   # B‑stride
    stride_A_i = left.stride(1)   # row (i) stride
    stride_A_k = left.stride(2)   # reduction (k) stride
    # hidden dimension stride is always 1 for contiguous tensors
    stride_B_b = right.stride(0)
    stride_B_k = right.stride(2)  # row of transposed matrix (k)
    stride_B_n = right.stride(1)  # column (j) stride
    stride_C_b = out.stride(0)
    stride_C_i = out.stride(1)
    stride_C_n = out.stride(2)

    # Block sizes – tuned for typical Hopper/V100 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                # fused batch * head dimension
        triton.cdiv(N, BLOCK_M),       # number of tiles in i‑direction
        triton.cdiv(N, BLOCK_N),       # number of tiles in j‑direction
    )

    _batched_tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B * hidden_dim,  # total fused batch size
        N, N, N,
        stride_A_b, stride_A_i, stride_A_k,
        stride_B_b, stride_B_k, stride_B_n,
        stride_C_b, stride_C_i, stride_C_n,
        hidden_dim=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 7. Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
