@triton.jit
def tri_mul_kernel(
    lhs_ptr, rhs_ptr, out_ptr,
    BATCH, M, N, K,
    stride_lhs_batch, stride_lhs_m, stride_lhs_k,
    stride_rhs_batch, stride_rhs_k, stride_rhs_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_STAGES: tl.constexpr = 2,
):
    pid = tl.program_id(0)
    # total number of blocks: batch * (M//BLOCK_M) * (N//BLOCK_N)
    batch_id = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_id = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = block_id // tl.cdiv(N, BLOCK_N)
    pid_n = block_id % tl.cdiv(N, BLOCK_N)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this block
    a_ptrs = lhs_ptr + batch_id * stride_lhs_batch + offs_m[:, None] * stride_lhs_m + tl.arange(0, BLOCK_K)[None, :] * stride_lhs_k
    b_ptrs = rhs_ptr + batch_id * stride_rhs_batch + tl.arange(0, BLOCK_K)[:, None] * stride_rhs_k + offs_n[None, :] * stride_rhs_n

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k + tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_lhs_k
        b_ptrs += BLOCK_K * stride_rhs_k

    # store result
    c_ptrs = out_ptr + batch_id * stride_out_batch + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the input tensor (B, N, N, C) in float32.
2. Cast to float16 for the compute‑heavy stages.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) are performed with torch.nn.functional.linear
   (cuBLAS) on FP16 tensors.
4. Optional mask is fused with the projections and gates using broadcasting.
5. Core O(N³·H) reduction is a batched matrix multiplication:
       out[b, i, j, h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton kernel that performs a batched GEMM
   (batch = B*H) on FP16 data with FP32 accumulation.
6. The result is reshaped back to (B, N, N, H) and cast to float32.
7. Output LayerNorm, output‑gate and final linear projection back to
   the original channel dimension.
8. Cast the final tensor back to the input dtype.

Only the core reduction (step 5) is performed in Triton; all other steps
use PyTorch primitives for simplicity.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (C = A @ B)
# A shape: (batch, M, K)   B shape: (batch, K, N)   -> C shape: (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    batch, M, N, K,                     # problem size
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # Number of tiles per matrix
    tiles_m = tl.cdiv(M, BLOCK_M)
    tiles_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = tiles_m * tiles_n

    batch_id = pid // blocks_per_batch
    block_id = pid % blocks_per_batch
    pid_m = block_id // tiles_n
    pid_n = block_id % tiles_n

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile in A and B
    a_ptrs = a_ptr + batch_id * stride_a_batch \
                   + offs_m[:, None] * stride_a_m \
                   + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_ptrs = b_ptr + batch_id * stride_b_batch \
                   + tl.arange(0, BLOCK_K)[:, None] * stride_b_k \
                   + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Store the result
    c_ptrs = c_ptr + batch_id * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) or None
            - weights: dict of model parameters (torch.Tensors)
            - config: dict containing at least "dim" and "hidden_dim",
                      optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑intensive steps
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast weights to FP16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 6. Apply mask (if present) and gates
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # mask broadcasted over hidden dimension
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction: batched GEMM using Triton
    #    out[b, i, j, h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # Bring hidden dimension next to batch for easy flattening
    left = left.permute(0, 3, 1, 2).contiguous()        # (B, H, N, N)
    right = right.permute(0, 3, 2, 1).contiguous()      # (B, H, N, N)  (transpose K ↔ N)

    total_batches = B * H
    left_flat  = left.view(total_batches, N, N)
    right_flat = right.view(total_batches, N, N)

    # Allocate output tensor (FP16)
    out_flat = torch.empty_like(left_flat)

    # Block sizes – tuned for typical B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = total_batches * grid_m * grid_n

    tri_mul_kernel[(total_blocks,)](
        left_flat,
        right_flat,
        out_flat,
        total_batches,
        N,
        N,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion before proceeding
    torch.cuda.synchronize(device)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # --------------------------------------------------------------
    out = out.float()  # convert to FP32 for LayerNorm
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w.float(),
        bias=to_out_norm_b.float(),
    )
    out = out * out_gate.float()
    out = F.linear(out, to_out_w.float())   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
