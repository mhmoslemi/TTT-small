"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Key ideas
---------
1. LayerNorm on the input (float32 for stability).
2. Linear projections + sigmoid gates are performed with torch.nn.functional.linear
   (half precision) – these are already highly optimized GEMMs.
3. A Triton kernel fuses the element‑wise mask, left/right gate, and gate
   multiplication.  The kernel works per (batch, i, j) pair and processes the
   hidden dimension in vectorized blocks, loading each mask value only once.
4. The core O(N³·H) contraction is carried out as a batched GEMM
   `left @ rightᵀ` using torch.bmm on half tensors.
5. Output LayerNorm, output gate and final linear projection are applied in
   half precision and the result is cast back to the input dtype.

The kernel reduces memory traffic compared to the naïve implementation
by avoiding repeated mask loads and by performing vector loads over the
hidden dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,                     # *float16, shape [total_pairs * hidden_dim]
    right_ptr,                    # *float16, same shape
    left_gate_ptr,                # *float16, same shape
    right_gate_ptr,               # *float16, same shape
    mask_ptr,                     # *float16, shape [total_pairs] (optional)
    total_pairs, hidden_dim,      # scalar metadata
    HAS_MASK: tl.constexpr,       # compile‑time bool
    BLOCK_H: tl.constexpr,        # compile‑time block size for hidden dim
):
    """
    In‑place fusion of:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    One program per (b, i, j) pair; hidden dimension is processed in
    BLOCK_H‑sized chunks.  Mask is broadcast across the hidden dimension.
    """
    pid = tl.program_id(0)                # pair index = b * N * N + i * N + j
    pair_idx = pid

    # Base pointer for this pair (offset into the flattened tensors)
    base = pair_idx * hidden_dim

    # Load mask once per pair if present
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + pair_idx)          # scalar float16
    else:
        mask_val = tl.full([1], 1.0, dtype=tl.float16)[0]

    # Loop over the hidden dimension
    offs = tl.arange(0, BLOCK_H)
    for off in range(0, hidden_dim, BLOCK_H):
        cur = off + offs
        active = cur < hidden_dim

        # Load vectors
        left      = tl.load(left_ptr      + base + cur, mask=active, other=0.0)
        right     = tl.load(right_ptr     + base + cur, mask=active, other=0.0)
        left_gate = tl.load(left_gate_ptr + base + cur, mask=active, other=0.0)
        right_gate= tl.load(right_gate_ptr+ base + cur, mask=active, other=0.0)

        # Apply gates and optional mask
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val

        # Store results back (in‑place)
        tl.store(left_ptr  + base + cur, left,  mask=active)
        tl.store(right_ptr + base + cur, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor (torch.Tensor): shape (B, N, N, C)
            - mask (torch.Tensor or None): shape (B, N, N)
            - weights (dict): model parameters
            - config (dict): must contain "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor: shape (B, N, N, C) with the same dtype as input_tensor.
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for compute‑heavy parts
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gates
    # -----------------------------------------------------------------
    left  = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)         # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 4. Fuse mask and gate multiplication (Triton)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # mask broadcasted over hidden dimension
        mask_fp = mask.to(device).to(torch.float16)
        # Flatten tensors for the kernel
        B, N, _, _ = left.shape
        total_pairs = B * N * N

        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (total_pairs,)

        # Choose a block size that comfortably fits hidden_dim (<= 128 works well)
        BLOCK_H = 64  # compile‑time constant used by the kernel

        grid = (total_pairs,)
        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total_pairs,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_H=BLOCK_H,
        )
        # Reshape back to original layout
        left  = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask → direct element‑wise multiplication with gates
        left  = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ rightᵀ) over the sequence dim
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (left @ rightᵀ)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the original dtype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
