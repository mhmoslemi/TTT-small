TriMul (outgoing) forward pass – Triton‑accelerated core reduction.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) using the supplied weight/bias.
2. Cast to fp16 for the compute‑intensive part.
3. Four linear projections (left/right and their gates) are performed with

7. Cast the result back to the input dtype (float32).

The kernel processes a tile of size (BLOCK_M, BLOCK_N) for each
(b, d) pair, looping over the reduction dimension k in BLOCK_K sized
chunks.  Accumulation is done in float32 for numerical stability and the
final result is stored as fp16.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: core TriMul reduction (batched GEMM over the k dimension)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,               # fp16*  [B, N, N, H]
    right_ptr,              # fp16*  [B, N, N, H]
    out_ptr,                # fp16*  [B, N, N, H]
    B, N, H,                # runtime ints
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,   # right strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # 0‑dim : batch*hidden, 1‑dim : i‑tile, 2‑dim : j‑tile
    batch_id = tl.program_id(0)
    i_block = tl.program_id(1)
    j_block = tl.program_id(2)

    # Decompose batch_id into batch index b and hidden index d
    b = batch_id // H
    d = batch_id % H

    # Starting coordinates of the tile
    i0 = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j0 = j_block * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / columns
    mask_i = i0 < N
    mask_j = j0 < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in BLOCK_K sized chunks
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_k):
        k0 = k_iter * BLOCK_K
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ----- load left block (M x K) -----
        left_offset = (
            b * stride_lb
            + i0[:, None] * stride_li
            + k[None, :] * stride_lk
            + d * stride_ld
        )
        left_mask = mask_i[:, None] & mask_k[None, :]
        left = tl.load(left_ptr + left_offset, mask=left_mask, other=0.0)
        left = left.to(tl.float32)

        # ----- load right block (K x N) -----
        # note: we need the matrix with rows = k, cols = j
        right_offset = (
            b * stride_rb
            + k[:, None] * stride_rk
            + j0[None, :] * stride_rj
            + d * stride_rd
        )
        right_mask = mask_k[:, None] & mask_j[None, :]
        right = tl.load(right_ptr + right_offset, mask=right_mask, other=0.0)
        right = right.to(tl.float32)

        # ----- accumulate -----
        acc += tl.dot(left, right)          # (BLOCK_M, BLOCK_N)

    # ----- write out -----
    out_offset = (
        b * stride_ob
        + i0[:, None] * stride_oi
        + j0[None, :] * stride_oj
        + d * stride_od
    )
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptr + out_offset, acc.to(tl.float16), mask=out_mask)


# ----------------------------------------------------------------------
# Triton helper for the core reduction
# ----------------------------------------------------------------------
def _tri_mul_reduce(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out = Σ_k left[...,k,d] * right[...,k,d] using the Triton kernel.
    left, right : [B, N, N, H]  (fp16, contiguous)
    Returns out : [B, N, N, H]  (fp16)
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    device = left.device

    out = torch.empty_like(left)

    # stride values in element units
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # Tile sizes – 32 is a good default for most GPUs
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    total_batches = B * H
    grid = (
        total_batches,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------- unpack --------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # -------------------- LayerNorm (input) --------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -------------------- cast to fp16 for compute --------------------
    x = x.to(torch.float16)

    # -------------------- cast weights --------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------- linear projections --------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,H]
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # -------------------- gates --------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # [B,N,N,H]

    # -------------------- optional mask --------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguity for the kernel
    left = left.contiguous()
    right = right.contiguous()

    # -------------------- core reduction (Triton) --------------------
    out = _tri_mul_reduce(left, right)          # [B,N,N,H], fp16

    # -------------------- second LayerNorm --------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------- output gating --------------------
    out = out * out_gate

    # -------------------- final linear projection --------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]

    # -------------------- cast back to original dtype --------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
