def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. LayerNorm
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # 2. Cast to fp16
    x = x.to(torch.float16)

    # 3. Linear projections
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)   # B,N,N,H
    right = F.linear(x, right_proj_w)
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # 4. Apply mask and gates
    if mask is not None and not nomask:
        # Cast mask to fp16
        mask_fp = mask.to(device).to(torch.float16)
        # flatten tensors
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_fp.view(-1)
        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # 5. Core TriMul via Triton kernel
    B, N, _, H = left.shape
    # Rearrange to (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)
    out_mat = torch.empty_like(left_mat)  # same dtype, shape (B*H,N,N)

    # compute strides for kernel
    stride_left_batch = left_mat.stride(0)
    stride_left_m = left_mat.stride(1)
    stride_left_k = left_mat.stride(2)

    stride_right_batch = right_mat.stride(0)
    stride_right_m = right_mat.stride(1)
    stride_right_n = right_mat.stride(2)  # using for original matrix

    stride_out_batch = out_mat.stride(0)
    stride_out_m = out_mat.stride(1)
    stride_out_n = out_mat.stride(2)

    # grid dimensions
    num_batch = B * H
    num_block_m = triton.cdiv(N, 64)  # block size set maybe 64
    num_block_n = triton.cdiv(N, 64)
    grid = (num_batch, num_block_m, num_block_n)

    # Choose block sizes via constexpr parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        num_batch,
        N,
        N,
        N,
        # strides
        stride_left_batch,
        stride_left_m,
        stride_left_k,
        stride_right_batch,
        stride_right_m,
        stride_right_n,
        stride_out_batch,
        stride_out_m,
        stride_out_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Reshape out_mat back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Output LayerNorm
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # apply output gate
    out = out * out_gate

    # 7. Final linear projection to dim
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)

    # Cast back to original dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH, M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_m, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for M,N out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for left: (M,K)
        left_ptrs = left_ptr + pid_batch * stride_l_batch + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k
        a = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # pointers for right transposed: we need (K,N)
        # right original shape (M,N). Use stride_r_n for row (K) and stride_r_m for column (N)
        right_ptrs = right_ptr + pid_batch * stride_r_batch + offs_k[:, None] * stride_r_n + offs_n[None, :] * stride_r_m
        b = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)
    # Write back
    out_ptrs = out_ptr + pid_batch * stride_o_batch + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n
    c = acc.to(tl.float16)
    tl.store(out_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
```

    right = tl.load(right_ptr + offs, mask=active, other=0.0)

        right = right * right_gate

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input (float32) → fp16.
2. Linear projections to hidden dimension (no bias).
3. Compute left/right gates and output gate (sigmoid) and fuse mask (if any)
   with a tiny Triton kernel that performs the element‑wise multiplications in‑place.
4. Core contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched GEMM (L @ Rᵀ) for each hidden channel.
   We implement a custom Triton GEMM kernel that treats the hidden
   dimension as an extra batch (B·H matrices of size N×N).
5. Output layer‑norm, output‑gate scaling and final linear projection.
6. Cast back to the original dtype.

The implementation fuses as many element‑wise ops as possible and keeps the
heavy O(N³·H) work inside a highly‑optimized Triton GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # one mask value per (b,i,k) -> broadcast over hidden dimension
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH, M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_m, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: out = left @ rightᵀ  (left & right are (M,K) and (M,N))"""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left: (M,K)
        ptr_l = left_ptr + pid_batch * stride_l_batch \
                + offs_m[:, None] * stride_l_m \
                + offs_k[None, :] * stride_l_k
        a = tl.load(ptr_l, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # rightᵀ: (K,N)  – we load transposed view using swapped strides
        ptr_r = right_ptr + pid_batch * stride_r_batch \
                + offs_k[:, None] * stride_r_n \
                + offs_n[None, :] * stride_r_m
        b = tl.load(ptr_r, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)          # fp16·fp16 → fp32 accumulation

    # write result
    ptr_o = out_ptr + pid_batch * stride_o_batch \
            + offs_m[:, None] * stride_o_m \
            + offs_n[None, :] * stride_o_n
    c = acc.to(tl.float16)
    tl.store(ptr_o, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------- #
    # 1. Input LayerNorm (float32 for stability)
    # --------------------------------------------------------------------- #
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------------- #
    # 2. Cast to fp16 for the compute‑heavy path
    # --------------------------------------------------------------------- #
    x = x.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 3. Linear projections (bias‑less)
    # --------------------------------------------------------------------- #
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left  = F.linear(x, left_proj_w)      # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------------- #
    # 4. Fuse mask and gates (if a mask is present)
    # --------------------------------------------------------------------- #
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten everything for the kernel
        left_f    = left.contiguous().view(-1)
        right_f   = right.contiguous().view(-1)
        left_g_f  = left_gate.contiguous().view(-1)
        right_g_f = right_gate.contiguous().view(-1)
        mask_f    = mask_fp.contiguous().view(-1)

        total = left_f.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        _gate_mask_kernel[grid](
            left_f,
            right_f,
            left_g_f,
            right_g_f,
            mask_f,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # simple element‑wise fusion without mask
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Core O(N³·H) contraction via a custom Triton GEMM
    # --------------------------------------------------------------------- #
    B, N, _, H = left.shape   # H == hidden_dim
    # reshape to (B*H, N, N) – each hidden channel becomes a separate batch
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_mat   = torch.empty_like(left_mat)

    # strides for the kernel
    stride_l_batch = left_mat.stride(0)
    stride_l_m     = left_mat.stride(1)
    stride_l_k     = left_mat.stride(2)

    stride_r_batch = right_mat.stride(0)
    stride_r_m     = right_mat.stride(1)   # original row stride (N)
    stride_r_n     = right_mat.stride(2)   # original col stride (1)

    stride_o_batch = out_mat.stride(0)
    stride_o_m     = out_mat.stride(1)
    stride_o_n     = out_mat.stride(2)

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    num_batch = B * H
    grid = (num_batch,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        num_batch,
        N, N, N,
        stride_l_batch, stride_l_m, stride_l_k,
        stride_r_batch, stride_r_m, stride_r_n,
        stride_o_batch, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 6. Output LayerNorm, output‑gate and final linear projection
    # --------------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # --------------------------------------------------------------------- #
    # 7. Cast back to original dtype
    # --------------------------------------------------------------------- #
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
