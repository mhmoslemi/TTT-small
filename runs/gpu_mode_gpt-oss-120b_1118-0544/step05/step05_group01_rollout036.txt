"""
custom_kernel: Triton‑accelerated outgoing TriMul forward pass.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (`dim`).
2. Cast the normalized tensor to FP16.
3. Linear projections (no bias) to a hidden dimension `H`:
       left_proj, right_proj, left_gate, right_gate, out_gate.
4. Apply sigmoid to the three gate tensors.
5. Fuse gate, projection and optional mask:
       left  = left_proj  * left_gate  * mask
       right = right_proj * right_gate * mask
6. Core reduction – batched matrix multiplication for every
   (batch, hidden) pair:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton kernel that reads the tensors
   directly without materialising the (B·H,N,N) reshapes used by
   torch.bmm, thus saving a large memory copy.
7. Layer‑norm over the hidden dimension, multiplicative out‑gate,
   and a final linear projection back to `dim`.
8. Cast the result back to the original dtype (usually FP32).

The kernel handles arbitrary batch size `B`, sequence length `N`,
input dimension `dim` and hidden dimension `H` (named `hidden_dim`).
Only the reduction step is written in Triton; all other operations
use highly‑optimized PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr,          # fp16 * (B, N, N, H)
    right_ptr,         # fp16 * (B, N, N, H)
    out_ptr,           # fp16 * (B, N, N, H)
    B: tl.constexpr,   # batch size
    N: tl.constexpr,   # sequence length
    H: tl.constexpr,   # hidden dimension
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_bh = tl.program_id(2)         # combined batch‑hidden index

    # decode batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # tile start offsets
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # range of indices in the tile
    i_idx = i_base + tl.arange(0, BLOCK_M)
    j_idx = j_base + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows / cols
    i_mask = i_idx < N
    j_mask = j_idx < N

    # base pointers for the current (b, h)
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for pk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = pk * BLOCK_K
        k_idx = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_idx < N

        # left[i, k, h]
        left_off = (left_base
                    + i_idx[:, None] * stride_li
                    + k_idx[None, :] * stride_lk)
        left_tile = tl.load(left_off,
                            mask=i_mask[:, None] & k_mask[None, :],
                            other=0.0)

        # right[j, k, h]  – we need the transposed view (k, j)
        right_off = (right_base
                     + k_idx[:, None] * stride_rk
                     + j_idx[None, :] * stride_rj)
        right_tile = tl.load(right_off,
                             mask=k_mask[:, None] & j_mask[None, :],
                             other=0.0)

        # dot‑product over the K dimension
        acc += tl.dot(left_tile, right_tile)   # FP16→FP32 accumulation

    # write the result (cast back to FP16)
    out_off = (out_base
               + i_idx[:, None] * stride_oi
               + j_idx[None, :] * stride_oj)
    tl.store(out_off,
             acc.to(tl.float16),
             mask=i_mask[:, None] & j_mask[None, :])


# ----------------------------------------------------------------------
# Helper: layer‑norm that works with FP16 weights/biases
# ----------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor) -> torch.Tensor:
    # x: (..., C)   weight/bias: (C,)
    return torch.nn.functional.layer_norm(x, x.shape[-1:], weight=weight, bias=bias)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] (binary) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim), dtype = input dtype

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)      # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5. Fuse projection, gate and optional mask
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape (B, N, N) -> broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)          # (B,N,N,1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with a custom Triton kernel to avoid costly
    #    reshapes and copies.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)   # (B,N,N,H), fp16

    # strides (in elements) of the three tensors
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)

    _tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_lb=stride_lb,
        stride_li=stride_li,
        stride_lk=stride_lk,
        stride_lh=stride_lh,
        stride_rb=stride_rb,
        stride_rj=stride_rj,
        stride_rk=stride_rk,
        stride_rh=stride_rh,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_oh=stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Output layer‑norm, apply out‑gate and final projection
    # ------------------------------------------------------------------
    out = _layer_norm(
        out,
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
