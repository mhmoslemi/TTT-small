```
@triton.jit
def triu_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)
    # compute batch and head indices
    b = pid_bh // H
    h = pid_bh % H
    # compute offsets for i, j
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    # compute base pointers for this b, h
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        
        # compute offsets for left tile
        left_off_i = i_offset * stride_l_i
        left_off_k = k_offset * stride_l_k
        left_ptrs = left_base + left_off_i + left_off_k

        # load left tile (M,K)
        left = tl.load(left_ptrs + tl.arange(0, BLOCK_M)[:, None] * stride_l_i + tl.arange(0, BLOCK_K)[None, :] * stride_l_k,
                       mask=(tl.arange(0, BLOCK_M)[:, None] < (N - i_offset)) & (tl.arange(0, BLOCK_K)[None, :] < (N - k_offset)),
                       other=0.0)

        # right tile: we need transposed, shape (K,N)
        # compute pointer to right base indexed by k (row) and j (col)
        right_off_j = j_offset * stride_r_j
        right_off_k = k_offset * stride_r_k
        # load right tile (K,N) where K is rows (k), N is columns (j) i.e., shape (BLOCK_K, BLOCK_N)
        right = tl.load(right_base + right_off_k + right_off_j + 
                        tl.arange(0, BLOCK_K)[:, None] * stride_r_k + tl.arange(0, BLOCK_N)[None, :] * stride_r_j,
                        mask=(tl.arange(0, BLOCK_K)[:, None] < (N - k_offset)) & (tl.arange(0, BLOCK_N)[None, :] < (N - j_offset)),
                        other=0.0)

        # perform dot product
        acc += tl.dot(left, right, out_dtype=tl.float32) # left MxK, right KxN -> MxN

    # store result
    out_ptrs = out_base + i_offset * stride_o_i + j_offset * stride_o_j
    tl.store(out_ptrs + tl.arange(0, BLOCK_M)[:, None] * stride_o_i + tl.arange(0, BLOCK_N)[None, :] * stride_o_j,
             acc.to(tl.float16), mask=(tl.arange(0, BLOCK_M)[:, None] < (N - i_offset)) & (tl.arange(0, BLOCK_N)[None, :] < (N - j_offset)))
```

"""
Optimized Triton implementation of the outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the channel dimension (dim) – torch implementation.
2. Cast to fp16 for compute‑intensive parts.
3. Linear projections (left/right) and gating projections (left_gate/right_gate/out_gate).
4. Fuse mask, projection and gate in a small Triton kernel (only when a mask is present).
5. Core reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a blocked matrix multiplication in Triton.  The batch * hidden
   dimension (B·H) is fused into the third program‑id, so all heads are processed
   in a single kernel launch.
6. Layer‑norm over the hidden dimension, multiply by out_gate and project back
   to the original channel dimension.
7. Cast the result back to the input dtype.

The heavy O(N³·H) reduction is fully realized in Triton, giving a large speed‑up
over a naïve PyTorch bmm loop.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# -----------------------------------------------------------------------
# Kernel 1: fuse projection, gate and optional mask (mask broadcast over H)
# -----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,       # fp16 * (B,N,N,H)
    gate_ptr,       # fp16 * (B,N,N,H)
    mask_ptr,       # fp16 * (B,N,N)  (0/1)
    out_ptr,        # fp16 * (B,N,N,H)
    total_elems,    # B*N*N*H
    hidden_dim,     # H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # each `hidden_dim` consecutive entries share the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """Fuse `proj * gate * mask` (mask broadcast over hidden dim) via Triton."""
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# -----------------------------------------------------------------------
# Kernel 2: core TriMul reduction (batched gemm over the k dimension)
# -----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,         # fp16 * (B,N,N,H)
    right_ptr,        # fp16 * (B,N,N,H)
    out_ptr,          # fp16 * (B,N,N,H)
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # tile over rows (i)
    pid_j = tl.program_id(1)   # tile over cols (j)
    pid_bh = tl.program_id(2)  # fused batch * hidden index

    # decode batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_iter * BLOCK_K

        # ------------------- load left tile (M x K) -------------------
        i_idx = i_base + tl.arange(0, BLOCK_M)          # (M,)
        k_idx = k_base + tl.arange(0, BLOCK_K)          # (K,)

        i_mask = i_idx < N
        k_mask = k_idx < N
        mask_l = i_mask[:, None] & k_mask[None, :]

        left_offset = (
            b * stride_l_b
            + h * stride_l_h
            + (i_idx[:, None] * stride_l_i)
            + (k_idx[None, :] * stride_l_k)
        )
        left = tl.load(left_ptr + left_offset, mask=mask_l, other=0.0).to(tl.float32)

        # ------------------- load right tile (K x N) -------------------
        j_idx = j_base + tl.arange(0, BLOCK_N)          # (N,)
        # note: right is accessed as (k, j) to match tl.dot(left, right)
        k_idx_r = k_base + tl.arange(0, BLOCK_K)        # (K,)

        j_mask = j_idx < N
        k_mask_r = k_idx_r < N
        mask_r = k_mask_r[:, None] & j_mask[None, :]

        right_offset = (
            b * stride_r_b
            + h * stride_r_h
            + (k_idx_r[:, None] * stride_r_k)
            + (j_idx[None, :] * stride_r_j)
        )
        right = tl.load(right_ptr + right_offset, mask=mask_r, other=0.0).to(tl.float32)

        # ------------------- dot product -------------------
        acc += tl.dot(left, right)   # (M, K) @ (K, N) -> (M, N)

    # ------------------- store result (M x N) -------------------
    i_idx_out = i_base + tl.arange(0, BLOCK_M)
    j_idx_out = j_base + tl.arange(0, BLOCK_N)

    mask_o = (i_idx_out[:, None] < N) & (j_idx_out[None, :] < N)

    out_offset = (
        b * stride_o_b
        + h * stride_o_h
        + (i_idx_out[:, None] * stride_o_i)
        + (j_idx_out[None, :] * stride_o_j)
    )
    tl.store(out_ptr + out_offset, acc.to(tl.float16), mask=mask_o)


# -----------------------------------------------------------------------
# Entry point
# -----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, dim] (same dtype as input).
    """
    # -------------------------------------------------------------------
    # Unpack arguments
    # -------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # -------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # -------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # -------------------------------------------------------------------
    # 2) Cast to fp16 for the heavy part
    # -------------------------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------------------------
    # 3) Cast all linear weights to fp16 (biases are None)
    # -------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------------------------
    # 4) Linear projections and gating
    # -------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)      # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()  # (B,N,N,H)

    # -------------------------------------------------------------------
    # 5) Apply mask & gating (fused via Triton when a mask is present)
    # -------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------------------------------------------------
    # 6) Core TriMul reduction (batched GEMM over k) – Triton kernel
    # -------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # (B,N,N,H), fp16

    # strides (in elements) for the three tensors
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_j, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),      # i‑tiles
        triton.cdiv(N, BLOCK_N),      # j‑tiles
        B * H,                        # fused batch*head dimension
    )

    _trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        s_l_b, s_l_i, s_l_k, s_l_h,
        s_r_b, s_r_j, s_r_k, s_r_h,
        s_o_b, s_o_i, s_o_j, s_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final projection back to `dim`
    # -------------------------------------------------------------------
    out = out.to(torch.float32)  # cast for stable layer‑norm

    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate.to(torch.float32)
    out = torch.nn.functional.linear(out, to_out_w)      # (B,N,N,dim)

    # -------------------------------------------------------------------
    # 8) Cast back to the original dtype
    # -------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
