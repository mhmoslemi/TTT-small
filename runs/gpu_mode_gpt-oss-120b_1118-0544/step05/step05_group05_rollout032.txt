@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, 
    stride_bm, stride_bn, # For original B (not transposed)
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BATCH: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    total_blocks_per_batch = num_blocks_m * num_blocks_n
    batch_idx = pid // total_blocks_per_batch
    block_idx = pid % total_blocks_per_batch
    block_m = block_idx // num_blocks_n
    block_n = block_idx % num_blocks_n

    # compute offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to batch start
    a_batch_ptr = a_ptr + batch_idx * batch_stride_a
    b_batch_ptr = b_ptr + batch_idx * batch_stride_b

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B block transposed: we need B^T, so rows correspond to N, cols correspond to K
        # Original B layout: rows=K, cols=N with stride_bm (row stride), stride_bn (col stride)
        # For B^T we need to load block (BLOCK_K, BLOCK_N) where rows=K, cols=N
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bm + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Multiply
        acc += tl.dot(a, b)  # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

        # advance pointers for next K block: not needed because we recompute each loop using k_base
        # but we could increment pointers, not needed.

    # store result
    c_ptrs = c_ptr + batch_idx * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

```
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for batched matrix multiplication where each batch corresponds to a hidden channel.
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,   # strides for A (row, col)
    stride_bm, stride_bn,   # strides for B (row, col) – we will read B as transposed
    stride_cm, stride_cn,   # strides for C (row, col)
    batch_stride_a, batch_stride_b, batch_stride_c,
    # compile‑time constants
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ---------------------------------------------------------------
    # Program ID layout:
    #   pid = batch * (num_blocks_m * num_blocks_n) + block_m * num_blocks_n + block_n
    # ---------------------------------------------------------------
    pid = tl.program_id(0)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    batch_idx = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    block_m = block_idx // num_blocks_n
    block_n = block_idx % num_blocks_n

    # ---------------------------------------------------------------
    # Offsets for the output tile
    # ---------------------------------------------------------------
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---------------------------------------------------------------
    # Pointers to the start of the current batch
    # ---------------------------------------------------------------
    a_batch_ptr = a_ptr + batch_idx * batch_stride_a
    b_batch_ptr = b_ptr + batch_idx * batch_stride_b

    # ---------------------------------------------------------------
    # Accumulator in float32 for accuracy
    # ---------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # Loop over the K dimension in BLOCK_K steps
    # ---------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile from A   (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load a tile from B transposed (BLOCK_K, BLOCK_N)
        # B is stored as (K, N) with stride_bm (row) and stride_bn (col)
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bm + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Dot product
        acc += tl.dot(a, b)

    # ---------------------------------------------------------------
    # Write back the result tile (cast to fp16)
    # ---------------------------------------------------------------
    c_ptrs = c_ptr + batch_idx * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    The implementation follows the reference PyTorch code but replaces the
    O(N³·H) contraction with a custom Triton batched GEMM kernel that
    computes, for each hidden channel, the matrix product
        out[..., i, j, h] = Σ_k left[..., i, k, h] * right[..., j, k, h].
    This kernel fuses the inner‑product across the sequence dimension and
    reduces launch overhead compared to a naïve `torch.bmm` loop.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # -----------------------------------------------------------------
    # Configuration
    # -----------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # Load weights (convert to FP16 for compute)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)

    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections
    # -----------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # -----------------------------------------------------------------
    # 3. Gating (sigmoid) and optional mask fusion
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    if mask is not None and not nomask:
        # mask: (B, N, N) -> (B, N, N, 1) for broadcasting
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left  = left * left_gate * mask_fp
        right = right * right_gate * mask_fp
    else:
        left  = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 4. Core contraction: batched GEMM over the sequence dimension
    # -----------------------------------------------------------------
    B, N, _, H = left.shape          # B, seq_len, seq_len, hidden_dim
    # Rearrange to (B*H, N, N) where each hidden channel becomes a batch
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Prepare output container
    out_mat = torch.empty_like(left_mat)   # (B*H, N, N), fp16

    # Strides (row‑major layout)
    stride_am = left_mat.stride(1)   # row stride = N
    stride_ak = left_mat.stride(2)   # col stride = 1
    stride_bm = right_mat.stride(1)  # same as above
    stride_bn = right_mat.stride(2)

    batch_stride_a = left_mat.stride(0)   # N*N
    batch_stride_b = right_mat.stride(0)
    batch_stride_c = out_mat.stride(0)

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = (B * H) * num_blocks_m * num_blocks_n
    grid = (total_programs,)

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_am, stride_ak,
        stride_bm, stride_bn,
        N, 1,                     # stride_cm, stride_cn for output (row‑major)
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5. Output LayerNorm, output gate, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # Cast back to the input dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation mirrors the reference PyTorch version but replaces the
O(N³·H) contraction with a custom Triton batched GEMM kernel.  For each
hidden channel the kernel computes

    out[..., i, j, h] = Σ_k left[..., i, k, h] * right[..., j, k, h]

by treating the hidden dimension as an additional batch dimension.
All heavy arithmetic is performed in float16 with float32 accumulation,
while the final result is cast back to the input dtype.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A strides (row, col)
    stride_bm, stride_bn,          # B strides (row, col) – we read B as transposed
    stride_cm, stride_cn,          # C strides (row, col)
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM where each batch corresponds to one hidden channel.
    Computes C = A @ Bᵀ with A,B ∈ ℝ^{M×K} and Bᵀ ∈ ℝ^{K×N}.
    """
    pid = tl.program_id(0)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    batch_idx = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    block_m = block_idx // num_blocks_n
    block_n = block_idx % num_blocks_n

    # Offsets for the output tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the beginning of the current batch
    a_batch = a_ptr + batch_idx * batch_stride_a
    b_batch = b_ptr + batch_idx * batch_stride_b

    # Accumulator in float32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B tile transposed: (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bm + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Dot product and accumulation
        acc += tl.dot(a, b)

    # Store the result tile (cast back to fp16)
    c_ptrs = c_ptr + batch_idx * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as the input.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # Load weights (cast to the precision used for compute)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)

    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability) → fp16 for compute
    # -----------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections
    # -----------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 3. Gating (sigmoid) and optional mask fusion
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left  = left * left_gate * mask_fp
        right = right * right_gate * mask_fp
    else:
        left  = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 4. Core contraction via custom Triton batched GEMM
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # Shape (B*H, N, N) where each hidden channel becomes a batch entry
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_mat = torch.empty_like(left_mat)  # (B*H, N, N), fp16

    # Strides (row‑major)
    stride_am = left_mat.stride(1)   # N
    stride_ak = left_mat.stride(2)   # 1
    stride_bm = right_mat.stride(1)  # N
    stride_bn = right_mat.stride(2)  # 1

    batch_stride_a = left_mat.stride(0)   # N * N
    batch_stride_b = right_mat.stride(0)
    batch_stride_c = out_mat.stride(0)

    stride_cm = out_mat.stride(1)   # N
    stride_cn = out_mat.stride(2)   # 1

    # -----------------------------------------------------------------
    # Launch configuration
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_progs = (B * H) * num_blocks_m * num_blocks_n
    grid = (total_progs,)

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_am, stride_ak,
        stride_bm, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # Cast back to the original dtype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
