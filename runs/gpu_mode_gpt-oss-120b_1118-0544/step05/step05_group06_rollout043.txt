    num_pid_n = tl.cdiv(N, BLOCK_N)

@triton.jit
def _trimul_core_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid = tl.program_id(0)
    pid_h = tl.program_id(1)   # hidden channel index

    # compute batch and tile indices
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_b = pid // (num_pid_m * num_pid_n)
    pid_rem = pid % (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n
    pid_n = pid_rem % num_pid_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for given batch and channel
    left_base = left_ptr + pid_b * stride_left_b + pid_h * stride_left_h
    right_base = right_ptr + pid_b * stride_right_b + pid_h * stride_right_h
    out_base = out_ptr + pid_b * stride_out_b + pid_h * stride_out_h

    # compute pointer offsets for left (i,k) and right (k,j)
    a_ptrs = left_base + offs_m[:, None] * stride_left_i + tl.arange(0, BLOCK_K)[None, :] * stride_left_k
    b_ptrs = right_base + tl.arange(0, BLOCK_K)[:, None] * stride_right_k + offs_n[None, :] * stride_right_j

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, N, BLOCK_K):
        # set mask for k dimension
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product across K dimension
        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # write output
    c = acc.to(tl.float16)

    out_ptrs = out_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
Highly‑optimized Triton‑accelerated forward pass for the outgoing TriMul
operator (AlphaFold3 style).

Algorithm
---------
1. Layer‑normalize the 4‑D input (B,N,N,C) with the provided LN weights.
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) are performed with FP16 weights.
4. If a mask is supplied, a small Triton kernel fuses
   `proj * gate * mask` (broadcast over the hidden dimension H).
5. Core reduction:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication per hidden channel.
   We implement this as a Triton GEMM where the hidden channel index is
   part of the program grid, thus avoiding any data copy or reshape.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate and project back to the original channel dimension.
7. Cast back to the input dtype (usually float32).

At least one operation (mask‑gate fusion) and the main O(N³·H) reduction
are executed in custom Triton kernels.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,                # *[total_elems]   fp16
    gate_ptr,                # *[total_elems]   fp16
    mask_ptr,                # *[mask_elems]    fp16
    out_ptr,                 # *[total_elems]   fp16
    total_elems,             # i64
    hidden_dim,              # i64  (size of the last dimension H)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # map flat index -> mask index (broadcast H)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Element‑wise ``proj * gate * mask`` where ``mask`` has shape (B, N, N)
    and is broadcast over the hidden dimension H.
    All tensors must be FP16 and contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],   # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM per hidden channel (core TriMul reduction)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_core_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes:
        out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
    where left/right/out are in FP16.  Accumulation is in FP32.
    The hidden channel ``h`` is taken from `tl.program_id(1)`.
    """
    pid = tl.program_id(0)          # encodes (batch, tile_i, tile_j)
    pid_h = tl.program_id(1)        # hidden channel index

    # ---- decode program ids ------------------------------------------------
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    pid_b = pid // (num_pid_m * num_pid_n)
    pid_rem = pid % (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n
    pid_n = pid_rem % num_pid_n

    # ---- offsets for the current tile --------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ---- base pointers for this batch and hidden channel --------------------
    left_base  = left_ptr  + pid_b * stride_left_b  + pid_h * stride_left_h
    right_base = right_ptr + pid_b * stride_right_b + pid_h * stride_right_h
    out_base   = out_ptr   + pid_b * stride_out_b   + pid_h * stride_out_h

    # ---- pointers to the K‑dimension tiles ---------------------------------
    a_ptrs = left_base + offs_i[:, None] * stride_left_i + tl.arange(0, BLOCK_K)[None, :] * stride_left_k
    b_ptrs = right_base + tl.arange(0, BLOCK_K)[:, None] * stride_right_k + offs_j[None, :] * stride_right_i

    # ---- accumulator in FP32 -----------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K --------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)          # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)          # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)               # sum over K

        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # ---- write result -------------------------------------------------------
    c = acc.to(tl.float16)
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])


def _core_trimul(left: torch.Tensor, right: torch.Tensor,
                 hidden_dim: int,
                 BLOCK_M: int = 16, BLOCK_N: int = 16, BLOCK_K: int = 32) -> torch.Tensor:
    """
    Launches the Triton kernel that performs the O(N³·H) reduction.
    """
    B, N, _, H = left.shape
    assert H == hidden_dim, "hidden dimension mismatch"

    out = torch.empty_like(left)

    # strides in element units (not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # grid: one program per (batch, tile_i, tile_j) and one program per hidden channel
    num_tiles_m = math.ceil(N / BLOCK_M)
    num_tiles_n = math.ceil(N / BLOCK_N)
    grid = (B * num_tiles_m * num_tiles_n, hidden_dim)

    _trimul_core_kernel[grid](
        left,
        right,
        out,
        B, N,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict containing all linear and LayerNorm parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for the heavy part
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all linear weights to FP16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections
    # --------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Apply mask + gate (Triton kernel) or just gate
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core O(N³·H) reduction via custom Triton GEMM
    # --------------------------------------------------------------
    out = _core_trimul(left, right, hidden_dim,
                       BLOCK_M=16, BLOCK_N=16, BLOCK_K=32)

    # --------------------------------------------------------------
    # 8. Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)          # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype (usually float32)
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
