@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    # maybe other consts
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # This ordering is arbitrary; for simplicity we can use program_id(0), (1), (2)
    # Let's assign batch id to pid_z (2)
    # Actually typical launch grid = (ceil(N/BLOCK_M), ceil(N/BLOCK_N), M)
    # So program_id(0) = pid_m (block row), program_id(1) = pid_n (block col), program_id(2) = pid_b (batch)
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_b = tl.program_id(2)  # batch

    # Compute starting indices
    start_m = pid_m * BLOCK_M
    start_n = pid_n * BLOCK_N

    # offsets for rows and cols
    offs_m = start_m + tl.arange(0, BLOCK_M)
    offs_n = start_n + tl.arange(0, BLOCK_N)

    # Compute masks for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for A and B at block start
    # A: shape (M,N,K) -> each block reads rows (offs_m) and k tile
    # B_T: shape (M,K,N) -> reads k tile and cols (offs_n)
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        # compute k index range
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile
        a_ptrs = a_ptr + pid_b * stride_am + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B_T tile
        b_ptrs = b_ptr + pid_b * stride_bm + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)  # returns (BLOCK_M, BLOCK_N) in float32

    # Write back results
    c_ptrs = c_ptr + pid_b * stride_cm + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1) LayerNorm input
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"].to(device), bias=weights["norm.bias"].to(device))
    # cast to fp16
    x = x.to(torch.float16)

    # 2) Cast all linear weights to fp16
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # 3) Linear projections
    left = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w) # (B,N,N,H)

    # 4) Compute gates and out_gate
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # 5) Apply mask and gates via Triton kernel
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_tensor = mask.to(torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    total_elements = left.numel()
    BLOCK = 4096  # threads per program
    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)
    grid = (triton.cdiv(total_elements, BLOCK),)
    _apply_mask_gate_kernel[grid](
        left, right,
        mask_tensor.view(-1),
        left_gate,
        right_gate,
        left_out,
        right_out,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # 6) Prepare for batched GEMM
    # rearrange to (B, H, N, N) then flatten batch dimension
    left_mat = left_out.permute(0, 3, 1, 2).contiguous()
    right_mat = right_out.permute(0, 3, 1, 2).contiguous()
    M = B * hidden_dim
    left_flat = left_mat.view(M, N, N)
    right_flat = right_mat.view(M, N, N)
    out_flat = torch.empty_like(left_flat)

    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

        right_flat,

        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # 9) Apply out gate
    out = out * out_gate

    out = F.linear(out, to_out_w)

    # 11) Cast back to original dtype
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)
    return out

    M, N, K,

    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_ci, stride_cj,

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C).
2. Project to hidden dimension H with two linear layers (left/right).
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. Fuse mask (if any) and left/right gates in a tiny Triton kernel.
5. Perform the core O(N³·H) reduction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (left @ rightᵀ) where the batch
   dimension is B·H.  Implemented with a custom Triton GEMM kernel.
6. Reshape to (B,N,N,H), apply a second LayerNorm over H, multiply by the
   output gate and project back to the original channel dimension.
7. Cast the result back to the original dtype.

Both the mask‑gate fusion and the batched GEMM are written in Triton, the
remaining linear layers and LayerNorms use PyTorch (highly‑optimized cuBLAS
and CUDA kernels).  The implementation works for any supported batch size,
sequence length and hidden dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask and left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,               # const fp16* left   (B,N,N,H) flattened
    right_ptr,              # const fp16* right
    mask_ptr,               # const fp16* mask   (B,N,N) flattened
    left_gate_ptr,          # const fp16* left_gate
    right_gate_ptr,         # const fp16* right_gate
    left_out_ptr,           # fp16* left_out
    right_out_ptr,          # fp16* right_out
    H: tl.constexpr,        # hidden dimension (compile‑time)
    TOTAL_ELEMS: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL_ELEMS

    # Load source tensors
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Broadcast mask over the hidden dimension
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Fuse mask and gates
    tl.store(left_out_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) where A,B ∈ (B*H,N,N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr,                # const fp16* A  (batch,M,K)  → (B*H,N,N)
    b_ptr,                # const fp16* B  (batch,N,K)  → (B*H,N,N) interpreted as (batch,K,N)
    c_ptr,                # fp16* C       (batch,M,N)
    BATCH: tl.constexpr,  # number of independent GEMMs = B*H
    M: tl.constexpr,      # rows per GEMM = N
    N: tl.constexpr,      # cols per GEMM = N
    K: tl.constexpr,      # reduction dim  = N
    stride_am, stride_ai, stride_ak,   # strides for A
    stride_bm, stride_bk, stride_bj,   # strides for B (interpreted as transposed)
    stride_cm, stride_ci, stride_cj,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # 3‑D grid: (row_tile, col_tile, batch)
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_b = tl.program_id(2)   # batch index

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Offsets for the current batch
    batch_offset_a = pid_b * stride_am
    batch_offset_b = pid_b * stride_bm
    batch_offset_c = pid_b * stride_cm

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + batch_offset_a + \
                 offs_m[:, None] * stride_ai + \
                 offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (transposed): shape (BLOCK_K, BLOCK_N)
        # B is stored as (batch,N,K); we interpret it as (batch,K,N)
        b_ptrs = b_ptr + batch_offset_b + \
                 offs_k[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # GEMM accumulation
        acc += tl.dot(a, b)

    # Write back the result in FP16
    c_ptrs = c_ptr + batch_offset_c + \
             offs_m[:, None] * stride_ci + \
             offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float16/float32)
            - mask        : torch.Tensor of shape [B, N, N] or None
            - weights     : dict of model parameters (float32 tensors)
            - config      : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ----------------------------------------------------------
    # Unpack
    # ----------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------
    # 1) Input LayerNorm (float32 precision)
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)  # work in fp16 for the heavy part

    # ----------------------------------------------------------
    # 2) Cast linear weights to fp16
    # ----------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ----------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ----------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w)  # (B,N,N,H)

    # ----------------------------------------------------------
    # 4) Compute sigmoid gates
    # ----------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,H)

    # ----------------------------------------------------------
    # 5) Apply mask and gates (fused Triton kernel)
    # ----------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_tensor = mask.to(torch.float16)
    else:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)

    total_elems = left.numel()                # B*N*N*H
    BLOCK = 4096                               # threads per program (tuned)

    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = (triton.cdiv(total_elems, BLOCK),)
    _apply_mask_gate_kernel[grid](
        left, right,
        mask_tensor.view(-1),
        left_gate,
        right_gate,
        left_out,
        right_out,
        hidden_dim,
        total_elems,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ----------------------------------------------------------
    # 6) Core reduction: batched GEMM (left @ rightᵀ)
    #    Shape after view: (B*H, N, N)
    # ----------------------------------------------------------
    # Rearrange to (B, H, N, N) and flatten batch dimension
    left_mat = left_out.permute(0, 3, 1, 2).contiguous()
    right_mat = right_out.permute(0, 3, 1, 2).contiguous()

    batch = B * hidden_dim                     # B*H
    left_flat = left_mat.view(batch, N, N)     # (batch, N, N)
    right_flat = right_mat.view(batch, N, N)   # (batch, N, N)
    out_flat = torch.empty_like(left_flat)     # (batch, N, N)

    # Strides (all tensors are contiguous)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    # B is interpreted as transposed (K,N)
    stride_bk = right_flat.stride(2)   # original third dim -> K
    stride_bj = right_flat.stride(1)   # original second dim -> N

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # GEMM tile sizes (tuned for A100/B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch,                     # independent GEMMs
    )
    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch,
        N, N, N,                  # M, N, K (all equal to seq_len)
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ----------------------------------------------------------
    # 7) Reshape back to (B, N, N, H)
    # ----------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------
    # 8) Output LayerNorm over hidden dimension
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----------------------------------------------------------
    # 9) Apply output gate
    # ----------------------------------------------------------
    out = out * out_gate

    # ----------------------------------------------------------
    # 10) Final linear projection back to original dim
    # ----------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ----------------------------------------------------------
    # 11) Cast back to the original dtype
    # ----------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
```
