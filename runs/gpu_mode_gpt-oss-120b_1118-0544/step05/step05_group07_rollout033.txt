def layernorm_gate_kernel(
    inp_ptr,   # fp16* (total_vectors * hidden_dim)
    out_ptr,  # fp16* (total_vectors * hidden_dim)
    weight_ptr, # fp16* (hidden_dim,)
    bias_ptr,   # fp16* (hidden_dim,)
    gate_ptr,   # fp16* (total_vectors * hidden_dim)
    hidden_dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    # Each program handles one vector (size hidden_dim)
    vec_id = tl.program_id(0)
    offset = vec_id * hidden_dim
    # Load a block of hidden_dim elements; assume hidden_dim <= BLOCK_SIZE
    # Use range
    idx = tl.arange(0, BLOCK_SIZE)
    mask = idx < hidden_dim
    x = tl.load(inp_ptr + offset + idx, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offset + idx, mask=mask, other=0.0)

    # Compute mean
    sum_x = tl.sum(x, axis=0)  # sum of elements
    mean = sum_x / hidden_dim

    # Compute variance
    diff = x - mean
    var = tl.sum(diff * diff, axis=0) / hidden_dim

    norm = (x - mean) / denom

    b = tl.load(bias_ptr + idx, mask=mask, other=0.0)

    tl.store(out_ptr + offset + idx, out, mask=mask)

"""
TriMul (outgoing) forward pass – Triton‑fused mask+gate and LayerNorm+gate.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) (float32) → fp16.
2. Fuse the two linear projections (left/right) into a single matmul, and
   similarly fuse the three gate linear layers.
3. Apply mask (if present) and element‑wise gate via a tiny Triton kernel.
4. Compute the core O(N³·H) reduction:
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d].
   Implemented as a batched GEMM (torch.bmm) over the combined batch
   dimension (B·H).
5. Fuse the second LayerNorm, its affine parameters and the output gate
   into one Triton kernel.
6. Final linear projection back to the original channel dimension.
7. Cast the result back to the input dtype.

Both the mask‑gate fusion and the final norm‑gate fusion are executed on
GPU with Triton, while the large matrix multiplication uses cuBLAS/Tensor
Core via torch.bmm for maximum throughput.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Kernel: proj * gate * mask (mask is broadcast over the hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (total_elems,)
    gate_ptr,          # fp16 * (total_elems,)
    mask_ptr,          # fp16 * (mask_elems,)
    out_ptr,           # fp16 * (total_elems,)
    total_elems,       # i32
    hidden_dim: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # map flat offset to mask offset (mask is not expanded over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(out_ptr + offs, proj * gate * mask_val, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """Fuse `proj * gate * mask` via Triton. All tensors must be fp16 and contiguous."""
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],               # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel: LayerNorm (affine) + output gate
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_gate_kernel(
    inp_ptr,       # fp16 * (vecs * hidden_dim,)
    out_ptr,       # fp16 * (vecs * hidden_dim,)
    weight_ptr,    # fp16 * (hidden_dim,)
    bias_ptr,      # fp16 * (hidden_dim,)
    gate_ptr,      # fp16 * (vecs * hidden_dim,)
    HIDDEN_DIM: tl.constexpr,
    EPS: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    vec_id = tl.program_id(0)                 # each program works on one vector (B*N*N)
    offset = vec_id * HIDDEN_DIM
    idx = tl.arange(0, BLOCK_SIZE)
    mask = idx < HIDDEN_DIM

    # load input and gate
    x = tl.load(inp_ptr + offset + idx, mask=mask, other=0.0).to(tl.float32)
    g = tl.load(gate_ptr + offset + idx, mask=mask, other=0.0).to(tl.float32)

    # mean
    mean = tl.sum(x, axis=0) / HIDDEN_DIM
    # variance
    diff = x - mean
    var = tl.sum(diff * diff, axis=0) / HIDDEN_DIM
    # normalize
    norm = (x - mean) / tl.sqrt(var + EPS)

    # affine
    w = tl.load(weight_ptr + idx, mask=mask, other=0.0).to(tl.float32)
    b = tl.load(bias_ptr + idx, mask=mask, other=0.0).to(tl.float32)
    norm = norm * w + b

    # apply output gate
    out = norm * g

    # store as fp16
    tl.store(out_ptr + offset + idx, out.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, C]
            - mask  : torch.Tensor [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (dtype matches input)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Combined linear projections
    # ------------------------------------------------------------------
    # left/right projections
    proj_weight = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(torch.float16)                     # (2*hidden_dim, dim)
    proj = torch.nn.functional.linear(x, proj_weight)   # (B,N,N,2*hidden_dim)
    left_proj, right_proj = torch.split(proj, hidden_dim, dim=-1)

    # combined gates (left_gate, right_gate, out_gate)
    gate_weight = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # (3*hidden_dim, dim)
    gate = torch.nn.functional.linear(x, gate_weight)   # (B,N,N,3*hidden_dim)
    left_gate, right_gate, out_gate = torch.split(gate, hidden_dim, dim=-1)
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3. Mask & gate fusion (Triton) – optional
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4. Core O(N³·H) reduction via batched GEMM
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # (B, H, N, N) → (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N)

    # batched matmul: left @ rightᵀ
    out = torch.bmm(left, right.transpose(1, 2))          # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Second LayerNorm + output gate (fused Triton kernel)
    # ------------------------------------------------------------------
    total_vectors = B * N * N
    out_flat = out.view(-1, hidden_dim).contiguous()
    out_gate_flat = out_gate.view(-1, hidden_dim).contiguous()
    out_norm_gate = torch.empty_like(out_flat)

    grid = (total_vectors,)
    _layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"].to(torch.float16),
        weights["to_out_norm.bias"].to(torch.float16),
        out_gate_flat,
        HIDDEN_DIM=hidden_dim,
        EPS=1e-5,
        BLOCK_SIZE=hidden_dim,
    )
    # reshape to (B,N,N,hidden_dim)
    out = out_norm_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 6. Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)        # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 7. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
