"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input (float32) → x.
2. Cast x to fp16 and compute linear projections:
   left, right (hidden), left_gate, right_gate, out_gate (all fp16).
3. Apply optional mask and fuse with gates (element‑wise).
4. Perform the core contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   via a custom Triton batched GEMM kernel that treats each (batch, d)
   pair as a separate GEMM (shape N×N).  The kernel uses tiling over
   M/N/K dimensions, accumulates in fp32 and writes fp32 results.
5. Reshape the result back to (B,N,N,hidden), apply a second LayerNorm,
   multiply by out_gate and a final linear projection back to the original
   channel dimension.
6. Return the result in the original dtype (float32).

The heavy O(N³·H) work is fully handled by the Triton kernel, while the
lighter linear & norm operations are performed with PyTorch (cuBLAS/Tensor
Cores).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to A, B (transposed), C
    M, N, K,                           # matrix sizes (all = seq_len)
    stride_ab, stride_am, stride_ak,   # strides for A: batch, row, col
    stride_bb, stride_bk, stride_bn,   # strides for B (already transposed)
    stride_cb, stride_cm, stride_cn,   # strides for C: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_b = tl.program_id(2)  # batch (combined batch*hidden)

    # offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bb
    c_batch = c_ptr + pid_b * stride_cb

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # compute pointers for the tiles
        a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        # load tiles (fp16)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # dot product accumulates into fp32
        acc += tl.dot(a, b)

    # store the result (fp32)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing:
                * "dim" (int)           – input channel dimension C
                * "hidden_dim" (int)    – hidden dimension H
                * "nomask" (bool, opt) – if True, ignore mask

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # Load model parameters onto the device and appropriate dtype
    # ------------------------------------------------------------------ #
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)

    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------ #
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)
    x = x.to(torch.float16)  # cast to half for the heavy path

    # ------------------------------------------------------------------ #
    # 2. Linear projections (bias‑less)
    # ------------------------------------------------------------------ #
    left  = F.linear(x, left_proj_w)    # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)   # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------ #
    # 3. Apply optional mask and fuse with gates
    # ------------------------------------------------------------------ #
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left  = left  * left_gate  * mask_fp
        right = right * right_gate * mask_fp
    else:
        left  = left  * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------ #
    # 4. Core O(N³·H) contraction via custom Triton batched GEMM
    # ------------------------------------------------------------------ #
    B, N_seq, _, H = left.shape  # N_seq == N
    # reshape to (B*H, N, N) for batched GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N_seq, N_seq)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N_seq, N_seq)

    # right matrix transposed view (no extra copy)
    right_mat_t = right_mat.transpose(-2, -1)  # shape (B*H, N, N) with swapped strides

    # output buffer (fp32 accumulator)
    out_mat = torch.empty((B * H, N_seq, N_seq), dtype=torch.float32, device=device)

    # strides for the kernel
    stride_ab = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_bb = right_mat_t.stride(0)
    stride_bk = right_mat_t.stride(1)
    stride_bn = right_mat_t.stride(2)

    stride_cb = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # Tiling parameters (tuned for typical GPUs)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N_seq, BLOCK_M),
        triton.cdiv(N_seq, BLOCK_N),
        B * H,
    )
    batched_matmul_kernel[grid](
        left_mat,
        right_mat_t,
        out_mat,
        N_seq,
        N_seq,
        N_seq,
        stride_ab,
        stride_am,
        stride_ak,
        stride_bb,
        stride_bk,
        stride_bn,
        stride_cb,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------ #
    # 5. Reshape back to (B,N,N,hidden_dim)
    # ------------------------------------------------------------------ #
    out = out_mat.view(B, H, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------ #
    # 6. Output LayerNorm, gating, and final projection
    # ------------------------------------------------------------------ #
    # LayerNorm on hidden dimension (use half precision for speed)
    out = out.to(torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    out = out * out_gate  # apply output gate (both half)

    # Final linear projection back to original channel dimension
    out = F.linear(out, to_out_w)  # (B,N,N,dim) in half

    # ------------------------------------------------------------------ #
    # 7. Cast back to the original dtype and return
    # ------------------------------------------------------------------ #
    if out.dtype != dtype_in:
        out = out.to(dtype_in)
    return out
