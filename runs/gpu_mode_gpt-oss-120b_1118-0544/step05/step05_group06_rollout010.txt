"""
TriMul (outgoing) forward pass – highly‑optimized Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑normalize the 4‑D input (B, N, N, C) (FP32 → FP16).
2. Linear projections to a hidden dimension H (FP16 weights) for left/right streams.
3. Compute per‑element gates (sigmoid of linear projections) in FP16.
4. Fuse element‑wise multiplication of projection, gate and optional mask
   with a custom Triton kernel (mask broadcast over H).
5. Core reduction:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (left @ rightᵀ) for every
   (b, h) pair.  We use torch.matmul on tensors shaped (B, H, N, N) which
   invokes cuBLAS tensor‑core GEMM.
6. Second LayerNorm over the hidden dimension, multiplication by the output
   gate and a final linear projection back to the original channel dimension.
7. Cast the result back to the input dtype (usually float32).

Only the gate‑mask fusion is performed in Triton; all heavy linear algebra
uses highly‑optimized PyTorch (cuBLAS) kernels.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,            # *[total_elements]   fp16
    gate_ptr,            # *[total_elements]   fp16
    mask_ptr,            # *[mask_elements]    fp16
    out_ptr,             # *[total_elements]   fp16
    total_elems,         # i64
    hidden_dim,          # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds accesses
    mask = offs < total_elems

    # Because mask has shape (B, N, N) we need to map each flat index
    # to the corresponding mask element.  Hidden dimension H is the
    # innermost stride, so the mask index is floor(offs / H).
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Element‑wise ``proj * gate * mask`` where ``mask`` has shape (B, N, N)
    and is broadcast over the hidden dimension H.
    All tensors must be contiguous FP16.
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],   # hidden_dim == H
        BLOCK_SIZE=block_size,
    )
    return out

# ----------------------------------------------------------------------

        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Cast weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (B, N, N, H)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Compute gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse projection, gate and mask (if present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core reduction: batched matrix multiplication per hidden channel
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as torch.matmul on (B, H, N, N) tensors.
    # ------------------------------------------------------------------
    # move hidden dim to second axis for batched GEMM
    left = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2) # (B, H, N, N)

    # matrix multiplication with right transposed on the last two axes
    out = torch.matmul(left, right.transpose(-1, -2))  # (B, H, N, N)

    # restore original layout
    out = out.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, C)

    # ------------------------------------------------------------------
    # 8. Cast back to original dtype
    # ------------------------------------------------------------------
    return out.to(dtype)
```
