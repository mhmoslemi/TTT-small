"""
TriMul (outgoing) forward pass – Triton‑accelerated mask & gate fusion
with a single fused linear projection.

Algorithm
---------
1. Layer‑norm over the input channel dimension `dim`.
2. Cast to FP16 for compute‑heavy ops.
3. Concatenate the five weight matrices (left/right proj, left/right gate,
   out gate) into one large matrix and perform a single linear projection:
   y = x @ W_all.T   -> shape (B, N, N, 5*hidden_dim)
   Split y into the five tensors needed.
4. Apply sigmoid to the three gate tensors.
5. Fuse element‑wise mask, gate and projection for `left` and `right`
   using a small Triton kernel (if a mask is present).
6. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (torch.bmm) after reshaping to
   (B*hidden_dim, N, N).
7. Output LayerNorm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.
8. Cast back to the original dtype.

The heavy linear work is reduced from 5 separate matmuls to a single one,
and the mask‑gate fusion avoids extra temporaries, giving a fast
implementation suitable for the B200 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: (proj * gate * mask) – mask broadcast over hidden_dim
# --------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # int: B*N*N*H
    hidden_dim,        # int: H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard against OOB
    valid = offs < total_elems

    # Index of the mask element (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 2048) -> torch.Tensor:
    """
    Fuse projection, gate and mask using a Triton kernel.
    All tensors must be fp16, contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)   (binary 0/1, broadcast over H)
    Returns proj * gate * mask (shape (B,N,N,H)).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out

            - config       : dict with keys
                  * "dim" – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Concatenate all linear weights (5 * hidden_dim output)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Each weight: (hidden_dim, dim)
    weight_all = torch.cat(
        [left_proj_w, right_proj_w,
         left_gate_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(torch.float16)  # shape (5*hidden_dim, dim)

    # ------------------------------------------------------------------
    # 4. Single fused linear projection
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    x_flat = x.reshape(B * N * N, dim)          # (B*N*N, dim)
    y_flat = F.linear(x_flat, weight_all)       # (B*N*N, 5*hidden_dim)
    y = y_flat.view(B, N, N, 5 * hidden_dim)    # (B, N, N, 5*H)

    # ------------------------------------------------------------------
    # 5. Split and apply sigmoid to gates
    # ------------------------------------------------------------------
    left_proj  = y[..., 0:hidden_dim]
    right_proj = y[..., hidden_dim:2 * hidden_dim]
    left_gate  = y[..., 2 * hidden_dim:3 * hidden_dim].sigmoid()
    right_gate = y[..., 3 * hidden_dim:4 * hidden_dim].sigmoid()
    out_gate   = y[..., 4 * hidden_dim:5 * hidden_dim].sigmoid()

    # ------------------------------------------------------------------
    # 6. Mask & gate fusion (Triton) or simple elementwise multiply
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16, contiguous, shape (B,N,N)
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype (float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)
    return out
