def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', False)
    # use same device
    device = input_tensor.device
    # Cast and compute
    # 1. LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(input_tensor.dtype),
        bias=weights["norm.bias"].to(input_tensor.dtype))
    # 2. Cast x to half for compute (if not already half)
    x_fp16 = x.to(torch.float16)
    # 3. Prepare combined weights
    left_concat_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    right_concat_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    # 4. Projections
    left_concat = torch.nn.functional.linear(x_fp16, left_concat_weight)   # (B,N,N,2*H)
    right_concat = torch.nn.functional.linear(x_fp16, right_concat_weight) # (B,N,N,2*H)

    left_proj = left_concat[..., :hidden_dim]
    left_gate = left_concat[..., hidden_dim:].sigmoid()
    right_proj = right_concat[..., :hidden_dim]
    right_gate = right_concat[..., hidden_dim:].sigmoid()

    # 5. Apply mask and gating via Triton or elementwise
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16)
        left, right = _apply_gate_mask_triton(left_proj, left_gate, right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 6. Compute out_gate
    out_gate = torch.nn.functional.linear(x_fp16, out_gate_weight).sigmoid()

    # 7. Reduction: out = sum_k left[:,i,k,:] * right[:,j,k,:]
    B, N, _, H = left.shape
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 8. Output norm, gate, final projection
    out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(out,
        normalized_shape=(hidden_dim,),
        weight=out_norm_weight,
        bias=out_norm_bias)
    out = out * out_gate

    out = torch.nn.functional.linear(out, to_out_weight)

    # 9. Cast back
    out = out.to(input_tensor.dtype)

"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to fp16 for all compute‑heavy ops.
3. Fuse left/right linear projections and their gates:
      * concat(left_proj, left_gate)  → weight (2*H, C)
      * concat(right_proj, right_gate) → weight (2*H, C)
   One `torch.nn.functional.linear` call yields both projection and pre‑gate
   values; the gate part is passed through a sigmoid.
4. If a mask is present, a Triton kernel fuses
      left = left_proj * left_gate * mask
      right = right_proj * right_gate * mask
   (mask is broadcast over the hidden dimension).
   Without a mask the element‑wise product is done with PyTorch.
5. Core multiplicative update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM using `torch.bmm` on the view
   (B*H, N, N).
6. Output layer‑norm, multiply by `out_gate` (computed from the same
   normalized input), and a final linear projection back to the original
   channel dimension.
7. Cast the result back to the input dtype.

A single Triton kernel is used for the mask‑gate fused multiplication,
fulfilling the “use a kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse left/right projection‑gate‑mask multiplication.

    Arguments
    ---------
    left_proj_ptr, left_gate_ptr, right_proj_ptr, right_gate_ptr :
        pointers to fp16 tensors of shape (B, N, N, H) each.
    mask_ptr : pointer to fp16 mask of shape (B, N, N) (broadcast over H).
    left_out_ptr, right_out_ptr : output pointers (same shape as proj tensors).
    total_elems : total number of elements in a projection tensor (B*N*N*H).
    hidden_dim : H (size of the hidden dimension).
    BLOCK_SIZE : compile‑time constant – threads per program.
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Compute the corresponding mask index (broadcast over H)
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_gate_mask(left_proj, left_gate, right_proj, right_gate, mask, block_size: int = 4096):
    """
    Apply (proj * gate * mask) for left and right tensors using Triton.
    All tensors must be fp16 and contiguous.

    Returns
    -------
    left_out, right_out : torch.Tensor of shape (B, N, N, H)
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_fuse_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out

        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(dtype_in).to(device),
        bias=weights["norm.bias"].to(dtype_in).to(device),
    )

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy part
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Prepare concatenated projection+gate weights (fp16)
    # -----------------------------------------------------------------
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16).to(device)          # (2*H, C)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16).to(device)          # (2*H, C)
    out_gate_w = weights["out_gate.weight"].to(torch.float16).to(device)  # (H, C)

    # -----------------------------------------------------------------
    # 4. Linear projections + gate logits (single matmul per side)
    # -----------------------------------------------------------------
    left_concat = F.linear(x, left_concat_w)      # (B, N, N, 2*H)
    right_concat = F.linear(x, right_concat_w)    # (B, N, N, 2*H)

    left_proj = left_concat[..., :hidden_dim]                     # (B,N,N,H)
    left_gate = left_concat[..., hidden_dim:].sigmoid()           # (B,N,N,H)

    right_proj = right_concat[..., :hidden_dim]                   # (B,N,N,H)
    right_gate = right_concat[..., hidden_dim:].sigmoid()         # (B,N,N,H)

    # -----------------------------------------------------------------
    # 5. Apply mask and gating (fused via Triton when mask is present)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_gate_mask(
            left_proj.contiguous(),
            left_gate.contiguous(),
            right_proj.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
            block_size=4096,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 6. Out‑gate (separate linear + sigmoid)
    # -----------------------------------------------------------------
    out_gate = F.linear(x, out_gate_w).sigmoid()      # (B,N,N,H)

    # -----------------------------------------------------------------
    # 7. Core multiplicative update (batched GEMM)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate scaling and final projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16).to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16).to(device)
    to_out_w = weights["to_out.weight"].to(torch.float16).to(device)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)      # (B, N, N, C)

    # -----------------------------------------------------------------
    # 9. Cast back to the original dtype
    # -----------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
