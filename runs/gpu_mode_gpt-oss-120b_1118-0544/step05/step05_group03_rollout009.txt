"""
TriMul (outgoing) forward pass.
- LayerNorm on input channels.
- Linear projections & gate projections are computed in FP16 on the flattened (B*N*N) matrix.
- Mask, gate and optional mask multiplication are fused in a Triton kernel.
- Core reduction (∑_k left[b,i,k,h] * right[b,j,k,h]) is performed as a batched GEMM
  using torch.bmm (leverages cuBLAS tensor cores).
- Output LayerNorm, out‑gate multiplication and final linear projection back to the
  original dimension.
The final result is cast back to the input dtype (typically float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask (broadcast over hidden_dim)
# --------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,   # fp16* (total_elems,)
    gate_ptr,   # fp16* (total_elems,)
    mask_ptr,   # fp16* (mask_elems,)
    out_ptr,    # fp16* (total_elems,)
    total_elems: tl.int64,
    hidden_dim: tl.int64,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # guard
    mask = offs < total_elems

    # each hidden channel shares the same mask element
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * mval
    tl.store(out_ptr + offs, out, mask=mask)

def _apply_gate_mask(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor, block_size: int = 16384) -> torch.Tensor:
    """
    proj, gate : (B, N, N, H)  fp16, contiguous
    mask       : (B, N, N)     fp16, contiguous
    Returns proj * gate * mask (broadcast mask over H).
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

# --------------------------------------------------------------

    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (float32) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim" and "hidden_dim" (int), optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm over channel dimension `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)  float32

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy path
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast weights to FP16
    # --------------------------------------------------------------
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out = weights["to_out.weight"].to(torch.float16)

    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 4. Flatten for linear ops (B*N*N, dim)
    # --------------------------------------------------------------
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)

    # --------------------------------------------------------------
    # 5. Linear projections and gate pre‑activations (all FP16)
    # --------------------------------------------------------------
    left_proj = F.linear(x_flat, w_left).view(B, N, N, hidden_dim)
    right_proj = F.linear(x_flat, w_right).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, w_left_gate).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, w_right_gate).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, w_out_gate).view(B, N, N, hidden_dim).sigmoid()

    # --------------------------------------------------------------
    # 6. Fuse mask and gate (Triton) or simple elementwise mul
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
    )
    out = out * out_gate

    # final linear back to original dim
    out_flat2 = out.view(-1, hidden_dim)                     # (B*N*N, H)
    out = F.linear(out_flat2, w_to_out).view(B, N, N, dim)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)

    return out
