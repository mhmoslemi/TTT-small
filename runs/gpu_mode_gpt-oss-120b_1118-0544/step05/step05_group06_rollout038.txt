def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # program id axes
    pid = tl.program_id(axis=0)  # batch id (B*H)
    pid_m = tl.program_id(axis=1)  # block row (i)
    pid_n = tl.program_id(axis=2)  # block col (j)

    # compute offsets for block start
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    # compute base pointers for left and right (transposed)
    # left[pid, i, k]
    left_ptrs = left_ptr + pid * stride_left_b + i[:, None] * stride_left_i + k[None, :] * stride_left_j
    # right[pid, j, k] where we want right[j,k] (no transpose). We'll load as right[j, k].
    # But we need right^T, which means we need to load as right[pid, k, j] or load right[k, j] with swapped dims.
    # Let's load right tile as right[pid, k, j] by swapping the offsets.
    right_ptrs = right_ptr + pid * stride_right_b + k[:, None] * stride_right_i + j[None, :] * stride_right_j
    # The shape of left tile: (BLOCK_M, BLOCK_K), right tile: (BLOCK_K, BLOCK_N)
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        # load tiles with masking for bounds
        left = tl.load(left_ptrs, mask=(i < N)[:, None] & (k < N)[None, :], other=0.0)

        # dot product
        acc += tl.dot(left, right)
        # increment pointers
        left_ptrs += BLOCK_K * stride_left_j
        right_ptrs += BLOCK_K * stride_right_i

        k += BLOCK_K   # ??? cannot modify k directly after loop; instead we compute offset differently each iteration.

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
import math

# Kernel for elementwise projection * gate * mask
@triton.jit
def _gate_mask_kernel(
    proj_ptr,    # *[total]
    gate_ptr,    # *[total]
    mask_ptr,    # *[mask_elems]
    out_ptr,     # *[total]
    total_elems, # i64
    hidden_dim,  # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # broadcasting mask over hidden_dim
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(out_ptr + offs, proj * gate * m, mask=mask)

def _apply_gate_and_mask(proj, gate, mask, block_size=1024):
    """
    proj, gate : fp16 tensors of shape [B, N, N, H]
    mask       : fp16 tensor of shape [B, N, N] (no trailing dim)
    returns elementwise proj*gate*mask broadcasted over H
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

# Core TriMul reduction kernel: out = left @ right^T
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_h, N,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)          # batch * head
    pid_m = tl.program_id(1)        # block row (i)
    pid_n = tl.program_id(2)        # block col (j)

    # block start indices
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for i/j within bounds
    mask_i = i < N
    mask_j = j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_base = k_iter * BLOCK_K
        k_idx = k_base + tl.arange(0, BLOCK_K)

        # mask for k within bounds
        mask_k = k_idx < N

        # pointers for this tile
        left_tile_ptr  = left_ptr  + pid * stride_l_batch \
                         + i[:, None] * stride_l_i \
                         + k_idx[None, :] * stride_l_k
        right_tile_ptr = right_ptr + pid * stride_r_batch \
                         + k_idx[:, None] * stride_r_i \
                         + j[None, :] * stride_r_k

        left = tl.load(left_tile_ptr,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)
        right = tl.load(right_tile_ptr,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # dot (accumulate in FP32)
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # store result (convert back to FP16)
    out_tile_ptr = out_ptr + pid * stride_o_batch \
                   + i[:, None] * stride_o_i \
                   + j[None, :] * stride_o_k
    tl.store(out_tile_ptr,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update
    (TriMul) from AlphaFold3.

    The implementation:
    1. Layer‑norm the input.
    2. FP16 linear projections and gate logits.
    3. Fuse projection * gate * (optional) mask with a tiny Triton kernel.
    4. Compute the O(N³·H) reduction with a custom batched GEMM kernel
       (left @ rightᵀ) where H = hidden_dim.
    5. Apply output LayerNorm, out‑gate, and final linear projection.
    6. Cast back to the original dtype.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- 1. Input LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- 2. Cast to FP16 ----------
    x = x.to(torch.float16)

    # ---------- 3. Cast weights ----------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ---------- 4. Linear projections ----------
    left_proj = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ---------- 5. Gate logits ----------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # ---------- 6. Apply mask & gating ----------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ---------- 7. Core reduction ----------
    # reshape to (B*H, N, N)
    B, N, _, H = left.shape
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                 # batch*head dimension
        (N + BLOCK_M - 1) // BLOCK_M,          # i blocks
        (N + BLOCK_N - 1) // BLOCK_N,          # j blocks
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # restore layout (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- 8. Output LayerNorm, gating and final projection ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)

    # ---------- 9. Cast back ----------
    out = out.to(input_tensor.dtype)
    return out
```

"""
Highly‑optimized Triton forward for the outgoing TriMul (AlphaFold3).

Algorithm
---------
1. Layer‑norm the 4‑D input.
2. Cast to FP16 for the compute‑intensive path.
3. Linear projections (`left_proj`, `right_proj`) and gate logits
   (`left_gate`, `right_gate`, `out_gate`).
4. Fuse element‑wise `proj * gate * mask` (mask is broadcast over the hidden
   dimension) with a tiny Triton kernel.
5. Core O(N³·H) reduction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM kernel (left @ rightᵀ) that processes the
   (B·H) batches in parallel.
6. Output LayerNorm → multiply by `out_gate` → final linear projection.
7. Cast back to the original dtype (usually float32).

Only the element‑wise mask‑gate fuse and the massive reduction are
implemented in Triton; all other ops use PyTorch’s highly‑optimized kernels.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Element‑wise proj * gate * mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # *[total]    fp16
    gate_ptr,          # *[total]    fp16
    mask_ptr,          # *[mask]     fp16
    out_ptr,           # *[total]    fp16
    total_elems,       # i64
    hidden_dim,        # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # broadcast mask over the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=active, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=active, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)

    tl.store(out_ptr + offs, proj * gate * m, mask=active)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Fuse ``proj * gate * mask`` where ``mask`` has shape (B, N, N) and is
    broadcast over the hidden dimension H. All tensors must be fp16 and
    contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)          # (B*N*N,)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# 2️⃣  Core reduction kernel: out = left @ rightᵀ  (batched GEMM)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_h, N,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)          # batch * head index
    pid_m = tl.program_id(1)        # block row (i)
    pid_n = tl.program_id(2)        # block col (j)

    # coordinates of the tile
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator in FP32 for numeric stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_base = k_iter * BLOCK_K
        k_idx = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # pointers for this k‑tile
        left_tile_ptr = left_ptr + pid * stride_l_batch \
                        + i[:, None] * stride_l_i \
                        + k_idx[None, :] * stride_l_k
        right_tile_ptr = right_ptr + pid * stride_r_batch \
                         + k_idx[:, None] * stride_r_i \
                         + j[None, :] * stride_r_k

        left = tl.load(left_tile_ptr,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)
        right = tl.load(right_tile_ptr,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        acc += tl.dot(left, right)          # FP16 dot → accumulates in FP32

    out_tile_ptr = out_ptr + pid * stride_o_batch \
                   + i[:, None] * stride_o_i \
                   + j[None, :] * stride_o_k
    tl.store(out_tile_ptr,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (may be None)
        - weights      : dict with all linear / LayerNorm parameters
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------- 1. Input LayerNorm -------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------- 2. FP16 cast -------------------------
    x = x.to(torch.float16)

    # ------------------- 3. FP16 weights ----------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------- 4. Linear projections ----------------
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------- 5. Gate logits -----------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # ------------------- 6. Apply mask & gating ---------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate          # (B, N, N, H)
        right = right_proj * right_gate

    # ------------------- 7. Core reduction --------------------
    B, N, _, H = left.shape                     # left: (B, N, N, H)

    # reshape to (B*H, N, N) → contiguous layout
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)      # FP16 output buffer

    # block sizes (tuned for N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                   # batch*head dimension
        (N + BLOCK_M - 1) // BLOCK_M,            # i‑blocks
        (N + BLOCK_N - 1) // BLOCK_N,            # j‑blocks
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Restore (B, N, N, H) layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- 8. Output LayerNorm & projection ----
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)               # (B, N, N, dim)

    # ------------------- 9. Cast back ------------------------
    out = out.to(input_tensor.dtype)
    return out
