@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)  # combined batch*hidden
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    batch_idx = pid // H
    hidden_idx = pid % H

    # compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # out of bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # compute base pointers
    left_base = left_ptr + batch_idx * stride_left_b + hidden_idx * stride_left_h
    right_base = right_ptr + batch_idx * stride_right_b + hidden_idx * stride_right_h
    out_base = out_ptr + batch_idx * stride_out_b + hidden_idx * stride_out_h

    # offsets for left: rows i, columns k
    # offsets for right: rows k (original i), columns j (original j) after transpose
    # We'll load k blocks
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        # Column indices for left (k)
        offs_k = k_start + tl.arange(0, BLOCK_K)
        # mask for k dimension
        mask_k = offs_k < N

        # Load A (left) tile
        a_ptrs = left_base + (offs_m[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B (right transposed) tile: we need element right[b, j, k] as B[k,j]
        # For B we need row index k (value from offs_k) and column index j (offs_n)
        # Offsets: k as original i dimension (stride_right_i) and j as original k dimension (stride_right_k)
        b_ptrs = right_base + (offs_n[:, None] * stride_right_k) + (offs_k[None, :] * stride_right_i)
        # Wait shape: we want B[k,j] where k is row (original i) and j is column (original k)
        # So row offset: k * stride_right_i (since original i dimension stride), column offset: j * stride_right_k (original k stride)
        # So we compute pointer: base + k * stride_right_i + j * stride_right_k
        # But using broadcasting we want shape (BLOCK_K, BLOCK_N). So we need to create pointer [k, j].
        # We'll compute b_ptrs = right_base + (offs_k[:, None] * stride_right_i) + (offs_n[None, :] * stride_right_k)
        b_ptrs = right_base + (offs_k[:, None] * stride_right_i) + (offs_n[None, :] * stride_right_k)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute dot product: a (M x K) times b (K x N)
        # a shape (BLOCK_M, BLOCK_K), b shape (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)

    # Store result
    out_ptrs = out_base + (offs_m[:, None] * stride_out_i) + (offs_n[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
import math

@triton.jit
def gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                     total_elems, hidden_dim,
                     BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # map linear index to mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    tl.store(out_ptr + offs, proj * gate * m, mask=mask)

def apply_gate_mask(proj, gate, mask, block_size=1024):
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (math.ceil(total / block_size),)
    mask_flat = mask.contiguous().view(-1)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # first axis: combined batch * hidden
    bh_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    hidden = H
    batch = bh_id // hidden
    h = bh_id % hidden

    # compute row and column offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointer for this batch and hidden channel
    left_base = left_ptr + batch * stride_left_b + h * stride_left_h
    right_base = right_ptr + batch * stride_right_b + h * stride_right_h
    out_base = out_ptr + batch * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A block (left)
        a_ptrs = left_base + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B block (right transposed)
        # right[b, j, k, h] -> B[k, j]
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # multiply-accumulate
        acc += tl.dot(a, b)

    # store result
    out_ptrs = out_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def tri_mul_gemm(left, right):
    """
    Batched GEMM for TriMul: out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    left, right: torch tensors of shape (B,N,N,H), dtype torch.float16
    Returns out tensor same shape.
    """
    B, N, _, H = left.shape
    # ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # strides
    s_left_b, s_left_i, s_left_k, s_left_h = left.stride()
    s_right_b, s_right_i, s_right_k, s_right_h = right.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()

    # block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_bh = B * H
    grid = (
        total_bh,
        math.ceil(N / BLOCK_M),
        math.ceil(N / BLOCK_N),
    )
    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_left_b,
        s_left_i,
        s_left_k,
        s_left_h,
        s_right_b,
        s_right_i,
        s_right_k,
        s_right_h,
        s_out_b,
        s_out_i,
        s_out_j,
        s_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3).
    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: [B, N, N, C] torch.Tensor (float32)
            mask: [B, N, N] torch.Tensor (float32) or None
            weights: dict of model parameters
            config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. LayerNorm on input
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # 2. Cast to fp16 for compute intensive ops
    x = x.to(torch.float16)

    # 3. Cast weights needed for fp16 ops
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # 4. Linear projections
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # 5. Gate projections + sigmoid
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # 6. Apply mask and gate (fused Triton kernel) if mask present
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 7. Core reduction: batched GEMM using Triton
    out = tri_mul_gemm(left, right)   # shape (B,N,N,hidden_dim), fp16

    # 8. Output LayerNorm (hidden dim), gating and final projection
    # Convert to float32 for stable LayerNorm
    out = out.to(torch.float32)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, to_out_w.to(torch.float32))

    # 9. Cast back to original dtype (usually float32)
    out = out.to(input_tensor.dtype)
    return out

"""
Highly‑optimized Triton implementation of the outgoing TriMul operator
(AlphaFold3 style). The forward pass consists of:
1. Layer‑norm on the input.
2. Linear projections (left/right) and gate projections (left/right/out).
3. Optional mask + gate fusion via a small Triton kernel.
4. Core reduction   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM (left @ rightᵀ) with a custom Triton
   kernel that processes all hidden channels in a single launch.
5. Output layer‑norm, out‑gate multiplication and final linear projection.
All heavy compute is performed in fp16, with fp32 accumulation for
numerical stability. The final result is returned in the original dtype.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,                # *[total_elems]   fp16
    gate_ptr,                # *[total_elems]   fp16
    mask_ptr,                # *[mask_elems]    fp16
    out_ptr,                 # *[total_elems]   fp16
    total_elems,             # i64
    hidden_dim,              # i64  (size of the last dimension H)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # map flat index -> mask index (broadcast over hidden dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(out_ptr + offs, proj * gate * m, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Element‑wise ``proj * gate * mask`` where ``mask`` has shape (B, N, N)
    and is broadcast over the hidden dimension H.
    All tensors must be FP16 and contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the core TriMul reduction
#   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # first axis encodes (batch, hidden)
    bh_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    hidden = H
    batch = bh_id // hidden
    h = bh_id % hidden

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this (batch, hidden)
    left_base = left_ptr + batch * stride_left_b + h * stride_left_h
    right_base = right_ptr + batch * stride_right_b + h * stride_right_h
    out_base = out_ptr + batch * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left tile (M x K) -----
        a_ptrs = left_base + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # ----- right tile (K x N)  (transposed access) -----
        # we need right[b, j, k, h] -> B[k, j]
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # ----- store result (M x N) -----
    out_ptrs = out_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[:, None])


def _tri_mul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for TriMul: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    left/right: (B, N, N, H), fp16
    Returns out of same shape and dtype.
    """
    B, N, _, H = left.shape
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # strides
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_bh = B * H
    grid = (
        total_bh,
        math.ceil(N / BLOCK_M),
        math.ceil(N / BLOCK_N),
    )
    _tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_lb,
        s_li,
        s_lk,
        s_lh,
        s_rb,
        s_ri,
        s_rk,
        s_rh,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) or None
            - weights: dict of model parameters (all torch.Tensors)
            - config: dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy stages
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast all linear / norm weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gate projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6) Apply mask + gate (fused Triton kernel) or just gate
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core O(N³·H) reduction via Triton GEMM kernel
    # ------------------------------------------------------------------
    out = _tri_mul_gemm(left, right)          # (B, N, N, H), fp16

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = out.to(torch.float32)                # fp32 for stable LN
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, to_out_w.to(torch.float32))   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
