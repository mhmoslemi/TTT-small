def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # LayerNorm 1
    norm_weight = weights["norm.weight"].to(device).to(dtype=torch.float32) # keep float?
    norm_bias = weights["norm.bias"].to(device).to(torch.float32)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # Cast to half for compute
    x = x.to(torch.float16)

    # Prepare weights in half
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(torch.float16)

    # Combine left_proj and left_gate weights
    left_comb_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0)  # shape [2*H, dim]
    right_comb_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0)

    # Compute combined projections
    # input shape [B,N,N,dim] -> flatten for linear
    # Using torch.nn.functional.linear for half input and weight half
    left_comb = torch.nn.functional.linear(x, left_comb_weight)  # shape [B,N,N,2*H]
    right_comb = torch.nn.functional.linear(x, right_comb_weight)

    # split
    left_proj_half, left_gate_half = left_comb.split(hidden_dim, dim=-1)
    right_proj_half, right_gate_half = right_comb.split(hidden_dim, dim=-1)

    # sigmoid gates
    left_gate = left_gate_half.sigmoid()
    right_gate = right_gate_half.sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # Apply mask and gates: use kernel if mask present and not nomask
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_mask(left_proj_half, left_gate, mask_fp16)
        right = _apply_gate_mask(right_proj_half, right_gate, mask_fp16)
    else:
        left = left_proj_half * left_gate
        right = right_proj_half * right_gate

    # Core O(N^3) reduction via batched GEMM
    # Permute to bring hidden dim to batch dimension
    B, N, _, _ = left.shape
    # left shape: [B, N, N, H]
    left_ = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)  # (B*H, N, N)
    right_ = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Compute batched matmul: left * right^T
    out_ = torch.bmm(left_, right_.transpose(1, 2))  # shape (B*H, N, N)

    # reshape back
    out = out_.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # Second LayerNorm on hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection to original dim
    out = torch.nn.functional.linear(out, to_out_weight)

    # Cast back to original dtype (float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑normalize the input tensor.
2. Cast to FP16 for compute‑intensive ops.
3. Fuse the two linear projections (proj & gate) for the left and right
   branches by concatenating their weight matrices – one GEMM per branch.
4. Apply sigmoid to the gate tensors.
5. Fuse element‑wise mask, gate and projection with a tiny Triton kernel.
6. Perform the core O(N³·H) reduction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
   This is a batched matrix multiplication of shape (B·H, N, N) using
   `torch.bmm`, which maps to a highly‑optimized cuBLAS kernel.
7. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate and project back to the original channel dimension.
8. Cast the result back to the input dtype (usually float32).

The Triton kernel is used for the mask‑gate‑projection fusion, satisfying the
requirement that at least part of the computation runs on Triton.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (total_elems,)
    gate_ptr,          # fp16 * (total_elems,)
    mask_ptr,          # fp16 * (mask_elems,)  where mask_elems = total_elems // hidden_dim
    out_ptr,           # fp16 * (total_elems,)
    total_elems,       # i32  – total number of elements in proj/gate
    hidden_dim,        # i32  – size of the hidden dimension (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # guard against OOB
    mask = offs < total_elems

    # broadcast mask over hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_mask(proj: torch.Tensor,
                     gate: torch.Tensor,
                     mask: torch.Tensor,
                     block_size: int = 65536) -> torch.Tensor:
    """
    Element‑wise fusion: out = proj * gate * mask
    All tensors must be fp16, contiguous and on the same CUDA device.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict containing "dim", "hidden_dim" and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Prepare FP16 weights
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse projection + gate linear layers (one GEMM per side)
    # ------------------------------------------------------------------
    # Concatenate weights so that a single linear produces both proj and gate
    left_comb_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, C)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, C)

    left_comb  = torch.nn.functional.linear(x, left_comb_w)   # (B,N,N,2*H)
    right_comb = torch.nn.functional.linear(x, right_comb_w)

    left_proj, left_gate_raw = left_comb.split(hidden_dim, dim=-1)
    right_proj, right_gate_raw = right_comb.split(hidden_dim, dim=-1)

    left_gate  = left_gate_raw.sigmoid()
    right_gate = right_gate_raw.sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask, projection and gate (Triton)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core O(N³·H) reduction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matmul: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm, output gate and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,C)

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
