@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_i, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program IDs
    pid_m = tl.program_id(0)   # block row of output matrix (i)
    pid_n = tl.program_id(1)   # block col of output matrix (j)
    pid_batch = tl.program_id(2)  # linear index over B*H

    # compute batch and head indices
    b = pid_batch // H
    h = pid_batch % H

    # Compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Compute pointers for the current block
        left_offset = (
            b * stride_left_b
            + h * stride_left_h
            + offs_m[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        right_offset = (
            b * stride_right_b
            + h * stride_right_h
            + offs_n[:, None] * stride_right_i
            + offs_k[None, :] * stride_right_k
        )

        # Load blocks
        left = tl.load(left_ptr + left_offset, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        right = tl.load(right_ptr + right_offset, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # Compute dot product, note we want left @ right^T -> left (M,K) @ right (N,K) with trans_b=True
        acc += tl.dot(left, right, trans_b=True)

    # Write back result
    out_offset = (
        b * stride_out_b
        + h * stride_out_h
        + offs_m[:, None] * stride_out_i
        + offs_n[None, :] * stride_out_j
    )
    out = acc.to(tl.float16)
    tl.store(out_ptr + out_offset, out, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. LayerNorm over dim
    # We'll use torch's F.layer_norm with weight and bias
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )   # result dtype as original (float32)

    # Cast to fp16 for compute
    x = x.to(torch.float16)

    # Cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)

    BLOCK_N = 64

    # Cast back to original dtype (likely float32)
    out = out.to(input_tensor.dtype)
    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor.
2. Cast to fp16 for compute‑intensive stages.
3. Linear projections (left/right) and gated‑projections (left_gate,
   right_gate, out_gate) using torch.nn.functional.linear (cuBLAS) on fp16.
4. Fuse mask, projection and gate with a tiny Triton kernel (if a mask is
   provided).
5. Core O(N³·H) reduction: for each batch * hidden_dim pair we compute
   out = left @ rightᵀ using a custom Triton matmul kernel (no cuBLAS).
   The kernel works on blocks (BLOCK_M × BLOCK_N) and accumulates in FP32.
6. Apply a second LayerNorm, multiply by the output gate and linearly
   project back to the original dimension.
7. Cast the result back to the input dtype.

Only the mask‑gate fusion and the batched matmul are written in Triton;
all other operations use PyTorch for high‑throughput GPU kernels.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: elementwise proj * gate * mask (mask broadcasted over H)
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    proj_ptr,          # *[total_elems]   fp16
    gate_ptr,          # *[total_elems]   fp16
    mask_ptr,          # *[mask_elems]    fp16
    out_ptr,           # *[total_elems]   fp16
    total_elems,       # int
    hidden_dim,        # int
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # each hidden_dim entries share the same mask element
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Fuse ``proj * gate * mask`` where ``mask`` has shape (B, N, N) and is
    broadcast over the hidden dimension H.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.view(-1)  # (B*N*N,)
    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the outgoing TriMul reduction
#   out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
# ----------------------------------------------------------------------
@triton.jit
def trmul_outgoing_kernel(
    left_ptr,   # *[B, H, N, N]  fp16
    right_ptr,  # *[B, H, N, N]  fp16
    out_ptr,    # *[B, H, N, N]  fp16
    B, H, N,
    # strides for left
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    # strides for right
    stride_right_b, stride_right_h, stride_right_i, stride_right_k,
    # strides for out
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_m = tl.program_id(0)   # block row (i)
    pid_n = tl.program_id(1)   # block col (j)
    pid_batch = tl.program_id(2)  # linear index over B*H

    # decode batch and head
    b = pid_batch // H
    h = pid_batch % H

    # coordinates within the output matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for left block (M x K)  -> (i, k)
        left_ptrs = (
            b * stride_left_b
            + h * stride_left_h
            + offs_m[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        # pointers for right block (N x K) -> (j, k)
        right_ptrs = (
            b * stride_right_b
            + h * stride_right_h
            + offs_n[:, None] * stride_right_i
            + offs_k[None, :] * stride_right_k
        )

        left = tl.load(left_ptr + left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)
        right = tl.load(right_ptr + right_ptrs,
                        mask=mask_n[:, None] & mask_k[None, :],
                        other=0.0)

        # left: (M, K), right: (N, K). We need left @ rightᵀ → use trans_b=True.
        acc += tl.dot(left, right, trans_b=True)

    # write back
    out_ptrs = (
        b * stride_out_b
        + h * stride_out_h
        + offs_m[:, None] * stride_out_i
        + offs_n[None, :] * stride_out_j
    )
    out = acc.to(tl.float16)
    tl.store(out_ptr + out_ptrs,
             out,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # x : (B, N, N, D), dtype = input dtype (usually fp32)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive stages
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6. Fuse mask with projection and gate (Triton) or simple multiply
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left, left_gate, mask_fp16)
        right = apply_gate_and_mask(right, right_gate, mask_fp16)
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction using a custom Triton kernel
    #    Compute left @ rightᵀ for each (batch, head) pair.
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for convenient batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    out_perm = torch.empty_like(left_perm, dtype=torch.float16)

    # Strides (in elements)
    stride_left_b = left_perm.stride(0)
    stride_left_h = left_perm.stride(1)
    stride_left_i = left_perm.stride(2)
    stride_left_k = left_perm.stride(3)

    stride_right_b = right_perm.stride(0)
    stride_right_h = right_perm.stride(1)
    stride_right_i = right_perm.stride(2)
    stride_right_k = right_perm.stride(3)

    stride_out_b = out_perm.stride(0)
    stride_out_h = out_perm.stride(1)
    stride_out_i = out_perm.stride(2)
    stride_out_j = out_perm.stride(3)

    # Block sizes – tuned for typical B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    trmul_outgoing_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B,
        H,
        N,
        stride_left_b,
        stride_left_h,
        stride_left_i,
        stride_left_k,
        stride_right_b,
        stride_right_h,
        stride_right_i,
        stride_right_k,
        stride_out_b,
        stride_out_h,
        stride_out_i,
        stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to original layout (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, D)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
