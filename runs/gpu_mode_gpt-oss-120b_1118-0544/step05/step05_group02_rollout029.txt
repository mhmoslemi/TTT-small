Define kernel:

@triton.jit
def gated_mask_fuse_kernel(
   left_ptr, right_ptr,
   left_gate_ptr, right_gate_ptr,
   mask_ptr,
   out_left_ptr, out_right_ptr,
   total_elems, hidden_dim,
   BLOCK_SIZE: tl.constexpr
):
  pid = tl.program_id(0)
  offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
  mask = offs < total_elems

  # Compute mask index = offs // hidden_dim (since mask repeats per hidden dim)
  mask_idx = offs // hidden_dim

  left = tl.load(left_ptr + offs, mask=mask, other=0.0)
  right = tl.load(right_ptr + offs, mask=mask, other=0.0)
  left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
  right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
  m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0) # mask broadcast: default 1

  # apply: left*left_gate*m, right*right_gate*m
  out_left = left * left_gate * m
  out_right = right * right_gate * m

  tl.store(out_left_ptr + offs, out_left, mask=mask)
  tl.store(out_right_ptr + offs, out_right, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input layer norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # 2. Cast to fp16
    x = x.to(torch.float16)

    # 3. Convert relevant weights to fp16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # 4. Fuse all linear projections (proj + gates + out_gate)
    B, N, _, C = x.shape
    X_flat = x.reshape(-1, C)  # (B*N*N, C)

    # Concatenate weight matrices
    # order: left_proj, right_proj, left_gate, right_gate, out_gate
    W = torch.cat([
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
    ], dim=0)  # shape (5*hidden_dim, C)

    Y = torch.nn.functional.linear(X_flat, W)  # (B*N*N, 5*hidden_dim)

    # 5. Slice outputs
    H = hidden_dim
    left_proj = Y[:, 0:H]
    right_proj = Y[:, H:2*H]
    left_gate = Y[:, 2*H:3*H]
    right_gate = Y[:, 3*H:4*H]
    out_gate = Y[:, 4*H:5*H]

    # 6. Activations
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # 7. Reshape to (B, N, N, H)
    left_proj = left_proj.view(B, N, N, H)
    right_proj = right_proj.view(B, N, N, H)
    left_gate = left_gate.view(B, N, N, H)
    right_gate = right_gate.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # 8. Apply mask and gating
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        # fused kernel to multiply proj*gate*mask
        left, right = apply_gate_mask_fused(
            left_proj, left_gate, right_proj, right_gate, mask_fp16
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # We can delete unused tensors to free memory (optional)
    # torch.cuda.empty_cache() not needed

    # 9. Core reduction (batched GEMM)
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # 10. Output layernorm
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 11. Apply out_gate and final linear
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)

    # 12. Cast back to original dtype
    out = out.to(input_tensor.dtype)

    return out

"""
TriMul (outgoing) forward pass – fully fused implementation.

Algorithm
---------
1. Input LayerNorm (over the channel dimension).
2. Cast to FP16 for compute‑intensive stages.
3. Fuse all linear projections (left/right projections, gates and out_gate) into a
   single matrix multiplication:
       Y = X_flat @ W.T   where   W = [left_proj; right_proj;
                                     left_gate; right_gate; out_gate]
4. Slice Y into the five tensors, apply sigmoid to the three gate tensors and
   reshape to [B, N, N, H].
5. If a mask is provided, fuse mask application and the left/right gate
   multiplications with a small Triton kernel (no extra memory copies).
   Otherwise perform the element‑wise products with torch.
6. Core reduction:
       out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   which is a batched GEMM (left @ rightᵀ) over the hidden dimension.
7. Output LayerNorm, multiply by the out‑gate and a final linear projection.
8. Cast back to the original dtype.

The heavy linear projection is performed by cuBLAS (via torch.nn.functional.linear),
while the mask‑gate fusion is realised with a custom Triton kernel.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_gate_mask_kernel(
    left_ptr,      # *[total]   fp16
    left_gate_ptr, # *[total]   fp16
    right_ptr,     # *[total]   fp16
    right_gate_ptr,# *[total]   fp16
    mask_ptr,      # *[mask]    fp16
    out_left_ptr,  # *[total]   fp16 (can be same as left_ptr)
    out_right_ptr, # *[total]   fp16 (can be same as right_ptr)
    total_elems: tl.constexpr,
    hidden_dim: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # mask index = element // hidden_dim  (broadcast mask over H)
    mask_idx = offs // hidden_dim

    left      = tl.load(left_ptr + offs,      mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs,mask=mask, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr + offs,mask=mask, other=0.0)
    m         = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out_left  = left  * left_gate  * m
    out_right = right * right_gate * m

    tl.store(out_left_ptr  + offs, out_left,  mask=mask)
    tl.store(out_right_ptr + offs, out_right, mask=mask)


def apply_gate_mask_fused(left, left_gate, right, right_gate, mask, block_size=1024):
    """
    In‑place fusion of (proj * gate * mask) for left and right tensors.
    All tensors are expected to be contiguous FP16 with shape [B, N, N, H].
    """
    total = left.numel()
    hidden_dim = left.shape[-1]
    grid = (math.ceil(total / block_size),)

    mask_flat = mask.view(-1)  # (B*N*N,)

    fused_gate_mask_kernel[grid](
        left,
        left_gate,
        right,
        right_gate,
        mask_flat,
        left,          # write result back into left
        right,         # write result back into right
        total,
        hidden_dim,
        BLOCK_SIZE=block_size,
    )
    return left, right

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Fuse all linear projections (proj + gates + out_gate)
    # ------------------------------------------------------------------
    B, N, _, C = x.shape
    X_flat = x.view(-1, C)                         # (B*N*N, C)

    # Stack weight matrices in the required order
    W = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    )                                              # (5*H, C)

    Y = F.linear(X_flat, W)                        # (B*N*N, 5*H)

    # ------------------------------------------------------------------
    # 5. Slice and activate
    # ------------------------------------------------------------------
    H = hidden_dim
    left_proj   = Y[:, 0:H]
    right_proj  = Y[:, H:2*H]
    left_gate   = Y[:, 2*H:3*H].sigmoid()
    right_gate  = Y[:, 3*H:4*H].sigmoid()
    out_gate    = Y[:, 4*H:5*H].sigmoid()

    # reshape to 4‑D tensors
    left_proj   = left_proj.view(B, N, N, H)
    right_proj  = right_proj.view(B, N, N, H)
    left_gate   = left_gate.view(B, N, N, H)
    right_gate  = right_gate.view(B, N, N, H)
    out_gate    = out_gate.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Apply mask + gate (fused Triton kernel if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = apply_gate_mask_fused(
            left_proj, left_gate, right_proj, right_gate, mask_fp16
        )
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction using batched GEMM
    # ------------------------------------------------------------------
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B*H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))          # (B*H, N, N)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)        # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
