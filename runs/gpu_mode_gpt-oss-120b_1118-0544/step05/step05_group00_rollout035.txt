@triton.jit
def layernorm_kernel(
    inp_ptr, out_ptr,
    weight_ptr, bias_ptr,
    eps,
    total_rows, dim,  # runtime scalars
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    row = pid
    if row >= total_rows:
        return

    offs = tl.arange(0, BLOCK_SIZE)
    mask = offs < dim
    # compute pointer to start of this row
    row_offset = row * dim

    x = tl.load(inp_ptr + row_offset + offs, mask=mask, other=0.0)
    # compute mean
    sum_x = tl.sum(x, axis=0, mask=mask)  # Actually sum across vector
    mean = sum_x / dim

    # compute variance
    diff = x - mean
    var = tl.sum(diff * diff, axis=0, mask=mask) / dim

    # norm
    inv_std = 1.0 / tl.sqrt(var + eps)

    w = tl.load(weight_ptr + offs, mask=mask, other=1.0)  # weight default 1
    b = tl.load(bias_ptr + offs, mask=mask, other=0.0)

    y = (x - mean) * inv_std * w + b

    tl.store(out_ptr + row_offset + offs, y, mask=mask)
```

def layernorm_kernel(
    inp_ptr, out_ptr,
    weight_ptr, bias_ptr,
    eps,
    total_rows, dim,
    BLOCK_SIZE: tl.constexpr,
    OUTPUT_DTYPE: tl.constexpr
):
    pid = tl.program_id(0)
    if pid >= total_rows:
        return

    row_offset = pid * dim

    # first pass compute sums
    sum = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    for offset in range(0, dim, BLOCK_SIZE):
        cur_len = tl.minimum(BLOCK_SIZE, dim - offset)
        offs = tl.arange(0, BLOCK_SIZE)
        mask = offs < cur_len
        x = tl.load(inp_ptr + row_offset + offset + offs, mask=mask).to(tl.float32)
        sum += tl.sum(x, mask=mask)
        sum_sq += tl.sum(x * x, mask=mask)

    mean = sum / dim
    var = sum_sq / dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # second pass write output
    for offset in range(0, dim, BLOCK_SIZE):

        w = tl.load(weight_ptr + offset + offs, mask=mask).to(tl.float32)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    # config: dim, hidden_dim, etc.
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    # Load layernorm weights
    norm_weight = weights["norm.weight"].to(device).to(torch.float32)
    norm_bias = weights["norm.bias"].to(device).to(torch.float32)
    # Flatten and apply layernorm via kernel
    B, N, N2, C = input_tensor.shape
    assert N == N2
    total_rows = B * N * N
    # Ensure contiguous
    x_flat = input_tensor.contiguous().view(total_rows, C)
    out_flat = torch.empty_like(x_flat, dtype=torch.float32)
    # Launch kernel
    BLOCK = 128
    grid = (total_rows,)
    layernorm_kernel[grid](
        x_flat,
        out_flat,
        norm_weight,
        norm_bias,
        eps,
        total_rows,
        C,
        BLOCK_SIZE=BLOCK,
        OUTPUT_DTYPE=tl.float32,
        num_warps=4
    )
    x = out_flat.view(B, N, N, C).to(torch.float16)
    # Linear projections
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)  # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Apply mask if present and not nomask
    nomask = config.get("nomask", False)
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp
        right = right * mask_fp

    # Core contraction via batched matmul
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2)
    right_perm = right.permute(0, 3, 1, 2)
    # bmm via matmul
    out_mat = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # (B, hidden_dim, N, N)
    out = out_mat.permute(0, 2, 3, 1)  # (B, N, N, hidden_dim)

    # Output layernorm using kernel (half)
    total_rows_out = total_rows
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(total_rows_out, hidden_dim)
    out_norm_flat = torch.empty_like(out_flat, dtype=torch.float16)
    layernorm_kernel[grid](
        out_flat,
        out_norm_flat,
        to_out_norm_w,
        to_out_norm_b,
        eps,
        total_rows_out,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        OUTPUT_DTYPE=tl.float16,
        num_warps=4
    )
    out_norm = out_norm_flat.view(B, N, N, hidden_dim)

    # Apply out gate
    out = out_norm * out_gate

    # Final projection
    out = F.linear(out, to_out_w)  # (B,N,N,dim) half

    # Cast back to original dtype (probably float32)
    if input_tensor.dtype != out.dtype:
        out = out.to(input_tensor.dtype)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1.  Layer‑norm over the input channel dimension using a custom Triton
    kernel (float‑32 accumulation, output type selectable).  This avoids
    the PyTorch Python‑level reduction.
2.  Cast to float16 for the compute‑intensive path.
3.  Linear projections for left/right values and three gates (no bias).
4.  Fuse element‑wise mask (if present) and gate multiplications by simple
    PyTorch broadcasting.
5.  Core O(N³·H) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    Implemented as a batched matrix‑multiply using `torch.matmul`
    on tensors reshaped to (B, H, N, N) – this leverages cuBLAS tensor
    cores without extra copies.
6.  Output layer‑norm (again via the Triton kernel), multiply by the
    output gate and a final linear projection back to the original
    dimension.
7.  Cast result back to the input dtype.

Only the layer‑norms are implemented in Triton; the remaining
operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    inp_ptr, out_ptr,            # pointers to (total_rows, dim)
    weight_ptr, bias_ptr,        # (dim,)
    eps,                         # epsilon (float)
    total_rows, dim,             # runtime scalars
    BLOCK_SIZE: tl.constexpr,   # compile‑time tile size for the dim axis
    OUTPUT_DTYPE: tl.constexpr, # tl.float32 or tl.float16
):
    """Fused LayerNorm:
    out = (inp - mean) / sqrt(var + eps) * weight + bias
    Computed per row of shape (dim,).  Uses float32 accumulator for
    both float32 and float16 inputs.
    """
    pid = tl.program_id(0)
    if pid >= total_rows:
        return

    row_offset = pid * dim

    # ---------- first pass: compute mean and variance ----------
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    # loop over the dim axis in chunks of BLOCK_SIZE
    for offset in range(0, dim, BLOCK_SIZE):
        # active lanes for the last chunk
        cur_len = dim - offset
        mask = tl.arange(0, BLOCK_SIZE) < cur_len
        x = tl.load(inp_ptr + row_offset + offset + tl.arange(0, BLOCK_SIZE),
                    mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ---------- second pass: write normalized output ----------
    for offset in range(0, dim, BLOCK_SIZE):
        cur_len = dim - offset
        mask = tl.arange(0, BLOCK_SIZE) < cur_len

        x = tl.load(inp_ptr + row_offset + offset + tl.arange(0, BLOCK_SIZE),
                    mask=mask, other=0.0).to(tl.float32)

        w = tl.load(weight_ptr + offset + tl.arange(0, BLOCK_SIZE),
                    mask=mask, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + offset + tl.arange(0, BLOCK_SIZE),
                    mask=mask, other=0.0).to(tl.float32)

        y = (x - mean) * inv_std * w + b
        y = y.to(OUTPUT_DTYPE)

        tl.store(out_ptr + row_offset + offset + tl.arange(0, BLOCK_SIZE),
                 y, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 accumulation, output float32)
    # -----------------------------------------------------------------
    B, N, N2, C = input_tensor.shape
    assert N == N2 and C == dim

    total_rows = B * N * N
    BLOCK = 128  # works for dim up to 384

    # flatten to (total_rows, dim)
    inp_flat = input_tensor.contiguous().view(total_rows, dim)
    out_flat = torch.empty_like(inp_flat, dtype=torch.float32)

    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)

    grid = (total_rows,)
    layernorm_kernel[grid](
        inp_flat,
        out_flat,
        norm_w,
        norm_b,
        eps,
        total_rows,
        dim,
        BLOCK_SIZE=BLOCK,
        OUTPUT_DTYPE=tl.float32,
        num_warps=4,
    )

    x = out_flat.view(B, N, N, C).to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections and gates (half precision)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)          # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)        # (B,N,N,hidden_dim)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    left = left * left_gate
    right = right * right_gate

    if mask is not None and not nomask:
        # mask broadcast over the hidden dimension
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 3. Core O(N³·H) contraction via batched matrix multiplication
    # -----------------------------------------------------------------
    # reshape to (B, hidden_dim, N, N) without materialising a copy
    left_perm = left.permute(0, 3, 1, 2)       # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2)     # (B, H, N, N)

    # batched matmul: left @ rightᵀ  (per hidden dim)
    out_mat = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # (B, H, N, N)

    # back to (B, N, N, H)
    out = out_mat.permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 4. Output LayerNorm (half precision) via Triton kernel
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    # flatten (B*N*N, hidden_dim)
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(total_rows, hidden_dim)
    out_norm_flat = torch.empty_like(out_flat, dtype=torch.float16)

    layernorm_kernel[grid](
        out_flat,
        out_norm_flat,
        to_out_norm_w,
        to_out_norm_b,
        eps,
        total_rows,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        OUTPUT_DTYPE=tl.float16,
        num_warps=4,
    )
    out_norm = out_norm_flat.view(B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 5. Final gating, projection and type cast
    # -----------------------------------------------------------------
    out = out_norm * out_gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # Cast back to the original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)
    return out
