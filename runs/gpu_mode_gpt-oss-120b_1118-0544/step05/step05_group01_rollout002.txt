```python
"""TriMul outgoing operator (forward only) with Triton kernel for gate+mask.

The implementation follows:
1. LayerNorm over channel dim.
2. Linear projections (left/right) and gate projections.
3. Fuse projection * gate * mask using a small Triton kernel when a mask is present.
4. Compute the core reduction out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a batched matrix multiplication (torch.bmm) on fp16 tensors.
5. LayerNorm over hidden dim, multiply by out_gate, final linear projection.
All heavy arithmetic is performed in fp16; final result is cast back to the
input dtype (typically float32).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,   # *fp16, shape (B,N,N,H) flattened
    gate_ptr,   # *fp16, same shape
    mask_ptr,   # *fp16, shape (B,N,N) flattened
    out_ptr,    # *fp16, same shape as proj/gate
    total_elems,  # int64, total number of elements B*N*N*H
    hidden_dim,   # int64, H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # mask for valid elements
    valid = offs < total_elems

    # each consecutive hidden_dim entries share the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Element‑wise multiplication proj * gate * mask (mask broadcast over the hidden dim).
    All tensors must be fp16, contiguous.
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C]
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Load weights (biasless linear layers) and cast to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

"""TriMul outgoing operator (forward only) with Triton kernel for gate+mask.

The implementation follows:
1. LayerNorm over the channel dimension.
2. Linear projections (left/right) and gate projections.
3. If a mask is present, fuse projection * gate * mask using a small
   Triton kernel (broadcasted over the hidden dimension).
4. Core reduction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   performed as a batched matrix multiplication on fp16 tensors.
5. LayerNorm over the hidden dimension, element‑wise multiplication with
   the output gate, and a final linear projection back to ``dim``.
All heavy arithmetic is in fp16; the final output is cast back to the
input dtype (usually float32)."""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,   # *fp16, shape (B,N,N,H) flattened
    gate_ptr,   # *fp16, same shape
    mask_ptr,   # *fp16, shape (B,N,N) flattened
    out_ptr,    # *fp16, same shape as proj/gate
    total_elems,  # i64, total number of elements B*N*N*H
    hidden_dim,   # i64, H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # each consecutive hidden_dim entries share the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Element‑wise ``proj * gate * mask`` (mask broadcast over hidden dim).
    All tensors must be fp16 and contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out
