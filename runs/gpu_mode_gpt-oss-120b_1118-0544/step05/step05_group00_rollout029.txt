    left_ptr, right_ptr, out_ptr,

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,

    batch_idx = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))

    right_proj_w = weights["right_proj.weight"].to(device).half()

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Input LayerNorm (float32 for stability).
2. Cast to fp16 for the compute‑heavy path.
3. Linear projections without bias: left, right, left_gate, right_gate, out_gate.
4. Fuse element‑wise mask (if present) and sigmoid gates with a tiny Triton kernel.
5. Core O(N³·H) contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication L @ Rᵀ.
   Implemented with a custom Triton GEMM kernel that works on the
   (B·H, N, N) reshaped tensors.
6. Output LayerNorm, multiply by out_gate and final linear projection.
7. Cast back to the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    """Fuse mask and sigmoid gates into left/right tensors in‑place."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask_active = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask_active, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask_active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask_active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask_active, other=0.0)

    if HAS_MASK:
        # mask is broadcast over hidden dimension
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask_active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=mask_active)
    tl.store(right_ptr + offs, right, mask=mask_active)


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ B   where
        A : (batch, M, K)
        B : (batch, K, N)  (already transposed view of the original right tensor)
        C : (batch, M, N)
    """
    batch_id = tl.program_id(0)   # batch dimension
    pid_m   = tl.program_id(1)   # tile row
    pid_n   = tl.program_id(2)   # tile col

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the start of the batch slice
    a_batch = A_ptr + batch_id * stride_a_batch
    b_batch = B_ptr + batch_id * stride_b_batch
    c_batch = C_ptr + batch_id * stride_c_batch

    # Masks for edges
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = (k_base + offs_k) < K

        # Load A tile (M x K)
        a = tl.load(
            a_batch + offs_m[:, None] * stride_a_m + (k_base + offs_k)[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (K x N) – note B is already the transposed view
        b = tl.load(
            b_batch + (k_base + offs_k)[:, None] * stride_b_k + offs_n[None, :] * stride_b_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate
        acc += tl.dot(a, b)

    # Write back
    c = acc.to(tl.float16)
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # -------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy path
    # -------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 3. Linear projections (bias‑less) – all in fp16
    # -------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left       = F.linear(x, left_proj_w)      # (B,N,N,hidden_dim)
    right      = F.linear(x, right_proj_w)     # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # -------------------------------------------------
    # 4. Fuse mask and gates (if mask is present)
    # -------------------------------------------------
    if mask is not None and not nomask:
        # Prepare tensors for the kernel
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left  = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # Simple element‑wise gating without mask
        left  = left * left_gate
        right = right * right_gate

    # -------------------------------------------------
    # 5. Core O(N³·H) contraction via custom Triton GEMM
    # -------------------------------------------------
    B, N, _, H = left.shape
    batch = B * H

    # Reshape to (batch, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(batch, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(batch, N, N)

    out_mat = torch.empty_like(left_mat)

    # Block sizes (tuned for fp16 Tensor‑Core usage)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    n_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    n_tiles_n = (N + BLOCK_N - 1) // BLOCK_N

    # Launch the batched GEMM kernel
    _batched_matmul_kernel[
        (batch, n_tiles_m, n_tiles_n)
    ](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        # Right is transposed (K,N) view: swap the last two strides
        right_mat.stride(0), right_mat.stride(2), right_mat.stride(1),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate   # apply output gate (half‑precision)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # -------------------------------------------------
    # 7. Cast back to original dtype and return
    # -------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)
    return out
