       B, N, H,

       stride_r_b, stride_r_i, stride_r_j, stride_r_h,

@triton.jit
def _gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, total_elems, hidden_dim, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # Compute mask index for broadcasting
    mask_idx = offs // hidden_dim

    p = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = p * g * m
    tl.store(out_ptr + offs, out, mask=mask)
```

"""
custom_kernel: high‑performance Triton implementation of the outgoing
Triangle Multiplicative Update (TriMul) used in AlphaFold‑3 style models.

Algorithm
---------
1. Layer‑norm the input tensor `[B, N, N, C]`.
2. Cast to FP16 for compute‑heavy steps.
3. Four linear projections (left/right plus their gates) are performed with
   torch.nn.functional.linear (FP16 weights).
4. Fuse element‑wise `proj * gate * mask` using a tiny Triton kernel
   (`_gate_mask_kernel`).
5. Core O(N³·H) reduction:
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   This is a batched matrix‑multiply (B·H independent N×N GEMMs) realised
   by a custom Triton kernel (`_trimul_kernel`) that tiles the mats with
   BLOCK_M×BLOCK_N output tiles and BLOCK_K reduction tiles.
6. Second Layer‑norm over the hidden dimension, multiplication by the output
   gate, and final linear projection back to the original channel dimension.
7. Cast the result back to the input dtype.

Only the mask‑gate fusion and the N³ reduction are performed in Triton,
while the remaining linear layers and layer‑norms leverage highly‑optimized
PyTorch/cuBLAS kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) with broadcasting of mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # Broadcast mask across hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """Fuse `proj * gate * mask` with a Triton kernel."""
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the core TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,   # fp16 * (B, N, N, H)
    right_ptr,  # fp16 * (B, N, N, H)
    out_ptr,    # fp16 * (B, N, N, H)
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_group = tl.program_id(0)   # combined batch * hidden
    pid_i = tl.program_id(1)       # block row
    pid_j = tl.program_id(2)       # block column

    total_groups = B * H
    b = pid_group // H
    h = pid_group % H

    # Offsets for i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (b,h) pair
    left_base  = left_ptr  + b * stride_l_b  + h * stride_l_h
    right_base = right_ptr + b * stride_r_b  + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b  + h * stride_o_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base \
            + (offs_i[:, None] * stride_l_i) \
            + (offs_k[None, :] * stride_l_j)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right tile: shape (BLOCK_K, BLOCK_N)  (k as rows, j as columns)
        b_ptrs = right_base \
            + (offs_k[:, None] * stride_r_j) \
            + (offs_j[None, :] * stride_r_i)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Accumulate (use FP32 for precision)
        acc += tl.dot(a, b).to(tl.float32)

    # Write result back (cast to FP16)
    out_ptrs = out_base \
        + (offs_i[:, None] * stride_o_i) \
        + (offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------- LayerNorm (input) -------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to FP16 for the compute‑intensive part
    x = x.to(torch.float16)

    # ----------------------- cast weights to FP16 -----------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ----------------------- linear projections -------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------- mask & gate fusion (Triton) --------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguous layout for the core kernel
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, H = left.shape

    # --------------------- core O(N³·H) reduction -----------------------
    out = torch.empty_like(left)  # (B,N,N,H)
    left_cont = left
    right_cont = right
    out_cont = out

    # Strides (int64)
    slb, sli, slj, slh = (left_cont.stride(0), left_cont.stride(1),
                          left_cont.stride(2), left_cont.stride(3))
    srb, sri, srj, srh = (right_cont.stride(0), right_cont.stride(1),
                          right_cont.stride(2), right_cont.stride(3))
    sob, soi, soj, soh = (out_cont.stride(0), out_cont.stride(1),
                          out_cont.stride(2), out_cont.stride(3))

    # Tuning parameters – adaptable, but 64/64/32 works well on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _trimul_kernel[grid](
        left_cont,
        right_cont,
        out_cont,
        B, N, H,
        slb, sli, slj, slh,
        srb, sri, srj, srh,
        sob, soi, soj, soh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------ cast back to input dtype ----------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
