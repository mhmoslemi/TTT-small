@triton.jit
def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lbn, stride_lbk, stride_lbh, stride_lb0,
    stride_rbn, stride_rbk, stride_rbh, stride_rb0,
    stride_obn, stride_obk, stride_obh, stride_ob0,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_head = tl.program_id(2)

    # Determine batch and head (hidden) indexes
    b = pid_head // H
    h = pid_head % H

    # Pointers offset for particular batch and head
    left_base = left_ptr + b * stride_lb0 + h * stride_lbh
    right_base = right_ptr + b * stride_rb0 + h * stride_rbh
    out_base = out_ptr + b * stride_ob0 + h * stride_obh

    # Compute start offsets for i and j
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Compute masks for i,j,k out of bounds
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # Load A tile (left): shape (BLOCK_M, BLOCK_K)
        # A indexing: i (first N) as rows, k (second N) as cols
        a_ptrs = left_base + (offs_i[:, None] * stride_lbn + offs_k[None, :] * stride_lbk)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B_T tile (right transposed): shape (BLOCK_K, BLOCK_N)
        # need B at indices (k, j) i.e., row k, col j (original right as [j,k])
        b_ptrs = right_base + (offs_k[:, None] * stride_rbk + offs_j[None, :] * stride_rbn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)  # a: (M,K), b: (K,N) -> (M,N)

    # Write output
    # out indexing: i, j
    out_ptrs = out_base + (offs_i[:, None] * stride_obn + offs_j[None, :] * stride_obk)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
custom_kernel: Triton‑accelerated forward pass for the outgoing TriMul operator.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C).
2. Cast to float16 for compute‑heavy stages.
3. Linear projections: left_proj, right_proj, left_gate, right_gate, out_gate.
4. Apply sigmoid to gates and (optional) mask → element‑wise
   left = left_proj * left_gate * mask, right = right_proj * right_gate * mask.
5. Core O(N³·H) reduction:
   For each batch b and hidden head h compute
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (left @ rightᵀ) implemented as a Triton
   tiled matrix multiplication with program_id(0)=i‑tile,
   program_id(1)=j‑tile, program_id(2)=b·H+h.
6. Layer‑norm over the hidden dimension, multiply by out_gate,
   linear projection back to C.
7. Cast back to the input dtype (float32) and return.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the TriMul reduction.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,               # *[B, N, N, H]  (float16)
    right_ptr,              # *[B, N, N, H]  (float16)
    out_ptr,                # *[B, N, N, H]  (float16)
    B, N, H,                # compile‑time constants
    stride_lb0, stride_lbn, stride_lbk, stride_lbh,   # left strides
    stride_rb0, stride_rbn, stride_rbk, stride_rbh,   # right strides
    stride_ob0, stride_obn, stride_obk, stride_obh,   # out strides
    BLOCK_M: tl.constexpr,  # tile size in i dimension
    BLOCK_N: tl.constexpr,  # tile size in j dimension
    BLOCK_K: tl.constexpr,  # tile size in k dimension
):
    pid_i = tl.program_id(0)                # i‑tile
    pid_j = tl.program_id(1)                # j‑tile
    pid_head = tl.program_id(2)             # batch*head

    # Decode batch index and hidden head index
    b = pid_head // H
    h = pid_head % H

    # Base pointers for this (b, h)
    left_base = left_ptr + b * stride_lb0 + h * stride_lbh
    right_base = right_ptr + b * stride_rb0 + h * stride_rbh
    out_base = out_ptr + b * stride_ob0 + h * stride_obh

    # Offsets for the i/j tiles
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for boundary checking
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- Load left tile (i,k) ----
        # left[b,i,k,h] -> offset = i*stride_lbn + k*stride_lbk
        a_ptrs = left_base + (offs_i[:, None] * stride_lbn
                              + offs_k[None, :] * stride_lbk)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # ---- Load right tile transposed (k,j) ----
        # right[b,j,k,h] accessed as (k,j) -> offset = k*stride_rbk + j*stride_rbn
        b_ptrs = right_base + (offs_k[:, None] * stride_rbk
                               + offs_j[None, :] * stride_rbn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # ---- Accumulate ----
        acc += tl.dot(a, b)   # a:(M,K), b:(K,N) → (M,N)

    # ---- Write back the result ----
    out_ptrs = out_base + (offs_i[:, None] * stride_obn
                           + offs_j[None, :] * stride_obk)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None (float32)
            - weights: dict with all model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to float16 for compute‑intensive part
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast weights to float16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections
    # --------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)      # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Apply mask (if any) and gating
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B,N,N) → broadcast over hidden dim
        mask = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguous layout for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape   # (B, N, N, H)

    # --------------------------------------------------------------
    # 7. Core TriMul reduction via Triton (batched GEMM)
    # --------------------------------------------------------------
    out = torch.empty_like(left)   # (B,N,N,H), fp16

    # Strides of the tensors (B,N,N,H) in contiguous layout
    stride_lb0, stride_lbn, stride_lbk, stride_lbh = left.stride()
    stride_rb0, stride_rbn, stride_rbk, stride_rbh = right.stride()
    stride_ob0, stride_obn, stride_obk, stride_obh = out.stride()

    # Tile sizes – tuned for typical A100/B200 shared‑memory budget
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_lb0, stride_lbn, stride_lbk, stride_lbh,
        stride_rb0, stride_rbn, stride_rbk, stride_rbh,
        stride_ob0, stride_obn, stride_obk, stride_obh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 8. Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype (usually float32) and return
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
