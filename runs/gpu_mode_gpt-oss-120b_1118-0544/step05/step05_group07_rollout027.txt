@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # tile id within batch
    batch_id = tl.program_id(1)  # which (b*h) slice

    # Compute tile coordinates
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute mask
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for a and b for this batch
    a_ptrs = a_ptr + batch_id * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        # Advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Cast to fp16
    c = acc.to(tl.float16)

    # Store result
    c_ptrs = c_ptr + batch_id * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass – Triton‑fused mask/gate and batched GEMM.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using provided LN parameters.
2. Cast to fp16 for compute‑intensive operations.
3. Linear projections (left/right) and their gates are computed in fp16.
4. Fuse element‑wise mask, projection and gate with a tiny Triton kernel.
5. Core O(N³·H) reduction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batch of H independent GEMMs of size N×N.
   Implemented with a custom batched‑matmul Triton kernel that processes
   all (batch * hidden_dim) GEMMs in parallel.
6. Second LayerNorm + output gate (fp16) using torch ops.
7. Final linear projection back to the original channel dimension.
8. Cast result back to the input dtype (float32).

All heavy compute is performed in fp16 with float32 accumulation for
numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask  (mask broadcasted over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 *
    gate_ptr,          # fp16 *
    mask_ptr,          # fp16 *
    out_ptr,           # fp16 *
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – size of the hidden dimension (for mask broadcast)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # find the corresponding mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    msk  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * msk
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using Triton.
    All tensors must be on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],          # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A : (B*H, M, K)   (row‑major)
#   B : (B*H, K, N)   (row‑major)   (i.e. Bᵀ from the original tensor)
#   C : (B*H, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # strides for A (row, col)
    stride_bk, stride_bn,          # strides for B (row, col)   (B is K×N)
    stride_cm, stride_cn,          # strides for C (row, col)
    batch_stride_a, batch_stride_b, batch_stride_c,   # stride for leading batch dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_tile = tl.program_id(0)          # tile index within a batch
    pid_batch = tl.program_id(1)         # which (batch * hidden) matrix

    # compute tile coordinates
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid_tile // num_pid_n
    pid_n = pid_tile % num_pid_n

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers offset by batch
    a_batch_ptr = a_ptr + pid_batch * batch_stride_a
    b_batch_ptr = b_ptr + pid_batch * batch_stride_b
    c_batch_ptr = c_ptr + pid_batch * batch_stride_c

    # compute base pointers for this tile
    a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM where `b` is already transposed (i.e. we compute a @ bᵀ).
    a, b: shape (B*H, N, N), dtype fp16, contiguous.
    Returns c of same shape and dtype.
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch_head, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "inner dimensions must match"
    device = a.device

    # output allocation
    c = torch.empty_like(a)

    # strides
    stride_am, stride_ak = a.stride(1), a.stride(2)
    stride_bk, stride_bn = b.stride(1), b.stride(2)
    stride_cm, stride_cn = c.stride(1), c.stride(2)
    batch_stride_a = a.stride(0)
    batch_stride_b = b.stride(0)
    batch_stride_c = c.stride(0)

    # block sizes (tuned for fp16 on B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_tiles_m = (M + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_tiles_m * num_tiles_n, batch_head)

    _batched_matmul_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return c

    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C]
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # LayerNorm on input (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # Cast to fp16 for heavy compute
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # cast weights to fp16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # Linear projections (fp16)
    # --------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)      # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)    # (B, N, N, H)

    # --------------------------------------------------------------
    # Gates (fp16)
    # --------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # --------------------------------------------------------------
    # Mask & gate fusion (Triton)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # Core O(N³·H) reduction via batched GEMM (Triton)
    # --------------------------------------------------------------
    B, N, _, _ = left.shape

    # reshape to (B*H, N, N) for both operands
    left_ = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)   # (B*H, N, N)
    # right needs to be transposed on the last two axes before reshaping
    right_ = right.permute(0, 3, 2, 1).contiguous().view(-1, N, N)  # (B*H, N, N) where dim1 is K, dim2 is J

    # batched matmul: left_ @ right_ᵀ  (right_ already has K dim first, J dim second)
    out_ = _batched_matmul(left_, right_)  # (B*H, N, N), fp16

    # reshape back to (B, N, N, H)
    out = out_.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # Second LayerNorm + output gate (fp16)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # Final linear projection back to dim (fp16)
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
