"""
TriMul (outgoing) forward pass – fully fused implementation.

Algorithm
---------
1. Input LayerNorm  (affine) on the last channel dimension (dim).
2. Cast to FP16 for the heavy compute path.
3. Linear projections:
   - left_proj, right_proj  :   x @ W (no bias)
   - left_gate, right_gate, out_gate : x @ W + sigmoid
4. Fuse element‑wise mask (optional), gate and projection with a small Triton
   kernel (proj * gate * mask).  When there is no mask the kernel is bypassed
   and a simple element‑wise multiplication is used.
5. Core reduction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (B·H, N, N) where each hidden
   channel h performs an N×N matmul.  We reshape the tensors, use
   torch.bmm (cuBLAS) which is heavily optimized on NVIDIA GPUs.
6. Output LayerNorm over the hidden dimension, point‑wise multiplication with
   out_gate and a final linear projection back to dim.
7. Cast back to the input dtype (usually float32).

All heavy element‑wise work (proj * gate * mask) is performed in a Triton
kernel; the rest relies on highly‑optimized PyTorch primitives.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and optional mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    proj_ptr,        # *[total_elems]   fp16
    gate_ptr,        # *[total_elems]   fp16
    mask_ptr,        # *[mask_elems]    fp16 (broadcasted)
    out_ptr,         # *[total_elems]   fp16
    total_elems,     # i64
    hidden_dim,      # i64  (size of H)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # Mapping from flat index to mask index (broadcast H)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_proj_gate_mask(proj: torch.Tensor, gate: torch.Tensor,
                          mask: torch.Tensor, block_size: int = 8192) -> torch.Tensor:
    """
    Element‑wise ``proj * gate * mask`` where ``mask`` has shape (B, N, N)
    and is broadcast over the hidden dimension H.
    All inputs must be FP16, contiguous, and on the same device.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)
    _proj_gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],   # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return out

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be None)
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (same dtype as input_tensor)
    """
    # Unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (affine)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑intensive stages
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all linear weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B, N, N, H) in FP16
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Fuse mask, gate and projection (Triton) or simple multiply
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_proj_gate_mask(left_proj, left_gate, mask_fp16)
        right = _apply_proj_gate_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # left shape (B, N, N, H)

    # Prepare left: (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2)
    # Prepare rightᵀ: (B, H, N, N) where last two dims are (k, j)
    right_perm_T = right.permute(0, 3, 2, 1)

    # Collapse batch and hidden dimensions for bmm
    left_mat = left_perm.reshape(B * H, N, N).contiguous()
    right_mat = right_perm_T.reshape(B * H, N, N).contiguous()

    # Batched matrix multiplication (FP16 GEMM)
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N)

    # Restore (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm over hidden dimension, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
