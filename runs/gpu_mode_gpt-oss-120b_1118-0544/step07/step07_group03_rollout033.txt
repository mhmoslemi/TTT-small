"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections (left/right) and gating (left/right/out) – all in FP16.
4. Optional mask multiplication (broadcasted over the hidden dimension).
5. Core TriMul reduction:
   For each batch `b` and hidden slice `h` we compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (L @ Rᵀ) with shape (B*H, N, N).
   Implemented with a custom Triton kernel that tiles the three dimensions
   (BLOCK_M, BLOCK_N, BLOCK_K) and accumulates in FP32.
6. Output LayerNorm, element‑wise out‑gate, and final linear projection back
   to the original channel dimension `dim`.
7. Cast the result back to the input dtype.

Only the reduction step is done in Triton; the rest uses highly‑optimized
PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr,          # fp16 * (BH, N, N)
    right_ptr,         # fp16 * (BH, N, N)
    out_ptr,           # fp16 * (BH, N, N)
    N: tl.int32,
    # strides for left
    stride_left_batch, stride_left_row, stride_left_col,
    # strides for right
    stride_right_batch, stride_right_row, stride_right_col,
    # strides for out
    stride_out_batch, stride_out_row, stride_out_col,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    i0 = block_i * BLOCK_M
    j0 = block_j * BLOCK_N

    # row / column indices for this tile
    i_range = i0 + tl.arange(0, BLOCK_M)
    j_range = j0 + tl.arange(0, BLOCK_N)

    # mask for output positions (handles edge tiles)
    mask_i = i_range < N
    mask_j = j_range < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        k_range = k0 + tl.arange(0, BLOCK_K)

        mask_k = k_range < N

        # -----------------------------------------------------------------
        # Load left tile   (M, K)  : left[batch_id, i_range, k_range]
        # -----------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + batch_id * stride_left_batch
            + i_range[:, None] * stride_left_row
            + k_range[None, :] * stride_left_col
        )
        left_mask = mask_i[:, None] & mask_k[None, :]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0)

        # -----------------------------------------------------------------
        # Load right tile transposed (K, N) : right[batch_id, j_range, k_range]
        # (we treat k as the row dimension, j as the column dimension)
        # -----------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + batch_id * stride_right_batch
            + k_range[:, None] * stride_right_col   # k → row of the transposed view
            + j_range[None, :] * stride_right_row   # j → column of the transposed view
        )
        right_mask = mask_k[:, None] & mask_j[None, :]
        right = tl.load(right_ptrs, mask=right_mask, other=0.0)

        # Accumulate the dot product (M,K)·(K,N) → (M,N)
        acc += tl.dot(left, right)

    # -----------------------------------------------------------------
    # Write the tile back to output
    # -----------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch_id * stride_out_batch
        + i_range[:, None] * stride_out_row
        + j_range[None, :] * stride_out_col
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_ij)

    if mask is not None and not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp
        right = right_proj * right_gate * mask_fp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:  # mask present but user asked to ignore it -> still broadcast for safety
            mask_fp = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask_fp
            right = right * mask_fp

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction (batched GEMM) – Triton kernel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # Move hidden dimension to second axis for easy flattening
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    BH = B * H
    left_flat = left.view(BH, N, N)    # (BH, N, N)
    right_flat = right.view(BH, N, N)  # (BH, N, N)

    out_flat = torch.empty_like(left_flat)  # (BH, N, N)

    # Tile sizes – tuned for typical AlphaFold3 workloads
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final projection back to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to the input dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
