    stride_r_b, stride_r_i, stride_r_k, stride_r_h,

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for fast compute.
3. Linear projections (left/right) and gate projections (left/right/out) are
   performed with fp16 weights.  The left/right tensors are multiplied by their
   respective gates and (optionally) by a mask – this fusion is done in a
   custom Triton kernel.
4. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication for each hidden slice `h`.  This is
   realised with a fully‑fused Triton kernel that tiles the output matrix and
   accumulates with tensor‑core friendly fp16 · fp16 → fp32 dot products.
5. A second LayerNorm over the hidden dimension, multiplication by the
   out‑gate, and a final linear projection back to `dim`.
6. The result is cast back to the input dtype (usually float32).

Only the mask‑gate fusion and the O(N³) reduction are executed in Triton;
all other ops use optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask index (B,N,N) = flatten index // hidden_dim
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(left_out_ptr + offs, lp * lg * m, mask=valid)
    tl.store(right_out_ptr + offs, rp * rg * m, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """Fuse mask, projection and gate for left and right tensors."""
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Triton kernel: core TriMul reduction (batched GEMM across hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_core_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # 2‑D grid for output tiles, 1‑D for (batch, hidden) combined
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile col
    pid_z = tl.program_id(2)  # combined batch*hidden index

    # decode batch and hidden indices
    b = pid_z // H
    h = pid_z % H

    # base offsets for this (b,h)
    offset_l = b * stride_l_b + h * stride_l_h
    offset_r = b * stride_r_b + h * stride_r_h
    offset_o = b * stride_o_b + h * stride_o_h

    # tile offsets in i and j dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (i, k)
        left_ptrs = left_ptr + offset_l + \
            offs_i[:, None] * stride_l_i + \
            offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile: (j, k) => we need right[b, k, j, h] layout
        right_ptrs = right_ptr + offset_r + \
            offs_j[:, None] * stride_r_i + \
            offs_k[None, :] * stride_r_k
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0)

        # dot product (M,K) · (N,K) -> (M,N)
        acc += tl.dot(left, right)

    # write back
    out_ptrs = out_ptr + offset_o + \
        offs_i[:, None] * stride_o_i + \
        offs_j[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections and gates (fp16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate = F.linear(x, left_gate_w).sigmoid()    # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid()  # (B, N, N, H)
    out_gate = F.linear(x, out_gate_w).sigmoid()      # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (fused via Triton when needed)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            block_size=1024,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 6) Core reduction (batched GEMM) via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)   # (B, N, N, H)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    _tri_mul_core_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to the original dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
