"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input over the channel dimension (float32 for stability).
2. Cast to FP16 and apply linear projections (left/right) to the hidden dimension.
3. Compute three sigmoid gates (left, right, out) via linear layers.
4. Fuse mask (if present) and gates into the left/right tensors with a
   custom Triton kernel (in‑place, no temporaries).
5. Core O(N³·H) reduction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a tiled batched GEMM in Triton (no extra transposes).
6. Layer‑normalize the hidden representation, apply the output gate,
   and project back to the original channel dimension.
7. Cast the result back to the input dtype.

Only the mask‑gate fusion and the O(N³·H) reduction are executed in Triton.
All other steps use PyTorch (high‑performance cuBLAS / cuDNN kernels).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (B,N,N,H) flattened
    right_ptr,         # fp16 * right (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and gates:
       left  = left  * mask * left_gate
       right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # Load values
    l = tl.load(left_ptr + offs, mask=valid)
    r = tl.load(right_ptr + offs, mask=valid)
    lg = tl.load(left_gate_ptr + offs, mask=valid)
    rg = tl.load(right_gate_ptr + offs, mask=valid)

    # Compute mask index (one mask per (b,i,k) element)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=valid)

    # Apply fusion and store back
    tl.store(left_ptr + offs,  l * m * lg, mask=valid)
    tl.store(right_ptr + offs, r * m * rg, mask=valid)


@triton.jit
def _tri_mul_reduce_kernel(
    left_ptr,   # fp16 * left  (B,N,N,H)
    right_ptr,  # fp16 * right (B,N,N,H)
    out_ptr,    # fp16 * out   (B,N,N,H)
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,
    stride_la: tl.int64, stride_li: tl.int64, stride_lk: tl.int64, stride_lh: tl.int64,
    stride_rb: tl.int64, stride_rj: tl.int64, stride_rk: tl.int64, stride_rh: tl.int64,
    stride_oa: tl.int64, stride_oi: tl.int64, stride_oj: tl.int64, stride_oh: tl.int64,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]"""

    pid0 = tl.program_id(0)   # combined batch * hidden
    pid_m = tl.program_id(1)   # tile row (i)
    pid_n = tl.program_id(2)   # tile col (j)

    # decode batch and hidden indices
    b = pid0 // H
    h = pid0 % H

    # compute start indices for this tile
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # ranges within the tile
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)
    k_base = tl.arange(0, BLOCK_K)      # will be offset inside the loop

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k = kk * BLOCK_K + k_base

        # Pointers for A (left) tile: shape (M, K)
        a_ptrs = left_ptr + \
                 b * stride_la + \
                 i[:, None] * stride_li + \
                 k[None, :] * stride_lk + \
                 h * stride_lh

        # Pointers for B (right) tile: shape (K, N)
        b_ptrs = right_ptr + \
                 b * stride_rb + \
                 j[None, :] * stride_rj + \
                 k[:, None] * stride_rk + \
                 h * stride_rh

        # Masks for out‑of‑bounds loads
        a_mask = (i[:, None] < N) & (k[None, :] < N)
        b_mask = (k[:, None] < N) & (j[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write the result back (cast to FP16)
    c_ptrs = out_ptr + \
             b * stride_oa + \
             i[:, None] * stride_oi + \
             j[None, :] * stride_oj + \
             h * stride_oh

    c_mask = (i[:, None] < N) & (j[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] (binary) or None
            - weights: dict of model parameters (see keys in docstring)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to FP16 for compute‑heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w) # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse mask and gates (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elems = left.numel()                       # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(total_elems, BLOCK),)
        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elems,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: batched matrix multiplication using Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # (B, N, N, H) FP16

    # Strides (in elements) for each tensor
    s_la, s_li, s_lk, s_lh = left.stride()
    s_rb, s_rj, s_rk, s_rh = right.stride()
    s_oa, s_oi, s_oj, s_oh = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _tri_mul_reduce_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        s_la, s_li, s_lk, s_lh,
        s_rb, s_rj, s_rk, s_rh,
        s_oa, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
