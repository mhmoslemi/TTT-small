def custom_kernel(data):
    x, mask, weights, config = data
    B,N,N,C = x.shape
    dim = config['dim']; hidden_dim = config['hidden_dim']; nomask = config.get('nomask', False)

    # LayerNorm (fp32)
    x_norm = F.layer_norm(x, (dim,), weight=weights['norm.weight'].to(dtype=torch.float32), bias=weights['norm.bias'].to(dtype=torch.float32))
    # Cast to fp16 for heavy ops
    x_norm = x_norm.to(torch.float16)

    # Linear combine:
    # left_proj + left_gate
    left_concat_w = torch.cat([weights['left_proj.weight'], weights['left_gate.weight']], dim=1)  # shape (C, 2*H)
    right_concat_w = torch.cat([weights['right_proj.weight'], weights['right_gate.weight']], dim=1)  # shape (C, 2*H)

    # Matmul: flatten B,N,N -> (B*N*N, C)
    x_flat = x_norm.reshape(-1, dim)  # (B*N*N, C)

    left_concat = torch.matmul(x_flat, left_concat_w.to(torch.float16))  # (B*N*N, 2*H)
    right_concat = torch.matmul(x_flat, right_concat_w.to(torch.float16))

    # Reshape
    left_concat = left_concat.view(B, N, N, 2*hidden_dim)
    right_concat = right_concat.view(B, N, N, 2*hidden_dim)

    # split
    left_proj, left_gate_raw = left_concat[..., :hidden_dim], left_concat[..., hidden_dim:]
    right_proj, right_gate_raw = right_concat[..., :hidden_dim], right_concat[..., hidden_dim:]

    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # Apply mask (if any)
    if mask is not None and not nomask:
        mask = mask.to(torch.float16).unsqueeze(-1)  # shape (B,N,N,1)
    else:
        mask = torch.ones((B, N, N, 1), dtype=torch.float16, device=x.device)

    left = left_proj * mask * left_gate
    right = right_proj * mask * right_gate

    # Now compute core reduction using bmm

    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # out_flat = left_flat @ right_flat.transpose(-2, -1)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1,2))

    # reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # second layer norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights['to_out_norm.weight'].to(torch.float16), bias=weights['to_out_norm.bias'].to(torch.float16))

    # out_gate (computed earlier)
    out_gate = torch.sigmoid(torch.matmul(x_flat, weights['out_gate.weight'].to(torch.float16)).view(B,N,N,hidden_dim))

    out = out * out_gate

    # final projection
    out = torch.matmul(out.view(-1, hidden_dim), weights['to_out.weight'].to(torch.float16)).view(B,N,N,dim)

    # cast back to original dtype
    out = out.to(x.dtype)

    return out

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    # shape unpack
    B, N, _, C = input_tensor.shape
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', False)

    # LayerNorm (fp32)
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights['norm.weight'].to(dtype=torch.float32, device=input_tensor.device),
                     bias=weights['norm.bias'].to(dtype=torch.float32, device=input_tensor.device))
    # Cast to fp16
    x = x.to(torch.float16)

    # Combine weights for left and right (proj + gate)
    left_concat_w = torch.cat([weights['left_proj.weight'], weights['left_gate.weight']], dim=1)  # (C, 2*H)
    right_concat_w = torch.cat([weights['right_proj.weight'], weights['right_gate.weight']], dim=1)

    # Ensure weights cast to fp16
    left_concat_w = left_concat_w.to(torch.float16)
    right_concat_w = right_concat_w.to(torch.float16)

    # Flatten input for matmul
    x_flat = x.reshape(-1, dim)  # (B*N*N, C)

    # Compute concatenated results
    left_concat = torch.matmul(x_flat, left_concat_w)  # (B*N*N, 2*H)
    right_concat = torch.matmul(x_flat, right_concat_w)

    # Reshape
    left_concat = left_concat.view(B, N, N, 2*hidden_dim)
    right_concat = right_concat.view(B, N, N, 2*hidden_dim)

    # split
    left_proj = left_concat[..., :hidden_dim]
    left_gate_raw = left_concat[..., hidden_dim:]
    right_proj = right_concat[..., :hidden_dim]
    right_gate_raw = right_concat[..., hidden_dim:]

    # compute sigmoid gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # compute out_gate
    out_gate_w = weights['out_gate.weight'].to(torch.float16)
    out_gate_raw = torch.matmul(x_flat, out_gate_w).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(out_gate_raw)

    # Prepare mask broadcast
    if mask is not None and not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
    else:
        mask_fp = torch.ones((B, N, N, 1), dtype=torch.float16, device=input_tensor.device)

    # Fuse mask multiplication with gate using Triton
    total_elements = left_proj.numel()
    BLOCK_SIZE = 1024

    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    _mask_gate_fusion_kernel[grid](
        left_ptr=left_proj,
        right_ptr=right_proj,
        mask_ptr=mask_fp.view(-1),
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        left_out_ptr=left_out,
        right_out_ptr=right_out,
        H=hidden_dim,
        total_elements=total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
        # maybe add num_stages=2 default
    )

    # Core reduction 
    # reshape for batched matmul
    left_flat = left_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B,N,N,hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # second layer norm

                       (hidden_dim,),

    out = torch.matmul(out.view(-1, hidden_dim), to_out_w).view(B, N, N, dim)

    # Cast back to original dtype
    out = out.to(input_tensor.dtype)

    return out

custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor (fp32 for stability).
2. Fuse the two linear projections (proj + gate) for the left and right
   branches by concatenating the weight matrices and a single
   `F.linear` call (half‑precision).
3. Compute the three sigmoid gates (left, right, out).
4. Fuse mask, projection and gate multiplication in a tiny Triton kernel:
       left_out  = left_proj  * mask * left_gate
       right_out = right_proj * mask * right_gate
5. Core reduction: for each batch and hidden channel compute
       out[b,i,j,d] = Σ_k left_out[b,i,k,d] * right_out[b,j,k,d]
   Implemented as a batched matrix multiplication via `torch.bmm`.
6. Apply a second LayerNorm over the hidden dimension.
7. Multiply by the output gate and project back to the original dimension.
8. Cast the result to the original dtype.

Only the mask‑gate fusion is performed in Triton; all other heavy
linear operations use highly‑optimized cuBLAS via `torch.nn.functional.linear`
and `torch.bmm`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # const * left  (B,N,N,H) flattened
    right_ptr,         # const * right (B,N,N,H) flattened
    mask_ptr,          # const * mask  (B,N,N)   flattened
    left_gate_ptr,     # const * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # const * right_gate (B,N,N,H) flattened
    left_out_ptr,      # * left_out  (B,N,N,H) flattened
    right_out_ptr,     # * right_out (B,N,N,H) flattened
    H: tl.constexpr,                # hidden dimension
    total_elements: tl.constexpr,   # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask and gates:
       left_out  = left  * mask * left_gate
       right_out = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load values
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Compute index into mask (mask repeats for each hidden dim)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply mask and gates, store results
    tl.store(left_out_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (LayerNorm, Linear weights, etc.)
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, if True ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to half‑precision for the heavy computations
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Fuse proj + gate linear layers (left & right)
    # ------------------------------------------------------------------
    # Concatenate weights along the output‑feature dimension
    left_weight_cat = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)               # (2*H, C)
    right_weight_cat = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)               # (2*H, C)

    # Single linear call per side (no bias)
    left_cat = F.linear(x, left_weight_cat)   # (B, N, N, 2*H)
    right_cat = F.linear(x, right_weight_cat) # (B, N, N, 2*H)

    # Split projection and gate components
    left_proj = left_cat[..., :hidden_dim]          # (B, N, N, H)
    left_gate_raw = left_cat[..., hidden_dim:]      # (B, N, N, H)
    right_proj = right_cat[..., :hidden_dim]        # (B, N, N, H)
    right_gate_raw = right_cat[..., hidden_dim:]    # (B, N, N, H)

    # Sigmoid gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # ------------------------------------------------------------------
    # 3) Output gate
    # ------------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_raw = F.linear(x, out_gate_weight)      # (B, N, N, H)
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 4) Prepare mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    B, N, _, _ = left_proj.shape
    if mask is not None and not nomask:
        mask_fp = mask.to(device=device, dtype=torch.float16)   # (B, N, N)
    else:
        mask_fp = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 5) Fuse mask and gate multiplication in Triton
    # ------------------------------------------------------------------
    total_elements = left_proj.numel()          # B * N * N * H
    BLOCK_SIZE = 1024

    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)
    _mask_gate_fusion_kernel[grid](
        left_ptr=left_proj,
        right_ptr=right_proj,
        mask_ptr=mask_fp.view(-1),
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        left_out_ptr=left_out,
        right_out_ptr=right_out,
        H=hidden_dim,
        total_elements=total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6) Core reduction: batched GEMM (left_out @ right_outᵀ)
    #    Reshape to (B*H, N, N) for torch.bmm.
    # ------------------------------------------------------------------
    left_flat = left_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication (cuBLAS)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 7) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)   # (B, N, N, H)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 8) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float16),
    )

    # ------------------------------------------------------------------
    # 9) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10) Final projection back to original channel dimension C
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_weight)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 11) Cast back to the original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
