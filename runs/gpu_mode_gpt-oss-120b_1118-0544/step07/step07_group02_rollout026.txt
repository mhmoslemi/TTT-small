@triton.jit
def batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH, M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start indices
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for M,N out-of-bounds
    mask_m = off_m < M
    mask_n = off_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, K, BLOCK_K):
        # compute k range
        off_k = k + tl.arange(0, BLOCK_K)
        mask_k = off_k < K

        # load A tile (M x K)
        a_ptrs = left_ptr + batch_id * stride_l_batch + off_m[:, None] * stride_l_m + off_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # load B tile (K x N)
        b_ptrs = right_ptr + batch_id * stride_r_batch + off_k[:, None] * stride_r_k + off_n[None, :] * stride_r_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

        # compute dot
        acc += tl.dot(a, b)

    # store output
    out_ptrs = out_ptr + batch_id * stride_o_batch + off_m[:, None] * stride_o_m + off_n[None, :] * stride_o_n
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input LayerNorm
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    # Use torch.nn.functional.layer_norm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # 2. Cast to fp16
    x = x.to(torch.float16)

    # 3. Linear projections and gates
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = torch.nn.functional.linear(x, left_proj_w)  # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # 4. Apply mask and gates
    if mask is not None and not nomask:
        # mask shape [B,N,N], need broadcast to hidden dim
        mask = mask.to(device).to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask
    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # 5. Core contraction using Triton
    B, N, _, H = left.shape
    # reshape to (batch, M, K) and (batch, K, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B,H,N,N) but K,N swapped

    batch = B * H
    left_flat = left_perm.view(batch, N, N)
    right_flat = right_perm.view(batch, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)  # shape (batch, N, N)

    # get strides
    stride_l_batch, stride_l_m, stride_l_k = left_flat.stride()
    stride_r_batch, stride_r_k, stride_r_n = right_flat.stride()
    stride_o_batch, stride_o_m, stride_o_n = out_flat.stride()

    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch,
        N,
        N,
        N,
        stride_l_batch, stride_l_m, stride_l_k,
        stride_r_batch, stride_r_k, stride_r_n,
        stride_o_batch, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Output layernorm, out gate, final linear
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to float32
    out = out.to(torch.float32)
    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Input LayerNorm (float32 for stability).
2. Cast to fp16 for the compute‑heavy path.
3. Linear projections (left/right) and three sigmoid gates (left, right, out).
4. Apply optional mask and fuse it with the left/right gates (pure PyTorch).
5. Core O(N³·H) contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (L @ Rᵀ) using a custom Triton kernel.
6. Output LayerNorm, multiply by output gate, final linear projection back to dim.
7. Cast result to float32.

The kernel in step 5 processes all hidden channels (batch*hidden_dim) as an
extra batch dimension and performs a blocked matrix multiplication with
float16 inputs and float32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(
    left_ptr,            # fp16, shape (BATCH, M, K)
    right_ptr,           # fp16, shape (BATCH, K, N)  (already transposed)
    out_ptr,             # fp16, shape (BATCH, M, N)
    BATCH: tl.constexpr,
    M: tl.constexpr,
    N: tl.constexpr,
    K: tl.constexpr,
    stride_l_batch: tl.constexpr,
    stride_l_m: tl.constexpr,
    stride_l_k: tl.constexpr,
    stride_r_batch: tl.constexpr,
    stride_r_k: tl.constexpr,
    stride_r_n: tl.constexpr,
    stride_o_batch: tl.constexpr,
    stride_o_m: tl.constexpr,
    stride_o_n: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = (left_ptr
                  + batch_id * stride_l_batch
                  + offs_m[:, None] * stride_l_m
                  + offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B tile: (BLOCK_K, BLOCK_N)   (right is already transposed)
        b_ptrs = (right_ptr
                  + batch_id * stride_r_batch
                  + offs_k[:, None] * stride_r_k
                  + offs_n[None, :] * stride_r_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product (float16 inputs, float32 accumulate)
        acc += tl.dot(a, b)

    # write results
    out_ptrs = (out_ptr
                + batch_id * stride_o_batch
                + offs_m[:, None] * stride_o_m
                + offs_n[None, :] * stride_o_n)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)   (float32)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int)          – input channel dimension C
                * "hidden_dim" (int)   – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with dtype torch.float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for the compute‑heavy part
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gates (all fp16)
    # -----------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)                    # (B, N, N, H)
    right = F.linear(x, right_proj_w)                  # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w)) # (B, N, N, H)

    # -----------------------------------------------------------------
    # 4. Apply mask (if present) and gates
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp
        right = right * mask_fp

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core O(N^3·H) contraction via Triton batched GEMM
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape for batched GEMM
    # left: (B, H, M=N, K=N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right needs to be transposed on the last two dimensions: (B, H, K=N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    batch = B * H
    left_flat = left_perm.view(batch, N, N)    # (batch, M, K)
    right_flat = right_perm.view(batch, N, N)  # (batch, K, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)  # (batch, M, N)

    # strides (in elements)
    stride_l_batch, stride_l_m, stride_l_k = left_flat.stride()
    stride_r_batch, stride_r_k, stride_r_n = right_flat.stride()
    stride_o_batch, stride_o_m, stride_o_n = out_flat.stride()

    # block sizes – tuned for typical GPUs; they also work for the test sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch,
        N,          # M
        N,          # N
        N,          # K
        stride_l_batch, stride_l_m, stride_l_k,
        stride_r_batch, stride_r_k, stride_r_n,
        stride_o_batch, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # -----------------------------------------------------------------
    # 7. Cast back to float32 and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
