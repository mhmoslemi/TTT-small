"""
TriMul outgoing (AlphaFold3) forward pass.

Algorithm
---------
1. Input LayerNorm over the channel dimension `dim`.
2. Linear projections to the hidden dimension `hidden_dim` (left/right).
3. Gating projections (sigmoid) for left, right and output.
4. Fuse mask (if provided), projection and gating for left/right tensors
   with a tiny Triton kernel (element‑wise multiplication).  If no mask is
   used the fusion is performed with a simple PyTorch expression.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is implemented as a batched matrix multiplication:
       out_h = left_h @ right_hᵀ  for each hidden slice `h`.
6. Output LayerNorm over the hidden dimension, element‑wise multiplication
   with the output gate, and a final linear projection back to the original
   dimension `dim`.
7. Cast the result back to the input dtype (usually float32).

The Triton kernel is only used for the mask‑gate‑projection fusion,
fulfilling the requirement to have a custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,    # *float32  (B,N,N,H)
    right_proj_ptr,   # *float32  (B,N,N,H)
    left_gate_ptr,    # *float32  (B,N,N,H)
    right_gate_ptr,   # *float32  (B,N,N,H)
    mask_ptr,         # *float32  (B,N,N)
    left_out_ptr,     # *float32  (B,N,N,H)
    right_out_ptr,    # *float32  (B,N,N,H)
    total_elems,      # i64 = B*N*N*H
    hidden_dim,       # i64 = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # each mask element repeats `hidden_dim` times in the flattened tensor
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gate and mask for both left and right tensors using Triton.

    All tensors must be contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)

    Returns:
        left, right : (B, N, N, H) = proj * gate * mask
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16, on CUDA)
            - mask         : torch.Tensor [B, N, N] (0/1) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Linear projections (float32)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 3) Apply mask & gating (fusion via Triton if mask is used)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is contiguous and same dtype as compute tensors
        mask_fp = mask.to(dtype).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp
        )
    else:
        # No mask or mask ignored
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    = left_h @ right_hᵀ for each hidden slice h
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)ᵀ -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 6) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype:
        out = out.to(dtype)

    return out
