@triton.jit
def _mask_gate_fuse_kernel(left_ptr, right_ptr,
                           left_gate_ptr, right_gate_ptr,
                           mask_ptr,
                           total_elems, hidden_dim,
                           HAS_MASK: tl.constexpr,
                           BLOCK_SIZE: tl.constexpr):
    ...

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', False)

    # layernorm
    norm_weight = weights['norm.weight'].to(device)
    norm_bias = weights['norm.bias'].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    x = x.to(torch.float16)

    # combine left weights
    left_proj_weight = weights['left_proj.weight'].to(device).to(torch.float16)
    left_gate_weight = weights['left_gate.weight'].to(device).to(torch.float16)
    left_combined_weight = torch.cat([left_proj_weight, left_gate_weight], dim=1)  # (dim, 2*hidden)

    # compute combined left
    left_combined = F.linear(x, left_combined_weight)  # (B,N,N,2*hidden)
    left_proj, left_gate = left_combined[..., :hidden_dim], left_combined[..., hidden_dim:]
    left_gate = torch.sigmoid(left_gate)

    # combine right
    right_proj_weight = weights['right_proj.weight'].to(device).to(torch.float16)
    right_gate_weight = weights['right_gate.weight'].to(device).to(torch.float16)
    right_combined_weight = torch.cat([right_proj_weight, right_gate_weight], dim=1)
    right_combined = F.linear(x, right_combined_weight)
    right_proj, right_gate = right_combined[..., :hidden_dim], right_combined[..., hidden_dim:]
    right_gate = torch.sigmoid(right_gate)

    # out gate
    out_gate_weight = weights['out_gate.weight'].to(device).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # apply mask and gates (fuse)
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten
        left_proj_f = left_proj.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_proj_f = right_proj.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_fp.contiguous().view(-1)

        total = left_proj_f.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _mask_gate_fuse_kernel[grid](
            left_proj_f,
            right_proj_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_proj_f.view_as(left_proj)
        right = right_proj_f.view_as(right_proj)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # contraction: batched GEMM
    B, N, _, H = left.shape
    # reshape each batch and hidden as batch matrix dim
    # We'll compute left @ right^T for each hidden channel
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # output layernorm
    to_out_norm_weight = weights['to_out_norm.weight'].to(device).to(torch.float16)
    to_out_norm_bias = weights['to_out_norm.bias'].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # multiply by out_gate
    out = out * out_gate

    # final projection
    to_out_weight = weights['to_out.weight'].to(device).to(torch.float16)
    out = F.linear(out, to_out_weight)

    # cast back to float32
    return out.to(torch.float32)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Input LayerNorm (float32 for numerical stability).
2. Cast to fp16 for the compute‑heavy path.
3. Fuse the two linear projections (proj & gate) of the left and right
   branches by concatenating their weight matrices, reducing the number
   of matmul calls from four to two.
4. Apply sigmoid to the gate parts.
5. Fuse mask, gate and element‑wise multiplication in a tiny Triton kernel
   (in‑place on the flattened tensors).
6. Core contraction: for each hidden channel h compute
       out_h = left_h @ right_hᵀ
   using a batched GEMM (torch.bmm).  This is the dominant O(N³·H) work.
7. Output LayerNorm, multiply by the output gate, and final linear projection.
8. Cast back to float32.
The kernel in step 5 is mandatory to satisfy the “use Triton” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,               # fp16 *, shape [B,N,N,H] flattened
    right_ptr,              # fp16 *, shape [B,N,N,H] flattened
    left_gate_ptr,          # fp16 *, same shape
    right_gate_ptr,         # fp16 *, same shape
    mask_ptr,               # fp16 *, shape [B,N,N] flattened
    total_elems,            # i32  scalar = B*N*N*H
    hidden_dim,             # i32  scalar = H
    HAS_MASK: tl.constexpr, # bool compile‑time flag
    BLOCK_SIZE: tl.constexpr # i32 compile‑time block size
):
    """In‑place fusion:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
       If HAS_MASK is False the mask is ignored (treated as 1).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # load data
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each group of `hidden_dim` consecutive elements share the same mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # store results back
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask:        torch.Tensor of shape (B, N, N) or None
            - weights:     dict with the following keys (all torch.float32):
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight"
                * "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict containing:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True masks are ignored

    Returns:
        torch.Tensor of shape (B, N, N, C) with dtype torch.float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for the compute‑heavy path
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Fuse left projection + left gate (and similarly for right)
    # -----------------------------------------------------------------
    # --- left branch ---
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)   # (H, C)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)   # (H, C)
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0)            # (2*H, C)

    left_combined = F.linear(x, left_combined_w)  # (B, N, N, 2*H)
    left_proj, left_gate = torch.split(left_combined, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate)

    # --- right branch ---
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)  # (H, C)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)  # (H, C)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0)          # (2*H, C)

    right_combined = F.linear(x, right_combined_w)  # (B, N, N, 2*H)
    right_proj, right_gate = torch.split(right_combined, hidden_dim, dim=-1)
    right_gate = torch.sigmoid(right_gate)

    # -----------------------------------------------------------------
    # 4. Output gate (separate linear + sigmoid)
    # -----------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)  # (H, C)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))                    # (B, N, N, H)

    # -----------------------------------------------------------------
    # 5. Fuse mask, gate and element‑wise multiplication (Triton)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).contiguous()

        # flatten tensors for the kernel
        left_proj_f = left_proj.contiguous().view(-1)
        right_proj_f = right_proj.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_proj_f.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _mask_gate_fuse_kernel[grid](
            left_proj_f,
            right_proj_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_proj_f.view_as(left_proj)
        right = right_proj_f.view_as(right_proj)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 6. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (left @ rightᵀ) per hidden channel.
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) where each slice corresponds to a hidden channel
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiply
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # restore original shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7. Output LayerNorm, apply output gate, final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)  # (C, H)
    out = F.linear(out, to_out_w)  # (B, N, N, C)

    # -----------------------------------------------------------------
    # 8. Cast back to float32 and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
