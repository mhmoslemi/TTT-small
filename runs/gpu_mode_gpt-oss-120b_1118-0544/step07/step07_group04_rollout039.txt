def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_stride, N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_j, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offset for i and j
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers base for this batch
    left_batch_ptr = left_ptr + pid_batch * stride_left_batch
    right_batch_ptr = right_ptr + pid_batch * stride_right_batch
    out_batch_ptr = out_ptr + pid_batch * stride_out_batch

   # 1. LayerNorm

   # cast to fp16
   x = x.to(torch.float16)

   # cast weights to fp16
   # Linear weights:
   left_proj_w = weights["left_proj.weight"].to(torch.float16)
   right_proj_w = weights["right_proj.weight"].to(torch.float16)
   left_gate_w = weights["left_gate.weight"].to(torch.float16)
   right_gate_w = weights["right_gate.weight"].to(torch.float16)
   out_gate_w = weights["out_gate.weight"].to(torch.float16)
   to_out_norm_w = weights["to_out_norm.weight"]
   to_out_norm_b = weights["to_out_norm.bias"]
   to_out_w = weights["to_out.weight"]

   # compute projections
   left_proj = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
   right_proj = F.linear(x, right_proj_w)

   left_gate = F.linear(x, left_gate_w).sigmoid()
   right_gate = F.linear(x, right_gate_w).sigmoid()

       left = left * mask_
       right = right * mask_

"""
TriMul (outgoing) forward pass – Triton‑accelerated reduction.

Algorithm
---------
1. Layer‑norm the 4‑D input over the channel dimension (`dim`).
2. Cast to FP16 for compute‑heavy stages.
3. Linear projections → `left_proj`, `right_proj` (B,N,N,hidden_dim).
4. Gating projections → sigmoid(left_gate), sigmoid(right_gate), sigmoid(out_gate).
5. Fuse gating (and optional mask) with element‑wise multiplication (pure PyTorch,
   cheap compared to the N³ reduction).
6. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (A @ Bᵀ) using a custom Triton kernel that
   processes all (batch * hidden_dim) matrices in a single launch.
7. Output LayerNorm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.
8. Return the result in the original (float32) dtype.

The Triton kernel performs the N³ reduction with tiling (BLOCK_M×BLOCK_N
output tile, BLOCK_K reduction tile) and accumulates in FP32 for numerical
stability while storing the result in FP16.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_j, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM kernel for the outgoing triangle multiplicative update:
        out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]

    left, right, out are tensors of shape (B_total, N, N) where
    B_total = batch_size * hidden_dim. The kernel processes one
    (batch, hidden) pair per program_id(0) and tiles the (i,j) output
    plane with BLOCK_M×BLOCK_N.
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for the current batch (i.e. batch * hidden_dim).
    left_batch_ptr = left_ptr + pid_batch * stride_left_batch
    right_batch_ptr = right_ptr + pid_batch * stride_right_batch
    out_batch_ptr = out_ptr + pid_batch * stride_out_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K.
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile of A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_batch_ptr + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load a tile of Bᵀ: shape (BLOCK_K, BLOCK_N)
        # B is accessed as (k, j) so that the dot product computes Σ_k A[i,k] * B[j,k].
        b_ptrs = right_batch_ptr + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write the result back (cast to half).
    c_ptrs = out_batch_ptr + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32/float16/bool) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim` (FP32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy stages
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast linear weights to FP16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"]          # keep FP32
    to_out_norm_b = weights["to_out_norm.bias"]           # keep FP32
    to_out_w      = weights["to_out.weight"]              # keep FP32

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid (B,N,N,H)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Apply gating and optional mask (simple PyTorch ops – cheap)
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    if mask is not None and not nomask:
        # mask shape (B, N, N) → broadcast over hidden dimension
        mask_fp = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a single Triton kernel over (batch*hidden) matrices.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Rearrange to (B, H, N, N) and flatten batch*head dimension.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch_total = B * H

    left_flat  = left_perm.view(batch_total, N, N)
    right_flat = right_perm.view(batch_total, N, N)

    out_flat = torch.empty_like(left_flat)  # FP16 output

    # Strides for the flattened tensors (they are contiguous)
    s_left_batch = left_flat.stride(0)
    s_left_i     = left_flat.stride(1)
    s_left_k     = left_flat.stride(2)

    s_right_batch = right_flat.stride(0)
    s_right_j     = right_flat.stride(1)   # this is the "row" dimension (j)
    s_right_k     = right_flat.stride(2)   # column dimension (k)

    s_out_batch = out_flat.stride(0)
    s_out_i     = out_flat.stride(1)
    s_out_j     = out_flat.stride(2)

    # Tiling parameters – chosen to fit into registers/shared memory.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_total,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        s_left_batch,
        s_left_i,
        s_left_k,
        s_right_batch,
        s_right_j,
        s_right_k,
        s_out_batch,
        s_out_i,
        s_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = out.to(torch.float32)  # cast to FP32 for stable layer‑norm
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
