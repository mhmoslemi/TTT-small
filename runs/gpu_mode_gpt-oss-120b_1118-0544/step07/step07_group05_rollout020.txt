"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input (float32) → fp16.
2. Fuse each projection with its gate:
       (x·W_proj , x·W_gate) are computed together as a single GEMM
       (weight matrices are concatenated along the output dimension).
   Apply sigmoid to the gate tensor.
3. Apply optional mask and gate element‑wise (fused in a tiny Triton kernel).
4. Core O(N³·H) reduction:
       For each batch‑hidden pair compute
           out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
       Implemented as a batched fp16 GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ.
5. Layer‑normalize over the hidden dimension.
6. Multiply by the output gate (sigmoid of a linear projection).
7. Final linear projection back to the original channel dimension.
8. Cast the result back to the input dtype (float32).

Only the mask‑gate fusion (step 3) is performed in Triton; all heavy
linear / matmul work uses PyTorch’s highly‑optimized cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (B,N,N,H) flattened
    right_ptr,         # fp16 * right (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and left/right gates:
       left  = left  * mask * left_gate
       right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load values
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Broadcast mask: one mask value per hidden element
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply and store back in‑place
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see expected keys below)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability) → fp16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)  # work in fp16 for speed

    # ------------------------------------------------------------------
    # 2) Prepare fp16 weights
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Fuse projection and gate linear layers (two GEMMs → two GEMMs)
    # ------------------------------------------------------------------
    # Concatenate projection and gate weights along the output dimension
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, C)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, C)

    # Compute both projection and gate in one pass each
    left_combined = F.linear(x, left_combined_w)   # (B, N, N, 2*H)
    right_combined = F.linear(x, right_combined_w) # (B, N, N, 2*H)

    # Split projection and gate
    left = left_combined[..., :hidden_dim]          # (B,N,N,H)
    left_gate_pre = left_combined[..., hidden_dim:] # (B,N,N,H)
    right = right_combined[..., :hidden_dim]        # (B,N,N,H)
    right_gate_pre = right_combined[..., hidden_dim:]# (B,N,N,H)

    # Apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 4) Apply mask and gate (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()                     # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates have been applied; we can free them
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5) Output gate (separate linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate_pre = F.linear(x, out_gate_w)   # (B,N,N,H)
    out_gate = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------------
    # 6) Core reduction: batched matrix multiplication
    #    left and right: (B, N, N, H) → reshape to (B*H, N, N)
    # ------------------------------------------------------------------
    batch_size, seq_len, _, _ = left.shape
    left_b  = left.permute(0, 3, 1, 2).reshape(batch_size * hidden_dim, seq_len, seq_len)
    right_b = right.permute(0, 3, 1, 2).reshape(batch_size * hidden_dim, seq_len, seq_len)

    # Compute out_b = left_b @ right_bᵀ   (B*H, N, N)
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_b.view(batch_size, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
