@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_abatch,
    stride_bk, stride_bn, stride_bbatch,
    stride_cm, stride_cn, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # shape: a (M,K), b (K,N), c (M,N) for each batch/head
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Pointers offset to this batch/head
    a_base = a_ptr + batch_id * stride_abatch
    b_base = b_ptr + batch_id * stride_bbatch
    c_base = c_ptr + batch_id * stride_cbatch

    # Compute the start indices for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        k_valid = k_base + tl.arange(0, BLOCK_K)
        # Load A and B tiles
        a_ptrs = a_base + (offs_m[:, None] * stride_am + (k_valid[None, :] + 0) * stride_ak)
        b_ptrs = b_base + ((k_valid[:, None] + 0) * stride_bk + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (k_valid[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k_valid[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        # Convert to fp16? We'll load as fp16 and cast to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_base + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated gating, mask fusion,
and a custom batched GEMM kernel for the N³ reduction.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (`dim`).
2. Cast to FP16 for the compute‑intensive part.
3. Linear projections (`left_proj`, `right_proj`) and gate projections
   (`left_gate`, `right_gate`, `out_gate`) – all bias‑free.
4. Fuse `proj * gate * mask` (if a mask is present) with a tiny Triton
   kernel to avoid extra memory traffic.
5. Core reduction:
   For each batch element and hidden channel `h`,
   compute `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`,
   i.e. a batched matrix multiplication `left @ rightᵀ`.
   Implemented with a custom Triton GEMM kernel (no cuBLAS call).
6. Output LayerNorm, multiply by `out_gate`, and final linear projection
   back to the original dimension.
7. Return result in the original dtype (usually float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse projection, gate and optional mask (mask broadcasted)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # int: B*N*N*H
    hidden_dim,        # int: H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    # index of the mask element (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 8192) -> torch.Tensor:
    """
    Fuse projection, gate and mask using a Triton kernel.
    All tensors must be fp16, contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N) broadcast over H
    Returns proj * gate * mask (shape (B,N,N,H)).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel 2: batched matrix multiplication (A @ Bᵀ) for many heads
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_abatch,
    stride_bk, stride_bn, stride_bbatch,
    stride_cm, stride_cn, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes for each batch_head:
        C = A @ Bᵀ
    where A is (M, K), B is (N, K) (original B is (N, K) but we treat it as transposed).
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Pointers to the start of this batch/head
    a_batch = a_ptr + batch_id * stride_abatch
    b_batch = b_ptr + batch_id * stride_bbatch
    c_batch = c_ptr + batch_id * stride_cbatch

    # Compute the start offsets of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_base = k * BLOCK_K
        k_offs = k_base + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (k_offs[None, :] < K)
        mask_b = (k_offs[:, None] < K) & (offs_n[None, :] < N)

        # Load tiles
        a_ptrs = a_batch + (offs_m[:, None] * stride_am + k_offs[None, :] * stride_ak)
        b_ptrs = b_batch + (k_offs[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)

    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)

            - mask         : torch.Tensor of shape [B, N, N] (float32/float16/bool) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy stages
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all linear weights to FP16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5. Gate projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()   # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid()  # (B, N, N, H)
    out_gate   = F.linear(x, out_gate_w).sigmoid()    # (B, N, N, H)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for the batched kernel
    left_t  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_t   = torch.empty_like(left_t)   # (B*H, N, N), fp16

    # Strides
    stride_am = left_t.stride(1)    # stride between rows (M)
    stride_ak = left_t.stride(2)    # stride between columns (K)
    stride_abatch = left_t.stride(0)

    # For B we treat it as (K, N) = transpose of the original (N, K)
    stride_bk = right_t.stride(2)   # K dimension (original last)
    stride_bn = right_t.stride(1)   # N dimension (original middle)
    stride_bbatch = right_t.stride(0)

    stride_cm = out_t.stride(1)
    stride_cn = out_t.stride(2)
    stride_cbatch = out_t.stride(0)

    # Tile sizes – tuned for typical B200 resources

    BLOCK_N = 128
    BLOCK_K = 32

    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _batched_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N, N, N,
        stride_am, stride_ak, stride_abatch,
        stride_bk, stride_bn, stride_bbatch,
        stride_cm, stride_cn, stride_cbatch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_t.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,

    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
