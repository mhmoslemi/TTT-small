import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_Ab, stride_Am, stride_Ak,
                          stride_Bb, stride_Bk, stride_Bn,
                          stride_Cb, stride_Cm, stride_Cn,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # load A tile
        a_ptrs = A_ptr + pid_batch * stride_Ab + offs_m[:, None] * stride_Am + k_offset[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load B tile
        b_ptrs = B_ptr + pid_batch * stride_Bb + k_offset[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # compute dot product
        acc += tl.dot(a, b)

    # write back
    c_ptrs = C_ptr + pid_batch * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def _batched_matmul_triton(left, right):
    """
    left, right: torch tensors of shape (B, N, N, H)
    Compute out = einsum('... i k h, ... j k h -> ... i j h')
    using a batched GEMM kernel.
    """
    B, N, _, H = left.shape
    # permute to (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    right_flat = right_t.view(B * H, N, N)

    K = N

"""
TriMul outgoing (AlphaFold3) forward pass implemented with Triton.

Algorithm:
1. Layer‑norm the input tensor over the channel dimension.
2. Cast to FP16 for compute‑intensive parts.
3. Linear projections (left/right) and gating (left/right/out) using
   bias‑free Linear layers.  Gating uses sigmoid.
4. Fuse mask (if present) with the projected tensors.
5. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication A @ Bᵀ for every hidden slice.
   A custom Triton kernel performs the batched GEMM (A: (B*H,N,N),
   Bᵀ: (B*H,N,N)).
6. Layer‑norm over the hidden dimension, apply out‑gate and a final
   linear projection back to the original channel dimension.
7. Cast the result back to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ B (both batched, no bias).

    A, B, C are interpreted as batches of M×K and K×N matrices
    (row‑major).  The batch dimension is the first axis.
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / columns
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current tile of A and B
        a_ptrs = (A_ptr
                  + pid_batch * stride_Ab
                  + offs_m[:, None] * stride_Am
                  + offs_k[None, :] * stride_Ak)
        b_ptrs = (B_ptr
                  + pid_batch * stride_Bb
                  + offs_k[:, None] * stride_Bk
                  + offs_n[None, :] * stride_Bn)

        # Load tiles (mask out‑of‑bounds elements)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate the dot product
        acc += tl.dot(a, b)

    # Write the result back in fp16
    c_ptrs = (C_ptr
              + pid_batch * stride_Cb
              + offs_m[:, None] * stride_Cm
              + offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the TriMul reduction using a custom Triton batched GEMM.

    Arguments:
        left, right: tensors of shape (B, N, N, H) in fp16.
    Returns:
        out: tensor of shape (B, N, N, H) in fp16.
    """
    B, N, _, H = left.shape

    # Bring hidden dimension to the batch axis for the GEMM
    # (B, N, N, H) -> (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # Transpose the last two axes of right (A @ Bᵀ)
    right_t = right.transpose(-2, -1).contiguous()

    # Collapse batch & hidden into a single batch dimension
    left_flat = left.view(B * H, N, N)
    right_flat = right_t.view(B * H, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Strides (in elements)
    stride_Ab, stride_Am, stride_Ak = left_flat.stride()
    stride_Bb, stride_Bk, stride_Bn = right_flat.stride()
    stride_Cb, stride_Cm, stride_Cn = out_flat.stride()

    # Problem dimensions
    M = N
    N_ = N
    K = N

    # Tile sizes – chosen to fit into shared memory on a B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N_ + BLOCK_N - 1) // BLOCK_N)

    _batched_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        M, N_, K,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # Preserve original dtype for the final cast
    orig_dtype = x.dtype

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast all linear weights to fp16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (element‑wise)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction via a fused batched GEMM (Triton)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    out = _batched_matmul_triton(left, right)   # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out

"""
TriMul outgoing (AlphaFold3) forward pass implemented with Triton.

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to FP16 for compute‑heavy parts.
3. Linear projections (left / right) and gating (left / right / out) using
   bias‑free Linear layers.  Gating uses sigmoid.
4. Fuse the optional mask with the projected tensors (element‑wise).
5. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication A @ Bᵀ for every hidden slice.
   A custom Triton kernel performs the batched GEMM (A: (B*H,N,N),
   Bᵀ: (B*H,N,N)).
6. Layer‑norm over the hidden dimension, apply out‑gate and a final linear
   projection back to the original channel dimension.
7. Cast the result back to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ B   (A: (batch, M, K), B: (batch, K, N))
    The batch dimension is the first axis.
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = (A_ptr
                  + pid_batch * stride_Ab
                  + offs_m[:, None] * stride_Am
                  + offs_k[None, :] * stride_Ak)
        b_ptrs = (B_ptr
                  + pid_batch * stride_Bb
                  + offs_k[:, None] * stride_Bk
                  + offs_n[None, :] * stride_Bn)

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    c_ptrs = (C_ptr
              + pid_batch * stride_Cb
              + offs_m[:, None] * stride_Cm
              + offs_n[None, :] * stride_Cn)

    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the TriMul reduction using the Triton batched GEMM kernel.

    Parameters
    ----------
    left, right : torch.Tensor
        Tensors of shape (B, N, N, H) in fp16.

    Returns
    -------
    out : torch.Tensor
        Tensor of shape (B, N, N, H) in fp16.
    """
    B, N, _, H = left.shape

    # Move hidden dimension to the batch axis.
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Transpose the last two dimensions of `right` to get Bᵀ.
    right_t = right.transpose(-2, -1).contiguous()  # (B, H, N, N)

    # Collapse batch & hidden into a single batch dimension.
    left_flat = left.view(B * H, N, N)
    right_flat = right_t.view(B * H, N, N)

    # Allocate output.
    out_flat = torch.empty_like(left_flat)

    # Strides (in elements).
    stride_Ab, stride_Am, stride_Ak = left_flat.stride()
    stride_Bb, stride_Bk, stride_Bn = right_flat.stride()
    stride_Cb, stride_Cm, stride_Cn = out_flat.stride()

    M = N
    N_ = N
    K = N

    # Tile sizes (tuned for B200).
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N_ + BLOCK_N - 1) // BLOCK_N)

    _batched_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        M, N_, K,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H).
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
        - mask         : torch.Tensor [B, N, N] (0/1) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    orig_dtype = x.dtype

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (element‑wise)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction – batched GEMM via Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    out = _batched_matmul_triton(left, right)   # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
