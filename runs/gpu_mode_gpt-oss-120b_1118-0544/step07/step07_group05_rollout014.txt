"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑normalize the input (float32 for stability).
2. Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
   into a single GEMM to reduce cuBLAS launch overhead.
3. Apply sigmoid to the three gate tensors.
4. Fuse mask, left‑gate, and right‑gate multiplications in a Triton kernel
   (in‑place on `left` and `right`).
5. Core update: for every hidden dimension h,
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication (left @ rightᵀ).
6. Layer‑normalize over the hidden dimension.
7. Apply the output gate.
8. Final linear projection back to the original channel dimension.
9. Return the result in the input dtype.

The Triton kernel only handles the element‑wise mask‑gate fusion; all
dense linear layers and the O(N³) reduction use highly‑optimized cuBLAS
operations via PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (B,N,N,H) flattened
    right_ptr,         # fp16 * right (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and left/right gates.
       left  = left  * mask * left_gate
       right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load values (masked)
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # One mask value per (b,i,k) -> broadcast over H
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply and store back
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (see keys below)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # Cast to fp16 for the heavy compute
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Fuse the five linear projections into a single GEMM
    # ------------------------------------------------------------------
    # Concatenate weight matrices: shape (5*hidden_dim, dim)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)

    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)
    proj = F.linear(x_flat, fused_weight)        # (B*N*N, 5*hidden_dim)
    proj = proj.view(B, N, N, 5, hidden_dim)     # (B, N, N, 5, hidden_dim)

    left = proj[..., 0, :]    # (B, N, N, H)
    right = proj[..., 1, :]   # (B, N, N, H)
    left_gate = proj[..., 2, :]   # (B, N, N, H)
    right_gate = proj[..., 3, :]  # (B, N, N, H)
    out_gate = proj[..., 4, :]    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3) Apply sigmoid to gate tensors
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 4) Fuse mask and gate multiplications via Triton (in‑place)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()            # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # gates no longer needed
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5) Core reduction: batched matrix multiplication (left @ rightᵀ)
    # ------------------------------------------------------------------
    left_b = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    out_b = torch.matmul(left_b, right_b.transpose(-2, -1))   # (B, H, N, N)
    out = out_b.permute(0, 2, 3, 1).contiguous()              # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float16),
    )

    # ------------------------------------------------------------------
    # 7) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection H -> C
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
