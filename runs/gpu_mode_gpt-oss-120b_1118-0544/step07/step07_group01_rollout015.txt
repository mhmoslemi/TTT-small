    left_ptr, right_ptr, out_ptr,
    batch, size,

    stride_r_batch, stride_r_m, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,

):

   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)
   # offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

"""
custom_kernel: Triton‑accelerated outgoing TriMul forward (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for compute‑heavy stages.
3. Linear projections (left/right) and gate projections (left/right/out).
4. Fuse mask, projection and gate for the left/right tensors with a small
   Triton kernel (`_proj_gate_mask_kernel`).  This avoids an extra
   materialisation of the masked tensors.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication for each hidden slice `h`.
   Implemented with a dedicated Triton GEMM kernel (`_tri_mul_kernel`)
   that works on the flattened batch dimension `batch = B*H`.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   out‑gate, and project back to the original `dim`.
7. Cast the result back to the input dtype (typically float32).

Both Triton kernels are deliberately lightweight (no shared memory,
only `tl.dot`), ensuring high occupancy on an NVIDIA B200.
"""

@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B, N, N, H)
    right_proj_ptr,  # fp16 * (B, N, N, H)
    left_gate_ptr,   # fp16 * (B, N, N, H)

    mask_idx = offs // hidden_dim

    # load
    left_proj  = tl.load(left_proj_ptr  + offs, mask=active, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=active, other=0.0)
    left_gate  = tl.load(left_gate_ptr  + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)
    m          = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)

    # fused mask & gate
    left_out  = left_proj  * left_gate  * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr  + offs, left_out,  mask=active)
    tl.store(right_out_ptr + offs, right_out, mask=active)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse mask, projection and gate for left/right tensors.
    All tensors must be fp16 and contiguous.
    Returns (left, right) of shape (B,N,N,H).
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# 2. Core TriMul kernel (batched GEMM on (B*H, N, N))
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,   # fp16 * (batch, N, N)
    right_ptr,  # fp16 * (batch, N, N)
    out_ptr,    # fp16 * (batch, N, N)
    batch,      # i64  = B*H
    size,       # i64  = N
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_m, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    b_id = tl.program_id(0)          # batch dimension
    pid_m = tl.program_id(1)         # row tile
    pid_n = tl.program_id(2)         # col tile

    # tile start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # boundary masks
    mask_m = offs_m < size
    mask_n = offs_n < size

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(size, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < size

        # left tile: (M, K)
        a_ptrs = left_ptr + b_id * stride_l_batch + \
                 offs_m[:, None] * stride_l_m + \
                 offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right tile transposed: we need (K, N)
        b_ptrs = right_ptr + b_id * stride_r_batch + \
                 offs_k[:, None] * stride_r_m + \
                 offs_n[None, :] * stride_r_k
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product (M,K) x (K,N) -> (M,N)
        acc += tl.dot(a, b)

    # write out
    c_ptrs = out_ptr + b_id * stride_o_batch + \
             offs_m[:, None] * stride_o_m + \
             offs_n[None, :] * stride_o_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform batched matrix multiplication:
        out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]
    where left/right are (B*H, N, N) tensors (fp16, contiguous).
    Returns fp16 tensor of same shape.
    """
    batch, N, _ = left.shape
    out = torch.empty_like(left)

    # strides for contiguous (batch, M, K) layout
    stride_l_batch, stride_l_m, stride_l_k = left.stride()
    stride_r_batch, stride_r_m, stride_r_k = right.stride()
    stride_o_batch, stride_o_m, stride_o_n = out.stride()

    # block sizes – tuned for B200 (adjustable)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        batch,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_m,
        stride_r_k,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None (0/1)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast weights to FP16 (bias‑free linear layers)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)   # (B, N, N, H)

    left_gate  = F.linear(x, left_gate_w).sigmoid()   # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid()  # (B, N, N, H)
    out_gate   = F.linear(x, out_gate_w).sigmoid()    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (fused via Triton when needed)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16, contiguous, broadcastable
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            block_size=1024,
        )
    else:
        # No mask (or ignored) → simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM with a custom Triton kernel.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # bring hidden dim into the batch axis -> (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = _batched_gemm_triton(left_flat, right_flat)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
