"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor x ∈ [B,N,N,C] in FP32.
2. Cast to FP16 and compute linear projections (left/right) to hidden_dim H.
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. Fuse mask (if present) with left/right projections and their gates via a
   Triton kernel:
       left_out  = left  * mask * left_gate
       right_out = right * mask * right_gate
   (If no mask, the kernel is skipped and the product is done in PyTorch.)
5. Core reduction:
       out[b,i,j,h] = Σ_k left_out[b,i,k,h] * right_out[b,j,k,h]
   This is implemented as a batched matrix multiplication
   (B*H, N, N) @ (B*H, N, N)ᵀ using torch.bmm.
6. Apply a second LayerNorm over the hidden dimension H (FP32).
7. Multiply by the output gate and linearly project back to the original
   channel dimension C.
8. Cast the result back to the original dtype of the input tensor.

Only the mask‑and‑gate fusion is performed in Triton; all other
operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # const * left  [B,N,N,H] flattened (fp16)
    right_ptr,         # const * right [B,N,N,H] flattened (fp16)
    mask_ptr,          # const * mask  [B,N,N]   flattened (fp16)
    left_gate_ptr,     # const * left_gate  [B,N,N,H] flattened (fp16)
    right_gate_ptr,    # const * right_gate [B,N,N,H] flattened (fp16)
    left_out_ptr,      # * left_out  [B,N,N,H] flattened (fp16)
    right_out_ptr,     # * right_out [B,N,N,H] flattened (fp16)
    H: tl.constexpr,                # hidden dimension
    total_elements: tl.constexpr,   # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask and gates:
       left_out  = left  * mask * left_gate
       right_out = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elements

    # Load data
    l = tl.load(left_ptr + offs, mask=active)
    r = tl.load(right_ptr + offs, mask=active)
    lg = tl.load(left_gate_ptr + offs, mask=active)
    rg = tl.load(right_gate_ptr + offs, mask=active)

    # mask is duplicated across the hidden dimension; compute its index
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=active)

    # Apply mask and gates
    tl.store(left_out_ptr + offs, l * lg * m, mask=active)
    tl.store(right_out_ptr + offs, r * rg * m, mask=active)

    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:

            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict containing model parameters:
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with keys:
                "dim"         – input channel dimension C

    # ------------------------------------------------------------------

    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1) Input LayerNorm in FP32 (for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,

        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # 2) Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------

    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

        # Allocate output tensors
        left_out = torch.empty_like(left)
        right_out = torch.empty_like(right)

        total_elements = left.numel()  # B * N * N * H
        BLOCK_SIZE = 1024  # multiple of 32, fits in shared memory

        grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            mask_ptr=mask_flat,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            left_out_ptr=left_out,
            right_out_ptr=right_out,
            H=hidden_dim,
            total_elements=total_elements,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=4,
        )
    else:
        # No mask (or ignore it) – simple elementwise multiplication
        left_out = left * left_gate
        right_out = right * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction: batched GEMM (left_out @ right_outᵀ)
    #    Reshape to (B*H, N, N) for torch.bmm.
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_flat = left_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication (cuBLAS)
    out_flat = torch.bmm(left_flat, right_flat.transpose(-2, -1))   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 8) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)   # (B, N, N, H)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 9) Second LayerNorm over hidden dimension H (FP32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 10) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 12) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
