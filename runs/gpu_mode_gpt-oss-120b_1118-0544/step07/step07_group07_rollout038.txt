@triton.jit
def _mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems,
    hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

"""
custom_kernel: Triton‑accelerated outgoing TriMul (Triangle Multiplicative Update).

Algorithm
---------
1. LayerNorm on the input (float32) → x.
2. Cast x to float16 for the heavy compute.
3. Fuse each projection with its gate:
   * left = (x @ left_proj_w.T) * sigmoid(x @ left_gate_w.T)
   * right = (x @ right_proj_w.T) * sigmoid(x @ right_gate_w.T)
   The two linear layers per side are merged into a single matmul.
4. Apply the optional mask and the two gates in‑place with a small Triton kernel
   (supports both masked and unmasked paths).
5. Core contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication:
   left_perm  = left.permute(0,3,1,2)   # (B,H,N,N)
   right_perm = right.permute(0,3,2,1)  # (B,H,N,N) = rightᵀ on the last two axes
   out_perm   = torch.matmul(left_perm, right_perm)  # (B,H,N,N)
   out        = out_perm.permute(0,2,3,1)            # (B,N,N,H)
6. Apply LayerNorm on the hidden dimension, multiply by out_gate,
   and a final linear projection back to the original dimension.
7. Cast back to the input dtype and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr,          # *fp16  [B,N,N,H] flattened
    right_ptr,         # *fp16  same shape
    left_gate_ptr,     # *fp16  same shape
    right_gate_ptr,    # *fp16  same shape
    mask_ptr,          # *fp16  [B,N,N] flattened
    total_elems,       # i32
    hidden_dim,        # i32
    HAS_MASK: tl.constexpr,   # compile‑time bool
    BLOCK_SIZE: tl.constexpr, # compile‑time int
):
    """In‑place fusion:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
       If HAS_MASK=False the mask term is omitted.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each mask entry repeats `hidden_dim` times in the flattened view
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # store back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) (may be dummy)
            - weights: dict of parameter tensors (float32)
            - config: dict with keys
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections + gates (fused two matmuls per side)
    # ------------------------------------------------------------------
    # concatenate proj & gate weights for the left side
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*H, C)

    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, C)

    # fused linear -> (B,N,N,2*H)
    left_concat = F.linear(x, left_concat_w)   # (B,N,N,2*H)
    right_concat = F.linear(x, right_concat_w) # (B,N,N,2*H)

    # split proj and gate
    left = left_concat[..., :hidden_dim]                         # (B,N,N,H)
    left_gate = torch.sigmoid(left_concat[..., hidden_dim:])      # (B,N,N,H)

    right = right_concat[..., :hidden_dim]                       # (B,N,N,H)
    right_gate = torch.sigmoid(right_concat[..., hidden_dim:])    # (B,N,N,H)

    # out gate (separate linear)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4. Fuse mask and gates (in‑place) via Triton
    # ------------------------------------------------------------------
    # Prepare flattened views
    left_flat = left.view(-1)
    right_flat = right.view(-1)
    left_gate_flat = left_gate.view(-1)
    right_gate_flat = right_gate.view(-1)

    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        mask_flat = mask_fp.view(-1)  # (B*N*N,)
        has_mask = True
    else:
        # dummy mask (won't be read)
        mask_flat = torch.empty((1,), dtype=torch.float16, device=device)
        has_mask = False

    total = left_flat.numel()
    BLOCK = 1024  # reasonable block size

    grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

    _gate_mask_kernel[grid](
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        total,
        hidden_dim,
        HAS_MASK=has_mask,
        BLOCK_SIZE=BLOCK,
    )

    # reshape back (views, no copy)
    left = left_flat.view_as(left)
    right = right_flat.view_as(right)

    # ------------------------------------------------------------------
    # 5. Core contraction via batched matmul (left @ rightᵀ)
    # ------------------------------------------------------------------
    # permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1) # (B, H, N, N) = rightᵀ on last two axes
    out_perm = torch.matmul(left_perm, right_perm)  # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------
    # 7. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
