@triton.jit
def proj_gate_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load values
    left_proj = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    # Compute mask index
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # if no mask than m=1 in else; but we only call when mask present
    # Compute fused
    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_ah, stride_am, stride_ak,
    stride_bh, stride_bk, stride_bn,
    stride_ch, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile col
    pid_b = tl.program_id(2)  # batch

    # Compute offsets
    block_start_m = pid_m * BLOCK_M
    block_start_n = pid_n * BLOCK_N

    # Create range offsets inside block
    offs_m = block_start_m + tl.arange(0, BLOCK_M)
    offs_n = block_start_n + tl.arange(0, BLOCK_N)
    # mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        # compute offsets for k block
        block_start_k = k * BLOCK_K
        offs_k = block_start_k + tl.arange(0, BLOCK_K)

        # Mask for k
        mask_k = offs_k < K

        # Load A block (M x K)
        a_ptrs = A_ptr + pid_b * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        # Load B block (K x N)
        b_ptrs = B_ptr + pid_b * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product accumulation
        acc += tl.dot(a, b)

    # Write result
    c_ptrs = C_ptr + pid_b * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    # Cast to desired dtype (float16)
    acc_fp16 = acc.to(tl.float16)
    tl.store(c_ptrs, acc_fp16, mask=mask_m[:, None] & mask_n[None, :])

def _batched_matmul(A: torch.Tensor, B: torch.Tensor):
    # A shape (batch, M, K) fp16
    # B shape (batch, K, N) fp16
    # output shape (batch, M, N) fp16

    batch, M, K = A.shape
    _, _, N = B.shape
    out = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # compute strides
    stride_ah = A.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)

    stride_bh = B.stride(0)
    stride_bk = B.stride(1)
    stride_bn = B.stride(2)

    stride_ch = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    # choose block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N), batch)

    )

def custom_kernel(data):
   input_tensor, mask, weights, config = data
   dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)
   device = input_tensor.device
   dtype_in = input_tensor.dtype

   # Step1: layernorm
   x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

   # Cast to fp16
   x = x.to(torch.float16)

   # Cast weights
   left_proj_w = weights["left_proj.weight"].to(torch.float16)
   right_proj_w = weights["right_proj.weight"].to(torch.float16)
   left_gate_w = weights["left_gate.weight"].to(torch.float16)
   right_gate_w = weights["right_gate.weight"].to(torch.float16)
   out_gate_w = weights["out_gate.weight"].to(torch.float16)
   to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
   to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
   to_out_w = weights["to_out.weight"].to(torch.float16)

   # Linear projections
   left_proj = F.linear(x, left_proj_w)  # (B,N,N,H)
   right_proj = F.linear(x, right_proj_w)

   # Gates
   left_gate = F.linear(x, left_gate_w).sigmoid()
   right_gate = F.linear(x, right_gate_w).sigmoid()
   out_gate = F.linear(x, out_gate_w).sigmoid()

   # Apply mask+gate
   if mask is not None and not nomask:
       mask_fp16 = mask.to(torch.float16).contiguous()
       left, right = _apply_proj_gate_mask(left_proj, right_proj, left_gate, right_gate, mask_fp16)
   else:
       left = left_proj * left_gate
       right = right_proj * right_gate
       if mask is not None and not nomask:
           mask_fp16 = mask.to(torch.float16).contiguous()
           left = left * mask_fp16[..., None]
           right = right * mask_fp16[..., None]

   # Prepare for matmul
   B, N, _, H = left.shape
   # Permute to (B, H, N, N)
   left_perm = left.permute(0, 3, 1, 2).contiguous()
   right_perm = right.permute(0, 3, 1, 2).contiguous()
   # Transpose RHS matrices
   right_T = right_perm.transpose(2, 3).contiguous()

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast the normalized tensor and all linear weights to fp16.
3. Compute left/right projections and left/right/out gates (sigmoid).
4. Fuse mask, projection and gate with a tiny Triton kernel
   (only executed when a mask is present and `nomask==False`).
5. Perform the core reduction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication `left @ rightᵀ`.
   The multiplication is implemented with a custom Triton GEMM kernel
   (batch dimension = B·hidden_dim).
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   out‑gate and project back to the original `dim`.
7. Cast the result back to the input dtype (usually fp32).

Both the mask‑gate fusion and the batched GEMM are written in Triton,
while the remaining tensor‑wise ops use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B, N, N, H)
    right_proj_ptr,  # fp16 * (B, N, N, H)
    left_gate_ptr,   # fp16 * (B, N, N, H)
    right_gate_ptr,  # fp16 * (B, N, N, H)
    mask_ptr,        # fp16 * (B, N, N)
    left_out_ptr,    # fp16 * (B, N, N, H)
    right_out_ptr,   # fp16 * (B, N, N, H)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # flattened index of the mask (B,N,N) = element index // H
    mask_idx = offs // hidden_dim

    # load
    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # fuse
    left_out  = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs,  left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuses projection, gate and mask for the left/right tensors.
    All tensors must be fp16, contiguous and have shapes:
        (B, N, N, H) for the 4‑D tensors and (B, N, N) for the mask.
    Returns:
        left, right : (B, N, N, H)
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out  = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for the TriMul core reduction
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_ah, stride_am, stride_ak,   # A: (batch, M, K)
    stride_bh, stride_bk, stride_bn,   # B: (batch, K, N)
    stride_ch, stride_cm, stride_cn,   # C: (batch, M, N)
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_b = tl.program_id(2)   # batch index

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A block: (M, K)
        a_ptrs = A_ptr + pid_b * stride_ah + \
                 offs_m[:, None] * stride_am + \
                 offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B block: (K, N)
        b_ptrs = B_ptr + pid_b * stride_bh + \
                 offs_k[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # store result back, cast to fp16
    c_ptrs = C_ptr + pid_b * stride_ch + \
             offs_m[:, None] * stride_cm + \
             offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def _batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B where
        A: (B, M, K)
        B: (B, K, N)
    Returns C: (B, M, N) in the same dtype as A/B (fp16).
    """
    batch, M, K = A.shape
    _, _, N = B.shape

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # strides
    stride_ah = A.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)

    stride_bh = B.stride(0)
    stride_bk = B.stride(1)
    stride_bn = B.stride(2)

    stride_ch = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    # tile size – tuned for typical NVIDIA GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )

    _batched_gemm_kernel[grid](
        A, B, C,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_cm, stride_cn,
        M, N, K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None (0/1)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1) Input LayerNorm over channel dimension
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy ops
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast weights to fp16 (bias‑free linear layers)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections (fp16)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)   # (B, N, N, H)

    left_gate  = F.linear(x, left_gate_w).sigmoid()   # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid()  # (B, N, N, H)
    out_gate   = F.linear(x, out_gate_w).sigmoid()    # (B, N, N, H)

    # --------------------------------------------------------------
    # 5) Fuse mask, projection and gate (use Triton when mask present)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure fp16, contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16,
            block_size=1024,
        )
    else:
        # simple element‑wise fusion (mask ignored)
        left  = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16).contiguous()
            left  = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # --------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM (A @ Bᵀ) with a Triton kernel.
    # --------------------------------------------------------------
    B, N, _, H = left.shape

    # Rearrange to (B, H, N, N) and make contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Transpose the RHS matrices
    right_T = right_perm.transpose(2, 3).contiguous()    # (B, H, N, N)

    # Collapse batch & hidden dimensions
    left_flat  = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_T.view(B * H, N, N)    # (B*H, N, N)

    # Batched matrix multiplication via Triton
    out_flat = _batched_gemm(left_flat, right_flat)   # (B*H, N, N), fp16

    # Restore original shape (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 8) Cast back to the original dtype
    # --------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
