"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

The implementation follows the reference PyTorch module:
1. Layer‑norm the input (float32 for stability).
2. Cast to fp16 for the compute‑heavy path.
3. Linear projections (left / right) and three sigmoid gates
   (left, right, out) – all in fp16.
4. Fuse mask (if present) together with the left/right gates using a small
   Triton kernel that works in‑place on the flattened tensors.
5. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched GEMM (left @ rightᵀ) for each hidden channel.
6. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate, and a final linear projection back to the original dimension.
7. Cast the result to float32 and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,   # fp16 * (B*N*N*H)
    right_ptr,  # fp16 *
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,   # fp16 * (B*N*N)
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place element‑wise fusion:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
       If HAS_MASK is False the mask is ignored (treated as 1).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden‑dim element shares the same mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C) (float32)
            - mask: torch.Tensor of shape (B, N, N) or None (float32)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with dtype torch.float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for the compute‑heavy part
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gates (all in fp16)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w) # (B,N,N,H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,H)

    # -----------------------------------------------------------------
    # 4. Fuse mask and gates using Triton (in‑place)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is fp16 and contiguous
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten tensors for the kernel
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # no mask → simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core O(N³·H) contraction via batched GEMM (L @ Rᵀ)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched matmul
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) = (B*H, N, N) @ (B*H, N, N).T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # (B,N,N,dim)

    # -----------------------------------------------------------------
    # 7. Cast back to float32 and return
    # -----------------------------------------------------------------
    out = out.to(torch.float32)
    return out
