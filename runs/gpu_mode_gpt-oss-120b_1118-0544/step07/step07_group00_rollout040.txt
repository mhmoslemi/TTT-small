@triton.jit
def _tri_mul_kernel(
    left_proj_ptr, left_gate_ptr, right_proj_ptr, right_gate_ptr,
    mask_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_left_gate_b, stride_left_gate_i, stride_left_gate_k, stride_left_gate_h,
    stride_right_proj_b, stride_right_i, stride_right_k, stride_right_h,
    stride_right_gate_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # Compute number of blocks per dimension
    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)

    # Decode pid to b, h, block_i, block_j
    # pid = b * H * num_blocks_i * num_blocks_j + h * num_blocks_i * num_blocks_j + block_i * num_blocks_j + block_j
    tmp = pid
    b = tmp // (H * num_blocks_i * num_blocks_j)
    tmp = tmp % (H * num_blocks_i * num_blocks_j)
    h = tmp // (num_blocks_i * num_blocks_j)
    tmp = tmp % (num_blocks_i * num_blocks_j)
    block_i = tmp // num_blocks_j
    block_j = tmp % num_blocks_j

    # Compute i and j indices
    i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out-of-bounds
    mask_i = i < N
    mask_j = j < N

    # base pointers for this batch and head
    left_base = left_proj_ptr + b * stride_left_b + h * stride_left_h
    left_gate_base = left_gate_ptr + b * stride_left_gate_b + h * stride_left_gate_h
    right_base = right_proj_ptr + b * stride_right_proj_b + h * stride_right_h
    right_gate_base = right_gate_ptr + b * stride_right_gate_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h
    mask_base = mask_ptr + b * stride_mask_b

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        ks = k_offset + tl.arange(0, BLOCK_K)
        mask_k = ks < N

        # Left tile
        left_ptrs = left_base + i[:, None] * stride_left_i + ks[None, :] * stride_left_k
        left_gate_ptrs = left_gate_base + i[:, None] * stride_left_gate_i + ks[None, :] * stride_left_gate_k
        mask_left_ptrs = mask_base + i[:, None] * stride_mask_i + ks[None, :] * stride_mask_k

        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        left_gate = tl.load(left_gate_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        mask_left = tl.load(mask_left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=1.0)

        left_tile = left * left_gate * mask_left

        # Right tile
        right_ptrs = right_base + j[:, None] * stride_right_i + ks[None, :] * stride_right_k
        right_gate_ptrs = right_gate_base + j[:, None] * stride_right_i + ks[None, :] * stride_right_k
        mask_right_ptrs = mask_base + j[:, None] * stride_mask_i + ks[None, :] * stride_mask_k

        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        right_gate = tl.load(right_gate_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        mask_right = tl.load(mask_right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=1.0)

        right_tile = right * right_gate * mask_right

        # Accumulate dot product
        acc += tl.dot(left_tile, right_tile)

    # Store result
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    store_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=store_mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # unpack dim and hidden_dim
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # Layer norm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for compute
    x = x.to(torch.float16)

    # Convert weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections (no bias)
    left_proj = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    B, N, _, _ = left_proj.shape  # dim original not needed

    # mask handling
    if mask is None or nomask:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device, contiguous=True)
    else:
        mask_fp16 = mask.to(torch.float16).contiguous()

    # allocate output tensor
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # launch kernel
    _tri_mul_kernel_launcher(
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask_fp16,
        out,
        BLOCK_M=64,
        BLOCK_N=64,
        BLOCK_K=64,
    )

    # Apply output layer norm, out_gate, final linear
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)

    # Cast back to original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

"""
TriMul (outgoing) forward pass implementation with Triton.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for compute‑intensive parts.
3. Linear projections (no bias) → left_proj, right_proj.
4. Gating projections → left_gate, right_gate, out_gate (sigmoid).
5. Fuse mask, projection and gating inside a custom Triton kernel that
   also performs the core reduction:
        out[b,i,j,h] = Σ_k left_proj[b,i,k,h] * left_gate[b,i,k,h] *
                       right_proj[b,j,k,h] * right_gate[b,j,k,h] *
                       mask[b,i,k] * mask[b,j,k]
   This is a batched matrix multiplication for every hidden slice `h`.
6. Apply a second LayerNorm over the hidden dimension, multiply by
   `out_gate` and a final linear projection back to `dim`.
7. Cast the result back to the original dtype (usually FP32).

The Triton kernel processes a tile (BLOCK_M × BLOCK_N) of the output
for a particular batch and hidden slice, using FP16 loads and FP32
accumulation for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    # pointers
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr, out_ptr,
    # scalar arguments
    B, N, H,
    # strides (int64)
    stride_lpb, stride_lpi, stride_lpk, stride_lph,
    stride_lgb, stride_lgi, stride_lgk, stride_lgh,
    stride_rpb, stride_rpi, stride_rpk, stride_rph,
    stride_rgb, stride_rgi, stride_rgk, stride_rgh,
    stride_mpb, stride_mpi, stride_mpk,
    stride_ob, stride_oi, stride_oj, stride_oh,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of blocks in each output dimension
    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)

    # decode pid → (batch, head, block_i, block_j)
    tmp = pid
    b = tmp // (H * num_blocks_i * num_blocks_j)
    tmp = tmp % (H * num_blocks_i * num_blocks_j)
    h = tmp // (num_blocks_i * num_blocks_j)
    tmp = tmp % (num_blocks_i * num_blocks_j)
    block_i = tmp // num_blocks_j
    block_j = tmp % num_blocks_j

    # global indices of the output tile
    i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # base pointers for this batch/head
    left_base   = left_proj_ptr + b * stride_lpb + h * stride_lph
    leftg_base  = left_gate_ptr + b * stride_lgb + h * stride_lgh
    right_base  = right_proj_ptr + b * stride_rpb + h * stride_rph
    rightg_base = right_gate_ptr + b * stride_rgb + h * stride_rgh
    out_base    = out_ptr + b * stride_ob + h * stride_oh
    mask_base   = mask_ptr + b * stride_mpb

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K
        ks = k_off + tl.arange(0, BLOCK_K)
        mask_k = ks < N

        # ----- left side -----
        left_ptrs = left_base + i[:, None] * stride_lpi + ks[None, :] * stride_lpk
        leftg_ptrs = leftg_base + i[:, None] * stride_lgi + ks[None, :] * stride_lgk
        maskl_ptrs = mask_base + i[:, None] * stride_mpi + ks[None, :] * stride_mpk

        left   = tl.load(left_ptrs,  mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        leftg  = tl.load(leftg_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        maskl  = tl.load(maskl_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=1.0)

        left_tile = left * leftg * maskl    # (BLOCK_M, BLOCK_K)

        # ----- right side -----
        right_ptrs = right_base + j[:, None] * stride_rpi + ks[None, :] * stride_rpk
        rightg_ptrs = rightg_base + j[:, None] * stride_rgi + ks[None, :] * stride_rgk
        maskr_ptrs = mask_base + j[:, None] * stride_mpi + ks[None, :] * stride_mpk

        right   = tl.load(right_ptrs,  mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        rightg  = tl.load(rightg_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        maskr   = tl.load(maskr_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=1.0)

        right_tile = right * rightg * maskr   # (BLOCK_K, BLOCK_N)

        # accumulate dot product over k
        acc += tl.dot(left_tile, right_tile)

    # store the result (convert back to FP16)
    out_ptrs = out_base + i[:, None] * stride_oi + j[None, :] * stride_oj
    store_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=store_mask)


def _tri_mul_kernel_launcher(
    left_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_proj: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    out: torch.Tensor,
    BLOCK_M: int = 64,
    BLOCK_N: int = 64,
    BLOCK_K: int = 64,
):
    """
    Launches the Triton kernel that fuses mask, gating and the
    triangular multiplicative reduction.
    """
    B, N, _, H = left_proj.shape

    # strides (in elements)
    stride_lpb, stride_lpi, stride_lpk, stride_lph = left_proj.stride()
    stride_lgb, stride_lgi, stride_lgk, stride_lgh = left_gate.stride()
    stride_rpb, stride_rpi, stride_rpk, stride_rph = right_proj.stride()
    stride_rgb, stride_rgi, stride_rgk, stride_rgh = right_gate.stride()
    stride_mpb, stride_mpi, stride_mpk = mask.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    num_blocks_i = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_j = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * H * num_blocks_i * num_blocks_j

    grid = (total_programs,)

    _tri_mul_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask,
        out,
        B, N, H,
        stride_lpb, stride_lpi, stride_lpk, stride_lph,
        stride_lgb, stride_lgi, stride_lgk, stride_lgh,
        stride_rpb, stride_rpi, stride_rpk, stride_rph,
        stride_rgb, stride_rgi, stride_rgk, stride_rgh,
        stride_mpb, stride_mpi, stride_mpk,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32 or float16)
        - mask         : torch.Tensor of shape [B, N, N] (0/1) or None
        - weights      : dict with keys matching the PyTorch module parameters
        - config       : dict with at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast model weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16, no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    B, N, _, _ = left_proj.shape

    # ------------------------------------------------------------------
    # 5) Prepare mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if mask is None or nomask:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device, contiguous=True)
    else:
        mask_fp16 = mask.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction (fused with mask & gating) via Triton
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)
    _tri_mul_kernel_launcher(
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask_fp16,
        out,
        BLOCK_M=64,
        BLOCK_N=64,
        BLOCK_K=64,
    )  # out : (B, N, N, hidden_dim) in fp16

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
