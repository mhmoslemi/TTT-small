@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B_H, N,
    stride_l_bh, stride_l_i, stride_l_k,
    stride_r_bh, stride_r_j, stride_r_k,
    stride_o_bh, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    bh = tl.program_id(0)  # batch*hidden
    pid = tl.program_id(1) # tile id for i/j
    # compute tile coordinates
    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)
    tile_i = pid // num_blocks_j
    tile_j = pid % num_blocks_j

    # Offsets
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for OOB
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left tile: [M, K]
        left = tl.load(
            left_ptr + bh * stride_l_bh + offs_i[:, None] * stride_l_i + k_offset[None, :] * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load right tile transposed: we want shape [K, N]
        right = tl.load(
            right_ptr + bh * stride_r_bh + k_offset[:, None] * stride_r_k + offs_j[None, :] * stride_r_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )
        # Convert to fp32 for accumulation
        left = left.to(tl.float32)
        right = right.to(tl.float32)
        acc += tl.dot(left, right)  # (M,N)

    # Write back
    out_ptrs = out_ptr + bh * stride_o_bh + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def triton_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    # left, right shape (B, N, N, H)
    # Permute to (B, H, N, N) then view as (BH, N, N)
    B, N, _, H = left.shape
    left_flat = left.permute(0, 3, 1, 2).contiguous()
    right_flat = right.permute(0, 3, 1, 2).contiguous()
    BH = B * H
    left_flat = left_flat.view(BH, N, N)
    right_flat = right_flat.view(BH, N, N)
    out_flat = torch.empty_like(left_flat)

    # Compute strides
    stride_l_bh, stride_l_i, stride_l_k = left_flat.stride()
    stride_r_bh, stride_r_j, stride_r_k = right_flat.stride()
    stride_o_bh, stride_o_i, stride_o_j = out_flat.stride()

    # grid: (BH, num_tiles_i * num_tiles_j)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (BH, num_i * num_j)

    tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        N,
        stride_l_bh, stride_l_i, stride_l_k,
        stride_r_bh, stride_r_j, stride_r_k,
        stride_o_bh, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

   This is a batched matrix multiplication (A @ Bᵀ) for each hidden slice.
   Implemented with a custom Triton GEMM kernel that works on the
   (B·H, N, N) tensors.
6. Output LayerNorm, multiply by `out_gate`, final linear projection back to
   the original dimension.
7. Cast back to the original input dtype (usually float32).

Only the mask‑gate‑fusion and the O(N³) reduction are performed in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask (left & right)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B,N,N,H)
    right_proj_ptr,  # fp16 * (B,N,N,H)
    left_gate_ptr,   # fp16 * (B,N,N,H)
    right_gate_ptr,  # fp16 * (B,N,N,H)
    mask_ptr,        # fp16 * (B,N,N)
    left_out_ptr,    # fp16 * (B,N,N,H)
    right_out_ptr,   # fp16 * (B,N,N,H)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # OOB guard
    mask = offs < total_elems

    # mask is shared across hidden dim
    mask_idx = offs // hidden_dim

    left_proj = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gate and mask for both tensors.
    All tensors must be fp16, contiguous and have shapes:
        (B, N, N, H) for the four first arguments,
        (B, N, N) for mask.
    Returns:
        left, right : (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr,      # fp16 * (BH, N, N)   (i.e., B*H, i, k)
    right_ptr,     # fp16 * (BH, N, N)   (i.e., B*H, j, k)
    out_ptr,       # fp16 * (BH, N, N)   (i.e., B*H, i, j)
    BH, N,
    stride_l_bh, stride_l_i, stride_l_k,
    stride_r_bh, stride_r_j, stride_r_k,
    stride_o_bh, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    bh = tl.program_id(0)          # 0 … BH-1
    pid = tl.program_id(1)         # tile index over (i, j)

    # number of tiles in each dimension
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)

    tile_i = pid // num_tiles_j
    tile_j = pid % num_tiles_j

    # offsets within the tile
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M x K)
        left = tl.load(
            left_ptr
            + bh * stride_l_bh
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load right tile transposed (K x N)
        right = tl.load(
            right_ptr
            + bh * stride_r_bh
            + offs_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # Convert to float32 for accumulation
        acc += tl.dot(left.to(tl.float32), right.to(tl.float32))

    # Write back
    out_ptrs = (
        out_ptr
        + bh * stride_o_bh
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the core TriMul reduction using Triton.
    `left` / `right` : (B, N, N, H) fp16
    Returns:
        out : (B, N, N, H) fp16
    """
    B, N, _, H = left.shape
    # (B, H, N, N) → (BH, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)

    # strides (in elements)
    s_l_bh, s_l_i, s_l_k = left_flat.stride()
    s_r_bh, s_r_j, s_r_k = right_flat.stride()
    s_o_bh, s_o_i, s_o_j = out_flat.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_tiles_i = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_j = (N + BLOCK_N - 1) // BLOCK_N

    grid = (B * H, num_tiles_i * num_tiles_j)

    _tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        s_l_bh,
        s_l_i,
        s_l_k,
        s_r_bh,
        s_r_j,
        s_r_k,
        s_o_bh,
        s_o_i,
        s_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (fused in Triton when mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is contiguous FP16
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # Simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction via Triton GEMM
    # ------------------------------------------------------------------
    out = _tri_mul_triton(left, right)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
