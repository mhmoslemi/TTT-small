def batched_gemm_kernel(a_ptr, b_ptr, c_ptr,
                        batch, M, N, K,
                        stride_a_batch, stride_a_row, stride_a_col,
                        stride_b_batch, stride_b_row, stride_b_col,
                        stride_c_batch, stride_c_row, stride_c_col,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
                        NUM_STAGES: tl.constexpr, NUM_WARPS: tl.constexpr):

   pid_batch = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   # compute block offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

   # initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   for k in range(0, tl.cdiv(K, BLOCK_K)):
       k_base = k * BLOCK_K
       offs_k = k_base + tl.arange(0, BLOCK_K)

       # mask for out-of-bounds in each dimension
       mask_a = (offs_m < M)[:, None] & (offs_k < K)[None, :]
       mask_b = (offs_k < K)[:, None] & (offs_n < N)[None, :]

       a_ptrs = a_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
       b_ptrs = b_ptr + pid_batch * stride_b_batch + offs_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col

       a = tl.load(a_ptrs, mask=mask_a, other=0.0)
       b = tl.load(b_ptrs, mask=mask_b, other=0.0)

       acc += tl.dot(a, b)

   # cast to output dtype (fp16)
   c = acc.to(tl.float16)

   # store
   c_ptrs = c_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
   mask_c = (offs_m < M)[:, None] & (offs_n < N)[None, :]
   tl.store(c_ptrs, c, mask=mask_c)

"""
custom_kernel: Triton‑accelerated outgoing TriMul (Triangle Multiplicative Update).

Algorithm
---------
1. LayerNorm on the input (float32) -> x (B,N,N,C).
2. Cast x to float16 for heavy compute.
3. Linear projections (bias‑less) → left, right, left_gate, right_gate, out_gate.
   Gates are passed through sigmoid.  Mask (if present) is broadcast and
   multiplied together with the gates.
4. Core contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication left @ rightᵀ for each hidden
   dimension h.  Implemented with a custom Triton GEMM kernel that
   processes all (B·H) batches in one launch.
5. LayerNorm on the hidden dimension, element‑wise multiply by out_gate.
6. Final linear projection back to the original dimension.
7. Cast back to the input dtype.

The kernel in step 4 uses FP16 inputs, FP32 accumulation and stores the
result in FP16.  Mask/gate fusion is performed with regular PyTorch
operations (cheap compared to the GEMM).  At least one operation is
implemented in Triton, satisfying the requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers to left, right, output
    batch, M, N, K,               # dimensions: batch = B*H, M=N, N=N, K=N
    stride_aa, stride_ai, stride_aj,  # strides for a: (batch, row, col)
    stride_bb, stride_bi, stride_bj,  # strides for b: (batch, row, col) – we read b as (K,N)
    stride_cc, stride_ci, stride_cj,  # strides for c: (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_STAGES: tl.constexpr,
    NUM_WARPS: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  where A,B shape = (batch, N, N)."""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute the row / col offsets of the block.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator in FP32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads.
        mask_a = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        mask_b = (offs_k < K)[:, None] & (offs_n < N)[None, :]

        # Pointers for A (row i, col k) and B (row k, col j).
        a_ptrs = a_ptr + pid_batch * stride_aa \
                 + offs_m[:, None] * stride_ai \
                 + offs_k[None, :] * stride_aj
        b_ptrs = b_ptr + pid_batch * stride_bb \
                 + offs_k[:, None] * stride_bi \
                 + offs_n[None, :] * stride_bj

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)

    # Cast accumulator to output dtype (FP16) and store.
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_cc \
             + offs_m[:, None] * stride_ci \
             + offs_n[None, :] * stride_cj
    mask_c = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, c, mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑intensive ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (bias‑less) and gates (sigmoid)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    # Projections
    left = F.linear(x, left_proj_w)          # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)        # (B,N,N,hidden_dim)

    # Gates
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4. Apply mask (if any) and gate multiplication
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape: (B,N,N); broadcast on hidden dim
        mask = mask.to(device).to(torch.float16)
        mask = mask.unsqueeze(-1)               # (B,N,N,1)
        left = left * left_gate * mask
        right = right * right_gate * mask
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: batched GEMM left @ rightᵀ using Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Merge batch and hidden dimensions for the batched GEMM.
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Allocate output buffer
    out_mat = torch.empty_like(left_mat)

    # Strides (in elements) for the three tensors
    stride_aa = left_mat.stride(0)
    stride_ai = left_mat.stride(1)
    stride_aj = left_mat.stride(2)

    stride_bb = right_mat.stride(0)
    stride_bi = right_mat.stride(1)
    stride_bj = right_mat.stride(2)

    stride_cc = out_mat.stride(0)
    stride_ci = out_mat.stride(1)
    stride_cj = out_mat.stride(2)

    # Choose block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                              # batch dimension
        triton.cdiv(N, BLOCK_M),            # output rows
        triton.cdiv(N, BLOCK_N),            # output cols
    )

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        B * H,
        N, N, N,
        stride_aa, stride_ai, stride_aj,
        stride_bb, stride_bi, stride_bj,
        stride_cc, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        NUM_STAGES=3,
        NUM_WARPS=4,
    )

    # Reshape back to (B,N,N,hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # (B,N,N,dim) in FP16

    # ------------------------------------------------------------------
    # 7. Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
