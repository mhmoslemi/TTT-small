"""
TriMul (outgoing) – Triton‑accelerated forward pass
---------------------------------------------------
1. LayerNorm the input (float32 for numerical stability).
2. Linear projections to a hidden dimension `H` (float16).
3. Compute three sigmoid gates (float16).
4. Fuse mask, left/right gates and the core O(N³) reduction
   in a single Triton kernel:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * mask[b,i,k] * left_gate[b,i,k,h] *
                      right[b,k,j,h] * mask[b,k,j] * right_gate[b,k,j,h]
5. Reshape the kernel output to [B, N, N, H].
6. Apply a second LayerNorm over the hidden dimension,
   multiply by the output gate and project back to the original
   channel dimension `C`.
All heavy arithmetic is performed in float16 inside the kernel,
while the normalization steps stay in float32 for stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fused mask, gates and batched matrix multiplication
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # const * left   [BH, N, N]   (float16)
    right_ptr,         # const * right  [BH, N, N]   (float16) – already transposed (k,j)
    mask_ptr,          # const * mask   [B, N, N]    (float16)
    left_gate_ptr,     # const * left_gate  [BH, N, N] (float16)
    right_gate_ptr,    # const * right_gate [BH, N, N] (float16)
    out_ptr,           # * out   [BH, N, N]   (float16)
    B: tl.constexpr,   # batch size
    N: tl.constexpr,   # sequence length
    H: tl.constexpr,   # hidden dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute the outgoing Triangle Multiplicative Update."""
    pid_z = tl.program_id(0)          # combined batch * hidden index
    pid_m = tl.program_id(1)          # i‑block
    pid_n = tl.program_id(2)          # j‑block

    # ------------------------------------------------------------------
    # Compute absolute indices
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j
    k_base = 0                                          # will be updated each iteration

    # batch‑hidden index and original batch index
    bh_idx = pid_z                     # 0 .. B*H-1
    b_idx  = bh_idx // H               # batch index in [0, B)

    # strides for the (BH, N, N) layout (contiguous)
    stride_batch = N * N
    stride_row   = N
    stride_col   = 1

    # base offsets for this batch‑hidden slice
    batch_off = bh_idx * stride_batch

    # mask base offset (same for left and right, no hidden dim)
    mask_batch_off = b_idx * stride_batch

    # ------------------------------------------------------------------
    # Accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for _ in range(num_k_blocks):
        # current k indices
        offs_k = tl.arange(0, BLOCK_K) + k_base

        # ------------------------------------------------------------------
        # Bounds masks
        # ------------------------------------------------------------------
        mask_a = (offs_m[:, None] < N) & (offs_k[None, :] < N)   # left tile
        mask_b = (offs_k[:, None] < N) & (offs_n[None, :] < N)   # right tile
        mask_out = (offs_m[:, None] < N) & (offs_n[None, :] < N)

        # ------------------------------------------------------------------
        # Load left tile and apply gate & mask
        # ------------------------------------------------------------------
        a_ptr = left_ptr + (batch_off + offs_m[:, None] * stride_row + offs_k[None, :] * stride_col)
        a = tl.load(a_ptr, mask=mask_a, other=0.0)

        a_gate_ptr = left_gate_ptr + (batch_off + offs_m[:, None] * stride_row + offs_k[None, :] * stride_col)
        a_gate = tl.load(a_gate_ptr, mask=mask_a, other=0.0)

        mask_a_ptr = mask_ptr + (mask_batch_off + offs_m[:, None] * stride_row + offs_k[None, :] * stride_col)
        m_a = tl.load(mask_a_ptr, mask=mask_a, other=0.0)

        a = a * a_gate * m_a

        # ------------------------------------------------------------------
        # Load right tile (already transposed to (k, j)) and apply gate & mask
        # ------------------------------------------------------------------
        b_ptr = right_ptr + (batch_off + offs_k[:, None] * stride_row + offs_n[None, :] * stride_col)
        b = tl.load(b_ptr, mask=mask_b, other=0.0)

        b_gate_ptr = right_gate_ptr + (batch_off + offs_k[:, None] * stride_row + offs_n[None, :] * stride_col)
        b_gate = tl.load(b_gate_ptr, mask=mask_b, other=0.0)

        mask_b_ptr = mask_ptr + (mask_batch_off + offs_k[:, None] * stride_row + offs_n[None, :] * stride_col)
        m_b = tl.load(mask_b_ptr, mask=mask_b, other=0.0)

        b = b * b_gate * m_b

        # ------------------------------------------------------------------
        # Accumulate
        # ------------------------------------------------------------------
        acc += tl.dot(a, b)

        # ------------------------------------------------------------------
        # Advance k
        # ------------------------------------------------------------------
        k_base += BLOCK_K

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + (batch_off + offs_m[:, None] * stride_row + offs_n[None, :] * stride_col)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, C]  (float32 or float16)
            - mask:         [B, N, N] (binary) or None
            - weights: dict with all model parameters
            - config:  {"dim": C, "hidden_dim": H, "nomask": bool (optional)}

    Returns:
        Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float32)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w) # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Prepare mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is None or nomask:
        mask_fp = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp = mask.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 6) Permute tensors to (B, H, N, N) → flatten to (BH, N, N)
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()       # (B, H, N, N) – transposed k‑j
    left_gate_perm = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate_perm = right_gate.permute(0, 3, 2, 1).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)
    left_gate_flat = left_gate_perm.view(BH, N, N)
    right_gate_flat = right_gate_perm.view(BH, N, N)

    # ------------------------------------------------------------------
    # 7) Allocate output buffer (BH, N, N)
    # ------------------------------------------------------------------
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # 8) Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (BH, tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N))
    _tri_mul_kernel[grid](
        left_ptr=left_flat,
        right_ptr=right_flat,
        mask_ptr=mask_fp.view(-1),
        left_gate_ptr=left_gate_flat,
        right_gate_ptr=right_gate_flat,
        out_ptr=out_flat,
        B=B,
        N=N,
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 9) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)            # (B, N, N, H)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 10) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 11) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 12) Final projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 13) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
