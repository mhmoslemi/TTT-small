2. Cast to FP16 for compute‑heavy paths.
3. Fuse the five linear projections (left/right and three gates) into a
   single matmul by concatenating their weight matrices.
4. Split the result, apply sigmoid to the three gate tensors.
5. If a mask is present and not ignored, fuse “mask * projection * gate”
   for the left/right tensors with a tiny Triton kernel.
   (When no mask is required we simply multiply by the gates.)
6. Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication
        left_h @ right_hᵀ   (h runs over the hidden dimension).
   The batch dimension is B·H, so a single `torch.bmm` call does the whole
   reduction.
7. Apply a second LayerNorm over the hidden dimension, multiply by the
   output‑gate and project back to the original channel dimension.
8. Cast the result back to the input dtype (normally float32).

Only the mask‑gate‑fusion step is written in Triton; all other heavy
operations use PyTorch (which already calls highly‑optimized cuBLAS/cuDNN
kernels).  This balance keeps the implementation simple while still
providing a measurable speed‑up over a naïve PyTorch‑only version.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B, N, N, H)
    right_proj_ptr,  # fp16 * (B, N, N, H)
    left_gate_ptr,   # fp16 * (B, N, N, H)
    right_gate_ptr,  # fp16 * (B, N, N, H)
    mask_ptr,        # fp16 * (B, N, N)
    left_out_ptr,    # fp16 * (B, N, N, H)
    right_out_ptr,   # fp16 * (B, N, N, H)
    total_elems,     # int64 scalar = B*N*N*H
    hidden_dim,      # int64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    msk = offs < total_elems

    # index of the element in the mask tensor (flattened over B,N,N)
    mask_idx = offs // hidden_dim

    # load values (masked)
    left_proj = tl.load(left_proj_ptr + offs, mask=msk, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=msk, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=msk, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=msk, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=msk, other=1.0)

    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=msk)
    tl.store(right_out_ptr + offs, right_out, mask=msk)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse element‑wise mask, projection and gate for both left and right tensors.
    All tensors must be FP16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)
    Returns:
        left, right : (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None (0/1)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Fuse five linear projections into one matmul
    # ------------------------------------------------------------------
    # concatenate weight matrices (out_features, in_features)
    w_left = weights["left_proj.weight"]
    w_right = weights["right_proj.weight"]
    w_lgate = weights["left_gate.weight"]
    w_rgate = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]
    combined_weight = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0
    ).to(torch.float16)  # (5*H, C)

    # single linear pass
    combined = F.linear(x, combined_weight)  # (B, N, N, 5*H)

    # split the combined tensor
    splits = torch.split(combined, hidden_dim, dim=-1)
    left_proj, right_proj, left_gate, right_gate, out_gate = splits

    # sigmoid gates
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # ------------------------------------------------------------------
    # 4) Apply mask & gating (fused via Triton when needed)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16, block_size=1024
        )
    else:
        # simple element‑wise fusion (mask ignored)
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask present but ignored – still broadcast
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 5) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM across the hidden dimension.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 7) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
