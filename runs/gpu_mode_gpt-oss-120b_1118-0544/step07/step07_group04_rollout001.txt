"""
TriMul (outgoing) forward pass – Triton‑accelerated gate‑mask fusion.

Algorithm
---------
1. **LayerNorm** over the channel dimension `dim` (FP32 for numerical stability).
2. Cast to **FP16** for the compute‑heavy part.
3. **Linear projections** (`left_proj`, `right_proj`) and **gate projections**
   (`left_gate`, `right_gate`, `out_gate`) using bias‑free `nn.Linear` weights.
4. **Fuse** projection, gate and optional mask in a single Triton kernel:
       fused = proj * gate * mask   (mask broadcast over `hidden_dim`).
   This kernel is required to satisfy the “part in Triton” rule.
5. **Core reduction**: for each batch `b` and hidden channel `h`
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is exactly a batched matrix multiplication
   `left[b,:,:,h] @ right[b,:,:,h].T`.  We perform it with
   `torch.bmm` on tensors of shape `(B*hidden_dim, N, N)`.
6. **Output LayerNorm**, multiply by `out_gate`, and a final linear projection
   back to the original `dim`.
7. Cast the result back to the input dtype (usually `float32`).

The implementation mixes highly‑optimized PyTorch primitives with a small
Triton kernel for the gating‑mask fusion, achieving good performance while
remaining simple and correct.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise fusion of projection, gate and mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,      # fp16 * (B,N,N,H)
    gate_ptr,      # fp16 * (B,N,N,H)
    mask_ptr,      # fp16 * (B,N,N)
    out_ptr,       # fp16 * (B,N,N,H)
    total_elems,   # int: B*N*N*H
    hidden_dim,    # int: H
    BLOCK_SIZE: tl.constexpr,
):
    """
    out = proj * gate * mask   (mask broadcast over hidden_dim)
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # Index of the corresponding mask element (each mask value is repeated
    # hidden_dim times along the last dimension)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(out_ptr + offs, proj * gate * m, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 65536) -> torch.Tensor:
    """
    Fuse projection, gate and mask using a Triton kernel.
    All tensors must be fp16 and contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)   (binary 0/1, broadcast over H)
    Returns proj * gate * mask (shape (B,N,N,H)).
    """
    assert proj.is_contiguous() and gate.is_contiguous()
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (float32/float16/bool) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim` (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy stages
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all linear weights to FP16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid (B,N,N,H)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Apply mask + gating (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16, contiguous, binary (0/1)
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        # No mask → simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
