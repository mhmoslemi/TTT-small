"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input (float32 for stability) → half.
2. Linear projections (left/right) from C → H using fp16 GEMM (torch.nn.functional.linear).
3. Compute three sigmoid gates (left, right, out) with linear layers.
4. Fuse mask and gate multiplications in‑place via a Triton kernel
   (left  = left  * mask * left_gate,
    right = right * mask * right_gate).
5. Core reduction: for every batch and hidden channel compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication left @ rightᵀ.
   Implemented with a custom Triton kernel that processes all (B·H) batches.
6. Second LayerNorm over the hidden dimension H.
7. Apply the output gate (element‑wise).
8. Final linear projection H → C.
9. Cast the result back to float32.

All heavy element‑wise work (mask+gate fusion, batched GEMM) is done in Triton,
while LayerNorms and linear layers use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (B,N,N,H) flattened
    right_ptr,         # fp16 * right (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and gates."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load values
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # One mask value per (b,i,k) -> broadcast over H
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply and store back
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_ah, stride_am, stride_ak,   # A: (batch, M, K)
    stride_bh, stride_bk, stride_bn,   # B: (batch, N, K)  (we will read Bᵀ)
    stride_ch, stride_cm, stride_cn,   # C: (batch, M, N)
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    """Batched GEMM: C[b] = A[b] @ B[b].T   where A,B are (M,K) and (N,K)."""
    pid_b = tl.program_id(2)          # batch index (B*H)
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Pointers with correct strides
        a_ptrs = a_ptr + pid_b * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_b * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write result
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_ptr + pid_b * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see code for expected keys)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float32)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float32)

    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # Cast to fp16 for compute‑heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse mask and gates (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()                     # B * N * N * H
        BLOCK = 8192                                      # threads per program
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates are no longer needed
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: batched matrix multiplication (left @ rightᵀ)
    #    left/right shape: (B, N, N, H) --> (B, H, N, N) --> (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left_b  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    batch = B * hidden_dim
    A = left_b.view(batch, N, N)
    B_ = right_b.view(batch, N, N)

    C = torch.empty_like(A)  # (batch, N, N), fp16

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks over rows
        triton.cdiv(N, BLOCK_N),   # blocks over cols
        batch,                     # batch dimension
    )

    _batched_matmul_kernel[grid](
        A,
        B_,
        C,
        A.stride(0), A.stride(1), A.stride(2),   # stride_ah, stride_am, stride_ak
        B_.stride(0), B_.stride(2), B_.stride(1),  # stride_bh, stride_bk, stride_bn (note: B is (batch,N,K))
        C.stride(0), C.stride(1), C.stride(2),   # stride_ch, stride_cm, stride_cn
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to (B, N, N, H)
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast back to float32 (as required by the specification)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
