"""
Triton‑accelerated outgoing TriMul (Triangle Multiplicative Update).

Algorithm
---------
1. Input LayerNorm (FP32) on the [B, N, N, C] tensor.
2. Cast to FP16 for the heavy compute path.
3. Linear projections (bias‑less) → left, right, left_gate, right_gate,
   out_gate (all in FP16).  Gates use a sigmoid.
4. Fuse mask (if present) with the left/right gates in a small Triton kernel
   (in‑place on left/right).  When no mask is provided we simply multiply
   element‑wise with the gates.
5. Core contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (left @ rightᵀ) over B·H independent N×N
   matrix multiplications, implemented with torch.bmm (cuBLAS) for maximum
   speed.
6. Apply a second LayerNorm on the hidden dimension, multiply by out_gate,
   and a final linear projection back to the original dimension.
7. Cast the result back to the input dtype.

Only the mask‑gate fusion is performed in Triton; all other ops use
high‑performance PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr,               # *fp16  [B, N, N, H] flattened
    right_ptr,              # *fp16  same layout as left
    left_gate_ptr,          # *fp16  same layout as left
    right_gate_ptr,         # *fp16  same layout as left
    mask_ptr,               # *fp16  [B, N, N] flattened (no H dimension)
    total_elems,            # i32   total number of fp16 elements in left/right
    hidden_dim,             # i32   H
    HAS_MASK: tl.constexpr, # bool  compile‑time mask flag
    BLOCK_SIZE: tl.constexpr,  # i32 block size
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    If HAS_MASK is False the mask term is omitted.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load the necessary slices
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each mask entry repeats `hidden_dim` times in the flattened view
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * m
        right = right * right_gate * m
    else:
        left = left * left_gate
        right = right * right_gate

    # Store the results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

    """

    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------

    # 3. Linear projections (bias‑less) in FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4. Fuse mask and gates (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure contiguous layout for the kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp = mask.to(device).to(torch.float16).contiguous()

        # Flatten tensors to 1‑D views
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_fp.view(-1)          # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )

        # Restore original shapes
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – simply apply the gates
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Rearrange to (B*H, N, N) for batched matmul
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (cuBLAS)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate                     # apply output gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)            # (B, N, N, C) in fp16

    # ------------------------------------------------------------------
    # 7. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
