"""
TriMul (outgoing) forward pass accelerated with Triton.

Algorithm
---------
1. Layer‑norm over the channel dimension `dim`.
2. Cast to FP16 for compute‑heavy parts.
3. Linear projections (left/right) and gate projections (left/right/out).
4. Apply element‑wise mask (if provided) and gate sigmoid.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM per head using a custom Triton kernel.
6. Output LayerNorm over the hidden dimension, multiply by `out_gate`,
   and final linear projection back to the original `dim`.
7. Cast back to the input dtype.

Only the reduction step (step 5) is performed in Triton; all other
operations use highly‑optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul reduction (outgoing version)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr,               # fp16 * (B, N, N, H)
    right_ptr,              # fp16 * (B, N, N, H)
    out_ptr,                # fp16 * (B, N, N, H)

    B, N, H,                # i64 scalar dimensions

    # strides (in elements) for left tensor
    stride_lb, stride_li, stride_lk, stride_lh,
    # strides for right tensor
    stride_rb, stride_rj, stride_rk, stride_rh,
    # strides for out tensor
    stride_ob, stride_oi, stride_oj, stride_oh,

    # compile‑time block sizes
    BLOCK_M: tl.constexpr,  # rows of the output tile   (i dimension)
    BLOCK_N: tl.constexpr,  # cols of the output tile   (j dimension)
    BLOCK_K: tl.constexpr,  # reduction tile size       (k dimension)
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where left/right have shape (B, N, N, H) and out the same.
    """
    pid_bh = tl.program_id(2)               # batch * hidden index
    pid_m  = tl.program_id(0)               # tile row
    pid_n  = tl.program_id(1)               # tile col

    # Compute batch index and hidden index from the combined id
    b = pid_bh // H
    h = pid_bh % H

    # Base pointers for this (b, h)
    base_left  = left_ptr  + b * stride_lb + h * stride_lh
    base_right = right_ptr + b * stride_rb + h * stride_rh
    base_out   = out_ptr   + b * stride_ob + h * stride_oh

    # Tile coordinates
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator in fp32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for start_k in range(0, tl.cdiv(N, BLOCK_K)):
        k = start_k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            base_left
            + offs_i[:, None] * stride_li
            + k[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B block: shape (BLOCK_K, BLOCK_N)
        b_ = tl.load(
            base_right
            + offs_j[None, :] * stride_rj
            + k[:, None] * stride_rk,
            mask=mask_j[None, :] & mask_k[:, None],
            other=0.0,
        )
        # Accumulate
        acc += tl.dot(a, b_)

    # Write result (cast back to fp16)
    c = acc.to(tl.float16)
    tl.store(
        base_out
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj,
        c,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper to launch the Triton kernel
# ----------------------------------------------------------------------
def _tri_mul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the TriMul reduction using the Triton kernel.

    Args:
        left, right: tensors of shape (B, N, N, H), dtype float16, contiguous.

    Returns:
        out tensor of the same shape and dtype.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Strides in elements (not bytes)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Tunable block sizes (chosen for good occupancy on B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        B * H,                     # pid_bh
    )

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C]  (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None (bool/int/float)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) LayerNorm over the input channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast all linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 5) Apply mask and gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape: (B, N, N) -> broadcast to (B, N, N, 1)
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = left_proj * left_gate * mask_fp16[..., None]
        right = right_proj * right_gate * mask_fp16[..., None]
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # Ensure contiguity for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction via Triton
    # ------------------------------------------------------------------
    out = _tri_mul_outgoing_triton(left, right)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
