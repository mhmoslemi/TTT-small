def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # each program processes a tile of (BLOCK_M, BLOCK_N)
    batch_id = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_in_batch = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    # compute block indices within batch
    pid_m = pid_in_batch // tl.cdiv(N, BLOCK_N)
    pid_n = pid_in_batch % tl.cdiv(N, BLOCK_N)
    # offsets for a and b
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # offsets for k
    k = tl.arange(0, BLOCK_K)
    # compute pointers
    a_ptrs = a_ptr + batch_id * stride_ah + offs_m[:, None] * stride_am + k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bh + k[:, None] * stride_bk + offs_n[None, :] * stride_bn
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        # load a and b
        a = tl.load(a_ptrs, mask=offs_m < M and k < K ?, ... ) ...
        # but mask can be done using tl.where
        # Actually we compute mask_m = offs_m < M
        # mask_n = offs_n < N
        # mask_k = k < K
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(k < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        # dot
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_ak

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_ah, stride_am, stride_ak,
    stride_bh, stride_bk, stride_bn,
    stride_ch, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # Compute number of blocks in M and N
    num_m_blocks = tl.cdiv(M, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)
    # Compute batch id and block indices
    batch_id = pid // (num_m_blocks * num_n_blocks)
    pid_in_batch = pid % (num_m_blocks * num_n_blocks)
    pid_m = pid_in_batch // num_n_blocks
    pid_n = pid_in_batch % num_n_blocks

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute offsets for K block
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Compute pointers
        a_ptrs = a_ptr + batch_id * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch_id * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load with mask
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + batch_id * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The operator works on a tensor x of shape [B, N, N, C] and computes:
    left  = Linear(x) * mask * sigmoid(Linear(x))
    right = Linear(x) * mask * sigmoid(Linear(x))
    out   = Σ_k left[..., i, k, h] * right[..., j, k, h]   (for all i,j,h)
    out   = LayerNorm(out) * sigmoid(Linear(x))
    out   = Linear(out)   # back to channel dimension C

The expensive reduction Σ_k is performed with a custom Triton batched‑GEMM
kernel that treats each (batch, hidden) pair as an independent matrix
multiply (N×N)·(N×N)ᵀ.  All other operations use PyTorch (mixed‑precision
float16 for speed, float32 for numerical stability where needed).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    stride_ah, stride_am, stride_ak,   # a strides
    stride_bh, stride_bk, stride_bn,   # b strides
    stride_ch, stride_cm, stride_cn,   # c strides
    M, N, K,                           # matrix sizes (M=N=seq_len, K=N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiply C = A @ B  where
       A : (batch, M, K)   fp16
       B : (batch, K, N)   fp16
       C : (batch, M, N)   fp16
    """
    pid = tl.program_id(0)

    # number of blocks in each dimension
    num_m_blocks = tl.cdiv(M, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)

    # decode block indices
    batch_id = pid // (num_m_blocks * num_n_blocks)
    pid_in_batch = pid % (num_m_blocks * num_n_blocks)
    pid_m = pid_in_batch // num_n_blocks
    pid_n = pid_in_batch % num_n_blocks

    # coordinate offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in float32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)

        # pointers with proper offsets
        a_ptrs = a_ptr + batch_id * stride_ah \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak

        b_ptrs = b_ptr + batch_id * stride_bh \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # load tiles with bounds checks
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)

        # matrix multiplication of the tile
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N) in fp32

    # write back the result
    c_ptrs = c_ptr + batch_id * stride_ch \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn

    mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see keys in code)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float32)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Cast to fp16 for compute‑heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask (if any) and gates – elementwise multiplication
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp16 * left_gate
        right = right * mask_fp16 * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: batched matrix multiplication using Triton
    #    left: (B, N, N, H) -> (B, H, N, N)
    #    right: (B, N, N, H) -> (B, H, N, N) and then transpose K/N
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left_b = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # transpose the last two dimensions of right for GEMM
    right_t = right_b.transpose(-2, -1).contiguous()  # (B, H, N, N) -> (B, H, N, N) as (K,N)

    # flatten batch and hidden dimensions
    batch_h = B * hidden_dim
    a = left_b.view(batch_h, N, N)      # (batch_h, M=N, K=N)
    b = right_t.view(batch_h, N, N)    # (batch_h, K=N, N)

    # allocate output
    c = torch.empty((batch_h, N, N), dtype=torch.float16, device=device)

    # kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    num_m_blocks = triton.cdiv(N, BLOCK_M)
    num_n_blocks = triton.cdiv(N, BLOCK_N)
    grid = (batch_h * num_m_blocks * num_n_blocks,)

    _batched_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        stride_ah=a.stride(0),
        stride_am=a.stride(1),
        stride_ak=a.stride(2),
        stride_bh=b.stride(0),
        stride_bk=b.stride(1),
        stride_bn=b.stride(2),
        stride_ch=c.stride(0),
        stride_cm=c.stride(1),
        stride_cn=c.stride(2),
        M=N,
        N=N,
        K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to (B, H, N, N) then to (B, N, N, H)
    out = c.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
```
