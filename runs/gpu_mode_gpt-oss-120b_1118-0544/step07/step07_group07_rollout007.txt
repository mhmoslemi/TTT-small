"""
custom_kernel: Triton‑accelerated outgoing TriMul (Triangle Multiplicative Update).

Algorithm
---------
1. LayerNorm on the input tensor (float32 → float32).
2. Cast to FP16 for the heavy computation.
3. Linear projections (bias‑less) to obtain `left`, `right`,
   `left_gate`, `right_gate`, and `out_gate` (all FP16).
4. Fuse mask (if present) and gate multiplication in a single
   Triton kernel that updates `left` and `right` in‑place.
5. Core contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched GEMM (left @ rightᵀ) over B·H matrices using
   torch.bmm (cuBLAS).
6. Apply a second LayerNorm on the hidden dimension, multiply by
   `out_gate`, and a final linear projection back to the original
   dimension.
7. Cast back to the input dtype and return.

Only the mask‑gate fusion step is implemented in Triton; all other
operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr,               # fp16*   [B,N,N,H] flattened
    right_ptr,              # fp16*   same shape
    left_gate_ptr,          # fp16*   same shape
    right_gate_ptr,         # fp16*   same shape
    mask_ptr,               # fp16*   [B,N,N] flattened
    total_elems,            # i32 scalar, total number of elements in left/right
    hidden_dim,             # i32 scalar, H
    HAS_MASK: tl.constexpr, # compile‑time bool
    BLOCK_SIZE: tl.constexpr # compile‑time int
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    If HAS_MASK=False the mask term is omitted.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # Load vectors
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    if HAS_MASK:
        # One mask value per (b,i,k) repeats `hidden_dim` times in the flattened view.
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) (may be ignored)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (bias‑less) in fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)    # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)  # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4. Fuse mask and gates (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # Flatten all tensors for the kernel
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024  # reasonable block size

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back to original layout
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – simple element‑wise gating
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Merge batch and hidden dimensions for a batched matrix multiply
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B*H, N, N)

    # Batched GEMM (uses cuBLAS, fp16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, gating, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply the output gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------
    # 7. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
