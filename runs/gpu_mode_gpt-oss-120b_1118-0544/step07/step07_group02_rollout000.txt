"""
custom_kernel: Triton‑accelerated outgoing TriMul (Triangle Multiplicative Update)

The forward pass follows the reference PyTorch implementation:
1. LayerNorm on the input (float32 for stability).
2. Cast to fp16 for compute‑heavy paths.
3. Linear projections (left/right) and three sigmoid gates (left, right, out).
4. Fuse mask (if present) together with the left/right gates in a small Triton
   kernel that works in‑place on the flattened tensors.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to a batched GEMM (L @ Rᵀ) for each hidden channel.
6. Second LayerNorm on the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original dimension.
7. Cast the result to float32 and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # fp16*  [B,N,N,H] flattened
    right_ptr,              # fp16*  same shape
    left_gate_ptr,          # fp16*  same shape
    right_gate_ptr,         # fp16*  same shape
    mask_ptr,               # fp16*  [B,N,N] flattened (broadcast over H)
    total_elems,            # total number of elements in left/right tensors
    hidden_dim,             # H
    HAS_MASK: tl.constexpr, # compile‑time bool
    BLOCK_SIZE: tl.constexpr,  # compile‑time int
):
    """In‑place fusion:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load values (masked load returns 0 for out‑of‑range elements)
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Each hidden element shares the same mask entry.
        mask_idx = offs // hidden_dim            # index in [B,N,N]
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)   (float32)
            - mask:        torch.Tensor of shape (B, N, N) or None
            - weights:     dict of model parameters (float32 tensors)
            - config:      dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, mask is ignored

    Returns:
        torch.Tensor of shape (B, N, N, C) (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive ops
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections & gates (all fp16)
    # -----------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device, dtype=torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device, dtype=torch.float16)

    # Linear projections: shape (B,N,N,H)
    left = F.linear(x, left_proj_w)                 # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)               # (B,N,N,hidden_dim)

    # Gates: shape (B,N,N,hidden_dim)
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 4. Fuse mask (if any) with left/right gates via Triton
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp = mask.to(device, dtype=torch.float16).contiguous()
        # Flatten tensors for the kernel
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)   # shape (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back to original layout
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # Simple elementwise multiplication when mask is absent
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (L @ Rᵀ) per hidden channel.
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # (B*H, N, N) tensors
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (broadcasted)

    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float16)
    out = F.linear(out, to_out_w)  # (B,N,N,dim)

    # -----------------------------------------------------------------
    # 7. Cast back to float32 and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
