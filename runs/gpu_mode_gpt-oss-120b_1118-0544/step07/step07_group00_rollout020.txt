import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # compute index of mask element (each mask element repeats hidden_dim times)
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gating and optional mask for both left and right tensors.
    All tensors must be fp16 and contiguous.
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask_tensor:  torch.Tensor of shape [B, N, N] with 0/1 (or None)
            - weights_dict: contains all linear and layer‑norm parameters.
            - config_dict: must contain "dim" and "hidden_dim",
                           optional "nomask" (bool).

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over channel dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast weights to fp16 (no bias in these linear layers)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Fuse mask (if present) with projection and gating using Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is contiguous fp16
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # simple element‑wise multiplication, mask is ignored or all‑ones
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM (fp16) over (B*hidden_dim) matrices.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batch matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm over hidden dimension, apply out_gate and final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for the compute‑intensive part.
3. Linear projections (left/right) and gating projections (left_gate,

4. If a mask is supplied (and not ignored) fuse mask, projection and

   Otherwise perform simple element‑wise multiplication.
5. Core reduction:

6. Layer‑norm over the hidden dimension, multiply by the output gate,

7. Cast the result back to the original dtype (usually float32).

    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (fusion via Triton if mask is used)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is half and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask or mask is ignored → simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, H, N, N) → (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM: (B*H, N, N) @ (B*H, N, N)^T → (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
```
