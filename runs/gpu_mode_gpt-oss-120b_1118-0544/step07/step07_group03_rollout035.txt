def batched_matmul_kernel( a_ptr, b_ptr, c_ptr,
                           batch_stride_a, stride_am, stride_ak,
                           batch_stride_b, stride_bk, stride_bn,
                           batch_stride_c, stride_cm, stride_cn,
                           M, N, K,
                           BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
                           GROUP_M: tl.constexpr):
   pid = tl.program_id(0)
   # Determine batch index and block coordinates
   num_block_m = tl.cdiv(M, BLOCK_M)
   num_block_n = tl.cdiv(N, BLOCK_N)
   batch = pid // (num_block_m * num_block_n)
   block_id = pid % (num_block_m * num_block_n)
   pid_m = block_id // num_block_n
   pid_n = block_id % num_block_n

   # Compute offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

   # mask for out-of-bounds
   mask_m = offs_m < M
   mask_n = offs_n < N

   # Offsets for C
   c_ptrs = c_ptr + batch * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

   # Initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # Loop over K
   for k in range(0, tl.cdiv(K, BLOCK_K)):
       k_offset = k * BLOCK_K
       offs_k = k_offset + tl.arange(0, BLOCK_K)

       # mask for k dimension
       mask_k = offs_k < K

       # Load A block: shape (BLOCK_M, BLOCK_K)
       a_ptrs = a_ptr + batch * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
       a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

       # Load B block transposed: we need B[b, j, k] -> we want B_T block shape (BLOCK_K, BLOCK_N)
       b_ptrs = b_ptr + batch * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
       b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

       # Compute dot product
       acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) dot (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

   # Write back to C
   tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the 4‑D input over the channel dimension (dim).
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections (left/right) and gating (left/right/out) using FP16
   weights.  If a mask is present, fuse mask, projection and gate for both
   left and right tensors in a single Triton kernel.
4. Core TriMul reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication `C = A @ Bᵀ` where
   `A = left.reshape(B*H, N, N)` and `B = right.reshape(B*H, N, N)`.
   Implemented with a custom Triton kernel that performs the
   transposed GEMM with fp16 inputs and fp32 accumulation.
5. Output LayerNorm over the hidden dimension, element‑wise multiplication
   with `out_gate`, and final linear projection back to the original
   channel dimension.
6. Cast the result back to the input dtype (usually float32).

The implementation mixes highly‑optimized Triton kernels with PyTorch
primitives to keep the code simple while achieving a large speed‑up over
the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask for both left and right
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B*N*N*H,)
    right_proj_ptr,  # fp16 * (B*N*N*H,)
    left_gate_ptr,   # fp16 * (B*N*N*H,)
    right_gate_ptr,  # fp16 * (B*N*N*H,)
    mask_ptr,        # fp16 * (B*N*N,)
    left_out_ptr,    # fp16 * (B*N*N*H,)
    right_out_ptr,   # fp16 * (B*N*N*H,)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # OOB guard
    mask = offs < total_elems

    # mask is broadcast over the hidden dimension
    mask_idx = offs // hidden_dim

    left_proj = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gate and mask for both left and right tensors in a single
    Triton kernel.
    All tensors must be FP16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)
    Returns:
        left, right : (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of blocks in each dimension
    num_block_m = tl.cdiv(M, BLOCK_M)
    num_block_n = tl.cdiv(N, BLOCK_N)

    # decompose pid into batch, block-m, block-n
    batch = pid // (num_block_m * num_block_n)
    block_id = pid % (num_block_m * num_block_n)
    pid_m = block_id // num_block_n
    pid_n = block_id % num_block_n

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out-of-range indices
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A block: (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + batch * batch_stride_a \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # B block transposed: we need Bᵀ, i.e. access (k, n)
        b_ptrs = b_ptr + batch * batch_stride_b \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # write back
    c_ptrs = c_ptr + batch * batch_stride_c \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

    # reshape to (B*H, N, N) and make contiguous
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)

    # strides in element units
    batch_stride_a = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    batch_stride_b = right_flat.stride(0)
    stride_bk = right_flat.stride(2)   # K dimension (column) stride
    stride_bn = right_flat.stride(1)   # N dimension (row) stride

    batch_stride_c = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    M = N
    K = N

    # Blocking parameters (tuned for typical B200 hardware)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    GROUP_M = 8

    num_block_m = (M + BLOCK_M - 1) // BLOCK_M
    num_block_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B * H * num_block_m * num_block_n
    grid = (total_blocks,)

    _batched_matmul_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_stride_a, stride_am, stride_ak,
        batch_stride_b, stride_bk, stride_bn,
        batch_stride_c, stride_cm, stride_cn,
        M, N, K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_M=GROUP_M,
    )

    # Restore original shape (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
