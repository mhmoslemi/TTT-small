```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_bh, stride_l_m, stride_l_k,
    stride_r_bh, stride_r_k, stride_r_n,
    stride_o_bh, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # batch index
    bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for OOB
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block
        a_ptrs = left_ptr + bh * stride_l_bh + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B block (transposed)
        b_ptrs = right_ptr + bh * stride_r_bh + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Store results
    c_ptrs = out_ptr + bh * stride_o_bh + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# Triton kernel for batched GEMM (left @ right^T)
@triton.jit
def tri_mul_kernel(
    left_ptr,   # fp16 *
    right_ptr,  # fp16 *
    out_ptr,    # fp16 *
    # tensor dimensions
    N,
    # strides (in elements)
    stride_l_bh, stride_l_m, stride_l_k,
    stride_r_bh, stride_r_k, stride_r_n,
    stride_o_bh, stride_o_m, stride_o_n,
    # compile-time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # batch*hidden index
    bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block (left)
        a_ptrs = left_ptr + \
                 bh * stride_l_bh + \
                 offs_m[:, None] * stride_l_m + \
                 offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B block (right transposed)
        b_ptrs = right_ptr + \
                 bh * stride_r_bh + \
                 offs_k[:, None] * stride_r_k + \
                 offs_n[None, :] * stride_r_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # write back result
    c_ptrs = out_ptr + \
             bh * stride_o_bh + \
             offs_m[:, None] * stride_o_m + \
             offs_n[None, :] * stride_o_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.
    Implements LayerNorm, linear projections, gating, optional mask,
    and the core O(N³) reduction using a fused Triton GEMM kernel.
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype
    # Use FP16 for heavy compute unless input already FP16
    compute_dtype = torch.float16 if dtype_in == torch.float32 else dtype_in

    # ------------------------------------------------------------------
    # 1) LayerNorm on input (over channel dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to compute dtype
    x = x.to(compute_dtype)

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to compute dtype (no bias)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(compute_dtype)
    right_proj_w  = weights["right_proj.weight"].to(compute_dtype)
    left_gate_w   = weights["left_gate.weight"].to(compute_dtype)
    right_gate_w  = weights["right_gate.weight"].to(compute_dtype)
    out_gate_w    = weights["out_gate.weight"].to(compute_dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(compute_dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(compute_dtype)
    to_out_w      = weights["to_out.weight"].to(compute_dtype)

    # ------------------------------------------------------------------
    # 3) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 4) Apply mask (if any) and combine with gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(compute_dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * mask_f * left_gate
        right = right_proj * mask_f * right_gate
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguity before permute
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 5) Core reduction using Triton GEMM kernel: out = left @ rightᵀ
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # N is seq_len, H is hidden_dim
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    BH = B * H
    left_flat = left_flat.view(BH, N, N)
    right_flat = right_flat.view(BH, N, N)
    out_flat = torch.empty_like(left_flat)  # fp16

    # Strides in elements (not bytes)
    stride_l_bh = left_flat.stride(0)
    stride_l_m  = left_flat.stride(1)
    stride_l_k  = left_flat.stride(2)

    stride_r_bh = right_flat.stride(0)
    # For transposed access we swap the row/col strides
    stride_r_k = right_flat.stride(2)   # original column stride
    stride_r_n = right_flat.stride(1)   # original row stride

    stride_o_bh = out_flat.stride(0)
    stride_o_m  = out_flat.stride(1)
    stride_o_n  = out_flat.stride(2)

    # Block sizes – tuned for B200 (FP16)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (BH, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_l_bh, stride_l_m, stride_l_k,
        stride_r_bh, stride_r_k, stride_r_n,
        stride_o_bh, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 7) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

custom_kernel: Outgoing TriMul forward (AlphaFold3) using a Triton GEMM kernel.
Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to FP16 (or keep FP16) for compute‑heavy parts.
3. Linear projections (left/right) and gate projections (sigmoid).
4. Optional mask is fused with projections and gates (element‑wise).
5. Core O(N³) reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (left @ rightᵀ) with a tiled
   Triton kernel.  The batch*hidden dimension (B·H) is parallelised as the
   first kernel dimension.
6. Output LayerNorm, out‑gate multiplication and final linear projection.
7. Cast back to the original dtype.
"""

import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

def tri_mul_kernel(
    left_ptr,   # fp16 *
    right_ptr,  # fp16 *
    out_ptr,    # fp16 *

    # strides (in elements)
    stride_l_bh, stride_l_m, stride_l_k,
    stride_r_bh, stride_r_k, stride_r_n,
    stride_o_bh, stride_o_m, stride_o_n,
    # compile‑time block sizes

):

    # One program instance processes a tile (BLOCK_M×BLOCK_N) of the
    # output matrix for a specific batch‑hidden slice (bh).
    # ---------------------------------------------------------------
    bh   = tl.program_id(0)                     # batch * hidden index
    pid_m = tl.program_id(1)                    # tile row index
    pid_n = tl.program_id(2)                    # tile col index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over the reduction dimension k ---------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a block of A = left
        a_ptrs = left_ptr + \
                 bh * stride_l_bh + \
                 offs_m[:, None] * stride_l_m + \
                 offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load a block of B = rightᵀ (transposed on‑the‑fly)
        b_ptrs = right_ptr + \
                 bh * stride_r_bh + \
                 offs_k[:, None] * stride_r_k + \
                 offs_n[None, :] * stride_r_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # ---- write the result ----------------------------------------------------
    c_ptrs = out_ptr + \
             bh * stride_o_bh + \
             offs_m[:, None] * stride_o_m + \
             offs_n[None, :] * stride_o_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with model parameters (FP32)
            - config: dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    dtype_in = input_tensor.dtype
    compute_dtype = torch.float16 if dtype_in == torch.float32 else dtype_in

    # ---------------------------------------------------------------
    # 1) Input LayerNorm over channel dimension
    # ---------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(compute_dtype)

    # ---------------------------------------------------------------
    # 2) Cast linear weights to compute dtype (no bias)
    # ---------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(compute_dtype)
    right_proj_w  = weights["right_proj.weight"].to(compute_dtype)
    left_gate_w   = weights["left_gate.weight"].to(compute_dtype)
    right_gate_w  = weights["right_gate.weight"].to(compute_dtype)
    out_gate_w    = weights["out_gate.weight"].to(compute_dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(compute_dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(compute_dtype)
    to_out_w      = weights["to_out.weight"].to(compute_dtype)

    # ---------------------------------------------------------------
    # 3) Projections and gates
    # ---------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)                # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # ---------------------------------------------------------------
    # 4) Apply mask (if present) and combine with gates
    # ---------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(compute_dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left_proj * mask_f * left_gate
        right = right_proj * mask_f * right_gate
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ---------------------------------------------------------------
    # 5) Core reduction: out = left @ rightᵀ  (batched GEMM)
    # ---------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for the batched kernel
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_flat = torch.empty_like(left_flat)   # fp16 output

    # strides in *elements*
    stride_l_bh = left_flat.stride(0)
    stride_l_m  = left_flat.stride(1)
    stride_l_k  = left_flat.stride(2)

    stride_r_bh = right_flat.stride(0)
    # transposed access: swap the last two strides
    stride_r_k = right_flat.stride(2)   # original column stride
    stride_r_n = right_flat.stride(1)   # original row stride

    stride_o_bh = out_flat.stride(0)
    stride_o_m  = out_flat.stride(1)
    stride_o_n  = out_flat.stride(2)

    # block sizes (tuned for H100/B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_l_bh, stride_l_m, stride_l_k,
        stride_r_bh, stride_r_k, stride_r_n,
        stride_o_bh, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------------------------------------------------------------
    # 6) Output layer‑norm, out‑gate and final projection
    # ---------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ---------------------------------------------------------------
    # 7) Cast back to the original dtype
    # ---------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
