import torch.nn.functional as F

import triton.language as tl

# Kernel for mask and gate fusion (same as previous)
@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    H: tl.constexpr,
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    tl.store(left_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)

# Kernel for the core reduction (outgoing TriMul)
@triton.jit
def _trimul_reduce_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_j, stride_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
    num_warps: tl.constexpr):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    left/right/out are contiguous tensors with shape (B,N,N,H) and strides:
        stride_b = N*N*H, stride_i = N*H, stride_j = H, stride_h = 1
    """
    pid_m = tl.program_id(0)  # tile over i
    pid_n = tl.program_id(1)  # tile over j
    pid_z = tl.program_id(2)  # combines batch and hidden-tile

    # compute tiling offsets
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # number of hidden tiles
    total_h_tiles = (H + BLOCK_H - 1) // BLOCK_H
    b = pid_z // total_h_tiles
    h_tile = (pid_z % total_h_tiles) * BLOCK_H

    # range indices
    i_range = tl.arange(0, BLOCK_M)
    j_range = tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)
    h_range = tl.arange(0, BLOCK_H)

    # bounds masks
    i_mask = i0 + i_range < N
    j_mask = j0 + j_range < N
    h_mask = h_tile + h_range < H

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # loop over reduction dimension k
    for k0 in range(0, N, BLOCK_K):
        k_mask = k0 + k_range < N

        # compute absolute indices
        i_idx = i0 + i_range[:, None, None]          # (BM,1,1)
        j_idx = j0 + j_range[None, :, None]          # (1,BN,1)
        k_idx = k0 + k_range[None, None, :]          # (1,1,BK)
        h_idx = h_tile + h_range[None, None, :]      # (1,1,BH)

        # left address: (b,i,k,h)
        left_offsets = b * stride_b + i_idx * stride_i + k_idx * stride_j + h_idx * stride_h
        left = tl.load(left_ptr + left_offsets,
                       mask = i_mask[:, None, None] & k_mask[None, None, :] & h_mask[None, None, :],
                       other = 0.0)

        # right address: (b,j,k,h)
        right_offsets = b * stride_b + j_idx * stride_i + k_idx * stride_j + h_idx * stride_h
        right = tl.load(right_ptr + right_offsets,
                        mask = j_mask[:, None, None] & k_mask[None, None, :] & h_mask[None, None, :],
                        other = 0.0)

        # compute dot for each hidden slice
        for hh in range(0, BLOCK_H):
            # slice (BM,BK) and (BN,BK)
            l_slice = left[:, :, hh]   # (BM,BK)
            r_slice = right[:, :, hh]  # (BN,BK)
            # tl.dot expects (M,K) @ (K,N). r_slice needs to be transposed to (BK,BN)
            acc[:, :, hh] += tl.dot(l_slice, r_slice, trans_b=True)

    # store result
    i_idx = i0 + i_range[:, None, None]

    # ------------------------------------------------------------------

    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )

        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core O(N³·H) contraction using Triton
    # ------------------------------------------------------------------

    # ensure contiguous layout for simple stride computation
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # fp16 tensor for result

    stride_b, stride_i, stride_j, stride_h = left.stride()

    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 32

    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid_h = (H + BLOCK_H - 1) // BLOCK_H

        H=H,

        stride_h=stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------

        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,

    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:

    return out

# ----------------------------------------------------------------------
# 1) Fuse mask and gate (in‑place) – this part is unchanged from before
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (B,N,N,H) flattened
    right_ptr,         # fp16 * right (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):

    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # one mask value per hidden element
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)

        del left_gate, right_gate
    else:
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction using Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # fp16 output tensor

    # strides for shape (B,N,N,H) – guaranteed contiguous
    stride_b, stride_i, stride_j, stride_h = left.stride()

    # Tiling parameters (tuned for A100/B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 32

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid_h = (H + BLOCK_H - 1) // BLOCK_H
    grid_z = B * grid_h
    grid = (grid_m, grid_n, grid_z)

    _trimul_reduce_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_b=stride_b,
        stride_i=stride_i,
        stride_j=stride_j,
        stride_h=stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
