```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # which element in mask (which repeats hidden_dim times)
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(left_out_ptr + offs, lp * lg * m, mask=mask)
    tl.store(right_out_ptr + offs, rp * rg * m, mask=mask)

def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """Fuse projection, gate and optional mask with Triton.

    All tensors must be fp16, contiguous.
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    Computes:
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    where `left` and `right` are linear projections of the normalized input,
    optionally masked and gated. The implementation fuses the mask+gate
    multiplication with a small Triton kernel and uses a batched matrix
    multiplication (torch.matmul) for the N³ reduction.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- 1. Input LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- 2. Cast to FP16 for compute ----------
    x = x.to(torch.float16)

    # ---------- 3. Cast weights ----------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ---------- 4. Linear projections ----------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ---------- 5. Apply mask + gate (fused via Triton) ----------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        # If a mask exists but we ignore it, treat as ones (no op)

    # ---------- 6. Core reduction (batched matmul) ----------
    # reshape to (B, hidden_dim, N, N) for batched matmul
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    # out shape (B, hidden_dim, N, N)
    out = torch.matmul(left, right.transpose(-1, -2))

    # bring back to (B, N, N, hidden_dim)
    out = out.permute(0, 2, 3, 1).contiguous()

    # ---------- 7. Output LayerNorm, gating, final linear ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)

    # ---------- 8. Cast back to original dtype ----------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 (the heavy compute part runs in half‑precision).
3. Linear projections (`left_proj`, `right_proj`) and gating projections
   (`left_gate`, `right_gate`, `out_gate`) are performed with torch.nn.functional.linear.
4. If a mask is provided (and not ignored) we fuse:
       left  = left_proj  * left_gate  * mask
       right = right_proj * right_gate * mask
   with a small Triton kernel that broadcasts the mask across the hidden dimension.
   Otherwise the fusion is done with a simple element‑wise multiplication in PyTorch.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden slice:
       out_h = left_h @ right_hᵀ .
   Implemented with torch.matmul on tensors of shape (B, H, N, N).
6. Apply a second LayerNorm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.
7. Cast the result back to the original input dtype.

Only the mask‑gate fusion is performed in Triton; the rest leverages
high‑performance PyTorch/cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B*N*N*H,)
    right_proj_ptr,  # fp16 * (B*N*N*H,)
    left_gate_ptr,   # fp16 * (B*N*N*H,)
    right_gate_ptr,  # fp16 * (B*N*N*H,)
    mask_ptr,        # fp16 * (B*N*N,)
    left_out_ptr,    # fp16 * (B*N*N*H,)
    right_out_ptr,   # fp16 * (B*N*N*H,)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask is broadcasted over the hidden dimension:
    # each mask element repeats `hidden_dim` times in the flattened tensor.
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m  = tl.load(mask_ptr + mask_idx,   mask=valid, other=1.0)

    tl.store(left_out_ptr + offs,  lp * lg * m, mask=valid)
    tl.store(right_out_ptr + offs, rp * rg * m, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gating and mask using a Triton kernel.

    All tensors must be fp16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)

    Returns:
        left, right : (B, N, N, H) = proj * gate * mask
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],          # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out

            - mask         : torch.Tensor [B, N, N] (0/1) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """

    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------

    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (fusion via Triton if mask is used)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        # If a mask exists but we are instructed to ignore it,
        # treating it as all‑ones leaves the tensors unchanged.

    # ------------------------------------------------------------------
    # 6) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    = left_h @ right_hᵀ  for every hidden slice h
    # ------------------------------------------------------------------
    # reshape to (B, hidden_dim, N, N) for batched matmul
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # batched matrix multiplication
    out = torch.matmul(left, right.transpose(-1, -2))   # (B, hidden_dim, N, N)

    # reshape back to (B, N, N, hidden_dim)
    out = out.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
