Signature:

@triton.jit
def tri_mul_gemm_kernel(
    out_ptr, left_ptr, right_ptr,
    BATCH, M, N, K,
    stride_out_batch, stride_out_m, stride_out_n,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_n, stride_right_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Program IDs
    pid_b = tl.program_id(2)  # batch index
    pid_m = tl.program_id(0)  # row tile index
    pid_n = tl.program_id(1)  # column tile index

    # Compute offsets for blocks
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Mask for out of bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator (use fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute offsets for left and right blocks
        k_base = k * BLOCK_K
        # Offsets for left: (batch, m, k)
        left_ptrs = left_ptr + pid_b * stride_left_batch + offs_m[:, None] * stride_left_m + (k_base + offs_k)[None, :] * stride_left_k
        # Offsets for right: (batch, n, k) (since right has shape (batch, N, K))
        right_ptrs = right_ptr + pid_b * stride_right_batch + offs_n[:, None] * stride_right_n + (k_base + offs_k)[None, :] * stride_right_k

        # Load blocks
        left = tl.load(left_ptrs, mask=mask_m[:, None] & (k_base + offs_k < K), other=0.0)  # shape (BLOCK_M, BLOCK_K)
        right = tl.load(right_ptrs, mask=mask_n[:, None] & (k_base + offs_k < K), other=0.0) # shape (BLOCK_N, BLOCK_K)

        # Convert to fp32
        left = left.to(tl.float32)
        right = right.to(tl.float32)

        # Accumulate
        acc += tl.dot(left, right)  # shape (BLOCK_M, BLOCK_N)

    # Write back to out
    out_ptrs = out_ptr + pid_b * stride_out_batch + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    # Cast accumulator to fp16
    out = acc.to(tl.float16)

    # Store with mask
    tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input (float32 for stability).
2. Cast to fp16 for compute‑heavy path.
3. Linear projections (left/right) and three sigmoid gates (left, right, out).
4. Fuse mask (if present) together with left/right gates using a small Triton kernel.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM (L @ Rᵀ) with a custom Triton kernel that
   processes all hidden‑dim batches in parallel and accumulates in fp32.
6. Apply a second LayerNorm on the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original dimension.
7. Cast the result to float32 and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # [B,N,N,H] flattened
    right_ptr,              # same shape
    left_gate_ptr,          # same shape
    right_gate_ptr,         # same shape
    mask_ptr,               # [B,N,N] flattened, broadcast over H
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
       If HAS_MASK is False the mask is ignored (treated as 1).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden‑dim element shares the same mask entry
        mask_idx = offs // hidden_dim            # index in [B,N,N]
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


@triton.jit
def _tri_mul_gemm_kernel(
    out_ptr, left_ptr, right_ptr,
    BATCH, M, N, K,
    stride_out_batch, stride_out_m, stride_out_n,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_n, stride_right_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM for TriMul:
       out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]
       where each batch corresponds to a (batch * hidden) pair.
    """
    pid_b = tl.program_id(2)   # batch index (0 .. BATCH-1)
    pid_m = tl.program_id(0)   # tile row index
    pid_n = tl.program_id(1)   # tile col index

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        cur_k = k_base + offs_k
        mask_k = cur_k < K

        # left block: (M,K)
        left_ptrs = (
            left_ptr
            + pid_b * stride_left_batch
            + offs_m[:, None] * stride_left_m
            + cur_k[None, :] * stride_left_k
        )
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        # right block: (N,K)
        right_ptrs = (
            right_ptr
            + pid_b * stride_right_batch
            + offs_n[:, None] * stride_right_n
            + cur_k[None, :] * stride_right_k
        )
        right = tl.load(right_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # fp32 accumulation
        acc += tl.dot(left.to(tl.float32), right.to(tl.float32))

    # write back
    out_ptrs = (
        out_ptr
        + pid_b * stride_out_batch
        + offs_m[:, None] * stride_out_m
        + offs_n[None, :] * stride_out_n
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) (can be None)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int)          – input channel dimension C
                * "hidden_dim" (int)   – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with dtype torch.float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for the compute‑heavy part
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gates (all in fp16)
    # -----------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)                     # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)                   # (B,N,N,hidden_dim)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # -----------------------------------------------------------------
    # 4. Fuse mask and left/right gates using Triton (in‑place)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten tensors
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK_SIZE = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM (L @ Rᵀ) via Triton
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Move hidden dim to second axis and make contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    batch = B * H
    left_mat  = left_perm.view(batch, N, N)
    right_mat = right_perm.view(batch, N, N)

    out_mat = torch.empty((batch, N, N), dtype=torch.float16, device=device)

    # strides for the tensors
    stride_out_batch = out_mat.stride(0)
    stride_out_m     = out_mat.stride(1)
    stride_out_n     = out_mat.stride(2)

    stride_left_batch = left_mat.stride(0)
    stride_left_m     = left_mat.stride(1)
    stride_left_k     = left_mat.stride(2)

    stride_right_batch = right_mat.stride(0)
    stride_right_n     = right_mat.stride(1)   # corresponds to output column index j
    stride_right_k     = right_mat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # tiles over rows (i)
        triton.cdiv(N, BLOCK_N),   # tiles over cols (j)
        batch,                     # batch * hidden
    )

    _tri_mul_gemm_kernel[grid](
        out_mat,
        left_mat,
        right_mat,
        batch, N, N, N,
        stride_out_batch, stride_out_m, stride_out_n,
        stride_left_batch, stride_left_m, stride_left_k,
        stride_right_batch, stride_right_n, stride_right_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # restore shape to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (both fp16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)      # (B,N,N,dim)

    # -----------------------------------------------------------------
    # 7. Cast back to float32 and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
