```python
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_n, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)  # batch*head index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute mask
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute offsets for k slice
        k_base = k * BLOCK_K
        cur_k = k_base + offs_k
        mask_k_cur = cur_k < N

        # pointers
        left_offset = pid_b * stride_left_batch + (offs_m[:, None] * stride_left_m) + (cur_k[None, :] * stride_left_k)
        right_offset = pid_b * stride_right_batch + (cur_k[:, None] * stride_right_k) + (offs_n[None, :] * stride_right_n)
        # load
        a = tl.load(left_ptr + left_offset, mask=mask_m[:, None] & mask_k_cur[None, :], other=0.0)
        b = tl.load(right_ptr + right_offset, mask=mask_k_cur[:, None] & mask_n[None, :], other=0.0)
        # accumulate
        acc += tl.dot(a, b)
    # write back
    out_offset = pid_b * stride_out_batch + (offs_m[:, None] * stride_out_m) + (offs_n[None, :] * stride_out_n)
    tl.store(out_ptr + out_offset, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul outgoing kernel: Implements the forward pass of the outgoing Triangle
Multiplicative Update from AlphaFold3. The core reduction
out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] is executed with a
custom Triton kernel (batched FP16 GEMM). All other operations (LayerNorm,
linear projections, gating, optional masking) are performed with
torch.nn.functional for simplicity. The kernel uses fp16 arithmetic with
fp32 accumulation for numerical stability and returns the result in the
input dtype (typically float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    # strides
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_n, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    N,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    batch_head_id = tl.program_id(0)  # batch * hidden
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < N

        # compute pointers
        # left tile: (M, K)
        left_offset = batch_head_id * stride_left_batch + (offs_m[:, None] * stride_left_m) + (cur_k[None, :] * stride_left_k)
        # right tile (transposed): we need right_T[k, n] = right[batch_head_id, n, k]
        right_offset = batch_head_id * stride_right_batch + (cur_k[:, None] * stride_right_k) + (offs_n[None, :] * stride_right_n)

        # load with masks
        a = tl.load(left_ptr + left_offset, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(right_ptr + right_offset, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # write back
    out_offset = batch_head_id * stride_out_batch + (offs_m[:, None] * stride_out_m) + (offs_n[None, :] * stride_out_n)
    tl.store(out_ptr + out_offset, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] (bool/int/float) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # 1. Input layer norm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape (B, N, N, dim)

    # 2. Cast to fp16 for compute intensive parts
    x = x.to(torch.float16)

    # 3. Cast linear weights to fp16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # 4. Linear projections
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # 5. Gates
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # 6. Apply gating (and optional mask)
    left = left_proj * left_gate
    right = right_proj * right_gate

    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16)
        left = left * mask_fp16[..., None]
        right = right * mask_fp16[..., None]

    # 7. Core TriMul reduction via Triton
    B, N_seq, _, H = left.shape  # N_seq == N
    assert N_seq == N  # reuse variable N
    # reshape to (B, H, N, N) then flatten batch*head
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    # flatten batch*head dimension
    left_flat = left.view(B * H, N, N)
    right_flat = right.view(B * H, N, N)
    out_flat = torch.empty_like(left_flat)

    # strides (in elements)
    stride_left_batch, stride_left_m, stride_left_k = left_flat.stride()
    stride_right_batch, stride_right_n, stride_right_k = right_flat.stride()
    stride_out_batch, stride_out_m, stride_out_n = out_flat.stride()

    # block sizes (tuned for typical B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        stride_left_batch,
        stride_left_m,
        stride_left_k,
        stride_right_batch,
        stride_right_n,
        stride_right_k,
        stride_out_batch,
        stride_out_m,
        stride_out_n,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 8. Output layer norm, gating and final linear projection
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)

    # 9. Cast back to original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
