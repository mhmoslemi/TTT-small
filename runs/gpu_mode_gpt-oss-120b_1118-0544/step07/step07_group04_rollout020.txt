def trilmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # each program computes a tile of out (i_range, j_range) for a particular batch and hidden.
    b = tl.program_id(0) // (H)
    h = tl.program_id(0) % H

    # compute tile offsets for i and j.
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    # allocate accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K

        # Load blocks from left and right.
        # left: shape (B,N,N,H) but we only need left[b,i,k:h]. left strides: (stride_left_b, stride_left_i, stride_left_k, stride_left_h)
        # left block: (BLOCK_M, BLOCK_K)
        left_offset = (
            b * stride_left_b
            + i_offset * stride_left_i
            + k_offset * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(left_ptr + left_offset, mask=... )  # need mask for block bounds
    
        # right block: we need right[b,j,k,h], but note we need right[b,j,k,h] (j as row) and k as col.
        # then need to transpose on k dimension? We need right[b,j,k,h] for multiplication: left[i,k]*right[j,k] -> we want right block (BLOCK_N, BLOCK_K) indexed by j,k. So we can load right similarly.

@triton.jit
def _tri_mul_outgoing_kernel_impl(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)
    pid2 = tl.program_id(2)
    # Decode batch and hidden
    b = pid0 // H
    h = pid0 % H

    i_start = pid1 * BLOCK_M
    j_start = pid2 * BLOCK_N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_iter * BLOCK_K

        # Compute indices for left: i, k
        i_idx = i_start + tl.arange(0, BLOCK_M)[:, None]  # shape (BLOCK_M,1)
        k_idx = k_start + tl.arange(0, BLOCK_K)[None, :]  # shape (1,BLOCK_K)
        # mask for left
        mask_i = i_idx < N
        mask_k = k_idx < N
        mask_left = mask_i & mask_k

        # Load left block
        left_offset = (
            b * stride_left_b
            + i_idx * stride_left_i
            + k_idx * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(left_ptr + left_offset, mask=mask_left, other=0.0)

        # Compute indices for right transposed: k, j
        j_idx = j_start + tl.arange(0, BLOCK_N)[None, :]  # shape (1,BLOCK_N)
        # mask for right
        mask_j = j_idx < N
        mask_right = mask_k & mask_j

        right_offset = (
            b * stride_right_b
            + k_idx * stride_right_k
            + j_idx * stride_right_j
            + h * stride_right_h
        )
        right = tl.load(right_ptr + right_offset, mask=mask_right, other=0.0)

        # Dot product
        acc += tl.dot(left, right)  # left (M,K), right (K,N) -> (M,N)

    # Store result
    i_idx_out = i_start + tl.arange(0, BLOCK_M)[:, None]
    j_idx_out = j_start + tl.arange(0, BLOCK_N)[None, :]
    mask_i_out = i_idx_out < N
    mask_j_out = j_idx_out < N
    mask_out = mask_i_out & mask_j_out

    out_offset = (
        b * stride_out_b
        + i_idx_out * stride_out_i
        + j_idx_out * stride_out_j
        + h * stride_out_h
    )
    tl.store(out_ptr + out_offset, acc.to(tl.float16), mask=mask_out)

"""
TriMul (outgoing) – Triton‑accelerated core reduction.
Algorithm
---------
1. LayerNorm over the channel dimension (float32 for stability).
2. Cast to fp16, compute linear projections and gate values (no bias).
3. Fuse projection, gate and optional mask with a tiny Triton kernel.
4. Compute the O(N³) reduction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a custom Triton batched GEMM kernel (no extra transposes).
5. Output LayerNorm, out‑gate multiplication and final linear projection.
All heavy compute is in fp16; final result is cast back to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse projection, gate and mask (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,          # fp16 *
    gate_ptr,          # fp16 *
    mask_ptr,          # fp16 *
    out_ptr,           # fp16 *
    total_elems,       # i64
    hidden_dim,        # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # mask index (broadcast over hidden dimension)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(out_ptr + offs, proj * gate * m, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 2048) -> torch.Tensor:
    """
    Element‑wise: proj * gate * mask   (mask broadcast over hidden_dim)
    All tensors must be fp16 and contiguous.
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the core TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel_impl(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    out_ptr,           # fp16 *
    B, N, H,           # i64
    # strides for left   (b,i,k,h)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right  (b,j,k,h)
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides for out    (b,i,j,h)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)   # batch * hidden
    pid1 = tl.program_id(1)   # i tiles
    pid2 = tl.program_id(2)   # j tiles

    # decode batch and hidden dimension
    b = pid0 // H
    h = pid0 % H

    i_start = pid1 * BLOCK_M
    j_start = pid2 * BLOCK_N

    # accumulator in fp32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_iter * BLOCK_K

        # ------------------------------------------------------------------
        # Load left block   (i,k)
        # ------------------------------------------------------------------
        offs_i = i_start + tl.arange(0, BLOCK_M)
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # shape (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        mask_i = offs_i[:, None] < N
        mask_k = offs_k[None, :] < N
        left = tl.load(left_ptrs, mask=mask_i & mask_k, other=0.0)

        # ------------------------------------------------------------------
        # Load right block (k,j)   – transposed view of right[b,j,k,h]
        # ------------------------------------------------------------------
        offs_j = j_start + tl.arange(0, BLOCK_N)

        # shape (BLOCK_K, BLOCK_N)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
        )
        mask_j = offs_j[None, :] < N
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j, other=0.0)

        # ------------------------------------------------------------------
        # Dot product and accumulate
        # ------------------------------------------------------------------
        acc += tl.dot(left, right)   # (M,K)·(K,N) -> (M,N)

    # ----------------------------------------------------------------------
    # Store the result (cast back to fp16)
    # ----------------------------------------------------------------------
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    mask_store = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_store)


def _tri_mul_outgoing_kernel(left: torch.Tensor,
                             right: torch.Tensor,
                             block_m: int = 64,
                             block_n: int = 64,
                             block_k: int = 64) -> torch.Tensor:
    """
    Batched GEMM kernel implementing
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    left/right : fp16 tensors of shape (B, N, N, H) – contiguous.
    Returns fp16 tensor of the same shape.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape

    out = torch.empty_like(left)

    # strides for contiguous tensor (b,i,k,h) etc.
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    grid = (
        B * H,                     # program id 0  (batch * hidden)
        triton.cdiv(N, block_m),   # program id 1  (i tiles)
        triton.cdiv(N, block_n),   # program id 2  (j tiles)
    )

    _tri_mul_outgoing_kernel_impl[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=block_m, BLOCK_N=block_n, BLOCK_K=block_k,
    )
    return out


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections and gates (bias‑free)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4. Fuse projection, gate and optional mask
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Core reduction – batched GEMM via Triton
    # ------------------------------------------------------------------
    out = _tri_mul_outgoing_kernel(left, right,
                                   block_m=64, block_n=64, block_k=64)   # (B,N,N,H), fp16

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    # Cast to float32 for the layer‑norm (more stable)
    out = out.to(torch.float32)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply out‑gate (broadcasted)
    out = out * out_gate.to(out.dtype)

    # Final linear back to original dim
    out = F.linear(out, weights["to_out.weight"].to(out.dtype))

    # ------------------------------------------------------------------
    # 7. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
