"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor (float16 for speed, weight/bias cast to fp16).
2. Linear projections (left/right) from dim → hidden_dim (fp16 GEMM via torch.nn.functional.linear).
3. Compute three sigmoid gates (left, right, out) with linear layers.
4. Fuse mask and gate multiplications in a single Triton kernel:
       left  = left  * mask * left_gate
       right = right * mask * right_gate
   (In‑place, no extra temporaries; mask is broadcast over the hidden dimension.)
5. Core reduction: for each head `h` perform a batched matrix multiplication
       out_h = left_h @ right_hᵀ   (shape (B, hidden_dim, N, N) → (B, N, N, hidden_dim))
   Implemented with torch.matmul (cuBLAS) which is highly optimized for fp16.
6. Apply a second LayerNorm over the hidden dimension.
7. Multiply by the output gate (element‑wise).
8. Final linear projection hidden_dim → dim.
9. Cast the result back to the original dtype of the input.

Only the mask‑gate fusion is written in Triton; all other heavy
operations are delegated to cuBLAS / PyTorch for maximum throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (flattened B*N*N*H)
    right_ptr,         # fp16 * right (flattened B*N*N*H)
    mask_ptr,          # fp16 * mask  (flattened B*N*N)
    left_gate_ptr,     # fp16 * left_gate  (flattened B*N*N*H)
    right_gate_ptr,    # fp16 * right_gate (flattened B*N*N*H)
    H: tl.constexpr,   # hidden dimension (compile‑time constant)
    total_elements: tl.int32,   # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """
    In‑place mask & gate fusion:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    The mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load data
    l  = tl.load(left_ptr + offs, mask=mask)
    r  = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Compute mask index (remove hidden dim)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply and store back
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see expected keys)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Cast all model weights to the compute dtype (fp16) and move to device
    # ------------------------------------------------------------------
    # LayerNorm weights
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float16)
    norm_bias   = weights["norm.bias"].to(device=device, dtype=torch.float16)

    # Linear projection weights (bias=False)
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    # Gate projection weights
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # Output norm and linear weights
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float16 compute)
    # ------------------------------------------------------------------
    # F.layer_norm expects weight/bias same dtype as input
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    # Work in fp16 for the heavy parts
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))    # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))   # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Fuse mask and gate multiplications via Triton (in‑place)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and broadcastable
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()          # B * N * N * H
        BLOCK_SIZE = 1024
        grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

        _mask_gate_fusion_inplace[grid](
            left_ptr=left,
            right_ptr=right,
            mask_ptr=mask_fp16.view(-1),
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            H=hidden_dim,
            total_elements=total_elements,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=4,
        )
        # Gates are no longer needed
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5) Core reduction: batched matrix multiplication (N³·H)
    #    left and right are (B, N, N, H) → reshape to (B, H, N, N)
    # ------------------------------------------------------------------
    left_b  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # out_b shape: (B, H, N, N) = left_b @ right_bᵀ
    out_b = torch.matmul(left_b, right_b.transpose(-2, -1))

    # Back to (B, N, N, H)
    out = out_b.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7) Apply output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 9) Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
```
