def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # 1. LayerNorm on last dim
    x = F.layer_norm(input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device, dtype=torch.float32))
    # cast to float16 for compute
    x = x.to(torch.float16)

    # 2. Concatenate projection and gate weights
    # shape: (5*hidden_dim, dim)
    left_proj_w = weights["left_proj.weight"]  # shape (hidden, dim)
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    # stack vertically
    weight_concat = torch.cat([
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w,
        out_gate_w
    ], dim=0).to(device, dtype=torch.float16)

    # Flatten x to (B*N*N, dim)
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)  # (B*N*N, dim)
    # Linear transform
    y = torch.nn.functional.linear(x_flat, weight_concat)  # (B*N*N, 5*hidden_dim)
    y = y.view(B, N, N, 5 * hidden_dim)

    # Slice
    left = y[..., :hidden_dim]
    right = y[..., hidden_dim:2*hidden_dim]
    left_gate = torch.sigmoid(y[..., 2*hidden_dim:3*hidden_dim])
    right_gate = torch.sigmoid(y[..., 3*hidden_dim:4*hidden_dim])
    out_gate = torch.sigmoid(y[..., 4*hidden_dim:5*hidden_dim])

    # Mask handling
    if mask is not None and not nomask:
        mask_fp16 = mask.to(dtype=torch.float16, device=device)
    else:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    mask_fp16 = mask_fp16.unsqueeze(-1)  # (B,N,N,1)

    # Fuse mask and gates using Triton
    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    # Launch kernel if mask needed, else we can compute directly
    total_elements = left.numel()
    BLOCK_SIZE = 8192  # 8k loops
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    @triton.jit
    def mask_gate_fusion_kernel(
        left_ptr, right_ptr, mask_ptr,
        left_gate_ptr, right_gate_ptr,
        left_out_ptr, right_out_ptr,
        H, total_elements, BLOCK_SIZE):
        pid = tl.program_id(0)
        offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
        mask = offs < total_elements

        l = tl.load(left_ptr + offs, mask=mask)
        r = tl.load(right_ptr + offs, mask=mask)
        lg = tl.load(left_gate_ptr + offs, mask=mask)
        rg = tl.load(right_gate_ptr + offs, mask=mask)

        mask_idx = offs // H  # because mask has shape B*N*N
        m = tl.load(mask_ptr + mask_idx, mask=mask)

        tl.store(left_out_ptr + offs, l * m * lg, mask=mask)
        tl.store(right_out_ptr + offs, r * m * rg, mask=mask)

    mask_gate_fusion_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask_fp16.view(-1),
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        left_out_ptr=left_out,
        right_out_ptr=right_out,
        H=hidden_dim,
        total_elements=total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
        )

    # 5. Core reduction using batched matmul
    # left_out: (B,N,N,H) ; right_out: (B,N,N,H)
    # permute for matmul: (B,H,N,N) and (B,H,N,N) with right transposed
    left_perm = left_out.permute(0, 3, 1, 2)  # (B,H,N,N)
    right_perm = right_out.permute(0, 3, 2, 1)  # (B,H,N,N)
    out_perm = torch.matmul(left_perm, right_perm)  # (B,H,N,N)
    out = out_perm.permute(0, 2, 3, 1)  # (B,N,N,H)

    # 6. Second LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device, dtype=torch.float16),
    )

    # 7. Apply out_gate
    out = out * out_gate

    # 8. Final projection
    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # Cast back to original dtype
    out = out.to(orig_dtype)

    return out

"""
Fast Triton implementation of the outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor (fp32 for stability).
2. Cast to fp16 and compute all five linear projections (left, right,
   left_gate, right_gate, out_gate) in a single GEMM by stacking the
   weights, then split the result.
3. If a mask is present, fuse mask, left‑gate and right‑gate with a
   Triton kernel:
       left_out  = left  * mask * left_gate
       right_out = right * mask * right_gate
   Otherwise perform the element‑wise multiplication directly.
4. Core reduction:
       out[b,i,j,h] = Σ_k left_out[b,i,k,h] * right_out[b,j,k,h]
   Implemented as a batched matrix‑multiply using torch.matmul on the
   (B, H, N, N) view (no extra copies).
5. Apply a second LayerNorm over the hidden dimension.
6. Multiply by the output gate and project back to the original channel
   dimension.
7. Cast the result back to the original dtype.

Only the mask‑gate fusion is performed in Triton; the heavy linear
projections and the batched matrix multiplication use cuBLAS via
torch.nn.functional.linear / torch.matmul, which is optimal on the B200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # const * left  (B,N,N,H) flattened
    right_ptr,         # const * right (B,N,N,H) flattened
    mask_ptr,          # const * mask  (B,N,N)   flattened
    left_gate_ptr,     # const * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # const * right_gate (B,N,N,H) flattened
    left_out_ptr,      # * left_out  (B,N,N,H) flattened
    right_out_ptr,     # * right_out (B,N,N,H) flattened
    H: tl.constexpr,                # hidden dimension
    total_elements: tl.constexpr,   # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask and gates:
       left_out  = left  * mask * left_gate
       right_out = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load data
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Mask is broadcast over the hidden dimension
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Write fused results
    tl.store(left_out_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, if True ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to fp16 for the remaining computation
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Single fused linear projection for all gates/projections
    # ------------------------------------------------------------------
    # Stack all projection/gate weights (shape: (5*H, dim))
    weight_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)

    B, N, _, _ = x.shape
    # Flatten to (B*N*N, dim) and apply the fused linear
    x_flat = x.view(-1, dim)                              # (B*N*N, dim)
    y = F.linear(x_flat, weight_concat)                   # (B*N*N, 5*H)
    y = y.view(B, N, N, 5 * hidden_dim)                  # (B, N, N, 5*H)

    # ------------------------------------------------------------------
    # 3) Split projections and gates
    # ------------------------------------------------------------------
    left = y[..., :hidden_dim]
    right = y[..., hidden_dim : 2 * hidden_dim]
    left_gate = torch.sigmoid(y[..., 2 * hidden_dim : 3 * hidden_dim])
    right_gate = torch.sigmoid(y[..., 3 * hidden_dim : 4 * hidden_dim])
    out_gate = torch.sigmoid(y[..., 4 * hidden_dim :])

    # Make tensors contiguous for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # ------------------------------------------------------------------
    # 4) Fuse mask with gates (Triton) or simple elementwise if no mask
    # ------------------------------------------------------------------
    if mask is None or nomask:
        # No mask ⇒ just multiply by gates
        left_out = left * left_gate
        right_out = right * right_gate
    else:
        # Prepare mask (broadcast over hidden dimension)
        mask_fp16 = mask.to(dtype=torch.float16, device=device)
        mask_fp16 = mask_fp16.view(-1)                     # (B*N*N,)

        left_out = torch.empty_like(left)
        right_out = torch.empty_like(right)

        total_elements = left.numel()                     # B*N*N*H
        BLOCK_SIZE = 8192  # multiple of 32; adjusts occupancy
        grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            mask_ptr=mask_fp16,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            left_out_ptr=left_out,
            right_out_ptr=right_out,
            H=hidden_dim,
            total_elements=total_elements,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=4,
        )

    # ------------------------------------------------------------------
    # 5) Core reduction: batched GEMM (left_out @ right_outᵀ) per hidden channel
    # ------------------------------------------------------------------
    # Re‑interpret tensors as (B, H, N, N) for batched matmul
    left_perm = left_out.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right_out.permute(0, 3, 2, 1)  # (B, H, N, N)  (transpose k↔j)
    out_perm = torch.matmul(left_perm, right_perm)  # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1)               # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"]
        .to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"]
        .to(device=device, dtype=torch.float16),
    )

    # ------------------------------------------------------------------
    # 7) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"]
        .to(device=device, dtype=torch.float16),
    )  # (B, N, N, C)

    # ------------------------------------------------------------------
    # 9) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
