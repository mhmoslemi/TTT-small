"""
custom_kernel: Triton‑accelerated outgoing TriMul (Triangle Multiplicative Update).

Algorithm
---------
1. LayerNorm on the input tensor (float32).
2. Cast to fp16 and compute linear projections and sigmoid gates with PyTorch
   (no bias).  Shapes after projection: [B, N, N, H] where H = hidden_dim.
3. Fuse mask (if present), element‑wise gates and the core contraction
   Σₖ left[...,i,k,h] * right[...,j,k,h] in a single Triton kernel.
   The kernel performs a blocked batched GEMM (left @ rightᵀ) for each
   (batch, head) pair using fp16 values and a float32 accumulator.
4. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate and a final linear projection back to the original dimension.
5. Cast back to the input dtype and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_gemm_kernel(
    # pointers
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr, out_ptr,
    # sizes
    B, N, H,
    # strides for left/right (B,N,N,H)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    # strides for gates (same layout as left/right)
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    # strides for mask (B,N,N)
    stride_m_b, stride_m_i, stride_m_j,
    # strides for out (B,N,N,H)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time constants
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr
):
    """Batched GEMM with fused mask and gates.

    Computes for each batch b and head h:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * left_gate[b,i,k,h] *
                       right[b,j,k,h] * right_gate[b,j,k,h] *
                       (mask[b,i,k] if HAS_MASK else 1) *
                       (mask[b,j,k] if HAS_MASK else 1)
    """
    pid_bh = tl.program_id(2)               # index over B*H
    pid_m = tl.program_id(0)                 # tile over M (i)
    pid_n = tl.program_id(1)                 # tile over N (j)

    # decode batch and head indices
    b = pid_bh // H
    h = pid_bh % H

    # base offsets for this (b, h)
    base_l  = b * stride_l_b  + h * stride_l_h
    base_r  = b * stride_r_b  + h * stride_r_h
    base_lg = b * stride_lg_b + h * stride_lg_h
    base_rg = b * stride_rg_b + h * stride_rg_h
    base_o  = b * stride_o_b  + h * stride_o_h

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left block (M, K) and its gate -----
        left = tl.load(
            left_ptr + base_l +
            offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0)
        left_gate = tl.load(
            left_gate_ptr + base_lg +
            offs_m[:, None] * stride_lg_i + offs_k[None, :] * stride_lg_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0)

        # apply mask and gate to left
        if HAS_MASK:
            mask_l = tl.load(
                mask_ptr + b * stride_m_b +
                offs_m[:, None] * stride_m_i + offs_k[None, :] * stride_m_j,
                mask=mask_m[:, None] & mask_k[None, :],
                other=1.0)
            left = left * left_gate * mask_l
        else:
            left = left * left_gate

        # ----- load right block (K, N) and its gate -----
        right = tl.load(
            right_ptr + base_r +
            offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_i,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0)
        right_gate = tl.load(
            right_gate_ptr + base_rg +
            offs_k[:, None] * stride_rg_k + offs_n[None, :] * stride_rg_i,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0)

        # apply mask and gate to right
        if HAS_MASK:
            mask_r = tl.load(
                mask_ptr + b * stride_m_b +
                offs_n[:, None] * stride_m_i + offs_k[None, :] * stride_m_j,
                mask=mask_n[:, None] & mask_k[None, :],
                other=1.0)
            right = right * right_gate * mask_r
        else:
            right = right * right_gate

        # accumulate dot product (M,K) x (K,N) -> (M,N)
        acc += tl.dot(left, right)

    # store result (convert accumulator to fp16)
    out = acc.to(tl.float16)
    tl.store(
        out_ptr + base_o +
        offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j,
        out,
        mask=mask_m[:, None] & mask_n[None, :]
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, torch.float32)
    norm_b = weights["norm.bias"].to(device, torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,C)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (bias‑less) and gates (sigmoid)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device, torch.float16)

    left = F.linear(x, left_proj_w)               # (B,N,N,H)
    right = F.linear(x, right_proj_w)             # (B,N,N,H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,H)

    # Ensure contiguity
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    out_gate = out_gate.contiguous()

    # ------------------------------------------------------------------
    # 4. Core contraction with fused mask & gates (Triton)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # Prepare mask
    has_mask = (mask is not None) and (not nomask)
    if has_mask:
        mask_fp = mask.to(device, torch.float16).contiguous()
    else:
        # dummy mask (won't be accessed because HAS_MASK=False)
        mask_fp = torch.empty(0, dtype=torch.float16, device=device)

    out = torch.empty_like(left)  # (B,N,N,H)

    # Strides (int64)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h = left_gate.stride()
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h = right_gate.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    if has_mask:
        stride_m_b, stride_m_i, stride_m_j = mask_fp.stride()
    else:
        # dummy strides (won't be used)
        stride_m_b = stride_m_i = stride_m_j = 0

    # Block sizes – tuned for typical B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    _trimul_gemm_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        left_gate_ptr=left_gate.data_ptr(),
        right_gate_ptr=right_gate.data_ptr(),
        mask_ptr=mask_fp.data_ptr(),
        out_ptr=out.data_ptr(),
        B=B,
        N=N,
        H=H,
        stride_l_b=stride_l_b,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_l_h=stride_l_h,
        stride_r_b=stride_r_b,
        stride_r_i=stride_r_i,
        stride_r_k=stride_r_k,
        stride_r_h=stride_r_h,
        stride_lg_b=stride_lg_b,
        stride_lg_i=stride_lg_i,
        stride_lg_k=stride_lg_k,
        stride_lg_h=stride_lg_h,
        stride_rg_b=stride_rg_b,
        stride_rg_i=stride_rg_i,
        stride_rg_k=stride_rg_k,
        stride_rg_h=stride_rg_h,
        stride_m_b=stride_m_b,
        stride_m_i=stride_m_i,
        stride_m_j=stride_m_j,
        stride_o_b=stride_o_b,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        stride_o_h=stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HAS_MASK=int(has_mask),
    )

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, gating, and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (both fp16)

    to_out_w = weights["to_out.weight"].to(device, torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------
    # 6. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
