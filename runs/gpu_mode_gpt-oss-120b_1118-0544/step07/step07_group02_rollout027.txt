def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Norm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Cast to FP16
    x = x.to(torch.float16)

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    # Apply mask & gates
    if mask is not None and not nomask:
        # flatten and use triton kernel
        mask_fp = mask.to(torch.float16).contiguous()
        total = left.numel()
        # flatten all tensors
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_fp.view(-1)  # shape (B,N,N)
        # launch kernel
        BLOCK = 16384
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # Core contraction
    # Use batched matrix multiplication per head
    B, N, _, H = left.shape
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)

    # Output layernorm

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm Overview
------------------
1. **Input LayerNorm** (float32 for stability).
2. Cast to fp16 for the compute‑heavy path.
3. Linear projections (`left_proj`, `right_proj`) and three sigmoid gates
   (`left_gate`, `right_gate`, `out_gate`).
4. Fuse mask (if present) together with the left/right gates in a small
   Triton kernel that works in‑place on the flattened tensors.
5. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM (L @ Rᵀ) for every hidden channel.
6. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate, and a final linear projection back to the original dimension.
7. Cast the result to float32 and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask and element‑wise gates in‑place.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,                 # fp16*  [B,N,N,H] flattened
    right_ptr,                # fp16*  same shape
    left_gate_ptr,            # fp16*  same shape
    right_gate_ptr,           # fp16*  same shape
    mask_ptr,                 # fp16*  [B,N,N] flattened, broadcast over H
    total_elems, hidden_dim,  # runtime scalars
    HAS_MASK: tl.constexpr,   # compile‑time bool
    BLOCK_SIZE: tl.constexpr, # compile‑time int
):
    """In‑place:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
       If HAS_MASK is False the mask is ignored (treated as 1).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load values
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each (b,i,j) entry is shared across `hidden_dim` consecutive
        # elements in the flattened view
        mask_idx = offs // hidden_dim          # index in [B,N,N]
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store the fused results back
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
              required keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with keys:
                * "dim" (int)          – input channel dimension C
                * "hidden_dim" (int)   – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with dtype torch.float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy part
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections and gates (fp16)
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)                       # (B,N,N,H)
    right = F.linear(x, right_proj_w)                     # (B,N,N,H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,H)

    # --------------------------------------------------------------
    # 4. Fuse mask and gates (Triton)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Convert mask to fp16 and ensure contiguous layout
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # Flatten everything for the kernel
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 16384
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back to original tensors
        left  = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # Simple element‑wise multiplication when no mask is needed
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM (L @ Rᵀ) per hidden channel.
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # (B*H, N, N) matrices
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate                         # apply output gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                # (B,N,N,dim)

    # --------------------------------------------------------------
    # 7. Cast back to float32 and return
    # --------------------------------------------------------------
    return out.to(torch.float32)
