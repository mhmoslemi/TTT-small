    stride_right_b, stride_right_h, stride_right_i, stride_right_j,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(2)  # batch index (0..B*H-1)
    pid_i = tl.program_id(0)  # row block
    pid_j = tl.program_id(1)  # col block

    # compute batch offsets
    # compute b and h indices
    b = pid_bh // H
    h = pid_bh % H

    # compute start indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # create ranges
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    # mask for i, j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # offsets for output
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + \
               (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)

    # loop over k
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # load left tile (M,K)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # keep same dtype

    # Cast to fp16
    x = x.to(torch.float16)

    # Convert weights to fp16 and no bias
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # Mask
    if mask is not None and not nomask:
        # Convert mask to float16 and broadcast
        mask_fp = mask.to(torch.float16)
        mask_fp = mask_fp.unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * mask_fp
        right = right_proj * mask_fp
    else:
        left = left_proj
        right = right_proj

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # TriMul reduction using Triton
    B, N, _, H = left.shape

    # Permute to (B, H, N, N) and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    out_perm = torch.empty_like(left_perm)

    # Choose block sizes (tune for performance)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * H )

    triton_tri_mul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B,
        H,
        N,
        left_perm.stride(0), left_perm.stride(1), left_perm.stride(2), left_perm.stride(3),
        right_perm.stride(0), right_perm.stride(1), right_perm.stride(2), right_perm.stride(3),
        out_perm.stride(0), out_perm.stride(1), out_perm.stride(2), out_perm.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Permute back to (B,N,N,H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to original dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_h, stride_left_i, stride_left_j,
    stride_right_b, stride_right_h, stride_right_i, stride_right_j,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    b = pid_bh // H
    h = pid_bh % H

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # output pointer
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + \
               offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k0 = k_iter * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile (M, K)
        a_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + \
                 offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_j
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile (K, N) transposed
        b_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + \

        b_tile = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b_tile)

    # store result
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
Optimized outgoing TriMul forward pass (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to float16 for the heavy compute.
3. Linear projections (left / right) and three gate projections (left,
   right, out) – all in FP16.
4. If a mask is supplied and `nomask` is False, apply it to the
   projected tensors before gating (element‑wise).
5. Core O(N³) reduction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a custom Triton kernel that performs a batched
   matrix multiplication for each (batch, hidden) pair using block‑wise
   tiling (BLOCK_M×BLOCK_N tiles, BLOCK_K reduction blocks) with
   fp32 accumulation.
6. Output LayerNorm, multiply by the out‑gate, and a final linear
   projection back to the original channel dimension.
7. Cast the result back to the input dtype.

The kernel handles arbitrary batch sizes (B∈{1,2}), sequence lengths
(N≤1024) and hidden dimensions (H) while remaining fully
GPU‑resident.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def triton_tri_mul_kernel(
    left_ptr,            # fp16*  (B, H, N, N)
    right_ptr,           # fp16*  (B, H, N, N)
    out_ptr,             # fp16*  (B, H, N, N)
    B, H, N,             # scalar int64
    stride_l_b, stride_l_h, stride_l_i, stride_l_j,
    stride_r_b, stride_r_h, stride_r_i, stride_r_j,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_i = tl.program_id(0)          # row block index
    pid_j = tl.program_id(1)          # column block index
    pid_bh = tl.program_id(2)         # combined batch*hidden index

    # decode batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # block start coordinates
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # ranges within the block
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers to the output tile
    out_ptrs = out_ptr + b * stride_o_b + h * stride_o_h + \
               offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑blocks
    num_k = tl.cdiv(N, BLOCK_K)

    for k_iter in range(num_k):
        k0 = k_iter * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + b * stride_l_b + h * stride_l_h + \
                 offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_j
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right tile (transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + b * stride_r_b + h * stride_r_h + \
                 offs_k[:, None] * stride_r_j + offs_j[None, :] * stride_r_i
        b_tile = tl.load(b_ptrs,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0).to(tl.float32)

        # accumulate the product
        acc += tl.dot(a, b_tile)

    # write back the result
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast linear weights to FP16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections (FP16)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # [B, N, N, H]
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5) Apply mask (if present) before gating, exactly as in the reference
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left_proj * mask_fp
        right = right_proj * mask_fp
    else:
        left = left_proj
        right = right_proj

    # --------------------------------------------------------------
    # 6) Gate projections and sigmoid
    # --------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 7) Core TriMul reduction via Triton (batched GEMM)
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # reshape to (B, H, N, N) for the kernel and make contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    out_perm   = torch.empty_like(left_perm)

    # block sizes (tuned for typical GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # rows
        (N + BLOCK_N - 1) // BLOCK_N,          # cols
        B * H,                                 # batch*hidden
    )

    triton_tri_mul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B,
        H,
        N,
        left_perm.stride(0), left_perm.stride(1), left_perm.stride(2), left_perm.stride(3),
        right_perm.stride(0), right_perm.stride(1), right_perm.stride(2), right_perm.stride(3),
        out_perm.stride(0), out_perm.stride(1), out_perm.stride(2), out_perm.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # back to [B, N, N, dim]

    # --------------------------------------------------------------
    # 9) Cast back to the original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
