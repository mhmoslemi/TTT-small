```python
"""
TriMul outgoing forward pass with fused projection, gating and masking using a Triton kernel.
The core reduction is performed with a batched GEMM (torch.bmm) after reshaping the
[batch, seq, seq, hidden] tensors to (batch*hidden, seq, seq). This matches the
Einstein summation "... i k d, ... j k d -> ... i j d". The implementation
uses mixed‑precision (FP16) for heavy compute and casts back to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# Triton kernel: fuse linear projection results with gating and optional mask
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B*N*N*H,)
    right_proj_ptr,  # fp16 * (B*N*N*H,)
    left_gate_ptr,   # fp16 * (B*N*N*H,)
    right_gate_ptr,  # fp16 * (B*N*N*H,)
    mask_ptr,        # fp16 * (B*N*N,)
    left_out_ptr,    # fp16 * (B*N*N*H,)
    right_out_ptr,   # fp16 * (B*N*N*H,)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard
    mask = offs < total_elems

    # Each mask element repeats `hidden_dim` times in the flattened tensor
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gate and mask for both left and right tensors using a Triton kernel.
    All tensors must be fp16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)
    Returns:
        left, right : (B, N, N, H)  = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] (0/1) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over channel dimension `dim`
    # ------------------------------------------------------------------
    # Cast LayerNorm parameters to the same dtype as the input for efficiency
    norm_weight = weights["norm.weight"].to(dtype_in)
    norm_bias = weights["norm.bias"].to(dtype_in)

    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Load linear weights in FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = F.linear(x_fp16, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (fused via Triton if mask is used)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is half‑precision and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
            block_size=1024,
        )
    else:
        # No mask or mask ignored → simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction (TriMul):
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM over (i,k) and (j,k) per hidden slice.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) where rows are i (or j) and columns are k
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N).T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```

"""
TriMul outgoing forward pass with fused projection, gating and optional masking.
The mask/gate fusion is performed by a Triton kernel, while the O(N³) reduction
is executed as a batched GEMM (torch.bmm) after reshaping the tensors.
Mixed‑precision (FP16) is used for the heavy compute and the result is cast
back to the original input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and optional mask
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B*N*N*H,)
    right_proj_ptr,  # fp16 * (B*N*N*H,)
    left_gate_ptr,   # fp16 * (B*N*N*H,)
    right_gate_ptr,  # fp16 * (B*N*N*H,)
    mask_ptr,        # fp16 * (B*N*N,)
    left_out_ptr,    # fp16 * (B*N*N*H,)
    right_out_ptr,   # fp16 * (B*N*N*H,)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds accesses
    mask = offs < total_elems

    # Each mask element repeats `hidden_dim` times in the flattened tensor
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse linear projections, gates and mask for both left and right tensors
    using a Triton kernel.

    All tensors must be fp16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)

    Returns:
        left, right : (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] (0/1) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    dtype_in = input_tensor.dtype
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over channel dimension `dim`
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(dtype_in)
    norm_bias = weights["norm.bias"].to(dtype_in)

    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Load linear weights in FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x_fp16, left_proj_w)   # (B, N, N, hidden_dim)
    right_proj = F.linear(x_fp16, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
            block_size=1024,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) where rows are i/j and columns are k
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
