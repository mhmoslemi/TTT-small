@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2) # batch index
    pid_m = tl.program_id(0) # block row
    pid_n = tl.program_id(1) # block col

    # compute block start indices
    block_start_m = pid_m * BLOCK_M
    block_start_n = pid_n * BLOCK_N

    # create offsets
    offs_m = block_start_m + tl.arange(0, BLOCK_M)
    offs_n = block_start_n + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to start of block
    a_ptrs = A_ptr + pid_z * stride_az + offs_m[:,None] * stride_am + tl.arange(0, BLOCK_K)[None,:] * stride_ak
    b_ptrs = B_ptr + pid_z * stride_bz + tl.arange(0, BLOCK_K)[:,None] * stride_bk + offs_n[None,:] * stride_bn

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load A tile
        a = tl.load(a_ptrs, mask=mask_m[:,None] & (k*BLOCK_K+tl.arange(0, BLOCK_K) < K), other=0.0)
        # load B tile
        b = tl.load(b_ptrs, mask=(k*BLOCK_K+tl.arange(0, BLOCK_K) < K)[:,None] & mask_n[None,:], other=0.0)

        # compute dot product
        acc += tl.dot(a, b)

        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # cast to output dtype (FP16)
    c = acc.to(tl.float16)

    # store
    c_ptrs = C_ptr + pid_z * stride_cz + offs_m[:,None] * stride_cm + offs_n[None,:] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:,None] & mask_n[None,:])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The operator works on a tensor `x` of shape [B, N, N, C] and performs:
1. LayerNorm over the channel dimension C.
2. Linear projections to a hidden dimension H (left/right) and three gates.
3. Optional element‑wise mask on the projected tensors.
4. Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication for each hidden slice.
   Implemented with a custom Triton GEMM kernel (FP16 compute, FP32 accumulation).
5. LayerNorm over the hidden dimension, gating, and final linear projection back
   to the original dimension C.
All heavy compute is performed in FP16; the final result is returned in the
input dtype (typically float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton batched GEMM kernel (A @ B, where A: (batch,M,K), B: (batch,K,N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    stride_az, stride_am, stride_ak,         # strides for A
    stride_bz, stride_bk, stride_bn,         # strides for B
    stride_cz, stride_cm, stride_cn,         # strides for C
    M, N, K,                                 # problem size
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)                 # batch index
    pid_m = tl.program_id(1)                 # row block
    pid_n = tl.program_id(0)                 # column block

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tiles for this batch
    a_ptrs = a_ptr + pid_z * stride_az + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid_z * stride_bz + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K‑loop
    # --------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Store result
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_z * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel defined above.
    A: (batch, M, K), B: (batch, K, N)  (both contiguous, FP16)
    Returns C: (batch, M, N) in the same dtype as A.
    """
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Strides (in elements)
    stride_az, stride_am, stride_ak = A.stride()
    stride_bz, stride_bk, stride_bn = B.stride()
    stride_cz, stride_cm, stride_cn = C.stride()

    # Tunable block sizes (chosen for good occupancy on A200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_N),    # pid_n
        triton.cdiv(M, BLOCK_M),    # pid_m
        batch,                      # pid_z
    )

    _batched_gemm_kernel[grid](
        A,
        B,
        C,
        stride_az, stride_am, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_cm, stride_cn,
        M, N, K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

    # --------------------------------------------------------------
    # 4) Linear projections (FP16)
    # --------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)        # (B, N, N, H)

    # --------------------------------------------------------------
    # 5) Optional mask (apply before gating, exactly as reference code)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B, N, N) -> (B, N, N, 1)
        mask_fp = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_fp
        right_proj = right_proj * mask_fp

    # --------------------------------------------------------------
    # 6) Gates (sigmoid) and gating multiplication
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    left = left_proj * left_gate      # (B, N, N, H)
    right = right_proj * right_gate   # (B, N, N, H)

    # --------------------------------------------------------------
    # 7) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM over the hidden dimension.
    # --------------------------------------------------------------
    B, N, _, H = left.shape

    # Rearrange to (B, H, N, N) where the last two dims are the matrix axes
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()        # (B, H, N, N)

    # Transpose the second matrix on the k‑axis
    right_t = right_perm.transpose(-2, -1).contiguous()       # (B, H, N, N) == (batch, K, N)

    # Collapse batch and hidden dimensions
    batch = B * H
    left_flat = left_perm.view(batch, N, N)    # (batch, M=N, K=N)
    right_flat = right_t.view(batch, N, N)    # (batch, K=N, N)

    # Batched GEMM via Triton
    out_flat = _batched_matmul_triton(left_flat, right_flat)   # (batch, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9) Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
