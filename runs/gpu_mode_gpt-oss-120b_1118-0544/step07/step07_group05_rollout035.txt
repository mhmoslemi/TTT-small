import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    H: tl.constexpr, total: tl.constexpr, BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # loads
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # mask index per (b,i,j) element
    idx = offs // H
    m = tl.load(mask_ptr + idx, mask=mask)
    # apply
    tl.store(left_ptr + offs, l * lg * m, mask=mask)
    tl.store(right_ptr + offs, r * rg * m, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

    The implementation fuses the mask‑gate multiplication in a small Triton
    kernel and merges the left/right linear projections (and their gates) into
    two concatenated Linear layers to reduce memory traffic. The core
    contraction `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]` is
    performed with a single torch.matmul (cuBLAS) on tensors shaped
    (B, H, N, N). The result is layer‑normalized, gated and projected back
    to the original channel dimension.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. First LayerNorm (float32 for stability)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to fp16 for the compute‑heavy part
    x = x.to(torch.float16)

    # 2. Prepare weights as fp16 tensors
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # 3. Fuse left/right projections: (B,N,N,2*H)
    proj_weight = torch.cat([left_proj_w, right_proj_w], dim=0)  # (2*H, C)
    proj = F.linear(x, proj_weight)  # (B, N, N, 2*H)
    left = proj[..., :hidden_dim]
    right = proj[..., hidden_dim:]

    # 4. Fuse left/right gates and apply sigmoid
    gate_weight = torch.cat([left_gate_w, right_gate_w], dim=0)  # (2*H, C)
    gates = torch.sigmoid(F.linear(x, gate_weight))  # (B, N, N, 2*H)
    left_gate = gates[..., :hidden_dim]
    right_gate = gates[..., hidden_dim:]

    # 5. Output gate (separate)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B, N, N, H)

    # 6. Apply mask and gates
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates are no longer needed
        del left_gate, right_gate
    else:
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # 7. Core reduction: batched matrix multiplication
    #   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #   Implemented as torch.matmul on (B, H, N, N) tensors.
    left_b = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_b = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) -> transpose k and j
    out_b = torch.matmul(left_b, right_b)            # (B, H, N, N)

    # reshape back to (B, N, N, H)
    out = out_b.permute(0, 2, 3, 1).contiguous()

    # 8. Second LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 9. Apply output gate
    out = out * out_gate

    # 10. Final linear projection back to original dim
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # 11. Cast back to original dtype if needed
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out

    right_gate_ptr,         # fp16 * right_gate [B,N,N,H] flattened
    H: tl.constexpr,        # hidden dimension
    TOTAL: tl.constexpr,    # total number of elements in left/right tensors
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and sigmoid gates.
    left  = left  * left_gate  * mask
    right = right * right_gate * mask
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < TOTAL

    # load values
    l  = tl.load(left_ptr + offs,  mask=valid)
    r  = tl.load(right_ptr + offs, mask=valid)
    lg = tl.load(left_gate_ptr + offs,  mask=valid)
    rg = tl.load(right_gate_ptr + offs, mask=valid)

    # one mask value per (b,i,j) -> repeat H times
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=valid)

    # apply and store back
    tl.store(left_ptr + offs,  l * lg * m, mask=valid)
    tl.store(right_ptr + offs, r * rg * m, mask=valid)


def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

    The implementation fuses mask‑gate multiplication in a small Triton kernel
    and merges the left/right linear projections (and their gates) into two
    concatenated Linear layers to reduce memory traffic.  The core contraction

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    is performed with a single torch.matmul (cuBLAS) on tensors shaped
    (B, H, N, N).  The result is layer‑normalized, gated and projected back to
    the original channel dimension.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to fp16 for the compute‑heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast all model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Fuse left/right projections (2*H outputs)
    # ------------------------------------------------------------------
    proj_weight = torch.cat([left_proj_w, right_proj_w], dim=0)  # (2*H, C)
    proj = F.linear(x, proj_weight)  # (B, N, N, 2*H)
    left  = proj[..., :hidden_dim]   # (B, N, N, H)
    right = proj[..., hidden_dim:]   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Fuse left/right gates (2*H outputs) + sigmoid
    # ------------------------------------------------------------------
    gate_weight = torch.cat([left_gate_w, right_gate_w], dim=0)  # (2*H, C)
    gates = torch.sigmoid(F.linear(x, gate_weight))  # (B, N, N, 2*H)
    left_gate  = gates[..., :hidden_dim]   # (B, N, N, H)
    right_gate = gates[..., hidden_dim:]   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Output gate (single H)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Apply mask and gates (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total = left.numel()                     # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # gates are no longer needed
        del left_gate, right_gate
    else:
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction: batched matrix multiplication
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) for cuBLAS
    left_b  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_b = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) – transposed over k/j

    out_b = torch.matmul(left_b, right_b)              # (B, H, N, N)

    # back to (B, N, N, H)
    out = out_b.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 11) Cast back to the original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
