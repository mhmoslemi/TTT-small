@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_an,
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pb = tl.program_id(2)  # batch index

    # compute starting offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # each offsets may exceed bounds; we mask later

    # load a and b tiles
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # accumulate in FP32 for precision, but we can keep FP16

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        a_ptrs = a_ptr + pb * stride_am + (offs_m[:, None] * stride_ak) + (offs_k[None, :] * stride_an)
        b_ptrs = b_ptr + pb * stride_bm + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs, mask= (offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask= (offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        # dot product: a (BLOCK_M x BLOCK_K) * b (BLOCK_K x BLOCK_N)
        acc += tl.dot(a, b)

    # write results
    c_ptrs = c_ptr + pb * stride_cm + (offs_m[:, None] * stride_cn) + (offs_n[None, :] * stride_cb)
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

@triton.jit
def gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each mask element repeats hidden_dim times
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_an,
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    batch = tl.program_id(2)  # batch index

    # compute global offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute mask for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # accumulate in FP32 for better precision

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers
        a_ptrs = a_ptr + batch * stride_am + (offs_m[:, None] * stride_ak) + (offs_k[None, :] * stride_an)
        b_ptrs = b_ptr + batch * stride_bm + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        # load
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        # compute
        acc += tl.dot(a, b)

    # store
    c_ptrs = c_ptr + batch * stride_cm + (offs_m[:, None] * stride_cn) + (offs_n[None, :] * stride_cb)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Step 1: LayerNorm
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16
    x = x.to(torch.float16)

    # Linear projections
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = torch.nn.functional.linear(x, left_proj_w)  # shape B,N,N,H
    right = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply mask/gating
    if not nomask:
        # mask shape B,N,N (float)
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_fp.contiguous().view(-1)  # shape B*N*N
        total = left_f.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
        _gate_mask_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # Step 5: core contraction using batched GEMM kernel
    B, N, _, H = left.shape
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose right on last two dims
    right_t = right_perm.transpose(2, 3).contiguous()
    # flatten batch*H
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_t.view(B * H, N, N)
    # allocate output
    out_mat = torch.empty_like(left_mat)

    # compute strides
    stride_am = left_mat.stride(0)
    stride_ak = left_mat.stride(1)
    stride_an = left_mat.stride(2)

    stride_bm = right_mat.stride(0)
    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)

    stride_cm = out_mat.stride(0)
    stride_cn = out_mat.stride(1)
    stride_cb = out_mat.stride(2)

    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)
    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_am, stride_ak, stride_an,
        stride_bm, stride_bk, stride_bn,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # Output layernorm
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply out_gate
    out = out * out_gate

    # Final linear projection
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```

"""
custom_kernel: Triton‑accelerated outgoing TriMul (Triangle Multiplicative Update).

Algorithm
---------
1. LayerNorm over the input channel dimension (float32 for stability).
2. Cast to FP16 for the heavy compute.
3. Linear projections (bias‑less) → left, right and three gating vectors.
4. Fuse mask (if present) and gating with a tiny Triton kernel (in‑place).
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM (left @ rightᵀ) in a custom Triton kernel.
6. Output LayerNorm over the hidden dimension, element‑wise multiplication with
   the output gate, and a final linear projection back to the original dimension.
7. Cast back to the input dtype.

Both Triton kernels are deliberately lightweight: the first one fuses element‑wise
operations, the second one is a block‑wise matrix multiply that works on the
(batched, hidden) dimension in parallel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. Mask‑gate fusion kernel (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened, fp16
    left_gate_ptr, right_gate_ptr,     # same shape, fp16
    mask_ptr,                          # [B,N,N] flattened, fp16
    total_elems, hidden_dim,           # runtime scalars (int32)
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each mask entry repeats `hidden_dim` times in the flattened view
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


# ----------------------------------------------------------------------
# 2. Batched GEMM kernel for the core contraction
#    Computes C[b] = A[b] @ B[b]  with A,B of shape (M,K) and (K,N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_an,   # A: (batch, M, K)
    stride_bm, stride_bk, stride_bn,   # B: (batch, K, N)
    stride_cm, stride_cn, stride_cb,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    batch = tl.program_id(2)   # batch index

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator (use FP32 for better precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for the current tile
        a_ptrs = a_ptr + batch * stride_am + (offs_m[:, None] * stride_ak) + (offs_k[None, :] * stride_an)
        b_ptrs = b_ptr + batch * stride_bm + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        # Load tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute dot product (FP16 * FP16 -> FP32 accumulation)
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + batch * stride_cm + (offs_m[:, None] * stride_cn) + (offs_n[None, :] * stride_cb)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 3. Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) (may be dummy)
            - weights: dict of parameter tensors (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (bias‑less) + gates
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)          # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)        # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4. Fuse mask and gates (in‑place) via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # ensure mask is float16
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten tensors for the kernel
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)  # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024

        grid = ( (total + BLOCK - 1) // BLOCK, )
        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # simple element‑wise gating when there is no mask
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via a batched GEMM Triton kernel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # reshape to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    # transpose the last two dimensions of right (so we have K x N)
    right_t = right_perm.transpose(2, 3).contiguous()   # (B,H,N,N)

    # flatten batch*H dimension
    left_mat  = left_perm.view(B * H, N, N)   # (B*H, M=N, K=N)
    right_mat = right_t.view(B * H, N, N)    # (B*H, K=N, N)

    # allocate output
    out_mat = torch.empty_like(left_mat)

    # strides (in elements)
    stride_am = left_mat.stride(0)
    stride_ak = left_mat.stride(1)
    stride_an = left_mat.stride(2)

    stride_bm = right_mat.stride(0)
    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)

    stride_cm = out_mat.stride(0)
    stride_cn = out_mat.stride(1)
    stride_cb = out_mat.stride(2)

    # tile sizes – tuned for typical GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_am, stride_ak, stride_an,
        stride_bm, stride_bk, stride_bn,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------
    # 7. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
