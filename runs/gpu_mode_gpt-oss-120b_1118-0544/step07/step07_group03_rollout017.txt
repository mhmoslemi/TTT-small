Implementation of kernel:

```
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # Compute batch and block indices
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    num_blocks_per_batch = num_pid_m * num_pid_n
    batch = pid // num_blocks_per_batch
    block_id = pid % num_blocks_per_batch
    pid_m = block_id // num_pid_n
    pid_n = block_id % num_pid_n

    # Offsets at the start of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers for the start of the batch
    a_batch_ptr = a_ptr + batch * batch_stride_a
    b_batch_ptr = b_ptr + batch * batch_stride_b
    c_batch_ptr = c_ptr + batch * batch_stride_c

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # Load A and B blocks
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)
```

    # Projections
    left = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    # Apply mask if any.
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # Gates
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Now perform batched GEMM
    B, N, _, H = left.shape

    # Permute to (B,H,N,N)
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    # Transpose right N dims -> (B,H,N,N) -> swap axes 2 and 3
    right_T = right_perm.permute(0,1,3,2).contiguous()

    # Flatten batch and hidden
    left_flat = left_perm.view(-1, N, N)
    right_flat = right_T.view(-1, N, N)

    # Output placeholder
    out_flat = torch.empty_like(left_flat)  # same dtype (fp16)

    # Compute strides
    a_batch_stride = left_flat.stride(0)
    a_row_stride = left_flat.stride(1)
    a_col_stride = left_flat.stride(2)
    b_batch_stride = right_flat.stride(0)
    b_row_stride = right_flat.stride(1)
    b_col_stride = right_flat.stride(2)
    c_batch_stride = out_flat.stride(0)
    c_row_stride = out_flat.stride(1)
    c_col_stride = out_flat.stride(2)

    # Determine block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid size
    total_batches = left_flat.shape[0]
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (total_batches * num_pid_m * num_pid_n,)

    # Launch kernel
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        a_batch_stride,
        b_batch_stride,
        c_batch_stride,
        a_row_stride,
        a_col_stride,
        b_row_stride,
        b_col_stride,
        c_row_stride,
        c_col_stride,
        N,  # M
        N,  # N
        N,  # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape output back
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()  # shape (B,N,N,H)

    # Output layernorm
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, to_out_w)

    # Cast back to input dtype
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

"""
custom_kernel: Triton‑accelerated outgoing TriMul forward (AlphaFold3).

The operator works on a 4‑D tensor [B, N, N, C] and performs:
1️⃣ LayerNorm over the channel dimension C.
2️⃣ Linear projections (left/right) and gated sigmoids (left_gate/right_gate/out_gate).
3️⃣ Optional masking (broadcast over the hidden dimension).
4️⃣ Core “TriMul” reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication A @ Bᵀ for every hidden slice h.
   This is implemented with a custom Triton kernel that fuses all batch
   instances into a single launch, using fp16 compute and fp32 accumulation.
5️⃣ Output LayerNorm, gating, and final linear projection back to C.
The final tensor is returned in the original input dtype (usually float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    batch_stride_a, batch_stride_b, batch_stride_c,  # batch strides
    stride_am, stride_ak,               # A row / col strides
    stride_bk, stride_bn,               # B row / col strides
    stride_cm, stride_cn,               # C row / col strides
    M, N, K,                            # matrix dims
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ B   where A,B have shape (batch, M, K) and (batch, K, N)."""
    pid = tl.program_id(0)

    # Number of blocks per matrix multiplication
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_pid_m * num_pid_n

    # Decode batch and block indices
    batch = pid // blocks_per_batch
    block_id = pid % blocks_per_batch
    pid_m = block_id // num_pid_n
    pid_n = block_id % num_pid_n

    # Offsets for the current block in M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch
    a_batch = a_ptr + batch * batch_stride_a
    b_batch = b_ptr + batch * batch_stride_b
    c_batch = c_ptr + batch * batch_stride_c

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A block [M, K] and B block [K, N]
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        acc += tl.dot(a, b)           # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # Store the result back to C (fp16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_store = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_store)


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # -------------------------------------------------
    # 1. Input LayerNorm
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy parts
    # -------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 3. Cast all linear weights to fp16 (biases are None)
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 4. Linear projections (no bias)
    # -------------------------------------------------
    left = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # -------------------------------------------------
    # 5. Optional mask (broadcast over hidden dim)
    # -------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 6. Gates (sigmoid)
    # -------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # -------------------------------------------------
    # 7. Apply gates
    # -------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 8. Core TriMul reduction via batched GEMM (A @ Bᵀ)
    # -------------------------------------------------
    B, N, _, H = left.shape

    # Rearrange to (B, H, N, N) and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Transpose the last two dimensions of right for A @ Bᵀ
    right_T = right_perm.permute(0, 1, 3, 2).contiguous()  # (B, H, N, N)

    # Collapse batch & hidden dimensions: (BH, N, N)
    left_flat = left_perm.view(-1, N, N)
    right_flat = right_T.view(-1, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # Strides for the flattened tensors (they are contiguous)
    a_batch_stride = left_flat.stride(0)
    a_row_stride   = left_flat.stride(1)
    a_col_stride   = left_flat.stride(2)

    b_batch_stride = right_flat.stride(0)
    b_row_stride   = right_flat.stride(1)
    b_col_stride   = right_flat.stride(2)

    c_batch_stride = out_flat.stride(0)
    c_row_stride   = out_flat.stride(1)
    c_col_stride   = out_flat.stride(2)

    # Block sizes – chosen to fit typical GPUs (B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: one program per (batch, block_m, block_n)
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_batches = left_flat.shape[0]
    grid = (total_batches * num_pid_m * num_pid_n,)

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        a_batch_stride,
        b_batch_stride,
        c_batch_stride,
        a_row_stride,
        a_col_stride,
        b_row_stride,
        b_col_stride,
        c_row_stride,
        c_col_stride,
        N,             # M
        N,             # N
        N,             # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 9. Output LayerNorm, out‑gate, final linear projection
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # -------------------------------------------------
    # 10. Cast back to the original dtype
    # -------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
