@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,  # maybe we need H? not needed
    stride_batch, stride_i, stride_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # pid0: batch_hidden index
    # pid1: block row index (i)
    # pid2: block col index (j)
    batch_hidden = tl.program_id(0)
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    # compute base pointer for this batch_hidden
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load A block (left) shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + batch_hidden * stride_batch + offs_i[:, None] * stride_i + k_offset[None, :] * stride_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B block transposed (right) shape (BLOCK_K, BLOCK_N)
        # right layout: (batch_hidden, j, k)
        # we need B_T[k,j] so load using k as first index, j as second
        b_ptrs = right_ptr + batch_hidden * stride_batch + k_offset[:, None] * stride_i + offs_j[None, :] * stride_k
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Compute dot product
        acc += tl.dot(a, b)

    # Store result
    out_ptrs = out_ptr + batch_hidden * stride_batch + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the 4‑D input over the channel dimension (float32 for stability).
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections (left/right) and gate projections (left/right/out) using
   bias‑free weights, followed by sigmoid for the gate tensors.
4. Fuse projection, gate and optional mask in a single Triton kernel
   (proj * gate * mask).  If no mask is provided the fusion reduces to
   element‑wise multiplication.
5. Core reduction – for each batch and hidden channel compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a tiled batched matrix‑multiplication kernel written in Triton.
   The kernel processes a tile of size BLOCK_M × BLOCK_N and iterates over
   K‑tiles, accumulating in FP32 and storing the result in FP16.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   out‑gate, and linearly project back to the original channel dimension.
7. Return the result in the original dtype (float32).

The implementation mixes PyTorch for the lightweight operations and Triton
for the two compute‑intensive parts (gate‑mask fusion and the O(N³) reduction).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # int: B*N*N*H
    hidden_dim,        # int: H
    BLOCK_SIZE: tl.constexpr,
):
    """
    out = proj * gate * mask   (mask broadcast over hidden_dim)
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


@triton.jit
def _tri_mul_matmul_kernel(
    left_ptr,    # fp16 * (B*H, N, N)
    right_ptr,   # fp16 * (B*H, N, N)
    out_ptr,     # fp16 * (B*H, N, N)
    N: tl.int64,
    stride_batch: tl.int64,
    stride_i: tl.int64,
    stride_k: tl.int64,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication for the TriMul core reduction:
        out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
    (right is accessed transposed on‑the‑fly).
    """
    batch_hidden = tl.program_id(0)   # index over B*H
    block_i = tl.program_id(1)        # tile row
    block_j = tl.program_id(2)        # tile column

    # offsets within the N × N matrix
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension K
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_tiles):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + batch_hidden * stride_batch \
                 + offs_i[:, None] * stride_i \
                 + offs_k[None, :] * stride_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right block (transposed on‑the‑fly):
        # we need right[b,h,k,j] which is right[b,h,j,k] with swapped axes.
        b_ptrs = right_ptr + batch_hidden * stride_batch \
                 + offs_k[:, None] * stride_i \
                 + offs_j[None, :] * stride_k
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # write result back in FP16
    out_ptrs = out_ptr + batch_hidden * stride_batch \
               + offs_i[:, None] * stride_i \
               + offs_j[None, :] * stride_k
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 2048) -> torch.Tensor:
    """
    Fuse projection, gate and (optional) mask using a Triton kernel.
    All tensors must be fp16, contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)  (binary 0/1, broadcast over H)
    Returns proj * gate * mask  with shape (B,N,N,H).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (float32/float16/bool) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim` (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy stages
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast linear weights to FP16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid (B,N,N,H)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Apply mask + gating (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16, contiguous, binary (0/1)
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Rearrange to (B, H, N, N) and flatten batch*hidden for the kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)    # (B*H, N, N)

    # Strides for the contiguous (B*H, N, N) layout
    stride_batch = left_flat.stride(0)  # N*N
    stride_i = left_flat.stride(1)      # N
    stride_k = left_flat.stride(2)      # 1

    # Tiling parameters (tuned for typical NVIDIA GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)

    grid = (B * H, num_blocks_m, num_blocks_n)

    _tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_batch,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    # LayerNorm over hidden dimension (float32 for stability)
    out = out.to(torch.float32)
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
    )
    # Apply out‑gate
    out = out * out_gate.to(torch.float32)
    # Final linear projection back to original dim
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
