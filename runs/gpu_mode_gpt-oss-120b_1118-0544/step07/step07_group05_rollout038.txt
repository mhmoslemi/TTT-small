def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr
):
    pid_z = tl.program_id(2)   # batch
    pid_m = tl.program_id(0)   # row block
    pid_n = tl.program_id(1)   # column block

    # Compute block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out-of-bound
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A and B tiles
        a = tl.load(
            a_ptr + pid_z * batch_stride_a + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        b = tl.load(
            b_ptr + pid_z * batch_stride_b + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # Convert to fp32
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        # Compute dot product
        acc += tl.dot(a, b)

        # Increment pointers?
        # Since we compute offsets each iteration, no need to incr.

    # Write back
    c = acc.to(tl.float16)
    tl.store(
        c_ptr + pid_z * batch_stride_c + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn),

    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,

    pid_n = tl.program_id(1)

"""
Custom Triton kernel for the outgoing Triangle Multiplicative Update (TriMul).
The implementation follows the PyTorch reference but accelerates the
mask+gate fusion (element‑wise) and the core O(N³) reduction with a
batched GEMM written in Triton. LayerNorms and linear projections are
performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernels
# --------------------------------------------------------------

@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (B,N,N,H) flattened
    right_ptr,         # fp16 * right (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and left/right gates."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # Load values
    l = tl.load(left_ptr + offs, mask=valid)
    r = tl.load(right_ptr + offs, mask=valid)
    lg = tl.load(left_gate_ptr + offs, mask=valid)
    rg = tl.load(right_gate_ptr + offs, mask=valid)

    # One mask value per (b,i,j); each group of H entries shares it
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=valid)

    # Apply mask and gates
    tl.store(left_ptr + offs,  l * m * lg, mask=valid)
    tl.store(right_ptr + offs, r * m * rg, mask=valid)


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    """Batched GEMM: C = A @ B, where
       A: (batch, M, K)
       B: (batch, K, N)
       C: (batch, M, N)"""
    pid_z = tl.program_id(2)   # batch index
    pid_m = tl.program_id(0)   # row block
    pid_n = tl.program_id(1)   # column block

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile
        a = tl.load(
            a_ptr + pid_z * batch_stride_a
                  + (offs_m[:, None] * stride_am)
                  + (offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load B tile (note B is (batch, K, N))
        b = tl.load(
            b_ptr + pid_z * batch_stride_b
                  + (offs_k[:, None] * stride_bk)
                  + (offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # Convert to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)

    # Write result
    c = acc.to(tl.float16)
    tl.store(
        c_ptr + pid_z * batch_stride_c
              + (offs_m[:, None] * stride_cm)
              + (offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :]
    )

        num_warps=4,
    )

    out_b = out_flat.view(B, hidden_dim, N, N)           # (B, H, N, N)
    out = out_b.permute(0, 2, 3, 1).contiguous()        # (B, N, N, H)

    # Release intermediate tensors we no longer need
    del left, right, left_b, right_b, left_flat, right_flat, out_flat, out_b

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------

"""
Custom Triton kernel for the outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the PyTorch reference but accelerates two
bottlenecks:
1) Element‑wise mask × gate fusion via a simple Triton kernel.
2) The O(N³) reduction (∑ₖ left[b,i,k,h] * right[b,j,k,h]) via a
   batched GEMM written in Triton (one GEMM per hidden channel).
All other operations (LayerNorms, linear layers, sigmoid) use PyTorch
for simplicity. The kernel works for variable batch size, sequence
length and hidden dimension, handling both masked and unmasked inputs.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernels
# --------------------------------------------------------------

@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (B,N,N,H) flattened
    right_ptr,         # fp16 * right (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and left/right gates."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # Load values
    l = tl.load(left_ptr + offs, mask=valid)
    r = tl.load(right_ptr + offs, mask=valid)
    lg = tl.load(left_gate_ptr + offs, mask=valid)
    rg = tl.load(right_gate_ptr + offs, mask=valid)

    # One mask value per (b,i,j); each group of H entries shares it
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=valid)

    # Apply mask and gates, store back in‑place
    tl.store(left_ptr + offs,  l * m * lg, mask=valid)
    tl.store(right_ptr + offs, r * m * rg, mask=valid)


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    """Batched GEMM: C = A @ B
    A: (batch, M, K)  (row‑major)
    B: (batch, K, N)  (row‑major, i.e. transposed version of the original RHS)
    C: (batch, M, N)
    """
    pid_z = tl.program_id(2)   # batch index
    pid_m = tl.program_id(0)   # row block
    pid_n = tl.program_id(1)   # column block

    # Block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile
        a = tl.load(
            a_ptr + pid_z * batch_stride_a
                  + (offs_m[:, None] * stride_am)
                  + (offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load B tile (B is (batch, K, N))
        b = tl.load(
            b_ptr + pid_z * batch_stride_b
                  + (offs_k[:, None] * stride_bk)
                  + (offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # fp16 → fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)

    # Write back result in fp16
    c = acc.to(tl.float16)
    tl.store(
        c_ptr + pid_z * batch_stride_c
              + (offs_m[:, None] * stride_cm)
              + (offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :]
    )

# --------------------------------------------------------------
# Custom kernel entry point
# --------------------------------------------------------------

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)   # switch to fp16 for the heavy part

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse mask and gates (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()                     # B*N*N*H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)
        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        del left_gate, right_gate
    else:
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction via a batched GEMM Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Rearrange to (B, H, N, N) for GEMM
    left_b = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Flatten batch and hidden dimensions → (batch, N, N)
    batch = B * hidden_dim
    left_flat = left_b.view(batch, N, N)                 # (batch, M, K)
    right_t = right_b.transpose(-2, -1).contiguous()    # (B, H, N, N) -> (batch, K, N)
    right_flat = right_t.view(batch, N, N)               # (batch, K, N)

    # Allocate output buffer
    out_flat = torch.empty((batch, N, N), dtype=torch.float16, device=device)

    # Kernel launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )
    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to (B, N, N, H)
    out_b = out_flat.view(B, hidden_dim, N, N)          # (B, H, N, N)
    out = out_b.permute(0, 2, 3, 1).contiguous()       # (B, N, N, H)

    # Cleanup temporaries
    del left, right, left_b, right_b, left_flat, right_flat, out_flat, out_b

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
