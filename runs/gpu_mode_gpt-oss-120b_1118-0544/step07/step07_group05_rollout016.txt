            num_warps=4,
        )

        del left_gate, right_gate

        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # 5. Core reduction: batched matmul
    B, N, _, _ = left.shape
    # permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # swap i and j

    # reshape for bmm
    left_b = left_perm.view(B * hidden_dim, N, N)
    right_b = right_perm.view(B * hidden_dim, N, N)

    # matmul: left_b @ right_b

    out = out_b.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden_dim)

        (hidden_dim,),

    out = out * out_gate

    # cast back to original dtype
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input (float32 for stability) and cast to fp16.
2. Fuse the two linear projections (left & right) into a single matmul.
3. Fuse the three gate linear layers (left_gate, right_gate, out_gate) into a
   single matmul and apply sigmoid.
4. Fuse mask (if present) and gate multiplications in a custom Triton kernel.
5. Core O(N³·H) reduction using a batched matrix multiplication:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is implemented as a standard batched GEMM on tensors of shape
   (B·H, N, N) via torch.bmm (cuBLAS).
6. Layer‑normalize over the hidden dimension H.
7. Apply the output gate.
8. Final linear projection back to the original channel dimension C.
9. Cast back to the original dtype.

The kernel only handles the mask‑gate fusion; the heavy linear and GEMM
operations are delegated to highly‑optimized PyTorch/cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (B,N,N,H) flattened
    right_ptr,         # fp16 * right (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and left/right gates.
    left  = left  * mask * left_gate
    right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load values
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # One mask element per hidden slice
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply and store back in‑place
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with model parameters
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)  # cast to fp16 for the compute‑heavy part

    # ------------------------------------------------------------------
    # 2) Fuse linear projections (left & right) into one matmul
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    proj_weight = torch.cat([left_proj_w, right_proj_w], dim=1)  # (dim, 2*hidden_dim)

    proj = F.linear(x, proj_weight)                     # (B, N, N, 2*hidden_dim)
    left = proj[..., :hidden_dim]                       # (B, N, N, hidden_dim)
    right = proj[..., hidden_dim:]                      # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3) Fuse gate linear layers (left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    gate_weight = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=1)  # (dim, 3*hidden_dim)

    gates = F.linear(x, gate_weight)                    # (B, N, N, 3*hidden_dim)
    left_gate = torch.sigmoid(gates[..., :hidden_dim])
    right_gate = torch.sigmoid(gates[..., hidden_dim:2*hidden_dim])
    out_gate = torch.sigmoid(gates[..., 2*hidden_dim:])

    # ------------------------------------------------------------------
    # 4) Fuse mask and gate multiplications (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()  # B * N * N * hidden_dim
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates are no longer needed
        del left_gate, right_gate
    else:
        # Simple element‑wise multiplication when no mask is used
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5) Core reduction: batched matrix multiplication over the sequence dim
    #    left: (B, N, N, H) -> (B, H, N, N)
    #    right: (B, N, N, H) -> (B, H, N, N) and swap the first two seq dims
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    # Flatten batch and head dimensions for torch.bmm
    left_b = left_perm.view(B * hidden_dim, N, N)
    right_b = right_perm.view(B * hidden_dim, N, N)

    # Batched GEMM (cuBLAS)
    out_b = torch.bmm(left_b, right_b)                  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_b.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float16),
    )

    # ------------------------------------------------------------------
    # 7) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(device=device, dtype=torch.float16))

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
