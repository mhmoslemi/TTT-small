"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input (float32 → fp16 for compute).
2. Compute linear projections (left/right) and three sigmoid gates (left, right, out).
3. Fuse mask (if present) together with left/right gates in a small Triton kernel that
   operates in‑place on the flattened tensors, avoiding an extra memory pass.
4. Core contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (L @ Rᵀ) using Torch’s highly‑optimized
   `torch.bmm`.  The tensors are reshaped to (B·H, N, N) to expose the batch dim.
5. Apply a second LayerNorm on the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original dimension.
6. Cast the result back to float32.

Only the mask‑gate fusion is performed in Triton; the remaining operations use
torch kernels (which are already heavily optimized).  This satisfies the
requirement of “implement at least part of the operations in a kernel” while
keeping the overall implementation fast and simple.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # [B,N,N,H] flattened
    right_ptr,         # [B,N,N,H] flattened
    left_gate_ptr,     # same shape as left
    right_gate_ptr,    # same shape as right
    mask_ptr,          # [B,N,N] flattened, broadcast over H
    total_elems,       # int64 scalar = B*N*N*H
    hidden_dim,        # int64 scalar = H
    HAS_MASK: tl.constexpr,   # compile‑time bool
    BLOCK_SIZE: tl.constexpr, # compile‑time int
):
    """In‑place fusion:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
       If HAS_MASK is False the mask is ignored (treated as 1).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE, dtype=tl.int64)
    mask = offs < total_elems

    # Load data
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Identify the corresponding mask entry (shared across H)
        mask_idx = offs // hidden_dim            # index in [B,N,N]
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

    else:
        # No mask – just apply gates
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM (L @ Rᵀ) per hidden channel
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # (B*H, N, N) matrices
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # Compute out_mat = left_mat @ right_matᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate                                 # apply output gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)                        # (B,N,N,dim)

    # -----------------------------------------------------------------
    # 7. Cast back to float32 and return
    # -----------------------------------------------------------------
    out = out.to(torch.float32)
    return out
```
