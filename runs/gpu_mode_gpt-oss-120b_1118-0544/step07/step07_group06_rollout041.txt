@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_gate_left_b, stride_gate_left_i, stride_gate_left_k, stride_gate_left_h,
    stride_gate_right_b, stride_gate_right_i, stride_gate_right_k, stride_gate_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # compute batch and hidden index
    pid_batch = tl.program_id(2)  # run over B*H total groups
    batch_idx = pid_batch // H
    h_idx = pid_batch % H

    # compute tile indices
    pid_m = tl.program_id(0)  # tile row index i
    pid_n = tl.program_id(1)  # tile col index j

    # compute start offsets
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    # masks for out-of-range
    mask_i = i < N
    mask_j = j < N

    # compute pointers for A (left) tile
    # left_ptr + batch_idx*stride_left_b + i*stride_left_i + k*stride_left_k + h_idx*stride_left_h
    a_ptrs = left_ptr + batch_idx * stride_left_b + i[:, None] * stride_left_i + k[None, :] * stride_left_k + h_idx * stride_left_h

    # load left values
    a = tl.load(a_ptrs, mask=mask_i[:, None] & (k[None, :] < N), other=0.0)

    # load left_gate values
    g_left_ptrs = left_gate_ptr + batch_idx * stride_gate_left_b + i[:, None] * stride_gate_left_i + k[None, :] * stride_gate_left_k + h_idx * stride_gate_left_h
    g_left = tl.load(g_left_ptrs, mask=mask_i[:, None] & (k[None, :] < N), other=0.0)

    # load mask for left (no hidden dim)
    mask_left_ptrs = mask_ptr + batch_idx * stride_mask_b + i[:, None] * stride_mask_i + k[None, :] * stride_mask_k
    mask_left = tl.load(mask_left_ptrs, mask=mask_i[:, None] & (k[None, :] < N), other=0.0)

    # multiply left scaled
    a_scaled = a * mask_left * g_left

    # compute pointers for B_T tile (right)
    # Right original shape: (b, j, k, h)
    # we need B_T element B_T[k, j] = right[b, j, k, h]
    b_ptrs = right_ptr + batch_idx * stride_right_b + j[:, None] * stride_right_i + k[None, :] * stride_right_k + h_idx * stride_right_h
    # note: the shape after indexing: (BLOCK_N, BLOCK_K) due to j along rows, k along columns
    b = tl.load(b_ptrs, mask=mask_j[:, None] & (k[None, :] < N), other=0.0)

    # load right gate
    g_right_ptrs = right_gate_ptr + batch_idx * stride_gate_right_b + j[:, None] * stride_gate_right_i + k[None, :] * stride_gate_right_k + h_idx * stride_gate_right_h
    g_right = tl.load(g_right_ptrs, mask=mask_j[:, None] & (k[None, :] < N), other=0.0)

    # load mask for right
    mask_right_ptrs = mask_ptr + batch_idx * stride_mask_b + j[:, None] * stride_mask_i + k[None, :] * stride_mask_k
    mask_right = tl.load(mask_right_ptrs, mask=mask_j[:, None] & (k[None, :] < N), other=0.0)

    b_scaled = b * mask_right * g_right

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k blocks
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        # compute offsets for this block
        cur_k = kk * BLOCK_K + k

        # load A block
        a_ptrs = left_ptr + batch_idx * stride_left_b + i[:, None] * stride_left_i + cur_k[None, :] * stride_left_k + h_idx * stride_left_h
        a = tl.load(a_ptrs, mask=mask_i[:, None] & (cur_k[None, :] < N), other=0.0)

        # load left gate and mask for this block
        g_left_ptrs = left_gate_ptr + batch_idx * stride_gate_left_b + i[:, None] * stride_gate_left_i + cur_k[None, :] * stride_gate_left_k + h_idx * stride_gate_left_h
        g_left = tl.load(g_left_ptrs, mask=mask_i[:, None] & (cur_k[None, :] < N), other=0.0)

        mask_left_ptrs = mask_ptr + batch_idx * stride_mask_b + i[:, None] * stride_mask_i + cur_k[None, :] * stride_mask_k
        mask_left = tl.load(mask_left_ptrs, mask=mask_i[:, None] & (cur_k[None, :] < N), other=0.0)

        a_scaled = a * mask_left * g_left

        # load B block (right)
        b_ptrs = right_ptr + batch_idx * stride_right_b + j[:, None] * stride_right_i + cur_k[None, :] * stride_right_k + h_idx * stride_right_h
        b = tl.load(b_ptrs, mask=mask_j[:, None] & (cur_k[None, :] < N), other=0.0)

        g_right_ptrs = right_gate_ptr + batch_idx * stride_gate_right_b + j[:, None] * stride_gate_right_i + cur_k[None, :] * stride_gate_right_k + h_idx * stride_right_h
        g_right = tl.load(g_right_ptrs, mask=mask_j[:, None] & (cur_k[None, :] < N), other=0.0)

        mask_right_ptrs = mask_ptr + batch_idx * stride_mask_b + j[:, None] * stride_mask_i + cur_k[None, :] * stride_mask_k
        mask_right = tl.load(mask_right_ptrs, mask=mask_j[:, None] & (cur_k[None, :] < N), other=0.0)

        b_scaled = b * mask_right * g_right

        # dot product over k axis
        acc += tl.dot(a_scaled, b_scaled) # shape (BLOCK_M, BLOCK_N)

    # after loop, store result in out tensor
    out_ptrs = out_ptr + batch_idx * stride_out_b + i[:, None] * stride_out_i + j[None, :] * stride_out_j + h_idx * stride_out_h
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float32)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # Cast to fp16 for subsequent ops
    x = x.to(torch.float16)

    # Cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # Linear projections
    left = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w) # (B,N,N,H)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Mask handling
    B, N, _, _ = left.shape
    if mask is None or nomask:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(device=device, dtype=torch.float16)

    # Allocate output tensor
    out_hidden = torch.empty_like(left)  # shape (B,N,N,H)

    # Compute kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    # Get strides
    left_stride_b, left_stride_i, left_stride_k, left_stride_h = left.stride()
    right_stride_b, right_stride_i, right_stride_k, right_stride_h = right.stride()
    left_gate_stride_b, left_gate_stride_i, left_gate_stride_k, left_gate_stride_h = left_gate.stride()
    right_gate_stride_b, right_gate_stride_i, right_gate_stride_k, right_gate_stride_h = right_gate.stride()
    mask_stride_b, mask_stride_i, mask_stride_k = mask_tensor.stride()
    out_stride_b, out_stride_i, out_stride_j, out_stride_h = out_hidden.stride()

    # Launch kernel
    _tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_tensor,
        out_ptr=out_hidden,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=left_stride_b,
        stride_left_i=left_stride_i,
        stride_left_k=left_stride_k,
        stride_left_h=left_stride_h,
        stride_right_b=right_stride_b,
        stride_right_i=right_stride_i,
        stride_right_k=right_stride_k,
        stride_right_h=right_stride_h,
        stride_gate_left_b=left_gate_stride_b,
        stride_gate_left_i=left_gate_stride_i,
        stride_gate_left_k=left_gate_stride_k,
        stride_gate_left_h=left_gate_stride_h,
        stride_gate_right_b=right_gate_stride_b,
        stride_gate_right_i=right_gate_stride_i,
        stride_gate_right_k=right_gate_stride_k,
        stride_gate_right_h=right_gate_stride_h,
        stride_mask_b=mask_stride_b,
        stride_mask_i=mask_stride_i,
        stride_mask_k=mask_stride_k,
        stride_out_b=out_stride_b,
        stride_out_i=out_stride_i,
        stride_out_j=out_stride_j,
        stride_out_h=out_stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        # maybe use num_stages=2 for shared memory usage, but default fine
    )

    # Second layernorm over hidden dim
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Apply output gate
    out_hidden = out_hidden * out_gate

    # Final linear projection back to dim
    out = F.linear(out_hidden, to_out_w)  # (B,N,N,dim)

    # Cast back to original dtype if needed
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out

@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_gate_left_b, stride_gate_left_i, stride_gate_left_k, stride_gate_left_h,
    stride_gate_right_b, stride_gate_right_i, stride_gate_right_k, stride_gate_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)
    batch_idx = pid_batch // H
    h_idx = pid_batch % H

    pid_m = tl.program_id(0) # tile row (i)
    pid_n = tl.program_id(1) # tile col (j)

    # Compute range of i and j
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N

    # Create ranges
    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)
    # Mask for out-of-bounds
    mask_i = i < N
    mask_j = j < N

    # Initialize accumulator

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the PyTorch reference:
1. Layer‑normalize the input (float32 for stability).
2. Linear projections to a hidden dimension (fp16).
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. Fuse mask, left‑gate and right‑gate scalings and perform the core
   reduction `out[b,i,j,h] = Σ_k left[b,i,k,h] * mask[b,i,k] * left_gate[b,i,k,h] *
                           right[b,j,k,h] * mask[b,j,k] * right_gate[b,j,k,h]`
   using a custom Triton kernel.  The kernel treats each pair (batch, hidden)
   as an independent matrix‑multiply of size N×N, thus performing B*H batched
   GEMMs without materialising the intermediate masked tensors.
5. Apply a second LayerNorm over the hidden dimension.
6. Multiply by the output gate.
7. Final linear projection back to the original channel dimension.
All heavy arithmetic is done in fp16; the final result is cast back to the
original dtype of the input tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM with on‑the‑fly mask and gate scaling
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    # pointers
    left_ptr,          # const * left  (B,N,N,H)
    right_ptr,         # const * right (B,N,N,H)
    left_gate_ptr,     # const * left_gate  (B,N,N,H)
    right_gate_ptr,    # const * right_gate (B,N,N,H)
    mask_ptr,          # const * mask  (B,N,N)
    out_ptr,           # * out  (B,N,N,H)

    # dimensions
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,

    # strides for left
    stride_left_b: tl.int32,
    stride_left_i: tl.int32,
    stride_left_k: tl.int32,
    stride_left_h: tl.int32,

    # strides for right
    stride_right_b: tl.int32,
    stride_right_i: tl.int32,
    stride_right_k: tl.int32,
    stride_right_h: tl.int32,

    # strides for left_gate
    stride_gate_left_b: tl.int32,
    stride_gate_left_i: tl.int32,
    stride_gate_left_k: tl.int32,
    stride_gate_left_h: tl.int32,

    # strides for right_gate
    stride_gate_right_b: tl.int32,
    stride_gate_right_i: tl.int32,
    stride_gate_right_k: tl.int32,
    stride_gate_right_h: tl.int32,

    # strides for mask
    stride_mask_b: tl.int32,
    stride_mask_i: tl.int32,
    stride_mask_k: tl.int32,

    # strides for output
    stride_out_b: tl.int32,
    stride_out_i: tl.int32,
    stride_out_j: tl.int32,
    stride_out_h: tl.int32,

    # compile‑time block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM with mask and gate fusion.
    Each program instance computes a (BLOCK_M x BLOCK_N) tile of the output
    for a specific (batch, hidden) pair.
    """
    pid_batch = tl.program_id(2)                     # batch * hidden index
    batch_idx = pid_batch // H                        # actual batch id
    h_idx = pid_batch % H                             # hidden channel id

    pid_m = tl.program_id(0)                         # tile over rows (i)
    pid_n = tl.program_id(1)                         # tile over cols (j)

    # ---- compute offsets ----
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N
    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)

    # bounds
    mask_i = i < N
    mask_j = j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_iter in range(0, num_k):
        k_base = k_iter * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ----- load left block (M x K) -----
        a_ptrs = left_ptr \
            + batch_idx * stride_left_b \
            + i[:, None] * stride_left_i \
            + k[None, :] * stride_left_k \
            + h_idx * stride_left_h
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # left gate
        a_gate_ptrs = left_gate_ptr \
            + batch_idx * stride_gate_left_b \
            + i[:, None] * stride_gate_left_i \
            + k[None, :] * stride_gate_left_k \
            + h_idx * stride_gate_left_h
        a_gate = tl.load(a_gate_ptrs,
                         mask=mask_i[:, None] & mask_k[None, :],
                         other=0.0)

        # mask for left
        mask_left_ptrs = mask_ptr \
            + batch_idx * stride_mask_b \
            + i[:, None] * stride_mask_i \
            + k[None, :] * stride_mask_k
        mask_left = tl.load(mask_left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        a_scaled = a * mask_left * a_gate   # (M, K)

        # ----- load right block (K x N) -----
        b_ptrs = right_ptr \
            + batch_idx * stride_right_b \
            + k[:, None] * stride_right_k \
            + j[None, :] * stride_right_i \
            + h_idx * stride_right_h
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # right gate
        b_gate_ptrs = right_gate_ptr \
            + batch_idx * stride_gate_right_b \
            + k[:, None] * stride_gate_right_k \
            + j[None, :] * stride_gate_right_i \
            + h_idx * stride_gate_right_h
        b_gate = tl.load(b_gate_ptrs,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0)

        # mask for right (mask[b, j, k])
        mask_right_ptrs = mask_ptr \
            + batch_idx * stride_mask_b \
            + j[None, :] * stride_mask_i \
            + k[:, None] * stride_mask_k
        mask_right = tl.load(mask_right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        b_scaled = b * mask_right * b_gate   # (K, N)

        # ----- accumulate -----
        acc += tl.dot(a_scaled, b_scaled)    # (M, N)

    # ---- write result ----
    out_ptrs = out_ptr \
        + batch_idx * stride_out_b \
        + i[:, None] * stride_out_i \
        + j[None, :] * stride_out_j \
        + h_idx * stride_out_h
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, if True ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float32)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float32)

    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # Cast to fp16 for the remaining heavy computation
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Prepare mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is None or nomask:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 6) Core reduction: fused mask, gates and batched GEMM via Triton
    # ------------------------------------------------------------------
    out_hidden = torch.empty_like(left)  # (B, N, N, H)

    # Block sizes – tuned for typical B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    # Strides for all tensors
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_gate_left_b, stride_gate_left_i, stride_gate_left_k, stride_gate_left_h = left_gate.stride()
    stride_gate_right_b, stride_gate_right_i, stride_gate_right_k, stride_gate_right_h = right_gate.stride()
    stride_mask_b, stride_mask_i, stride_mask_k = mask_tensor.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_hidden.stride()

    # Launch kernel
    _tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_tensor,
        out_ptr=out_hidden,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_gate_left_b=stride_gate_left_b,
        stride_gate_left_i=stride_gate_left_i,
        stride_gate_left_k=stride_gate_left_k,
        stride_gate_left_h=stride_gate_left_h,
        stride_gate_right_b=stride_gate_right_b,
        stride_gate_right_i=stride_gate_right_i,
        stride_gate_right_k=stride_gate_right_k,
        stride_gate_right_h=stride_gate_right_h,
        stride_mask_b=stride_mask_b,
        stride_mask_i=stride_mask_i,
        stride_mask_k=stride_mask_k,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out_hidden = out_hidden * out_gate

    # ------------------------------------------------------------------
    # 9) Final projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out_hidden, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
