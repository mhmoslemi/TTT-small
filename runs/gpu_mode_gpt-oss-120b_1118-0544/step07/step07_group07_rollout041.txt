def _matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bn, stride_bk, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)  # batch index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    # Cast to fp16
    x_fp16 = x.to(torch.float16)

    # Linear projections
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = torch.nn.functional.linear(x_fp16, left_proj_w)    # (B,N,N,H)
    right = torch.nn.functional.linear(x_fp16, right_proj_w)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_w))

    # Apply mask & gates via Triton kernel if needed
    if mask is not None and not nomask:
        # mask shape (B,N,N)
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten for kernel
        B, N, _, H = left.shape
        total = B * N * N * H
        # flatten
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)  # (B,N,N)

        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # Release gates to save memory
    del left_gate, right_gate

    # Core contraction: batched matmul using Triton
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    num_mats = B * H
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(num_mats, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(num_mats, N, N)  # transpose K and N

    out_mat = torch.empty_like(left_mat)

    # strides
    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_k = right_mat.stride(1)
    stride_b_n = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_m = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

            triton.cdiv(N, BLOCK_N))

        stride_a_m, stride_a_k, stride_a_batch,

        stride_c_m, stride_c_n, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layer norm
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply out_gate gating
    out = out * out_gate

    # Final linear projection
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to input dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

@triton.jit
def _matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)  # batch index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # shared acc
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Load a and b tiles
        a = tl.load(a_ptr + pid * stride_ab + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak,
                    mask=(offs_m[:, None] < M) & (cur_k[None, :] < K), other=0.0)
        b = tl.load(b_ptr + pid * stride_bb + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
                    mask=(cur_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)
        acc += tl.dot(a, b)

    # store results
    c = acc.to(tl.float16)
    tl.store(c_ptr + pid * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
             c, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

"""
custom_kernel: Triton‑accelerated outgoing TriMul (Triangle Multiplicative Update).

Algorithm
---------
1. LayerNorm on the input tensor (float32).
2. Cast to FP16 for the heavy compute.
3. Linear projections (bias‑less) → left, right, left_gate, right_gate, out_gate.
4. Fuse mask (if any) and gate multiplication in a tiny Triton kernel
   (in‑place on left/right).
5. Core contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM (left @ rightᵀ) via a custom Triton kernel
   that processes all hidden slices in one launch.
6. Apply a second LayerNorm on the hidden dimension, multiply by out_gate,
   and a final linear projection back to the original dimension.
7. Cast back to the input dtype and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Kernel 1: fused mask & gate multiplication (in‑place on left/right)
# -------------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened, fp16
    left_gate_ptr, right_gate_ptr,     # same shape, fp16
    mask_ptr,                          # [B,N,N] flattened, fp16
    total_elems, hidden_dim,           # runtime scalars (int32)
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each mask element repeats `hidden_dim` times in the flattened view
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # store back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


# -------------------------------------------------------------------------
# Kernel 2: batched matrix multiplication (left @ rightᵀ)
# -------------------------------------------------------------------------
@triton.jit
def _matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix sizes (all equal to seq_len)
    stride_am, stride_ak, stride_ab,          # a: (batch, row, col)
    stride_bk, stride_bn, stride_bb,          # b: (batch, row, col)  (row = K, col = N)
    stride_cm, stride_cn, stride_cb,          # c: (batch, row, col)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)        # batch index (B * hidden_dim)
    pid_m = tl.program_id(1)      # tile row
    pid_n = tl.program_id(2)      # tile col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # mask for out‑of‑bounds
        mask_a = (offs_m[:, None] < M) & (cur_k[None, :] < K)
        mask_b = (cur_k[:, None] < K) & (offs_n[None, :] < N)

        # load tiles (fp16)
        a = tl.load(a_ptr + pid * stride_ab
                         + offs_m[:, None] * stride_am
                         + cur_k[None, :] * stride_ak,
                    mask=mask_a, other=0.0)
        b = tl.load(b_ptr + pid * stride_bb
                         + cur_k[:, None] * stride_bk
                         + offs_n[None, :] * stride_bn,
                    mask=mask_b, other=0.0)

        # dot product (fp16 -> fp32 accumulation)
        acc += tl.dot(a, b)

    # write result (fp16)
    c = acc.to(tl.float16)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptr + pid * stride_cb
                     + offs_m[:, None] * stride_cm
                     + offs_n[None, :] * stride_cn,
              c, mask=mask_c)


# -------------------------------------------------------------------------
# Entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of parameter tensors (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)   # (B,N,N,dim)

    # -------------------------------------------------------------
    # 2. Cast to FP16 for heavy compute
    # -------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -------------------------------------------------------------
    # 3. Linear projections (bias‑less) in FP16
    # -------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)                      # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)                    # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))  # (B,N,N,hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # (B,N,N,hidden_dim)

    # -------------------------------------------------------------
    # 4. Fuse mask and gates (in‑place) via Triton
    # -------------------------------------------------------------
    if mask is not None and not nomask:
        # mask is (B,N,N); broadcast to hidden_dim inside the kernel
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        B, N, _, H = left.shape
        total = B * N * N * H
        BLOCK = 1024

        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)          # (B*N*N,)

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # simple element‑wise gating when no mask
        left = left * left_gate
        right = right * right_gate

    # free gates that are no longer needed
    del left_gate, right_gate

    # -------------------------------------------------------------
    # 5. Core contraction via a batched GEMM (left @ rightᵀ) in Triton
    # -------------------------------------------------------------
    B, N, _, H = left.shape
    num_mats = B * H

    # (B,H,N,N) -> (num_mats, N, N) where rows = i/k, cols = k/j
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(num_mats, N, N)
    # transpose the second N dimension of right before flattening
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(num_mats, N, N)

    out_mat = torch.empty_like(left_mat)

    # strides (in elements)
    stride_a_batch = left_mat.stride(0)
    stride_a_m     = left_mat.stride(1)
    stride_a_k     = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_k     = right_mat.stride(1)
    stride_b_n     = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_m     = out_mat.stride(1)
    stride_c_n     = out_mat.stride(2)

    # tile sizes – tuned for Hopper B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        num_mats,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_a_m, stride_a_k, stride_a_batch,
        stride_b_k, stride_b_n, stride_b_batch,
        stride_c_m, stride_c_n, stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # -------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate                     # apply output gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)            # (B,N,N,dim) in fp16

    # -------------------------------------------------------------
    # 7. Cast back to original dtype
    # -------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
