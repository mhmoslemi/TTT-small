    B, N, H,
    stride_b_l, stride_i_l, stride_k_l, stride_h_l,
    stride_b_r, stride_i_r, stride_k_r, stride_h_r,
    stride_b_o, stride_i_o, stride_k_o, stride_h_o,

):
    pid = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch and hidden index
    b = pid // H
    h = pid % H

    # Compute start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # initialize accumulator

        i_offset = i_start + tl.arange(0, BLOCK_M)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input (float32) → FP16.
2. Linear projections (left/right) from C → H using FP16 GEMM.
3. Compute three sigmoid gates (left, right, out) with a linear layer.
4. Fuse mask (if present) and gate multiplications in‑place via a Triton
   kernel (no extra temporaries).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton block‑matrix multiplication over the
   sequence dimension (B·H programs, each computes a (BLOCK_M×BLOCK_N) tile).
6. Layer‑norm over the hidden dimension H.
7. Apply the output gate (element‑wise).
8. Final linear projection H → C.
9. Cast result back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (B,N,N,H) flattened
    right_ptr,         # fp16 * right (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion: left  = left  * mask * left_gate
                       right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load values
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # One mask value per (b,i,k) → broadcast over H
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply and store back
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b_l, stride_i_l, stride_k_l, stride_h_l,
    stride_b_r, stride_i_r, stride_k_r, stride_h_r,
    stride_b_o, stride_i_o, stride_k_o, stride_h_o,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where left/right have shape (B,N,N,H).
    The kernel launches (B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N)) programs.
    """
    pid0 = tl.program_id(0)    # combined batch*hidden
    pid_i = tl.program_id(1)   # block row (i)
    pid_j = tl.program_id(2)   # block col (j)

    # Decode batch and hidden indices
    b = pid0 // H
    h = pid0 % H

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K

        # ---- Load left tile (i, k) ----
        i_idx = i_start + tl.arange(0, BLOCK_M)
        k_idx = k_start + tl.arange(0, BLOCK_K)

        mask_i = i_idx < N
        mask_k = k_idx < N

        left_ptrs = (left_ptr +
                     b * stride_b_l +
                     i_idx[:, None] * stride_i_l +
                     k_idx[None, :] * stride_k_l +
                     h * stride_h_l)

        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ---- Load right tile (j, k) ----
        j_idx = j_start + tl.arange(0, BLOCK_N)
        mask_j = j_idx < N

        right_ptrs = (right_ptr +
                      b * stride_b_r +
                      j_idx[None, :] * stride_i_r +   # j dimension uses stride_i
                      k_idx[:, None] * stride_k_r +   # k dimension uses stride_k
                      h * stride_h_r)

        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # ---- Accumulate ----
        acc += tl.dot(left, right)

    # ---- Store result ----
    i_idx = i_start + tl.arange(0, BLOCK_M)
    j_idx = j_start + tl.arange(0, BLOCK_N)

    mask_i = i_idx < N
    mask_j = j_idx < N

    out_ptrs = (out_ptr +
                b * stride_b_o +
                i_idx[:, None] * stride_i_o +
                j_idx[None, :] * stride_k_o +
                h * stride_h_o)

    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see keys in the function)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with dtype torch.float32.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to fp16 for compute‑heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse mask and gate multiplications (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(device=device, dtype=torch.float16).contiguous()
        total_elements = left.numel()  # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates are no longer needed
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    out = torch.empty_like(left)  # (B, N, N, H), fp16

    # Strides for left / right / out (all identical layout)
    stride_b_l, stride_i_l, stride_k_l, stride_h_l = left.stride()
    stride_b_r, stride_i_r, stride_k_r, stride_h_r = right.stride()
    stride_b_o, stride_i_o, stride_k_o, stride_h_o = out.stride()

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_b_l, stride_i_l, stride_k_l, stride_h_l,
        stride_b_r, stride_i_r, stride_k_r, stride_h_r,
        stride_b_o, stride_i_o, stride_k_o, stride_h_o,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C) in fp16

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype (float32)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
