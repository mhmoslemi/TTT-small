@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    B, H, M, N, K,
    stride_Ab, stride_Ah, stride_Am, stride_Ak,
    stride_Bb, stride_Bh, stride_Bk, stride_Bn,
    stride_Cb, stride_Ch, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    # compute batch index: pid // (num_blocks_m * num_blocks_n)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    block_m = block_idx // num_blocks_n
    block_n = block_idx % num_blocks_n
    # compute offsets for this block
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # load A and B tiles
        a_ptrs = A_ptr + batch*stride_Ab + 0*stride_Ah + offs_m[:, None]*stride_Am + offs_k[None, :]*stride_Ak
        b_ptrs = B_ptr + batch*stride_Bb + 0*stride_Bh + offs_k[:, None]*stride_Bk + offs_n[None, :]*stride_Bn
        a = tl.load(a_ptrs, mask=offs_m < M and offs_k < K, other=0.0)
        b = tl.load(b_ptrs, mask=offs_k < K and offs_n < N, other=0.0)
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) dot (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # write back
    c_ptrs = C_ptr + batch*stride_Cb + 0*stride_Ch + offs_m[:, None]*stride_Cm + offs_n[None, :]*stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=offs_m < M and offs_n < N)
```

"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑normalize the input tensor (dim‑wise) in fp32.
2. Project the normalized tensor to a hidden dimension `H` using fp16 linear layers:
       left  = x @ W_left
       right = x @ W_right
3. Compute three sigmoid gates (left, right, out) from the same normalized input.
4. Fuse mask (if present) with left/right gates in a Triton kernel:
       left_out  = left  * mask * left_gate
       right_out = right * mask * right_gate
5. Perform the core O(N³·H) reduction using a custom batched GEMM Triton kernel:
       out[b,h,i,j] = Σ_k left_out[b,h,i,k] * right_out[b,h,j,k]
   (right tensor is transposed on the reduction axis before multiplication.)
6. Reshape back to (B,N,N,H) and apply a second LayerNorm over the hidden dimension.
7. Multiply by the output gate and project back to the original channel dimension `C`.
8. Cast the result to float32 (the required output dtype).

Only the mask‑gate fusion and the batched GEMM are implemented in Triton; all
other operations use PyTorch (which internally calls highly‑optimized cuBLAS).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------
# 1. Triton kernel: fuse mask and left/right gates
# --------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # const * left  (B,N,N,H) flattened
    right_ptr,         # const * right (B,N,N,H) flattened
    mask_ptr,          # const * mask  (B,N,N) flattened
    left_gate_ptr,     # const * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # const * right_gate (B,N,N,H) flattened
    left_out_ptr,      # * left_out  (B,N,N,H) flattened
    right_out_ptr,     # * right_out (B,N,N,H) flattened
    H: tl.constexpr,                # hidden dimension (compile‑time)
    total_elements: tl.constexpr,   # total number of elements in left/right
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask and gates:
       left_out  = left  * mask * left_gate
       right_out = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # Load raw tensors
    l = tl.load(left_ptr + offs, mask=valid)
    r = tl.load(right_ptr + offs, mask=valid)
    lg = tl.load(left_gate_ptr + offs, mask=valid)
    rg = tl.load(right_gate_ptr + offs, mask=valid)

    # Index into mask: one mask element per (b,i,k) repeated over H
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=valid)

    # Apply mask and gates
    tl.store(left_out_ptr + offs,  l * m * lg, mask=valid)
    tl.store(right_out_ptr + offs, r * m * rg, mask=valid)


# --------------------------------------------------------------
# 2. Triton kernel: batched GEMM for the TriMul reduction
# --------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    B, H, M, N, K,
    stride_Ab, stride_Ah, stride_Am, stride_Ak,
    stride_Bb, stride_Bh, stride_Bk, stride_Bn,
    stride_Cb, stride_Ch, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute, for each batch b and hidden h:
        C[b,h,i,j] = Σ_k A[b,h,i,k] * B[b,h,k,j]
    All tensors are in fp16, accumulation is fp32.
    """
    pid = tl.program_id(0)

    # Determine block indices
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n
    batch_idx = pid // blocks_per_batch               # combined (b,h) index
    block_idx = pid % blocks_per_batch
    block_m = block_idx // num_blocks_n
    block_n = block_idx % num_blocks_n

    # Decode batch and hidden indices
    b = batch_idx // H
    h = batch_idx % H

    # Offsets for the tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers with correct strides
        a_ptrs = (A_ptr + b * stride_Ab + h * stride_Ah
                  + offs_m[:, None] * stride_Am
                  + offs_k[None, :] * stride_Ak)
        b_ptrs = (B_ptr + b * stride_Bb + h * stride_Bh
                  + offs_k[:, None] * stride_Bk
                  + offs_n[None, :] * stride_Bn)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute dot product; result accumulated in fp32
        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = (C_ptr + b * stride_Cb + h * stride_Ch
              + offs_m[:, None] * stride_Cm
              + offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# 3. Entry point: custom_kernel
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, C] (float32 or float16)
            - mask: [B, N, N] (binary) or None
            - weights: model parameters (LayerNorm, Linear weights, biases)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Load and cast model weights
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))    # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))   # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Prepare mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_fp = mask.to(device=device, dtype=torch.float16)
    else:
        mask_fp = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 7) Fuse mask and gates with Triton
    # ------------------------------------------------------------------
    total_elements = left.numel()                     # B * N * N * H
    BLOCK_SIZE = 1024                                 # tuneable; multiple of 32

    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)
    _mask_gate_fusion_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask_fp.view(-1),
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        left_out_ptr=left_out,
        right_out_ptr=right_out,
        H=hidden_dim,
        total_elements=total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 8) Prepare tensors for batched GEMM
    #    left_perm: (B, H, N, N) where axis 2 is i, axis 3 is k
    #    right_perm: (B, H, N, N) where axis 2 is k, axis 3 is j
    # ------------------------------------------------------------------
    left_perm  = left_out.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_out.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) with k,j order

    out_tensor = torch.empty(
        (B, hidden_dim, N, N), dtype=torch.float16, device=device
    )

    # --------------------------------------------------------------
    # 9) Batched matrix multiplication via Triton
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * hidden_dim * num_blocks_m * num_blocks_n,)

    _batched_matmul_kernel[grid](
        A_ptr=left_perm,
        B_ptr=right_perm,
        C_ptr=out_tensor,
        B=B,
        H=hidden_dim,
        M=N,
        N=N,
        K=N,
        stride_Ab=left_perm.stride(0),
        stride_Ah=left_perm.stride(1),
        stride_Am=left_perm.stride(2),
        stride_Ak=left_perm.stride(3),
        stride_Bb=right_perm.stride(0),
        stride_Bh=right_perm.stride(1),
        stride_Bk=right_perm.stride(2),
        stride_Bn=right_perm.stride(3),
        stride_Cb=out_tensor.stride(0),
        stride_Ch=out_tensor.stride(1),
        stride_Cm=out_tensor.stride(2),
        stride_Cn=out_tensor.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 10) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_tensor.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 11) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 12) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 13) Final projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 14) Cast to float32 (required output dtype)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
