@triton.jit
def batch_matmul_kernel(A_ptr, B_ptr, C_ptr,
                        stride_az, stride_ak, stride_ai,
                        stride_bz, stride_bk, stride_bi,
                        stride_cz, stride_cm, stride_ci,
                        M, N, K,
                        BLOCK_M: tl.constexpr,
                        BLOCK_N: tl.constexpr,
                        BLOCK_K: tl.constexpr):
    # program ids
    pid_z = tl.program_id(2)  # batch index
    pid_m = tl.program_id(0)  # row block
    pid_n = tl.program_id(1)  # col block

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # load A and B blocks
        a_ptrs = A_ptr + pid_z * stride_az + offs_m[:, None] * stride_ak + offs_k[None, :] * stride_ai
        b_ptrs = B_ptr + pid_z * stride_bz + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bi

        a = tl.load(a_ptrs, mask=(offs_m < M) & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K) & (offs_n < N), other=0.0)

        # accumulate

    c_ptrs = C_ptr + pid_z * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_ci

def _batched_matmul(A: torch.Tensor, B: torch.Tensor, block_m=64, block_n=64, block_k=64) -> torch.Tensor:
    # A: (batch, M, K)
    # B: (batch, K, N)
    # both contiguous
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2
    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)
    grid = (tl.cdiv(M, block_m), tl.cdiv(N, block_n), batch)
    # compute strides
    stride_az = A.stride(0)
    stride_ak = A.stride(1)
    stride_ai = A.stride(2)
    stride_bz = B.stride(0)
    stride_bk = B.stride(1)
    stride_bi = B.stride(2)
    stride_cz = C.stride(0)
    stride_cm = C.stride(1)
    stride_ci = C.stride(2)
    batched_matmul_kernel[grid](
        A, B, C,
        stride_az, stride_ak, stride_ai,
        stride_bz, stride_bk, stride_bi,
        stride_cz, stride_cm, stride_ci,
        M, N, K,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return C

"""
custom_kernel: Triton‑accelerated outgoing TriMul forward (AlphaFold3).

Algorithm
---------
1. Layer‑Norm the 4‑D input over the channel dimension (`dim`).
2. Cast the normalized tensor and all linear weights to FP16.
3. Compute left/right projections and left/right/out gates (sigmoid).
4. Fuse mask, projection and gate multiplication with a tiny Triton kernel
   (only when a mask is present and not ignored).
5. Core reduction: for every batch and hidden slice compute
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication (left @ rightᵀ).  Implemented
   with a hand‑written Triton GEMM kernel that works on the flattened
   (batch*hidden, N, N) tensors.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   out‑gate, and linearly project back to the original `dim`.
7. Cast the result back to the original input dtype.

All heavy element‑wise work (mask‑gate fusion) and the N³ reduction are
performed on the GPU with Triton, while the rest uses cheap PyTorch ops.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, projection and gate for left and right tensors
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B, N, N, H)
    right_proj_ptr,  # fp16 * (B, N, N, H)
    left_gate_ptr,   # fp16 * (B, N, N, H)
    right_gate_ptr,  # fp16 * (B, N, N, H)
    mask_ptr,        # fp16 * (B, N, N)
    left_out_ptr,    # fp16 * (B, N, N, H)
    right_out_ptr,   # fp16 * (B, N, N, H)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # map flattened offset to mask offset (each mask element repeats for H hidden channels)
    mask_idx = offs // hidden_dim

    left_proj = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse mask, projection and gate for both left and right tensors.
    All tensors must be fp16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)
    Returns:
        left, right : (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ B) for tensors of shape
# (batch, M, K) and (batch, K, N) → (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_ak, stride_ai,
    stride_bz, stride_bk, stride_bi,
    stride_cz, stride_cm, stride_ci,
    M, N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)          # batch index
    pid_m = tl.program_id(0)          # row block
    pid_n = tl.program_id(1)          # column block

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Pointers with proper strides
        a_ptrs = A_ptr + pid_z * stride_az + offs_m[:, None] * stride_ak + offs_k[None, :] * stride_ai
        b_ptrs = B_ptr + pid_z * stride_bz + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bi

        a = tl.load(a_ptrs, mask=(offs_m < M) & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K) & (offs_n < N), other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = C_ptr + pid_z * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_ci
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m < M) & (offs_n < N))


def _batched_matmul(A: torch.Tensor, B: torch.Tensor,
                    block_m: int = 64, block_n: int = 64, block_k: int = 64) -> torch.Tensor:
    """
    Batched matrix multiplication using Triton.
    A: (batch, M, K)  FP16, contiguous
    B: (batch, K, N)  FP16, contiguous
    Returns C: (batch, M, N) FP16
    """
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    grid = (triton.cdiv(M, block_m), triton.cdiv(N, block_n), batch)

    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        M, N, K,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return C


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None (0/1)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy operations
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast weights to FP16 (all linear layers are bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections & gates (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate = F.linear(x, left_gate_w).sigmoid()   # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid() # (B, N, N, H)
    out_gate = F.linear(x, out_gate_w).sigmoid()     # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (fused via Triton when needed)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            block_size=1024,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask present but ignored → still broadcast if wanted
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM on (B*H, N, N) tensors.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # we need rightᵀ for the equation; transpose last two dims
    right_t = right_flat.permute(0, 2, 1).contiguous()

    # batched matmul via Triton
    out_flat = _batched_matmul(left_flat, right_t,
                               block_m=64, block_n=64, block_k=64)  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
