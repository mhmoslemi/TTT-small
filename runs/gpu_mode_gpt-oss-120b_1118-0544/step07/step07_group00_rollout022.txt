@triton.jit
def _apply_mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total, hidden_dim,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # compute index for mask: each element repeats hidden_dim times
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

def _apply_mask_gate(proj, gate, mask, block_size=1024):

    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _apply_mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    nomask = config.get("nomask", False)
    B,N,_,C = input_tensor.shape
    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)
    # reshape
    x_flat = x.view(B * N * N, dim)

    # combine left_proj + left_gate weights
    left_comb_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    right_comb_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)

    # compute combined linear
    left_comb = F.linear(x_flat, left_comb_weight)  # (BNN, 2H)
    right_comb = F.linear(x_flat, right_comb_weight)

    # split
    left_proj = left_comb[:, :hidden_dim].view(B, N, N, hidden_dim)
    left_gate = torch.sigmoid(left_comb[:, hidden_dim:].view(B, N, N, hidden_dim))
    right_proj = right_comb[:, :hidden_dim].view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(right_comb[:, hidden_dim:].view(B, N, N, hidden_dim))

    # out_gate linear
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_weight)).view(B, N, N, hidden_dim)

    # mask handling
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # not needed
            pass

    # core reduction
    # reshape for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # output layer norm
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    out = out * out_gate

    # final linear
    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_weight)

    # cast back
    out = out.to(input_tensor.dtype)

    return out

"""
TriMul outgoing (AlphaFold3) forward pass – Triton‑accelerated.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (`dim`).
2. Cast to fp16 for compute‑heavy kernels.
3. Fuse the left/right projections with their respective gating linear layers:
   - One matmul computes both projection and gate (weights concatenated).
   - Split the result and apply sigmoid to the gate part.
4. Compute the output gate (single linear + sigmoid).
5. If a mask is present and not ignored, fuse mask, projection and gate
   with a small Triton element‑wise kernel (`proj * gate * mask`).
   Otherwise perform the element‑wise product in PyTorch.
6. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   realized as a batched GEMM over the hidden dimension:
   (B*H, N, N) @ (B*H, N, N)ᵀ.
7. Apply a second Layer‑norm over the hidden dimension, multiply by the
   output gate and project back to the original `dim`.
8. Cast the result back to the input dtype.

Only the mask‑gate fusion is implemented in Triton; the large linear
projections and the reduction use highly‑optimized PyTorch/cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i64 scalar = B*N*N*H
    hidden_dim,        # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    in_bounds = offs < total_elems

    # load projection and gate
    proj = tl.load(proj_ptr + offs, mask=in_bounds, other=0.0, dtype=tl.float16)
    gate = tl.load(gate_ptr + offs, mask=in_bounds, other=0.0, dtype=tl.float16)

    # each mask element repeats `hidden_dim` times in the flattened tensor
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=in_bounds, other=1.0, dtype=tl.float16)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=in_bounds)


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor,
                     mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Fuse projection, gate and mask using Triton.

    All tensors must be half‑precision, contiguous and have shapes:
        proj, gate : (B, N, N, H)
        mask      : (B, N, N)

    Returns:
        out : (B, N, N, H) = proj * gate * mask
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)

    _mask_gate_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

            - mask         : torch.Tensor [B, N, N] (0/1) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive kernels
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    B, N, _, _ = x.shape
    flat_dim = B * N * N

    # ------------------------------------------------------------------
    # 3) Fuse projection + gating linear layers
    # ------------------------------------------------------------------
    # Concatenate projection and gate weights (out_features, in_features)
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, C)

    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, C)

    # Reshape input for a single matmul
    x_flat = x.view(flat_dim, dim)  # (B*N*N, C)

    # Combined matmuls
    left_comb = F.linear(x_flat, left_comb_weight)   # (B*N*N, 2*H)
    right_comb = F.linear(x_flat, right_comb_weight) # (B*N*N, 2*H)

    # Split projection and gate, apply sigmoid to gates
    left_proj = left_comb[:, :hidden_dim].view(B, N, N, hidden_dim)
    left_gate = torch.sigmoid(left_comb[:, hidden_dim:].view(B, N, N, hidden_dim))

    right_proj = right_comb[:, :hidden_dim].view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(right_comb[:, hidden_dim:].view(B, N, N, hidden_dim))

    # ------------------------------------------------------------------
    # 4) Output gate
    # ------------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)  # (H, C)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_weight)).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (fusion via Triton if mask is used)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        # mask ignored if nomask=True or mask is None

    # ------------------------------------------------------------------
    # 6) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
```
