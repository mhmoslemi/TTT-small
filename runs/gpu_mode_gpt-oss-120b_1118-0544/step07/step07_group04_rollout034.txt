@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,  # dims for indexing
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # decode batch and head
    # program id maps to (b, h)
    b = pid // H
    h = pid % H

    # tile indices
    offs_m = tl.arange(0, BLOCK_M) + tl.program_id(1) * BLOCK_M
    offs_n = tl.arange(0, BLOCK_N) + tl.program_id(2) * BLOCK_N

    # compute mask for out of range
    mask_m = offs_m < N
    mask_n = offs_n < N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        # load left tile: shape BLOCK_M x BLOCK_K
        left_tile = tl.load(
            left_ptr + b*stride_left_b + h*stride_left_h + offs_m[:, None] * stride_left_i + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_left_k,

        )
        # load right tile: shape BLOCK_N x BLOCK_K (since we need right[b,j,k,h] with j index n)

            right_ptr + b*stride_right_b + h*stride_right_h + offs_n[:, None] * stride_right_j + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_right_k,

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid = tl.program_id(0)
    pid_m = tl.program_id(1)  # tile row
    pid_n = tl.program_id(2)  # tile col

    b = pid // H
    h = pid % H

    # compute offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b,h)
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = k_offset + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # load left tile (M x K)
        left = tl.load(
            left_base + offs_i[:, None] * stride_l_i + k_range[None, :] * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load right tile transposed (K x N)
        right = tl.load(
            right_base + k_range[:, None] * stride_r_k + offs_j[None, :] * stride_r_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # accumulate
        acc += tl.dot(left, right)

    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass – Triton‑accelerated reduction.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (float32 for stability).
2. Cast to fp16 and perform all linear projections / sigmoid gates in fp16.
3. If a mask is provided, broadcast it over the hidden dimension and
   multiply it into the gated left/right tensors.
4. Core O(N³) reduction:
     out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply (A @ Bᵀ) for each head `h`.
   It is implemented as a tiled Triton kernel (one program per (b,h)
   pair) that accumulates in float32 and writes back in fp16.
5. Layer‑norm over the hidden dimension, multiply by the out‑gate and
   apply the final linear projection back to the original dimension.
6. Return the result in the original dtype (float32).

The kernel fuses the reduction and thus satisfies the “at least part in
Triton” requirement while keeping the surrounding ops in PyTorch for
readability and speed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 * (B, N, N, H)
    right_ptr,         # fp16 * (B, N, N, H)
    out_ptr,           # fp16 * (B, N, N, H)
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute, for every batch `b` and head `h`:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using a tiled GEMM (A @ Bᵀ) where A = left, B = right.
    """
    pid0 = tl.program_id(0)   # combined batch * head
    pid_m = tl.program_id(1)  # tile row (i)
    pid_n = tl.program_id(2)  # tile col (j)

    # decode batch and head
    b = pid0 // H
    h = pid0 % H

    # tile start indices
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N

    # offsets within the tile
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (M, K)
        left = tl.load(
            left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile transposed: (K, N)
        right = tl.load(
            right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(left, right)   # (M, N) accumulation in fp32

    # store result in fp16
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (float32/float16/bool) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim` (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast linear weights to fp16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid (B,N,N,H)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

        right = right * mask_fp

    right = right.contiguous()

    # ------------------------------------------------------------------

    stride_r = right.stride()
    stride_o = out.stride()

    BLOCK_N = 128
    BLOCK_K = 32

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_ln = F.layer_norm(
        out_fp32,
        (H,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
    )
    out_ln = out_ln.to(torch.float16)

    # Gating with out_gate (already fp16)
    out = out_ln * out_gate

    # Final linear projection back to `dim`
    out = F.linear(out, to_out_w)   # shape (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
