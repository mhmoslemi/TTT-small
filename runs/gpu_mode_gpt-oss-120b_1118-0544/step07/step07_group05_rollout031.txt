@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

Now compute:

    pid = tl.program_id(0)

    # Compute total blocks in M and N
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    # total blocks per batch
    blocks_per_batch = num_m * num_n

    batch_id = pid // blocks_per_batch
    block_id = pid % blocks_per_batch

    pid_m = block_id // num_n
    pid_n = block_id % num_n

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # computing mask for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulators
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute k loop
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # compute offsets for a and b
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # mask for k
        mask_k = offs_k < K

        # pointers
        a_ptrs = a_ptr + batch_id * stride_batch_a + offs_m[:, None] * stride_m_a + offs_k[None, :] * stride_k_a
        b_ptrs = b_ptr + batch_id * stride_batch_b + offs_k[:, None] * stride_k_b + offs_n[None, :] * stride_n_b

        # load
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # compute dot
        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_ptr + batch_id * stride_batch_c + offs_m[:, None] * stride_m_c + offs_n[None, :] * stride_n_c
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input (float32) → fp16.
2. Linear projections (left/right) and three gating linear layers (fp16).
3. Fuse mask (if present) and gate multiplications in a Triton kernel
   (in‑place on left/right).
4. Core O(N³·H) reduction: batched GEMM implemented in Triton.
   left[b,i,k,h] · right[b,j,k,h] → out[b,i,j,h].
5. Layer‑norm over hidden dimension, apply output gate, final linear projection.
6. Cast back to the original dtype.

At least two parts of the forward are executed in custom Triton kernels
(mask‑gate fusion and the batched GEMM), meeting the requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: in‑place mask & gate fusion
# --------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (B,N,N,H) flattened
    right_ptr,         # fp16 * right (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Compute mask index (one mask value per hidden element)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Fuse
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


# --------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³·H) reduction
# --------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of blocks in each dimension
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_m * num_n

    batch_id = pid // blocks_per_batch
    block_id = pid % blocks_per_batch
    pid_m = block_id // num_n
    pid_n = block_id % num_n

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for A and B
        a_ptrs = a_ptr + batch_id * stride_batch_a \
                 + offs_m[:, None] * stride_m_a \
                 + offs_k[None, :] * stride_k_a
        b_ptrs = b_ptr + batch_id * stride_batch_b \
                 + offs_k[:, None] * stride_k_b \
                 + offs_n[None, :] * stride_n_b

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_ptr + batch_id * stride_batch_c \
             + offs_m[:, None] * stride_m_c \
             + offs_n[None, :] * stride_n_c
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see code for expected keys)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ----------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to fp16 for compute‑heavy part
    x = x.to(torch.float16)

    # ----------------------------------------------------------
    # 2) Cast linear weights to fp16
    # ----------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ----------------------------------------------------------
    # 3) Linear projections to hidden dimension
    # ----------------------------------------------------------
    left = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ----------------------------------------------------------
    # 4) Compute sigmoid gates
    # ----------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ----------------------------------------------------------
    # 5) Fuse mask and gates (in‑place) via Triton
    # ----------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()            # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates are no longer needed
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ----------------------------------------------------------
    # 6) Core reduction: batched matrix multiplication
    #    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ----------------------------------------------------------
    # Rearrange to (B, H, N, N) and make them contiguous
    left_b = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_b.shape
    batch = B * H

    # Transpose the second operand (right) to match GEMM semantics
    right_t = right_b.permute(0, 1, 3, 2).contiguous()  # (B, H, N, N)

    # Flatten batch dimension
    left_flat = left_b.view(batch, N, N)
    right_flat = right_t.view(batch, N, N)
    out_flat = torch.empty_like(left_flat)

    # Strides (in elements)
    stride_batch_a = left_flat.stride(0)
    stride_m_a = left_flat.stride(1)
    stride_k_a = left_flat.stride(2)

    stride_batch_b = right_flat.stride(0)
    stride_k_b = right_flat.stride(1)
    stride_n_b = right_flat.stride(2)

    stride_batch_c = out_flat.stride(0)
    stride_m_c = out_flat.stride(1)
    stride_n_c = out_flat.stride(2)

    # Launch kernel
    num_m = triton.cdiv(N, 64)
    num_n = triton.cdiv(N, 64)
    total_blocks = batch * num_m * num_n
    grid = (total_blocks,)

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        stride_batch_a, stride_m_a, stride_k_a,
        stride_batch_b, stride_k_b, stride_n_b,
        stride_batch_c, stride_m_c, stride_n_c,
        N, N, N,
        BLOCK_M=64, BLOCK_N=64, BLOCK_K=32,
        num_warps=4,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ----------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----------------------------------------------------------
    # 8) Apply output gate
    # ----------------------------------------------------------
    out = out * out_gate

    # ----------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ----------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ----------------------------------------------------------
    # 10) Cast back to original dtype if necessary
    # ----------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
```
