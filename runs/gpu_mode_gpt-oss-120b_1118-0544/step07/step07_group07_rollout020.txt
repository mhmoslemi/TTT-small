We'll need to compute stride for left & right matrices: each matrix shape (N, N) with stride arrangement: left has shape [B, N, N, H]. Simplify: we can view left as shape (B, H, N, N) via permute. In memory, after a contiguous operation .permute changes stride order, but we can circumvent by generating contiguous view before passing to kernel. For performance, we can compute left.permute(0, 3, 1, 2).contiguous() => shape (B, H, N, N). Similarly right.

But note that only permuting does not change the underlying storage arrangement; after .contiguous() we have new contiguous memory with layout (B, H, N, N). Then we can treat each matrix (b,h) as contiguous for GEMM. We'll then pass pointers and strides.

Alternatively, we can avoid permuting and just compute offsets inside kernel based on known strides: stride over batch = N*N*H, stride over hidden = N*N, stride over i = N, stride over k = 1. That is the standard layout. But easier is to compute left = input in shape (B, H, N, N) contiguous and similarly right. Then treat each as a separate matrix. And then compute contraction across K dimension using standard GEMM approach.

We'll also need to compute out matrix: shape (B, H, N, N). Then after computing, we will reshape back to (B, N, N, H) via .permute.

"""
TriMul (outgoing) forward pass with Triton‑accelerated batched GEMM.
---------------------------------------------------------------
Algorithm
1.  Layer‑norm the input (float32) → x.
2.  Cast to fp16, compute linear projections:
        left_proj, right_proj, left_gate, right_gate, out_gate.
    Apply sigmoid to gates and (optional) mask, then fuse
    gate·mask into left/right.
3.  Rearrange left/right to (B, H, N, N) contiguous view.
4.  Triton kernel `gemm_kernel` computes for every batch b
    and head h:
        out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
    (i.e. a batched GEMM with the second operand transposed).
5.  Reshape output to (B,N,N,H), apply LayerNorm on the hidden
    dimension, multiply by out_gate and a final linear projection
    back to the original channel dimension.
6.  Cast to the input dtype and return.
The kernel fuses the entire matrix multiplication for all heads,
eliminating the overhead of many small torch.bmm calls.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------
# Triton GEMM kernel (batched, no bias)
# ---------------------------------------------------------
@triton.jit
def gemm_kernel(
    left_ptr,                # *fp16   [B, H, N, N]
    right_ptr,               # *fp16   [B, H, N, N]
    out_ptr,                 # *fp16   [B, H, N, N]
    B, H, N,                 # int32 scalars
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,   # int32 strides for left
    stride_r_b, stride_r_h, stride_r_i, stride_r_k,   # int32 strides for right
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,   # int32 strides for out
    BLOCK_M: tl.constexpr,   # tile size in M dimension
    BLOCK_N: tl.constexpr,   # tile size in N dimension
    BLOCK_K: tl.constexpr,   # tile size in K dimension
):
    """Batched GEMM: out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]"""
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_batch = tl.program_id(2)      # batch * head

    # Resolve batch and head indices
    b = pid_batch // H
    h = pid_batch % H

    # Compute start offsets for this tile
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # Range of rows/cols for this tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # Masks for out-of‑bounds rows/cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, N, BLOCK_K):
        k_max = tl.minimum(k + BLOCK_K, N)
        cur_k = k_max - k
        offs_k = k + tl.arange(0, BLOCK_K)

        mask_k = offs_k < N

        # -----------------
        # Load left tile: shape (BLOCK_M, cur_k)
        # left layout: (b, h, i, k)
        a_ptr = (
            left_ptr
            + b * stride_l_b
            + h * stride_l_h
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
        )
        a = tl.load(
            a_ptr,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # -----------------
        # Load right tile (transposed): shape (cur_k, BLOCK_N)
        # right layout: (b, h, j, k)  --> we need (k, j)
        b_ptr = (
            right_ptr
            + b * stride_r_b
            + h * stride_r_h
            + offs_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_i
        )
        b = tl.load(
            b_ptr,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # -----------------
        # Dot product accumulation
        acc += tl.dot(a, b)

    # Store the result (cast back to fp16)
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + h * stride_o_h
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ---------------------------------------------------------
# Entry point
# ---------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, dim)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of parameter tensors (float32)
            - config: dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor of shape (B, N, N, dim) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # Config & constants
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (bias‑less) in fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    # Projections
    left = F.linear(x, left_proj_w)      # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)    # (B,N,N,hidden_dim)

    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4. Apply mask (if any) and fuse gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape (B,N,N) → broadcast over hidden_dim
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * left_gate * mask_fp
        right = right * right_gate * mask_fp
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Rearrange to (B, H, N, N) contiguous layout for GEMM
    # ------------------------------------------------------------------
    # Move hidden_dim to second axis
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left.shape  # _ == N (since shape is B,H,N,N)

    # Allocate output tensor for hidden representation
    out_hidden = torch.empty_like(left)  # (B, H, N, N), fp16

    # ------------------------------------------------------------------
    # 6. Triton batched GEMM kernel launch
    # ------------------------------------------------------------------
    # Strides for left / right (contiguous)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k = left.stride()
    stride_r_b, stride_r_h, stride_r_i, stride_r_k = right.stride()
    stride_o_b, stride_o_h, stride_o_i, stride_o_j = out_hidden.stride()

    # Tile sizes – chosen to fit typical B200 shared memory limits
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * H,                            # batch * head
    )

    gemm_kernel[grid](
        left,
        right,
        out_hidden,
        B,
        H,
        N,
        stride_l_b,
        stride_l_h,
        stride_l_i,
        stride_l_k,
        stride_r_b,
        stride_r_h,
        stride_r_i,
        stride_r_k,
        stride_o_b,
        stride_o_h,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Restore original (B, N, N, H) layout
    # ------------------------------------------------------------------
    out = out_hidden.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (broadcasted)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
