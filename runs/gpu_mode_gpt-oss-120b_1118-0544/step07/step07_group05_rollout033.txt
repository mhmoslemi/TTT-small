"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input (float32 for numerical stability) and cast to FP16.
2. Linear projections to hidden dimension (left/right) and three sigmoid gates
   (left_gate, right_gate, out_gate) using FP16 weights.
3. Fuse mask (if provided) and gate multiplications with a custom Triton kernel
   (in‑place on left/right).
4. Core reduction: batched GEMM (left @ rightᵀ) for each head using a dedicated
   Triton kernel.  The batch dimension is B·H, each GEMM is N×N·N.
5. Second LayerNorm over the hidden dimension, apply the output gate and final
   linear projection back to the original channel dimension.
6. Cast the result back to the original dtype.

Both the mask‑gate fusion and the batched GEMM are implemented in Triton,
while the remaining linear / norm operations use PyTorch (cuBLAS / cuDNN).
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left   (B,N,N,H) flattened
    right_ptr,         # fp16 * right  (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask   (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and left/right gates:
    left  = left  * mask * left_gate
    right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # load values
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # mask broadcast over hidden dim
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # apply and store back
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    Batch, M, N, K,
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_BLOCK_M: tl.constexpr, NUM_BLOCK_N: tl.constexpr,
):
    """Batched GEMM: C[b,m,n] = sum_k A[b,m,k] * B[b,k,n]  (B = Batch)"""
    pid = tl.program_id(0)

    # decode batch / block indices
    blocks_per_batch = NUM_BLOCK_M * NUM_BLOCK_N
    batch_id = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    block_m = block_idx // NUM_BLOCK_N
    block_n = block_idx % NUM_BLOCK_N

    # tile offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # pointers for this block (will add k offset inside loop)
    a_base = a_ptr + batch_id * stride_batch_a
    b_base = b_ptr + batch_id * stride_batch_b
    c_base = c_ptr + batch_id * stride_batch_c

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + offs_k
        k_valid = cur_k < K

        a_ptrs = a_base + (offs_m[:, None] * stride_m_a) + (cur_k[None, :] * stride_k_a)
        b_ptrs = b_base + (cur_k[:, None] * stride_k_b) + (offs_n[None, :] * stride_n_b)

        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & k_valid[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=k_valid[:, None] & (offs_n[None, :] < N), other=0.0)

        # matrix multiplication of the tiles
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_base + (offs_m[:, None] * stride_m_c) + (offs_n[None, :] * stride_n_c)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see code for expected keys)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to fp16 for compute‑heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse mask and gates (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure contiguous tensors
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        mask_fp16 = mask.to(device=device, dtype=torch.float16).contiguous()
        total_elements = left.numel()               # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # gates no longer needed
        del left_gate, right_gate
    else:
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: batched GEMM (left @ rightᵀ) per head
    # ------------------------------------------------------------------
    # permute to (B, H, N, N) and flatten batch*head
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    batch = B * H
    M = N
    K = N
    N_dim = N

    # reshape to (batch, M, K) and (batch, K, N_dim)
    left_mat = left_perm.view(batch, M, K)
    right_mat = right_perm.view(batch, N_dim, K)  # original layout (batch, N, K)

    # allocate output
    out_mat = torch.empty_like(left_mat)

    # compute strides (in elements, not bytes) for the kernels
    stride_batch_a = left_mat.stride(0)
    stride_m_a = left_mat.stride(1)
    stride_k_a = left_mat.stride(2)

    stride_batch_b = right_mat.stride(0)
    stride_k_b = right_mat.stride(2)   # K dimension after transposition
    stride_n_b = right_mat.stride(1)   # N dimension after transposition

    stride_batch_c = out_mat.stride(0)
    stride_m_c = out_mat.stride(1)
    stride_n_c = out_mat.stride(2)

    # tiling parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N_dim + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch,
        M,
        N_dim,
        K,
        stride_batch_a,
        stride_m_a,
        stride_k_a,
        stride_batch_b,
        stride_k_b,
        stride_n_b,
        stride_batch_c,
        stride_m_c,
        stride_n_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        NUM_BLOCK_M=num_blocks_m,
        NUM_BLOCK_N=num_blocks_n,
        num_warps=8,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
