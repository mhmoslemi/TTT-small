def batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_hidden, M, N, K,
    stride_bh, stride_m, stride_k,
    stride_br, stride_n, stride_kr,
    stride_out_bh, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)  # batch-hidden dimension
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block col

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # create mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + pid_bh * stride_bh + offs_m[:, None] * stride_m + offs_k[None, :] * stride_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile transposed: shape (BLOCK_K, BLOCK_N)
        # right layout (batch_hidden, N, K). We need right^T => (K, N)
        b_ptrs = right_ptr + pid_bh * stride_br + offs_k[:, None] * stride_kr + offs_n[None, :] * stride_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Compute tile
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = out_ptr + pid_bh * stride_out_bh + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    mask_n = offs_n < N

        offs_k = k_base + tl.arange(0, BLOCK_K)

    # 7. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

"""
Triton‑accelerated outgoing TriMul (Triangle Multiplicative Update).

The implementation mirrors the reference PyTorch module but offloads two
compute‑intensive steps to custom Triton kernels:

1. **Mask‑gate fusion** – multiplies the projected tensors by their gates
   and (optionally) by a pairwise mask, in‑place.
2. **Batched GEMM** – computes the core contraction
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`,
   i.e. a matrix multiplication `left @ rightᵀ` for every hidden dimension
   `h`.  The kernel treats the `(batch, hidden)` pair as a single batch axis,
   performing a standard blocked GEMM in fp16 with fp32 accumulation.

All heavy arithmetic is performed in fp16 while LayerNorms stay in fp32 for
numerical stability.  The final output is cast back to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@tritone.jit
def gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and gates onto `left` and `right`."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask_valid = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask_valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask_valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask_valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask_valid, other=0.0)

    if HAS_MASK:
        # each mask entry repeats `hidden_dim` times in the flattened view
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask_valid)
    tl.store(right_ptr + offs, right, mask=mask_valid)


@triton.jit
def batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_hidden, M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_n, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out = left @ rightᵀ for a batch of (M×K) × (N×K) matrices."""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (left) : (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + pid_batch * stride_l_batch \
                 + offs_m[:, None] * stride_l_m \
                 + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B tile (right) transposed : (BLOCK_K, BLOCK_N)
        # right layout is (batch, N, K); we need K×N
        b_ptrs = right_ptr + pid_batch * stride_r_batch \
                 + offs_k[:, None] * stride_r_k \
                 + offs_n[None, :] * stride_r_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    c_ptrs = out_ptr + pid_batch * stride_o_batch \
             + offs_m[:, None] * stride_o_m \
             + offs_n[None, :] * stride_o_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

        N,
        stride_l_batch, stride_l_m, stride_l_k,
        stride_r_batch, stride_r_n, stride_r_k,
        stride_o_batch, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection (still fp16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate

    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float16)
    out = F.linear(out, to_out_w)   # (B, N, N, dim) in fp16

    # --------------------------------------------------------------
    # 7. Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(dtype_in)
    return out
```
