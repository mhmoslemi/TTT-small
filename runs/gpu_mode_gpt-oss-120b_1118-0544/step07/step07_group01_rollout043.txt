class TriMul(nn.Module):
    def __init__(self, dim, hidden_dim):
        super().__init__()
        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x, mask):
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

@triton.jit
def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    pid_n = tl.program_id(1)

    # compute batch and block row
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    batch = pid // num_blocks_m
    pid_m = pid % num_blocks_m

    # compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute pointer to start of batch for a and b and c
    a_batch_ptr = a_ptr + batch * batch_stride_a
    b_batch_ptr = b_ptr + batch * batch_stride_b
    c_batch_ptr = c_ptr + batch * batch_stride_c

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # load a block
        a = tl.load(
            a_batch_ptr + (offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak),
            mask=mask_m[:, None] & (k_offset < K),
            other=0.0
        )
        # load b block (note b is transposed so its shape K,N)
        b = tl.load(
            b_batch_ptr + (k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=(k_offset < K)[:, None] & mask_n[None, :],
            other=0.0
        )
        # compute dot product
        acc += tl.dot(a, b)
        # advance? Not needed because we use offsets each iteration
    # cast to desired dtype (fp16)
    c = acc.to(tl.float16)
    # store
    tl.store(c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c, mask=mask_m[:, None] & mask_n[None, :])
    # Done

def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    pid_n = tl.program_id(1)

    num_blocks_m = tl.cdiv(M, BLOCK_M)
    batch = pid // num_blocks_m
    pid_m = pid % num_blocks_m

    # offsets within block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute base pointers for this batch
    a_batch_ptr = a_ptr + batch * batch_stride_a
    b_batch_ptr = b_ptr + batch * batch_stride_b
    c_batch_ptr = c_ptr + batch * batch_stride_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # load A block [M, K]

            a_batch_ptr + (offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak),

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched matrix multiplication C = A @ B for tensors of shape
        A : (BATCH, M, K)
        B : (BATCH, K, N)
    All tensors are assumed to be in FP16; accumulation is performed in FP32.
    """
    pid = tl.program_id(0)
    pid_n = tl.program_id(1)

    # number of blocks in the M dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)

    # batch index and block row index
    batch = pid // num_blocks_m
    pid_m = pid % num_blocks_m

    # compute the offsets inside the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the beginning of the current batch
    a_batch = a_ptr + batch * batch_stride_a
    b_batch = b_ptr + batch * batch_stride_b
    c_batch = c_ptr + batch * batch_stride_c

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # A block: [BLOCK_M, BLOCK_K]
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # B block: [BLOCK_K, BLOCK_N]  (B is already transposed)
        b = tl.load(
            b_batch + (k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        acc += tl.dot(a, b)

    # write back result in FP16
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :]
    )

    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    The kernel fuses the expensive N³ reduction using a custom batched GEMM.
    """

    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # 1) Input LayerNorm over channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w) # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction using Triton batched GEMM:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    This is equivalent to (left_h) @ (right_h).T for each hidden slice.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # bring hidden dimension to batch axis for a single GEMM launch
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch_dim = B * H
    left_flat = left_perm.view(batch_dim, N, N)
    right_flat = right_perm.view(batch_dim, N, N)

    # output buffer
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # 8) Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (batch_dim * num_blocks_m, num_blocks_n)

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_stride_a=left_flat.stride(0),
        stride_am=left_flat.stride(1),
        stride_ak=left_flat.stride(2),
        batch_stride_b=right_flat.stride(0),
        stride_bk=right_flat.stride(1),
        stride_bn=right_flat.stride(2),
        batch_stride_c=out_flat.stride(0),
        stride_cm=out_flat.stride(1),
        stride_cn=out_flat.stride(2),
        M=N,
        N=N,
        K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9) Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    # Cast to float32 for stable LayerNorm and final linear
    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_fp32,
        (H,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
    )
    out_norm = out_norm * out_gate.to(torch.float32)
    out_fp32 = F.linear(out_norm, to_out_w.to(torch.float32))

    # 10) Cast back to the input dtype (typically float32)
    # ------------------------------------------------------------------
    if out_fp32.dtype != dtype_in:
        out_fp32 = out_fp32.to(dtype_in)

    return out_fp32

"""
custom_kernel: Triton‑accelerated forward pass of the outgoing Triangle
Multiplicative Update (TriMul) from AlphaFold3.

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to FP16 and convert all linear weights to FP16.
3. Compute left/right projections and left/right/out gates (sigmoid).
4. Apply the optional mask to the projections **before** gating.
5. Fuse left/right projection with their respective gates.
6. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (left @ rightᵀ) performed by a
   custom Triton kernel that processes all hidden slices in a single launch.
7. Apply a second LayerNorm over the hidden dimension, multiply by the out‑gate,
   and project back to the original channel dimension.
8. Return the result in the original input dtype (float32 or float16).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM C = A @ B for tensors of shape
        A : (BATCH, M, K)   (FP16)
        B : (BATCH, K, N)   (FP16, already transposed)
        C : (BATCH, M, N)   (FP16)

    Accumulation is performed in FP32 for numerical stability.
    """
    pid = tl.program_id(0)
    pid_n = tl.program_id(1)

    num_blocks_m = tl.cdiv(M, BLOCK_M)

    batch = pid // num_blocks_m
    pid_m = pid % num_blocks_m

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a_batch = a_ptr + batch * batch_stride_a
    b_batch = b_ptr + batch * batch_stride_b
    c_batch = c_ptr + batch * batch_stride_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        b = tl.load(
            b_batch + (k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        acc += tl.dot(a, b)

    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :]
    )

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul.  `data` is a tuple:
        (input_tensor, mask, weights, config)

    Returns:
        Tensor of shape [B, N, N, dim] with the same dtype as the input.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    dtype_in = input_tensor.dtype
    device = input_tensor.device

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
