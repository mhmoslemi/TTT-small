@triton.jit
def _final_norm_linear_kernel(
    out_ptr, out_gate_ptr, out_norm_weight_ptr, out_norm_bias_ptr, to_out_weight_ptr, out_final_ptr,
    B, N, H, C,
    stride_out_b0, stride_out_b1, stride_out_b2, stride_out_b3,
    stride_gate_b0, stride_gate_b1, stride_gate_b2, stride_gate_b3,
    stride_norm_w, stride_norm_b,
    stride_weight_c, stride_weight_h,
    stride_out_final_b0, stride_out_final_b1, stride_out_final_b2, stride_out_final_b3,
    EPS: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    # Compute position index
    pid0 = tl.program_id(0)  # position
    pid1 = tl.program_id(1)  # channel block

    total_positions = B * N * N
    if pid0 >= total_positions:
        return

    # compute batch, i, j
    b = pid0 // (N * N)
    tmp = pid0 % (N * N)
    i = tmp // N
    j = tmp % N

    c_start = pid1 * BLOCK_C
    # mask for channel block
    mask_c = c_start + tl.arange(0, BLOCK_C) < C

    # Compute mean and variance across hidden dimension H
    sum_val = tl.float32(0.0)
    sq_sum_val = tl.float32(0.0)
    for off_h in range(0, H, BLOCK_H):
        cur_h = off_h + tl.arange(0, BLOCK_H)
        mask_h = cur_h < H

        # compute offset for out tensor
        # out_ptr + b*stride_out_b0 + i*stride_out_b1 + j*stride_out_b2 + cur_h*stride_out_b3
        ptr = out_ptr + b*stride_out_b0 + i*stride_out_b1 + j*stride_out_b2 + cur_h*stride_out_b3
        val = tl.load(ptr, mask=mask_h, other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=0)
        sq_sum_val += tl.sum(val * val, axis=0)

    mean = sum_val / H
    var = sq_sum_val / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # Prepare accumulator for output channels
    acc = tl.zeros([BLOCK_C], dtype=tl.float32)

    # Loop again over hidden dimension to compute normalized * gate * weight
    for off_h in range(0, H, BLOCK_H):
        cur_h = off_h + tl.arange(0, BLOCK_H)
        mask_h = cur_h < H

        # Load hidden
        out_off = out_ptr + b*stride_out_b0 + i*stride_out_b1 + j*stride_out_b2 + cur_h*stride_out_b3
        hidden = tl.load(out_off, mask=mask_h, other=0.0).to(tl.float32)

        # Load gate
        gate_off = out_gate_ptr + b*stride_gate_b0 + i*stride_gate_b1 + j*stride_gate_b2 + cur_h*stride_gate_b3
        gate = tl.load(gate_off, mask=mask_h, other=0.0).to(tl.float32)

        # Load norm weight and bias for hidden dimension
        wn_off = out_norm_weight_ptr + cur_h*stride_norm_w
        wb_off = out_norm_bias_ptr + cur_h*stride_norm_b
        wn = tl.load(wn_off, mask=mask_h, other=0.0).to(tl.float32)
        wb = tl.load(wb_off, mask=mask_h, other=0.0).to(tl.float32)

        # Normalize
        h_norm = (hidden - mean) * inv_std
        h_norm = h_norm * wn + wb
        # Apply gate
        h_norm_gate = h_norm * gate

        # Load weight block for this hidden block
        # shape (BLOCK_C, BLOCK_H)
        c_idxs = c_start + tl.arange(0, BLOCK_C)
        # mask for channel block already prepared; may need mask for c
        # Compute pointer offsets
        w_ptrs = to_out_weight_ptr + c_idxs[:, None] * stride_weight_c + cur_h[None, :] * stride_weight_h
        # Load with mask for c and h
        mask_wh = mask_c[:, None] & mask_h[None, :]
        w_block = tl.load(w_ptrs, mask=mask_wh, other=0.0).to(tl.float32)  # shape (BLOCK_C, BLOCK_H)

        # Compute dot product: (1, BLOCK_H) * (BLOCK_H, BLOCK_C) -> (1, BLOCK_C)
        # h_norm_gate shape (BLOCK_H)
        # We need to reshape to (1, BLOCK_H)
        prod = tl.dot(h_norm_gate[None, :], w_block)  # shape (1, BLOCK_C)
        # Accumulate
        acc += prod[0]

    # Write out accumulator to output
    # out_final_ptr + b*stride_out_final_b0 + i*stride_out_final_b1 + j*stride_out_final_b2 + c_start*stride_out_final_b3
    out_ptr_final = out_final_ptr + b*stride_out_final_b0 + i*stride_out_final_b1 + j*stride_out_final_b2 + c_start*stride_out_final_b3
    tl.store(out_ptr_final, acc.to(tl.float16), mask=mask_c)

That kernel will compute the final output for each position and channel block. Note: dst dtype should be float16 and then we can cast back.

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Step 1: Input LayerNorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
                     bias=weights["norm.bias"].to(device=device, dtype=torch.float32))

    # Cast to fp16
    x = x.to(torch.float16)

    # Step 2: Linear projections
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left = F.linear(x, left_proj_w)   # [B,N,N,H]
    right = F.linear(x, right_proj_w) # same

    # Step 3: Gates
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Step 4: Mask & gate fusion
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()
        BLOCK = 65536 # large block size but must be multiple of 128? We'll use 1024 as previous.
        grid = (triton.cdiv(total_elements, BLOCK),)
        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        del left_gate, right_gate
    else:
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # Step 5: Core reduction using matmul
    left_b = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)
    # Compute out_b = left_b @ right_b.transpose(-2, -1)
    out_b = torch.matmul(left_b, right_b.transpose(-2, -1))  # (B, H, N, N)
    # Permute back to (B,N,N,H)
    out = out_b.permute(0, 2, 3, 1).contiguous()

    # Step 6: Final Norm and Linear (fused kernel)
    # Prepare weight tensors
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)  # shape (dim, hidden_dim)

    # Prepare output tensor
    out_final = torch.empty((input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2], dim), device=device, dtype=torch.float16)

    B, N, _, H = out.shape
    C = dim

    # Compute strides
    # For out (B,N,N,H)
    stride_out_b0, stride_out_b1, stride_out_b2, stride_out_b3 = out.stride()
    # For out_gate (same shape)
    stride_gate_b0, stride_gate_b1, stride_gate_b2, stride_gate_b3 = out_gate.stride()
    # Norm weight stride: weight is 1D so stride = 1
    stride_norm_w = to_out_norm_w.stride(0) # should be 1
    stride_norm_b = to_out_norm_b.stride(0) # 1
    # For to_out_weight (C, H)
    stride_weight_c, stride_weight_h = to_out_w.stride()
    # For out_final (B,N,N,C)
    stride_out_final_b0, stride_out_final_b1, stride_out_final_b2, stride_out_final_b3 = out_final.stride()

    # Kernel launch grid
    total_positions = B * N * N
    BLOCK_C = 64  # up to 64
    grid = (total_positions, triton.cdiv(C, BLOCK_C))

    _final_norm_linear_kernel[grid](
        out,
        out_gate,
        to_out_norm_w,
        to_out_norm_b,
        to_out_w,
        out_final,
        B, N, H, C,
        stride_out_b0, stride_out_b1, stride_out_b2, stride_out_b3,
        stride_gate_b0, stride_gate_b1, stride_gate_b2, stride_gate_b3,
        stride_norm_w, stride_norm_b,
        stride_weight_c, stride_weight_h,
        stride_out_final_b0, stride_out_final_b1, stride_out_final_b2, stride_out_final_b3,
        EPS=1e-5,
        BLOCK_H=32,
        BLOCK_C=BLOCK_C,
        num_warps=8,
    )

    # Cast back to original dtype
    out_final = out_final.to(orig_dtype)

    return out_final

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the reference PyTorch module but fuses several
memory‑bound steps with custom Triton kernels:

1. Input LayerNorm (float32 → fp16)
2. Linear projections (left / right) to the hidden dimension.
3. Three sigmoid gates (left, right, out).
4. In‑place fusion of mask, left‑gate and right‑gate (triton).
5. Core reduction: batched matrix multiplication
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]   (implemented with cuBLAS).
6. Fused LayerNorm over the hidden dimension, output‑gate multiplication and
   final linear projection back to the original channel dimension (triton).

All heavy compute is performed in fp16; the final result is cast back to the
original input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,               # fp16 * left  (B,N,N,H) flattened
    right_ptr,              # fp16 * right (B,N,N,H) flattened
    mask_ptr,               # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,          # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,         # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,        # hidden dimension
    TOTAL_ELEMS: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask and gates in‑place:
    left  = left  * mask * left_gate
    right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL_ELEMS

    # load data
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # each mask value is repeated H times
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # write back
    tl.store(left_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


@triton.jit
def _final_norm_linear_kernel(
    out_ptr,                # fp16 * out          (B,N,N,H)
    out_gate_ptr,           # fp16 * out_gate     (B,N,N,H)
    norm_w_ptr,             # fp16 * to_out_norm.weight (H)
    norm_b_ptr,             # fp16 * to_out_norm.bias   (H)
    proj_w_ptr,             # fp16 * to_out.weight       (C,H)
    out_final_ptr,          # fp16 * output (B,N,N,C)

    B: tl.int64,
    N: tl.int64,
    H: tl.constexpr,        # hidden dimension (compile‑time)
    C: tl.int64,

    # strides for out and out_gate (in elements)
    STRIDE_OUT_B0, STRIDE_OUT_B1, STRIDE_OUT_B2, STRIDE_OUT_B3,
    STRIDE_GATE_B0, STRIDE_GATE_B1, STRIDE_GATE_B2, STRIDE_GATE_B3,

    # strides for norm weight/bias (1‑D tensors)
    STRIDE_NORM_W, STRIDE_NORM_B,

    # strides for projection weight (C,H)
    STRIDE_PROJ_C, STRIDE_PROJ_H,

    # strides for final output (B,N,N,C)
    STRIDE_OUT_FINAL_B0, STRIDE_OUT_FINAL_B1,
    STRIDE_OUT_FINAL_B2, STRIDE_OUT_FINAL_B3,

    EPS: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    """For each (b,i,j) compute:
       1) LayerNorm over hidden dim H of `out`.
       2) Multiply by `out_gate`.
       3) Linear projection using `proj_w` -> final tensor (B,N,N,C).
    """
    # ------------------------------------------------------------------
    # 1) identify (b,i,j) from the first program ID
    # ------------------------------------------------------------------
    pid = tl.program_id(0)                     # position index
    pid_c = tl.program_id(1)                   # output‑channel block

    total_pos = B * N * N
    if pid >= total_pos:
        return

    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # output‑channel block start
    c_start = pid_c * BLOCK_C
    c_range = c_start + tl.arange(0, BLOCK_C, dtype=tl.int64)
    mask_c = c_range < C

    # ------------------------------------------------------------------
    # 2) compute mean & variance of `out[b,i,j,:]`
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sq_sum_val = tl.float32(0.0)

    # base offsets for out and out_gate
    base_out = b * STRIDE_OUT_B0 + i * STRIDE_OUT_B1 + j * STRIDE_OUT_B2
    base_gate = b * STRIDE_GATE_B0 + i * STRIDE_GATE_B1 + j * STRIDE_GATE_B2

    for off_h in range(0, H, BLOCK_H):
        h_idx = off_h + tl.arange(0, BLOCK_H, dtype=tl.int64)
        mask_h = h_idx < H

        # load hidden values
        ptr = out_ptr + base_out + h_idx * STRIDE_OUT_B3
        val = tl.load(ptr, mask=mask_h, other=0.0).to(tl.float32)

        sum_val += tl.sum(val, axis=0)
        sq_sum_val += tl.sum(val * val, axis=0)

    mean = sum_val / H
    var = sq_sum_val / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # ------------------------------------------------------------------
    # 3) accumulate final output (C‑dim) in registers
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_C], dtype=tl.float32)

    for off_h in range(0, H, BLOCK_H):
        h_idx = off_h + tl.arange(0, BLOCK_H, dtype=tl.int64)
        mask_h = h_idx < H

        # load hidden
        out_ptr_h = out_ptr + base_out + h_idx * STRIDE_OUT_B3
        hidden = tl.load(out_ptr_h, mask=mask_h, other=0.0).to(tl.float32)

        # load gate
        gate_ptr_h = out_gate_ptr + base_gate + h_idx * STRIDE_GATE_B3
        gate = tl.load(gate_ptr_h, mask=mask_h, other=0.0).to(tl.float32)

        # load LayerNorm parameters for this block of hidden dim
        w_norm_ptr = norm_w_ptr + h_idx * STRIDE_NORM_W
        b_norm_ptr = norm_b_ptr + h_idx * STRIDE_NORM_B
        w_norm = tl.load(w_norm_ptr, mask=mask_h, other=0.0).to(tl.float32)
        b_norm = tl.load(b_norm_ptr, mask=mask_h, other=0.0).to(tl.float32)

        # normalize → weight & bias → gate
        h_norm = (hidden - mean) * inv_std
        h_norm = h_norm * w_norm + b_norm
        h_norm = h_norm * gate                     # apply out_gate

        # load projection weight block: shape (BLOCK_C, BLOCK_H)
        w_ptrs = proj_w_ptr + c_range[:, None] * STRIDE_PROJ_C + h_idx[None, :] * STRIDE_PROJ_H
        mask_wh = mask_c[:, None] & mask_h[None, :]
        w_block = tl.load(w_ptrs, mask=mask_wh, other=0.0).to(tl.float32)

        # matmul: (1, BLOCK_H) × (BLOCK_H, BLOCK_C) → (1, BLOCK_C)
        prod = tl.dot(h_norm[None, :], w_block)   # shape (1, BLOCK_C)
        acc += prod[0]                            # accumulate

    # ------------------------------------------------------------------
    # 4) write result
    # ------------------------------------------------------------------
    out_final_base = (b * STRIDE_OUT_FINAL_B0 +
                      i * STRIDE_OUT_FINAL_B1 +
                      j * STRIDE_OUT_FINAL_B2 +
                      c_start * STRIDE_OUT_FINAL_B3)
    tl.store(out_final_base + c_range * STRIDE_OUT_FINAL_B3,
             acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see code for expected keys)
            - config: dict with keys "dim" and "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2) Linear projections to hidden dimension
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w) # (B, N, N, H)

    # --------------------------------------------------------------
    # 3) Compute sigmoid gates
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 4) Fuse mask and gates (in-place) via Triton
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elems = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total_elems, BLOCK),)
        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elems,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        del left_gate, right_gate
    else:
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # --------------------------------------------------------------
    # 5) Core reduction: batched matrix multiplication
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    left_b = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # (B, H, N, N) @ (B, H, N, N)ᵀ → (B, H, N, N)
    out_b = torch.matmul(left_b, right_b.transpose(-2, -1))
    out = out_b.permute(0, 2, 3, 1).contiguous()    # (B, N, N, H)

    # --------------------------------------------------------------
    # 6) Fused LayerNorm + out_gate + final linear projection (Triton)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)  # (C, H)

    B, N, _, H = out.shape
    C = dim

    # allocate final output
    out_final = torch.empty((B, N, N, C), device=device, dtype=torch.float16)

    # strides (in elements)
    s_out_b0, s_out_b1, s_out_b2, s_out_b3 = out.stride()
    s_gate_b0, s_gate_b1, s_gate_b2, s_gate_b3 = out_gate.stride()
    s_norm_w = to_out_norm_w.stride(0)
    s_norm_b = to_out_norm_b.stride(0)
    s_proj_c, s_proj_h = to_out_w.stride()
    s_out_final_b0, s_out_final_b1, s_out_final_b2, s_out_final_b3 = out_final.stride()

    # launch grid: (position, channel‑block)
    total_positions = B * N * N
    BLOCK_C = 64
    grid = (total_positions, triton.cdiv(C, BLOCK_C))

    _final_norm_linear_kernel[grid](
        out,
        out_gate,
        to_out_norm_w,
        to_out_norm_b,
        to_out_w,
        out_final,
        B, N, hidden_dim, C,
        s_out_b0, s_out_b1, s_out_b2, s_out_b3,
        s_gate_b0, s_gate_b1, s_gate_b2, s_gate_b3,
        s_norm_w, s_norm_b,
        s_proj_c, s_proj_h,
        s_out_final_b0, s_out_final_b1, s_out_final_b2, s_out_final_b3,
        EPS=1e-5,
        BLOCK_H=32,
        BLOCK_C=BLOCK_C,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 7) Cast back to original dtype and return
    # --------------------------------------------------------------
    out_final = out_final.to(orig_dtype)
    return out_final
