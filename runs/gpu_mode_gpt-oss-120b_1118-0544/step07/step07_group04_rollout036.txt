    stride_a_batch, stride_b_batch, stride_c_batch,

):

    pid_n = tl.program_id(1)  # col block
    pid_b = tl.program_id(2)  # batch

"""
TriMul (outgoing) forward pass – Triton‑accelerated gating, mask fusion and
batched matrix multiplication.

Algorithm
---------
1. Layer‑norm the 4‑D input over the channel dimension (FP32 for stability).
2. Cast to FP16 for compute‑heavy stages.
3. Linear projections to the hidden dimension for left/right tensors.
4. Compute gate activations (sigmoid) for left/right and the output gate.
5. Fuse each projection with its gate and optional mask using a Triton kernel.
6. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (B·H independent matrix multiplications)
   using a custom Triton kernel.
7. Apply output LayerNorm, multiply by the output gate and project back to the
   original dimension.
8. Cast the result back to the input dtype (usually float32).

All heavy element‑wise operations are performed in Triton; the remaining steps
use PyTorch primitives.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and optional mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,   # fp16 * (B*N*N*H,)
    gate_ptr,   # fp16 * (B*N*N*H,)
    mask_ptr,   # fp16 * (B*N*N,)
    out_ptr,    # fp16 * (B*N*N*H,)
    total_elems,  # int: B*N*N*H
    hidden_dim,   # int: H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    in_bounds = offs < total_elems
    # mask index is the element without the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=in_bounds, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=in_bounds, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=in_bounds, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=in_bounds)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 2048) -> torch.Tensor:
    """
    Fuse proj * gate * mask (mask broadcast over hidden_dim) via Triton.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

# Triton kernel: batched matrix multiplication (fp16 input, fp32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,  # A: (batch, M, K)
    stride_bk, stride_bn,  # B: (batch, K, N)
    stride_cm, stride_cn,  # C: (batch, M, N)
    stride_a_batch, stride_b_batch, stride_c_batch,  # batch strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch index (B*H)

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # batch pointers
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers to blocks of A and B
        a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        # load with masks
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # promote to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # write back result in fp16
    c = acc.to(tl.float16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor,
                         block_m: int = 64,
                         block_n: int = 64,
                         block_k: int = 32) -> torch.Tensor:
    """
    Batched matrix multiplication for tensors:
        a: (B*H, M, K)  fp16
        b: (B*H, K, N)  fp16
    Returns:
        c: (B*H, M, N)  fp16
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_H, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((B_H, M, N), dtype=torch.float16, device=a.device)

    # strides
    stride_am = a.stride(1)
    stride_ak = a.stride(2)
    stride_bk = b.stride(1)
    stride_bn = b.stride(2)
    stride_cm = c.stride(1)
    stride_cn = c.stride(2)

    stride_a_batch = a.stride(0)
    stride_b_batch = b.stride(0)
    stride_c_batch = c.stride(0)

    grid = (triton.cdiv(M, block_m),
            triton.cdiv(N, block_n),
            B_H)

    _batched_matmul_kernel[grid](
        a, b, c,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=block_m, BLOCK_N=block_n, BLOCK_K=block_k,
    )
    return c

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask         : torch.Tensor [B, N, N] (float32/float16/bool) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, C] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over channel dimension (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy part
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast linear weights to FP16 (bias‑free)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B,N,N,hidden_dim)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)  # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5. Gates (sigmoid) (B,N,N,hidden_dim)
    # --------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Fuse projection, gate and (optional) mask via Triton
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) -> (B, H, K=N, N=j)

    left_mat = left_perm.view(B * H, N, N)   # (B*H, M=N, K=N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, K=N, N=N)

    out_mat = _batched_matmul_fp16(left_mat, right_mat)  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
