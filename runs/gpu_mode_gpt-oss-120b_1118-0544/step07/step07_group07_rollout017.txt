def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. LayerNorm
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # 2. Cast to fp16 for compute
    x = x.to(torch.float16)

    # 3. Linear projections (no bias)
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    # shape: [dim, hidden_dim]
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # same shape as left

    # 4. Apply mask & gates (maybe via triton kernel)
    if not nomask:
        # mask shape [B,N,N]
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        # Use triton kernel to fuse mask and gates at once
        _fuse_kernel(...)
    else:
        left = left * left_gate
        right = right * right_gate

    # 5. Core contraction using Triton batched GEMM
    B, N, _, H = left.shape
    total = B * H
    # permute
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(total, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(total, N, N)
    # allocate output
    out_mat = torch.empty_like(left_mat)  # shape (total, N, N), fp16

    # launch triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    # compute grid
    grid = lambda META: (
        triton.cdiv(N, META['BLOCK_M']),   # block rows per matrix
        triton.cdiv(N, META['BLOCK_N']),   # block cols per matrix
        total                              # batch dimension
    )
    matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_am=left_mat.stride(0), # stride between matrices? Actually we need stride for each dimension
        stride_ak=left_mat.stride(2), # Actually left_mat: (total, M, K)
    )

    # stride_ak = left_mat.stride(2) (K stride)
    # For b_ptr: stride_bk = right_mat.stride(1) (K stride), stride_bn = right_mat.stride(2) (N stride)
    # For c_ptr: stride_cm = out_mat.stride(1) (M stride), stride_cn = out_mat.stride(2) (N stride)

    # Reshape out_mat back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Output LayerNorm, gating, final projection
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)

    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)
    # cast back
    return out.to(dtype)

"""
TriMul (outgoing) forward pass accelerated with Triton.

Algorithm
---------
1. LayerNorm on the input (float32).
2. Cast to fp16 for the heavy part.
3. Linear projections (bias‑less) → left, right, left_gate, right_gate, out_gate.
4. Fuse mask and gate multiplications (in‑place) with a small Triton kernel.
5. Core contraction:
   For each batch b and hidden dimension h compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched GEMM (left @ rightᵀ) over B·H matrices of size N×N.
   Implemented with a custom Triton matmul kernel (accumulate in fp32, store fp16).
6. Output LayerNorm, apply out_gate, final linear projection.
7. Cast back to the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask and gate multiplication (in‑place on left/right)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,            # [total_elems] fp16
    left_gate_ptr, right_gate_ptr,  # [total_elems] fp16
    mask_ptr,                       # [mask_elems] fp16 or unused
    total_elems, hidden_dim,        # scalar ints
    HAS_MASK: tl.constexpr,         # bool compile‑time
    BLOCK_SIZE: tl.constexpr,       # int compile‑time
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each mask entry repeats `hidden_dim` times in the flattened view
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM (C = A @ Bᵀ) for many (batch*hidden) matrices
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr,
):
    # Program IDs
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column
    pid_batch = tl.program_id(2)      # which (batch * hidden) matrix

    # Compute start indices of the block
    block_m = pid_m
    block_n = pid_n

    # Compute offsets within the matrix
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current matrix
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch_ptr
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (already transposed): shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_batch_ptr
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Dot product
        acc += tl.dot(a, b)

    # Write out
    c = acc.to(tl.float16)
    tl.store(
        c_batch_ptr
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (bias‑less) in fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)          # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)        # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4. Fuse mask & gates (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B,N,N) -> (B,N,N,1) for broadcasting
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten tensors
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024  # reasonable block size

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # Simple element‑wise gating if no mask
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    total = B * H

    # Permute to (B,H,N,N) then flatten batch*hidden
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(total, N, N)   # (total, M=N, K=N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(total, N, N) # (total, K=N, N)

    # Allocate output
    out_mat = torch.empty_like(left_mat)

    # Strides for the three tensors
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    # Block sizes – tuned for typical N up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    GROUP_SIZE_M = 8  # recommended in Triton matmul example

    # Grid: (grid_m, grid_n, batch)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        total,
    )

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,               # M
        N,               # N
        N,               # K
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=GROUP_SIZE_M,
    )

    # Restore shape (B,N,N,hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------
    # 7. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
