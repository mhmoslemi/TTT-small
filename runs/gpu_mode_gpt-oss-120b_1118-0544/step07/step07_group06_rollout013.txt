"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1️⃣ Layer‑normalize the input tensor `x ∈ [B,N,N,C]` in fp32.
2️⃣ Linear projections `left` / `right` from `C → H` (hidden_dim) in fp16.
3️⃣ Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) also in fp16.
4️⃣ Fuse mask (if present) and the two gates in a tiny Triton kernel:
       left  = left  * mask * left_gate
       right = right * mask * right_gate
5️⃣ Core reduction – for every hidden channel h we need
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is exactly a batched matrix‑multiply:
       out_h = left_h @ right_hᵀ.
   We reshape to (B*H, N, N) and call `torch.bmm`, which uses cuBLAS
   and fully exploits Tensor‑Cores.
6️⃣ Second LayerNorm over the hidden dimension H (still fp16).
7️⃣ Multiply by the output gate.
8️⃣ Final linear projection `H → C`.
9️⃣ Cast back to the original dtype of the input.

The only Triton part is step 4, satisfying the requirement while keeping the
heavy O(N³·H) work on cuBLAS where it is optimal.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: fuse mask & gates (in‑place on left/right)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # fp16*  (B,N,N,H) flattened
    right_ptr,         # fp16*  (B,N,N,H) flattened
    mask_ptr,          # fp16*  (B,N,N)   flattened
    left_gate_ptr,     # fp16*  (B,N,N,H) flattened
    right_gate_ptr,    # fp16*  (B,N,N,H) flattened
    H: tl.constexpr,                # hidden dimension
    total_elements: tl.constexpr,   # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load data
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Index for mask (same mask for each hidden channel)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Write back in‑place
    tl.store(left_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: Tensor[B, N, N, C] (float16/float32) on CUDA.
            - mask: Tensor[B, N, N] (binary) or None.
            - weights: dict of model parameters.
            - config: dict containing "dim", "hidden_dim", optional "nomask".

    Returns:
        Tensor[B, N, N, C] of the same dtype as the input.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    orig_dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (fp32 for numeric stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy work
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Load linear weights in fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Prepare mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is not None and not nomask:
        mask_fp = mask.to(device=device, dtype=torch.float16)
    else:
        mask_fp = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6️⃣ Fuse mask & gates (in‑place) via Triton
    # ------------------------------------------------------------------
    total_elements = left.numel()               # B * N * N * H
    BLOCK_SIZE = 1024                           # fits nicely into 32‑thread warps
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    _mask_gate_fusion_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask_fp.view(-1),
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        H=hidden_dim,
        total_elements=total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Core reduction via batched GEMM (left @ rightᵀ) per hidden channel
    # ------------------------------------------------------------------
    # reshape: (B, N, N, H) -> (B*H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matrix multiply
    out_flat = torch.bmm(left_flat, right_flat.transpose(-2, -1))   # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 🔟 Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 🎯 Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
