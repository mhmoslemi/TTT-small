def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    total_i = tl.cdiv(N, BLOCK_M)
    total_j = tl.cdiv(N, BLOCK_N)
    total_bh = B * H
    bh = pid // (total_i * total_j)
    block_index = pid % (total_i * total_j)
    i_tile = block_index // total_j
    j_tile = block_index % total_j

    # Compute offsets
    b = bh // H
    h = bh % H

    # base offsets for batch and channel
    offset_bh = b * stride_left_b + h * stride_left_h  # same for left/right/out

    # Coordinates
    i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for OOB
    mask_i = i < N
    mask_j = j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Compute pointers
        a_ptrs = left_ptr + offset_bh + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        b_ptrs = right_ptr + offset_bh + k[:, None] * stride_right_k + j[None, :] * stride_right_i

        # Load with mask
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Accumulate in fp32
        acc += tl.dot(a, b)

    # Write back
    out_ptrs = out_ptr + offset_bh + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
custom_kernel: Triton‑accelerated outgoing TriMul forward (AlphaFold3).

Algorithm
---------
1. Input LayerNorm on the channel dimension (dim) – done in PyTorch (fp32).
2. Cast to fp16 for the heavy compute path.
3. Linear projections (left/right) and gating (left_gate, right_gate, out_gate)
   – PyTorch `F.linear` + sigmoid, all in fp16.
4. Optional mask handling: element‑wise multiplication with the mask
   (broadcasted over the hidden dimension).
5. Core triangular multiplication:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM for each hidden slice. Implemented with a
   custom Triton kernel that tiles the i‑j output plane (BLOCK_M×BLOCK_N)
   and loops over k in BLOCK_K steps, accumulating in fp32.
6. Output LayerNorm over the hidden dimension, multiply by out_gate,
   and a final linear projection back to the original dim.
7. Cast back to the input dtype (usually float32) and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    out_ptr,           # fp16 *
    B, N, H,           # i64 scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""

    pid = tl.program_id(0)
    total_i = tl.cdiv(N, BLOCK_M)
    total_j = tl.cdiv(N, BLOCK_N)
    total_bh = B * H

    # Decode program ids
    bh = pid // (total_i * total_j)                # combined batch+hidden index
    block_index = pid % (total_i * total_j)
    i_tile = block_index // total_j
    j_tile = block_index % total_j

    # Decode batch and hidden indices
    b = bh // H
    h = bh % H

    # Base offset for this (b, h)
    offset_bh = b * stride_left_b + h * stride_left_h

    # Tile coordinates
    i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows/cols
    mask_i = i < N
    mask_j = j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + offset_bh + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile: shape (BLOCK_K, BLOCK_N) – note the transposed layout
        b_ptrs = right_ptr + offset_bh + k[:, None] * stride_right_k + j[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)   # fp16×fp16 -> fp32

    # Write back result (cast to fp16)
    out_ptrs = out_ptr + offset_bh + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

# Main entry point
# ----------------------------------------------------------------------

    """

    Args:
        data: (input_tensor, mask, weights, config)

    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape (B, N, N, dim), dtype same as input

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------

    if mask is not None and not nomask:

    else:

            right = right * mask_fp16

    # Release intermediate tensors not needed any more
    del left_proj, right_proj, left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction via Triton GEMM kernel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)          # (B,N,N,hidden_dim) fp16

    # Strides for left/right (same layout)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    # Strides for output
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tiling parameters – chosen for good occupancy on A100/B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_i = (N + BLOCK_M - 1) // BLOCK_M
    total_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * H * total_i * total_j,)

    _tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_left_b,          # right has same strides as left
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Free left/right tensors early
    del left, right

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------

        out = out.to(orig_dtype)

    return out
