def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # Linear projections and gates
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float15??) # use fp16
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = torch.nn.functional.linear(x, left_proj_w)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # Apply mask and gates
    if mask is not None and not nomask:
        # mask shape [B,N,N], need broadcast to hidden_dim
        mask_ = mask.to(device).to(torch.float16).unsqueeze(-1) # [B,N,N,1]
        left = left * left_gate * mask_
        right = right * right_gate * mask_
    else:
        left = left * left_gate
        right = right * right_gate

    # Core contraction using triton
    B, N, _, H = left.shape
    out = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    # compute strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    total_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    total_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * H * total_blocks_m * total_blocks_n,)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # out now fp16 after kernel. Convert to float32 for next steps.
    out = out.to(torch.float32)

    # Output layer norm
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply out_gate (need cast to float32)
    out_gate_f32 = out_gate.to(torch.float32)
    out = out * out_gate_f32

    # Final linear projection
    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)
    out = torch.nn.functional.linear(out, to_out_w)

    return out

def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch_head = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)

    block_m = block_idx // num_blocks_n
    block_n = block_idx % num_blocks_n

    b = batch_head // H
    h = batch_head % H

    # row and col offsets
    row_offsets = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    col_offsets = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    row_mask = row_offsets < N
    col_mask = col_offsets < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_blocks):
        k_offsets = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < N

        # pointers
        left_ptrs = left_ptr + b * stride_l_b + h * stride_l_h + row_offsets[:, None] * stride_l_i + k_offsets[None, :] * stride_l_k
        right_ptrs = right_ptr + b * stride_r_b + h * stride_r_h + col_offsets[None, :] * stride_r_j + k_offsets[:, None] * stride_r_k

        left = tl.load(left_ptrs, mask=row_mask[:, None] & k_mask[None, :], other=0.0)

    out_ptrs = out_ptr + b * stride_o_b + h * stride_o_h + row_offsets[:, None] * stride_o_i + col_offsets[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=row_mask[:, None] & col_mask[None, :])

"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The heavy O(N³·H) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is implemented as a tiled batched matrix multiplication in Triton.
All other operations (LayerNorm, linear projections, sigmoid gates) are
performed with PyTorch for simplicity.  The final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    left  : [B, N, N, H]  (b,i,k,h)
    right : [B, N, N, H]  (b,j,k,h)
    out   : [B, N, N, H]  (b,i,j,h)
    """
    pid = tl.program_id(0)

    # number of tiles in each dimension
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # decode program id -> (batch, head, tile_i, tile_j)
    batch_head = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    tile_i = block_idx // num_blocks_n
    tile_j = block_idx % num_blocks_n

    b = batch_head // H
    h = batch_head % H

    # tile offsets
    i_offsets = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offsets = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_offsets < N
    j_mask = j_offsets < N

    # accumulator in fp16 (sufficient for the model)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_blocks):
        k_offsets = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < N

        # pointers for the current tile
        left_tile_ptr = (
            left_ptr
            + b * stride_l_b
            + h * stride_l_h
            + i_offsets[:, None] * stride_l_i
            + k_offsets[None, :] * stride_l_k
        )
        right_tile_ptr = (
            right_ptr
            + b * stride_r_b
            + h * stride_r_h
            + j_offsets[None, :] * stride_r_j
            + k_offsets[:, None] * stride_r_k
        )

        left = tl.load(
            left_tile_ptr,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0,
        )
        right = tl.load(
            right_tile_ptr,
            mask=k_mask[:, None] & j_mask[None, :],
            other=0.0,
        )
        acc += tl.dot(left, right)  # (BLOCK_M, BLOCK_N)

    # write back
    out_tile_ptr = (
        out_ptr
        + b * stride_o_b
        + h * stride_o_h
        + i_offsets[:, None] * stride_o_i
        + j_offsets[None, :] * stride_o_j
    )
    tl.store(
        out_tile_ptr,
        acc,
        mask=i_mask[:, None] & j_mask[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape (B, N, N, C)
        - mask         : torch.Tensor of shape (B, N, N) or None
        - weights      : dict of model parameters (float32 tensors)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape (B, N, N, C) with dtype torch.float32
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy path
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gates (all in fp16)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)                               # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)                             # (B,N,N,hidden_dim)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))          # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))        # (B,N,N,hidden_dim)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))            # (B,N,N,hidden_dim)

    # -----------------------------------------------------------------
    # 4. Apply mask (if present) and gates
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * left_gate * mask_fp
        right = right * right_gate * mask_fp
    else:
        left = left * left_gate
        right = right * right_gate

    left = left.contiguous()
    right = right.contiguous()
    out_gate = out_gate.contiguous()

    # -----------------------------------------------------------------
    # 5. Core O(N³·H) contraction via Triton‑based tiled GEMM
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # tile sizes (tuned for modern GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * H * grid_m * grid_n

    _tri_mul_kernel[total_programs](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_j,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    out = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate.to(torch.float32)          # apply output gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)
    out = F.linear(out, to_out_w)                   # (B,N,N,dim)

    return out
