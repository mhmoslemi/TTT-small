import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                           total_elems, hidden_dim,
                           BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    mask_idx = offs // hidden_dim
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0) # if mask missing, we bypass kernel
    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

def _apply_gate_and_mask(proj, gate, mask, block_size=2048):
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_fuse_kernel[grid](
        proj, gate, mask, out,
        total, H,
        BLOCK_SIZE=block_size,
    )
    return out

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing Triangle Multiplicative Update
    (TriMul). The kernel fuses projection, gating and optional masking.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # 1. Input LayerNorm (FP32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # 2. Cast to FP16 for compute heavy parts
    x = x.to(torch.float16)

    # 3. Prepare weights in FP16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # 4. Linear projections
    left_proj = F.linear(x, left_proj_w)    # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)  # (B,N,N,hidden_dim)

    # 5. Gate projections + sigmoid
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()  # (B,N,N,hidden_dim)

    # 6. Apply mask and gating (fused via Triton if mask provided)
    if mask is not None and not nomask:
        # Ensure mask is contiguous fp16
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 7. Core reduction: batched GEMM per hidden head
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matmul left @ right^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 8. Output LayerNorm, out_gate multiplication, final linear projection
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # 9. Cast back to input dtype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated gating & mask fusion.

Algorithm
---------
1. Layer‑norm the 4‑D input over the channel dimension (FP32 for stability).
2. Cast to FP16 for the compute‑intensive part.
3. Linear projections → `left_proj`, `right_proj`  (B,N,N,hidden_dim).
4. Gate projections (sigmoid) → `left_gate`, `right_gate`, `out_gate`.
5. Fuse projection, gate and optional mask in a single Triton kernel:
       fused = proj * gate * mask   (mask broadcast over hidden_dim)
   If no mask is provided the fusion reduces to element‑wise product.
6. Core reduction (batched GEMM):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented via a dense batched matrix multiplication (torch.bmm) on
   tensors of shape (B*hidden_dim, N, N); this leverages cuBLAS/Tensor‑cores.
7. Output LayerNorm, multiply by `out_gate`, and a final linear projection
   back to the original dimension.
8. Cast the result back to the input dtype (usually float32).

Only the gating/mask fusion is performed in Triton; the rest uses highly‑
optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # int: B*N*N*H
    hidden_dim,        # int: H
    BLOCK_SIZE: tl.constexpr,
):
    """
    out = proj * gate * mask   (mask broadcast over hidden_dim)
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # bounds
    mask = offs < total_elems

    # mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 2048) -> torch.Tensor:
    """
    Fuse projection, gate and mask using a Triton kernel.

    All tensors must be fp16 and contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)  (binary 0/1, broadcast over H)

    Returns:
        Tensor of shape (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out

    # reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (B*H, N, N) @ (B*H, N, N)^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
