"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to fp16 and fuse the five linear projections (left, right,
   left_gate, right_gate, out_gate) together with sigmoid gates and optional
   masking in a single Triton kernel.  This reduces memory traffic and kernel
   launch overhead.
3. Perform the core multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication over the hidden dimension.
4. Apply a second LayerNorm, multiply by the out‑gate, and a final linear
   projection back to the original channel dimension.
5. Return the result in the original dtype (usually float32).

Only the projection/gate/mask step is implemented in Triton; the remaining
operations use highly‑optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_kernel(
    # pointers
    x_ptr,               # input tensor          (B,N,N,dim)
    left_ptr,            # left output           (B,N,N,hidden)
    right_ptr,           # right output          (B,N,N,hidden)
    out_gate_ptr,        # out_gate output       (B,N,N,hidden)

    w_left_ptr,          # weight matrices (hidden,dim)
    w_right_ptr,
    w_lgate_ptr,
    w_rgate_ptr,
    w_outgate_ptr,

    mask_ptr,            # optional mask (B,N,N)

    # compile‑time integers
    B, N, DIM, H,
    TOTAL,               # B*N*N*H
    APPLY_MASK: tl.constexpr,
    BLOCK_D: tl.constexpr,   # inner dot block size
    BLOCK_SIZE: tl.constexpr,  # threads per program
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    valid = offs < TOTAL

    # ---- decode linear indices ----
    h = offs % H
    idx = offs // H
    b = idx // (N * N)
    rem = idx % (N * N)
    i = rem // N
    j = rem % N

    # base offsets in the flattened tensors
    x_base = ((b * N + i) * N + j) * DIM
    w_base = h * DIM
    out_offset = ((b * N + i) * N + j) * H + h
    mask_offset = (b * N + i) * N + j

    # ---- accumulate linear projections ----
    sum_l = tl.float32(0.0)
    sum_r = tl.float32(0.0)
    sum_lg = tl.float32(0.0)
    sum_rg = tl.float32(0.0)
    sum_og = tl.float32(0.0)

    for d in range(0, DIM, BLOCK_D):
        cur = d + tl.arange(0, BLOCK_D)
        dmask = cur < DIM

        x = tl.load(x_ptr + x_base + cur, mask=dmask, other=0.0)               # fp16
        w_l = tl.load(w_left_ptr + w_base + cur, mask=dmask, other=0.0)
        w_r = tl.load(w_right_ptr + w_base + cur, mask=dmask, other=0.0)
        w_lg = tl.load(w_lgate_ptr + w_base + cur, mask=dmask, other=0.0)
        w_rg = tl.load(w_rgate_ptr + w_base + cur, mask=dmask, other=0.0)
        w_og = tl.load(w_outgate_ptr + w_base + cur, mask=dmask, other=0.0)

        sum_l  += tl.dot(x, w_l)
        sum_r  += tl.dot(x, w_r)
        sum_lg += tl.dot(x, w_lg)
        sum_rg += tl.dot(x, w_rg)
        sum_og += tl.dot(x, w_og)

    # ---- sigmoid gates ----
    lgate = 1.0 / (1.0 + tl.exp(-sum_lg))
    rgate = 1.0 / (1.0 + tl.exp(-sum_rg))
    ogate = 1.0 / (1.0 + tl.exp(-sum_og))

    # ---- apply gates (and mask if present) ----
    left_val  = sum_l * lgate
    right_val = sum_r * rgate

    if APPLY_MASK:
        m = tl.load(mask_ptr + mask_offset, mask=valid, other=1.0)   # fp16 (0/1)
        left_val  = left_val * m
        right_val = right_val * m

    # ---- write results ----
    tl.store(left_ptr      + out_offset, tl.cast(left_val,  tl.float16), mask=valid)
    tl.store(right_ptr     + out_offset, tl.cast(right_val, tl.float16), mask=valid)
    tl.store(out_gate_ptr  + out_offset, tl.cast(ogate,    tl.float16), mask=valid)


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul) forward.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask:        torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights:     dict of model parameters (float32)
            - config:      dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) LayerNorm over input channels
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B,N,N,dim) in original dtype

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 3) Cast all linear weights to fp16 (no bias)
    # ------------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16).contiguous()
    w_right_proj  = weights["right_proj.weight"].to(torch.float16).contiguous()
    w_left_gate   = weights["left_gate.weight"].to(torch.float16).contiguous()
    w_right_gate  = weights["right_gate.weight"].to(torch.float16).contiguous()
    w_out_gate    = weights["out_gate.weight"].to(torch.float16).contiguous()
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16).contiguous()
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16).contiguous()
    w_to_out      = weights["to_out.weight"].to(torch.float16).contiguous()

    B, N, _, _ = x.shape
    H = hidden_dim

    # ------------------------------------------------------------------
    # 4) Allocate output tensors for the fused projection/gate step
    # ------------------------------------------------------------------
    left      = torch.empty((B, N, N, H), dtype=torch.float16, device=device)
    right     = torch.empty((B, N, N, H), dtype=torch.float16, device=device)
    out_gate  = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 5) Prepare mask (if any)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        apply_mask = True
    else:
        # dummy tensor – will never be accessed when apply_mask is False
        mask_fp16 = torch.empty((0,), dtype=torch.float16, device=device)
        apply_mask = False

    # ------------------------------------------------------------------
    # 6) Launch fused projection/gate (+mask) kernel
    # ------------------------------------------------------------------
    total = B * N * N * H
    BLOCK_SIZE = 1024          # threads per program
    BLOCK_D = 32               # dot‑product chunk size

    grid = (triton.cdiv(total, BLOCK_SIZE),)

    _proj_gate_kernel[grid](
        x,
        left,
        right,
        out_gate,
        w_left_proj,
        w_right_proj,
        w_left_gate,
        w_right_gate,
        w_out_gate,
        mask_fp16,
        B, N, dim, H,
        total,
        apply_mask,
        BLOCK_D=BLOCK_D,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ------------------------------------------------------------------
    # 7) Core triangle multiplicative reduction
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM over the hidden dimension.
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate multiplication and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
    )
    out = out * out_gate
    out = F.linear(out, w_to_out)          # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
