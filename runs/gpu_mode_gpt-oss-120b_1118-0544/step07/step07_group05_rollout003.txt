"""
TriMul (outgoing) forward pass accelerated with Triton.

Algorithm
---------
1. Layer‑normalize the input (float32 for stability) and cast to fp16.
2. Linear projections (left/right) from C → H using fp16 GEMM.
3. Compute sigmoid gates (left, right, out) with another linear + sigmoid.
4. Fuse mask, left‑gate and right‑gate in‑place via a Triton kernel
   (left = left * mask * left_gate, right = right * mask * right_gate).
5. Core contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication on tensors of shape
   (B, H, N, N) → (B, H, N, N).
6. Layer‑normalize over the hidden dimension H.
7. Apply the output gate (element‑wise).
8. Final linear projection H → C.
9. Cast back to the original dtype (float32).

The heavy linear layers and the O(N³·H) reduction are delegated to
cuBLAS/torch, while the mask‑gate fusion is performed by Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16* left  (B,N,N,H) flattened
    right_ptr,         # fp16* right (B,N,N,H) flattened
    mask_ptr,          # fp16* mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16* left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16* right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and gates:
    left  = left  * mask * left_gate
    right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # Load tensors
    l  = tl.load(left_ptr + offs,  mask=valid)
    r  = tl.load(right_ptr + offs, mask=valid)
    lg = tl.load(left_gate_ptr + offs, mask=valid)
    rg = tl.load(right_gate_ptr + offs, mask=valid)

    # Mask index (one mask value per hidden element)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=valid)

    # Apply mask and gates
    tl.store(left_ptr + offs,  l * m * lg, mask=valid)
    tl.store(right_ptr + offs, r * m * rg, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or fp16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see expected keys)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with dtype float32.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32)
        if "norm.weight" in weights else None,
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32)
        if "norm.bias" in weights else None,
    )

    # Cast to fp16 for compute‑heavy parts
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse mask and gates via Triton (in‑place)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()          # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.reshape(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates have been applied; free memory
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: batched matrix multiplication over the K dimension
    #    left/right: (B, N, N, H) → (B, H, N, N) → matmul → (B, H, N, N)
    # ------------------------------------------------------------------
    left_b  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Multiply left_b with right_bᵀ over the K dimension
    out_b = torch.matmul(left_b, right_b.transpose(-2, -1))   # (B, H, N, N)

    # Restore original ordering
    out = out_b.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate
    del out_gate  # free

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast back to float32 (required output dtype)
    # ------------------------------------------------------------------
    if out.dtype != torch.float32:
        out = out.to(torch.float32)

    return out
