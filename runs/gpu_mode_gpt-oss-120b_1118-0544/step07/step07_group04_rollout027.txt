@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    B, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # compute batch idx, block row, block col
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    block_m = block_idx // num_blocks_n
    block_n = block_idx % num_blocks_n

    # offsets for A and B
    a_ptrs = a_ptr + batch * stride_a_batch + (block_m * BLOCK_M + tl.arange(0, BLOCK_M))[:, None] * stride_a_m + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    # For B we need its transpose: we want B^T => need to multiply A (M,K) with B (N,K) as B^T (K,N) => we can load B tile as (BLOCK_K, BLOCK_N)
    b_ptrs = b_ptr + batch * stride_b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_b_k + (block_n * BLOCK_N + tl.arange(0, BLOCK_N))[None, :] * stride_b_n

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=( (block_m*BLOCK_M + tl.arange(0, BLOCK_M)) < M)[:, None] &
                              (tl.arange(0, BLOCK_K) < K - k*BLOCK_K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K) < K - k*BLOCK_K)[:, None] &
                              ((block_n*BLOCK_N + tl.arange(0, BLOCK_N)) < N)[None, :],
                    other=0.0)
        # compute dot product across K dimension
        acc += tl.dot(a, b)
        # update pointers for next K block
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k
    # store result
    c_ptrs = c_ptr + batch * stride_c_batch + (block_m * BLOCK_M + tl.arange(0, BLOCK_M))[:, None] * stride_c_m + (block_n * BLOCK_N + tl.arange(0, BLOCK_N))[None, :] * stride_c_n
    mask_c = ( (block_m*BLOCK_M + tl.arange(0, BLOCK_M)) < M)[:, None] & ((block_n*BLOCK_N + tl.arange(0, BLOCK_N)) < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)
```

    b_ptrs = b_batch_ptr + b_offset_row[:, None] * stride_b_k + b_offset_col[None, :] * stride_b_n

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input over the channel dimension (`dim`).
2. Cast to FP16 for compute‑heavy stages.
3. Linear projections (`left_proj`, `right_proj`) and gating projections
   (`left_gate`, `right_gate`, `out_gate`).
4. Fuse projection, gate and optional mask with a tiny Triton kernel
   (element‑wise multiplication, broadcast mask over the hidden dim).
5. Batched matrix multiplication:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   performed by a custom Triton GEMM kernel on tensors of shape
   (B*H, N, N) → (B*H, N, N).  This replaces the expensive `torch.bmm`
   and keeps the whole reduction on the device.
6. Output LayerNorm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.
7. Cast the result back to the original dtype (usually FP32).

The kernel operates on FP16 tensors; all LayerNorms and final linear
projection are executed in FP16 and the final result is cast to the
input dtype for correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # int: B*N*N*H
    hidden_dim,        # int: H
    BLOCK_SIZE: tl.constexpr,
):
    """out = proj * gate * mask   (mask broadcast over hidden_dim)"""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    # each hidden_dim elements share the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj, gate, mask, block_size=2048):
    """
    Fuse projection, gate and mask using Triton.
    All tensors are FP16 and contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)  (0/1, broadcast over H)
    Returns (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B^T for a batch of matrices.
    A : (BATCH, M, K)      row‑major
    B : (BATCH, N, K)      row‑major (will be transposed inside)
    C : (BATCH, M, N)
    All tensors are FP16; accumulation is in FP32.
    """
    pid = tl.program_id(0)

    # number of blocks in each dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch = pid // (num_blocks_m * num_blocks_n)
    block = pid % (num_blocks_m * num_blocks_n)
    block_m = block // num_blocks_n
    block_n = block % num_blocks_n

    # base pointers for the current batch
    a_batch = a_ptr + batch * stride_a_batch
    b_batch = b_ptr + batch * stride_b_batch
    c_batch = c_ptr + batch * stride_c_batch

    # row / column offsets for the output tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for the output tile
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_out = mask_m[:, None] & mask_n[None, :]

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        # column offsets for A  (K dimension)
        offs_k = k_start + tl.arange(0, BLOCK_K)
        # masks for the current K block
        mask_k_a = offs_k < K
        mask_k_b = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k_a[None, :], other=0.0)

        # B tile: we need B^T, so we load (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_k_b[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)        # (BLOCK_M, BLOCK_N)

    # store result
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_out)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (float32/float16/bool) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim` (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast linear weights to FP16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B, N, N, H)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid (B, N, N, H)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Apply mask + gating (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is FP16 and broadcastable
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        # No mask → simple element‑wise multiplication
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM via Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) – contiguous layout required by the kernel
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # strides for contiguous tensors (row‑major)
    stride_a_batch = left_flat.stride(0)
    stride_a_m     = left_flat.stride(1)   # = N
    stride_a_k     = left_flat.stride(2)   # = 1
    stride_b_batch = right_flat.stride(0)
    stride_b_n     = right_flat.stride(1)  # = N
    stride_b_k     = right_flat.stride(2)  # = 1
    stride_c_batch = out_flat.stride(0)
    stride_c_m     = out_flat.stride(1)    # = N
    stride_c_n     = out_flat.stride(2)    # = 1

    # Tiling parameters – tuned for Ampere/Hopper GPUs
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * H * num_blocks_m * num_blocks_n,)

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
