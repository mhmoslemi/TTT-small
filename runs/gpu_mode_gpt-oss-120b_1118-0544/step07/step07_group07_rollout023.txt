"""
custom_kernel: Triton‑accelerated outgoing TriMul (Triangle Multiplicative Update).

Algorithm
---------
1. Layer‑norm the input tensor (float32) → keep high‑precision statistics.
2. Cast to fp16 for the heavy work.
3. Linear projections (bias‑less) produce left/right tensors of shape
   [B, N, N, H] and three gate tensors (left/right/out) of the same shape.
4. Fuse mask (if present) and the left/right gates in a small Triton kernel.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is implemented as a batched GEMM (left @ rightᵀ) with a custom Triton
   kernel that processes all batches and hidden channels in one launch.
6. Apply the output LayerNorm, multiply by the out‑gate and project back
   to the original dimension with a final linear layer.
7. Cast back to the input dtype and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. Mask & gate fusion kernel
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened, fp16
    left_gate_ptr, right_gate_ptr,     # same shape, fp16
    mask_ptr,                          # [B,N,N] flattened, fp16
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # load
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    if HAS_MASK:
        # mask repeats `hidden_dim` times in the flattened view
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)
        left = left * lgate * m
        right = right * rgate * m
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


# ----------------------------------------------------------------------
# 2. Core TriMul contraction kernel (batched GEMM)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,    # const fp16 *
    right_ptr,   # const fp16 *
    out_ptr,     # fp16 *
    B, N, H,     # int32
    # strides for left
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # strides for out
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)      # tile row   (i)
    pid_n = tl.program_id(1)      # tile col   (j)
    pid_bh = tl.program_id(2)     # batch * hidden index

    b = pid_bh // H               # batch index
    h = pid_bh % H                # hidden channel index

    # offsets within the N×N matrix
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        k_idx = k_base + offs_k
        mask_k = k_idx < N

        # ----- load left tile: shape (BLOCK_M, BLOCK_K) -----
        # left[b, i, k, h]
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + (k_idx)[None, :] * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16 -> will be promoted in dot

        # ----- load right tile (transposed): shape (BLOCK_K, BLOCK_N) -----
        # we need right[b, j, k, h] but we load it as (k, j)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + (k_idx)[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i
            + h * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )  # fp16

        # accumulate
        acc += tl.dot(left, right)   # result is fp32

    # ----- write back result -----
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# 3. Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape (B, N, N, C)
            - mask         : torch.Tensor of shape (B, N, N) (may be None)
            - weights      : dict of torch.Tensor parameters (float32)
            - config       : dict with keys
                * "dim"         : input channel dimension C
                * "hidden_dim"  : hidden dimension H
                * "nomask"      : optional bool (True → ignore mask)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy work
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (bias‑less) in fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)          # (B,N,N,H)
    right = F.linear(x, right_proj_w)        # (B,N,N,H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4. Fuse mask and gates (in‑place) via Triton when needed
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B,N,N) → fp16
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten everything
        total = left.numel()
        BLOCK = 1024
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        lgate_f = left_gate.contiguous().view(-1)
        rgate_f = right_gate.contiguous().view(-1)
        mask_f = mask_fp.contiguous().view(-1)

        grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)
        _gate_mask_kernel[grid](
            left_f,
            right_f,
            lgate_f,
            rgate_f,
            mask_f,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # simple element‑wise gating when no mask
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    out = torch.empty_like(left)                 # (B,N,N,H) fp16
    # ensure contiguous memory
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tile sizes – chosen to fit in shared memory / registers
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        B * H,                     # pid_bh
    )
    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate                # apply output gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)       # (B,N,N,dim) fp16

    # ------------------------------------------------------------------
    # 7. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
