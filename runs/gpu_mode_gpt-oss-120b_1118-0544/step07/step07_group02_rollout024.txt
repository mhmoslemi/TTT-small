"""
Optimized outgoing TriMul forward pass.
...
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # index of (b,i,j) = offs // hidden_dim
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C) (float32)
            - mask: torch.Tensor of shape (B, N, N) (bool/float) or None
            - weights: dict containing model parameters (float32)
            - config: dict with keys:
                * "dim" (int): input channel dimension C
                * "hidden_dim" (int): hidden dimension H
                * "nomask" (bool, optional): if True, ignore mask

    Returns:
        torch.Tensor of shape (B, N, N, C) with dtype torch.float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy path
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections & gates (fp16)
    # -----------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)                # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)              # (B,N,N,hidden_dim)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 4. Fuse mask + gates (Triton kernel) or simple multiplication
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure contiguous tensors for in‑place kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp = mask.to(device).to(torch.float16).contiguous()

        # Flatten tensors
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_fp.view(-1)   # shape (B*N*N,)

        total = left_flat.numel()

"""
Optimized outgoing TriMul forward pass.

The implementation follows the PyTorch reference but moves the mask‑gate
fusion into a small Triton kernel and performs the heavy O(N³·H) contraction
with a batched GEMM (leveraging cuBLAS via torch.bmm).  The overall flow is:

1. Layer‑norm the input (float32 for numerical stability).
2. Cast to fp16 for the compute‑intensive path.
3. Linear projections for “left”, “right” and three sigmoid gates.
4. Fuse mask, left‑gate and right‑gate in‑place with a Triton kernel
   (skipped when no mask or when `nomask` is True).
5. Core contraction: for each hidden channel h,
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   Implemented as a batched matrix multiplication:
   (B*H, N, N) = (B*H, N, N) @ (B*H, N, N)ᵀ.
6. Apply a second LayerNorm, multiply by the output gate, and a final linear
   projection back to the original channel dimension.
7. Return the result in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # [B,N,N,H] flattened
    right_ptr,              # same shape
    left_gate_ptr,          # same shape
    right_gate_ptr,         # same shape
    mask_ptr,               # [B,N,N] flattened, broadcast over H
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """
    In‑place fusion kernel:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    If HAS_MASK is False the mask is ignored (treated as 1).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    if HAS_MASK:
        # each hidden‑dim element shares the same mask entry
        mask_idx = offs // hidden_dim                # index in [B,N,N]
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)

    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C) (float32)
            - mask: torch.Tensor of shape (B, N, N) (bool/float) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape (B, N, N, C) with dtype torch.float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy path
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections & sigmoid gates (fp16)
    # -----------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)                 # (B, N, N, H)
    right = F.linear(x, right_proj_w)               # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 4. Fuse mask and gates (Triton) or simple element‑wise multiplication
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure contiguous storage for in‑place modification
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp = mask.to(device).to(torch.float16).contiguous()

        # Flatten tensors (no copy because they are already contiguous)
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)

        )

        right = right_flat.view_as(right)
    else:

        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched matrix multiplication
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) = (B*H, N, N) @ (B*H, N, N)ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = F.linear(out, to_out_w)

    # -----------------------------------------------------------------
    # 7. Cast back to float32 and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
