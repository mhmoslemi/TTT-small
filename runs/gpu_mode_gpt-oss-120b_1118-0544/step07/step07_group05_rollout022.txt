@triton.jit
def _apply_mask_gate_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    H: tl.constexpr,
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    l_proj = tl.load(left_proj_ptr + offs, mask=mask)
    l_gate = tl.load(left_gate_ptr + offs, mask=mask)
    r_proj = tl.load(right_proj_ptr + offs, mask=mask)
    r_gate = tl.load(right_gate_ptr + offs, mask=mask)

    # Compute mask index: each hidden dim element belongs to same position in mask
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    left_out = l_proj * l_gate * m
    right_out = r_proj * r_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)
```

    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    B, N, _, _ = x.shape

    # Flatten for linear layers
    x_flat = x.reshape(-1, dim)  # (B*N*N, dim)

    # Concatenate left_proj and left_gate weights
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*hidden, dim)
    # Linear: (B*N*N, 2*hidden) = x_flat @ left_comb_w.T
    left_comb = torch.nn.functional.linear(x_flat, left_comb_w)  # shape (B*N*N, 2*hidden)

    # split
    left_proj = left_comb[:, :hidden_dim]    # (B*N*N, hidden)
    left_gate = left_comb[:, hidden_dim:]    # (B*N*N, hidden)
    left_gate = torch.sigmoid(left_gate)

    # right
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)
    right_comb = torch.nn.functional.linear(x_flat, right_comb_w)
    right_proj = right_comb[:, :hidden_dim]
    right_gate = right_comb[:, hidden_dim:]
    right_gate = torch.sigmoid(right_gate)

    # out_gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_w))

    # Reshape back to (B,N,N,hidden)
    left_proj = left_proj.reshape(B, N, N, hidden_dim)
    left_gate = left_gate.reshape(B, N, N, hidden_dim)
    right_proj = right_proj.reshape(B, N, N, hidden_dim)
    right_gate = right_gate.reshape(B, N, N, hidden_dim)
    out_gate = out_gate.reshape(B, N, N, hidden_dim)

    # Mask processing: ensure mask exists
    if mask is None or nomask:
        mask_fp16 = torch.ones((B, N, N), device=device, dtype=torch.float16)
    else:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)

    # Fuse mask and gates using Triton kernel
    total_elems = left_proj.numel()  # B*N*N*hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)

    # Allocate output tensors for left and right after gating
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    _apply_mask_gate_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask_fp16,
        left,
        right,
        hidden_dim,
        total_elems,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    # cleaning up intermediate tensors
    del left_proj, left_gate, right_proj, right_gate

    # Core reduction
    # Permute to (B, hidden, N, N)
    left_b = left.permute(0, 3, 1, 2).contiguous()
    right_b = right.permute(0, 3, 1, 2).contiguous()

    # Reshape for batched matmul: (B*hidden, N, N)
    left_b = left_b.view(B * hidden_dim, N, N)
    right_b = right_b.view(B * hidden_dim, N, N)

    # out_b: (B*hidden, N, N)
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))

    # Reshape back
    out = out_b.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Second LayerNorm over hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to original dtype if needed
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input (float32) → fp16.
2. Fuse the left/right projections and their gates:
   * Compute (proj || gate) = x @ [W_proj; W_gate]^T in a single matmul.
   * Split the result and apply sigmoid to the gate portion.
3. Compute the output gate (x @ W_out_gate^T + sigmoid).
4. Fuse mask, left‑gate and right‑gate in one Triton kernel:
   left  = left_proj  * left_gate  * mask
   right = right_proj * right_gate * mask
5. Core reduction – batched matrix multiplication over the sequence axis:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
6. Layer‑normalize over the hidden dimension.
7. Multiply by the output gate.
8. Final linear projection back to the original channel dimension.
9. Cast back to the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_mask_gate_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    H: tl.constexpr,                 # hidden dimension
    TOTAL_ELEMS: tl.constexpr,       # B * N * N * H
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse mask and gate multiplications for left and right tensors.
    Each element computes:
        left  = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask
    mask is broadcasted over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL_ELEMS

    # Load tensors
    lproj = tl.load(left_proj_ptr + offs, mask=mask)
    lgate = tl.load(left_gate_ptr + offs, mask=mask)
    rproj = tl.load(right_proj_ptr + offs, mask=mask)
    rgate = tl.load(right_gate_ptr + offs, mask=mask)

    # Convert linear offset to mask offset (mask has shape B*N*N)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply mask and gates
    left_val = lproj * lgate * m
    right_val = rproj * rgate * m

    # Store results
    tl.store(left_out_ptr + offs, left_val, mask=mask)
    tl.store(right_out_ptr + offs, right_val, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2) Prepare weight tensors (fp16)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    B, N, _, _ = x.shape  # (B, N, N, C)

    # --------------------------------------------------------------
    # 3) Flatten and compute fused projections + gates
    # --------------------------------------------------------------
    x_flat = x.reshape(-1, dim)                         # (B*N*N, C)

    # Left: project + gate in one matmul
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, C)
    left_comb = F.linear(x_flat, left_comb_w)                    # (B*N*N, 2*H)
    left_proj = left_comb[:, :hidden_dim]          # (B*N*N, H)
    left_gate = torch.sigmoid(left_comb[:, hidden_dim:])  # (B*N*N, H)

    # Right: project + gate in one matmul
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, C)
    right_comb = F.linear(x_flat, right_comb_w)                    # (B*N*N, 2*H)
    right_proj = right_comb[:, :hidden_dim]          # (B*N*N, H)
    right_gate = torch.sigmoid(right_comb[:, hidden_dim:])  # (B*N*N, H)

    # Output gate (separate)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w))  # (B*N*N, H)

    # Reshape back to [B, N, N, H]
    left_proj = left_proj.reshape(B, N, N, hidden_dim)
    left_gate = left_gate.reshape(B, N, N, hidden_dim)
    right_proj = right_proj.reshape(B, N, N, hidden_dim)
    right_gate = right_gate.reshape(B, N, N, hidden_dim)
    out_gate = out_gate.reshape(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 4) Fuse mask, left‑gate and right‑gate using Triton
    # --------------------------------------------------------------
    if mask is None or nomask:
        mask_fp16 = torch.ones((B, N, N), device=device, dtype=torch.float16)
    else:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)

    # Allocate output tensors for gated left/right
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    total_elements = left_proj.numel()                     # B * N * N * H
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    _apply_mask_gate_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask_fp16.view(-1),
        left,
        right,
        hidden_dim,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    # Release intermediate tensors
    del left_proj, left_gate, right_proj, right_gate

    # --------------------------------------------------------------
    # 5) Core reduction: batched matrix multiplication
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left_b = left.permute(0, 3, 1, 2).contiguous()
    right_b = right.permute(0, 3, 1, 2).contiguous()

    # Merge batch and hidden dimensions for strided batched GEMM
    left_b = left_b.view(B * hidden_dim, N, N)
    right_b = right_b.view(B * hidden_dim, N, N)

    # Batched matmul (uses Tensor Cores for fp16)
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))   # (B*H, N, N)

    # Restore shape (B, N, N, H)
    out = out_b.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6) Second LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 7) Apply output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 8) Final linear projection back to original channel dimension
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # --------------------------------------------------------------
    # 9) Cast back to original dtype if needed
    # --------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
