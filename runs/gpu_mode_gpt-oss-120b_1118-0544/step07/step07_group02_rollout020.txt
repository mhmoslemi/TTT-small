def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # fp32

    # cast to fp16
    x = x.to(torch.float16)

    # Projections:
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = torch.nn.functional.linear(x, left_proj_w)  # shape B,N,N,H
    right = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Mask apply
    if mask is not None and not nomask:
        # Ensure mask dtype is float16, shape B,N,N
        mask_fp = mask.to(torch.float16)
        # Broadcast mask to hidden dim
        mask_fp = mask_fp.unsqueeze(-1)  # B,N,N,1
        left = left * left_gate * mask_fp
        right = right * right_gate * mask_fp
    else:
        left = left * left_gate
        right = right * right_gate

    # Core contraction using batched GEMM
    B, N, _, H = left.shape
    # reshape for batched matmul
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.reshape(B, H, N, N).permute(0, 2, 3, 1)  # B,N,N,H

    # Output layernorm
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # apply out_gate
    out = out * out_gate

    # final linear
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back
    out = out.to(torch.float32)

    return out

@triton.jit
def fuse_mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    off = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = off < total_elems

    # load
    left = tl.load(left_ptr + off, mask=mask, other=0.0)
    right = tl.load(right_ptr + off, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + off, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + off, mask=mask, other=0.0)

    # compute mask index: each hidden_dim elements share same mask
    mask_idx = off // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left = left * left_gate * mask_val
    right = right * right_gate * mask_val

    tl.store(left_ptr + off, left, mask=mask)
    tl.store(right_ptr + off, right, mask=mask)

"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the reference PyTorch version but:
  • Uses fp16 for the compute‑heavy path.
  • Fuses mask application and gate multiplication for the left/right
    tensors in an in‑place Triton kernel.
  • Performs the O(N³·H) contraction as a batched half‑precision GEMM
    (torch.bmm), which is highly optimized in cuBLAS.
  • All layer‑norms and final projection are kept in fp32‑fp16 mixed precision.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape
    mask_ptr,                          # [B,N,N] flattened, broadcast over H
    total_elems, hidden_dim,           # runtime scalars
    BLOCK_SIZE: tl.constexpr,          # compile‑time block size
):
    """In‑place fusion:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
       Mask is broadcast over the hidden dimension H.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # each hidden‑dim element shares the same mask entry
    mask_idx = offs // hidden_dim                # index in [B,N,N]
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left = left * left_gate * mask_val
    right = right * right_gate * mask_val

    # store back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, mask is ignored

    Returns:
        torch.Tensor of shape (B, N, N, C) with dtype torch.float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for the compute‑heavy path
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gates (all in fp16)
    # -----------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)                 # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)               # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # -----------------------------------------------------------------
    # 4. Fuse mask and gate multiplication via a Triton kernel
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # Prepare tensors for the kernel
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # Flatten everything (row‑major order)
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _mask_gate_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back to original layout
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
        # gate tensors are no longer needed
        del left_gate, right_gate, left_gate_flat, right_gate_flat, mask_flat, mask_fp
    else:
        # Simple elementwise multiplication when no mask is used
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # -----------------------------------------------------------------
    # 5. Core O(N³·H) contraction via batched GEMM (L @ Rᵀ)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched matmul
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) = (B*H, N, N) @ (B*H, N, N).T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate                                 # apply output gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)                        # (B,N,N,dim)

    # -----------------------------------------------------------------
    # 7. Cast back to float32 and return
    # -----------------------------------------------------------------
    out = out.to(torch.float32)
    return out
