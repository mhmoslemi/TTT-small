    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets

"""
custom_kernel: Triton‑accelerated outgoing TriMul forward (AlphaFold‑3).

Algorithm
---------
1. Layer‑Norm the input over the channel dimension (dim) – FP32 → FP16.
2. Fuse the two linear projections (proj & gate) per side by concatenating
   the weights and splitting the result, reducing the number of GEMM calls.
3. Apply sigmoid to the gate logits and multiply with the projected values.
4. Optionally apply the pairwise mask (broadcasted over the hidden dim).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM using a custom Triton kernel that works on
   the flattened batch*hidden dimension, thereby avoiding extra copies.
6. Layer‑Norm over the hidden dimension, multiply by the output gate and
   project back to the original channel dimension.
7. Cast the result back to the input dtype (usually float32).

The implementation keeps the heavy linear work in FP16 to exploit Tensor
Cores, while the custom Triton kernel performs the O(N³) reduction with
FP16 inputs and FP32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ B  (A: (B, M, K), B: (B, K, N))
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,    # strides for A
    stride_Bb, stride_Bk, stride_Bn,    # strides for B
    stride_Cb, stride_Cm, stride_Cn,    # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # --------------------------------------------------------------
    # Compute offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers for this batch
    a_ptrs = A_ptr + batch_id * stride_Ab \
                     + offs_m[:, None] * stride_Am \
                     + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + batch_id * stride_Bb \
                     + offs_k[:, None] * stride_Bk \
                     + offs_n[None, :] * stride_Bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main loop over K
    # --------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        # fp16 * fp16 -> fp32 accumulation
        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # --------------------------------------------------------------
    # Write back
    # --------------------------------------------------------------
    c_ptrs = C_ptr + batch_id * stride_Cb \
                     + offs_m[:, None] * stride_Cm \
                     + offs_n[None, :] * stride_Cn
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper to launch the batched GEMM kernel
# ----------------------------------------------------------------------
def batched_gemm(A: torch.Tensor, B: torch.Tensor, block_m=64, block_n=64, block_k=64) -> torch.Tensor:
    """
    A: (B, M, K)  fp16, contiguous
    B: (B, K, N)  fp16, contiguous
    Returns C: (B, M, N) fp16
    """
    assert A.is_contiguous() and B.is_contiguous()
    Bdim, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((Bdim, M, N), dtype=torch.float16, device=A.device)

    # strides
    stride_Ab, stride_Am, stride_Ak = A.stride()
    stride_Bb, stride_Bk, stride_Bn = B.stride()
    stride_Cb, stride_Cm, stride_Cn = C.stride()

    grid = (
        Bdim,
        (M + block_m - 1) // block_m,
        (N + block_n - 1) // block_n,
    )

    batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_Ab,
        stride_Am,
        stride_Ak,
        stride_Bb,
        stride_Bk,
        stride_Bn,
        stride_Cb,
        stride_Cm,
        stride_Cn,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        num_warps=4,
    )
    return C

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None (0/1)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32 -> float16)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # Cast to half for the heavy part
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2) Prepare weights (FP16)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    # concatenate proj & gate weights to halve the number of GEMMs
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, C)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, C)

    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 3) Linear projections + gates (fused via weight concatenation)
    # --------------------------------------------------------------
    # left side
    left_comb = F.linear(x, left_comb_w)                     # (B, N, N, 2*H)
    left_proj, left_gate_raw = torch.split(left_comb, hidden_dim, dim=-1)
    left_gate = left_gate_raw.sigmoid()
    left = left_proj * left_gate                              # (B, N, N, H)

    # right side
    right_comb = F.linear(x, right_comb_w)                   # (B, N, N, 2*H)
    right_proj, right_gate_raw = torch.split(right_comb, hidden_dim, dim=-1)
    right_gate = right_gate_raw.sigmoid()
    right = right_proj * right_gate                           # (B, N, N, H)

    # out gate (used later)
    out_gate = F.linear(x, out_gate_w).sigmoid()              # (B, N, N, H)

    # --------------------------------------------------------------
    # 4) Optional mask (broadcasted)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape (B, N, N) → (B, N, N, 1) for broadcasting
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 5) Core triangle multiplication via a custom batched GEMM
    #    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, H = left.shape

    # Bring hidden dim to the batch axis (contiguous layout required)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) → transpose k‑j

    # Flatten batch*hidden into one dimension
    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched GEMM: (B*H, N, N) @ (B*H, N, N)^T  →  (B*H, N, N)
    out_flat = batched_gemm(left_flat, right_flat, block_m=64, block_n=64, block_k=64)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 7) Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
