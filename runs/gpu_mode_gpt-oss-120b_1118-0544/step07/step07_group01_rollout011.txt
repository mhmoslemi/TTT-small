"""
custom_kernel: Triton‑accelerated outgoing TriMul forward (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for compute‑heavy parts.
3. Linear projections (left/right) and gate projections (left/right/out) using
   bias‑free F.linear (weights are cast to fp16).
4. Fuse mask, projection and gate multiplication for the left/right tensors
   with a small Triton kernel (only executed when a mask is present and not
   ignored).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   realised as a strided‑batched GEMM on the hidden dimension (B*H, N, N).
6. Apply a second LayerNorm over the hidden dimension, multiply by ``out_gate``,
   and project back to the original ``dim``.
7. Cast the result back to the input dtype (usually float32).

Only the mask‑gate fusion is implemented in Triton; the remaining ops use
high‑performance cuBLAS/torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,    # fp16 * (B, N, N, H)
    right_proj_ptr,   # fp16 * (B, N, N, H)
    left_gate_ptr,    # fp16 * (B, N, N, H)
    right_gate_ptr,   # fp16 * (B, N, N, H)
    mask_ptr,         # fp16 * (B, N, N)
    left_out_ptr,     # fp16 * (B, N, N, H)
    right_out_ptr,    # fp16 * (B, N, N, H)
    total_elems,      # i64 scalar = B*N*N*H
    hidden_dim,       # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # mask for out‑of‑bounds
    active = offs < total_elems

    # index into the mask (B,N,N) -> each hidden slice shares the same mask entry
    mask_idx = offs // hidden_dim

    # load values (masked)
    left_proj  = tl.load(left_proj_ptr + offs,  mask=active, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=active, other=0.0)
    left_gate  = tl.load(left_gate_ptr + offs,  mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)
    mask_val   = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)

    # fused multiply‑add
    left_out  = left_proj * left_gate * mask_val
    right_out = right_proj * right_gate * mask_val

    tl.store(left_out_ptr + offs,  left_out,  mask=active)
    tl.store(right_out_ptr + offs, right_out, mask=active)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse mask, projection and gate for left and right tensors.

    All tensors must be fp16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)

    Returns
    -------
    left, right : torch.Tensor
        (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32/float16)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict containing at least
            * "dim"        – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask" (optional) – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy operations
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to fp16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)  # (B, N, N, H)

    left_gate  = F.linear(x, left_gate_w).sigmoid()   # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid()  # (B, N, N, H)
    out_gate   = F.linear(x, out_gate_w).sigmoid()    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse mask & gate (Triton kernel if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is contiguous fp16
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            block_size=1024,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask present but ignored -> still broadcast multiply
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a strided‑batched GEMM over the hidden dimension.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, H, N, N) -> reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # Batched matmul: (B*H, N, N) @ (B*H, N, N)^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
