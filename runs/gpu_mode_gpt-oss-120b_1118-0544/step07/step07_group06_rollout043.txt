@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_ptr,
    B, H, N,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    bh = tl.program_id(2)   # 0..B*H-1
    block_m = tl.program_id(0)
    block_n = tl.program_id(1)

    # compute batch and hidden index
    b = bh // H
    h = bh % H

    # offsets for tiles
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Masks for validity
        mask_i = offs_m < N
        mask_j = offs_n < N
        mask_k = offs_k < N

        # Load left projection tile
        left_proj_ptrs = left_ptr + \
            b * stride_left_b + \
            offs_m[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k + \
            h * stride_left_h
        left_proj = tl.load(left_proj_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load left gate tile
        left_gate_ptrs = left_gate_ptr + \
            b * stride_lgate_b + \
            offs_m[:, None] * stride_lgate_i + \
            offs_k[None, :] * stride_lgate_k + \
            h * stride_lgate_h
        left_gate = tl.load(left_gate_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load mask for left (shape BLOCK_M, BLOCK_K)
        mask_left_ptrs = mask_ptr + \
            b * stride_mask_b + \
            offs_m[:, None] * stride_mask_i + \
            offs_k[None, :] * stride_mask_k
        mask_left = tl.load(mask_left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Compute left tile after gating and mask
        a = left_proj * left_gate * mask_left

        # Load right projection tile (shape BLOCK_K, BLOCK_N) -- note order: rows = k, cols = j
        right_proj_ptrs = right_ptr + \
            b * stride_right_b + \
            offs_k[:, None] * stride_right_i + \
            offs_n[None, :] * stride_right_k + \
            h * stride_right_h
        # Wait: we need to map indices correctly: right[b,j,k,h] -> i-index = j, k-index = k.
        # Right tensor strides: stride_right_i (index i), stride_right_k (index j). Actually careful:
        # Right has shape (B,N,N,H) with indices (b,i,j,h) where i is first N, j second N.
        # We need B_T[k,j] = right[b,j,k,h]. So we need pointer with i = k (first N), j = j (second N). So:
        # row index = k (i dim), column index = j (j dim).
        # So pointer = base + b*stride_right_b + k*stride_right_i + j*stride_right_k + h*stride_right_h. That's what we wrote.

        right_proj = tl.load(right_proj_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Load right gate tile similarly
        right_gate_ptrs = right_gate_ptr + \
            b * stride_rgate_b + \
            offs_k[:, None] * stride_rgate_i + \
            offs_n[None, :] * stride_rgate_k + \
            h * stride_rgate_h
        right_gate = tl.load(right_gate_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Load mask for right (shape BLOCK_K, BLOCK_N)
        mask_right_ptrs = mask_ptr + \
            b * stride_mask_b + \
            offs_n[:, None] * stride_mask_i + \
            offs_k[None, :] * stride_mask_k
        # Wait: mask for right uses indices (b,j,k). So i=j, j=k.
        # mask_ptr shape (B,N,N) with indices (b,i,j). So for mask[b,j,k] we need i=j (first index), j=k (second index).
        # So pointer = base + b*stride_mask_b + j*stride_mask_i + k*stride_mask_k.
        # Here offs_n is j dimension, offs_k is k dimension. So pointer = b*stride_mask_b + offs_n[:, None] * stride_mask_i + offs_k[None, :] * stride_mask_k
        mask_right = tl.load(mask_right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Compute right tile after gating and mask

        # accumulate

    # Store result

    )

    left_proj_w = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16)

    mask_stride = mask_tensor.stride()
    out_stride = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

        right_ptr=right_proj,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_tensor,
        out_ptr=out,

        N=N,

        stride_left_k=left_stride[2],

        stride_mask_k=mask_stride[2],
        stride_out_b=out_stride[0],
        stride_out_i=out_stride[1],
        stride_out_j=out_stride[2],
        stride_out_h=out_stride[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # 2nd LayerNorm
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply out gate
    out = out * out_gate

    # Final projection
    out = F.linear(out, to_out_w)

    # cast to orig dtype
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input [B,N,N,C] (fp32 for stability) and cast to fp16.
2. Linear projections (left/right) and three sigmoid gates (left, right, out) from
   C → H (hidden_dim) using fp16.
3. Fuse mask (if present), projection and gate multiplications and perform the
   core triangular reduction in a single Triton kernel:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * mask[b,i,k] *
                  left_gate[b,i,k,h] *
                  right[b,j,k,h] * mask[b,j,k] *
                  right_gate[b,j,k,h]
   The kernel treats each (batch, hidden) pair as an independent GEMM:
   left_tile (M×K) × right_tileᵀ (K×N) → out_tile (M×N).
4. Apply a second LayerNorm over the hidden dimension H.
5. Multiply by the output gate and project back to the original channel dimension C.
6. Cast the result back to the input dtype.

Only the mask‑gate‑fusion and the O(N³·H) reduction are performed in Triton;
all other operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: fused mask, gates and triangular reduction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,            # const * left projection   [B,N,N,H]
    right_ptr,           # const * right projection  [B,N,N,H]
    left_gate_ptr,       # const * left gate         [B,N,N,H]
    right_gate_ptr,      # const * right gate        [B,N,N,H]
    mask_ptr,            # const * mask              [B,N,N]
    out_ptr,             # * out [B,N,N,H]
    B, H, N,             # runtime sizes
    # strides for left projection
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right projection
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # strides for left gate
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    # strides for right gate
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    # strides for mask
    stride_mask_b, stride_mask_i, stride_mask_k,
    # strides for output
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute outgoing TriMul for a tile of (i,j) and a given (batch, hidden)."""
    bh = tl.program_id(2)          # 0 … B*H-1
    block_m = tl.program_id(0)     # row tile
    block_n = tl.program_id(1)     # col tile

    # Decompose bh into batch and hidden indices
    b = bh // H
    h = bh % H

    # Tile offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Validity masks
        mask_i = offs_m < N
        mask_j = offs_n < N
        mask_k = offs_k < N

        # ---------- left side (i,k) ----------
        left_proj_ptrs = left_ptr + \
            b * stride_left_b + \
            offs_m[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k + \
            h * stride_left_h
        left_proj = tl.load(left_proj_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        left_gate_ptrs = left_gate_ptr + \
            b * stride_lgate_b + \
            offs_m[:, None] * stride_lgate_i + \
            offs_k[None, :] * stride_lgate_k + \
            h * stride_lgate_h
        left_gate = tl.load(left_gate_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        mask_left_ptrs = mask_ptr + \
            b * stride_mask_b + \
            offs_m[:, None] * stride_mask_i + \
            offs_k[None, :] * stride_mask_k
        mask_left = tl.load(mask_left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        a = left_proj * left_gate * mask_left   # (M, K)

        # ---------- right side (j,k) ----------
        # Load right projection transposed: shape (N, K) where rows=j, cols=k
        right_proj_ptrs = right_ptr + \
            b * stride_right_b + \
            offs_n[:, None] * stride_right_i + \
            offs_k[None, :] * stride_right_k + \
            h * stride_right_h
        right_proj_T = tl.load(right_proj_ptrs,
                               mask=mask_j[:, None] & mask_k[None, :],
                               other=0.0).to(tl.float32)

        right_gate_ptrs = right_gate_ptr + \
            b * stride_rgate_b + \
            offs_n[:, None] * stride_rgate_i + \
            offs_k[None, :] * stride_rgate_k + \
            h * stride_rgate_h
        right_gate_T = tl.load(right_gate_ptrs,
                               mask=mask_j[:, None] & mask_k[None, :],
                               other=0.0).to(tl.float32)

        mask_right_ptrs = mask_ptr + \
            b * stride_mask_b + \
            offs_n[:, None] * stride_mask_i + \
            offs_k[None, :] * stride_mask_k
        mask_right_T = tl.load(mask_right_ptrs,
                               mask=mask_j[:, None] & mask_k[None, :],
                               other=0.0).to(tl.float32)

        b_tile = right_proj_T * right_gate_T * mask_right_T   # (N, K)

        # Accumulate: a (M,K) • b_tileᵀ (K,N) → (M,N)
        acc += tl.dot(a, b_tile, trans_b=True)

    # Store the result
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        offs_m[:, None] * stride_out_i + \
        offs_n[None, :] * stride_out_j + \
        h * stride_out_h
    mask_out = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, C] (float32 or float16)
            - mask: [B, N, N] (binary) or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Prepare mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    B, N, _, _ = left_proj.shape
    if mask is not None and not nomask:
        mask_fp = mask.to(device=device, dtype=torch.float16)
    else:
        mask_fp = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6) Core reduction in Triton (fused mask + gates)
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Strides (in elements)
    l_str = left_proj.stride()
    r_str = right_proj.stride()
    lg_str = left_gate.stride()
    rg_str = right_gate.stride()
    m_str = mask_fp.stride()
    out_str = out.stride()

    # Block sizes (tuned for 1024‑wide sequences)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    tri_mul_outgoing_kernel[grid](
        left_ptr=left_proj,
        right_ptr=right_proj,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_fp,
        out_ptr=out,
        B=B,
        H=hidden_dim,
        N=N,
        stride_left_b=l_str[0], stride_left_i=l_str[1],
        stride_left_k=l_str[2], stride_left_h=l_str[3],
        stride_right_b=r_str[0], stride_right_i=r_str[1],
        stride_right_k=r_str[2], stride_right_h=r_str[3],
        stride_lgate_b=lg_str[0], stride_lgate_i=lg_str[1],
        stride_lgate_k=lg_str[2], stride_lgate_h=lg_str[3],
        stride_rgate_b=rg_str[0], stride_rgate_i=rg_str[1],
        stride_rgate_k=rg_str[2], stride_rgate_h=rg_str[3],
        stride_mask_b=m_str[0], stride_mask_i=m_str[1], stride_mask_k=m_str[2],
        stride_out_b=out_str[0], stride_out_i=out_str[1],
        stride_out_j=out_str[2], stride_out_h=out_str[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
