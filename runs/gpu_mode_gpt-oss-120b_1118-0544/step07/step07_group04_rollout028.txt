import triton.language as tl

@triton.jit
def gate_mask_fuse_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                         total_elems, hidden_dim,
                         BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    mask_idx = offs // hidden_dim
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

def apply_gate_and_mask(proj, gate, mask, block_size=1024):
    B,N,N,H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,

        BLOCK_SIZE=block_size,
    )

@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_hidden, N,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Compute offsets for A and B
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # A block (M x K)
        a_ptrs = a_ptr + bh * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        # B block (K x N)
        b_ptrs = b_ptr + bh * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        # Load with masks
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < N) & (offs_k[None, :] < N), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < N) & (offs_n[None, :] < N), other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

    # Write out result
    c_ptrs = c_ptr + bh * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m[:, None] < N) & (offs_n[None, :] < N))

    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()

    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(1)  # row index corresponds to K
    stride_b_n = right_flat.stride(2)  # column index corresponds to N

    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = triton.cdiv(N_seq, BLOCK_M)
    num_blocks_n = triton.cdiv(N_seq, BLOCK_N)

        right_flat,
        out_flat,
        B * H,
        N_seq,

        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,

    out = out_flat.view(B, H, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()

"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).
Implements the forward pass of AlphaFold3's TriMul using:
  * a fused gate‑mask kernel (proj * gate * mask) in Triton,
  * a batched GEMM kernel (left @ rightᵀ) in Triton for the O(N³) contraction,
  * remaining linear / LayerNorm steps in PyTorch.
Returns a tensor of shape [bs, seq_len, seq_len, dim] (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and (optional) mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # int: B*N*N*H
    hidden_dim,        # int: H
    BLOCK_SIZE: tl.constexpr,
):
    """
    out = proj * gate * mask   (mask broadcast over hidden_dim)
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # bounds
    mask = offs < total_elems

    # index into mask (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 2048) -> torch.Tensor:
    """Fuse projection, gate and mask using a Triton kernel."""
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) contraction
#   out[bh, i, j] = Σ_k left[bh,i,k] * right[bh,j,k]
#   where bh = batch * hidden_dim combined index.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # fp16 pointers
    batch_hidden, N,                         # int scalars
    stride_a_batch, stride_a_m, stride_a_k,  # int strides for A (left)
    stride_b_batch, stride_b_k, stride_b_n,  # int strides for B (right, transposed)
    stride_c_batch, stride_c_m, stride_c_n,  # int strides for C (out)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    bh   = tl.program_id(0)          # batch * hidden_dim index
    pid_m = tl.program_id(1)         # block row (i)
    pid_n = tl.program_id(2)         # block col (j)

    # offsets within the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # A block: (M x K)
        a_ptrs = a_ptr + \
                 bh * stride_a_batch + \
                 offs_m[:, None] * stride_a_m + \
                 offs_k[None, :] * stride_a_k
        # B block: (K x N) – right matrix is already in (K,N) order
        b_ptrs = b_ptr + \
                 bh * stride_b_batch + \
                 offs_k[:, None] * stride_b_k + \
                 offs_n[None, :] * stride_b_n

        # mask out‑of‑range elements
        a_mask = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        b_mask = (offs_k[:, None] < N) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # dot product accumulate
        acc += tl.dot(a, b)

    # write result back (C)
    c_ptrs = c_ptr + \
             bh * stride_c_batch + \
             offs_m[:, None] * stride_c_m + \
             offs_n[None, :] * stride_c_n
    c_mask = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)

    Args:
        data: tuple (input_tensor, mask, weights, config)

            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    """

    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    dtype_in = input_tensor.dtype
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    stride_a_k     = left_flat.stride(2)
