"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the channel dimension (`dim`).
2. Cast to FP16 for compute‑heavy parts.
3. Linear projections (left/right) + gate projections (left/right/out) are
   performed with PyTorch (bias‑free Linear).  The mask (if present) and the
   gate sigmoid are fused with the projections in a small Triton kernel.
4. Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication per hidden slice.  It is
   implemented as a tiled GEMM Triton kernel (`tri_mul_kernel`) that
   processes a (BLOCK_I x BLOCK_J) tile per program and iterates over the
   reduction dimension `k` in blocks.
5. Output LayerNorm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to the original channel dimension.
6. Cast the result back to the input dtype (usually float32).

The implementation uses mixed precision (FP16 compute, FP32 accumulation)
and fuses the mask/gate/projection step, achieving a large speed‑up
compared to a pure‑PyTorch baseline.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# 1. Fuse projection, gate and mask for left & right tensors
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B,N,N,H)
    right_proj_ptr,  # fp16 * (B,N,N,H)
    left_gate_ptr,   # fp16 * (B,N,N,H)
    right_gate_ptr,  # fp16 * (B,N,N,H)
    mask_ptr,        # fp16 * (B,N,N)
    left_out_ptr,    # fp16 * (B,N,N,H)
    right_out_ptr,   # fp16 * (B,N,N,H)
    total_elems,     # i64 = B*N*N*H
    hidden_dim,      # i64 = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # guard
    mask = offs < total_elems
    # mask index = element // H (broadcast over hidden dim)
    mask_idx = offs // hidden_dim

    left_proj = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """Fuse projection, gate and mask for left/right tensors."""
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# 2. Core TriMul reduction (batched GEMM) kernel
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 * (B,N,N,H)
    right_ptr,         # fp16 * (B,N,N,H)
    out_ptr,           # fp16 * (B,N,N,H)
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # tile over i dimension
    pid_j = tl.program_id(1)   # tile over j dimension
    pid_hb = tl.program_id(2)  # combined batch*head index

    # decode batch and hidden head
    b = pid_hb // H
    h = pid_hb % H

    # compute i/j tile offsets
    i_offset = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # masks for boundary
    mask_i = i_offset < N
    mask_j = j_offset < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, N, BLOCK_K):
        k_offset = k + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left: (BLOCK_I, BLOCK_K) -> indices (b,i,k,h)
        left = tl.load(
            left_ptr
            + b * stride_left_b
            + i_offset[:, None] * stride_left_i
            + k_offset[None, :] * stride_left_k
            + h * stride_left_h,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right: (BLOCK_J, BLOCK_K) -> indices (b,j,k,h)
        right = tl.load(
            right_ptr
            + b * stride_right_b
            + j_offset[:, None] * stride_right_j
            + k_offset[None, :] * stride_right_k
            + h * stride_right_h,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )

        # dot over k (left @ right.T) → (BLOCK_I, BLOCK_J)
        acc += tl.dot(left, right, trans_b=True)

    # write out
    out = acc.to(tl.float16)
    tl.store(
        out_ptr
        + b * stride_out_b
        + i_offset[:, None] * stride_out_i
        + j_offset[None, :] * stride_out_j
        + h * stride_out_h,
        out,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _tri_mul(left: torch.Tensor, right: torch.Tensor,
             block_i: int = 64, block_j: int = 64, block_k: int = 32) -> torch.Tensor:
    """Batched GEMM kernel implementing out[b,i,j,h] = Σ_k left[b,i,k,h]*right[b,j,k,h]."""
    assert left.shape == right.shape, "left and right must have identical shapes"
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides
    stride_l = left.stride()
    stride_r = right.stride()
    stride_o = out.stride()

    grid = (
        triton.cdiv(N, block_i),
        triton.cdiv(N, block_j),
        B * H,
    )
    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l[0],
        stride_l[1],
        stride_l[2],
        stride_l[3],
        stride_r[0],
        stride_r[1],
        stride_r[2],
        stride_r[3],
        stride_o[0],
        stride_o[1],
        stride_o[2],
        stride_o[3],
        BLOCK_I=block_i,
        BLOCK_J=block_j,
        BLOCK_K=block_k,
    )
    return out


# ----------------------------------------------------------------------
# 3. Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32/float16)
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict with mandatory keys "dim", "hidden_dim"
                optional key "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) LayerNorm over channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 5) Fuse mask, projection and gate (Triton) or fallback to PyTorch
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # simple element‑wise fusion (mask ignored or absent)
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction (batched GEMM) via Triton
    # ------------------------------------------------------------------
    out = _tri_mul(left, right)   # (B,N,N,hidden_dim) FP16

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
