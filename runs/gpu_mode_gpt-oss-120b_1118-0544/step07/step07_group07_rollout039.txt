```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_al, stride_ak,
    stride_br, stride_bk,
    stride_cl, stride_ck,
    batch_head_stride,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # block row index
    pid_n = tl.program_id(1)  # block col index
    pid_bh = tl.program_id(2) # batch*head index

    # offsets within the matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointer for this batch/head
    base = pid_bh * batch_head_stride

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of tiles in K dimension
    num_k = tl.cdiv(K, BLOCK_K)

    for k in range(0, num_k):
        k_offset = k * BLOCK_K
        # indices for A block [M,K]
        a_i = offs_m[:, None]
        a_k = k_offset + tl.arange(0, BLOCK_K)[None, :]

        # indices for B^T block [K,N]
        b_k = k_offset + tl.arange(0, BLOCK_K)[:, None]
        b_j = offs_n[None, :]

        # pointers
        a_ptrs = left_ptr + base + a_i * stride_al + a_k * stride_ak
        b_ptrs = right_ptr + base + b_k * stride_bk + b_j * stride_br

        # mask for out-of-bounds
        a_mask = (a_i < M) & (a_k < K)
        b_mask = (b_k < K) & (b_j < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # dot product and accumulate in FP32
        acc += tl.dot(a, b).to(tl.float32)

    # store result
    c_i = offs_m[:, None]
    c_j = offs_n[None, :]
    c_ptrs = out_ptr + base + c_i * stride_cl + c_j * stride_ck
    c_mask = (c_i < M) & (c_j < N)
    tl.store(c_ptrs, tl.cast(acc, tl.float16), mask=c_mask)
```

def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul (Triangle Multiplicative Update).
    Performs LayerNorm, linear projections, gating, optional mask,
    a fused batched GEMM via a Triton kernel, then output LayerNorm,
    gating and final linear projection. Returns tensor of shape [B,N,N,dim]
    in float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input LayerNorm (float32)
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # 2. Cast to fp16 for compute
    x = x.to(torch.float16)

    # 3. Linear projections (bias‑less) in fp16
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)               # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)             # (B,N,N,hidden_dim)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # 4. Apply mask and gates (element‑wise)
    if mask is not None and not nomask:
        # mask shape [B,N,N] -> [B,N,N,1] for broadcasting
        mask = mask.to(device).to(torch.float16).unsqueeze(-1)
        left = left * left_gate * mask
        right = right * right_gate * mask
    else:
        left = left * left_gate
        right = right * right_gate

    # 5. Core contraction via Triton GEMM (left @ rightᵀ) per head
    B, N, _, H = left.shape  # H == hidden_dim

    # permute to (B, H, N, N) and flatten batch*head dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # shape (B*H, N, N)
    left_mat = left_perm.view(-1, N, N)
    right_mat = right_perm.view(-1, N, N)

    # allocate output
    out_mat = torch.empty_like(left_mat)

    # strides (int64)
    stride_al = left_mat.stride(1)   # row stride (i)
    stride_ak = left_mat.stride(2)   # col stride (k)
    stride_br = right_mat.stride(1)  # row stride (j)
    stride_bk = right_mat.stride(2)  # col stride (k)
    stride_cl = out_mat.stride(1)
    stride_ck = out_mat.stride(2)
    batch_head_stride = left_mat.stride(0)   # same for all three

    # block sizes - tune for performance
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # launch grid
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        left_mat.shape[0],  # B*H
    )

    tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_al, stride_ak,
        stride_br, stride_bk,
        stride_cl, stride_ck,
        batch_head_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Output LayerNorm, gating, final projection (float32)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out_fp32 = out.to(torch.float32)
    out_fp32 = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out_fp32 = out_fp32 * out_gate.to(torch.float32)   # apply output gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)
    out_fp32 = F.linear(out_fp32, to_out_w)   # (B,N,N,dim)

    # Return in float32 (same as input if input was float32)
    return out_fp32

"""
custom_kernel: Triton‑accelerated outgoing TriMul (Triangle Multiplicative Update).

The implementation follows the reference PyTorch TriMul forward pass:
1. LayerNorm on the input tensor (float32).
2. Cast to fp16 and apply linear projections (left, right) and gating layers.
3. Optional mask multiplication (element‑wise).
4. Core contraction `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`
   performed as a batched GEMM (left @ rightᵀ) using a custom Triton kernel.
5. Output LayerNorm, output‑gate multiplication, and final linear projection.
The final output tensor has shape [B, N, N, dim] and dtype float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr,                # fp16 *, shape (BH, N, N)
    right_ptr,               # fp16 *, shape (BH, N, N)
    out_ptr,                 # fp16 *, shape (BH, N, N)
    M, N, K,                 # int32 matrix dims (M=N=K=seq_len)
    stride_al, stride_ak,    # int64, left matrix row/col strides
    stride_br, stride_bk,    # int64, right matrix row/col strides (original layout)
    stride_cl, stride_ck,    # int64, output matrix row/col strides
    batch_head_stride,       # int64, stride between successive (batch*head) matrices
    BLOCK_M: tl.constexpr,   # tile size in M dimension
    BLOCK_N: tl.constexpr,   # tile size in N dimension
    BLOCK_K: tl.constexpr,   # tile size in K dimension
):
    # Program IDs map to output tile coordinates and batch*head index
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_bh = tl.program_id(2)  # batch*head

    # Base offset for this batch*head
    base = pid_bh * batch_head_stride

    # Compute absolute indices of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k = tl.cdiv(K, BLOCK_K)

    for k in range(0, num_k):
        k_offset = k * BLOCK_K

        # ---- Load A block (left matrix) ----
        a_i = offs_m[:, None]                     # (BLOCK_M, 1)
        a_k = k_offset + tl.arange(0, BLOCK_K)[None, :]  # (1, BLOCK_K)
        a_ptrs = left_ptr + base + a_i * stride_al + a_k * stride_ak
        a_mask = (a_i < M) & (a_k < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)  # fp16

        # ---- Load Bᵀ block (right matrix transposed) ----
        # Bᵀ has shape (K, N); we load it as (BLOCK_K, BLOCK_N)
        b_k = k_offset + tl.arange(0, BLOCK_K)[:, None]   # (BLOCK_K, 1)
        b_n = offs_n[None, :]                             # (1, BLOCK_N)
        # Row stride for Bᵀ = original column stride, column stride = original row stride
        b_ptrs = right_ptr + base + b_k * stride_bk + b_n * stride_br
        b_mask = (b_k < K) & (b_n < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)  # fp16

        # ---- Dot product and accumulate (FP32) ----
        acc += tl.cast(tl.dot(a, b), tl.float32)

    # ---- Store result ----
    c_i = offs_m[:, None]
    c_j = offs_n[None, :]
    c_ptrs = out_ptr + base + c_i * stride_cl + c_j * stride_ck
    c_mask = (c_i < M) & (c_j < N)
    tl.store(c_ptrs, tl.cast(acc, tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, dim)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of parameter tensors (float32)
            - config: dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, dim) in float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (bias‑less) in fp16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)          # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)        # (B,N,N,hidden_dim)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4. Apply mask and gates (element‑wise)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * left_gate * mask_fp
        right = right * right_gate * mask_fp
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Rearrange to (B, H, N, N) and flatten batch*head dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(-1, N, N)    # (BH, N, N)
    right_mat = right_perm.view(-1, N, N)  # (BH, N, N)

    out_mat = torch.empty_like(left_mat)

    # Strides for the GEMM kernel
    stride_al = left_mat.stride(1)   # row stride (i)
    stride_ak = left_mat.stride(2)   # col stride (k)
    stride_br = right_mat.stride(1)  # row stride (j)
    stride_bk = right_mat.stride(2)  # col stride (k)
    stride_cl = out_mat.stride(1)    # output row stride
    stride_ck = out_mat.stride(2)    # output col stride
    batch_head_stride = left_mat.stride(0)  # stride between different (batch*head) matrices

    # Tile sizes – chosen to map well to Tensor Cores (multiples of 16)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks in M dimension
        triton.cdiv(N, BLOCK_N),   # blocks in N dimension
        left_mat.shape[0],         # BH = B * hidden_dim
    )

    _tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_al, stride_ak,
        stride_br, stride_bk,
        stride_cl, stride_ck,
        batch_head_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection (float32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)

    out_fp32 = out.to(torch.float32)
    out_fp32 = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out_fp32 = out_fp32 * out_gate.to(torch.float32)   # apply output gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)
    out_fp32 = F.linear(out_fp32, to_out_w)           # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 7. Return result (float32)
    # ------------------------------------------------------------------
    return out_fp32
