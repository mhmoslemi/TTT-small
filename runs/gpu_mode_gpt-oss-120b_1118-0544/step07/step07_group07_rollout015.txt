"""
custom_kernel: Triton‑accelerated outgoing TriMul (Triangle Multiplicative Update).

Algorithm
---------
1. Input LayerNorm (float32).
2. Cast to FP16 for all heavy compute.
3. Linear projections (bias‑less) → left, right and three gates.
4. Apply mask (if present) and element‑wise gating (torch ops, cheap).
5. Core contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM using a Triton kernel (left @ rightᵀ)
   where the batch dimension is B·hidden_dim.
6. Output LayerNorm on hidden dimension, multiply by out_gate,
   final linear projection back to original dim.
7. Cast back to the input dtype and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the batched matrix multiplication:
#   out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
# left : (B, N, N, H)      -- row major
# right: (B, N, N, H)      -- we use its transpose over the last two axes
# out  : (B, N, N, H)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # *fp16  [B, N, N, H]
    right_ptr,         # *fp16  [B, N, N, H] (transposed view)
    out_ptr,           # *fp16  [B, N, N, H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_k, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_bh = tl.program_id(2)  # batch * hidden index

    # Global indices
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Decompose batch index into (b, h)
    bh = pid_bh
    b = bh // H
    h = bh % H

    # Masks for out‑of‑bounds rows / cols
    mask_i = i < N
    mask_j = j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # Accumulator in FP32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Masks for loads
        mask_a = mask_i[:, None] & mask_k[None, :]   # left[i,k]
        mask_b = mask_k[:, None] & mask_j[None, :]   # right_t[k,j]

        # Pointers with proper strides
        a_ptrs = (left_ptr +
                  b * stride_left_b +
                  i[:, None] * stride_left_i +
                  k[None, :] * stride_left_k +
                  h * stride_left_h)
        b_ptrs = (right_ptr +
                  b * stride_right_b +
                  k[:, None] * stride_right_k +
                  j[None, :] * stride_right_j +
                  h * stride_right_h)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # tl.dot with fp16 inputs accumulates into FP32 (default)
        acc += tl.dot(a, b)

    # Write result back (convert to fp16)
    c_ptrs = (out_ptr +
              b * stride_out_b +
              i[:, None] * stride_out_i +
              j[None, :] * stride_out_j +
              h * stride_out_h)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_ij)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of parameter tensors (float32)
            - config: dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor of shape (B, N, N, dim) with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (bias‑less) in fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left  = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)  # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4. Apply mask (if present) and element‑wise gating (torch ops)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask
        right = right * mask

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Prepare transposed view of right for GEMM (k, j ordering)
    right_t = right.transpose(1, 2).contiguous()   # shape (B, N, N, H) with axes (B, K, J, H)

    # Allocate output tensor
    out = torch.empty_like(left, dtype=torch.float16)

    # Strides (in elements, not bytes)
    s_left_b, s_left_i, s_left_k, s_left_h = left.stride()
    s_right_b, s_right_k, s_right_j, s_right_h = right_t.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H                      # batch * hidden
    )

    _tri_mul_kernel[grid](
        left,
        right_t,
        out,
        B, N, H,
        s_left_b, s_left_i, s_left_k, s_left_h,
        s_right_b, s_right_k, s_right_j, s_right_h,
        s_out_b, s_out_i, s_out_j, s_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------
    # 7. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
