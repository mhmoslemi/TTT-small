    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    M, N, K,

    BATCHES: tl.constexpr,
):

    batch_id = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_remaining = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_aa, stride_am, stride_ak,
    stride_bb, stride_bk, stride_bn,
    stride_cc, stride_cm, stride_cn,
    M, N, K,
    BATCHES: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # Compute batch id and tile indices
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n
    batch_id = pid // blocks_per_batch
    block_id = pid % blocks_per_batch
    block_m = block_id // num_blocks_n
    block_n = block_id % num_blocks_n

    # Offsets for M,N
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Pointers for this batch
    a_batch_ptr = a_ptr + batch_id * stride_aa
    b_batch_ptr = b_ptr + batch_id * stride_bb
    c_batch_ptr = c_ptr + batch_id * stride_cc

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )
        # Load B: shape (BLOCK_K, BLOCK_N) where B is transposed (K,N)
        b = tl.load(
            b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        # Compute dot
        acc += tl.dot(a, b)

    # Write back
    c = acc.to(tl.float16)
    tl.store(
        c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )

"""
TriMul (outgoing) forward pass – Triton‑accelerated reduction.

Algorithm
---------
1. LayerNorm over the channel dimension (float32).
2. Cast to float16 for the heavy compute.
3. Linear projections → left_proj / right_proj (B,N,N,H).
4. Gate projections → left_gate / right_gate / out_gate (sigmoid).
5. Fuse projection, gate and optional mask with a tiny Triton kernel
   (element‑wise product, mask broadcast over H).
6. **Batched GEMM in Triton**:
   Compute for every batch‑head `h`:
        out[b,h,i,j] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication where each (b,h) pair
   multiplies an N×N matrix with the transpose of another N×N matrix.
   The kernel tiles the output (M×N) with BLOCK_M×BLOCK_N and iterates
   over K in BLOCK_K‑sized chunks, using fp16 loads and fp32 accumulation.
7. Output LayerNorm over the hidden dimension, multiply by out_gate,
   final linear projection back to `dim`.
8. Cast back to the input dtype (usually float32).

The only non‑trivial operation performed in Triton is the reduction
(step 6); the rest uses PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # int: B*N*N*H
    hidden_dim,        # int: H
    BLOCK_SIZE: tl.constexpr,
):
    """out = proj * gate * mask   (mask broadcast over hidden_dim)."""
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # index of the corresponding mask element (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # mask defaults to 1

    tl.store(out_ptr + offs, proj * gate * m, mask=valid)


def _apply_gate_and_mask(proj, gate, mask, block_size=2048):
    """Fuse projection, gate and mask using Triton."""
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    stride_aa, stride_am, stride_ak,    # a: (batch, M, K)
    stride_bb, stride_bk, stride_bn,    # b: (batch, K, N)   (K is the shared dim)
    stride_cc, stride_cm, stride_cn,    # c: (batch, M, N)
    M, N, K,
    BATCHES: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[b] = A[b] @ B[b]^T  for each batch."""
    pid = tl.program_id(0)

    # number of tiles per batch
    num_m_tiles = tl.cdiv(M, BLOCK_M)
    num_n_tiles = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_m_tiles * num_n_tiles

    batch_id = pid // tiles_per_batch
    tile_id = pid % tiles_per_batch
    tile_m = tile_id // num_n_tiles
    tile_n = tile_id % num_n_tiles

    # coordinate offsets for the output tile
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers to the beginning of this batch
    a_batch = a_ptr + batch_id * stride_aa
    b_batch = b_ptr + batch_id * stride_bb
    c_batch = c_ptr + batch_id * stride_cc

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )
        # Load B tile (K x N) – note B is stored as (K,N) already (right is transposed)
        b = tl.load(
            b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        acc += tl.dot(a, b)          # fp16·fp16 → fp32 accumulate

    # write back
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask         : torch.Tensor [B, N, N] (float32/float16/bool) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"        – input channel dimension C
                  * "hidden_dim" – hidden dimension H
                  * "nomask" (optional) – if True, ignore the mask
    Returns:
        torch.Tensor [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast linear weights to FP16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid (B,N,N,H)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Fuse projection, gate and optional mask (Triton kernel)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Rearrange to (B, H, N, N) and then view as (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    batch = B * H

    left_flat = left.view(batch, N, N)   # (batch, M=N, K=N)
    right_flat = right.view(batch, N, N) # (batch, K=N, N=N) – we will treat it as transposed

    # Allocate output tensor
    out_flat = torch.empty_like(left_flat)

    # Strides for the kernels
    stride_aa = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bb = right_flat.stride(0)
    stride_bk = right_flat.stride(1)   # K dimension stride
    stride_bn = right_flat.stride(2)   # N dimension stride

    stride_cc = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # Tile sizes – chosen to fit in shared memory and give good occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_m_tiles = (N + BLOCK_M - 1) // BLOCK_M
    num_n_tiles = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = batch * num_m_tiles * num_n_tiles

    grid = (total_tiles,)

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        stride_aa,
        stride_am,
        stride_ak,
        stride_bb,
        stride_bk,
        stride_bn,
        stride_cc,
        stride_cm,
        stride_cn,
        N,            # M
        N,            # N
        N,            # K
        BATCHES=batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
