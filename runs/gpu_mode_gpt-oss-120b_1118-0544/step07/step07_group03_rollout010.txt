@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Program IDs
    pid_batch = tl.program_id(0)  # batch_head index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute b and h
    b = pid_batch // H
    h = pid_batch % H

    # Compute base pointers for this batch+head
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Compute coordinate ranges
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for M,N dimensions
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_idx in range(num_k_blocks):
        k_base = k_idx * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers for left tile (M,K)
        left_ptrs = left_base + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        # Load left tile with mask
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Pointers for right tile (K,N)
        right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(left, right)  # automatic accumulation in fp32

    out_ptrs = out_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to float16 for the heavy compute.
3. Linear projections (left/right) and gate projections (left, right, out).
4. Apply the optional mask (broadcast over the hidden dimension) and
   multiply by the corresponding gates.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (left @ rightᵀ) for every (b,h) pair.
   Implemented with a custom Triton kernel that fuses the whole batch*head
   dimension, thus avoiding the overhead of many torch.bmm calls.
6. Output LayerNorm over the hidden dimension, multiply by `out_gate`,
   final linear projection back to `dim`.
7. Cast the result back to the input dtype (usually float32).

The Triton kernel loads tiles of `left` (M×K) and `right` (K×N),
accumulates in float32, and stores the result in float16.
All other ops use high‑performance PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel for the batched GEMM (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr,          # fp16 * (B,N,N,H)
    right_ptr,         # fp16 * (B,N,N,H)
    out_ptr,           # fp16 * (B,N,N,H)
    B, N, H,           # i64 scalar
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_batch = tl.program_id(0)   # combined batch * head index
    pid_m = tl.program_id(1)       # i‑block
    pid_n = tl.program_id(2)       # j‑block

    # --------------------------------------------------------------
    # Derive batch (b) and head (h) indices from the combined id
    # --------------------------------------------------------------
    b = pid_batch // H
    h = pid_batch % H

    # ------------------------------------------------------------------
    # Base pointers for this (b,h) slice
    # ------------------------------------------------------------------
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Accumulator (float32 for accuracy)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_k_blocks):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (M, K)
        left_ptrs = left_base + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile: (K, N)  <-- note the transposed layout
        right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        # Accumulate
        acc += tl.dot(left, right)   # result automatically accumulated in fp32

    # --------------------------------------------------------------
    # Store the result (cast back to fp16)
    # --------------------------------------------------------------
    out_ptrs = out_base + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper to launch the Triton GEMM
# ----------------------------------------------------------------------
def _trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM over the last hidden dimension:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    left, right: (B, N, N, H) float16, contiguous
    Returns out with the same shape and dtype.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides (in elements)
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_i, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    # Block sizes – tuned for typical B200 SM90 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                              # batch*head dimension
        (N + BLOCK_M - 1) // BLOCK_M,       # i‑blocks
        (N + BLOCK_N - 1) // BLOCK_N,       # j‑blocks
    )

    _trimul_gemm_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        s_l_b, s_l_i, s_l_k, s_l_h,
        s_r_b, s_r_i, s_r_k, s_r_h,
        s_o_b, s_o_i, s_o_j, s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"         – input channel dimension C
                  * "hidden_dim"  – hidden dimension H
                  * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (no bias in all Linear layers)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Projections and gates (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 5) Mask handling (optional) and gating fusion
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: [B,N,N] -> [B,N,N,1] in fp16
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * mask_fp16 * left_gate
        right = right_proj * mask_fp16 * right_gate
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguous layout for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction via Triton (batched GEMM)
    # ------------------------------------------------------------------
    out = _trimul_gemm(left, right)   # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim) fp16

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype (typically float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
