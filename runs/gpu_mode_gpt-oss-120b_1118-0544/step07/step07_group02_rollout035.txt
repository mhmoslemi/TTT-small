def _bmm_kernel(
    left_ptr, right_ptr, out_ptr,
    B_H, N,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    total_blocks = B_H * num_pid_m * num_pid_n
    # compute batch index
    batch = pid // (num_pid_m * num_pid_n)
    block_index = pid % (num_pid_m * num_pid_n)
    pid_m = block_index // num_pid_n
    pid_n = block_index % num_pid_n

    # compute offsets
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    # mask for out-of-range
    mask_i = i < N
    mask_j = j < N
    mask_k = k < N

    # compute base pointer offset for this batch
    batch_offset = batch * N * N
    # left pointer for block (i,k)
    left_ptrs = left_ptr + batch_offset + i[:, None] * N + k[None, :]
    # right pointer (k,j) for transposed
    right_ptrs = right_ptr + batch_offset + k[:, None] * 1 + j[None, :] * N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K tiles
    num_tiles_k = tl.cdiv(N, BLOCK_K)
    for t in range(num_tiles_k):
        # compute mask per tile
        cur_k = t * BLOCK_K
        # mask for each tile
        cur_mask_k = (cur_k + k) < N
        # load
        left_tile = tl.load(left_ptrs + t * BLOCK_K * 1, mask=mask_i[:, None] & cur_mask_k[None, :], other=0.0)
        right_tile = tl.load(right_ptrs + t * BLOCK_K * 1, mask=cur_mask_k[:, None] & mask_j[None, :], other=0.0)
        acc += tl.dot(left_tile, right_tile)
    # store
    out_ptrs = out_ptr + batch_offset + i[:, None] * N + j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input (float32 for stability).
2. Cast to fp16 for the compute‑heavy part.
3. Compute left/right projections and three sigmoid gates (left, right, out).
4. Fuse mask (if present) together with the left/right gate multiplications in a
   small Triton kernel that works in‑place on the flattened tensors.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM (L @ Rᵀ) with a custom Triton kernel.
6. Apply a second LayerNorm on the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original dimension.
7. Cast the result back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Kernel 1: fuse mask and gate multiplication (in‑place)
# -------------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened, fp16
    left_gate_ptr, right_gate_ptr,     # same shape, fp16
    mask_ptr,                          # [B,N,N] flattened, fp16
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden‑dim element shares the same mask entry
        mask_idx = offs // hidden_dim               # index in [B,N,N]
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

    num_blocks_n = tl.cdiv(N, BLOCK_N)

    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)

    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)

    right = F.linear(x, right_proj_w)        # (B, N, N, H)

    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core O(N³·H) contraction via custom batched GEMM (Triton)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_mat = torch.empty_like(left_mat)  # fp16

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_tasks = (B * H) * num_blocks_m * num_blocks_n
    grid = (total_tasks,)

    _bmm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        B * H,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # -----------------------------------------------------------------
    # 7. Cast back to float32
    # -----------------------------------------------------------------
    out = out.to(torch.float32)
    return out
```
