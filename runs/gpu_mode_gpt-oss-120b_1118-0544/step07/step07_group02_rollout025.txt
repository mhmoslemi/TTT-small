def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

Inside kernel:
    pid = tl.program_id(0)  # batch*head dimension index
    batch = pid // H
    head = pid % H

    pid_m = tl.program_id(1) 
    pid_n = tl.program_id(2)

    # Offsets
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_i = i < N
    mask_j = j < N

    # Initialize accumulator to zero (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k range
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Compute memory offsets
        # left: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + batch*stride_left_b + head*stride_left_h + i[:, None] * stride_left_i + k_offset[None, :] * stride_left_k
        # right: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + batch*stride_right_b + head*stride_right_h + k_offset[:, None] * stride_right_k + j[None, :] * stride_right_j

        # load with masks
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot: left (M,K) * right (K,N) => (M,N)
        acc += tl.dot(left, right)  # product shape (M,N)
    
    # cast acc to fp16
    acc_fp16 = acc.to(tl.float16)

    # store
    out_ptrs = out_ptr + batch*stride_out_b + head*stride_out_h + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc_fp16, mask=mask_i[:, None] & mask_j[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input (float32 → stable) and cast to fp16.
2. Linear projections (left/right) and three sigmoid gates (left, right, out).
3. Fuse mask (if present) together with the left/right gates in a small Triton
   kernel that works in‑place on the flattened tensors.
4. Core O(N³·H) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (L @ Rᵀ) with a Triton kernel that tiles
   the i‑ and j‑ dimensions while looping over k.
5. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate, and a final linear projection back to the original dimension.
6. Cast the result to float32 and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape
    mask_ptr,                          # [B,N,N] flattened, broadcast over H
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    """In‑place fusion:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
       If HAS_MASK is False the mask is ignored (treated as 1).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden‑dim element shares the same mask entry
        mask_idx = offs // hidden_dim               # index in [B,N,N]
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Grid: (batch*head, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    """
    pid = tl.program_id(0)            # combined batch * head index
    pid_m = tl.program_id(1)          # block row (i)
    pid_n = tl.program_id(2)          # block col (j)

    # derive batch and head indices
    batch = pid // H
    head = pid % H

    # tile offsets
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator in fp32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left: shape (BLOCK_M, BLOCK_K) -> left[b,i,k,h]
        left_ptrs = (left_ptr
                     + batch * stride_left_b
                     + head * stride_left_h
                     + i[:, None] * stride_left_i
                     + k[None, :] * stride_left_k)

        # right: shape (BLOCK_K, BLOCK_N) -> right[b,j,k,h] (k as row)
        right_ptrs = (right_ptr
                      + batch * stride_right_b
                      + head * stride_right_h
                      + k[:, None] * stride_right_k
                      + j[None, :] * stride_right_j)

        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # dot: (M,K) @ (K,N) -> (M,N), accumulation in fp32
        acc += tl.dot(left_tile, right_tile)

    # write back result in fp16
    out_ptrs = (out_ptr
                + batch * stride_out_b
                + head * stride_out_h
                + i[:, None] * stride_out_i
                + j[None, :] * stride_out_j)

    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int)          – input channel dimension C
                * "hidden_dim" (int)   – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with dtype torch.float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy path
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gates (all in fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    left  = F.linear(x, left_proj_w)                # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)               # (B,N,N,hidden_dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # -----------------------------------------------------------------
    # 4. Fuse mask and gates (Triton) or fallback to PyTorch
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        # flatten tensors for the kernel
        B, N, _, _ = left.shape
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core O(N³·H) contraction via custom Triton GEMM
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    assert H == hidden_dim, "hidden dimension mismatch"

    out = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    # strides for left, right, and out (in elements)
    stride_left = left.stride()   # (b, i, k, h)
    stride_right = right.stride()
    stride_out = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)

    # LayerNorm in fp32 for stability
    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,),
                            weight=to_out_norm_w, bias=to_out_norm_b)
    out_norm = out_norm.to(torch.float16)

    # Apply output gate
    out = out_norm * out_gate

    # Final linear back to original dim
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # -----------------------------------------------------------------
    # 7. Cast back to float32 and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
