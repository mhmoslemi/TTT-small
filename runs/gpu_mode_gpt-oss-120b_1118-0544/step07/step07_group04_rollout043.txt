The TriMul forward steps (as per reference):

1. LayerNorm over dim on the input: x_norm = layer_norm(x, weight=norm.weight, bias=norm.bias). Over last dimension (dim)

2. Compute left_proj = Linear(x_norm, left_proj.weight) -> shape (B,N,N,H)
   right_proj = Linear(x_norm, right_proj.weight) -> shape (B,N,N,H)

3. Multiply by mask if provided: left_proj = left_proj * mask[..., None]; right_proj = right_proj*mask[..., None]

4. Compute gates: left_gate = sigmoid(Linear(x_norm, left_gate.weight)), input shape (B,N,N,H). right_gate = sigmoid(Linear(x_norm, right_gate.weight)), out_gate = sigmoid(Linear(x_norm, out_gate.weight)); shape same.

5. Apply gating: left = left_proj * left_gate, right = right_proj * right_gate

6. Compute out via einsum: out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h] (this is a bilinear contraction, akin to matrix multiplication across k dimension). Equivalent to batched GEMM.

7. Apply to_out_norm (LayerNorm over hidden_dim) on out: out_norm = layer_norm(out, weight=to_out_norm.weight, bias=to_out_norm.bias)

8. Multiply by out_gate (broadcasted): out = out_norm * out_gate

9. Linear projection back to dim: output = Linear(out, to_out.weight) -> shape (B,N,N,dim)

10. Return output.

# Gate+mask fusion kernel
@triton.jit
def gate_mask_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # compute mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

def apply_gate_and_mask(proj, gate, mask, block_size=8192):
    # proj, gate: (B,N,N,H) fp16 contiguous
    # mask: (B,N,N) fp16 contiguous
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size), )
    gate_mask_fuse_kernel[grid](
        proj, gate, mask, out,
        total, H,
        BLOCK_SIZE=block_size,
    )
    return out

@triton.jit
def batched_gemm_kernel(
    A, B, C,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets for output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = A + batch * stride_am
    b_base = B + batch * stride_bm
    c_base = C + batch * stride_cm

    # pointers for rows and cols
    a_ptrs = a_base + offs_m[:, None] * stride_ai
    b_ptrs = b_base + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        k_range = cur_k + tl.arange(0, BLOCK_K)

        # load A block
        a = tl.load(a_ptrs + cur_k * stride_ak,
                    mask=mask_m[:, None] & (k_range[None, :] < K),
                    other=0.0)

        # load B block
        b = tl.load(b_ptrs + cur_k * stride_bk,
                    mask=(k_range[:, None] < K) & mask_n[None, :],
                    other=0.0)

        # compute
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_base + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Includes fused gate+mask computation and a custom batched GEMM kernel for the
    contraction over the sequence dimension.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # 1. Input LayerNorm (float32 for stability)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(torch.float32),
        bias=weights["norm.bias"].to(torch.float32)
    )   # (B,N,N,dim)

    # 2. Cast to fp16 for compute heavy parts
    x = x.to(torch.float16)

    # 3. Prepare weights in fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # 4. Linear projections
    left_proj = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # 5. Gate projections
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # 6. Apply mask and gate (fused via Triton)
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Free unused tensors to save memory
    del left_proj, right_proj, left_gate, right_gate

    # 7. Core reduction: batched GEMM
    B, N, _, H = left.shape  # B,N,N,H
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # transpose right for GEMM (K,N)
    right_t = right_flat.transpose(1, 2).contiguous()

    # Prepare output tensor
    out_flat = torch.empty_like(left_flat)

    # Strides
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the channel dimension (float32 for stability).
2. Cast to fp16 for compute‑heavy stages.
3. Linear projections → left/right (B,N,N,hidden_dim) and gate projections.
4. Fuse projection, gate and optional mask in a Triton kernel:
       fused = proj * gate * mask   (mask broadcast over hidden_dim)
5. Batched GEMM (custom Triton kernel):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   The batch dimension is B·hidden_dim, each batch performs a matrix‑multiply
   (N×N) @ (N×N)ᵀ using tiled loads and accumulation in float32.
6. Output LayerNorm over hidden_dim, multiply by out_gate, final linear
   projection back to the original dimension.
7. Cast back to the input dtype (usually float32).

The kernel implements both the gate‑mask fusion and the O(N³) reduction,
providing a fully Triton‑based forward pass.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. Gate + mask fusion kernel
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,          # fp16 * (B,N,N,H)
    gate_ptr,          # fp16 * (B,N,N,H)
    mask_ptr,          # fp16 * (B,N,N)
    out_ptr,           # fp16 * (B,N,N,H)
    total_elems,       # int: B*N*N*H
    hidden_dim,        # int: H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 8192) -> torch.Tensor:
    """Fuse projection, gate and mask using Triton."""
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# 2. Batched GEMM kernel (A @ Bᵀ) for the O(N³) reduction
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A, B, C,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = A + batch_id * stride_am
    b_batch = B + batch_id * stride_bm
    c_batch = C + batch_id * stride_cm

    # row/col pointers (without the K dimension)
    a_ptrs = a_batch + offs_m[:, None] * stride_ai
    b_ptrs = b_batch + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        k_offset = cur_k + tl.arange(0, BLOCK_K)  # (BLOCK_K,)

        a = tl.load(
            a_ptrs + k_offset[None, :] * stride_ak,
            mask=mask_m[:, None] & (k_offset[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_ptrs + k_offset[:, None] * stride_bk,
            mask=(k_offset[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_batch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 3. Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation using Triton.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters (float32 tensors)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(torch.float32),
        bias=weights["norm.bias"].to(torch.float32),
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)

    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections and gates (fp16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)        # (B, N, N, hidden_dim)

    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()

    if mask is not None and not nomask:

    # free intermediate tensors
    del left_proj, right_proj, left_gate, right_gate

    # ------------------------------------------------------------------
    # 6. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # transpose right for A @ Bᵀ
    right_t = right_flat.transpose(1, 2).contiguous()
    out_flat = torch.empty_like(left_flat)

    # strides for the kernel
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_t.stride(0)
    stride_bk = right_t.stride(1)
    stride_bn = right_t.stride(2)

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    M = left_flat.shape[0]      # B * hidden_dim
    N_dim = N
    K = N

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

        K,

        stride_ai,

        stride_bm,

        stride_cm,
        stride_ci,

        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # free large temporaries
    del left, right, left_flat, right_flat, right_t, out_flat

    # ------------------------------------------------------------------

    out_norm = out_norm * out_gate  # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------

    return output
