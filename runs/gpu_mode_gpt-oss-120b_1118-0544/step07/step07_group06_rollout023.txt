def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = torch.nn.functional.layer_norm(
         input_tensor,
         (dim,),
         weight=weights["norm.weight"].to(device).float(),
         bias=weights["norm.bias"].to(device).float(),
    )
    x = x.to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"].to(device).half())
    right = torch.nn.functional.linear(x, weights["right_proj.weight"].to(device).half())

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"].to(device).half()))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"].to(device).half()))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"].to(device).half()))

    B, N, _, _ = left.shape

    if mask is None or nomask:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(dtype=torch.float16, device=device)

    # Fuse mask and gates using triton
    total_elements = left.numel()
    BLOCK_SIZE = 16384  # 2^14 = 16384, multiple of 32

    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)
    mask_gate_fusion_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask_tensor.view(-1),
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        left_out_ptr=left_out,
        right_out_ptr=right_out,
        H=hidden_dim,
        total_elements=total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # Core reduction using batched GEMM
    # reshape to (B*H, N, N)
    left_flat = left_out.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right_out.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Second LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device).half(),
        bias=weights["to_out_norm.bias"].to(device).half(),
    )

    # Apply out_gate
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(device).half())

    # Cast back to original dtype
    out = out.to(orig_dtype)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor x ∈ [B,N,N,C] in fp32.
2. Project x to a hidden dimension H with two linear layers (left/right).
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. Fuse mask, projection and gates in a Triton kernel:
        left_out  = left  * mask * left_gate
        right_out = right * mask * right_gate
5. Core reduction (the O(N³) step):
        out[b,i,j,h] = Σ_k left_out[b,i,k,h] * right_out[b,j,k,h]
   Implemented as a batched GEMM on (B·H, N, N) tensors.
6. Apply a second LayerNorm over the hidden dimension H.
7. Multiply by the output gate and project back to the original channel dimension C.
8. Return the result in the original dtype of the input tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: fuse mask and gates
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # const * left  (B,N,N,H) flattened
    right_ptr,         # const * right (B,N,N,H) flattened
    mask_ptr,          # const * mask  (B,N,N)   flattened
    left_gate_ptr,     # const * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # const * right_gate (B,N,N,H) flattened
    left_out_ptr,      # * left_out  (B,N,N,H) flattened
    right_out_ptr,     # * right_out (B,N,N,H) flattened
    H: tl.constexpr,                # hidden dimension
    total_elements: tl.constexpr,   # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask and gates:
       left_out  = left  * mask * left_gate
       right_out = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load tensors
    l  = tl.load(left_ptr + offs,      mask=mask)
    r  = tl.load(right_ptr + offs,     mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs,mask=mask)

    # Mask index repeats every H elements
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply mask and gates
    tl.store(left_out_ptr  + offs, l * m * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * m * rg, mask=mask)

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:

    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict with model parameters (LayerNorm, Linear weights)
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, if True ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    x = x.to(torch.float16)

    # 2) Prepare linear weights in fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Prepare mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is None or nomask:
        mask_fp = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp = mask.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 6) Fuse mask and gates with Triton
    # ------------------------------------------------------------------
    total_elements = left.numel()                     # B * N * N * H
    BLOCK_SIZE = 16384                                 # multiple of 32, reasonable launch size

    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)
    _mask_gate_fusion_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask_fp.view(-1),
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        left_out_ptr=left_out,
        right_out_ptr=right_out,
        H=hidden_dim,
        total_elements=total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Core reduction: batched GEMM (left_out @ right_outᵀ)
    #    Reshape to (B*H, N, N) for torch.bmm.
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_flat  = left_out.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right_out.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Batched matrix multiplication (cuBLAS)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 8) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)   # (B, N, N, H)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 9) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 10) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 12) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
