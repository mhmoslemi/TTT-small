"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input over the channel dimension `dim`.
2. Cast to FP16 for the heavy compute.
3. Linear projections (left/right) and gate projections (left/right/out) are
   performed with FP16 weights.
4. If a mask is present (and not ignored) a single Triton kernel fuses
   mask, projection and gate multiplications for both the left and right
   tensors.
5. Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton kernel that performs a batched GEMM
   for every hidden slice `h` without any data reshaping or copies.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate and linearly project back to the original dimension.
7. Cast the result back to the input dtype (float32).

The implementation keeps all large‑scale tensor operations in FP16 and
accumulates in FP32 for numerical stability.  Only the mask‑gate fusion
and the GEMM are written in Triton; the remaining light‑weight operations
use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, projection and gate
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B, N, N, H)
    right_proj_ptr,  # fp16 * (B, N, N, H)
    left_gate_ptr,   # fp16 * (B, N, N, H)
    right_gate_ptr,  # fp16 * (B, N, N, H)
    mask_ptr,        # fp16 * (B, N, N)
    left_out_ptr,    # fp16 * (B, N, N, H)
    right_out_ptr,   # fp16 * (B, N, N, H)
    total_elems,     # int64 scalar = B*N*N*H
    hidden_dim,      # int64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # mask for out‑of‑range threads
    valid = offs < total_elems

    # compute flattened index for the mask (B,N,N)
    mask_idx = offs // hidden_dim

    # load tensors (masked loads)
    left_proj   = tl.load(left_proj_ptr + offs,   mask=valid, other=0.0)
    right_proj  = tl.load(right_proj_ptr + offs,  mask=valid, other=0.0)
    left_gate   = tl.load(left_gate_ptr + offs,   mask=valid, other=0.0)
    right_gate  = tl.load(right_gate_ptr + offs,  mask=valid, other=0.0)
    m           = tl.load(mask_ptr + mask_idx,   mask=valid, other=1.0)

    left_out  = left_proj  * left_gate  * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr  + offs, left_out,  mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse mask, projection and gate for both left and right tensors.
    All tensors must be FP16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)
    Returns:
        left, right : (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Triton kernel: core TriMul reduction (batched GEMM over the hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr,   # fp16 * (B,N,N,H)
    right_ptr,  # fp16 * (B,N,N,H)
    out_ptr,    # fp32 * (B,N,N,H)
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)   # combined batch*hidden*tile_i
    pid1 = tl.program_id(1)   # tile_j

    # compute number of tiles along the i dimension
    num_tiles_i = tl.cdiv(N, BLOCK_M)

    # decode pid0 into (batch, hidden, tile_i)
    tile_i = pid0 % num_tiles_i
    bh_idx = pid0 // num_tiles_i          # batch*hidden index
    batch = bh_idx // H
    hidden = bh_idx % H

    i_start = tile_i * BLOCK_M
    j_start = pid1 * BLOCK_N

    # range vectors
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (M, K)
        left = tl.load(
            left_ptr +
                batch * stride_left_b +
                offs_i[:, None] * stride_left_i +
                offs_k[None, :] * stride_left_k +
                hidden * stride_left_h,
            mask = mask_i[:, None] & mask_k[None, :],
            other = 0.0,
        )
        # right tile (K, N) -> we need right[b, k, j, h]
        right = tl.load(
            right_ptr +
                batch * stride_right_b +
                offs_k[:, None] * stride_right_k +
                offs_j[None, :] * stride_right_j +
                hidden * stride_right_h,
            mask = mask_k[:, None] & mask_j[None, :],
            other = 0.0,
        )
        # accumulate product; tl.dot promotes to fp32
        acc += tl.dot(left, right)

    # write back
    tl.store(
        out_ptr +
            batch * stride_out_b +
            offs_i[:, None] * stride_out_i +
            offs_j[None, :] * stride_out_j +
            hidden * stride_out_h,
        acc,
        mask = mask_i[:, None] & mask_j[None, :],
    )


def _tri_mul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    left/right: (B, N, N, H) FP16
    Returns out: (B, N, N, H) FP32
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32)

    # strides in element counts
    stride_l = left.stride()
    stride_r = right.stride()
    stride_o = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H * ((N + BLOCK_M - 1) // BLOCK_M),
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l[0],
        stride_l[1],
        stride_l[2],
        stride_l[3],
        stride_r[0],
        stride_r[2],   # stride for j dimension (original second N)
        stride_r[1],   # stride for k dimension (original first N)
        stride_r[3],
        stride_o[0],
        stride_o[1],
        stride_o[2],
        stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)  # (B, N, N, H)

    left_gate = F.linear(x, left_gate_w).sigmoid()    # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid()  # (B, N, N, H)
    out_gate = F.linear(x, out_gate_w).sigmoid()      # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (fused via Triton when required)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16, contiguous and without the trailing dim
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
            block_size=1024,
        )
    else:
        # Simple element‑wise fusion (mask ignored)
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask present but explicitly ignored → treat as all‑ones
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a custom Triton GEMM across the hidden dimension.
    # ------------------------------------------------------------------
    out = _tri_mul_gemm(left, right)          # (B, N, N, H) FP32
    out = out.to(torch.float16)                # cast back to FP16 for LayerNorm

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim), still FP16

    # ------------------------------------------------------------------
    # 8) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
