    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,

    # compute batch index and tile coordinates:

    num_blocks_n = tl.cdiv(N, BLOCK_N)

"""
custom_kernel: Triton‑accelerated outgoing TriMul forward (AlphaFold3).

Algorithm
---------
1. Apply LayerNorm on the channel dimension (dim) of the input tensor.
2. Cast to FP16 for compute‑heavy parts.
3. Linear projections (left_proj, right_proj) and gating projections
   (left_gate, right_gate, out_gate) are performed with torch.nn.functional.linear.
4. If a mask is present and `nomask` is False, fuse mask by simple
   element‑wise multiplication (the cost is negligible compared to the
   O(N³) reduction).
5. Fuse projection and gate by element‑wise multiplication:
        left  = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask
6. Core O(N³) reduction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication C = A @ Bᵀ where
   A and B have shape (batch*hidden, N, N).  It is implemented
   as a custom Triton GEMM kernel that iterates over the K dimension
   in tiles, using FP16 arithmetic with FP32 accumulation.
7. Apply a second LayerNorm over the hidden dimension, multiply by
   `out_gate`, and project back to the original dimension with a final linear.
8. Cast the result back to the original input dtype.

The heavy reduction step (step 6) is the only part written in Triton;
all other operations use fast PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                           # pointers
    batch_stride_a, stride_am, stride_ak,           # A strides
    batch_stride_b, stride_bk, stride_bn,           # B strides
    batch_stride_c, stride_cm, stride_cn,           # C strides
    M, N, K,                                        # matrix dimensions (all int64)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_BLOCKS_M: tl.constexpr, NUM_BLOCKS_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1) Decode program ID into batch index and tile (i,j) indices
    # --------------------------------------------------------------
    batch_idx = pid // (NUM_BLOCKS_M * NUM_BLOCKS_N)
    tile_idx  = pid % (NUM_BLOCKS_M * NUM_BLOCKS_N)
    pid_m = tile_idx // NUM_BLOCKS_N       # block row
    pid_n = tile_idx % NUM_BLOCKS_N        # block column

    # --------------------------------------------------------------
    # 2) Compute offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # --------------------------------------------------------------
    # 3) Accumulate over K dimension
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A tile [M, K]
        a_ptrs = a_ptr + batch_idx * batch_stride_a \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # Load B tile [K, N] (we need Bᵀ, so we load B as [k, n])
        b_ptrs = b_ptr + batch_idx * batch_stride_b \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # 4) Write back the result tile
    # --------------------------------------------------------------
    c_ptrs = c_ptr + batch_idx * batch_stride_c \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to FP16 for the heavy compute
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast weights to FP16 (no bias in linear layers)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections and gates
    # --------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)                       # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)                     # (B, N, N, H)

    left_gate = F.linear(x, left_gate_w).sigmoid()            # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid()          # (B, N, N, H)
    out_gate = F.linear(x, out_gate_w).sigmoid()              # (B, N, N, H)

    # --------------------------------------------------------------
    # 5) Apply mask (if any) and fuse projection with gate
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask_fp16
            right = right * mask_fp16

    # --------------------------------------------------------------
    # 6) Core O(N³) reduction via a custom Triton batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_flat = torch.empty_like(left_flat)

    # strides (in elements) for A, B, C
    batch_stride_a = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    batch_stride_b = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    batch_stride_c = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # Tile sizes (tuned for Hopper B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    NUM_BLOCKS_M = (N + BLOCK_M - 1) // BLOCK_M
    NUM_BLOCKS_N = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = (B * H) * NUM_BLOCKS_M * NUM_BLOCKS_N

    grid = (total_programs,)

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_stride_a, stride_am, stride_ak,
        batch_stride_b, stride_bk, stride_bn,
        batch_stride_c, stride_cm, stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        NUM_BLOCKS_M=NUM_BLOCKS_M,
        NUM_BLOCKS_N=NUM_BLOCKS_N,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final projection

        bias=to_out_norm_b,
