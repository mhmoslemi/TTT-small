def custom_kernel(data):
    input_tensor, mask, weights, config = data
    ...
    # 1. LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Cast to fp16
    x = x.to(torch.float16)
    # Weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections
    left_proj = F.linear(x, left_proj_w)  # shape B,N,N,hidden_dim
    right_proj = F.linear(x, right_proj_w)

    # Gating
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply mask gating
    if mask is not None and not config.get("nomask", False):
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(left_proj, right_proj, left_gate, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not config.get("nomask", False):
            # If mask exists but ignored; but main condition ensures we ignore mask.
            pass

    # Core reduction using Triton batched matmul
    B, N, _, H = left.shape
    # reshape
    left_flat = left.permute(0, 3, 1, 2).contiguous()  # shape (B, H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous()
    # flatten batch and H
    left_flat = left_flat.view(B * H, N, N)
    right_flat = right_flat.view(B * H, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)  # same dtype fp16

    # get strides
    stride_a_m = left_flat.stride(1)  # stride for rows (M)
    stride_a_k = left_flat.stride(2)  # stride for cols (K)
    stride_b_n = right_flat.stride(1)  # rows (N) for B (since B shape (batch, N, K))
    stride_b_k = right_flat.stride(2)  # cols (K)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    batch_stride_a = left_flat.stride(0)
    batch_stride_b = right_flat.stride(0)
    batch_stride_c = out_flat.stride(0)

    # grid
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # compute number of blocks
    num_tiles_m = triton.cdiv(N, BLOCK_M)
    num_tiles_n = triton.cdiv(N, BLOCK_N)

    grid = (num_tiles_m, num_tiles_n, B * H)

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_a_m, stride_a_k,
        stride_b_k, stride_b_n,
        stride_c_m, stride_c_n,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # output layer norm, gating, final linear
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)

    # cast back to original dtype
    out = out.to(input_tensor.dtype)

    return out

"""
TriMul outgoing (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for compute‑intensive parts.
3. Linear projections (no bias) → left_proj, right_proj.
4. Gating projections → left_gate, right_gate, out_gate (sigmoid).
5. Fuse mask, projection and gating with a tiny Triton kernel when a mask is used.
6. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (left @ rightᵀ) for each
   hidden slice `h`.  Implemented with a custom Triton kernel that
   processes many N×N matrix products in parallel.
7. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.
8. Cast back to the original dtype (usually float32).

The implementation uses Triton for two parts:
* element‑wise mask‑gate fusion,
* the batched matrix multiplication (the dominant N³ work).
All other operations are performed with PyTorch (FP16 where possible)
for simplicity and to keep the code concise.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, projection and gate for left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B*N*N*H,)
    right_proj_ptr,  # fp16 * (B*N*N*H,)
    left_gate_ptr,   # fp16 * (B*N*N*H,)
    right_gate_ptr,  # fp16 * (B*N*N*H,)
    mask_ptr,        # fp16 * (B*N*N,)
    left_out_ptr,    # fp16 * (B*N*N*H,)
    right_out_ptr,   # fp16 * (B*N*N*H,)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    in_range = offs < total_elems

    # mask index repeats `hidden_dim` times
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=in_range, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=in_range, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=in_range, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=in_range, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=in_range, other=1.0)

    left_out  = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs,  left_out,  mask=in_range)
    tl.store(right_out_ptr + offs, right_out, mask=in_range)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gate and mask for both left and right tensors using Triton.

    All tensors must be fp16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)

    Returns:
        left, right : (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
# A: (M, K)   B: (N, K)   C: (M, N)
# Batch dimension is encoded in the third program ID.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix sizes
    stride_am, stride_ak,               # A strides (row, col)
    stride_bk, stride_bn,               # B strides (col, row) – note Bᵀ access
    stride_cm, stride_cn,               # C strides (row, col)
    batch_stride_a, batch_stride_b, batch_stride_c,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_b = tl.program_id(2)   # batch index (b * hidden)

    # Offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers that include the batch offset
    a_batch_offset = pid_b * batch_stride_a
    b_batch_offset = pid_b * batch_stride_b
    c_batch_offset = pid_b * batch_stride_c

    # Accumulator in FP32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = k_offset + tl.arange(0, BLOCK_K)
        mask_k = k_range < K

        # Load A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + a_batch_offset + \
                 offs_m[:, None] * stride_am + \
                 k_range[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load Bᵀ tile: (BLOCK_K, BLOCK_N)
        # B is stored as (N, K); we want Bᵀ so we index (k, n)
        b_ptrs = b_ptr + b_batch_offset + \
                 k_range[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Write result back to C
    c_ptrs = c_ptr + c_batch_offset + \
             offs_m[:, None] * stride_cm + \
             offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] (0/1) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast all linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (fusion via Triton if mask is used)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask or mask ignored → simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction (TriMul) using a batched matmul kernel
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)  # fp16 output

    # strides (row‑major layout)
    stride_a_m = left_flat.stride(1)   # row stride (N)
    stride_a_k = left_flat.stride(2)   # col stride (1)
    stride_b_k = right_flat.stride(2)  # col stride (1)
    stride_b_n = right_flat.stride(1)  # row stride (N)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)
    batch_stride_a = left_flat.stride(0)   # N*N
    batch_stride_b = right_flat.stride(0)  # N*N
    batch_stride_c = out_flat.stride(0)    # N*N

    # Tile sizes – tuned for typical B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # tiles over rows (i)
        triton.cdiv(N, BLOCK_N),   # tiles over cols (j)
        B * H,                     # batch * hidden
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                           # M, N, K
        stride_a_m, stride_a_k,
        stride_b_k, stride_b_n,
        stride_c_m, stride_c_n,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
