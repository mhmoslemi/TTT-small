"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input (float32 for stability) → fp16.
2. Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
   into a single matrix‑multiply to reduce kernel launch overhead.
3. Apply sigmoid to the three gate outputs.
4. Fuse mask, left‑gate and right‑gate multiplications in a Triton kernel
   (in‑place on left/right).
5. Perform the core O(N³·H) reduction as a batched matrix multiplication:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to left @ rightᵀ for each (b,h) pair.
6. Layer‑normalize over the hidden dimension, multiply by the output gate,
   and project back to the original channel dimension.
7. Cast the result back to the original dtype (float32).

Only the mask‑gate fusion is done in Triton; all other heavy
operations use highly‑optimized torch kernels (Tensor‑cores for fp16).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (B,N,N,H) flattened
    right_ptr,         # fp16 * right (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    TOTAL: tl.constexpr,  # total number of elements = B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and gate multiplications:
       left  = left  * mask * left_gate
       right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL

    # Load data
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # One mask value per (b,i,k) position, broadcast over H
    mask_idx = offs // H          # integer division
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply and store back in‑place
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or fp16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see keys below)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # Cast to fp16 for the compute‑heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Merge the five linear layers into one matrix multiply
    # ------------------------------------------------------------------
    # Cast all required weights to fp16
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # Concatenate the five weight matrices (order: left, right, left_gate, right_gate, out_gate)
    combined_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)                     # (B*N*N, dim)

    # Single fused linear projection
    all_out = F.linear(x_flat, combined_weight)     # (B*N*N, 5*hidden_dim)
    all_out = all_out.view(B, N, N, 5, hidden_dim)  # (B, N, N, 5, H)

    # Split results
    left      = all_out[..., 0, :]                 # (B, N, N, H)
    right     = all_out[..., 1, :]                 # (B, N, N, H)
    left_gate = torch.sigmoid(all_out[..., 2, :])  # (B, N, N, H)
    right_gate= torch.sigmoid(all_out[..., 3, :])  # (B, N, N, H)
    out_gate  = torch.sigmoid(all_out[..., 4, :])  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3) Fuse mask, left‑gate and right‑gate (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()                     # B*N*N*H
        BLOCK = 32768                                     # threads per program
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates no longer needed after fusion
        del left_gate, right_gate
    else:
        # Simple element‑wise multiplication when no mask is present
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 4) Core reduction: batched matrix multiplication (left @ rightᵀ)
    # ------------------------------------------------------------------
    # Reshape to (B*H, N, N) for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matmul using Tensor Cores (fp16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Restore (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5) Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float16),
    )

    # ------------------------------------------------------------------
    # 6) Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].to(device=device, dtype=torch.float16)
    )   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 8) Cast back to the original dtype (float32)
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
