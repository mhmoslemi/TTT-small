Define kernel:

@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr, right_ptr,
    mask_ptr,
    left_gate_ptr, right_gate_ptr,
    H: tl.constexpr,
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    l = l * m * lg
    r = r * m * rg

    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm Overview
------------------
1. Layer‑normalize the input tensor (fp32 for numerical stability).
2. Cast to fp16 and compute the linear projections and their sigmoid gates.
   - Left projection & left gate are computed together by a single matmul
     using a concatenated weight matrix (2·H × C).
   - Right projection & right gate are analogously computed.
3. Compute the output gate (sigmoid of a linear projection).
4. Fuse mask, left/right gates and the projections with a small Triton kernel:
       left  ← left  * mask * left_gate
       right ← right * mask * right_gate
   The kernel works in‑place, avoiding extra memory copies.
5. Core TriMul reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication (torch.bmm) on
   reshaped tensors (B·H, N, N), which lets cuBLAS handle the O(N³·H)
   work efficiently.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate and project back to the original channel dimension.
7. Cast the result back to the original input dtype.

The implementation fuses as many elementwise operations as possible into
the Triton kernel while keeping the heavy O(N³·H) reduction in cuBLAS,
yielding a fast forward pass suitable for the target test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # fp16*  (B,N,N,H) flattened, also output
    right_ptr,         # fp16*  (B,N,N,H) flattened, also output
    mask_ptr,          # fp16*  (B,N,N)   flattened
    left_gate_ptr,     # fp16*  (B,N,N,H) flattened
    right_gate_ptr,    # fp16*  (B,N,N,H) flattened
    H: tl.constexpr,                # hidden dimension
    total_elements: tl.constexpr,   # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and sigmoid gates:
       left  = left  * mask * left_gate
       right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # Load data
    l = tl.load(left_ptr + offs, mask=valid)
    r = tl.load(right_ptr + offs, mask=valid)
    lg = tl.load(left_gate_ptr + offs, mask=valid)
    rg = tl.load(right_gate_ptr + offs, mask=valid)

    # Convert flat offset to mask index (mask repeats for each hidden dim)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=valid)

    # Apply mask and gates
    l = l * m * lg
    r = r * m * rg

    # Store results back in‑place
    tl.store(left_ptr + offs, l, mask=valid)
    tl.store(right_ptr + offs, r, mask=valid)


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (LayerNorm, Linear weights)
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, if True ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # --------------------------------------------------------------
    # 0) Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (fp32 for stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )  # (B, N, N, C), fp32

    # --------------------------------------------------------------
    # 2) Cast to fp16 for the heavy computations
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Prepare fused projection+gate weights (fp16)
    # --------------------------------------------------------------
    # Left: projection + left gate
    left_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)          # (2*H, C)

    # Right: projection + right gate
    right_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)          # (2*H, C)

    # Output gate weight
    out_gate_weight = weights["out_gate.weight"].to(
        device=device, dtype=torch.float16
    )  # (H, C)

    # --------------------------------------------------------------
    # 4) Compute projections + gates in a fused manner
    # --------------------------------------------------------------
    # Left: (B,N,N,2*H) -> split into proj and gate
    left_combined = F.linear(x, left_weight)          # (B, N, N, 2*H)
    left_proj, left_gate = torch.split(
        left_combined, [hidden_dim, hidden_dim], dim=-1
    )
    left_gate = torch.sigmoid(left_gate)              # (B,N,N,H)

    # Right: (B,N,N,2*H) -> split
    right_combined = F.linear(x, right_weight)        # (B, N, N, 2*H)
    right_proj, right_gate = torch.split(
        right_combined, [hidden_dim, hidden_dim], dim=-1
    )
    right_gate = torch.sigmoid(right_gate)            # (B,N,N,H)

    # Output gate (no split)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # (B,N,N,H)

    # --------------------------------------------------------------
    # 5) Prepare mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    B, N, _, _ = left_proj.shape
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
    else:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # --------------------------------------------------------------
    # 6) Fuse mask and gates with Triton (in‑place on left_proj/right_proj)
    # --------------------------------------------------------------
    total_elements = left_proj.numel()  # B * N * N * H
    BLOCK_SIZE = 1024  # multiple of 32, good occupancy

    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    _mask_gate_fusion_kernel[grid](
        left_ptr=left_proj,
        right_ptr=right_proj,
        mask_ptr=mask_fp16.view(-1),
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        H=hidden_dim,
        total_elements=total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 7) Core reduction: batched GEMM (left_proj @ right_projᵀ)
    #    Shapes: (B, H, N, N) -> (B*H, N, N)
    # --------------------------------------------------------------
    left_flat = left_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication (cuBLAS)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, H)

    # --------------------------------------------------------------
    # 8) Second LayerNorm over hidden dimension H
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"]
        .to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"]
        .to(device=device, dtype=torch.float16),
    )

    # --------------------------------------------------------------
    # 9) Apply output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 10) Final projection back to original channel dimension C
    # --------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].to(device=device, dtype=torch.float16),
    )  # (B, N, N, C), fp16

    # --------------------------------------------------------------
    # 11) Cast back to original dtype if needed
    # --------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
