"""
TriMul outgoing kernel (AlphaFold3) – forward pass.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (`dim`).
2. Linear projections to a hidden dimension `hidden_dim` for left/right tensors.
3. Compute sigmoid gates (left, right, out) from the normalized input.
4. Fuse mask (if present), projection and gate with a tiny Triton kernel.
   The kernel works in the same dtype as the tensors (float32 by default).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply for each hidden slice.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   output‑gate and finally project back to the original `dim`.
7. Return the result in the original input dtype.

The implementation mixes PyTorch for the heavy matrix‑multiply (cuBLAS) and a
custom Triton kernel for the element‑wise mask‑gate fusion, satisfying the
requirement to use Triton while keeping the code simple and correct.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # * (B, N, N, H)   - fp32
    right_proj_ptr,  # * (B, N, N, H)   - fp32
    left_gate_ptr,   # * (B, N, N, H)   - fp32
    right_gate_ptr,  # * (B, N, N, H)   - fp32
    mask_ptr,        # * (B, N, N)      - fp32 (0/1)
    left_out_ptr,    # * (B, N, N, H)   - fp32
    right_out_ptr,   # * (B, N, N, H)   - fp32
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gate and (optional) mask for left/right tensors."""
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # index of the mask element (B,N,N) for the current element of left/right
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out  = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs,  left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Apply mask, projection and gate in a single Triton kernel.
    All tensors must be contiguous and have matching dtype.
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict containing model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Linear projections (no bias) and gates
    # --------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])   # (B, N, N, hidden_dim)
    right_proj = F.linear(x, weights["right_proj.weight"]) # (B, N, N, hidden_dim)

    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # --------------------------------------------------------------
    # 3) Fuse mask, projection and gate (Triton kernel) if needed
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is float32 and contiguous
        mask_fp = mask.to(torch.float32).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp, block_size=1024
        )
    else:
        # Simple element‑wise multiplication (mask ignored or absent)
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask present but instructed to ignore – still broadcast
            mask_fp = mask.to(torch.float32).contiguous()
            left = left * mask_fp[..., None]
            right = right * mask_fp[..., None]

    # --------------------------------------------------------------
    # 4) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with batched GEMM (cuBLAS) for each hidden slice.
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) -> perform bmm per hidden slice
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5) Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 6) Cast back to original dtype if needed
    # --------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
