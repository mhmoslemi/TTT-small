@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab, # batch stride for a (maybe stride_ab)
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # pointers to the start of this batch
    a_batch_ptr = a_ptr + batch_id * stride_ab
    b_batch_ptr = b_ptr + batch_id * stride_bb
    c_batch_ptr = c_ptr + batch_id * stride_cb

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # load A and B tiles
        a = tl.load(a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        # B is transposed: we need B^T, so we can load B with swapped indices (k, n) -> B_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        b = tl.load(b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        # compute
        acc += tl.dot(a, b)

        # increment pointers for next k block (not necessary with offsets recomputed each iteration)
    # write out
    c = acc.to(tl.float16)
    tl.store(c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
             c,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

But note: we used a dot on (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> result (BLOCK_M, BLOCK_N). However tl.dot expects shape (BLOCK_M, BLOCK_K) and (BLOCK_K, BLOCK_N) and returns (BLOCK_M, BLOCK_N). Good.

@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ama, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cma, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute masks for valid indices
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer base for this batch
    a_batch_ptr = a_ptr + batch_id * stride_ab
    b_batch_ptr = b_ptr + batch_id * stride_bb
    c_batch_ptr = c_ptr + batch_id * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile
        a = tl.load(
            a_batch_ptr + (offs_m[:, None] * stride_ama + offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )

        # load B tile (transposed)
        b = tl.load(
            b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )

        acc += tl.dot(a, b)

    # store result
    c = acc.to(tl.float16)
    tl.store(
        c_batch_ptr + (offs_m[:, None] * stride_cma + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :]
    )

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Input LN
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # cast to fp16
    x = x.to(torch.float16)

    # cast weights
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # Projections
    left_proj = F.linear(x, left_proj_w)   # B,N,N,H
    right_proj= F.linear(x, right_proj_w)

    # mask apply
    if mask is not None and not nomask:
        mask = mask.to(torch.float16).unsqueeze(-1)  # shape (B,N,N,1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # Gates
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    left = left_proj * left_gate
    right= right_proj * right_gate

    # Core reduction using Triton
    B, N, _, H = left.shape
    # permute to (B, H, N, N)
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    # flatten batch*H
    left_flat = left_perm.view(B*H, N, N)
    right_flat = right_perm.view(B*H, N, N)
    out_flat = torch.empty_like(left_flat)  # shape (B*H,N,N)

    # compute strides
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(1)  # row stride for K dimension
    stride_b_n = right_flat.stride(2)  # col stride for N dimension

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B*H, tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N))
    # Actually triton uses grid tuple of ints; we must compute using math.ceil.
    import math
    grid = (B*H, math.ceil(N / BLOCK_M), math.ceil(N / BLOCK_N))

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_a_m, stride_a_k, stride_a_batch,
        stride_b_k, stride_b_n, stride_b_batch,
        stride_c_m, stride_c_n, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape output back
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()

    # Output LayerNorm
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # out gate
    out = out * out_gate

    # final linear
    out = F.linear(out, to_out_w)

    # cast back
    out = out.to(input_tensor.dtype)

    return out

"""
TriMul outgoing implementation (AlphaFold3).

- Input LayerNorm over the channel dimension.
- Linear projections (left/right) and gating (left/right/out) are bias‑free.
- Optional binary mask is fused with the projections (element‑wise multiply).
- Core reduction:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
  implemented as a batched matrix multiplication (left @ rightᵀ) using a
  custom Triton kernel that processes each (batch, hidden) pair independently.
- Output LayerNorm, output‑gate, and final linear projection back to the
  original dimension.  Result is returned in the original dtype (float32/float16).
"""

import math
from typing import Tuple, Dict

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr,               # fp16 * (batch, M, K)
    b_ptr,               # fp16 * (batch, K, N)   (we will read it as transposed)
    c_ptr,               # fp16 * (batch, M, N)
    M, N, K,             # matrix sizes (int)
    stride_am, stride_ak, stride_ab,   # strides for A (row, col, batch)
    stride_bk, stride_bn, stride_bb,   # strides for B (row, col, batch) – B is (K,N)
    stride_cm, stride_cn, stride_cb,   # strides for C (row, col, batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  (A,B,C are fp16, accumulation in fp32)."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this batch
    a_batch = a_ptr + batch_id * stride_ab
    b_batch = b_ptr + batch_id * stride_bb
    c_batch = c_ptr + batch_id * stride_cb

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile transposed: we need Bᵀ so we read B as (k, n)
        b = tl.load(
            b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot product, accumulate in fp32
        acc += tl.dot(a, b)

    # write result back as fp16
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] or None (binary)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Optional mask (fused as element‑wise multiply)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B, N, N) → (B, N, N, 1) for broadcasting
        mask_fp = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_fp
        right_proj = right_proj * mask_fp

    # ------------------------------------------------------------------
    # 6) Gating (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    left  = left_proj * left_gate   # (B, N, N, H)
    right = right_proj * right_gate # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (left @ rightᵀ) via Triton.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reorder to (B, H, N, N) and make contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # flatten batch*head dimension
    left_flat  = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # allocate output buffer
    out_flat = torch.empty_like(left_flat)

    # strides for the contiguous tensors
    stride_a_batch = left_flat.stride(0)
    stride_am = left_flat.stride(1)   # row stride (M)
    stride_ak = left_flat.stride(2)   # col stride (K)

    stride_b_batch = right_flat.stride(0)
    stride_bk = right_flat.stride(1)   # row stride (K) for B (since B is K×N)
    stride_bn = right_flat.stride(2)   # col stride (N)

    stride_c_batch = out_flat.stride(0)
    stride_cm = out_flat.stride(1)    # row stride (M)
    stride_cn = out_flat.stride(2)    # col stride (N)

    # block sizes – tuned for typical GPU shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        math.ceil(N / BLOCK_M),
        math.ceil(N / BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        stride_am,
        stride_ak,
        stride_a_batch,
        stride_bk,
        stride_bn,
        stride_b_batch,
        stride_cm,
        stride_cn,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
