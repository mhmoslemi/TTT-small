def triton_matmul_kernel(A_ptr, B_ptr, C_ptr,
                         M, N, K,
                         stride_am, stride_ak,
                         stride_bk, stride_bn,
                         stride_cm, stride_cn,
                         batch_stride_a, batch_stride_b, batch_stride_c,
                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
   pid_b = tl.program_id(0)   # batch index
   pid_m = tl.program_id(1)   # tile row
   pid_n = tl.program_id(2)   # tile col

   # block start offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

   # compute mask for out-of-bounds
   mask_m = offs_m < M
   mask_n = offs_n < N

   # get pointers for A, B with batch offset
   a_ptrs = A_ptr + pid_b * batch_stride_a + (offs_m[:, None] * stride_am) + (tl.arange(0, BLOCK_K)[None, :] * stride_ak)
   b_ptrs = B_ptr + pid_b * batch_stride_b + (tl.arange(0, BLOCK_K)[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

   # initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # iterate over K dimension
   for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute K offsets
        k_offset = k * BLOCK_K
        a = tl.load(a_ptrs + k_offset * stride_ak, mask=mask_m[:, None] & (k_offset + tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs + k_offset * stride_bk, mask=(k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
        # dot product
        acc += tl.dot(a, b)

   # cast back to fp16
   c = acc.to(tl.float16)

   # store
   c_ptrs = C_ptr + pid_b * batch_stride_c + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
   tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def _batched_matmul(left: torch.Tensor, right: torch.Tensor, block_m: int = 64, block_n: int = 64, block_k: int = 64):
    # left,right shape (B,N,N,H)
    B, N, _, H = left.shape
    # Permute to (B,H,N,N) and view to (B*H,N,N)
    left_ = left.permute(0, 3, 1, 2).contiguous()
    right_ = right.permute(0, 3, 1, 2).contiguous()

"""
TriMul outgoing (AlphaFold3) forward pass with Triton fusion.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections (left/right) and gating projections (left/right/out) – all
   without bias, all in FP16.
4. Fuse projection, gating and optional mask with a small Triton kernel.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix‑multiply using a custom Triton kernel.
6. Output LayerNorm, multiply by `out_gate`, final linear projection back to `dim`.
7. Cast back to the original dtype (float32).

At least one operation (projection + gate + mask) and the core O(N³) reduction
are executed in Triton kernels for maximum performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Kernel 1: fuse projection, gating and mask
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask:
       left_out  = left_proj  * left_gate  * mask
       right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    # index in the (B,N,N) mask (no hidden dim)
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # default 1.0 = no mask

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 4096,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launches the Triton kernel that fuses projection, gating and mask.
    All tensors must be fp16 and contiguous.
    Returns (left, right) each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Kernel 2: batched matrix‑multiply for the O(N³) reduction
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes for each batch index:
        C[b, i, j] = Σ_k A[b, i, k] * B[b, j, k]
    where A, B, C are (batch, M, K), (batch, N, K), (batch, M, N) respectively.
    B is interpreted as transposed inside the dot product.
    """
    pid_b = tl.program_id(0)          # batch * hidden
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = A_ptr + pid_b * batch_stride_a + (offs_m[:, None] * stride_am)
    b_base = B_ptr + pid_b * batch_stride_b + (offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_base + (offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile as (BLOCK_K, BLOCK_N) because we need B^T
        b = tl.load(
            b_base + (offs_k[:, None] * stride_bk),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N) accumulation in fp32

    # Write back result in fp16
    c = acc.to(tl.float16)
    c_ptr = C_ptr + pid_b * batch_stride_c + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptr, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul(
    left: torch.Tensor,
    right: torch.Tensor,
    block_m: int = 64,
    block_n: int = 64,
    block_k: int = 64,
) -> torch.Tensor:
    """
    Perform the core TriMul reduction using the custom Triton GEMM kernel.
    left, right : (B, N, N, H) fp16 tensors.
    Returns out  : (B, N, N, H) fp16 tensor.
    """
    B, N, _, H = left.shape

    # Rearrange to (B, H, N, N) and flatten batch*H
    left_ = left.permute(0, 3, 1, 2).contiguous()
    right_ = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_.view(B * H, N, N)
    right_flat = right_.view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)

    batch = B * H
    grid = (
        batch,
        triton.cdiv(N, block_m),
        triton.cdiv(N, block_n),
    )
    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        N,
        left_flat.stride(1),   # stride_am
        left_flat.stride(2),   # stride_ak
        right_flat.stride(2),  # stride_bk (K dimension)
        right_flat.stride(1),  # stride_bn (N dimension)
        out_flat.stride(1),    # stride_cm
        out_flat.stride(2),    # stride_cn
        left_flat.stride(0),   # batch stride A
        right_flat.stride(0),  # batch stride B
        out_flat.stride(0),    # batch stride C
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) – may be all‑ones.
            - weights: dict containing model parameters (float32)
            - config: dict with keys:
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid, FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Fuse mask, projection and gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul) – batched GEMM over the hidden dim
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    out = _batched_matmul(left, right, block_m=64, block_n=64, block_k=64)

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim), still fp16

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype (float32)
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
