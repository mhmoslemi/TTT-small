```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])
    # cast to fp16
    x = x.to(torch.float16)

    # cast weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # projections
    left_proj = F.linear(x, left_proj_w)  # shape (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)
    
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # apply mask+gate
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(left_proj, right_proj,
                                            left_gate, right_gate,
                                            mask_fp16,
                                            block_size=1024)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # core reduction using Triton GEMM (batched)
    # reshape for batched GEMM
    B, N, _, H = left.shape
    left_flat = left.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous()
    # combine batch and head
    left_batch = left_flat.view(B * H, N, N)
    right_batch = right_flat.view(B * H, N, N)

    out_batch = _triton_batched_gemm(left_batch, right_batch)

    # reshape back
    out = out_batch.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # output layernorm
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # out_gate multiplication
    out = out * out_gate

    # final linear
    out = F.linear(out, to_out_w)

    # cast back to float32
    out = out.to(torch.float32)

    return out
```

```python
def _triton_batched_gemm(A, B):
    # A and B: torch tensors of shape (BATCH, M, K) and (BATCH, N, K)
    # Returns C: (BATCH, M, N)
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2

    # Output
    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Strides
    stride_Ab = A.stride(0)
    stride_Am = A.stride(1)
    stride_Ak = A.stride(2)

    stride_Bb = B.stride(0)
    stride_Bn = B.stride(1)  # N dimension
    stride_Bk = B.stride(2)

    stride_Cb = C.stride(0)
    stride_Cm = C.stride(1)
    stride_Cn = C.stride(2)

    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (tl.cdiv(M, BLOCK_M), tl.cdiv(N, BLOCK_N), batch)

    _gemm_batch_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_Ab,
        stride_Am,
        stride_Ak,
        stride_Bb,
        stride_Bn,
        stride_Bk,
        stride_Cb,
        stride_Cm,
        stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C
```

    offset_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute batch offset
    a_batch_ptr = A_ptr + pid_b * stride_Ab
    b_batch_ptr = B_ptr + pid_b * stride_Bb
    c_batch_ptr = C_ptr + pid_b * stride_Cb

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension

        offset_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A block (M x K)

    mask = (offset_m[:, None] < M) & (offset_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

"""
custom_kernel: Triton‑accelerated outgoing TriMul forward (AlphaFold3).

The implementation follows the PyTorch reference:
1️⃣ LayerNorm over the channel dimension.
2️⃣ Linear projections (left/right) and gates (left/right/out) cast to fp16.
3️⃣ Optional mask application fused with projection & gating in a Triton kernel.
4️⃣ Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (A @ Bᵀ) using a custom Triton kernel that
   processes the (B·H) independent matrix multiplications with tiling.
5️⃣ Output LayerNorm, multiplication by out‑gate and final linear projection.
6️⃣ Cast back to float32.

All heavy element‑wise work (proj × gate × mask) and the N³ reduction are
performed on the GPU, giving a large speed‑up over the pure‑PyTorch version.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, projection and gate for left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B, N, N, H)
    right_proj_ptr,  # fp16 * (B, N, N, H)
    left_gate_ptr,   # fp16 * (B, N, N, H)
    right_gate_ptr,  # fp16 * (B, N, N, H)
    mask_ptr,        # fp16 * (B, N, N)
    left_out_ptr,    # fp16 * (B, N, N, H)
    right_out_ptr,   # fp16 * (B, N, N, H)
    total_elems,     # i64 = B * N * N * H
    hidden_dim,      # i64 = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # Bounds check
    mask = offs < total_elems

    # Index of the (b,i,j) element that each hidden slice belongs to
    idx_3d = offs // hidden_dim

    # Load tensors
    left_proj = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + idx_3d, mask=mask, other=1.0)

    # Fuse: (proj * gate) * mask
    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse mask, projection and gate for both left and right tensors using Triton.
    All tensors must be fp16 and contiguous.
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for the core TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _gemm_batch_kernel(
    A_ptr,                 # fp16 * (BATCH, M, K)
    B_ptr,                 # fp16 * (BATCH, N, K)   (will be read as K x N)
    C_ptr,                 # fp16 * (BATCH, M, N)
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bn, stride_Bk,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block index over M
    pid_n = tl.program_id(1)   # block index over N
    pid_b = tl.program_id(2)   # batch (BATCH)

    # Offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch
    a_batch = A_ptr + pid_b * stride_Ab
    b_batch = B_ptr + pid_b * stride_Bb
    c_batch = C_ptr + pid_b * stride_Cb

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A block [M, K]
        a_ptrs = a_batch + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # Load Bᵀ block [K, N] (read B as (N, K) and transpose on‑the‑fly)
        b_ptrs = b_batch + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

    # Write result
    c_ptrs = c_batch + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def _triton_batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM: C = A @ Bᵀ
    A, B : (batch, M, K) and (batch, N, K)  (fp16, contiguous)
    Returns C : (batch, M, N) (fp16)
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Strides (in elements)
    stride_Ab = A.stride(0)
    stride_Am = A.stride(1)
    stride_Ak = A.stride(2)

    stride_Bb = B.stride(0)
    stride_Bn = B.stride(1)
    stride_Bk = B.stride(2)

    stride_Cb = C.stride(0)
    stride_Cm = C.stride(1)
    stride_Cn = C.stride(2)

    # Tiling parameters (tuned for typical N ≤ 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )

    _gemm_batch_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_Ab,
        stride_Am,
        stride_Ak,
        stride_Bb,
        stride_Bn,
        stride_Bk,
        stride_Cb,
        stride_Cm,
        stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Main entrypoint
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys:
            * "dim"        – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask" (optional) – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) LayerNorm over channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections and gates (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))     # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask + gating (fused via Triton when mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            block_size=1024,
        )
    else:
        # No mask or mask ignored
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # Mask present but explicitly ignored → still broadcast
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (A @ Bᵀ) using Triton.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Rearrange to (B*H, N, N) for the batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = _triton_batched_gemm(left_flat, right_flat)  # (B*H, N, N)

    # Restore shape (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to float32 (required output dtype)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
