@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_idx = tl.program_id(2)
    bi = tl.program_id(0)
    bj = tl.program_id(1)

    # derive batch and hidden
    b = batch_idx // H
    h = batch_idx % H

    h_offset_left = h * stride_left_h
    h_offset_right = h * stride_right_h
    h_offset_out = h * stride_out_h

    i0 = bi * BLOCK_M
    j0 = bj * BLOCK_N

    i_offsets = i0 + tl.arange(0, BLOCK_M)
    j_offsets = j0 + tl.arange(0, BLOCK_N)

    mask_i = i_offsets < N
    mask_j = j_offsets < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        k_offsets = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < N

        # load A tile
        a_ptrs = left_ptr + b * stride_left_b + i_offsets[:, None] * stride_left_i + k_offsets[None, :] * stride_left_k + h_offset_left
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load B_T tile (right)
        b_ptrs = right_ptr + b * stride_right_b + j_offsets[None, :] * stride_right_i + k_offsets[:, None] * stride_right_k + h_offset_right
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # store
    out_ptrs = out_ptr + b * stride_out_b + i_offsets[:, None] * stride_out_i + j_offsets[None, :] * stride_out_j + h_offset_out
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Cast
    x = x.to(torch.float16)

    # Convert weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections
    left_proj = F.linear(x, left_proj_w)  # shape B,N,N,hidden_dim
    right_proj = F.linear(x, right_proj_w)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Fuse with mask
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()

    # Core reduction using Triton
    out = tri_mul_kernel_launch(left, right) # we need to define

    # Output layernorm
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection
    out = F.linear(out, to_out_w)  # (B,N,N,dim)

    # Cast back
    out = out.to(input_tensor.dtype)

    return out

        a_ptrs = left_ptr + b * stride_left_b + i_range[:, None] * stride_left_i + k_range[None, :] * stride_left_k + off_h_left
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # right: B_T[k,j] = right[b, j, k, h]
        b_ptrs = right_ptr + b * stride_right_b + j_range[None, :] * stride_right_i + k_range[:, None] * stride_right_k + off_h_right
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # store result
    out_ptrs = out_ptr + b * stride_out_b + i_range[:, None] * stride_out_i + j_range[None, :] * stride_out_j + off_h_out
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h] using Triton."""

    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # block sizes (tuned)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)

        left,
        right,
        out,
        B,
        N,
        H,

        stride_left[1],
        stride_left[2],

        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

    out = F.linear(out, to_out_w)

    out = out.to(input_tensor.dtype)

"""
Outgoing TriMul (AlphaFold3) forward pass.
- LayerNorm on the input.
- Linear projections + sigmoid gates.
- Optional mask fused with projection & gate (elementwise).
- Core N‑wise reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
  implemented as a batched matrix multiplication using a custom Triton kernel.
- Output LayerNorm, out‑gate, final linear projection.
All heavy compute is performed in FP16; final result is cast back to the
input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]   .
    The kernel is launched with a 3‑D grid:
        (num_i_tiles, num_j_tiles, B*H)
    where each program processes a (BLOCK_M×BLOCK_N) tile of the output
    for a particular (batch, hidden) pair.
    """
    pid_bh = tl.program_id(2)      # index over batch*hidden
    pid_i = tl.program_id(0)       # tile row
    pid_j = tl.program_id(1)       # tile column

    # derive batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # hidden‑dimension offsets
    off_h_left = h * stride_left_h
    off_h_right = h * stride_right_h
    off_h_out = h * stride_out_h

    # start indices of the tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    i_range = i0 + tl.arange(0, BLOCK_M)
    j_range = j0 + tl.arange(0, BLOCK_N)

    mask_i = i_range < N
    mask_j = j_range < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        k_range = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # left tile:  A[i,k]  (shape BLOCK_M×BLOCK_K)
        a_ptrs = (
            left_ptr
            + b * stride_left_b
            + i_range[:, None] * stride_left_i
            + k_range[None, :] * stride_left_k
            + off_h_left
        )
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # right tile (transposed):  B_T[k,j] = right[b, j, k, h]
        b_ptrs = (
            right_ptr
            + b * stride_right_b
            + j_range[None, :] * stride_right_i
            + k_range[:, None] * stride_right_k
            + off_h_right
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # write the result
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i_range[:, None] * stride_out_i
        + j_range[None, :] * stride_out_j
        + off_h_out
    )
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched N‑wise reduction using the Triton kernel above.
    left, right: (B, N, N, H)  fp16, contiguous.
    Returns out of the same shape and dtype.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides for each tensor
    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

    # tile sizes – chosen to give good occupancy on A100/B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_l[0],
        s_l[1],
        s_l[2],
        s_l[3],
        s_r[0],
        s_r[1],
        s_r[2],
        s_r[3],
        s_o[0],
        s_o[1],
        s_o[2],
        s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]

    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.
    Arguments
    ---------
    data : (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, C]
        mask         : torch.Tensor [B, N, N] or None
        weights      : dict with model parameters
        config       : dict with 'dim' and 'hidden_dim' (and optional 'nomask')
    Returns
    -------
    torch.Tensor [B, N, N, C]  (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast model weights to fp16 (except the first LayerNorm already used)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)      # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Fuse mask with projection & gate
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # broadcast mask over the hidden dimension
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguous layout for Triton
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 7) Core TriMul reduction via Triton
    # ------------------------------------------------------------------
    out = _tri_mul_triton(left, right)   # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)

    return out
