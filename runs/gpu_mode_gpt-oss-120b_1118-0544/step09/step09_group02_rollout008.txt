"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections (left/right) and gating projections (left_gate,
   right_gate, out_gate).  All linear layers are bias‑free.
4. Fuse projection, gating and optional mask with a tiny Triton kernel.
5. Core reduction (tri‑multiply):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is implemented as a batched GEMM on the (B·H) dimension.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and apply the final linear projection back to `dim`.
7. Cast the result back to the original dtype.

At least one operation (projection + gate + mask) is performed in a
Triton kernel, satisfying the requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask:
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # guard
    mask = offs < total_elems

    # each mask element corresponds to one (b,i,k) entry, i.e. every hidden_dim
    # values share the same mask value
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch the Triton kernel that fuses projection, gating and mask.
    All tensors must be fp16, contiguous and on the same device.
    Returns (left, right) each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ----------------------------------------------------------------------
    # 1) Input LayerNorm over channel dim
    # ----------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(orig_dtype).to(device)
    norm_b = weights["norm.bias"].to(orig_dtype).to(device)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # (B, N, N, dim)

    # ----------------------------------------------------------------------
    # 2) Cast to FP16 for the heavy part
    # ----------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ----------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (no bias)
    # ----------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16).to(device)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16).to(device)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16).to(device)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16).to(device)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16).to(device)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16).to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16).to(device)
    to_out_w      = weights["to_out.weight"].to(torch.float16).to(device)

    # ----------------------------------------------------------------------
    # 4) Projections (FP16)
    # ----------------------------------------------------------------------
    left_proj  = F.linear(x_fp16, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x_fp16, right_proj_w)

    # ----------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ----------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    # ----------------------------------------------------------------------
    # 6) Apply mask & fuse with projection & gating
    # ----------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask is expected to be broadcastable to (B, N, N)
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask (or ignored)
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ----------------------------------------------------------------------
    # 7) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ----------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ----------------------------------------------------------------------
    # 9) Cast back to original dtype
    # ----------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
