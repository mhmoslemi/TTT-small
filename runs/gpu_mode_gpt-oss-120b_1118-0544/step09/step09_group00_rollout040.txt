def batched_tri_mul(left, right):
    B, N, _, H = left.shape
    # reshape to (B*H,N,N)
    left_f = left.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    right_f = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_f.view(B*H, N, N)
    right_flat = right_f.view(B*H, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # get strides
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)
    stride_b_batch = right_flat.stride(0)
    stride_b_m = right_flat.stride(1)
    stride_b_k = right_flat.stride(2)
    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_k = out_flat.stride(2)

    # launch kernel
    grid = (tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N), B*H)
    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_m=stride_b_m,
        stride_b_k=stride_b_k,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_k=stride_c_k,
    )
    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

"""
TriMul (outgoing) forward pass implemented with a Triton kernel for the
core reduction.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for compute‑heavy parts.
3. Linear projections (left/right) and gating projections (left_gate,
   right_gate, out_gate) – bias‑free.
4. Apply mask (if present) and fuse with gating (element‑wise).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication per hidden slice
   (A @ Bᵀ).  Implemented in Triton with tiling over M, N and K.
6. Output layer‑norm, multiply by `out_gate`, final linear projection
   back to `dim`.
7. Cast back to the original dtype.

The Triton kernel processes each (batch * hidden) matrix independently
using three program‑ids: (block‑row, block‑col, batch).  Accumulation is
performed in FP32 for numerical stability and the result is stored as
FP16.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_matmul_kernel(
    A_ptr,                     # fp16 *
    B_ptr,                     # fp16 *
    C_ptr,                     # fp16 *
    N,                         # i32 scalar (sequence length)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_m, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ Bᵀ
    A, B, C have shape (batch, N, N).  The operation performed is:
        C[i, j] = Σ_k A[i, k] * B[j, k]
    """
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_b = tl.program_id(2)  # batch index (batch * hidden)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # A: (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + k_offset[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # B: (BLOCK_K, BLOCK_N) – we load with k as the first dimension
        b_ptrs = (
            B_ptr
            + pid_b * stride_b_batch
            + k_offset[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_m
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    c_ptrs = (
        C_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_k
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the batched reduction `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
    using the Triton kernel above.
    left/right: tensors of shape (B, N, N, H) in FP16, contiguous.
    Returns tensor of the same shape.
    """
    B, N, _, H = left.shape

    # reshape to (B*H, N, N) for the kernel
    left_f = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_f = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_f.view(B * H, N, N)
    right_flat = right_f.view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)

    # strides for contiguous (batch, M, K)
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_m, stride_b_k = right_flat.stride()
    stride_c_batch, stride_c_m, stride_c_k = out_flat.stride()

    # tile sizes – tuned for typical GPU shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_m,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights: dict with keys:
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Apply mask and fuse with projection & gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp
        right = right_proj * right_gate * mask_fp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction using Triton (batched matmul with transpose)
    # ------------------------------------------------------------------
    out = _batched_tri_mul(left, right)  # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
