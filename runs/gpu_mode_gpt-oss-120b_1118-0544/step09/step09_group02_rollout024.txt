@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid = tl.program_id(0)
    # Compute number of blocks
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    # flatten dims
    total_blocks = B * H * num_pid_m * num_pid_n
    # compute indices
    bh_idx = pid // (num_pid_m * num_pid_n)
    rem = pid % (num_pid_m * num_pid_n)
    pid_m = rem // num_pid_n
    pid_n = rem % num_pid_n

    # compute b and h
    b_idx = bh_idx // H
    h_idx = bh_idx % H

    # compute offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # compute base pointers
    # left: (b,i,k,h)
    left_base = left_ptr + b_idx * stride_l_b + h_idx * stride_l_h
    # right: (b,j,k,h) -> we need transposed (k,j)
    right_base = right_ptr + b_idx * stride_r_b + h_idx * stride_r_h

    # offsets for left and right
    # left: (i,k)
    left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
    # right transposed: (k,j)
    right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        # load
        a = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        # dot
        acc += tl.dot(a, b)
        # increment pointers
        left_ptrs += BLOCK_K * stride_l_k
        right_ptrs += BLOCK_K * stride_r_k

    # store result (cast to fp16)
    out_base = out_ptr + b_idx * stride_o_b + h_idx * stride_o_h
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass with a fused Triton kernel for the
core N×N×N reduction.

Algorithm
---------
1. Layer‑norm over the input channel dimension `dim`.
2. Cast to fp16 for compute‑heavy parts.
3. Linear projections (left/right) and gating projections
   (left_gate, right_gate, out_gate) – all bias‑free.
4. Optional mask fusion (performed in PyTorch).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication per head and is
   implemented by a custom Triton kernel (FP16 compute, FP32
   accumulation).  The kernel processes (B·H) independent
   (N×N) matrix‑multiplications.
6. Layer‑norm over the hidden dimension, gating with `out_gate`,
   and final linear projection back to `dim`.
7. Cast back to the original dtype.

The kernel handles arbitrary batch size (≤2), sequence length
(≤1024) and hidden dimension (≤384).  All tensors are assumed to
be on the CUDA device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    out_ptr,           # fp16 *
    B, N, H,           # i32 scalars
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication for the TriMul core:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    The kernel processes (B*H) independent (N×N) multiplications.
    """

    pid = tl.program_id(0)

    # number of blocks along M and N dimensions
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    # decode program id into (batch*head, block_m, block_n)
    bh_idx = pid // (num_pid_m * num_pid_n)
    rem   = pid % (num_pid_m * num_pid_n)
    pid_m = rem // num_pid_n
    pid_n = rem % num_pid_n

    # decode batch and head indices
    b_idx = bh_idx // H
    h_idx = bh_idx % H

    # offsets within the current block
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds accesses
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # -------------------------------------------------------------
    # Pointers to the left (A) and right (B) matrices.
    # left  : (b, i, k, h)   -> we need (i,k)
    # right : (b, j, k, h)   -> we need (k,j) i.e. transposed
    # -------------------------------------------------------------
    left_base  = left_ptr  + b_idx * stride_l_b + h_idx * stride_l_h
    right_base = right_ptr + b_idx * stride_r_b + h_idx * stride_r_h

    # (BLOCK_M, BLOCK_K) from left
    a_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
    # (BLOCK_K, BLOCK_N) from right (transposed)
    b_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)
        acc += tl.dot(a, b)               # fp16·fp16 → fp32 accumulation
        a_ptrs += BLOCK_K * stride_l_k     # advance K‑dimension
        b_ptrs += BLOCK_K * stride_r_k

    # -------------------------------------------------------------
    # Store the result (cast back to fp16)
    # -------------------------------------------------------------
    out_base = out_ptr + b_idx * stride_o_b + h_idx * stride_o_h
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm over channel dimension `dim`
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # -----------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive parts
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3) Linear projections (fp16)
    # -----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    # -----------------------------------------------------------------
    # 4) Apply mask (if present) and fuse with projection & gating
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape [B, N, N] -> broadcast to [B, N, N, 1]
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5) Core reduction (TriMul) via Triton kernel
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    assert N == left.shape[2] and H == hidden_dim

    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)      # (B, N, N, H), fp16

    # strides for the three tensors (b, i, k, h) order
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # block sizes – tuned for typical N up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * H * num_pid_m * num_pid_n
    grid = (total_programs,)

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 7) Cast back to original dtype (usually float32)
    # -----------------------------------------------------------------
    out = out.to(input_tensor.dtype)

    return out
