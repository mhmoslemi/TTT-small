def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # dims
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # layer norm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # cast to fp16
    x = x.to(torch.float16)

    # cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Projections
    left_proj = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # Gate projections
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(left_proj, right_proj, left_gate, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Core reduction
    # left,right shape B,N,N,hidden_dim
    out = _tri_mul_kernel(left, right)  # returns B,N,N,hidden_dim tensor fp16

    # Output layer norm
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)  # shape B,N,N,dim

    # cast back to original dtype
    out = out.to(input_tensor.dtype)
    return out

@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k, stride_h,
    stride_b_out, stride_i_out, stride_j_out, stride_h_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    batch_idx = pid_batch // H
    head_idx = pid_batch % H

    # Compute base offsets
    # batch offset
    batch_offset = batch_idx * stride_b
    # head offset
    head_offset = head_idx * stride_h

    # Offsets for i and j within tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # Load left tile (M,K)
        left_ptrs = left_ptr + batch_offset + head_offset + i[:, None] * stride_i + k_idx[None, :] * stride_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile (K,N) (note transposed)
        right_ptrs = right_ptr + batch_offset + head_offset + k_idx[:, None] * stride_k + j[None, :] * stride_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_ptr + batch_offset + head_offset + i[:, None] * stride_i_out + j[None, :] * stride_j_out
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
custom_kernel: Optimized outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections (left/right) and gating projections (left_gate,
   right_gate, out_gate) – all fused with the optional mask via a
   small Triton kernel.
4. Core reduction (TriMul):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (per head) with a custom Triton
   kernel that tiles the i/j dimensions (BLOCK_M×BLOCK_N) and reduces
   over k (BLOCK_K). Accumulation is performed in FP32.
5. Output LayerNorm, element‑wise out‑gate, final linear projection back
   to `dim`.
6. Cast back to the original dtype (usually FP32).

Only the projection‑gate‑mask fusion and the TriMul reduction are
implemented in Triton; all other ops use PyTorch (which is already
highly optimized).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# 1) Fuse projection, gating and optional mask
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask:
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    All tensors are flattened in row‑major order.
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    # each element repeats `hidden_dim` times across the hidden dimension
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # 1.0 = no masking

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr  + offs, left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch the Triton kernel that fuses projection, gating and mask.
    All inputs must be contiguous FP16 tensors.
    Returns (left, right) each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# 2) Core TriMul reduction (batched GEMM) in Triton
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,    # fp16 *
    right_ptr,   # fp16 *
    out_ptr,     # fp16 *
    B, N, H,                     # int64 scalars
    stride_b, stride_i, stride_k, stride_h,               # int64 strides for left/right
    stride_b_out, stride_i_out, stride_j_out, stride_h_out,  # int64 strides for out
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    The tensors have layout (B, N, N, H) in row‑major order.
    """
    pid_batch = tl.program_id(0)   # encodes (batch, head)
    pid_i     = tl.program_id(1)   # tile row index
    pid_j     = tl.program_id(2)   # tile column index

    # decode batch and head
    batch_idx = pid_batch // H
    head_idx  = pid_batch % H

    # base offsets for this (batch, head)
    batch_offset = batch_idx * stride_b
    head_offset  = head_idx * stride_h

    # tile offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # left tile: (M, K)
        left_ptrs = (left_ptr + batch_offset + head_offset
                     + i[:, None] * stride_i
                     + k_idx[None, :] * stride_k)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right tile: (K, N) – note that we load transposed so we can use tl.dot
        right_ptrs = (right_ptr + batch_offset + head_offset
                      + k_idx[:, None] * stride_k
                      + j[None, :] * stride_i)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)   # (M, N)

    # write result
    out_ptrs = (out_ptr + batch_offset + head_offset
                + i[:, None] * stride_i_out
                + j[None, :] * stride_j_out)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def _apply_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched TriMul reduction using the Triton kernel above.
    left, right: (B, N, N, H) FP16, contiguous.
    Returns out of same shape and dtype.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides for left/right (identical)
    stride_b, stride_i, stride_k, stride_h = left.stride()
    # strides for out
    stride_b_out, stride_i_out, stride_j_out, stride_h_out = out.stride()

    # tile sizes – chosen to fit in shared memory and registers
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_b, stride_i, stride_k, stride_h,
        stride_b_out, stride_i_out, stride_j_out, stride_h_out,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# 3) Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights: dict with the following keys (all float32):
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Fuse mask, projection and gating (Triton if needed)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is FP16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # simple element‑wise fusion when no mask (or ignored)
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul)
    # ------------------------------------------------------------------
    out = _apply_tri_mul(left, right)   # (B, N, N, hidden_dim), FP16

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype (typically float32)
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)

    return out
