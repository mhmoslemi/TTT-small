@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Get program ids
    pid_i = tl.program_id(0) # block row
    pid_j = tl.program_id(1) # block col
    pid_bh = tl.program_id(2) # batch*head

    # Determine batch and head indices
    b = pid_bh // H
    h = pid_bh % H

    # Compute start offsets
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Compute offsets for A (left) tile: shape (BLOCK_M, BLOCK_K)
        # Offsets: i_offset + range(M), k_offset + range(K)
        k_offset = k * BLOCK_K

        # Compute pointer for left tile
        # left[b, i, k, h] => ptr = left_ptr + b*stride_l_b + (i_offset+mi)*stride_l_i + (k_offset+mk)*stride_l_k + h*stride_l_h
        a_ptrs = left_ptr + b*stride_l_b + (i_offset + tl.arange(0, BLOCK_M)[:, None]) * stride_l_i + (k_offset + tl.arange(0, BLOCK_K)[None, :]) * stride_l_k + h*stride_l_h

        # Compute pointer for right tile
        # right[b, j, k, h] but we need transposed: we need shape (BLOCK_K, BLOCK_N) where first dim is k, second is j.
        # So pointer should be: right_ptr + b*stride_r_b + (j_offset + tl.arange(0, BLOCK_N)[None, :]) * stride_r_j + (k_offset + tl.arange(0, BLOCK_K)[:, None]) * stride_r_k + h*stride_r_h
        # However to get shape (BLOCK_K, BLOCK_N), we need k as leading dimension. So we will create pointer where first index is k_offset + mk (size BLOCK_K) and second is j_offset + nj (size BLOCK_N). We'll use broadcasting accordingly.
        b_ptrs = right_ptr + b*stride_r_b + (j_offset + tl.arange(0, BLOCK_N)[None, :]) * stride_r_j + (k_offset + tl.arange(0, BLOCK_K)[:, None]) * stride_r_k + h*stride_r_h

        # Load tiles
        a = tl.load(a_ptrs, mask=(i_offset + tl.arange(0, BLOCK_M)[:, None] < N) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < N), other=0.0)
        b = tl.load(b_ptrs, mask=(j_offset + tl.arange(0, BLOCK_N)[None, :] < N) & (k_offset + tl.arange(0, BLOCK_K)[:, None] < N), other=0.0)

        # Multiply-add: a (M,K) * b (K,N) -> (M,N)
        acc += tl.dot(a, b)

    # Write output tile
    # out[b, i, j, h]
    out_ptrs = out_ptr + b*stride_o_b + (i_offset + tl.arange(0, BLOCK_M)[:, None]) * stride_o_i + (j_offset + tl.arange(0, BLOCK_N)[None, :]) * stride_o_j + h*stride_o_h

    # store, with mask
    tl.store(out_ptrs, acc.to(tl.float16), mask=(i_offset + tl.arange(0, BLOCK_M)[:, None] < N) & (j_offset + tl.arange(0, BLOCK_N)[None, :] < N))

    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,

def _tri_mul(left: torch.Tensor, right: torch.Tensor, block_m=64, block_n=64, block_k=64):
    B, N, _, H = left.shape
    out = torch.empty_like(left)
    # strides
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()
    grid = (triton.cdiv(N, block_m), triton.cdiv(N, block_n), B * H)
    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return out

"""
TriMul outgoing (AlphaFold3) forward pass – Triton fused implementation.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for compute‑heavy parts.
3. Linear projections (left/right) and gating projections (left_gate,
   right_gate, out_gate) – all bias‑free.
4. If a mask is provided (and not ignored) apply it element‑wise after
   gating.
5. Core reduction (TriMul):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a blocked batched matrix multiplication in Triton
   (one kernel, no Python loops).
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and apply the final linear projection back to `dim`.
7. Cast the result back to the original dtype.

The reduction kernel works on tiles of size (BLOCK_M, BLOCK_N) and
iterates over the `k` dimension in blocks of size BLOCK_K.
All arithmetic is performed in fp32, inputs/outputs are fp16.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse left/right projection, gating and optional mask
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask:
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    All tensors are flattened in row‑major order.
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # each element repeats `hidden_dim` times across the hidden dimension
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=valid, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=valid, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # 1.0 = no masking

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch the Triton kernel that fuses projection, gating and mask.
    All inputs must be contiguous fp16 tensors.
    Returns (left, right) each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Triton kernel: TriMul core reduction (batched GEMM across hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,   # fp16 *
    right_ptr,  # fp16 *
    out_ptr,    # fp16 *
    B, N, H,    # i64 scalars
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # block row (i)
    pid_j = tl.program_id(1)   # block col (j)
    pid_bh = tl.program_id(2)  # batch * hidden

    # decode batch and head indices
    b = pid_bh // H
    h = pid_bh % H

    # start offsets for this tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_iter * BLOCK_K

        # offets for left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            left_ptr
            + b * stride_lb
            + (i_start + tl.arange(0, BLOCK_M)[:, None]) * stride_li
            + (k_start + tl.arange(0, BLOCK_K)[None, :]) * stride_lk
            + h * stride_lh
        )
        a_mask = (i_start + tl.arange(0, BLOCK_M)[:, None] < N) & (k_start + tl.arange(0, BLOCK_K)[None, :] < N)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # offsets for right tile: we need shape (BLOCK_K, BLOCK_N)
        # right layout is (b, j, k, h) → we treat k as leading dim
        b_ptrs = (
            right_ptr
            + b * stride_rb
            + (j_start + tl.arange(0, BLOCK_N)[None, :]) * stride_rj
            + (k_start + tl.arange(0, BLOCK_K)[:, None]) * stride_rk
            + h * stride_rh
        )
        b_mask = (j_start + tl.arange(0, BLOCK_N)[None, :] < N) & (k_start + tl.arange(0, BLOCK_K)[:, None] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)

    # store result tile
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + (i_start + tl.arange(0, BLOCK_M)[:, None]) * stride_oi
        + (j_start + tl.arange(0, BLOCK_N)[None, :]) * stride_oj
        + h * stride_oh
    )
    out_mask = (i_start + tl.arange(0, BLOCK_M)[:, None] < N) & (j_start + tl.arange(0, BLOCK_N)[None, :] < N)
    tl.store(out_ptrs, acc.to(tl.float16), mask=out_mask)


def _tri_mul(left: torch.Tensor, right: torch.Tensor,
             block_m: int = 64, block_n: int = 64, block_k: int = 64) -> torch.Tensor:
    """Batched TriMul reduction implemented in Triton."""
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides (as int64)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    grid = (
        triton.cdiv(N, block_m),   # rows
        triton.cdiv(N, block_n),   # cols
        B * H,                     # batch * hidden
    )
    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights: dict with required weight tensors (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to fp16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Apply mask and fuse with projection & gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is contiguous fp16 and broadcastable
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Make tensors contiguous for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul) via Triton
    # ------------------------------------------------------------------
    out = _tri_mul(left, right, block_m=64, block_n=64, block_k=64)  # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
