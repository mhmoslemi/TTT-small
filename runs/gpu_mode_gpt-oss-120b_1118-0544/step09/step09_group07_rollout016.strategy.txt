def _normalize_layer_kernel(in_ptr, out_ptr,
                             grid_size, channel_size,
                             group_size, BLOCK_SIZE: tl.constexpr):
    # Each grid element processes "group_size" channel blocks
    # in out_ptr

    # Transpose for software reuse
    in_transposed = in_ptr + tl.arange(grid_size) * (grid_size) * tl.arange(channel_size)
    # Normalize along channel dimension
    mean = tl.sum(tl.load(in_transposed + pid * grid_size * channel_size)) / channel_size
    variance = tl.sum(tl.pow(tl.load(in_transposed + pid * grid_size * channel_size), 2)) / channel_size
    out = ((in_transposed + pid * grid_size * channel_size) - mean) / tl.sqrt(variance + 1e-8)
    tl.store(out_ptr + pid * grid_size * channel_size, out)
