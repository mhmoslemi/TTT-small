def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get('nomask', False)

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor,
                (dim,),
                weight=weights["norm.weight"],
                bias=weights["norm.bias"])

    # Cast to fp16
    x = x.to(torch.float16)

    # Linear weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)  # (B,N,N,hidden_dim)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply mask and gating
    if mask is not None and not nomask:
        # mask shape (B,N,N). Expand to (B,N,N,1)
        mask = mask.to(torch.float16).unsqueeze(-1)  # shape (B,N,N,1)
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguity
    left = left.contiguous()
    right = right.contiguous()

    # Core reduction with Triton
    B, N, _, H = left.shape

    out = torch.empty_like(left)  # shape (B,N,N,H)

    # Compute strides
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()
    # Unpack
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = stride_left
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = stride_right
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = stride_out

    # Kernel launch parameters
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    total_i = (N + BLOCK_M - 1) // BLOCK_M
    total_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * H * total_i * total_j,)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Output layernorm
    out = torch.nn.functional.layer_norm(out,
                (hidden_dim,),
                weight=to_out_norm_w,
                bias=to_out_norm_b)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to original dtype
    if out.dtype != dtype:
        out = out.to(dtype)

    return out

"""
TriMul outgoing forward pass (AlphaFold3) with Triton fusion.

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to fp16 for the compute‑heavy parts.
3. Linear projections (left, right) and gating projections
   (left_gate, right_gate, out_gate) – all bias‑free.
4. Apply optional mask and element‑wise gating (torch).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton kernel that performs a batched
   GEMM over the hidden dimension (B·H independent GEMMs of size N×N).
6. Output layer‑norm, multiply by out_gate and final linear projection.
7. Cast back to the original dtype.

At least the expensive reduction (step 5) is executed in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    out_ptr,           # fp16 *
    B, N, H,           # scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where left/right have shape (B, N, N, H) and out the same.
    """
    pid = tl.program_id(0)

    # Tiles in the i and j dimensions
    total_i = tl.cdiv(N, BLOCK_M)
    total_j = tl.cdiv(N, BLOCK_N)
    tiles_per_bh = total_i * total_j

    # Decode program id into (batch*hidden), i‑tile, j‑tile
    bh = pid // tiles_per_bh
    tile_id = pid % tiles_per_bh
    i_tile = tile_id // total_j
    j_tile = tile_id % total_j

    b = bh // H
    h = bh % H

    i_start = i_tile * BLOCK_M
    j_start = j_tile * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, h] -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = (left_ptr + b * stride_left_b
                     + offs_i[:, None] * stride_left_i
                     + offs_k[None, :] * stride_left_k
                     + h * stride_left_h)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right[b, j, k, h] -> shape (BLOCK_N, BLOCK_K)
        right_ptrs = (right_ptr + b * stride_right_b
                      + offs_j[:, None] * stride_right_j
                      + offs_k[None, :] * stride_right_k
                      + h * stride_right_h)
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)

        # Accumulate dot product over k
        acc += tl.dot(left, right)

    # Store result as fp16
    out = acc.to(tl.float16)
    out_ptrs = (out_ptr + b * stride_out_b
                + offs_i[:, None] * stride_out_i
                + offs_j[None, :] * stride_out_j
                + h * stride_out_h)
    tl.store(out_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) – may be all‑ones.
            - weights: dict containing model parameters (float32)
            - config: dict with keys:
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid, FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Apply optional mask and combine with gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B, N, N) -> (B, N, N, 1)
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguous layout for Triton
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul) – batched GEMM over the hidden dim
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # (B, N, N, H)

    # Strides (in elements) for pointer arithmetic
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    (s_left_b, s_left_i, s_left_k, s_left_h) = stride_left
    (s_right_b, s_right_j, s_right_k, s_right_h) = stride_right
    (s_out_b, s_out_i, s_out_j, s_out_h) = stride_out

    # Triton launch configuration
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    tiles_i = (N + BLOCK_M - 1) // BLOCK_M
    tiles_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * H * tiles_i * tiles_j,)

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_left_b,
        s_left_i,
        s_left_k,
        s_left_h,
        s_right_b,
        s_right_j,
        s_right_k,
        s_right_h,
        s_out_b,
        s_out_i,
        s_out_j,
        s_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
