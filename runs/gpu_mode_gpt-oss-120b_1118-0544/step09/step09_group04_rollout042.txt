    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast
    x = x.to(torch.float16)

    # Convert weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    B, N, _, _ = x.shape

    X_flat = x.view(-1, dim)  # (M, dim) where M = B*N*N

    left_cat_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*H, dim)
    left_cat = torch.nn.functional.linear(X_flat, left_cat_w)  # (M, 2H)
    left_proj, left_gate_raw = torch.split(left_cat, hidden_dim, dim=1)

   implemented as a batched GEMM (B·H, N, N) @ (B·H, N, N)ᵀ using a Triton kernel.
5. Output layer‑norm, element‑wise out‑gate, and final linear projection back
   to the original channel dimension.
6. Cast back to the input dtype.

All heavy linear work uses a single fused linear per side, and the N³ reduction
is performed by a custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bn, stride_bk,
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C[m,i,j] = Σ_k A[m,i,k] * B[m,j,k] for all batches m."""
    pid_i = tl.program_id(0)          # block row   (i)
    pid_j = tl.program_id(1)          # block col   (j)
    pid_b = tl.program_id(2)          # batch index (m)

    # Offsets for the current block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < K

    # Pointers to the start of the block
    a_ptrs = a_ptr + pid_b * stride_am + offs_i[:, None] * stride_an + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_bm + offs_j[:, None] * stride_bn + offs_k[None, :] * stride_bk

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store the result
    c_ptrs = c_ptr + pid_b * stride_cm + offs_i[:, None] * stride_cn + offs_j[None, :] * stride_ck
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """TriMul outgoing forward pass.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, C] (float32)
            - mask: [B, N, N] (float32) – may be None
            - weights: dict of model parameters (float32)
            - config: {'dim': C, 'hidden_dim': H, 'nomask': bool (optional)}

    Returns:
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                               # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    B, N, _, _ = x.shape
    M = B * N * N                     # total pair positions

    # ------------------------------------------------------------------
    # 4) Fused projection + gate for left and right sides
    # ------------------------------------------------------------------
    x_flat = x.view(M, dim)           # (M, dim)

    # left side: [proj | gate] -> (2*H, dim)
    left_cat_w = torch.cat([left_proj_w, left_gate_w], dim=0)
    left_cat = F.linear(x_flat, left_cat_w)               # (M, 2H)
    left_proj, left_gate_raw = torch.split(left_cat, hidden_dim, dim=1)
    left_gate = torch.sigmoid(left_gate_raw)

    # right side: [proj | gate] -> (2*H, dim)
    right_cat_w = torch.cat([right_proj_w, right_gate_w], dim=0)
    right_cat = F.linear(x_flat, right_cat_w)             # (M, 2H)
    right_proj, right_gate_raw = torch.split(right_cat, hidden_dim, dim=1)
    right_gate = torch.sigmoid(right_gate_raw)

    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).view(M, 1)    # (M,1)
        left  = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # reshape to (B, N, N, H)
    left  = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5) Core reduction (batched GEMM) using Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for batched matmul
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_flat = torch.empty_like(left_flat)

    # strides for the Triton kernel
    stride_am, stride_an, stride_ak = left_flat.stride()
    stride_bm, stride_bn, stride_bk = right_flat.stride()
    stride_cm, stride_cn, stride_ck = out_flat.stride()

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * hidden_dim,
        N,
        N,
        stride_am, stride_an, stride_ak,
        stride_bm, stride_bn, stride_bk,
        stride_cm, stride_cn, stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # out‑gate (sigmoid of a linear over the normalized input)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)
    out = out * out_gate

    # final projection back to `dim`
    out = F.linear(out, to_out_w)       # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 7) Cast back to the original dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
```
