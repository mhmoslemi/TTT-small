"""
TriMul outgoing kernel (AlphaFold3) – fast forward path.

Algorithm
---------
1. Layer‑norm the input (FP16).
2. Fuse each pair of projection+gate linear layers by concatenating the
   weight matrices (dim → 2·hidden_dim) and splitting the result.
   Apply sigmoid to the gate part.
3. Multiply projection and gate; optionally apply a binary mask using a
   small Triton kernel (elementwise * mask broadcast over the hidden dim).
4. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]  (i.e. batched GEMM
   left @ rightᵀ for each hidden channel).  Implemented with torch.bmm.
5. Second layer‑norm, output‑gate (linear+sigmoid), elementwise gating,
   final linear projection back to the original dimension.
6. Cast back to the input dtype.

The only Triton‑implemented part is the mask‑fusion kernel, which avoids
extra Python loops and keeps the main compute in cuBLAS‑accelerated
operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _mask_kernel(
    tensor_ptr,   # fp16 * (B, N, N, H)
    mask_ptr,     # fp16 * (B, N, N)
    out_ptr,      # fp16 * (B, N, N, H)
    total_elems,  # i64 = B*N*N*H
    hidden_dim,   # i64 = H
    BLOCK_SIZE: tl.constexpr,
):
    """Apply a binary mask (broadcast over the hidden dimension)."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    in_range = offs < total_elems

    # Index into mask (mask has no hidden dim)
    mask_idx = offs // hidden_dim

    x = tl.load(tensor_ptr + offs, mask=in_range, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=in_range, other=1.0)

    tl.store(out_ptr + offs, x * m, mask=in_range)


def _apply_mask(tensor: torch.Tensor, mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """Fuses a mask into `tensor` via a Triton kernel."""
    total = tensor.numel()
    out = torch.empty_like(tensor)
    grid = (triton.cdiv(total, block_size),)
    _mask_kernel[grid](
        tensor,
        mask,
        out,
        total,
        tensor.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

    norm_b = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2) Fuse projection + gate for left and right sides
    # ------------------------------------------------------------------
    # Concatenate weights (out_features, in_features)
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)

    # Linear (no bias)
    left_comb = F.linear(x, left_comb_w)  # (B, N, N, 2*H)
    right_comb = F.linear(x, right_comb_w)

    # Split projection / gate and apply sigmoid to gates
    left_proj, left_gate = left_comb[..., :hidden_dim], left_comb[..., hidden_dim:]
    right_proj, right_gate = right_comb[..., :hidden_dim], right_comb[..., hidden_dim:]

    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()

    # ------------------------------------------------------------------
    # 3) Apply mask and combine projection with gate
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    if mask is not None and not nomask:
        # Ensure mask is half‑precision and contiguous
        mask_h = mask.to(torch.float16).contiguous()
        left = _apply_mask(left, mask_h, block_size=1024)
        right = _apply_mask(right, mask_h, block_size=1024)

    # ------------------------------------------------------------------
    # 4) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM (left @ rightᵀ) over hidden channels.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5) Output LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B,N,N,H)
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, C)

    # ------------------------------------------------------------------
    # 6) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)
    return out
```
