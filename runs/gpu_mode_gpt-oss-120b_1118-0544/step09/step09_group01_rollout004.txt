"""
TriMul forward (outgoing) – Triton‑accelerated elementwise projection/gating.

Algorithm
---------
1️⃣ Layer‑norm the input over the channel dimension.
2️⃣ Cast to FP16 and compute linear projections (left/right) and gates.
3️⃣ Fuse projection·gate·mask (when a mask is present) with a tiny Triton kernel.
4️⃣ Core reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (torch.bmm) after reshaping to
   (B*H, N, N) → (B*H, N, N)ᵀ.
5️⃣ Output LayerNorm → out‑gate → final linear projection back to dim.
6️⃣ Return FP32 tensor of shape [B, N, N, dim].

Only the element‑wise fusion (proj × gate × mask) is performed in Triton;
the large matrix multiplication leverages Torch’s highly‑optimized batched GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B, N, N, H)
    right_proj_ptr,  # fp16 * (B, N, N, H)
    left_gate_ptr,   # fp16 * (B, N, N, H)
    right_gate_ptr,  # fp16 * (B, N, N, H)
    mask_ptr,        # fp16 * (B, N, N)
    left_out_ptr,    # fp16 * (B, N, N, H)
    right_out_ptr,   # fp16 * (B, N, N, H)
    total_elems,     # i64 = B * N * N * H
    hidden_dim,      # i64 = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse (proj * gate) and optional mask for left/right branches."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    within = offs < total_elems

    # each element belongs to a (b,i,j) triple; compute its mask index
    mask_idx = offs // hidden_dim

    # Load
    lp = tl.load(left_proj_ptr + offs, mask=within, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=within, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=within, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=within, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=within, other=1.0)  # 1.0 when no mask

    # Fuse
    left_out  = lp * lg * m
    right_out = rp * rg * m

    # Store
    tl.store(left_out_ptr + offs, left_out, mask=within)
    tl.store(right_out_ptr + offs, right_out, mask=within)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Triton‑based element‑wise fusion of projection, gate and mask.
    All tensors must be FP16, contiguous and have shapes:
        (B, N, N, H) for the four 4‑D tensors and (B, N, N) for mask.
    Returns left, right tensors of shape (B, N, N, H).
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (fp32 or fp16)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (fp32)
        - config       : dict with keys ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool) to ignore the mask.

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy lifting
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast linear weights to FP16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections and gates in FP16
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate = F.linear(x, left_gate_w).sigmoid()   # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid() # (B, N, N, H)
    out_gate = F.linear(x, out_gate_w).sigmoid()     # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse projection·gate·mask (Triton) or do it element‑wise
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Convert mask to FP16 (1.0 for keep, 0.0 for mask out)
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16, block_size=1024
        )
    else:
        # No mask (or mask ignored) – simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask present but explicitly ignored → still broadcast
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 6️⃣ Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented via batched GEMM.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, N, N, H) → (B*H, N, N)   where dim0 = i, dim1 = k
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # (B, N, N, H) → (B*H, N, N)   where dim0 = j, dim1 = k
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: (i,k) · (k,j)ᵀ → (i,j)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm → out‑gate → final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (required output dtype)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
