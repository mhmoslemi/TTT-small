import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# Triton kernel
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,    # fp16 *
    right_proj_ptr,   # fp16 *
    left_gate_ptr,    # fp16 *
    right_gate_ptr,   # fp16 *
    mask_ptr,         # fp16 *
    left_out_ptr,     # fp16 *
    right_out_ptr,    # fp16 *
    total_elems,      # i64
    hidden_dim,       # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # each hidden_dim contiguous elements correspond to one (b,i,j)
    mask_idx = offs // hidden_dim

    # load
    left_proj = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # mask broadcasted
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gating and mask for left/right tensors using Triton.
    All tensors must be fp16, contiguous and have shapes:
        (B, N, N, H) for proj/gate tensors and (B, N, N) for mask.
    Returns (left, right) tensors of shape (B, N, N, H).
    """
    assert left_proj.is_contiguous() and right_proj.is_contiguous()
    assert left_gate.is_contiguous() and right_gate.is_contiguous()
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Steps:
    1. Layer‑norm the input over the channel dimension.
    2. Cast to fp16 and compute linear projections and gates.
    3. Fuse projection, gate and optional mask with a Triton kernel.
    4. Perform the core reduction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
       via a batched GEMM on (B*H, N, N).
    5. Output LayerNorm, out‑gate multiplication and final linear projection.
    6. Cast back to float32.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # 1. Input LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B,N,N,dim) float32

    # 2. Cast to fp16 for compute‑heavy ops
    x = x.to(torch.float16)

    # 3. Cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # 4. Linear projections
    left_proj = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w) # (B,N,N,hidden_dim)

    # 5. Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B,N,N,hidden_dim)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))     # (B,N,N,hidden_dim)

    # 6. Apply mask + gating (fused via Triton if mask is present)
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            block_size=1024,
        )
    else:
        # No mask or mask ignored: simple elementwise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask present but ignored – broadcast as ones
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # 7. Core reduction using batched GEMM
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N).transpose(1,2) -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden_dim)

    # 8. Output LayerNorm, out_gate, linear projection
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B,N,N,dim) fp16

    # 9. Cast back to float32
    out = out.to(torch.float32)

    return out

"""
custom_kernel: Triton‑accelerated outgoing TriMul forward (AlphaFold3).

The forward pass consists of:
1️⃣ LayerNorm over the channel dimension.
2️⃣ Linear projections (left/right) and gates (left/right/out) in FP16.
3️⃣ Optional mask application.  The mask, projection and gate are fused
   in a single Triton kernel for the left and right branches.
4️⃣ Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (torch.bmm) after reshaping to
   (B*H, N, N).
5️⃣ Output LayerNorm, out‑gate multiplication and final linear projection
   back to the original channel dimension.
6️⃣ Cast result to float32.

All heavy element‑wise work (proj × gate × mask) is performed by the
Triton kernel `_proj_gate_mask_kernel`.  The remaining matrix‑multiplication
step uses PyTorch's highly‑optimized batched GEMM.

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, projection and gate for left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B, N, N, H)
    right_proj_ptr,  # fp16 * (B, N, N, H)
    left_gate_ptr,   # fp16 * (B, N, N, H)
    right_gate_ptr,  # fp16 * (B, N, N, H)
    mask_ptr,        # fp16 * (B, N, N)
    left_out_ptr,    # fp16 * (B, N, N, H)
    right_out_ptr,   # fp16 * (B, N, N, H)
    total_elems,     # i64 = B * N * N * H
    hidden_dim,      # i64 = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds
    mask = offs < total_elems

    # Each contiguous block of `hidden_dim` elements corresponds to one (b,i,j)
    mask_idx = offs // hidden_dim

    # Load tensors (masked)
    left_proj = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Fuse: (proj * gate) * mask
    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    # Store results
    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

    # 6) Apply mask + gating (fused via Triton when mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            block_size=1024,
        )
    else:
        # No mask or mask ignored: simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Reshape to (B*H, N, N) for batched matrix multiplication
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim) fp16

    # ------------------------------------------------------------------
    # 9) Cast back to float32 (required output dtype)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
```
