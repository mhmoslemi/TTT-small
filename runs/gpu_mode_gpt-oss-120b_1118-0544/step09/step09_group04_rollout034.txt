def _gate_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems,
    hidden_dim,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    # Compute mask index: each element corresponds to one hidden_dim slice, so mask_idx = offs // hidden_dim
    mask_idx = offs // hidden_dim

    # load values
    l = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)

    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # default 1.0

    right_out = r * rg * m

    tl.store(right_out_ptr + offs, right_out, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16
    x = x.to(torch.float16)

    # Load weights and cast to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Combine projection and gate weights
    combined_w = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    # Flatten x and apply combined linear
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)  # (B*N*N, dim)
    combined = torch.nn.functional.linear(x_flat, combined_w)  # (B*N*N, 5*hidden_dim)
    combined = combined.view(B, N, N, 5 * hidden_dim)

    # Split
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        combined, hidden_dim, dim=-1
    )
    # Apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Apply mask and gating using Triton kernel (if mask required)
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Core reduction
    B, N, _, H = left.shape
    # reshape for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm, gate, final linear
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to original dtype
    if out.dtype != dtype:
        out = out.to(dtype)

    return out

def _apply_gate_mask(left_proj, right_proj, left_gate, right_gate, mask, block_size=1024):

        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the input (float32) and cast to float16 for compute‑heavy ops.
2. Fuse the five linear projections (left_proj, right_proj, left_gate, right_gate,
   out_gate) into a single matrix multiplication with a concatenated weight matrix.
3. Split the combined result, apply sigmoid to the three gate tensors.
4. Fuse mask, gating and element‑wise multiplication for left/right tensors via a
   Triton kernel (skipped when mask is absent or `nomask=True`).
5. Compute the core reduction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM using torch.bmm on tensors of shape (B*H, N, N).
6. Apply a second LayerNorm, multiply by the out‑gate, and a final linear
   projection back to the original dimension.
7. Cast the result back to the input dtype.

At least one operation (mask + gate fusion) is executed in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _gate_mask_kernel(
    left_proj_ptr,   # half *
    right_proj_ptr,  # half *
    left_gate_ptr,   # half *
    right_gate_ptr,  # half *
    mask_ptr,        # half *
    left_out_ptr,    # half *
    right_out_ptr,   # half *
    total_elems,     # i64
    hidden_dim,      # i64
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask, gating and element‑wise multiplication."""
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds
    m = offs < total_elems

    # Index into the (B,N,N) mask (broadcasted over hidden_dim)
    mask_idx = offs // hidden_dim

    # Load values
    l = tl.load(left_proj_ptr + offs, mask=m, other=0.0)
    r = tl.load(right_proj_ptr + offs, mask=m, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=m, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=1.0)  # default 1.0 = no mask

    # Apply mask and gates
    left_out = l * lg * mask_val
    right_out = r * rg * mask_val

    # Store results
    tl.store(left_out_ptr + offs, left_out, mask=m)
    tl.store(right_out_ptr + offs, right_out, mask=m)


def _apply_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launches the Triton kernel that fuses mask, gating and multiplication.
    All tensors must be torch.float16 and contiguous.
    Returns (left, right) each of shape (B, N, N, hidden_dim).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32); can be all‑ones.
            - weights: dict containing model parameters (float32)
            - config: dict with keys:
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Fuse the five linear projections into a single matmul
    # ------------------------------------------------------------------
    # Combine weights: order = left_proj, right_proj, left_gate, right_gate, out_gate
    combined_w = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)
    combined = F.linear(x_flat, combined_w)      # (B*N*N, 5*hidden_dim)
    combined = combined.view(B, N, N, 5 * hidden_dim)

    # ------------------------------------------------------------------
    # 5) Split projections and gates, apply sigmoid to gates
    # ------------------------------------------------------------------
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        combined, hidden_dim, dim=-1
    )
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Ensure tensors are contiguous before the Triton kernel
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # ------------------------------------------------------------------
    # 6) Apply mask, gating and element‑wise multiplication via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul) – batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype (float32)
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
