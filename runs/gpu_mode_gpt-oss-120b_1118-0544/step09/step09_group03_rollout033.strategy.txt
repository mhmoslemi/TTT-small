        self.right_gate_w = weights["right_gate.weight"]
        self.out_gate_w = weights["out_gate.weight"]
        self.to_out_norm_w = weights["to_out_norm.weight"]
        self.to_out_norm_b = weights["to_out_norm.bias"]
        self.to_out_w = weights["to_out.weight"]
        
        # Normalize input tensor and weights
        self.input = input_tensor.to(torch.float32)  # FP32 for LayerNorm
        
    def forward(self, *args, **kwargs):
        # 1) LayerNorm on the channel dimension (dim=C)
        x = self.input
        x_norm = F.layer_norm(x, (self.dim,), weight=self.weights["norm.weight"], bias=self.weights["norm.bias"])
        
        # 2) Cast to float16
        x_fp16 = x_norm.to(torch.float16)
        
        # 3) Linear projections using tied weights (contiguous)
        left_proj = F.linear(x_fp16[:, :, :, 0:self.hidden_dim], self.left_proj_w[:, :, :, 0]).contiguous()
        right_proj = F.linear(x_fp16[:, :, :, 0:self.hidden_dim], self.right_proj_w[:, :, :, 0]).contiguous()
        
        # 4) Gating projections
        left_gate = F.linear(x_fp16[:, :, 0, 0], self.left_gate_w[:, :, :, 0])
        left_gate = left_gate.sigmoid()
        
        right_gate = F.linear(x_fp16[:, :, 0, 0], self.right_gate_w[:, :, :, 0])
        right_gate = right_gate.sigmoid()
        
        out_gate = F.linear(x_fp16[:, :, 0], self.out_gate_w[:, :, :, 0])
        out_gate = out_gate.sigmoid()
        
        # 5) Apply mask if needed
        if self.use_mask:
            left_applied = left_proj[:, :, :, None] * left_gate[:, :, :, None]
            left_applied = left_applied * self.mask[:, :, :, None]
            right_applied = right_proj[:, :, :, None] * right_gate[:, :, :, None]
            right_applied = right_applied * self.mask[:, :, :, None]
        else:
            left_applied = left_proj[:, :, :, None] * left_gate[:, :, :, None]
            right_applied = right_proj[:, :, :, None] * right_gate[:, :, :, None]
        
        # 6) Bulk operation: Compute TriMul matrix multiplication in a Triton kernel
        # Inputs: left_applied, right_applied (float16), output_dtype=torch.float16
        # Output: out

        # 7) Compute LayerNorm and linear projection for the output
        out_fp32 = F.layer_norm(out, (self.hidden_dim,), 
                                weight=self.to_out_norm_w, 
                                bias=self.to_out_norm_b)
        out_final = out_fp32 @ self.to_out_w
        
        # 8) Convert back to original dtype if needed
        if out_final.dtype != self.input.dtype:
            out_final = out_final.to(self.input.dtype)
            
        return out_final
