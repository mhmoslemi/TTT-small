"""
TriMul (outgoing) forward pass using Triton for fused projection/gate/mask.
...
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # half *
    right_proj_ptr,  # half *
    left_gate_ptr,   # half *
    right_gate_ptr,  # half *
    mask_ptr,        # half *
    left_out_ptr,    # half *
    right_out_ptr,   # half *
    total_elems,     # i64
    hidden_dim,      # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    # which mask element for this idx (same for each hidden dim)
    mask_idx = offs // hidden_dim

    left = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gating and mask using Triton.
    All tensors must be torch.float16 and contiguous.
    Returns left, right each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out

def custom_kernel(

    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape (B, N, N, dim), dtype same as input

    # ------------------------------------------------------------------
    # 2) Cast to float16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # 6) Apply mask & fuse with projection & gating

        mask_fp16 = mask.to(torch.float16).contiguous()

        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T  -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to input dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

"""
TriMul (outgoing) forward pass using Triton for fused projection, gating and mask.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension `dim` (float32).
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (left/right) and gating projections (left_gate,
   right_gate, out_gate) – all bias‑free.
4. If a mask is present and not ignored, fuse mask, projection and
   gating using a Triton kernel; otherwise perform the element‑wise
   fusion in PyTorch.
5. Core reduction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication per hidden slice.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate, and finally project back to the original dimension.
7. Cast the result back to the input dtype.

Only the mask‑gate‑projection fusion is executed in Triton; the rest
uses high‑performance PyTorch (batched GEMM via cuBLAS).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # half *
    right_proj_ptr,  # half *
    left_gate_ptr,   # half *
    right_gate_ptr,  # half *
    mask_ptr,        # half *
    left_out_ptr,    # half *
    right_out_ptr,   # half *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask:
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # each hidden channel shares the same mask element
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=valid, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=valid, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m     = tl.load(mask_ptr + mask_idx,   mask=valid, other=1.0)

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)

    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
