def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # extract dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # cast to fp16
    x = x.to(torch.float16)

    # Linear projections (fp16)
    left_proj = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x, weights["right_proj.weight"].to(torch.float16))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    # Apply mask - elementwise
    if mask is not None and not nomask:
        mask = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # rearrange for GEMM
    B, N, _, H = left.shape
    # permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # reshape to (B*H, N, N)
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # compute strides
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)
    stride_b_batch = right_flat.stride(0)
    stride_b_n = right_flat.stride(1)  # row dimension for B (actually it's same as M)
    stride_b_k = right_flat.stride(2)
    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    batched_gemm_kernel[grid](
        left_flat, right_flat, out_flat,
        B * H, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # reshape out_flat back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # To-out LayerNorm and final linear
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"].to(torch.float16), bias=weights["to_out_norm.bias"].to(torch.float16))
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast back to original dtype
    out = out.to(input_tensor.dtype)

    return out

"""
TriMul (outgoing) forward pass using Triton for the core batched matrix multiplication.
1. LayerNorm on input.
2. Linear projections and gating (fp16).
3. Optional mask applied elementwise.
4. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (left @ right.T) with a custom Triton kernel.
5. LayerNorm on hidden dimension, gating, final linear projection.
6. Cast back to original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ B.T for a batch of matrices.
    A, B, C are interpreted as (batch, M, K), (batch, N, K), (batch, M, N).
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start indices for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out-of-bounds rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile in A and B
    a_ptrs = (A_ptr + pid_batch * stride_a_batch
              + offs_m[:, None] * stride_a_m
              + tl.arange(0, BLOCK_K)[None, :] * stride_a_k)

    b_ptrs = (B_ptr + pid_batch * stride_b_batch
              + tl.arange(0, BLOCK_K)[:, None] * stride_b_k
              + offs_n[None, :] * stride_b_n)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute actual K offset for this iteration
        k_offset = k * BLOCK_K
        # Masks for the K dimension
        mask_k = (k_offset + tl.arange(0, BLOCK_K)) < K

        # Load A and B tiles, applying masks
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑block
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Write the result back
    c_ptrs = (C_ptr + pid_batch * stride_c_batch
              + offs_m[:, None] * stride_c_m
              + offs_n[None, :] * stride_c_n)

    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (fp16) and gating
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"].to(torch.float16))   # (B,N,N,hidden)
    right_proj = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    # ------------------------------------------------------------------
    # 4) Apply mask (if any) and fuse with projection & gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Core reduction – batched GEMM implemented in Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, hidden_dim)

    # Rearrange to (B, H, N, N) and flatten batch*head
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Block sizes – tuned for typical B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides (in elements, not bytes)
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_n = right_flat.stride(1)   # row stride for B (j dimension)
    stride_b_k = right_flat.stride(2)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Launch kernel
    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H, N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 7) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out

"""
TriMul (outgoing) forward pass using Triton for the core batched matrix multiplication.

Algorithm
---------
1. Layer‑norm over the channel dimension `dim`.
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (left/right) and gating (left_gate, right_gate, out_gate).
4. Apply optional mask (element‑wise) and fuse with projection & gating.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (left @ rightᵀ) with a custom Triton kernel.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.
7. Cast back to the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ Bᵀ for a batch of matrices.
    A : (batch, M, K)   (float16)
    B : (batch, N, K)   (float16)   – we treat it as (batch, K, N) inside the kernel
    C : (batch, M, N)   (float16)
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current tile in A and B
    a_ptrs = (A_ptr + pid_batch * stride_a_batch
              + offs_m[:, None] * stride_a_m
              + tl.arange(0, BLOCK_K)[None, :] * stride_a_k)

    b_ptrs = (B_ptr + pid_batch * stride_b_batch
              + tl.arange(0, BLOCK_K)[:, None] * stride_b_k
              + offs_n[None, :] * stride_b_n)

    # Accumulator in fp32 for higher precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        mask_k = (k_offset + tl.arange(0, BLOCK_K)) < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N)

        # Advance to next K‑block
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Store the result
    c_ptrs = (C_ptr + pid_batch * stride_c_batch
              + offs_m[:, None] * stride_c_m
              + offs_n[None, :] * stride_c_n)

    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)

            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with keys
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
              (all float32, on the correct device)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    """

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(input_tensor.dtype),

    B, N, _, H = left.shape               # (B, N, N, hidden_dim)

    out_flat = torch.empty_like(left_flat)

    BLOCK_K = 64

    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_n = right_flat.stride(1)      # row stride of B (j dimension)

    stride_c_n = out_flat.stride(2)

        stride_b_batch, stride_b_n, stride_b_k,

        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
