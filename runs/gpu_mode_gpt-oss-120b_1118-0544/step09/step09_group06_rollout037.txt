"""
Optimized outgoing TriMul forward pass using Triton.

The implementation:
1. LayerNorm on input.
2. FP16 cast for compute.
3. Linear projections (left/right) and gating (left_gate, right_gate, out_gate).
4. Optional mask application fused with projection and gating in a Triton kernel.
5. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] implemented as
   batched matrix multiplication (torch.bmm) over the hidden dimension.
6. Output LayerNorm, out‑gate multiplication and final linear projection.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel to fuse projection, gating and optional mask
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64
    hidden_dim,      # i64
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask:
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    off = pid * BLOCK_SIZE
    offs = off + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds
    mask = offs < total_elems

    # Index of the corresponding mask element (each mask entry is repeated `hidden_dim` times)
    mask_idx = offs // hidden_dim

    # Load values
    left  = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Fuse
    left_out  = left * lgate * m
    right_out = right * rgate * m

    # Store
    tl.store(left_out_ptr + offs,  left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch the Triton kernel that fuses projection, gating and (optional) mask.
    All tensors must be fp16, contiguous and on the same CUDA device.
    Returns (left_out, right_out) each with shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    # Compute grid size
    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],   # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing:
                * "dim" (int)          – input channel dimension C
                * "hidden_dim" (int)   – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for the compute‑intensive part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast model weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Apply mask (if any) and fuse with projection & gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16, contiguous and on the same device
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask (or mask ignored): simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for a batched GEMM
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: left_flat @ right_flatᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

    # Store
    tl.store(left_out_ptr + offs,  left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch the Triton kernel that fuses projection, gating and (optional) mask.
    All tensors must be fp16, contiguous and on the same CUDA device.
    Returns (left_out, right_out) each with shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],   # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing:
                * "dim" (int)          – input channel dimension C
                * "hidden_dim" (int)   – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for the compute‑intensive part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast model weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Apply mask (if any) and fuse with projection & gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask (or mask ignored): simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Reshape to (B*H, N, N) for batched GEMM
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: left_flat @ right_flatᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
