# Triton kernel for fusing projection, gating and mask multiplication
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64
    hidden_dim,      # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # compute mask index for broadcasting over hidden_dim
    mask_idx = offs // hidden_dim

    left_proj = tl.load(left_proj_ptr + offs, mask=mask)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:

    Fuse (proj * gate * mask) for both left and right tensors via a Triton kernel.
    All tensors must be contiguous fp16.

    return out

"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (left/right) and gating (left/right/out) are performed in fp16.
4. If a mask is provided (and not ignored) a Triton kernel fuses
   projection * gate * mask for both left and right tensors.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is implemented as a batched GEMM (B*H, N, N) @ (B*H, N, N)^T
   using `torch.bmm`.
6. Output LayerNorm, element‑wise multiplication by `out_gate`,
   and a final linear projection back to the original dimension.
7. Cast the result back to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse projection, gate and mask:
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask
    The mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # bounds
    valid = offs < total_elems

    # broadcast mask index (mask shape B,N,N)
    mask_idx = offs // hidden_dim

    left_proj = tl.load(left_proj_ptr + offs, mask=valid)
    right_proj = tl.load(right_proj_ptr + offs, mask=valid)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid)
    m = tl.load(mask_ptr + mask_idx, mask=valid)

    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)

    All tensors must be contiguous fp16.
    Shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)
    Returns:
        left, right : (B, N, N, H)
    """

    B, N, _, H = left_proj.shape

    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

        left_out,
        right_out,
        total,
        H,

    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
