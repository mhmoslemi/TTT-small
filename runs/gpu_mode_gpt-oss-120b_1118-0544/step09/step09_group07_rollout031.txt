"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for the compute‑heavy part.
3. Fuse the two projection linear layers (`left_proj`, `left_gate`) and
   (`right_proj`, `right_gate`) by concatenating their weight matrices
   and performing a single matmul for each side.
4. Apply sigmoid to the gate partitions.
5. If a mask is present (and `nomask` is False) a single Triton kernel
   fuses:   left = left_proj * left_gate * mask,
           right = right_proj * right_gate * mask
   (mask is broadcast over the hidden dimension).
6. Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (B*H, N, N) @ (B*H, N, N)ᵀ using torch.bmm.
7. Output layer‑norm, element‑wise multiply by `out_gate`,
   and final linear projection back to `dim`.
8. Cast the result back to the original input dtype.

The kernel in step 5 is required by the task; all other operations
use highly‑optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B*N*N*H)
    left_gate_ptr,   # fp16 * (B*N*N*H)
    right_proj_ptr,  # fp16 * (B*N*N*H)
    right_gate_ptr,  # fp16 * (B*N*N*H)
    mask_ptr,        # fp16 * (B*N*N)
    left_out_ptr,    # fp16 * (B*N*N*H)
    right_out_ptr,   # fp16 * (B*N*N*H)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse projection, gate and mask:
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask
    Mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # bounds check
    valid = offs < total_elems

    # mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=valid)
    lg = tl.load(left_gate_ptr + offs, mask=valid)
    rp = tl.load(right_proj_ptr + offs, mask=valid)
    rg = tl.load(right_gate_ptr + offs, mask=valid)
    m  = tl.load(mask_ptr + mask_idx, mask=valid)

    left_out  = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs,  left_out,  mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_proj: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Triton kernel that multiplies proj * gate * mask for both left and right tensors.
    All tensors must be contiguous fp16.
    Shapes: (B, N, N, H) for proj/gate, (B, N, N) for mask.
    Returns left, right tensors of shape (B, N, N, H).
    """
    total = left_proj.numel()
    B, N, _, H = left_proj.shape

    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
        - mask         : torch.Tensor [B, N, N] (bool/int/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys
            * "dim"        – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask" (optional) – if True, ignore mask

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Fuse projection + gate linear layers (FP16)
    # ------------------------------------------------------------------
    # Concatenate weights for left side: [proj; gate]
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)

    # Concatenate weights for right side: [proj; gate]
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)

    # Linear transformations (single matmul per side)
    left_concat = F.linear(x, left_concat_w)   # (B, N, N, 2*H)
    right_concat = F.linear(x, right_concat_w) # (B, N, N, 2*H)

    # Split into proj and gate parts
    left_proj = left_concat[..., :hidden_dim].contiguous()
    left_gate_pre = left_concat[..., hidden_dim:].contiguous()
    right_proj = right_concat[..., :hidden_dim].contiguous()
    right_gate_pre = right_concat[..., hidden_dim:].contiguous()

    # Sigmoid gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 4) Out gate (separate linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (fused in Triton when mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, left_gate, right_proj, right_gate, mask_fp16
        )
    else:
        # No mask or mask ignored – simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask exists but ignored – still apply as multiplication
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM on (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
