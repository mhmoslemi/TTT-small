    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_al, stride_ai, stride_ak,
    stride_br, stride_rk, stride_rn,
    stride_ol, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets for this block
    block_row = pid_m * BLOCK_M
    block_col = pid_n * BLOCK_N

    # Compute linear index for batch (b_h)
    # We treat B*H as batch dimension; so total_batches = B * H
    # But we don't need B or H separately in kernel; we just use total_batch.

    # Compute pointer base for this batch:
    left_ptr += pid_b * stride_al
    right_ptr += pid_b * stride_br
    out_ptr += pid_b * stride_ol

    offs_n = block_col + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_hidden, N,
    stride_al, stride_ai, stride_ak,
    stride_br, stride_rk, stride_rn,
    stride_ol, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    # map block index to row and col
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers for this batch
    left_batch_ptr = left_ptr + batch * stride_al
    right_batch_ptr = right_ptr + batch * stride_br
    out_batch_ptr = out_ptr + batch * stride_ol

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block
        left = tl.load(
            left_batch_ptr + (offs_m[:, None] * stride_ai) + (offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load right block (transposed)
        # right shape (N,N) -> we need right_T[K,N] where K is k and N is j.
        right = tl.load(
            right_batch_ptr + (offs_k[:, None] * stride_rk) + (offs_n[None, :] * stride_rn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # Compute dot product; left: (BLOCK_M, BLOCK_K), right: (BLOCK_K, BLOCK_N)
        acc += tl.dot(left, right)

    # Write output
    out = acc.to(tl.float16)
    tl.store(
        out_batch_ptr + (offs_m[:, None] * stride_oi) + (offs_n[None, :] * stride_oj),
        out,
        mask=mask_m[:, None] & mask_n[None, :]
    )

"""
Triton implementation of the outgoing Triangle Multiplicative Update (TriMul)
used in AlphaFold3. The forward pass consists of:

1. Layer‑norm over the channel dimension.
2. Linear projections (left/right) and gating projections (left_gate,
   right_gate, out_gate). All linear layers are bias‑free.
3. Optional mask application (broadcast over the hidden dimension).
4. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM (left @ rightᵀ) in a custom Triton
   kernel, accumulating in fp32 and storing in fp16.
5. Layer‑norm over the hidden dimension, multiplication by out_gate,
   and final linear projection back to the original channel size.
6. Cast back to the input dtype.

Only the compute‑heavy reduction (step 4) is written in Triton; the rest
uses highly‑optimized PyTorch ops.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def triton_trimul_kernel(
    left_ptr,          # fp16 *  (batch*H, N, N)
    right_ptr,         # fp16 *  (batch*H, N, N)
    out_ptr,           # fp16 *  (batch*H, N, N)
    N,                 # i64 scalar = sequence length
    stride_al, stride_ai, stride_ak,   # left strides (batch, row i, col k)
    stride_br, stride_rk, stride_rn,   # right strides (batch, k, col j) – note transposed view
    stride_ol, stride_oi, stride_oj,   # out strides (batch, row i, col j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start indices of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the current batch
    left_batch_ptr = left_ptr + batch_id * stride_al
    right_batch_ptr = right_ptr + batch_id * stride_br
    out_batch_ptr = out_ptr + batch_id * stride_ol

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block [M, K]
        left = tl.load(
            left_batch_ptr
            + (offs_m[:, None] * stride_ai)
            + (offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load right block transposed [K, N]
        # right is stored as (batch, j, k); we need right_T[k, j]
        right = tl.load(
            right_batch_ptr
            + (offs_k[:, None] * stride_rk)
            + (offs_n[None, :] * stride_rn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate dot product
        acc += tl.dot(left, right)

    # Write result (convert back to fp16)
    out = acc.to(tl.float16)
    tl.store(
        out_batch_ptr
        + (offs_m[:, None] * stride_oi)
        + (offs_n[None, :] * stride_oj),
        out,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with model parameters (float32)
            - config: dict with keys:
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Linear projections (fp16)
    # ------------------------------------------------------------------
    # Cast to fp16 for the compute‑heavy part
    x_fp16 = x.to(torch.float16)

    left_proj = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))   # (B,N,N,H)
    right_proj = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16)) # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3) Apply mask (if present) before gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape (B,N,N) -> (B,N,N,1) for broadcasting
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 4) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)))

    # ------------------------------------------------------------------
    # 5) Fuse projection with gating (element‑wise)
    # ------------------------------------------------------------------
    left = left_proj * left_gate    # (B,N,N,H)
    right = right_proj * right_gate # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction using Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Rearrange to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)  # fp16 output

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    triton_trimul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(2),  # stride over K (original column)
        right_flat.stride(1),  # stride over N (original row) – transposed view
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = (
        out_flat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(out.dtype),
        bias=weights["to_out_norm.bias"].to(out.dtype),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(out.dtype))  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
