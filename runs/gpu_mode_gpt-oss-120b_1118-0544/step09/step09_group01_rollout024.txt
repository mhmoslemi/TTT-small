"""
custom_kernel: Triton‑accelerated outgoing TriMul forward pass.

The forward consists of:
1️⃣ LayerNorm over the channel dimension (FP16).
2️⃣ Linear projections (left/right) and gate logits (left/right/out) (FP16).
3️⃣ Optional mask application fused with projection and gate using a
   Triton kernel (`_proj_gate_mask_kernel`).  This kernel computes
   `proj * sigmoid(gate) * mask` for the left and right branches.
4️⃣ Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM on reshaped tensors
   (B*H, N, N) × (B*H, N, N)ᵀ.
5️⃣ Output LayerNorm, out‑gate multiplication and final linear projection
   back to the original channel dimension.
All heavy element‑wise work is performed in FP16; the final result is cast
to float32 as required by the model.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    proj_left_ptr,   # fp16 * (B, N, N, H)
    proj_right_ptr,  # fp16 * (B, N, N, H)
    gate_left_ptr,   # fp16 * (B, N, N, H)
    gate_right_ptr,  # fp16 * (B, N, N, H)
    mask_ptr,        # fp16 * (B, N, N)
    out_left_ptr,    # fp16 * (B, N, N, H)
    out_right_ptr,   # fp16 * (B, N, N, H)
    total_elems,     # i64 = B * N * N * H
    hidden_dim,      # i64 = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, sigmoid‑gate and optional mask."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds
    mask = offs < total_elems

    # Index of the (b,i,j) triple for the mask (one mask value per (b,i,j))
    mask_idx = offs // hidden_dim

    # Load tensors
    proj_l = tl.load(proj_left_ptr + offs, mask=mask, other=0.0)
    proj_r = tl.load(proj_right_ptr + offs, mask=mask, other=0.0)
    gate_l = tl.load(gate_left_ptr + offs, mask=mask, other=0.0)
    gate_r = tl.load(gate_right_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Sigmoid on the gate (approximation via native tl.sigmoid)
    gate_l_sig = tl.sigmoid(gate_l)
    gate_r_sig = tl.sigmoid(gate_r)

    # Fuse: (proj * sigmoid(gate)) * mask
    out_l = proj_l * gate_l_sig * m
    out_r = proj_r * gate_r_sig * m

    # Store results
    tl.store(out_left_ptr + offs, out_l, mask=mask)
    tl.store(out_right_ptr + offs, out_r, mask=mask)


def _apply_proj_gate_mask(
    proj_left: torch.Tensor,
    proj_right: torch.Tensor,
    gate_left: torch.Tensor,
    gate_right: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gate (with sigmoid) and mask for both left and right
    branches using Triton.

    All tensors must be FP16, contiguous and have shapes:
        proj_left/right, gate_left/right : (B, N, N, H)
        mask                           : (B, N, N)
    Returns:
        left, right : (B, N, N, H) = proj * sigmoid(gate) * mask
    """
    B, N, _, H = proj_left.shape
    total = proj_left.numel()
    left_out = torch.empty_like(proj_left)
    right_out = torch.empty_like(proj_right)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        proj_left,
        proj_right,
        gate_left,
        gate_right,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32/float16)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys:
            * "dim"        – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask" (optional) – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over channel dimension (keep in FP16 for speed)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Cast weights to FP16 (bias‑free linear layers)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    # Shape: (B, N, N, hidden_dim)
    left_proj = F.linear(x, left_proj_w)          # no bias
    right_proj = F.linear(x, right_proj_w)

    # Gate logits (still FP16)
    left_gate = F.linear(x, left_gate_w)          # will sigmoid in kernel
    right_gate = F.linear(x, right_gate_w)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask + sigmoid‑gate (fused via Triton) if needed
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            block_size=1024,
        )
    else:
        # No mask → simple element‑wise fusion (sigmoid + multiply)
        left_gate_sig = torch.sigmoid(left_gate)
        right_gate_sig = torch.sigmoid(right_gate)
        left = left_proj * left_gate_sig
        right = right_proj * right_gate_sig
        if mask is not None and not nomask:
            # mask present but explicitly ignored → still broadcast
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 5️⃣ Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM on (B*H, N, N) tensors.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Rearrange to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # Batched matrix multiplication
    # (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to float32 (model expects FP32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
