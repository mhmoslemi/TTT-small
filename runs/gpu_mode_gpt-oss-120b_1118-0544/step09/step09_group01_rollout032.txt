import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # * fp16
    right_proj_ptr,  # * fp16
    left_gate_ptr,   # * fp16
    right_gate_ptr,  # * fp16
    mask_ptr,        # * fp16
    left_out_ptr,    # * fp16
    right_out_ptr,   # * fp16
    total_elems,     # i64
    hidden_dim,      # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # each group of hidden_dim consecutive elements belong to the same (b,i,j)
    mask_idx = offs // hidden_dim

    left_proj = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gating and optional mask for both left and right tensors
    using a Triton kernel.
    All tensors must be half-precision and contiguous.
    Returns:
        left_out, right_out : torch.Tensor of shape (B,N,N,hidden_dim)
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : Tensor[B, N, N, C] (float32 or float16)
        - mask         : Tensor[B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys:
            * "dim"        – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask" (optional) – if True, ignore mask

    Returns
    -------
    Tensor[B, N, N, C] (float32) – result of the TriMul module.
    """
    # Unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1) Input LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for most compute
    x = x.to(torch.float16)

    # 2) Cast weights to fp16 (bias‑free)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # 3) Linear projections and gates (half precision)
    left_proj = F.linear(x, left_proj_w)          # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # 4) Fuse mask, projection and gate
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16, block_size=1024
        )
    else:
        # simple elementwise fusion; mask is ignored or absent
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask present but ignored => treat as ones (no effect)
            pass

    # 5) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # 6) Output LayerNorm, gate, final linear
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)

    # Cast back to float32
    out = out.to(torch.float32)
    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm over the channel dimension of the input tensor.
2. Cast to fp16 and apply three bias‑free linear layers:
   * left_proj  :  dim → hidden_dim
   * right_proj :  dim → hidden_dim
   * left_gate / right_gate / out_gate : dim → hidden_dim  (followed by sigmoid)
3. Fuse projection, gating and optional mask with a Triton kernel
   (produces `left` and `right` tensors of shape B×N×N×H).
4. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM after reshaping to (B*H, N, N).
5. Output layer‑norm, element‑wise multiplication with `out_gate`,
   and a final linear projection back to the original channel dimension.
6. Cast result back to float32.

Only the mask‑projection‑gate fusion is performed in Triton; the
remaining heavy lifting uses torch's highly‑optimized GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16* (B, N, N, H)
    right_proj_ptr,  # fp16* (B, N, N, H)
    left_gate_ptr,   # fp16* (B, N, N, H)
    right_gate_ptr,  # fp16* (B, N, N, H)
    mask_ptr,        # fp16* (B, N, N)
    left_out_ptr,    # fp16* (B, N, N, H)
    right_out_ptr,   # fp16* (B, N, N, H)
    total_elems,     # i64 = B*N*N*H
    hidden_dim,      # i64 = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # guard against out‑of‑bounds
    mask = offs < total_elems

    # each group of `hidden_dim` consecutive elements corresponds to one (b,i,j)
    mask_idx = offs // hidden_dim

    left_proj   = tl.load(left_proj_ptr  + offs, mask=mask, other=0.0)
    right_proj  = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    left_gate   = tl.load(left_gate_ptr  + offs, mask=mask, other=0.0)
    right_gate  = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m           = tl.load(mask_ptr       + mask_idx, mask=mask, other=1.0)

    left_out  = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr  + offs, left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections and gates (half precision)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate = F.linear(x, left_gate_w).sigmoid()   # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid() # (B, N, N, H)
    out_gate = F.linear(x, out_gate_w).sigmoid()     # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse mask, projection and gate
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            block_size=1024,
        )
    else:
        # simple element‑wise fusion; mask is ignored or absent
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to float32
    # ------------------------------------------------------------------
    out = out.to(torch.float32)
    return out
```
