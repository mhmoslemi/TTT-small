"""
TriMul (outgoing) forward pass optimized for GPU.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (float32).
2. Cast to fp16 for compute‑heavy parts.
3. Fuse each pair of projection + gate linear layers:
       [proj, gate] = Linear(x, concat([proj_weight, gate_weight]))
   then split and apply sigmoid to the gate part.
4. If a mask is present (and not ignored) fuse mask, projection and gating
   with a small Triton kernel to avoid extra memory passes.
5. Core reduction (the triangle multiplicative update):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply per hidden slice:
       out_h = left_h @ right_hᵀ   for each h.
   Implemented via torch.bmm on fp16 tensors (uses cuBLAS).
6. Apply output LayerNorm, multiply by out‑gate and a final linear projection.
7. Cast back to the original dtype (float32).

Only the mask‑gate‑projection fusion is performed in Triton; the rest uses
high‑performance PyTorch operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask:
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    # each mask element corresponds to hidden_dim consecutive entries
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launches the Triton kernel that fuses projection, gating and mask.
    All tensor arguments must be fp16 and contiguous.
    Returns (left, right) each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],   # hidden_dim H
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension `dim` (float32)
    # ----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ----------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ----------------------------------------------------------------------
    x = x.to(torch.float16)

    # ----------------------------------------------------------------------
    # 3) Prepare fused projection+gate weights (FP16)
    # ----------------------------------------------------------------------
    # left side
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    left_fused_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)  # (2H, dim)

    # right side
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_fused_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)  # (2H, dim)

    # out gate
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # ----------------------------------------------------------------------
    # 4) Linear projections + gates (fused)
    # ----------------------------------------------------------------------
    # Shape after linear: (B, N, N, 2H)
    left_combined = F.linear(x, left_fused_w)   # (B, N, N, 2H)
    right_combined = F.linear(x, right_fused_w) # (B, N, N, 2H)

    left_proj = left_combined[..., :hidden_dim]                     # (B,N,N,H)
    left_gate = torch.sigmoid(left_combined[..., hidden_dim:])      # (B,N,N,H)

    right_proj = right_combined[..., :hidden_dim]                   # (B,N,N,H)
    right_gate = torch.sigmoid(right_combined[..., hidden_dim:])    # (B,N,N,H)

    out_gate = torch.sigmoid(F.linear(x, out_gate_w))               # (B,N,N,H)

    # ----------------------------------------------------------------------
    # 5) Apply mask and fuse with projection & gating
    # ----------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is contiguous fp16
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ----------------------------------------------------------------------
    # 6) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM per hidden slice.
    # ----------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ----------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim), still fp16

    # ----------------------------------------------------------------------
    # 8) Cast back to original dtype (float32)
    # ----------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
```
