def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B_H, N,
    stride_l_batch, stride_l_row, stride_l_col,
    stride_r_batch, stride_r_row, stride_r_col,
    stride_o_batch, stride_o_row, stride_o_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    total_batches = B_H
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_pid_m * num_pid_n)
    pid_rem = pid % (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n
    pid_n = pid_rem % num_pid_n

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers
    a_ptrs = left_ptr + batch * stride_l_batch + offs_m[:, None] * stride_l_row + offs_k[None, :] * stride_l_col
    b_ptrs = right_ptr + batch * stride_r_batch + offs_n[:, None] * stride_r_row + offs_k[None, :] * stride_r_col

    # load with mask
    a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < N), other=0.0)
    b = tl.load(b_ptrs, mask=mask_n[:, None] & (offs_k < N), other=0.0)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < N), other=0.0)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & (offs_k < N), other=0.0)

        a_ptrs += BLOCK_K * stride_l_col

    # store

"""
TriMul outgoing (AlphaFold3) forward pass fused with Triton.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (`dim`).
2. Cast to fp16 for compute‑intensive parts.
3. Linear projections (left/right) and gating projections (left_gate,
   right_gate, out_gate) – bias‑free.
4. Apply optional mask, fuse with projection & gating (pure PyTorch;
   a Triton kernel is used for the heavy N³ reduction).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication (B*H, N, N) @ (B*H, N, N)ᵀ
   using a custom Triton kernel.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and apply the final linear projection back to `dim`.
7. Cast the result back to the original dtype.

The kernel processes each (batch, hidden) slice independently, using
blocking (128×128×32) and fp16 arithmetic with fp32 accumulation.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _trimul_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    out_ptr,           # fp16 *
    B_H,               # i64 scalar = B * hidden_dim
    N,                 # i64 scalar = sequence length
    stride_l_batch,    # i64
    stride_l_row,      # i64
    stride_l_col,      # i64
    stride_r_batch,    # i64
    stride_r_row,      # i64
    stride_r_col,      # i64
    stride_o_batch,    # i64
    stride_o_row,      # i64
    stride_o_col,      # i64
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul where out = left @ rightᵀ.
    left, right, out have shape (B_H, N, N)."""
    pid = tl.program_id(0)

    # Compute which batch/hidden slice and which block within that slice
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_pid_m * num_pid_n)
    pid_rem = pid % (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n
    pid_n = pid_rem % num_pid_n

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # OOB masks for rows and columns
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the start of the current block
    a_ptrs = (
        left_ptr
        + batch * stride_l_batch
        + offs_m[:, None] * stride_l_row
        + offs_k[None, :] * stride_l_col
    )
    b_ptrs = (
        right_ptr
        + batch * stride_r_batch
        + offs_n[:, None] * stride_r_row
        + offs_k[None, :] * stride_r_col
    )

    # Accumulator in fp32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in blocks of size BLOCK_K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K
        a = tl.load(
            left_ptr
            + batch * stride_l_batch
            + offs_m[:, None] * stride_l_row
            + (cur_k + offs_k)[None, :] * stride_l_col,
            mask=mask_m[:, None] & ((cur_k + offs_k) < N),
            other=0.0,
        )
        b = tl.load(
            right_ptr
            + batch * stride_r_batch
            + offs_n[:, None] * stride_r_row
            + (cur_k + offs_k)[None, :] * stride_r_col,
            mask=mask_n[:, None] & ((cur_k + offs_k) < N),
            other=0.0,
        )
        acc += tl.dot(a, b)  # fp16 * fp16 -> fp32 accumulation

    # Write back result in fp16
    c = acc.to(tl.float16)
    c_ptrs = (
        out_ptr
        + batch * stride_o_batch
        + offs_m[:, None] * stride_o_row
        + offs_n[None, :] * stride_o_col
    )
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with model parameters (float32)
            - config: dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim` (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Load weights (bias‑free) and cast to fp16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid, fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Apply mask (if present) and fuse with projection & gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B, N, N) -> (B, N, N, 1)
        mask = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction using Triton: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # allocate output tensor
    out_flat = torch.empty_like(left_flat)

    # strides (all tensors are contiguous)
    stride_l_batch, stride_l_row, stride_l_col = left_flat.stride()
    stride_r_batch, stride_r_row, stride_r_col = right_flat.stride()
    stride_o_batch, stride_o_row, stride_o_col = out_flat.stride()

    # choose block sizes (tuned for B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_pid_m = math.ceil(N / BLOCK_M)
    num_pid_n = math.ceil(N / BLOCK_N)
    total_batches = B * H
    grid = (total_batches * num_pid_m * num_pid_n,)

    _trimul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        total_batches,
        N,
        stride_l_batch,
        stride_l_row,
        stride_l_col,
        stride_r_batch,
        stride_r_row,
        stride_r_col,
        stride_o_batch,
        stride_o_row,
        stride_o_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    # Cast to float32 for the final layer norm (better numeric stability)
    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # apply out gate (convert gate to float32)
    out_norm = out_norm * out_gate.to(torch.float32)
    # final linear projection back to `dim`
    out_final = F.linear(out_norm, to_out_w)  # (B, N, N, dim) in fp32

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out_final.dtype != input_tensor.dtype:
        out_final = out_final.to(input_tensor.dtype)

    return out_final
