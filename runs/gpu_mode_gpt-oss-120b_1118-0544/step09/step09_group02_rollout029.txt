@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_bh, stride_left_i, stride_left_k,
    stride_right_bh, stride_right_i, stride_right_k,
    stride_out_bh, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(2)   # batch*hidden index
    pid_m = tl.program_id(0)    # block row
    pid_n = tl.program_id(1)    # block col

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A block [M, K]
        a_ptrs = left_ptr + pid_bh * stride_left_bh + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        # Load B block [K, N] (transposed)
        b_ptrs = right_ptr + pid_bh * stride_right_bh + offs_k[:, None] * stride_right_i + offs_n[None, :] * stride_right_k

        a = tl.load(a_ptrs, mask=(offs_m < N)[:, None] & (offs_k < N)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < N)[:, None] & (offs_n < N)[None, :], other=0.0)

        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Store result
    c_ptrs = out_ptr + pid_bh * stride_out_bh + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    mask_store = (offs_m < N)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_store)
```

def _tri_mul(left: torch.Tensor, right: torch.Tensor, block_m=64, block_n=64, block_k=64):
    # left, right shape: (B, N, N, H)
    B, N, _, H = left.shape
    # permute to (B, H, N, N) and view as (BH, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * H
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # strides
    stride_left_bh = left_flat.stride(0)
    stride_left_i = left_flat.stride(1)
    stride_left_k = left_flat.stride(2)

    stride_right_bh = right_flat.stride(0)
    stride_right_i = right_flat.stride(1)
    stride_right_k = right_flat.stride(2)

    stride_out_bh = out_flat.stride(0)
    stride_out_i = out_flat.stride(1)
    stride_out_j = out_flat.stride(2)

    grid = (tl.cdiv(N, block_m), tl.cdiv(N, block_n), BH)
    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        H,
        stride_left_bh, stride_left_i, stride_left_k,
        stride_right_bh, stride_right_i, stride_right_k,
        stride_out_bh, stride_out_i, stride_out_j,
        BLOCK_M=block_m, BLOCK_N=block_n, BLOCK_K=block_k,
    )
    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
```

"""
TriMul (outgoing) forward pass using Triton fusion.

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to float16 for compute‑heavy parts.
3. Linear projections (left/right) and gating projections (left_gate,
   right_gate, out_gate) – all bias‑free.
4. Fuse projection, gating and optional mask with a small Triton kernel.
5. Core reduction (Triangle multiplicative update):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (A @ Bᵀ) per hidden channel using a
   Triton block‑wise kernel.
6. Layer‑norm over the hidden dimension, multiply by out_gate,
   and final linear projection back to the original dimension.
7. Cast back to the input dtype.

Only the projection‑gate‑mask fusion and the core reduction are executed
in Triton; the remaining operations use high‑performance PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------------------- #
# Triton kernel: fuse projection, gating and mask
# --------------------------------------------------------------------------- #
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask:
       left_out  = left_proj  * left_gate  * mask
       right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    in_range = offs < total_elems
    # each hidden element shares the same mask entry
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=in_range, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=in_range, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=in_range, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=in_range, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=in_range, other=1.0)

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=in_range)
    tl.store(right_out_ptr + offs, right_out, mask=in_range)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launches the Triton kernel that fuses projection, gating and mask.
    All tensor arguments must be fp16 and contiguous.
    Returns (left, right) each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],   # hidden_dim H
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# --------------------------------------------------------------------------- #
# Triton kernel: batched matrix multiplication for the triangle update
# --------------------------------------------------------------------------- #
@triton.jit
def _tri_mul_kernel(
    left_ptr,       # fp16 *
    right_ptr,      # fp16 *
    out_ptr,        # fp16 *
    N,              # i64 scalar (seq length)
    stride_l_bh, stride_l_i, stride_l_k,
    stride_r_bh, stride_r_i, stride_r_k,
    stride_o_bh, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes for each batch*hidden:
        out = left @ rightᵀ
    where left/right have shape (N, N).
    """
    pid_bh = tl.program_id(2)   # batch * hidden index
    pid_m  = tl.program_id(0)   # block row
    pid_n  = tl.program_id(1)   # block column

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A block [M, K]
        a_ptrs = left_ptr + pid_bh * stride_l_bh \
                 + offs_m[:, None] * stride_l_i \
                 + offs_k[None, :] * stride_l_k
        # Load B block [K, N]   (transposed view of right)
        b_ptrs = right_ptr + pid_bh * stride_r_bh \
                 + offs_k[:, None] * stride_r_i \
                 + offs_n[None, :] * stride_r_k

        a = tl.load(a_ptrs,
                    mask=(offs_m < N)[:, None] & (offs_k < N)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k < N)[:, None] & (offs_n < N)[None, :],
                    other=0.0)

        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Store result as fp16
    c_ptrs = out_ptr + pid_bh * stride_o_bh \
             + offs_m[:, None] * stride_o_i \
             + offs_n[None, :] * stride_o_j
    mask_store = (offs_m < N)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_store)


def _tri_mul(left: torch.Tensor, right: torch.Tensor,
             block_m: int = 64, block_n: int = 64, block_k: int = 64) -> torch.Tensor:
    """
    Batched triangle multiplication:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    Input tensors are (B, N, N, H) in fp16.
    Returns out in fp16 with the same shape.
    """
    B, N, _, H = left.shape

    # (B, H, N, N) -> (BH, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * H
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    out_flat = torch.empty_like(left_flat)

    # strides for the flattened tensors
    stride_l_bh, stride_l_i, stride_l_k = left_flat.stride()
    stride_r_bh, stride_r_i, stride_r_k = right_flat.stride()
    stride_o_bh, stride_o_i, stride_o_j = out_flat.stride()

    grid = (triton.cdiv(N, block_m), triton.cdiv(N, block_n), BH)
    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_l_bh, stride_l_i, stride_l_k,
        stride_r_bh, stride_r_i, stride_r_k,
        stride_o_bh, stride_o_i, stride_o_j,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# --------------------------------------------------------------------------- #
# Entry point
# --------------------------------------------------------------------------- #
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) LayerNorm over channel dimension `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast linear weights to FP16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections (FP16)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 6) Fuse mask, projection and gating (if needed)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B, N, N) -> fp16 contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask (or mask ignored): simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7) Core reduction (Triangle multiplicative update)
    # --------------------------------------------------------------
    out = _tri_mul(left, right)   # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9) Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
