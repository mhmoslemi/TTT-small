@triton.jit
def batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH, M, N, K,
    stride_left_b, stride_left_m, stride_left_k,
    stride_right_b, stride_right_k, stride_right_n,
    stride_out_b, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_m = tl.program_id(0)  # over M blocks
    pid_n = tl.program_id(1)  # over N blocks
    pid_b = tl.program_id(2)  # batch dimension

    # compute base offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + pid_b * stride_left_b + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load right tile: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + pid_b * stride_right_b + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product accumulate
        acc += tl.dot(left, right)

    # write output
    out_ptrs = out_ptr + pid_b * stride_out_b + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for the compute‑heavy parts.
3. Linear projections (left/right) and gating projections (left_gate,
   right_gate, out_gate) – all bias‑free.
4. Fuse mask, projection and gating with a small Triton kernel
   (if a mask is present and not ignored).
5. Core reduction (the “tri‑mul”):
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (left @ rightᵀ) using a
   Triton kernel that tiles over the i‑ and j‑axes.
6. Output layer‑norm, gating, and final linear projection back to `dim`.
7. Cast back to the original input dtype.

Only the forward pass is provided; gradients are not required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,    # fp16 *
    right_proj_ptr,   # fp16 *
    left_gate_ptr,    # fp16 *
    right_gate_ptr,   # fp16 *
    mask_ptr,         # fp16 *
    left_out_ptr,     # fp16 *
    right_out_ptr,    # fp16 *
    total_elems,      # i64 scalar = B*N*N*H
    hidden_dim,       # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask:
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # each element corresponds to a hidden dimension slice; compute
    # the index of the corresponding mask entry (B,N,N) by dividing out H
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=valid, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=valid, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launches the Triton kernel that fuses projection, gating and mask.
    All tensor arguments must be fp16 and contiguous.
    Returns (left, right) each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


@triton.jit
def _batched_matmul_kernel(
    left_ptr,   # fp16 *
    right_ptr,  # fp16 *
    out_ptr,    # fp16 *
    BATCH, M, N, K,
    stride_l_b, stride_l_m, stride_l_k,
    stride_r_b, stride_r_k, stride_r_n,
    stride_o_b, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: out[b, i, j] = Σ_k left[b, i, k] * right[b, k, j]
    left  : (BATCH, M, K)
    right : (BATCH, K, N)
    out   : (BATCH, M, N)
    """
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_b = tl.program_id(2)  # batch index

    # Coordinates of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load left tile (M x K)
        left_ptrs = left_ptr + pid_b * stride_l_b \
                    + offs_m[:, None] * stride_l_m \
                    + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # Load right tile (K x N)
        right_ptrs = right_ptr + pid_b * stride_r_b \
                     + offs_k[:, None] * stride_r_k \
                     + offs_n[None, :] * stride_r_n
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(left, right)

    # Write back
    out_ptrs = out_ptr + pid_b * stride_o_b \
               + offs_m[:, None] * stride_o_m \
               + offs_n[None, :] * stride_o_n
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast linear weights to FP16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections (FP16)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 6) Apply mask & fuse with projection & gating
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7) Core reduction (TriMul) – batched GEMM in Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # (_ == N)

    # Rearrange tensors for batched GEMM:
    # left : (B, N, N, H) -> (B, H, N, N) where axes are (b, h, i, k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: (B, N, N, H) -> (B, H, N, N) where axes are (b, h, k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # Collapse batch and hidden dimensions
    batch_dim = B * H
    left_flat = left_perm.view(batch_dim, N, N)
    right_flat = right_perm.view(batch_dim, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Strides
    stride_l_b, stride_l_m, stride_l_k = left_flat.stride()
    stride_r_b, stride_r_k, stride_r_n = right_flat.stride()
    stride_o_b, stride_o_m, stride_o_n = out_flat.stride()

    # Block sizes (tuned for typical B200 GPUs)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_dim,
    )
    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_dim,
        N,
        N,
        N,
        stride_l_b,
        stride_l_m,
        stride_l_k,
        stride_r_b,
        stride_r_k,
        stride_r_n,
        stride_o_b,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9) Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
