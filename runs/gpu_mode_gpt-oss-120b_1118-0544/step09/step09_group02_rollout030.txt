@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_head = tl.program_id(0)   # combined batch and head
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # bound check
    mask_m = offs_m < N
    mask_n = offs_n < N

    # compute pointer base for this batch_head
    # batch_head index -> batch = batch_head // H, head = batch_head % H
    batch_idx = batch_head // H
    head_idx = batch_head % H

    # compute base offsets
    # pointer to left[batch, i, k, head] after permuting to (B, H, N, N)
    # We will compute left_ptr offset: batch*stride_l_batch + head*stride_l_h + i*stride_l_i + k*stride_l_k

    # But after permute we have shape (B, H, N, N). So strides: stride_l_batch (B*H dimension), stride_l_h (head), stride_l_i (row i), stride_l_k (col k). We can compute base = left_ptr + batch_idx*stride_l_batch + head_idx*stride_l_h

    # We'll calculate base pointer for left:
    left_base = left_ptr + batch_idx * stride_l_batch + head_idx * stride_l_h

    # Similarly for right:
    right_base = right_ptr + batch_idx * stride_r_batch + head_idx * stride_r_h

    # out base
    out_base = out_ptr + batch_idx * stride_o_batch + head_idx * stride_o_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left tile (M,K)
        left_ptrs = left_base + (offs_m[:, None] * stride_l_i) + (k_offset[None, :] * stride_l_k)
        a = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load right tile (K,N) by loading right with k as first dim, j as second
        right_ptrs = right_base + (k_offset[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_j)
        b = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # multiply accumulate
        acc += tl.dot(a, b)

    # write output
    out_ptrs = out_base + (offs_m[:, None] * stride_o_i) + (offs_n[None, :] * stride_o_j)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])

    # cast to fp16
    x = x.to(torch.float16)

    # linear projections
    left_proj = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x, weights["right_proj.weight"].to(torch.float16))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    if mask is not None and not nomask:
        # mask shape [B,N,N]; need to broadcast to hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Now compute core reduction using Triton
    B, N, _, H = left.shape

    # Permute to (B, H, N, N) and contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Allocate output: same shape
    out = torch.empty_like(left_perm)  # (B, H, N, N), dtype fp16

    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        math.ceil(N / BLOCK_M),
        math.ceil(N / BLOCK_N),
    )
    trimul_kernel[grid](
        left_perm,
        right_perm,
        out,
        B,
        N,
        H,
        left_perm.stride(0), left_perm.stride(2), left_perm.stride(3), left_perm.stride(1),
        right_perm.stride(0), right_perm.stride(2), right_perm.stride(3), right_perm.stride(1),
        out.stride(0), out.stride(2), out.stride(3), out.stride(1),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # out currently shape (B, H, N, N); need to permute back to (B, N, N, H)
    out = out.permute(0, 2, 3, 1).contiguous()

    # post processing: layer norm over hidden dim
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"].to(torch.float16), bias=weights["to_out_norm.bias"].to(torch.float16))
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # cast back to original dtype
    out = out.to(input_tensor.dtype)
    return out

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    batch_head = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch and head indices
    batch_idx = batch_head // H
    head_idx = batch_head % H

    # offsets for M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch/head
    left_base = left_ptr + batch_idx * stride_l_batch + head_idx * stride_l_h
    right_base = right_ptr + batch_idx * stride_r_batch + head_idx * stride_r_h
    out_base = out_ptr + batch_idx * stride_o_batch + head_idx * stride_o_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left tile (M,K)
        left_ptrs = left_base + (offs_m[:, None] * stride_l_i) + (k_offset[None, :] * stride_l_k)
        a = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load right tile (K,N) -> we need B transposed, so load k as first dim, j as second
        right_ptrs = right_base + (k_offset[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_j)
        b = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)   # (M,K) @ (K,N) -> (M,N)

    # store result
    out_ptrs = out_base + (offs_m[:, None] * stride_o_i) + (offs_n[None, :] * stride_o_j)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass – highly optimized Triton implementation.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Cast to fp16 for compute‑heavy parts.
3. Linear projections (left/right) and gating projections (left_gate,
   right_gate, out_gate) – all bias‑free.
4. Optional mask fusion (element‑wise multiplication, cheap).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM in Triton (left @ rightᵀ) with
   fp16 inputs and fp32 accumulation.
6. Layer‑norm over the hidden dimension, multiply by out_gate,
   final linear projection back to `dim`.
7. Cast back to the input dtype.

The kernel processes each (batch, head) pair as an independent GEMM,
using tiled loads and `tl.dot` for high arithmetic intensity.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _trimul_kernel(
    left_ptr,               # fp16 *
    right_ptr,              # fp16 *
    out_ptr,                # fp16 *
    B, N, H,                # i64 scalars
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM kernel computing
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where left/right are stored as (B, H, N, N) after a permute.
    """
    # program ids
    batch_head = tl.program_id(0)          # combined batch * head
    pid_m = tl.program_id(1)               # block row
    pid_n = tl.program_id(2)               # block col

    # recover batch and head indices
    batch_idx = batch_head // H
    head_idx = batch_head % H

    # offsets within the N x N matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this (batch, head)
    left_base = left_ptr + batch_idx * stride_l_batch + head_idx * stride_l_h
    right_base = right_ptr + batch_idx * stride_r_batch + head_idx * stride_r_h
    out_base = out_ptr + batch_idx * stride_o_batch + head_idx * stride_o_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction dimension K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left tile: (M, K)
        left_ptrs = left_base + (offs_m[:, None] * stride_l_i) + (k_offset[None, :] * stride_l_k)
        a = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right tile: we need rightᵀ, so load as (K, N)
        right_ptrs = right_base + (k_offset[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_j)
        b = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # accumulate
        acc += tl.dot(a, b).to(tl.float32)

    # write output (convert back to fp16)
    out_ptrs = out_base + (offs_m[:, None] * stride_o_i) + (offs_n[None, :] * stride_o_j)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config) where
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # unpack arguments (support both 3‑ or 4‑tuple inputs)
    if len(data) == 4:
        input_tensor, mask, weights, config = data
    else:
        input_tensor, weights, config = data
        mask = None

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) LayerNorm over the channel dimension
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Linear projections (FP16)
    # --------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 4) Gating projections (sigmoid, FP16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    # --------------------------------------------------------------
    # 5) Optional mask fusion (simple element‑wise)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6) Core reduction: batched GEMM (left @ rightᵀ) via Triton
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Rearrange to (B, H, N, N) for a conventional GEMM layout
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Allocate output tensor
    out_perm = torch.empty_like(left_perm)  # fp16, shape (B, H, N, N)

    # Tiling parameters – chosen to fit in registers and shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                 # batch * head
        math.ceil(N / BLOCK_M),                # rows
        math.ceil(N / BLOCK_N),                # cols
    )

    # Launch the Triton kernel
    _trimul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B,
        N,
        H,
        left_perm.stride(0), left_perm.stride(2), left_perm.stride(3), left_perm.stride(1),
        right_perm.stride(0), right_perm.stride(2), right_perm.stride(3), right_perm.stride(1),
        out_perm.stride(0), out_perm.stride(2), out_perm.stride(3), out_perm.stride(1),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Re‑shape back to [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 7) Output LayerNorm, gating, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 8) Cast back to the original dtype
    # --------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
