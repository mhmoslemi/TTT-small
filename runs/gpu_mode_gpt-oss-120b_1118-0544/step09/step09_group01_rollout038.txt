def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, M, N, K,
    stride_lbm, stride_lbk, stride_lbn,
    stride_rbm, stride_rbk, stride_rbn,
    stride_obm, stride_obn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block start
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # mask for K dim
        mask_k = offs_k < K

        # Load left block (M x K)
        left_ptrs = left_ptr + b * stride_lbm + offs_m[:, None] * stride_lbk + offs_k[None, :] * stride_lbn

        # Load right block transposed (K x N)
        # right shape (batch, N, K); we want (K,N)
        right_ptrs = right_ptr + b * stride_rbm + offs_k[:, None] * stride_rbk + offs_n[None, :] * stride_rbn

        # Multiply-accumulate

    # Cast accumulator to fp16
    out = acc.to(tl.float16)

    # Store results
    out_ptrs = out_ptr + b * stride_obm + offs_m[:, None] * stride_obn + offs_n[None, :]

Potential pitfalls:

- We need to ensure Torch's tensor shape after linear is (B,N,N,hidden_dim). That is a 4D tensor. The permute order for flatten: .permute(0,3,1,2) rearranges to (B,H,N,N). Then view.

- The final linear layer expects input shape (..., hidden_dim). So we can apply it directly on out (B,N,N,H). We'll use F.linear.

"""
custom_kernel: Triton‑accelerated outgoing TriMul forward (AlphaFold3).

Algorithm
---------
1️⃣ Layer‑norm over the channel dimension `dim`.
2️⃣ Cast to fp16 and compute linear projections:
   left_proj, right_proj, left_gate, right_gate, out_gate.
3️⃣ Fuse mask (if present) with projections and gates:
   left  = left_proj  * mask * left_gate
   right = right_proj * mask * right_gate
4️⃣ Core reduction:
   For each head `h` and batch `b`,
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM L @ Rᵀ and is implemented in Triton
   (kernel `triton_trimul_kernel`).
5️⃣ Output LayerNorm, out‑gate multiplication and final linear projection.
6️⃣ Cast back to float32.

The heavy O(N³) work (step 4) is the only part done in Triton;
all other ops use highly‑optimized torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul core reduction
# ----------------------------------------------------------------------
@triton.jit
def triton_trimul_kernel(
    left_ptr,          # fp16 * (B*H, M, K)      left matrix
    right_ptr,         # fp16 * (B*H, N, K)      right matrix (will be used transposed)
    out_ptr,           # fp16 * (B*H, M, N)      output
    M, N, K,           # matrix dimensions
    stride_lbm, stride_lbk, stride_lbn,   # strides for left (batch, M, K)
    stride_rbm, stride_rbk, stride_rbn,   # strides for right (batch, N, K)
    stride_obm, stride_obn,               # strides for out (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """ Compute out[b, i, j] = Σ_k left[b, i, k] * right[b, j, k] """
    b = tl.program_id(0)          # batch index (B*H)
    pid_m = tl.program_id(1)      # block index over M (i)
    pid_n = tl.program_id(2)      # block index over N (j)

    # --------------------------------------------------------------
    # Compute offsets for the current block
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left block: (M x K)
        left_ptrs = left_ptr + \
            b * stride_lbm + \
            offs_m[:, None] * stride_lbk + \
            offs_k[None, :] * stride_lbn
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # right block transposed: we need (K x N)
        # right shape is (batch, N, K); we index as (batch, k, n)
        right_ptrs = right_ptr + \
            b * stride_rbm + \
            offs_k[:, None] * stride_rbk + \
            offs_n[None, :] * stride_rbn
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        # accumulate dot product over K
        acc += tl.dot(left, right)

    # --------------------------------------------------------------
    # Write back the result (cast to fp16)
    # --------------------------------------------------------------
    out = acc.to(tl.float16)
    out_ptrs = out_ptr + \
        b * stride_obm + \
        offs_m[:, None] * stride_obn + \
        offs_n[None, :]
    tl.store(out_ptrs, out,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys:
            * "dim"        – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask" (optional) – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to fp16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections and gates (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate = F.linear(x, left_gate_w).sigmoid()   # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid() # (B, N, N, H)
    out_gate = F.linear(x, out_gate_w).sigmoid()     # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask + gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape -> (B, N, N, 1) for broadcasting over hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        left = left_proj * mask_fp16 * left_gate
        right = right_proj * mask_fp16 * right_gate
    else:
        # No mask (or mask ignored)
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and nomask:
            # mask present but explicitly ignored → still broadcast
            mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
            left = left * mask_fp16
            right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6) Core reduction: batched GEMM L @ R^T via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                    # B,N,N,H
    # reshape to (B*H, N, N) for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Block sizes (tuned for A100/B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    batch = left_flat.shape[0]
    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # launch the kernel
    triton_trimul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                     # M
        N,                     # N
        N,                     # K
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(2),  # stride for K dimension in transposed view
        right_flat.stride(1),  # stride for N dimension
        out_flat.stride(0),
        out_flat.stride(1),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to float32 (required output dtype)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
