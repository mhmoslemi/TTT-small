def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B,N,N,dim)

    # Cast to half
    x = x.to(torch.float16)

    B, N, _, _ = x.shape

    # flatten
    x_flat = x.reshape(-1, dim)  # (B*N*N, dim)

    # get weights in half
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # combine left projection and gate weights
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    # linear
    left_comb = torch.nn.functional.linear(x_flat, left_comb_w)  # (B*N*N, 2*H)
    right_comb = torch.nn.functional.linear(x_flat, right_comb_w)

    # split
    H = hidden_dim
    left_proj = left_comb[:, :H]
    left_gate = torch.sigmoid(left_comb[:, H:])
    right_proj = right_comb[:, :H]
    right_gate = torch.sigmoid(right_comb[:, H:])

    # out_gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_w))

    # Reshape to (B,N,N,H)
    left_proj = left_proj.view(B, N, N, H)
    left_gate = left_gate.view(B, N, N, H)
    right_proj = right_proj.view(B, N, N, H)
    right_gate = right_gate.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Apply mask and gating using Triton kernel if needed
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        # mask shape (B,N,N)
        # Flatten for kernel
        left_out, right_out = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16,
        )
    else:
        left_out = left_proj * left_gate
        right_out = right_proj * right_gate

    # Core reduction
    # left_out shape (B,N,N,H)
    # Permute to (B, H, N, N)
    left_perm = left_out.permute(0, 3, 1, 2).contiguous()
    right_perm = right_out.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # output layernorm
    out = torch.nn.functional.layer_norm(

        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # scale by out_gate
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, to_out_w)  # input shape (B,N,N,H) -> output (B,N,N,dim)

    # cast back to original dtype
    if out.dtype != dtype:

    return out

   Apply sigmoid to the gate part and optionally a mask.
4. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM:
   (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N).
5. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.
6. Cast the result back to the original dtype.

A small Triton kernel fuses the element‑wise mask, projection and gating
step for both the left and right branches.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse projection, gating and mask:
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask
    The mask has shape (B, N, N) and is broadcast over the hidden dim.
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    # Index of the element in the (B,N,N) mask tensor (without the hidden dim)
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # default 1.0 = no mask

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launches the Triton kernel that fuses projection, gating and mask.

    All tensors must be fp16 and contiguous.
    Returns (left, right) each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) – can be all‑ones or None.
            - weights: dict containing model parameters (float32)
            - config: dict with keys:
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy part
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    B, N, _, _ = x.shape
    HW = hidden_dim

    # --------------------------------------------------------------
    # 3) Prepare weight matrices (FP16)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4) Fuse projection + gate via concatenated matmul
    # --------------------------------------------------------------
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)

    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    left_comb = F.linear(x_flat, left_comb_w)    # (B*N*N, 2*H)
    right_comb = F.linear(x_flat, right_comb_w)  # (B*N*N, 2*H)

    left_proj = left_comb[:, :HW]
    left_gate = torch.sigmoid(left_comb[:, HW:])
    right_proj = right_comb[:, :HW]
    right_gate = torch.sigmoid(right_comb[:, HW:])

    # out_gate (separate linear)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w))

    # --------------------------------------------------------------
    # 5) Reshape tensors back to (B, N, N, H)
    # --------------------------------------------------------------
    left_proj = left_proj.view(B, N, N, HW)
    left_gate = left_gate.view(B, N, N, HW)
    right_proj = right_proj.view(B, N, N, HW)
    right_gate = right_gate.view(B, N, N, HW)
    out_gate = out_gate.view(B, N, N, HW)

    # --------------------------------------------------------------
    # 6) Apply mask, projection and gating (Triton fusion)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()  # (B, N, N)
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7) Core reduction (batched GEMM)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * HW, N, N)
    right_flat = right_perm.view(B * HW, N, N)

    # (B*H, N, N) @ (B*H, N, N)ᵀ -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, HW, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (HW,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9) Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
