@triton.jit
def matmul_triton_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b_left, stride_i_left, stride_k_left, stride_h_left,
    stride_b_right, stride_i_right, stride_k_right, stride_h_right,
    stride_b_out,  stride_i_out,  stride_j_out,  stride_h_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # grid: (ceil(N/BLOCK_M), ceil(N/BLOCK_N), B*H)
    pid_m = tl.program_id(0) # i tile
    pid_n = tl.program_id(1) # j tile
    pid_z = tl.program_id(2) # batch*hidden index

    # compute batch and head
    b = pid_z // H
    h = pid_z % H

    # start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute mask for within bounds
    mask_m = offs_m < N
    mask_n = offs_n < N
    # k mask will be applied within loop, but we can load with out-of-bounds zeros.

    # compute base pointers for left and right for this batch and head
    left_base = left_ptr + b * stride_b_left + h * stride_h_left
    right_base = right_ptr + b * stride_b_right + h * stride_h_right
    out_base = out_ptr + b * stride_b_out + h * stride_h_out

    # compute pointers with offsets
    # left: (i,k) shape (BLOCK_M, BLOCK_K)
    a_ptrs = left_base + offs_m[:, None] * stride_i_left + offs_k[None, :] * stride_k_left
    # right: (j,k) shape (BLOCK_N, BLOCK_K)
    b_ptrs = right_base + offs_n[:, None] * stride_i_right + offs_k[None, :] * stride_k_right

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute real k offset
        k_offset = k * BLOCK_K
        # compute current k indices
        cur_k = k_offset + offs_k
        cur_k_mask = cur_k < N

        # load A and B
        a = tl.load(a_ptrs + k_offset * stride_k_left, mask=mask_m[:, None] & cur_k_mask[None, :], other=0.0)
        b = tl.load(b_ptrs + k_offset * stride_k_right, mask=mask_n[:, None] & cur_k_mask[None, :], other=0.0)

        # accumulate
        acc += tl.dot(a, b)

    # write out
    out_ptrs = out_base + offs_m[:, None] * stride_i_out + offs_n[None, :] * stride_j_out
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16
    x = x.to(torch.float16)

    # Convert weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)  # (B, N, N, hidden_dim)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # Gating
    left_gate = torch.nn.functional.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.nn.functional.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.nn.functional.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply mask and gate
    if mask is not None and not nomask:
        # mask shape (B,N,N), expand to (B,N,N,1)
        mask = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()

"""
TriMul outgoing (AlphaFold3) forward pass with Triton fused reduction.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (left/right) and gating projections (left_gate,
   right_gate, out_gate) – all without bias.
4. Apply the mask (if provided) and element‑wise gating.
5. Core reduction (TriMul):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication per hidden slice
   (left_h @ right_hᵀ) using a custom Triton kernel.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and apply the final linear projection back to `dim`.
7. Cast the result back to the original dtype.

The heavy O(B·H·N³) reduction is performed entirely on the GPU
with Triton, while the remaining cheap operations use PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _triton_trimul_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    out_ptr,           # fp16 *
    B, N, H,           # i32 scalar
    stride_b_left, stride_i_left, stride_k_left, stride_h_left,
    stride_b_right, stride_i_right, stride_k_right, stride_h_right,
    stride_b_out, stride_i_out, stride_j_out, stride_h_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication for TriMul:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    The kernel treats each (b, h) pair as an independent matrix
    multiplication of size (N, N)."""
    pid_m = tl.program_id(0)   # i‑tile
    pid_n = tl.program_id(1)   # j‑tile
    pid_z = tl.program_id(2)   # combined batch * hidden index

    # De‑compose batch and head indices
    b = pid_z // H
    h = pid_z % H

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds rows/cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this (b, h) pair
    left_base  = left_ptr  + b * stride_b_left  + h * stride_h_left
    right_base = right_ptr + b * stride_b_right + h * stride_h_right
    out_base   = out_ptr   + b * stride_b_out   + h * stride_h_out

    # Pointers to tiles.
    # left: (M, K)   -> indices (i, k)
    a_ptrs = left_base + offs_m[:, None] * stride_i_left + offs_k[None, :] * stride_k_left
    # right: we need rightᵀ so we load (K, N) -> indices (k, j)
    b_ptrs = right_base + offs_k[:, None] * stride_k_right + offs_n[None, :] * stride_i_right

    # Accumulator in float32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for cur_k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = cur_k * BLOCK_K
        cur_k_idx = k_offset + offs_k
        mask_k = cur_k_idx < N

        # Load tiles with masking
        a = tl.load(a_ptrs + k_offset * stride_k_left,
                    mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs + k_offset * stride_k_right,
                    mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate dot product
        acc += tl.dot(a, b)

    # Compute output pointers (M, N)
    out_ptrs = out_base + offs_m[:, None] * stride_i_out + offs_n[None, :] * stride_j_out

    # Store result cast back to fp16
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights: dict with keys:
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast model weights to FP16 (no bias terms)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Apply mask and fuse with projection & gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and broadcastable
        mask_fp16 = mask.to(torch.float16).to(device).unsqueeze(-1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Make tensors contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul) via Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    out = torch.empty_like(left)  # (B, N, N, H), fp16

    # Strides (in element counts)
    stride_b_left, stride_i_left, stride_k_left, stride_h_left = left.stride()
    stride_b_right, stride_i_right, stride_k_right, stride_h_right = right.stride()
    stride_b_out, stride_i_out, stride_j_out, stride_h_out = out.stride()

    # Block sizes – tuned for typical B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _triton_trimul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_b_left, stride_i_left, stride_k_left, stride_h_left,
        stride_b_right, stride_i_right, stride_k_right, stride_h_right,
        stride_b_out, stride_i_out, stride_j_out, stride_h_out,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
