"""
TriMul outgoing kernel (AlphaFold3).

Implements the forward pass of the “outgoing” Triangle Multiplicative Update.
The steps are:
1. LayerNorm over the input channel dimension.
2. Linear projections (left/right) and gating (left/right/out) (bias‑free).
3. Optional mask is fused with projection and gating via a Triton kernel.
4. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   realized as a batched matrix multiplication (left @ rightᵀ) in FP16.
5. Output LayerNorm, output‑gate, and final linear projection back to the
   original dimension.  Result is returned in the input dtype.

At least part of the work (proj + gate + mask) is performed in a Triton
kernel; the expensive reduction leverages PyTorch’s highly‑optimized
batched GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask for left/right tensors.
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B, N, N, H)
    right_proj_ptr,  # fp16 * (B, N, N, H)
    left_gate_ptr,   # fp16 * (B, N, N, H)
    right_gate_ptr,  # fp16 * (B, N, N, H)
    mask_ptr,        # fp16 * (B, N, N)
    left_out_ptr,    # fp16 * (B, N, N, H)
    right_out_ptr,   # fp16 * (B, N, N, H)
    total_elems,     # i64 = B*N*N*H
    hidden_dim,      # i64 = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gate and mask for left/right tensors."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # bounds check
    mask = offs < total_elems

    # index of the mask (mask has no hidden dim)
    mask_idx = offs // hidden_dim

    # load values (zero‑fill out‑of‑bounds)
    lproj = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rproj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # fuse
    lout = lproj * lgate * m
    rout = rproj * rgate * m

    tl.store(left_out_ptr + offs,  lout, mask=mask)
    tl.store(right_out_ptr + offs, rout, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """Fuse mask, projection and gate for both left and right tensors."""
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out

    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Batched matrix multiplication; result (B, H, N, N)
    out_perm = torch.matmul(
        left_perm, right_perm.transpose(-2, -1)
    )  # (B, H, N, N)

    # Back to (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
