"""
TriMul outgoing kernel (AlphaFold3) – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm over the input channel dimension.
2. Linear projections (left/right) and gating (left/right/out) are performed
   with FP16 weights.  If a mask is present (and not disabled) the mask,
   projection and gate are fused in a Triton kernel.
3. Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is a batched matrix multiplication (left @ rightᵀ) for each
   (batch, hidden) pair.  This is implemented with a custom Triton
   kernel that processes the batch dimension explicitly.
4. Second layer‑norm, output‑gate and final linear projection back to the
   original dimension.
5. Output is cast back to the input dtype.

The implementation mixes PyTorch for high‑level ops (norms, linear layers)
and Triton for the two performance‑critical parts:
* Fuse mask, projection and gate.
* Batched GEMM for the N³ reduction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# 1. Mask / projection / gate fusion kernel
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B, N, N, H)
    right_proj_ptr,  # fp16 * (B, N, N, H)
    left_gate_ptr,   # fp16 * (B, N, N, H)
    right_gate_ptr,  # fp16 * (B, N, N, H)
    mask_ptr,        # fp16 * (B, N, N)
    left_out_ptr,    # fp16 * (B, N, N, H)
    right_out_ptr,   # fp16 * (B, N, N, H)
    total_elems,     # i64 = B*N*N*H
    hidden_dim,      # i64 = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and optional mask."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # Index of the mask element (mask has no hidden dim)
    mask_idx = offs // hidden_dim

    lproj = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rproj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    lout = lproj * lgate * m
    rout = rproj * rgate * m

    tl.store(left_out_ptr + offs,  lout, mask=mask)
    tl.store(right_out_ptr + offs, rout, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """Fuse mask, projection and gate for both left and right tensors."""
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out

def _batched_matmul(
    left: torch.Tensor,
    right: torch.Tensor,
    block_m: int = 128,
    block_n: int = 128,
    block_k: int = 32,
) -> torch.Tensor:
    """
    Perform batched matrix multiplication `left @ rightᵀ` where
    left, right have shape (BATCH, N, N).  Returns tensor of the same shape.
    """
    assert left.is_contiguous() and right.is_contiguous()
    BATCH, M, K = left.shape
    _, K2, N = right.shape
    assert K == K2, "Inner dimensions must match"

    out = torch.empty((BATCH, M, N), dtype=left.dtype, device=left.device)

    # strides (in elements)
    stride_ab = left.stride(0)
    stride_am = left.stride(1)
    stride_ak = left.stride(2)

    stride_bb = right.stride(0)
    stride_bk = right.stride(1)
    stride_bn = right.stride(2)

    stride_cb = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    grid = (
        BATCH,
        triton.cdiv(M, block_m),
        triton.cdiv(N, block_n),
    )

    _batched_matmul_kernel[grid](
        left,
        right,
        out,
        BATCH, M, N, K,
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cb, stride_cm, stride_cn,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return out

# 3. Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] (binary) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections and gates (FP16)
    # ------------------------------------------------------------------
    # shape after linear: (B, N, N, hidden_dim)
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate = F.linear(x, left_gate_w).sigmoid()   # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid() # (B, N, N, H)
    out_gate = F.linear(x, out_gate_w).sigmoid()     # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse mask, projection and gate (Triton kernel when needed)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
            block_size=1024,
        )
    else:
        # simple element‑wise multiplication (mask treated as all ones)
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # we need rightᵀ; we achieve this by transposing the last two dims
    right_t = right_flat.transpose(1, 2).contiguous()

    out_flat = _batched_matmul(left_flat, right_t,
                               block_m=128, block_n=128, block_k=32)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
