pid_n = tl.program_id(1)  # col tile index

# Compute offsets
i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

# Initialize accumulator
acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

# Loop over k
for k in range(0, tl.cdiv(K, BLOCK_K)):
    k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

    # Load A block

# Write back
c_ptrs = out_ptr + pid_batch * stride_out_batch + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j

"""
TriMul (outgoing) forward pass – Triton‑accelerated core reduction.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (float32).
2. Cast to fp16 for the compute‑heavy path.
3. Fuse the five linear projections (left, right, left_gate, right_gate,
   out_gate) into a single GEMM to reduce memory traffic.
4. Apply sigmoid to the three gate tensors, multiply by the optional mask
   (if present and not ignored) and fuse the gates with the projections.
5. Compute the TriMul reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matmul `left @ rightᵀ`.  This is performed by a
   custom Triton kernel that iterates over `k` in tiles.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   out‑gate, and a final linear projection back to the original channel
   dimension.
7. Cast the result back to the input dtype.

Only the core reduction (step 5) is written in Triton; all other
operations use PyTorch primitives for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # dimensions: batch*H, seq, seq
    stride_a_batch, stride_a_i, stride_a_k,   # strides for A (left)
    stride_b_batch, stride_b_k, stride_b_i,   # strides for Bᵀ (right transposed)
    stride_c_batch, stride_c_i, stride_c_j,   # strides for C (output)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ Bᵀ   (A,B shape = [M,N,K])"""
    pid_batch = tl.program_id(2)               # batch*hidden index
    pid_m = tl.program_id(0)                   # tile row index
    pid_n = tl.program_id(1)                   # tile col index

    # Offsets for the output tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialise accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + offs_i[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a_mask = (offs_i[:, None] < N) & (offs_k[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # Load B tile transposed: original B is [M,N,K]; we need Bᵀ of shape (K,N)
        # Row stride = stride_b_k (original K‑stride), col stride = stride_b_i (original N‑stride)
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_j[None, :] * stride_b_i
        )
        b_mask = (offs_k[:, None] < K) & (offs_j[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Dot product and accumulate
        acc += tl.dot(a, b)

    # Write back the result in fp16
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_i[:, None] * stride_c_i
        + offs_j[None, :] * stride_c_j
    )
    c_mask = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Fuse all five linear projections into a single GEMM
    # --------------------------------------------------------------
    # Combine weight matrices: shape (5*H, C)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # (5*H, C)

    # Single linear transform -> (B, N, N, 5*H)
    combined = F.linear(x, combined_weight)  # bias is None

    # Split the combined tensor
    split = torch.split(combined, hidden_dim, dim=-1)
    left_proj, right_proj, left_gate, right_gate, out_gate = split

    # --------------------------------------------------------------
    # 4) Apply sigmoid to gate tensors
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # --------------------------------------------------------------
    # 5) Apply mask (if present) and fuse with projections/gates
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape: [B, N, N] -> broadcast to hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6) Core reduction: out = left @ rightᵀ   (batched GEMM)
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # Merge batch and hidden dimensions for the kernel
    M = B * H
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(M, N, N)   # (M,N,N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(M, N, N)  # (M,N,N)

    out_flat = torch.empty_like(left_flat)  # fp16 output

    # Tile sizes – chosen to fit well on NVIDIA GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),  # rows
        triton.cdiv(N, BLOCK_N),  # cols
        M,                        # batch*hidden
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M,
        N,
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(2),   # row stride for Bᵀ (original K‑stride)
        right_flat.stride(1),   # col stride for Bᵀ (original N‑stride)
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = (
        out_flat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, H)

    # --------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 8) Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)

    return out
