@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)  # batch-head index
    pid1 = tl.program_id(1)  # block row i
    pid2 = tl.program_id(2)  # block col j

    # compute batch index and head index
    b = pid0 // H
    h = pid0 % H

    # compute row and col offsets
    i_offset = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_i = i_offset < N
    mask_j = j_offset < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # compute pointers for left tile
        left_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            i_offset[:, None] * stride_left_i + \
            k_offset[None, :] * stride_left_k

        # load left tile
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # compute pointers for right tile (transposed)
        right_ptrs = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            k_offset[:, None] * stride_right_k + \
            j_offset[None, :] * stride_right_j

        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        i_offset[:, None] * stride_out_i + \
        j_offset[None, :] * stride_out_j

    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x_fp = x.to(torch.float16)

    # Linear projections to hidden dim
    left_proj = torch.nn.functional.linear(x_fp, weights["left_proj.weight"].to(torch.float16))
    right_proj = torch.nn.functional.linear(x_fp, weights["right_proj.weight"].to(torch.float16))

    # Gating projections + sigmoid
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_fp, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_fp, weights["out_gate.weight"].to(torch.float16)))

    if mask is not None and not nomask:
        # mask shape [B,N,N]; unsqueeze(-1)
        mask_fp = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # fuse proj, gate, mask using kernel
        left, right = _apply_proj_gate_mask(left_proj, right_proj, left_gate, right_gate, mask_fp)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # core reduction via Triton
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # shape (B,N,N,H)
    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (B * H, (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)
    left_ptr = left
    right_ptr = right
    out_ptr = out

    # compute strides
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()
    tri_mul_kernel[grid](
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # out now shape (B,N,N,H) fp16
    # Output layernorm
    out = torch.nn.functional.layer_norm(

    out = out * out_gate.to(out.dtype)
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(out.dtype))

"""
TriMul (outgoing) forward pass with Triton.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for compute‑heavy parts.
3. Linear projections (left/right) and gating projections
   (left_gate, right_gate, out_gate) – bias‑free.
4. Optional mask fusion (projection * gate * mask) via a small Triton kernel.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication using a custom Triton kernel
   (one program per (batch, head) tile).  The kernel computes
   out_h = left_h @ right_hᵀ for each hidden slice `h`.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and apply a final linear projection back to `dim`.
7. Cast the result back to the input dtype.

The implementation fuses the mask‑gate‑projection step (when a mask is
present) and the N³ reduction into Triton kernels, while reusing
high‑performance PyTorch for the small linear layers and layer‑norms.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gating and mask (if present)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64
    hidden_dim,      # i64
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask:
       left_out  = left_proj  * left_gate  * mask
       right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    within = offs < total_elems
    # each element corresponds to (b,i,j,hidden); mask has shape (b,i,j)
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=within, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=within, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=within, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=within, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=within, other=1.0)

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=within)
    tl.store(right_out_ptr + offs, right_out, mask=within)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """Launch the Triton kernel that fuses projection, gating and mask.
    All tensors must be FP16 and contiguous.
    Returns (left, right) with shape (B, N, N, H)."""
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,   # fp16 *
    right_ptr,  # fp16 *
    out_ptr,    # fp16 *
    B, N, H,    # i64 scalars
    # strides for left (B,N,N,H)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right (B,N,N,H)
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides for out (B,N,N,H)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid0 = tl.program_id(0)  # batch * head index
    pid1 = tl.program_id(1)  # block row (i)
    pid2 = tl.program_id(2)  # block col (j)

    # recover batch and head indices
    b = pid0 // H
    h = pid0 % H

    # offsets for i and j dimensions
    i_offset = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            i_offset[:, None] * stride_left_i + \
            k_offset[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile (transposed): shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            k_offset[:, None] * stride_right_k + \
            j_offset[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # store the result
    out_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        i_offset[:, None] * stride_out_i + \
        j_offset[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (input channel dimension C)
                * "hidden_dim" (hidden dimension H)
                * "nomask" (optional, bool) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy parts
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Linear projections (FP16)
    # --------------------------------------------------------------
    left_proj = F.linear(
        x_f16, weights["left_proj.weight"].to(torch.float16)
    )  # (B, N, N, hidden_dim)
    right_proj = F.linear(
        x_f16, weights["right_proj.weight"].to(torch.float16)
    )

    # --------------------------------------------------------------
    # 4) Gating projections (sigmoid, FP16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_f16, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x_f16, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x_f16, weights["out_gate.weight"].to(torch.float16))
    )  # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5) Apply mask and fuse with projection & gating
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape: [B, N, N] -> [B, N, N, 1] -> FP16
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6) Core reduction (TriMul) via Triton
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # (B, N, N, H), FP16

    # Block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Strides (in elements)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    # Convert to FP32 for layer norm (more stable)
    out_fp32 = out.to(torch.float32)

    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_fp32 = out_fp32 * out_gate.to(out_fp32.dtype)
    out_fp32 = F.linear(
        out_fp32, weights["to_out.weight"].to(out_fp32.dtype)
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 8) Cast back to original dtype
    # --------------------------------------------------------------
    if out_fp32.dtype != input_tensor.dtype:
        out_fp32 = out_fp32.to(input_tensor.dtype)

    return out_fp32
