"""
TriMul outgoing (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for compute‑heavy parts.
3. Fuse the four linear projections (`left_proj`, `right_proj`,
   `left_gate`, `right_gate`) into a single GEMM and apply sigmoid to the
   gate slices.  An optional mask is broadcasted and applied.
4. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix‑multiply (B·H, N, N) @ (B·H, N, N)ᵀ
   with a custom Triton kernel.
5. Layer‑norm over the hidden dimension, element‑wise multiplication by
   `out_gate`, and a final linear projection back to the original `dim`.
6. Cast the result back to the original dtype.

At least one operation (the reduction) is performed with a Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    batch, M, N, K,                           # dimensions
    stride_ah, stride_am, stride_ak,          # strides for A (batch, M, K)
    stride_bh, stride_bk, stride_bn,          # strides for B (batch, K, N) – B is stored as (batch, N, K) and accessed transposed
    stride_ch, stride_cm, stride_cn,          # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM with the second operand transposed: C = A @ Bᵀ."""
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)

        # pointers for A and Bᵀ
        a_ptrs = (
            a_ptr
            + batch_idx * stride_ah
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + batch_idx * stride_bh
            + offs_k[:, None] * stride_bk   # K dimension
            + offs_n[None, :] * stride_bn   # N dimension
        )

        # out‑of‑bounds masks
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) * (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # write result
    c_ptrs = (
        c_ptr
        + batch_idx * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) – can be all‑ones.
            - weights: dict containing model parameters (float32)
            - config: dict with keys:
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias = weights["norm.bias"].to(dtype)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Fuse the four projections/gates into a single GEMM
    # ------------------------------------------------------------------
    # weights: (out_features, in_features) in fp16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

    # stack them -> shape (4*H, dim)
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # (4*hidden_dim, dim)

    # combined linear
    combined = F.linear(x, combined_weight)  # (B, N, N, 4*H)

    # split
    h = hidden_dim
    left_proj = combined[..., :h]
    left_gate = torch.sigmoid(combined[..., h : 2 * h])
    right_proj = combined[..., 2 * h : 3 * h]
    right_gate = torch.sigmoid(combined[..., 3 * h :])

    # ------------------------------------------------------------------
    # 4) Apply optional mask
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Core reduction (batched GEMM) using Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)  # (B*H, N, N)

    # strides (in elements)
    a_batch_stride, a_m_stride, a_k_stride = left_flat.stride()
    b_batch_stride, b_k_stride, b_n_stride = right_flat.stride()  # note: we will access Bᵀ
    c_batch_stride, c_m_stride, c_n_stride = out_flat.stride()

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        batch=B * H,
        M=N,
        N=N,
        K=N,
        stride_ah=a_batch_stride,
        stride_am=a_m_stride,
        stride_ak=a_k_stride,
        stride_bh=b_batch_stride,
        stride_bk=b_k_stride,   # K‑dim stride (original N dimension)
        stride_bn=b_n_stride,   # N‑dim stride (original N dimension)
        stride_ch=c_batch_stride,
        stride_cm=c_m_stride,
        stride_cn=c_n_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B, N, N, H)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 7) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype:
        out = out.to(dtype)

    return out
