def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_j, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)  # batch*hidden index
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block col

    # compute starting indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # used for k inside loop
    # create mask for out of bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left tile (M,K)
        left_ptrs = left_ptr + pid_b * stride_left_batch + offs_m[:, None] * stride_left_i + k_offset[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile transposed (K,N)
        # right indices: (batch, j, k) -> we need k as row, j as col
        right_ptrs = right_ptr + pid_b * stride_right_batch + k_offset[:, None] * stride_right_k + offs_n[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_ptr + pid_b * stride_out_batch + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # get weights
    ... cast to fp16.

    # 1) norm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)

    # 2) projections
    left_proj = F.linear(x, left_proj_w)  # shape (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # 3) mask and gating
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(left_proj, right_proj, left_gate, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 4) core reduction using Triton kernel
    B, N, _, H = left.shape
    # Reorder to (B, H, N, N) for contiguous layout
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # view as (B*H, N, N)
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)
    out_flat = torch.empty_like(left_flat)

    # compute strides
    stride_left_batch = left_flat.stride(0)
    stride_left_i = left_flat.stride(1)
    stride_left_k = left_flat.stride(2)  # k dimension is last

    stride_right_batch = right_flat.stride(0)
    stride_right_j = right_flat.stride(1)  # j dimension (row)
    stride_right_k = right_flat.stride(2)  # k dimension (col)

    stride_out_batch = out_flat.stride(0)
    stride_out_i = out_flat.stride(1)
    stride_out_j = out_flat.stride(2)

    # Choose block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        stride_left_batch, stride_left_i, stride_left_k,
        stride_right_batch, stride_right_j, stride_right_k,
        stride_out_batch, stride_out_i, stride_out_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    out = F.linear(out, to_out_w)

    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_j, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (M,K)
        left_ptrs = left_ptr + pid_b * stride_left_batch + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right tile transposed (K,N) i.e., we need (K,N) layout

        right_ptrs = right_ptr + pid_b * stride_right_batch + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j

        # cast to float32
        left_f = left.to(tl.float32)
        right_f = right.to(tl.float32)

        acc += tl.dot(left_f, right_f)

    # store
    out_ptrs = out_ptr + pid_b * stride_out_batch + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass – Triton implementation.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (dim).
2. Cast to fp16 for compute‑intensive parts.
3. Linear projections (left/right) and gating projections
   (left_gate, right_gate, out_gate) – bias‑free.
4. Fuse projection, gating and optional mask with a small Triton kernel.
5. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (A @ Bᵀ) with a custom Triton kernel,
   where each (batch, hidden) pair is a separate matrix multiplication.
6. Output layer‑norm, multiply by out_gate and final linear projection.
7. Cast back to the original dtype.

Only the forward pass is needed; gradients are not computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# 1. Kernel: fuse projection, gating and mask
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask:
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # each mask element corresponds to (b,i,j) → one mask per hidden entry
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=valid, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=valid, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m     = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launches the Triton kernel that fuses projection, gating and mask.
    All tensors must be fp16 and contiguous.
    Returns (left, right) each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],  # hidden_dim H
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# 2. Kernel: batched matrix multiplication for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    out_ptr,           # fp16 *
    batch,             # i64 scalar = B*H
    N,                 # i64 scalar
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_j, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute for each batch*hidden slice:
        out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]
    (left: (M,K), right: (N,K) → we load right transposed to (K,N)).
    """
    pid_b = tl.program_id(0)          # batch * hidden index
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block col

    # offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (M, K)
        left_ptrs = left_ptr + pid_b * stride_left_batch \
                    + offs_m[:, None] * stride_left_i \
                    + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile transposed: we need (K, N)
        # right layout is (batch, j, k); we load as (k, j)
        right_ptrs = right_ptr + pid_b * stride_right_batch \
                     + offs_k[:, None] * stride_right_k \
                     + offs_n[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        # accumulate in fp32
        acc += tl.dot(left.to(tl.float32), right.to(tl.float32))

    # store result (fp16)
    out_ptrs = out_ptr + pid_b * stride_out_batch \
               + offs_m[:, None] * stride_out_i \
               + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 3. Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 0) Cast weights to fp16 where possible
    # --------------------------------------------------------------
    def to_fp16(t):
        return t.to(torch.float16)

    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = to_fp16(weights["left_proj.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w = to_fp16(weights["to_out.weight"])

    # --------------------------------------------------------------
    # 1) Input LayerNorm over channel dimension `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Linear projections (fp16)
    # --------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 4) Gating projections (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 5) Apply mask and fuse with projection & gating
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6) Core reduction (TriMul) via Triton batched GEMM
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Rearrange to (B, H, N, N) for contiguous memory, then view as (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)
    out_flat = torch.empty_like(left_flat)

    # Strides (in elements)
    s_lb = left_flat.stride(0)
    s_li = left_flat.stride(1)
    s_lk = left_flat.stride(2)

    s_rb = right_flat.stride(0)
    s_rj = right_flat.stride(1)   # row index (j)
    s_rk = right_flat.stride(2)   # column index (k)

    s_ob = out_flat.stride(0)
    s_oi = out_flat.stride(1)
    s_oj = out_flat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                     # batch * hidden
        triton.cdiv(N, BLOCK_M),  # rows
        triton.cdiv(N, BLOCK_N),  # cols
    )
    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        s_lb, s_li, s_lk,
        s_rb, s_rj, s_rk,
        s_ob, s_oi, s_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 8) Cast back to original dtype if needed
    # --------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
