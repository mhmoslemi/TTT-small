@triton.jit
def _tri_mul_batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lbh, stride_lrow, stride_lcol,
    stride_rbh, stride_rrow, stride_rcol,
    stride_obh, stride_orow, stride_ocol,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_head = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # Compute starting indices
    start_m = pid_m * BLOCK_M
    start_n = pid_n * BLOCK_N

    # Compute offsets for each tile
    # allocate accumulators
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Compute offsets for load
        offset_l = batch_head * stride_lbh + (start_m + tl.arange(0, BLOCK_M)) * stride_lrow + (k * BLOCK_K + tl.arange(0, BLOCK_K)) * stride_lcol
        offset_r = batch_head * stride_rbh + (k * BLOCK_K + tl.arange(0, BLOCK_K)) * stride_rrow + (start_n + tl.arange(0, BLOCK_N)) * stride_rcol

        # Load blocks
        block_l = tl.load(left_ptr + offset_l, mask= (start_m + tl.arange(0, BLOCK_M) < N)[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < N)[None, :], other=0.0).to(tl.float32)
        block_r = tl.load(right_ptr + offset_r, mask= (k * BLOCK_K + tl.arange(0, BLOCK_K) < N)[:, None] & (start_n + tl.arange(0, BLOCK_N) < N)[None, :], other=0.0).to(tl.float32)

        # Multiply-accumulate
        acc += tl.dot(block_l, block_r)  # shape (BLOCK_M, BLOCK_N)

    # Write back
    out_offset = batch_head * stride_obh + (start_m + tl.arange(0, BLOCK_M)) * stride_orow + (start_n + tl.arange(0, BLOCK_N)) * stride_ocol
    mask_o = (start_m + tl.arange(0, BLOCK_M) < N)[:, None] & (start_n + tl.arange(0, BLOCK_N) < N)[None, :]
    tl.store(out_ptr + out_offset, acc.to(tl.float16), mask=mask_o)

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    # Ensure contiguous
    input_tensor = input_tensor.contiguous()
    if mask is not None:
        mask = mask.contiguous()

    # Extract config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm on input
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Cast to fp16
    x = x.to(torch.float16)

    # Convert weight tensors to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections
    left_proj = F.linear(x, left_proj_w)  # shape (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply mask and gating
    if mask is not None and not nomask:
        mask_fp = mask.unsqueeze(-1).to(x.dtype)  # (B,N,N,1)
        left = left_proj * left_gate * mask_fp
        right = right_proj * right_gate * mask_fp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Permute to (B, H, N, N) for batched gemm
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose right for multiplication
    right_perm_T = right_perm.permute(0, 1, 3, 2).contiguous()  # (B, H, N, N) with k,j

    B, H, N, _ = left_perm.shape
    # flatten batch*head
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm_T.view(B * H, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # Get strides
    stride_l0, stride_l1, stride_l2 = left_flat.stride()
    stride_r0, stride_r1, stride_r2 = right_flat.stride()
    stride_o0, stride_o1, stride_o2 = out_flat.stride()

    # Determine block sizes

"""
TriMul outgoing kernel (AlphaFold3) – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑normalize the input tensor (float32) and cast to fp16.
2. Linear projections (left/right) and gating streams are computed with fp16
   weights.  Mask (if present) is fused by element‑wise multiplication.
3. The core O(N³) reduction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is realised as a batched matrix multiplication
        out = left @ rightᵀ
   with a custom Triton GEMM kernel that processes each (batch, head) pair
   as a separate GEMM (block‑wise tiling, fp16→fp32 accumulation).
4. An output LayerNorm, multiplicative out‑gate and a final linear projection
   map back to the original channel dimension.
5. The result is returned in the original input dtype (float32).

The implementation fuses projection, gating and mask in pure PyTorch (cheap
element‑wise ops) and replaces the expensive batched GEMM with a hand‑written
Triton kernel, delivering a large speed‑up over a naïve PyTorch version.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_batched_gemm_kernel(
    left_ptr,          # fp16 * (B*H, N, N)
    right_ptr,         # fp16 * (B*H, N, N)  (already transposed on the last two dims)
    out_ptr,           # fp16 * (B*H, N, N)
    N,                 # sequence length
    stride_l0, stride_l1, stride_l2,   # left strides
    stride_r0, stride_r1, stride_r2,   # right strides
    stride_o0, stride_o1, stride_o2,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM:  out[bh, i, j] = Σₖ left[bh, i, k] * right[bh, k, j]
    Each program processes one (batch*head) slice and a tile (BLOCK_M, BLOCK_N)
    of the output matrix.
    """
    bh = tl.program_id(0)                     # batch*head index
    pid_m = tl.program_id(1)                  # tile row
    pid_n = tl.program_id(2)                  # tile col

    # tile start coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # bounds
        mask_m = offs_m < N
        mask_n = offs_n < N
        mask_k = offs_k < N

        # pointers for the current tile
        left_index  = bh * stride_l0 + offs_m[:, None] * stride_l1 + offs_k[None, :] * stride_l2
        right_index = bh * stride_r0 + offs_k[:, None] * stride_r1 + offs_n[None, :] * stride_r2

        # load tiles (fp16 → fp32)
        a = tl.load(left_ptr + left_index,
                    mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(right_ptr + right_index,
                    mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)

    # write‑back
    out_index = bh * stride_o0 + offs_m[:, None] * stride_o1 + offs_n[None, :] * stride_o2
    mask_o = mask_m[:, None] & mask_n[None, :]
    tl.store(out_ptr + out_index, acc.to(tl.float16), mask=mask_o)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections and gates (fp16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)                     # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)                   # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))      # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))    # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))       # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Apply mask and gating (element‑wise, fp16)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.unsqueeze(-1).to(x.dtype)  # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp
        right = right_proj * right_gate * mask_fp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Core reduction: batched GEMM via Triton
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()       # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()     # (B, H, N, N)

    # Transpose right on the last two dimensions => (B, H, N, N) with (k, j)
    right_perm_T = right_perm.transpose(-2, -1).contiguous()

    B, H, N, _ = left_perm.shape
    # flatten batch*head dimension
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm_T.view(B * H, N, N)

    # allocate output buffer
    out_flat = torch.empty_like(left_flat)

    # strides (in elements)
    stride_l0, stride_l1, stride_l2 = left_flat.stride()
    stride_r0, stride_r1, stride_r2 = right_flat.stride()
    stride_o0, stride_o1, stride_o2 = out_flat.stride()

    # tile sizes – chosen to give good occupancy on B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # launch grid: (batch*head, tiles_m, tiles_n)
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_l0, stride_l1, stride_l2,
        stride_r0, stride_r1, stride_r2,
        stride_o0, stride_o1, stride_o2,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)          # (B, N, N, C)

    # ------------------------------------------------------------------
    # 7) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
