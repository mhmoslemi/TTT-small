```python
def batched_gemm_kernel(A_ptr, B_ptr, C_ptr,

                        stride_bm, stride_bn, stride_bb,
                        stride_cm, stride_cn, stride_cb,

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

    a_ptrs = A_ptr + pid_h * stride_ab + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak

        a = tl.load(A_ptr + pid_h * stride_ab + offs_m[:, None] * stride_am + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak,
```

```
def matmul_kernel(A_ptr, B_ptr, C_ptr,
                 M, N, K,
                 stride_a_b, stride_a_m, stride_a_k,
                 stride_b_b, stride_b_k, stride_b_n,
                 stride_c_b, stride_c_m, stride_c_n,
                 BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    pid_b = tl.program_id(axis=2)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # create masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for A, B (with offset for bh)
    a_ptrs = A_ptr + pid_b * stride_a_b + offs_m[:, None] * stride_a_m + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_ptrs = B_ptr + pid_b * stride_b_b + (tl.arange(0, BLOCK_K)[:, None]) * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # Compute offsets for this iteration
        a = tl.load(A_ptr + pid_b * stride_a_b + offs_m[:, None] * stride_a_m + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_a_k,
                    mask=mask_m[:, None] & ((k_offset + tl.arange(0, BLOCK_K)) < K), other=0.0)
        b = tl.load(B_ptr + pid_b * stride_b_b + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_b_k + offs_n[None, :] * stride_b_n,
                    mask=((k_offset + tl.arange(0, BLOCK_K)) < K)[:, None] & mask_n[None, :], other=0.0)

        # Convert to float16
        # ensure dtype
        a = a.to(tl.float16) if a.dtype != tl.float16 else a
        b = b.to(tl.float16) if b.dtype != tl.float16 else b

        acc += tl.dot(a, b)

    # Write output
    c = acc
    c_ptrs = C_ptr + pid_b * stride_c_b + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
```

```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # layernorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # mask (if required)
    if not config.get("nomask", True):
        mask_exp = mask[..., None].type_as(left)
        left = left * mask_exp
        right = right * mask_exp
    # gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()
    # apply gates
    left = left * left_gate
    right = right * right_gate
    # Prepare for GEMM
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()
    right_T_f16 = right_f16.permute(0,2,1,3).contiguous()
    BH = B * hidden_dim
    left_ = left_f16.view(BH, N, N)
    right_T_ = right_T_f16.view(BH, N, N)
    out_f16 = torch.empty_like(left_)
    # get strides
    a_stride_b = left_.stride(0)
    a_stride_m = left_.stride(1)
    a_stride_k = left_.stride(2)
    b_stride_b = right_T_.stride(0)
    b_stride_k = right_T_.stride(1)
    b_stride_n = right_T_.stride(2)
    c_stride_b = out_f16.stride(0)
    c_stride_m = out_f16.stride(1)
    c_stride_n = out_f16.stride(2)
    # launch kernel
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), BH)
    matmul_kernel[grid](
        left_, right_T_, out_f16,
        N, N, N,
        a_stride_b, a_stride_m, a_stride_k,
        b_stride_b, b_stride_k, b_stride_n,
        c_stride_b, c_stride_m, c_stride_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # reshape output
    out = out_f16.to(torch.float32).view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()
    # final layernorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # multiply by out_gate
    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
```

```
@triton.jit
def matmul_kernel(
    A, B, C,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    pid_b = tl.program_id(axis=2)
    # Compute start offsets for M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer offsets for A and B
    a_ptrs = A + pid_b * stride_Ab + offs_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptrs = B + pid_b * stride_Bb + (tl.arange(0, BLOCK_K)[:, None]) * stride_Bk + offs_n[None, :] * stride_Bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        # compute offsets for this tile
        a = tl.load(A + pid_b * stride_Ab + offs_m[:, None] * stride_Am + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_Ak,
                    mask = mask_m[:, None] & ((k_offset + tl.arange(0, BLOCK_K)) < K), other=0.0)
        b = tl.load(B + pid_b * stride_Bb + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_Bk + offs_n[None, :] * stride_Bn,
                    mask = ((k_offset + tl.arange(0, BLOCK_K)) < K)[:, None] & mask_n[None, :], other=0.0)

        # Convert to fp16 if necessary
        a = a.to(tl.float16)
        b = b.to(tl.float16)

        acc += tl.dot(a, b)
    # Write back
    c_ptrs = C + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask = mask_m[:, None] & mask_n[None, :])
```

import torch
import triton
import triton.language as tl

@triton.jit
def matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(axis=0)  # i tile
    pid_n = tl.program_id(axis=1)  # j tile
    pid_b = tl.program_id(axis=2)  # batch * head id

    # Compute tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        # Load A block
        a = tl.load(
            A_ptr
            + pid_b * stride_Ab
            + offs_m[:, None] * stride_Am
            + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_Ak,
            mask=mask_m[:, None] & ((k_offset + tl.arange(0, BLOCK_K)) < K),
            other=0.0,
        )
        # Load B block (KxN)
        b = tl.load(
            B_ptr
            + pid_b * stride_Bb
            + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_Bk
            + offs_n[None, :] * stride_Bn,
            mask=((k_offset + tl.arange(0, BLOCK_K)) < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Convert to fp16 if needed (kernel works with fp16 inputs)
        a = a.to(tl.float16)
        b = b.to(tl.float16)

        acc += tl.dot(a, b)

    # Write back
    c = acc
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    The kernel handles the cubic‑complex reduction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    using a batched GEMM written in Triton. All other
    linear/LN/gating steps are performed in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---- LayerNorm on the input -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---- Linear projections ------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---- Optional mask -----------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [bs, seq_len, seq_len] -> [bs, seq_len, seq_len, 1]
        mask_exp = mask.unsqueeze(-1).type_as(left)
        left = left * mask_exp
        right = right * mask_exp

    # ---- Gates -------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # ---- Apply gates -------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ---- Prepare for Triton GEMM -------------------------------------------------
    # Convert to fp16 for the heavy reduction
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # Transpose the second pair of axes in `right` (j,k) -> (k,j)
    right_t = right.permute(0, 2, 1, 3).contiguous()

    # Move the hidden dimension to the second axis and fuse batch*head
    left_ = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, seq_len, seq_len)
    right_ = right_t.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, seq_len, seq_len)

    # Output buffer (fp16) – same fused shape
    out_fp16 = torch.empty_like(left_)

    # Strides for the kernel (all tensors are contiguous)
    a_stride_b = left_.stride(0)
    a_stride_m = left_.stride(1)
    a_stride_k = left_.stride(2)

    b_stride_b = right_.stride(0)
    b_stride_k = right_.stride(1)
    b_stride_n = right_.stride(2)

    c_stride_b = out_fp16.stride(0)
    c_stride_m = out_fp16.stride(1)
    c_stride_n = out_fp16.stride(2)

    # Block sizes – tuned for typical A100/B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
        bs * hidden_dim,
    )

    matmul_kernel[grid](
        left_,
        right_,
        out_fp16,
        seq_len,
        seq_len,
        seq_len,
        a_stride_b,
        a_stride_m,
        a_stride_k,
        b_stride_b,
        b_stride_k,
        b_stride_n,
        c_stride_b,
        c_stride_m,
        c_stride_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ---- Reshape back to [bs, seq_len, seq_len, hidden_dim] -----------------------
    out = out_fp16.to(torch.float32).view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # ---- Final LayerNorm, gating and projection ----------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul outgoing kernel (forward only).

The heavy O(N³) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched GEMM in Triton:
    left (B,N,N,D) → (B·D,N,N)
    right (B,N,N,D) → transpose j/k → (B·D,N,N)
    out = left @ rightᵀ   (per hidden channel)

All other operations (LayerNorms, linear projections, gates, final
normalisation and projection) are performed with PyTorch.  The kernel
operates on fp16 tensors for speed and casts the result back to fp32.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def matmul_kernel(
    A_ptr,
    B_ptr,
    C_ptr,
    M,
    N,
    K,
    stride_Ab,
    stride_Am,
    stride_Ak,
    stride_Bb,
    stride_Bk,
    stride_Bn,
    stride_Cb,
    stride_Cm,
    stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched fp16 GEMM: C = A @ B where A,B,C are (batch, M, K)/(K,N)/(M,N)."""
    pid_m = tl.program_id(axis=0)  # block row
    pid_n = tl.program_id(axis=1)  # block col
    pid_b = tl.program_id(axis=2)  # batch*head index

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K

        # Load A block (M×K)
        a = tl.load(
            A_ptr
            + pid_b * stride_Ab
            + offs_m[:, None] * stride_Am
            + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_Ak,
            mask=mask_m[:, None] & ((k_offset + tl.arange(0, BLOCK_K)) < K),
            other=0.0,
        )
        # Load B block (K×N)
        b = tl.load(
            B_ptr
            + pid_b * stride_Bb
            + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_Bk
            + offs_n[None, :] * stride_Bn,
            mask=((k_offset + tl.arange(0, BLOCK_K)) < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        # a,b are fp16; tl.dot promotes to fp32 accumulation
        acc += tl.dot(a, b)

    # Store the result
    c_ptr = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(c_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (may be ignored)
        - weights      : dict of torch.Tensors containing all linear / LN weights
        - config       : dict containing at least "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------ LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------- Linear proj
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------- Optional mask
    if not config.get("nomask", True):
        # mask: [B, N, N] → [B, N, N, 1] for broadcasting
        mask_exp = mask.unsqueeze(-1).type_as(left)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------- Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------- Prepare for Triton
    # fp16 for the cubic reduction
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # Transpose right's (j,k) → (k,j) so the reduction becomes a matrix multiplication
    right_t = right.permute(0, 2, 1, 3).contiguous()  # (B, N, N, D)

    # Move hidden dimension next to batch and make the tensor contiguous,
    # then fuse batch & hidden dimension: (B*D, N, N)
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_ = right_t.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (fp16) with the fused leading dimension
    out_fp16 = torch.empty_like(left_)

    # ------------------------------------------------------------------- Strides
    a_stride_b = left_.stride(0)
    a_stride_m = left_.stride(1)
    a_stride_k = left_.stride(2)

    b_stride_b = right_.stride(0)
    b_stride_k = right_.stride(1)
    b_stride_n = right_.stride(2)

    c_stride_b = out_fp16.stride(0)
    c_stride_m = out_fp16.stride(1)
    c_stride_n = out_fp16.stride(2)

    # --------------------------------------------------------------- Launch Triton
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    matmul_kernel[grid](
        left_,
        right_,
        out_fp16,
        N,
        N,
        N,
        a_stride_b,
        a_stride_m,
        a_stride_k,
        b_stride_b,
        b_stride_k,
        b_stride_n,
        c_stride_b,
        c_stride_m,
        c_stride_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------- Reshape back
    out = (
        out_fp16.to(torch.float32)
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # shape [B, N, N, hidden_dim]

    # ---------------------------------------------------------- Final LN, gate & linear
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
