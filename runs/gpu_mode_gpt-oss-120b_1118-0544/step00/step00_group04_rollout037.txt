@triton.jit
def bmm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    batch = tl.program_id(1)
    # compute block coordinates
    # dimensions: pid = i_grid * grid_n + j_grid
    # need grid dimensions: grid_m = tl.cdiv(M, BLOCK_M)
    # grid_n = tl.cdiv(N, BLOCK_N)
    grid_m = tl.cdiv(M, BLOCK_M)
    grid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // grid_n
    pid_n = pid % grid_n

    # Offsets for block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Use mask to limit out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers
    a_ptrs = a_ptr + batch * batch_stride_a + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + batch * batch_stride_b + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        a = tl.load(a_ptrs + k_offset * stride_ak, mask=mask_m[:, None] & (k_offset + tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs + k_offset * stride_bk, mask=(k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
        # cast to float32
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)
        # increment pointers maybe not needed as we recompute each time using offsets
    # store result
    c_ptrs = c_ptr + batch * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,

    batch = tl.program_id(1)
    # compute grid
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    pid_n = pid % num_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    # layernorm
    eps = 1e-5

    # Norm
    x = input_tensor
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    # compute mean,var
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    if w_norm is not None:
        x_norm = x_norm * w_norm + b_norm
    # linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # mask
    if not nomask:
        # mask shape [B,N,N], unsqueeze for broadcast
        mask_exp = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp
    # gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    )

    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,hidden)
    # Convert to float32 for norm
    out = out.to(torch.float32)

"""
TriMul outgoing forward (AlphaFold3).

The implementation follows the reference PyTorch code for all cheap ops
(normalization, linear projections, gating) and uses a custom Triton kernel
to accelerate the heavy O(N³) reduction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

The reduction is rewritten as a batched matrix multiplication
    left_reshape @ right_reshapeᵀ
where each hidden channel d is treated as an independent batch.
The Triton kernel computes C = A @ B for many batches in a single launch.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _bmm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ B.
    A: (batch, M, K)   B: (batch, K, N)   C: (batch, M, N)
    All tensors are assumed to be contiguous in row‑major order.
    """
    pid = tl.program_id(0)          # tile id   (covers M×N)
    batch_id = tl.program_id(1)     # which batch / hidden channel

    # number of blocks in each dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_blocks_n
    pid_n = pid % num_blocks_n

    # offsets within the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for the current (batch, block)
    a_base = a_ptr + batch_id * batch_stride_a + offs_m[:, None] * stride_am
    b_base = b_ptr + batch_id * batch_stride_b + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = tl.minimum(k + tl.arange(0, BLOCK_K), K - 1)
        cur_k_len = tl.minimum(BLOCK_K, K - k)

        # load a block of A (size BLOCK_M × cur_k_len)
        a = tl.load(
            a_base + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & (cur_k < K),
            other=0.0,
        )
        # load a block of B (size cur_k_len × BLOCK_N)
        b = tl.load(
            b_base + cur_k[:, None] * stride_bk,
            mask=(cur_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        # compute contribution of this k‑slice
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # write back the result
    c = c_ptr + batch_id * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """Forward pass of the outgoing TriMul operator using a Triton kernel.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask:        torch.Tensor [B, N, N] (may be ignored if config["nomask"] is True)
            - weights:     dict of model parameters.
            - config:      dict with at least "dim", "hidden_dim", and optional "nomask".

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (over the last dimension)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Batched GEMM for the O(N³) reduction
    #    Convert (B, N, N, hidden) -> (B*hidden, N, N) and compute
    #    C = A @ Bᵀ where Bᵀ is obtained by swapping the two N axes.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B = batch size, N = sequence length

    # bring hidden dim to the batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, hidden, N, N)

    # A = left_perm reshaped to (B*hidden, N, N)
    A = left_perm.view(B * hidden_dim, N, N)
    # Bᵀ = right_perm with last two axes swapped, then reshaped
    B_mat = right_perm.permute(0, 1, 3, 2).contiguous().view(B * hidden_dim, N, N)

    # Output tensor
    C = torch.empty_like(A, dtype=torch.float32)  # (B*hidden, N, N)

    # Tile sizes – tuned for typical B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (tiles over M×N, batches)
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_m * grid_n, B * hidden_dim)

    # Strides for A, B, C
    a_batch_stride, a_stride_m, a_stride_k = A.stride()
    b_batch_stride, b_stride_k, b_stride_n = B_mat.stride()
    c_batch_stride, c_stride_m, c_stride_n = C.stride()

    # Launch the Triton kernel
    _bmm_kernel[grid](
        A,
        B_mat,
        C,
        a_batch_stride,
        a_stride_m,
        a_stride_k,
        b_batch_stride,
        b_stride_k,
        b_stride_n,
        c_batch_stride,
        c_stride_m,
        c_stride_n,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden)
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 6) LayerNorm on the hidden dimension, gating, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # broadcasted over hidden dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
