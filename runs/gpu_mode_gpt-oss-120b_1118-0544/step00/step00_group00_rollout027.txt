@triton.jit
def mask_gate_kernel(L_ptr, R_ptr, M_ptr, LG_ptr, RG_ptr, L_out_ptr, R_out_ptr,
                     B, N, H,
                     stride_L_b, stride_L_i, stride_L_j, stride_L_h,
                     stride_R_b, stride_R_i, stride_R_j, stride_R_h,
                     stride_M_b, stride_M_i, stride_M_j, stride_M_h,
                     stride_LG_b, stride_LG_i, stride_LG_j, stride_LG_h,
                     stride_RG_b, stride_RG_i, stride_RG_j, stride_RG_h,
                     BLOCK: tl.constexpr):
    idx = tl.program_id(0)
    offs = idx * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # compute coordinates
    b = (offs // (N * N * H))
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem = rem % (N * H)
    j = rem // H
    h = rem % H

    # compute flat offset for each ptr
    L_offset = b * stride_L_b + i * stride_L_i + j * stride_L_j + h * stride_L_h
    R_offset = b * stride_R_b + i * stride_R_i + j * stride_R_j + h * stride_R_h
    M_offset = b * stride_M_b + i * stride_M_i + j * stride_M_j + h * stride_M_h
    LG_offset = b * stride_LG_b + i * stride_LG_i + j * stride_LG_j + h * stride_LG_h
    RG_offset = b * stride_RG_b + i * stride_RG_i + j * stride_RG_j + h * stride_RG_h

    # load values
    L = tl.load(L_ptr + L_offset, mask=mask, other=0.0).to(tl.float32)
    R = tl.load(R_ptr + R_offset, mask=mask, other=0.0).to(tl.float32)
    Mval = tl.load(M_ptr + M_offset, mask=mask, other=0.0).to(tl.float32) if M_ptr != 0 else 1.0
    LG = tl.load(LG_ptr + LG_offset, mask=mask, other=0.0).to(tl.float32)
    RG = tl.load(RG_ptr + RG_offset, mask=mask, other=0.0).to(tl.float32)

    L_out = L * LG * Mval
    R_out = R * RG * Mval

    tl.store(L_out_ptr + L_offset, L_out.to(tl.float16), mask=mask)
    tl.store(R_out_ptr + R_offset, R_out.to(tl.float16), mask=mask)

@triton.jit
def mask_gate_kernel(L_ptr, R_ptr, M_mask_ptr,
                     LG_ptr, RG_ptr,
                     L_out_ptr, R_out_ptr,
                     B, N, H,
                     stride_L_b, stride_L_i, stride_L_j, stride_L_h,
                     stride_R_b, stride_R_i, stride_R_j, stride_R_h,
                     stride_M_b, stride_M_i, stride_M_j,
                     stride_LG_b, stride_LG_i, stride_LG_j, stride_LG_h,
                     stride_RG_b, stride_RG_i, stride_RG_j, stride_RG_h,
                     BLOCK: tl.constexpr):
    idx = tl.program_id(0)
    offs = idx * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # compute coords
    tmp = offs
    b = tmp // (N * N * H)
    tmp = tmp % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    # offsets
    L_off = b * stride_L_b + i * stride_L_i + j * stride_L_j + h * stride_L_h
    R_off = b * stride_R_b + i * stride_R_i + j * stride_R_j + h * stride_R_h
    LG_off = b * stride_LG_b + i * stride_LG_i + j * stride_LG_j + h * stride_LG_h
    RG_off = b * stride_RG_b + i * stride_RG_i + j * stride_RG_j + h * stride_RG_h
    M_off = b * stride_M_b + i * stride_M_i + j * stride_M_j  # ignore h

    L_val = tl.load(L_ptr + L_off, mask=mask, other=0.0).to(tl.float32)
    R_val = tl.load(R_ptr + R_off, mask=mask, other=0.0).to(tl.float32)
    LG_val = tl.load(LG_ptr + LG_off, mask=mask, other=0.0).to(tl.float32)
    RG_val = tl.load(RG_ptr + RG_off, mask=mask, other=0.0).to(tl.float32)
    mask_val = tl.load(M_mask_ptr + M_off, mask=mask, other=1.0).to(tl.float32)

    L_out = L_val * LG_val * mask_val
    R_out = R_val * RG_val * mask_val

    tl.store(L_out_ptr + L_off, L_out.to(tl.float16), mask=mask)
    tl.store(R_out_ptr + R_off, R_out.to(tl.float16), mask=mask)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # Ensure dtype float32 as input maybe float32
    x = input_tensor
    # layernorm
    eps = 1e-5
    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Cast to float16 for heavy ops
    x_norm = x_norm.to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"].to(torch.float16))
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"].to(torch.float16)))

    # Mask handling
    if config.get("nomask", True):
        # ignore mask, just use ones
        mask = torch.ones((x.shape[0], x.shape[1], x.shape[2]), dtype=torch.float16, device=device)
    else:
        mask = mask_tensor.to(torch.float16)
    # broadcast mask
    mask = mask.unsqueeze(-1)  # shape (B,N,N,1)

    # Apply mask and gates using Triton kernel
    left, right = apply_mask_gate_triton(left, right, mask, left_gate, right_gate)

    # Compute pairwise product via batched matmul
    B, N, _, H = left.shape
    # reshape for batched matmul
    L = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H, N, N)
    R = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H, N, N)
    # matrix multiplication L @ R^T
    out = torch.bmm(L, R.transpose(1, 2))
    out = out.reshape(B, H, N, N).permute(0, 2, 3, 1)   # (B,N,N,H)

    # Output layernorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply out_gate scaling
    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))
    # Cast to float32 for output
    out = out.to(torch.float32)

    return out

"""
TriMul (outgoing) forward pass implemented with mixed‑precision torch ops and a
small Triton kernel for the element‑wise mask & gate fusion.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor.
2. Linear projections (left/right) and gating projections are computed with
   half‑precision weights (torch.nn.functional.linear + sigmoid).
3. A Triton kernel fuses:   left = left * mask * left_gate,
                            right = right * mask * right_gate.
   This avoids extra memory passes in Python.
4. For each hidden channel (batch * hidden_dim) we perform a batched matrix
   multiplication  L @ Rᵀ  where L/R are (N, N) matrices.  This implements the
   einsum `... i k d, ... j k d -> ... i j d` efficiently using cuBLAS.
5. Apply a second layer‑norm over the hidden dimension, scale by the output
   gate and a final linear projection back to `dim`.
6. The result is returned as float32 tensor.

The heavy O(N³) reduction is handled by torch.bmm (highly optimized), while the
mask‑gate fusion demonstrates the required Triton usage.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mask_gate_kernel(
    L_ptr, R_ptr, M_ptr,
    LG_ptr, RG_ptr,
    L_out_ptr, R_out_ptr,
    B, N, H,
    stride_L_b, stride_L_i, stride_L_j, stride_L_h,
    stride_R_b, stride_R_i, stride_R_j, stride_R_h,
    stride_M_b, stride_M_i, stride_M_j,
    stride_LG_b, stride_LG_i, stride_LG_j, stride_LG_h,
    stride_RG_b, stride_RG_i, stride_RG_j, stride_RG_h,
    BLOCK: tl.constexpr,
):
    """Fuse mask and per‑element gates into left/right tensors."""
    idx = tl.program_id(0)
    offs = idx * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # Decode linear index into (b,i,j,h)
    tmp = offs
    b = tmp // (N * N * H)
    tmp = tmp % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    # Compute raw offsets for each tensor
    L_off = b * stride_L_b + i * stride_L_i + j * stride_L_j + h * stride_L_h
    R_off = b * stride_R_b + i * stride_R_i + j * stride_R_j + h * stride_R_h
    LG_off = b * stride_LG_b + i * stride_LG_i + j * stride_LG_j + h * stride_LG_h
    RG_off = b * stride_RG_b + i * stride_RG_i + j * stride_RG_j + h * stride_RG_h
    # mask has no hidden dimension → broadcast over h
    M_off = b * stride_M_b + i * stride_M_i + j * stride_M_j

    # Load values (float16 → float32 for arithmetic)
    L = tl.load(L_ptr + L_off, mask=mask, other=0.0).to(tl.float32)
    R = tl.load(R_ptr + R_off, mask=mask, other=0.0).to(tl.float32)
    LG = tl.load(LG_ptr + LG_off, mask=mask, other=0.0).to(tl.float32)
    RG = tl.load(RG_ptr + RG_off, mask=mask, other=0.0).to(tl.float32)
    M = tl.load(M_ptr + M_off, mask=mask, other=1.0).to(tl.float32)

    # Fuse mask and gates
    L_out = L * LG * M
    R_out = R * RG * M

    # Store back as float16
    tl.store(L_out_ptr + L_off, L_out.to(tl.float16), mask=mask)
    tl.store(R_out_ptr + R_off, R_out.to(tl.float16), mask=mask)


def _apply_mask_gate_triton(left, right, mask, left_gate, right_gate):
    """Run the Triton kernel that fuses mask & gates."""
    B, N, _, H = left.shape
    # Allocate output tensors
    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    BLOCK = 1024
    total = B * N * N * H
    grid = ( (total + BLOCK - 1) // BLOCK, )

    # Strides (in elements)
    sL = left.stride()
    sR = right.stride()
    sLG = left_gate.stride()
    sRG = right_gate.stride()
    sM = mask.stride()      # mask shape (B,N,N,1) → we use first three strides only

    mask_gate_kernel[grid](
        left, right, mask,
        left_gate, right_gate,
        left_out, right_out,
        B, N, H,
        sL[0], sL[1], sL[2], sL[3],
        sR[0], sR[1], sR[2], sR[3],
        sM[0], sM[1], sM[2],
        sLG[0], sLG[1], sLG[2], sLG[3],
        sRG[0], sRG[1], sRG[2], sRG[3],
        BLOCK=BLOCK,
    )
    return left_out, right_out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool/int/float)
            - weights_dict: mapping from weight names to tensors (float32)
            - config_dict: contains 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (keep in FP32 for numerical stability)
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)          # cast to FP16 for the heavy path

    # -----------------------------------------------------------------
    # 2) Linear projections and gates (all in FP16)
    # -----------------------------------------------------------------
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left = F.linear(x_norm, w_left)                # (B,N,N,hidden_dim)
    right = F.linear(x_norm, w_right)

    left_gate = torch.sigmoid(F.linear(x_norm, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_norm, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_norm, w_out_gate))

    # -----------------------------------------------------------------
    # 3) Apply mask and gates using a Triton kernel
    # -----------------------------------------------------------------
    if config.get("nomask", True):
        mask = torch.ones((input_tensor.shape[0],
                           input_tensor.shape[1],
                           input_tensor.shape[2]), dtype=torch.float16,
                          device=device)
    else:
        mask = mask_tensor.to(torch.float16)

    mask = mask.unsqueeze(-1)      # shape (B,N,N,1) for broadcasting
    left, right = _apply_mask_gate_triton(left, right, mask, left_gate, right_gate)

    # -----------------------------------------------------------------
    # 4) Core pairwise multiplication: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    L = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    R = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)

    out = torch.bmm(L, R.transpose(1, 2))                         # (B*H, N, N)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1)                # (B, N, N, H)

    # -----------------------------------------------------------------
    # 5) Output LayerNorm, gate and final linear projection
    # -----------------------------------------------------------------
    w_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_out_norm = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_out_norm,
        bias=b_out_norm,
        eps=eps,
    )
    out = out * out_gate

    w_to_out = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, w_to_out)            # (B,N,N,dim) in FP16
    out = out.to(torch.float32)              # cast back to FP32 for the API contract

    return out
