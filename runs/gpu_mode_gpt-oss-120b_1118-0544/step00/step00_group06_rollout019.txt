    stride_rbh, stride_rk, stride_rj,

```

We'll compute offsets:

- block_i = tl.program_id(0)
- block_j = tl.program_id(1)
- batch_idx = tl.program_id(2)

i_offsets = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
j_offsets = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

Now for k loop we tile (BLOCK_K). Initialize accumulator accumulator = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

Loop over k_tile in range(0, tl.cdiv(N, BLOCK_K)):
   k_offsets = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
   # mask out-of-bounds
   mask_k = k_offsets < N

   # load left tile: shape BLOCK_M x BLOCK_K
   left_ptrs = left_ptr + batch_idx * stride_lbh + i_offsets[:, None] * stride_li + k_offsets[None, :] * stride_lk
   left_tile = tl.load(left_ptrs, mask= (i_offsets[:, None] < N) & mask_k[None, :], other=0.0).to(tl.float32)

   # load right tile: shape BLOCK_K x BLOCK_N (since right_T is k x j)

   # accumulate
   accumulator += tl.dot(left_tile, right_tile)
```

```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_lbh, stride_li, stride_lk,
    stride_rbh, stride_rk, stride_rj,
    stride_obh, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # compute offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left tile [M,K]
        left_ptrs = left_ptr + pid_bh * stride_lbh + i_offset[:, None] * stride_li + k_offset[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right tile [K,N] (right_T is K x N)
        right_ptrs = right_ptr + pid_bh * stride_rbh + k_offset[:, None] * stride_rk + j_offset[None, :] * stride_rj
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_ptr + pid_bh * stride_obh + i_offset[:, None] * stride_oi + j_offset[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

def custom_kernel(data):
    x, mask, weights, config = data
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    # LAYER NORM
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Linear projections for left/right
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()
    # Apply mask if present and not nomask (mask may be None)
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp
    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # cast to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # reshape for kernel
    B, N, _, H = left.shape
    batch_hidden = B * H

    left_perm = left.permute(0, 3, 1, 2).contiguous()

    right_perm = right.permute(0, 3, 2, 1).contiguous()
    right_flat = right_perm.view(batch_hidden, N, N)

    # Allocate output
    out_flat = torch.empty((batch_hidden, N, N), dtype=torch.float16, device=x.device)

    # strides in elements
    stride_lbh, stride_li, stride_lk = left_flat.stride()
    stride_rbh, stride_rk, stride_rj = right_flat.stride()
    stride_obh, stride_oi, stride_oj = out_flat.stride()

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), batch_hidden)
    # call kernel
    tri_mul_kernel[grid](
        left_flat, right_flat, out_flat,
        N,
        stride_lbh, stride_li, stride_lk,
        stride_rbh, stride_rk, stride_rj,
        stride_obh, stride_oi, stride_oj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back to (B,N,N,H)
    out_hidden = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape B,N,N,H

    out_hidden = out_hidden.to(torch.float32)

        bias=weights["to_out_norm.bias"]
    )

    # final linear
    out = torch.nn.functional.linear(out_hidden, weights["to_out.weight"])
    return out

"""
tri_mul Kernel
----------------
Implements the forward pass of the “outgoing” TriMul module from AlphaFold3.
The heavy O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is rewritten as a batched matrix multiplication:
    left  : (B, H, N, N)   → shape (B*H, N, N)  with rows = i, cols = k
    right : (B, H, N, N)   → transpose to (B, H, N, N) with rows = k, cols = j
    out   = left @ right_T  → (B*H, N, N), later reshaped to (B,N,N,H)

The contraction is executed in a custom Triton kernel using tiling on the
sequence dimensions (BLOCK_M, BLOCK_N) and the reduction dimension k
(BLOCK_K).  The rest of the TriMul pipeline (LayerNorms, linear
projections, gating, final Linear) is performed with PyTorch for
convenience.  The kernel works with fp16 inputs and accumulates in fp32
to retain numeric stability.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def tri_mul_kernel(
    left_ptr,          # *float16   [batch_hidden, N, N]
    right_ptr,         # *float16   [batch_hidden, N, N]  (k, j) layout
    out_ptr,           # *float16   [batch_hidden, N, N]
    N: tl.constexpr,   # sequence length
    stride_lbh, stride_li, stride_lk,
    stride_rbh, stride_rk, stride_rj,
    stride_obh, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_i = tl.program_id(0)               # tile row
    pid_j = tl.program_id(1)               # tile col
    pid_bh = tl.program_id(2)              # batch * hidden idx

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for start_k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = start_k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load left[i, k] ----
        left_ptrs = (
            left_ptr
            + pid_bh * stride_lbh
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ---- load right_T[k, j] ---- (right already transposed)
        right_ptrs = (
            right_ptr
            + pid_bh * stride_rbh
            + offs_k[:, None] * stride_rk
            + offs_j[None, :] * stride_rj
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # ---- accumulate dot product ----
        acc += tl.dot(left, right)

    # ---- store result ----
    out_ptrs = (
        out_ptr
        + pid_bh * stride_obh
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the TriMul module using a Triton kernel for the
    heavy contraction.
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ---------- LayerNorm on input ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Gating ----------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()   # [B,N,N,H]

    # ---------- Apply mask (if any) ----------
    if mask is not None:
        # mask is [B,N,N]; broadcast on hidden dim
        mask = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Apply element‑wise gates ----------
    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to fp16 for kernel ----------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape
    batch_hidden = B * H

    # ---- reshape for batched matmul (B*H, N, N) ----
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    left_flat = left_perm.view(batch_hidden, N, N)

    # right needs to be transposed on the last two dims: (k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N) with (k, j)
    right_flat = right_perm.view(batch_hidden, N, N)

    # allocate output buffer
    out_flat = torch.empty((batch_hidden, N, N), dtype=torch.float16, device=device)

    # strides in elements
    stride_lbh, stride_li, stride_lk = left_flat.stride()
    stride_rbh, stride_rk, stride_rj = right_flat.stride()
    stride_obh, stride_oi, stride_oj = out_flat.stride()

    # launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),          # i tiles
        triton.cdiv(N, BLOCK_N),          # j tiles
        batch_hidden,                     # batch*hidden
    )
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_lbh, stride_li, stride_lk,
        stride_rbh, stride_rk, stride_rj,
        stride_obh, stride_oi, stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---- reshape back to (B, N, N, H) ----
    out_hidden = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]
    out_hidden = out_hidden.to(torch.float32)  # needed for LayerNorm

    # ---------- to_out_norm ----------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ---------- Apply out gate ----------
    out_hidden = out_hidden * out_gate

    # ---------- Final linear ----------
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
