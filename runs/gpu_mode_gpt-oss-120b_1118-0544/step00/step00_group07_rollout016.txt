@triton.jit
def apply_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_c,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_c,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_C: tl.constexpr
):
    # get block ids
    pid_i = tl.program_id(0)  # block in i dimension
    pid_j = tl.program_id(1)  # block in j dimension
    pid_c = tl.program_id(2)  # block in c dimension (includes batch)
    # Compute batch index and c block
    num_c_blocks = tl.cdiv(H, BLOCK_C)
    b = pid_c // num_c_blocks
    cid = pid_c % num_c_blocks
    # compute start indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    c0 = cid * BLOCK_C

    # offsets within block
    off_i = i0 + tl.arange(0, BLOCK_M)
    off_j = j0 + tl.arange(0, BLOCK_N)
    off_c = c0 + tl.arange(0, BLOCK_C)

    # create masks for out-of-range
    mask_i = off_i < N
    mask_j = off_j < N
    mask_c = off_c < H
    mask_ij = mask_i[:, None] & mask_j[None, :]
    # final mask for 3d: combine with c dimension
    mask_ijk = mask_ij[:, :, None] & mask_c[None, None, :]

    # calculate offsets for proj and gate
    # shape (BLOCK_M, BLOCK_N, BLOCK_C)
    proj_off = (off_i[:, None, None] * stride_proj_i
                + off_j[None, :, None] * stride_proj_j
                + off_c[None, None, :] * stride_proj_c
                + b * stride_proj_b)
    gate_off = (off_i[:, None, None] * stride_gate_i
                + off_j[None, :, None] * stride_gate_j
                + off_c[None, None, :] * stride_gate_c
                + b * stride_gate_b)
    mask_off = (off_i[:, None] * stride_mask_i
                + off_j[None, :] * stride_mask_j
                + b * stride_mask_b)

    # load data
    proj = tl.load(proj_ptr + proj_off, mask=mask_ijk, other=0.0)
    gate = tl.load(gate_ptr + gate_off, mask=mask_ijk, other=0.0)
    mask_val = tl.load(mask_ptr + mask_off, mask=mask_ij, other=0.0)

    # expand mask to shape (BLOCK_M, BLOCK_N, 1) then broadcast
    mask_exp = mask_val[:, :, None]

    out = proj * gate * mask_exp

    # store
    out_off = (off_i[:, None, None] * stride_out_i
               + off_j[None, :, None] * stride_out_j
               + off_c[None, None, :] * stride_out_c
               + b * stride_out_b)
    tl.store(out_ptr + out_off, out, mask=mask_ijk)
```

"""
TriMul (outgoing) forward pass with a fused Triton kernel for the
left/right projection‑gate‑mask fusion. The expensive cubic update
is performed as a batched matrix multiplication using torch.bmm,
while the elementwise mask+gate fusion is done in a custom Triton
kernel. No gradients are computed.
"""

import torch
import triton
import triton.language as tl
import math
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: out = proj * gate * mask   (proj,gate:[B,N,N,C], mask:[B,N,N])
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,      # pointers
    B, N, C,                                   # tensor sizes
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_c,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_c,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_C: tl.constexpr
):
    # program ids
    pid_i = tl.program_id(0)               # i‑block
    pid_j = tl.program_id(1)               # j‑block
    pid_bc = tl.program_id(2)              # combined batch‑and‑c block

    # decompose batch and channel block
    num_c_blocks = tl.cdiv(C, BLOCK_C)
    b = pid_bc // num_c_blocks
    c_block = pid_bc % num_c_blocks

    # global start indices for this block
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    c0 = c_block * BLOCK_C

    # offsets within the block
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_c = c0 + tl.arange(0, BLOCK_C)

    # bound checks
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_c = offs_c < C
    mask_ij = mask_i[:, None] & mask_j[None, :]
    mask_ijk = mask_ij[:, :, None] & mask_c[None, None, :]

    # ------------------------------------------------------------------
    # Load proj, gate and mask (broadcast mask over C)
    # ------------------------------------------------------------------
    proj_off = (offs_i[:, None, None] * stride_proj_i +
                offs_j[None, :, None] * stride_proj_j +
                offs_c[None, None, :] * stride_proj_c +
                b * stride_proj_b)
    gate_off = (offs_i[:, None, None] * stride_gate_i +
                offs_j[None, :, None] * stride_gate_j +
                offs_c[None, None, :] * stride_gate_c +
                b * stride_gate_b)

    mask_off = (offs_i[:, None] * stride_mask_i +
                offs_j[None, :] * stride_mask_j +
                b * stride_mask_b)

    proj = tl.load(proj_ptr + proj_off, mask=mask_ijk, other=0.0)
    gate = tl.load(gate_ptr + gate_off, mask=mask_ijk, other=0.0)
    mask_val = tl.load(mask_ptr + mask_off, mask=mask_ij, other=0.0)

    # broadcast mask over the channel axis
    mask_val = mask_val[:, :, None]

    # ------------------------------------------------------------------
    # Compute output = proj * gate * mask
    # ------------------------------------------------------------------
    out = proj * gate * mask_val

    # Store result
    out_off = (offs_i[:, None, None] * stride_out_i +
               offs_j[None, :, None] * stride_out_j +
               offs_c[None, None, :] * stride_out_c +
               b * stride_out_b)
    tl.store(out_ptr + out_off, out, mask=mask_ijk)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (float32, 0/1)
            - weights_dict: mapping from weight name to torch.nn.Parameter
            - config_dict: must contain keys "dim" and "hidden_dim"
                           and optionally "nomask" (bool)
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on input (over the last dimension)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]               # (dim, hidden_dim)

    # Projections
    left_proj = F.linear(x, left_proj_weight)              # [B,N,N,hidden_dim]
    right_proj = F.linear(x, right_proj_weight)

    # Gates (pre‑sigmoid)
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # Apply mask and elementwise gates with a Triton kernel (if needed)
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # Ensure mask is float32 on the same device
        mask = mask.to(dtype=torch.float32, device=device)

        # Allocate tensors for gated & masked results
        left = torch.empty_like(left_proj, device=device)
        right = torch.empty_like(right_proj, device=device)

        # Strides (in elements) for the arguments
        # left_proj / right_proj share the same layout
        stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_c = left_proj.stride()
        stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_c = left_gate.stride()
        stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_c = right_proj.stride()
        stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_c = right_gate.stride()
        stride_m_b, stride_m_i, stride_m_j = mask.stride()
        # output strides are identical to input layout
        stride_out_b, stride_out_i, stride_out_j, stride_out_c = left.stride()

        # Kernel launch configuration
        BLOCK_M = 32
        BLOCK_N = 32
        BLOCK_C = 32
        grid_i = triton.cdiv(N, BLOCK_M)
        grid_j = triton.cdiv(N, BLOCK_N)
        num_c_blocks = triton.cdiv(hidden_dim, BLOCK_C)
        grid_bc = B * num_c_blocks
        grid = (grid_i, grid_j, grid_bc)

        # Left tensor
        apply_gate_kernel[grid](
            left_proj, left_gate, mask, left,
            B, N, hidden_dim,
            stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_c,
            stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_c,
            stride_m_b, stride_m_i, stride_m_j,
            stride_out_b, stride_out_i, stride_out_j, stride_out_c,
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_C=BLOCK_C,
        )

        # Right tensor (reuse same grid)
        apply_gate_kernel[grid](
            right_proj, right_gate, mask, right,
            B, N, hidden_dim,
            stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_c,
            stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_c,
            stride_m_b, stride_m_i, stride_m_j,
            stride_out_b, stride_out_i, stride_out_j, stride_out_c,
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_C=BLOCK_C,
        )

    # ------------------------------------------------------------------
    # Batched matrix multiplication: out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
    # ------------------------------------------------------------------
    # Reshape to (B*hidden_dim, N, N) and perform bmm with right transposed
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_mat = right.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden_dim
    left_mat = left_mat.view(BH, N, N)
    right_mat = right_mat.view(BH, N, N)

    # GEMM: left @ right^T  (right transposed over the N dimension)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output layernorm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)
    out = out * out_gate
    out = F.linear(out, to_out_weight)    # [B,N,N,dim]

    return out
