"""
TriMul (outgoing) forward pass with a Triton kernel for the core contraction.

The input tensor x ∈ ℝ^{B×N×N×C} is layer‑normalized, projected,
gated and optionally masked. The main O(N³·C_z) operation is

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication
    out_{b,d} = left_{b,d} @ right_{b,d}ᵀ
where left/right_{b,d} ∈ ℝ^{N×N} for each hidden dimension d.
The kernel computes this contraction for all batches and hidden dimensions
in a fused fashion using Triton.  After the contraction a second
LayerNorm, a gating term and a final linear projection produce the
output of shape [B, N, N, C].

Only the contraction is performed in Triton; all other steps use
regular PyTorch ops.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    L_ptr, R_ptr, O_ptr,                # pointers
    N,                                   # sequence length
    stride_l_batch, stride_l_head, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_head, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_head, stride_o_i, stride_o_j,
    B, H,                                # batch size, hidden dim
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out = left @ rightᵀ for each (batch, head) pair.
    left and right have layout (B, H, N, N) with dimensions
        - i (row) : stride_l_i
        - k (col) : stride_l_k
    rightᵀ is accessed by swapping its i/k strides.
    """
    pid_z = tl.program_id(2)              # index over (batch * hidden)
    pid_m = tl.program_id(0)              # i‑tile index
    pid_n = tl.program_id(1)              # j‑tile index

    batch_idx = pid_z // H
    head_idx  = pid_z %  H

    # Offsets for the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- Load left tile (M × K) -----
        L_ptrs = (L_ptr
                  + batch_idx * stride_l_batch
                  + head_idx  * stride_l_head
                  + offs_m[:, None] * stride_l_i
                  + offs_k[None, :] * stride_l_k)
        L = tl.load(L_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # ----- Load right tile transposed (K × N) -----
        # rightᵀ element (k, j) = right[j, k]
        R_ptrs = (R_ptr
                  + batch_idx * stride_r_batch
                  + head_idx  * stride_r_head
                  + offs_k[:, None] * stride_r_k   # row = k (original column)
                  + offs_n[None, :] * stride_r_i)  # col = j (original row)
        R = tl.load(R_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # ----- Accumulate -----
        acc += tl.dot(L, R)   # (M×K)·(K×N) → (M×N) in FP32

    # ----- Store result -----
    O_ptrs = (O_ptr
              + batch_idx * stride_o_batch
              + head_idx  * stride_o_head
              + offs_m[:, None] * stride_o_i
              + offs_n[None, :] * stride_o_j)
    tl.store(O_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, C] (float32) on the same device as the input.
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, C = input_tensor.shape  # C == dim

    # ---------- LayerNorm ----------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ---------- Linear projections ----------
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]    # (dim, hidden_dim)

    left  = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ---------- Optional mask ----------
    if mask is not None and not config.get("nomask", False):
        mask_f = mask.to(dtype).unsqueeze(-1)                 # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ---------- Gates ----------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ---------- Prepare tensors for Triton kernel ----------
    # Permute to (B, H, N, N) layout for contiguous memory access
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Output buffer in FP32
    out_perm = torch.empty_like(left_perm, dtype=torch.float32)

    # Strides (convert to 64‑bit ints for Triton)
    stride_l_batch, stride_l_head, stride_l_i, stride_l_k = left_perm.stride()
    stride_r_batch, stride_r_head, stride_r_i, stride_r_k = right_perm.stride()
    stride_o_batch, stride_o_head, stride_o_i, stride_o_j = out_perm.stride()

    # Block sizes – tuned for typical Nvidia GPUs (B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid dimensions
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m, num_blocks_n, B * hidden_dim)

    # Launch Triton kernel
    tri_mul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        N,
        stride_l_batch, stride_l_head, stride_l_i, stride_l_k,
        stride_r_batch, stride_r_head, stride_r_i, stride_r_k,
        stride_o_batch, stride_o_head, stride_o_i, stride_o_j,
        B, hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Post‑processing ----------
    # out_perm: shape [B, hidden_dim, N, N] → [B, N, N, hidden_dim]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # Apply the output gate
    out = out * out_gate

    # Second LayerNorm over the hidden dimension
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # Final linear projection back to `dim`
    out = F.linear(out, to_out_weight)

    return out
