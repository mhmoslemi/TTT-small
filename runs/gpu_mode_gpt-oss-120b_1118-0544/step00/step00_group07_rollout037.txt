def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_WARPS: tl.constexpr,
):

    pid_n = tl.program_id(1)
    pid_b = tl.program_id(2)  # batch index

    # compute block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute pointer offsets for A and B blocks
    a_ptrs = a_ptr + pid_b * batch_stride_a + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak

"""
TriMul forward kernel (outgoing version) optimized with Triton.

Algorithm
---------
1. Layer‑normalize the input tensor (last dimension = dim).
2. Linear projections to hidden dimension:
   left = x_norm @ left_proj_weight
   right = x_norm @ right_proj_weight
3. Optional mask (broadcasted on the hidden dim) is applied to left/right.
4. Gating:
   left  *= sigmoid(x_norm @ left_gate_weight)
   right *= sigmoid(x_norm @ right_gate_weight)
5. Compute the core O(N³·H) contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication: for each hidden channel d
   (and each batch) we compute  left_d @ right_dᵀ.
   The batched GEMM is performed by a custom Triton kernel that
   processes many (batch·hidden) matrices in parallel, using fp16
   arithmetic and accumulation in fp32.
6. Layer‑normalize the resulting tensor over the hidden dimension,
   apply the output gate (sigmoid of a linear), and a final linear
   projection back to the original dimension.
7. Return a float32 tensor of shape [B, N, N, dim].

The Triton kernel works on tensors of shape [B·H, N, N] (contiguous
row‑major) and computes C = A @ Bᵀ for each batch element.
"""
import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr, num_stages: tl.constexpr
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_b = tl.program_id(2)  # batch (includes hidden dimension)

    # offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the A and B tiles (k = 0)
    a_ptrs = a_ptr + pid_b * batch_stride_a + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * batch_stride_b + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # k‑range used for masking
        k_range = k * BLOCK_K + tl.arange(0, BLOCK_K)

        mask_a = mask_m[:, None] & (k_range[None, :] < K)
        mask_b = (k_range[:, None] < K) & mask_n[None, :]

        a = tl.load(a_ptrs, mask=mask_a, other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0, dtype=tl.float16)

        acc += tl.dot(a, b)

        # Advance pointers to the next k‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store the result (convert to fp16 to keep memory footprint low)
    c_ptrs = c_ptr + pid_b * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module using a Triton‑accelerated batched GEMM.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be ignored if config["nomask"] is True)
            - weights: dict of model weights (see skeleton)
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    bs, seq_len, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # helpers for layer norm
    # ------------------------------------------------------------------
    eps = config.get("eps", 1e-5)

    # LayerNorm over the last dimension
    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (bias=False)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare for the Triton GEMM (convert to fp16 for bandwidth)
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange to [B, hidden_dim, N, N] and make contiguous
    left_h = left_h.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    # Collapse batch and hidden dimensions => shape (B*H, N, N)
    bh = bs * hidden_dim
    left_mat = left_h.view(bh, seq_len, seq_len)
    right_mat = right_h.view(bh, seq_len, seq_len)

    # Output buffer
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(seq_len, BLOCK_M),   # blocks over M dimension (i)
        triton.cdiv(seq_len, BLOCK_N),   # blocks over N dimension (j)
        bh,                              # batch*hidden
    )

    batched_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        batch_stride_a=left_mat.stride(0),
        stride_am=left_mat.stride(1),
        stride_ak=left_mat.stride(2),
        batch_stride_b=right_mat.stride(0),
        # Note: we treat B as transposed inside the kernel:
        stride_bk=right_mat.stride(2),   # original column stride (k)
        stride_bn=right_mat.stride(1),   # original row stride (j)
        batch_stride_c=out_mat.stride(0),
        stride_cm=out_mat.stride(1),
        stride_cn=out_mat.stride(2),
        M=seq_len,
        N=seq_len,
        K=seq_len,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] and continue on the PyTorch side
    # ------------------------------------------------------------------
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # Output layer‑norm (over hidden_dim)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # gating after norm
    out = out * out_gate

    # final linear projection back to original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
