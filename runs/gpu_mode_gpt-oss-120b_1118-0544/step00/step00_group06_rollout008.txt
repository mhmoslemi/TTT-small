def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x_norm.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Move hidden dim to batch
    B, N, _, _ = left.shape
    left_batch = left.permute(0,3,1,2).contiguous()   # (B, hidden_dim, N, N)
    right_batch = right.permute(0,3,1,2).contiguous()

    batch = B * hidden_dim
    A = left_batch.view(batch, N, N)
    B_ = right_batch.view(batch, N, N)

    # Cast to half for kernel
    A_fp16 = A.to(torch.float16)
    B_fp16 = B_.to(torch.float16)
    C_fp16 = torch.empty_like(A_fp16)

    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_gemm_kernel[grid](
        A_fp16, B_fp16, C_fp16,
        batch, N, N, N,
        A_fp16.stride(0), A_fp16.stride(1), A_fp16.stride(2),
        B_fp16.stride(0), B_fp16.stride(1), B_fp16.stride(2),
        C_fp16.stride(0), C_fp16.stride(1), C_fp16.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Cast back to float32
    out_hidden = C_fp16.to(dtype=torch.float32)

    # Reshape to original layout
    out_hidden = out_hidden.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()   # shape B,N,N,hidden_dim

    # Output layer norm
    out_norm = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply out gate
    out_gated = out_norm * out_gate

    # Final projection
    out = torch.nn.functional.linear(out_gated, weights["to_out.weight"])

    return out
```

"""
High‑performance Triton implementation of the “outgoing” TriMul operator (AlphaFold3).

The expensive core is a batch of independent GEMMs:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which we compute as a set of GEMMs (one per hidden dimension) using a custom
Triton kernel. All other operations (LayerNorm, Linear projections, sigmoid
gates) are performed with PyTorch for simplicity.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM computing L @ R^T for many batches.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    batch, M, N, K,                     # dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)   # batch index
    pid_m = tl.program_id(1)       # row‑tile index
    pid_n = tl.program_id(2)       # col‑tile index

    # Compute global offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for this batch
    a_batch_ptr = A_ptr + pid_batch * stride_a_batch
    b_batch_ptr = B_ptr + pid_batch * stride_b_batch
    c_batch_ptr = C_ptr + pid_batch * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a block of A (M x K)
        a = tl.load(
            a_batch_ptr
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a block of B (N x K) – we need B transposed, so we load B[j,k]
        b = tl.load(
            b_batch_ptr
            + offs_n[:, None] * stride_b_n
            + offs_k[None, :] * stride_b_k,
            mask=mask_n[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Accumulate dot product over K‑block
        acc += tl.dot(a, b)

    # Write result
    c = acc  # already FP32
    tl.store(
        c_batch_ptr
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )

# ----------------------------------------------------------------------
# Entry point expected by the evaluator.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module using a Triton kernel for the
    O(N³·hidden_dim) core operation.
    Args:
        data: tuple (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → broadcast to [B, N, N, 1]
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core Triton GEMM: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    This is a batch of (hidden_dim) independent GEMMs.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Move hidden dimension into the batch axis and make tensors contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch = B * hidden_dim
    A = left_perm.view(batch, N, N)   # [batch, M=N, K=N]
    B_ = right_perm.view(batch, N, N)  # same layout

    # Allocate output tensor (float32)
    C = torch.empty_like(A, dtype=torch.float32, device=device)

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        A,
        B_,
        C,
        batch,
        N,
        N,
        N,
        A.stride(0),
        A.stride(1),
        A.stride(2),
        B_.stride(0),
        B_.stride(1),
        B_.stride(2),
        C.stride(0),
        C.stride(1),
        C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out_hidden = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_gated = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out_gated, weights["to_out.weight"])

    return out
