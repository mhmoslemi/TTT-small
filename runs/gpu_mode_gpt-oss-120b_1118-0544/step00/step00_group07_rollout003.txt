"""TriMul (outgoing) forward pass.

This implementation follows the PyTorch reference and fuses the core
pair‑wise multiplication Σ_k left[b,i,k,d] * right[b,j,k,d] into a
custom Triton kernel.  The kernel computes a batched matrix multiplication
for each (batch, hidden) slice:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to `left @ right.T` per hidden dimension.
All surrounding operations (LayerNorm, linear projections, gating, final
projection) are performed with PyTorch to keep the code concise, while the
expensive O(N³) contraction is accelerated by Triton.

The kernel works for arbitrary batch size (B∈{1,2}), sequence length N (up
to 1024) and hidden dimension H (≤128).  It handles an optional mask
(by applying it before the Triton kernel) and returns a float32 tensor of
shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # B=batch, N=seq_len, H=hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = left @ right.T for each (b, h) pair.

    left  : [B, N, N, H]   (i, k)
    right : [B, N, N, H]   (j, k)  -> accessed as (k, j) via swapped strides
    out   : [B, N, N, H]   (i, j)
    """
    pid = tl.program_id(0)                     # combines batch and hidden dim
    bh = pid
    b = bh // H                                 # batch index
    h = bh % H                                  # hidden index

    # block indices for output matrix (i, j)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute start indices of the tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid i/j positions
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers to the beginning of the current (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension (the summed axis)
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for the current K‑tile
        # left: (i, k)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        # right accessed as (k, j) → use stride_right_k for rows, stride_right_j for columns
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i

        # load with masks
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate dot product
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N) in FP32

    # write results back as FP16 (kernel works on half precision)
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be all‑ones)
            - weights: dict of model parameters
            - config: dict containing 'dim' and 'hidden_dim' (and optionally 'eps')

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Gates (sigmoid) and mask application
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    if mask is not None:
        mask = mask.to(dtype=left.dtype)
        mask = mask.unsqueeze(-1)  # (B,N,N,1)
        left = left * mask
        right = right * mask

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to half precision for the heavy O(N³) contraction
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    B, N, _, _ = left_h.shape

    # output buffer in half precision
    out_h = torch.empty_like(left_h, dtype=torch.float16, device=device)

    # strides for Triton (must be int64)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_h.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_h.stride()
    stride_out_b, stride_out_i, stride_out_k, stride_out_h = out_h.stride()

    # ------------------------------------------------------------------
    # Triton kernel launch
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        B * hidden_dim,                         # axis 0: batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,           # axis 1: i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,           # axis 2: j‑tiles
    )
    tri_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_k,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Convert back to FP32 for the remaining ops
    out = out_h.to(torch.float32)

    # ------------------------------------------------------------------
    # Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
