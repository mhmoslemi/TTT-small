def tri_mul_kernel(A_ptr, B_ptr, C_ptr, M, N, K,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
                  BLOCK_K: tl.constexpr):
   pid_i = tl.program_id(0)
   pid_j = tl.program_id(1)
   pid_batch = tl.program_id(2)

   # compute start offsets
   offs_m = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

   # mask out-of-bounds
   mask_m = offs_m < M
   mask_n = offs_n < N

   # Initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # compute pointer to batch slice
   offset_batch = pid_batch * M * K
   A_base = A_ptr + offset_batch
   B_base = B_ptr + pid_batch * K * N   # B shape [K,N] after transpose (since B is KxN)
   # We'll use for k loops:
   for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # compute block loads
        a_ptrs = A_base + (offs_m[:, None] * K + (k_offset + tl.arange(0, BLOCK_K))[None, :])
        b_ptrs = B_base + ((k_offset + tl.arange(0, BLOCK_K))[:, None] * N + offs_n[None, :])
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k_offset + tl.arange(0, BLOCK_K) < K), other=0.0, dtype=tl.float16)

@triton.jit
def tri_mul_kernel(A_ptr, B_ptr, C_ptr, M, N, K,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_batch = tl.program_id(2)

    # compute offsets for tiles
    offs_m = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    # pointer to start of batch
    A_batch = A_ptr + pid_batch * M * K
    B_batch = B_ptr + pid_batch * K * N
    C_batch = C_ptr + pid_batch * M * N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # compute pointers for this K block
        a_ptrs = A_batch + (offs_m[:, None] * K) + (k_offset + k_range)[None, :]
        b_ptrs = B_batch + (k_offset + k_range)[:, None] * N + offs_n[None, :]

        # masks
        mask_a = (offs_m[:, None] < M) & ((k_offset + k_range)[None, :] < K)
        mask_b = ((k_offset + k_range)[:, None] < K) & (offs_n[None, :] < N)

        # load
        a = tl.load(a_ptrs, mask=mask_a, other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0, dtype=tl.float16)

        # convert to float32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = C_batch + (offs_m[:, None] * N) + offs_n[None, :]
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)

"""
TriMul (outgoing) implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Project to a hidden dimension (left/right) and compute gating values.
3. Apply optional pairwise mask and element‑wise gates.
4. Cast to float16 and reshape:
   left  -> [B, H, N, N]   (i,k)
   right -> [B, H, N, N]   (k,j)   (right is transposed on the two N axes)
   Flatten batch & head:  [B*H, N, N].
5. Triton kernel computes the batched matrix product
        out[b,i,j] = Σₖ left[b,i,k] * right[b,k,j]
   for every head simultaneously (standard GEMM with row‑major layout).
6. Reshape back to [B, N, N, H] and cast to float32.
7. Apply the output LayerNorm, output gate and final linear projection.

Only the heavy O(N³·H) multiplication is performed in Triton; all other
operations use PyTorch (which is fast for the remaining work).
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel : batched GEMM (A @ B) where A:[batch, M, K], B:[batch, K, N]
# --------------------------------------------------------------------------- #
@triton.jit
def _triton_trimul_kernel(
    A_ptr, B_ptr, C_ptr,    # pointers
    M, N, K,                # matrix dimensions (int)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)          # block row (i)
    pid_j = tl.program_id(1)          # block col (j)
    pid_b = tl.program_id(2)          # batch * head index

    # --------------------------------------------------------------- #
    # Offsets for the current tile
    # --------------------------------------------------------------- #
    offs_m = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointer for this batch (row‑major layout)
    A_batch = A_ptr + pid_b * M * K
    B_batch = B_ptr + pid_b * K * N
    C_batch = C_ptr + pid_b * M * N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------- #
    # Loop over reduction dimension K
    # --------------------------------------------------------------- #
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K

        # Pointers to current K‑tile
        a_ptrs = A_batch + (offs_m[:, None] * K) + (k_offset + offs_k)[None, :]
        b_ptrs = B_batch + (k_offset + offs_k)[:, None] * N + offs_n[None, :]

        # Masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & ((k_offset + offs_k)[None, :] < K)
        mask_b = ((k_offset + offs_k)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0, dtype=tl.float16)

        # Convert to fp32 for the dot
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # --------------------------------------------------------------- #
    # Write back result
    # --------------------------------------------------------------- #
    c_ptrs = C_batch + (offs_m[:, None] * N) + offs_n[None, :]
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


# --------------------------------------------------------------------------- #
# Entry point used by the evaluator
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, C]   (float32)
        - mask         : torch.Tensor[B, N, N] or None (float32/bool)
        - weights      : dict of torch.Tensor containing the model parameters
        - config       : dict with keys 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------- #
    # Unpack arguments
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------- #
    # LayerNorm on the last dimension
    # ------------------------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------- #
    # Linear projections (float32)
    # ------------------------------------------------------------------- #
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # Gating (float32) and optional mask
    # ------------------------------------------------------------------- #
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    if mask is not None:
        # ensure mask has same device and dtype as tensors
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # Cast to half precision for the heavy GEMM
    # ------------------------------------------------------------------- #
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape  # B, N, N, hidden_dim

    # ------------------------------------------------------------------- #
    # Prepare tensors for the Triton batched GEMM
    #   left  : [B, H, N, N] (i,k)
    #   right : [B, H, N, N] (k,j) -> transpose on the two N axes
    # ------------------------------------------------------------------- #
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # [B, H, N, N]

    # Flatten batch and head dimension -> [B*H, N, N]
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=left.device)

    # ------------------------------------------------------------------- #
    # Triton launch configuration
    # ------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,      # i dimension
        (N + BLOCK_N - 1) // BLOCK_N,      # j dimension
        B * H                              # batch*head
    )

    # Launch kernel
    _triton_trimul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,        # M
        N,        # N
        N,        # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------- #
    # Reshape back to [B, N, N, hidden_dim] (float32)
    # ------------------------------------------------------------------- #
    out_perm = out_flat.view(B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out = out.to(torch.float32)

    # ------------------------------------------------------------------- #
    # Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
