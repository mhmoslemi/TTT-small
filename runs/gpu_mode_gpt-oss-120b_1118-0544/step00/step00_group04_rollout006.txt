def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_k, stride_right_i, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Get program ids
    bh = tl.program_id(0)
    b = bh // H
    h = bh % H

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute offsets for the tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # Create mask for valid i, j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers base offset for this (b,h) slice
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, N, BLOCK_K):

"""
tri_mul_triton.py

Fast Triton implementation of the AlphaFold3 “outgoing” TriMul operator.
The heavy cubic contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed by a custom Triton kernel that treats the operation as a
batched GEMM (left @ rightᵀ) for each hidden dimension slice.
All other components (LayerNorm, linear projections, sigmoids, final
normalisation/projection) are done with PyTorch for simplicity.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,               # *float32
    right_ptr,              # *float32
    out_ptr,                # *float32
    B, N, H,                # int32
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_k, stride_right_i, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute for each (b, h) slice:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    This is equivalent to a matrix multiplication left @ rightᵀ.
    """

    # --------------------------------------------------------------------
    # 1) Identify which batch / head (hidden dim) we are processing
    # --------------------------------------------------------------------
    pid = tl.program_id(0)
    b = pid // H               # batch index
    h = pid % H                # hidden‑dim index

    # 2) Tile indices for i and j dimensions
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # 3) Compute base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # 4) Accumulator for the tile (BLOCK_I × BLOCK_J)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # --------------------------------------------------------------------
    # 5) Loop over the reduction dimension k in blocks
    # --------------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k] tile – shape (BLOCK_I, BLOCK_K)
        a_ptrs = left_base + (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float32)

        # rightᵀ[k, j] tile – we need rows = k, cols = j
        # (BLOCK_K, BLOCK_J)
        b_ptrs = right_base + (offs_k[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_i)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0,
                    dtype=tl.float32)

        # GEMM accumulate: (BLOCK_I, BLOCK_K) @ (BLOCK_K, BLOCK_J) → (BLOCK_I, BLOCK_J)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------------
    # 6) Write the computed tile back to memory
    # --------------------------------------------------------------------
    out_ptrs = out_base + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

    # Load weights
    # --------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # --------------------------------------------------------------------
    # Stage 1: LayerNorm and linear projections
    # --------------------------------------------------------------------
    x_norm = F.layer_norm(input_tensor,
                          normalized_shape=(dim,),
                          weight=norm_weight,
                          bias=norm_bias)

    # Projections (no bias)
    left  = F.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_weight)  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------------
    # Stage 2: Gating (sigmoid) and optional mask
    # --------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # Apply mask if provided (mask has shape [B,N,N])
    if mask is not None:
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left  * mask_f
        right = right * mask_f

    # Apply gates
    left  = left  * left_gate
    right = right * right_gate

    # Ensure contiguous layout for kernel
    left  = left.contiguous()
    right = right.contiguous()

    # --------------------------------------------------------------------
    # Stage 3: Cubic contraction via Triton kernel
    # --------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Compute strides (int64) for each tensor
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Block sizes – tuned for the B200 GPU
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                     # one program per (batch, hidden_dim)
        math.ceil(N / BLOCK_I),            # tiles along i
        math.ceil(N / BLOCK_J),            # tiles along j
    )

    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_k, stride_r_i, stride_r_h,   # note k and i swapped for transposition
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_I, BLOCK_J, BLOCK_K,
        num_warps=4,  # empirically good for this size
    )
    # Ensure kernel completion before using the result
    torch.cuda.synchronize(device)

    # --------------------------------------------------------------------
    # Stage 4: Second LayerNorm, gating, and final projection
    # --------------------------------------------------------------------
    out_norm = F.layer_norm(out,
                            normalized_shape=(hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias)

    out_norm = out_norm * out_gate  # broadcast over hidden dim

    # Final linear projection back to original dim
    output = F.linear(out_norm, to_out_weight)  # [B,N,N,dim]

    return output
```
