def custom_kernel(data):

    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5  # typical

    # Get weights
    norm_weight = weights["norm.weight"]  # shape [dim]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]  # shape [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]  # shape [hidden_dim]
    to_out_norm_bias = weights["to_out_norm.bias"]   # shape [hidden_dim]
    to_out_weight = weights["to_out.weight"]  # shape [dim, hidden_dim]

    # Cast to appropriate dtype
    # assume input_tensor dtype is float32

    # LayerNorm on dim
    # compute mean and var over last dim
    x = input_tensor
    # shape: [B,N,N,dim]
    # flatten for LN per instance: can compute using torch.mean & var over -1
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Linear projections left/right: shape [B,N,N,hidden_dim]
    # using torch.nn.functional.linear: it expects input shape (..., dim) and weight shape (out_features, in_features). So:
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # no bias
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if needed
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Compute core multiplication using Triton kernel
    # Permute to bring hidden_dim to batch for fused dimension
    B, N, _, _ = left.shape
    # Permute left/right to shape (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    right_flat = right_perm.view(-1, N, N)

    out_flat = torch.empty_like(left_flat, dtype=left_flat.dtype)  # shape (B*hidden_dim, N, N)

    right_fp16 = right_flat.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    BHM = left_fp16.shape[0]  # batch*hidden_dim
    seq_len = N

    b_batch_stride, b_i_stride, b_k_stride = right_fp16.stride()
    c_batch_stride, c_i_stride, c_j_stride = out_fp16.stride()
    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = ( (seq_len + BLOCK_M - 1) // BLOCK_M,
             (seq_len + BLOCK_N - 1) // BLOCK_N,

    # Call kernel
    tri_mul_gemm_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        seq_len, seq_len, seq_len,
        a_batch_stride, a_i_stride, a_k_stride,
        b_batch_stride, b_i_stride, b_k_stride,
        c_batch_stride, c_i_stride, c_j_stride,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # Convert output back to float32
    out = out_fp16.to(torch.float32)

    # Reshape back to (B, hidden_dim, N, N) then permute back to (B,N,N,hidden_dim)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Apply to_out_norm (LayerNorm over hidden_dim)
    # LN dims: last dim hidden_dim
    # compute mean and var across hidden_dim dimension
    mean_out = out.mean(dim=-1, keepdim=True)
    var_out = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_out) * torch.rsqrt(var_out + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # Multiply by out_gate (broadcast)
    out_norm = out_norm * out_gate

    # Final linear to convert hidden_dim to dim
    # weight shape (dim, hidden_dim)
    out_final = torch.nn.functional.linear(out_norm, to_out_weight)  # no bias

    return out_final

TriMul outgoing implementation (forward only).

Algorithm
---------

1. **Layer‑norm on the input channels** (dim).
2. **Linear projections**:
   - `left  = x_norm @ left_proj_weight`   → (B,N,N,hidden)
   - `right = x_norm @ right_proj_weight` → (B,N,N,hidden)
3. Optional *mask* multiplication on the projected tensors.
4. **Gating**:
   - Compute three sigmoids from `x_norm` (left_gate, right_gate, out_gate).
   - Apply left/right gates element‑wise to `left` and `right`.
5. **Core O(N³) reduction**:
   - Compute for each hidden channel `h`
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] .
   - This is a batched matrix multiply `left @ rightᵀ` where the hidden
     dimension is fused into the batch dimension.
   - Implemented in Triton (`tri_mul_gemm_kernel`) with 64×64×64 tiling,
     fp16 inputs and fp32 accumulation.
6. **Layer‑norm on the hidden dimension** (to_out_norm) followed by the
   `out_gate` element‑wise multiplication.
7. **Final linear projection** back to `dim`.

Only the O(N³) reduction is performed in Triton; all surrounding

@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix sizes (all equal to seq_len)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ Bᵀ for a batch of square matrices.
    A: (batch, M, K)  – fp16
    B: (batch, N, K)  – fp16   (we read it as transposed: Bᵀ has shape K×N)
    C: (batch, M, N)  – fp16 (accumulated in fp32)
    """

    pid_batch = tl.program_id(2)                     # fused batch (B * hidden_dim)
    pid_m = tl.program_id(0)                         # block row
    pid_n = tl.program_id(1)                         # block col

    # block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for this batch
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k in range(num_k_tiles):
        k_offset = k * BLOCK_K

        # ----- load A tile (M×K) -----
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_a_m + (k_offset + offs_k)[None, :] * stride_a_k
        mask_a = (offs_m[:, None] < M) & ((k_offset + offs_k)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)           # fp16

        # ----- load Bᵀ tile (K×N) -----
        # Bᵀ has rows = k (original row of B), cols = n (original col of B)
        b_ptrs = b_batch_ptr + (k_offset + offs_k)[:, None] * stride_b_n + offs_n[None, :] * stride_b_k
        mask_b = ((k_offset + offs_k)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)           # fp16

        # ----- dot product -----
        acc += tl.dot(a_tile, b_tile)            # fp32 accumulation

    # ----- store result -----
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Expected input:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    # ----------------------------------------------------------------------
    # Unpack config
    # ----------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------------------
    # Unpack weights
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----------------------------------------------------------------------
    # LayerNorm over the channel dimension (dim)
    # ----------------------------------------------------------------------
    x = input_tensor
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias                     # [B,N,N,dim]

    # ----------------------------------------------------------------------
    # Linear projections (no bias)
    # ----------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # ----------------------------------------------------------------------
    # Optional mask (broadcast over hidden dim)
    # ----------------------------------------------------------------------
    if not nomask and mask is not None:
        mask = mask.to(dtype=dtype)
        mask = mask.unsqueeze(-1)                                 # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------------------
    # Gating (sigmoid of linear)
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # Core O(N³) reduction via Triton: out = left @ rightᵀ per hidden channel
    # ----------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Fuse hidden dimension into the batch dimension for the kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()            # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()          # [B, hidden, N, N]

    left_flat = left_perm.view(-1, N, N)                         # [B*hidden, N, N]
    right_flat = right_perm.view(-1, N, N)

    # Cast to fp16 for the kernel (inputs) – accumulator stays fp32 inside kernel
    a_fp16 = left_flat.to(torch.float16)
    b_fp16 = right_flat.to(torch.float16)
    c_fp16 = torch.empty_like(a_fp16, dtype=torch.float16)

    # Strides for the kernel (batch, row, col)
    stride_a_batch, stride_a_m, stride_a_k = a_fp16.stride()
    stride_b_batch, stride_b_n, stride_b_k = b_fp16.stride()
    stride_c_batch, stride_c_i, stride_c_j = c_fp16.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks along M (i)
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks along N (j)
        a_fp16.shape[0]                 # fused batch = B * hidden_dim
    )

    tri_mul_gemm_kernel[grid](
        a_fp16, b_fp16, c_fp16,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_i, stride_c_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Convert back to fp32 and reshape to original layout
    out = c_fp16.to(torch.float32)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,hidden]

    # ----------------------------------------------------------------------
    # LayerNorm over hidden dimension (to_out_norm) and out_gate multiplication
    # ----------------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    out_norm = out_norm * out_gate   # element‑wise gate

    # ----------------------------------------------------------------------
    # Final linear projection back to `dim`
    # ----------------------------------------------------------------------
    out_final = torch.nn.functional.linear(out_norm, to_out_weight)   # [B,N,N,dim]

    return out_final
```

This implementation follows the reference PyTorch logic, fusing the expensive triply‑nested sum into a high‑performance Triton GEMM kernel while keeping the rest of the pipeline in PyTorch for simplicity and correctness.
