@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    # compute group index for batch and hidden dim
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)
    tiles_per_batch_hidden = num_tiles_i * num_tiles_j
    # group index = gid = pid // tiles_per_batch_hidden
    gid = pid // tiles_per_batch_hidden
    tile_idx = pid % tiles_per_batch_hidden
    # compute batch and hidden dim ids
    batch_id = gid // D
    d_id = gid % D
    # compute tile coordinates
    i_tile = tile_idx // num_tiles_j
    j_tile = tile_idx % num_tiles_j
    # Compute block coordinates
    i_offset = i_tile * BLOCK_M
    j_offset = j_tile * BLOCK_N

    # offsets for left and right
    # We'll accumulate over k in inner loop
    # Create offset vectors
    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    # mask for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers base for this batch and d
    left_base = left_ptr + batch_id * stride_left_batch + d_id * stride_left_d
    right_base = right_ptr + batch_id * stride_right_batch + d_id * stride_right_d
    out_base = out_ptr + batch_id * stride_out_batch + d_id * stride_out_d

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over kBlocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        # compute k offsets
        k_offs = k_offset + offs_k
        # mask for k
        mask_k = k_offs < N

        # load left tile [M, K]
        left_ptrs = left_base + (offs_i[:, None] * stride_left_i) + (k_offs[None, :] * stride_left_k)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # cast to float16 maybe
        left = left.to(tl.float16)

        # load right tile [K, N] (note swapped order)
        right_ptrs = right_base + (k_offs[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_i)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        right = right.to(tl.float16)

        # dot product accumulate
        acc += tl.dot(left, right)  # shape (BLOCK_M, BLOCK_N)

        # increment pointers: we rely on offsets within loop and recompute each iteration

    # Write output
    out_ptrs = out_base + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])
```

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_kernel(left_ptr, right_ptr, out_ptr,
                  B, N, D,
                  stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
                  stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
                  stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
   # param: program_id 0 (1D)
   pid = tl.program_id(0)
   # number of tiles per batch*hidden
   num_tiles_i = tl.cdiv(N, BLOCK_M)
   num_tiles_j = tl.cdiv(N, BLOCK_N)
   tiles_per_batch_hidden = num_tiles_i * num_tiles_j
   # compute group id = batch*hidden_dim + d
   group = pid // tiles_per_batch_hidden
   tile_idx = pid % tiles_per_batch_hidden
   # batch = group // D, d = group % D
   batch = group // D
   d = group % D
   # tile coordinates
   i_tile = tile_idx // num_tiles_j
   j_tile = tile_idx % num_tiles_j

   # base offsets for i and j
   i_offset = i_tile * BLOCK_M
   j_offset = j_tile * BLOCK_N

   # compute offsets within tile
   offs_i = i_offset + tl.arange(0, BLOCK_M)
   offs_j = j_offset + tl.arange(0, BLOCK_N)
   mask_i = offs_i < N
   mask_j = offs_j < N

   # accumulate
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # pointers to start of batch and d
   left_base = left_ptr + batch * stride_left_batch + d * stride_left_d
   right_base = right_ptr + batch * stride_right_batch + d * stride_right_d
   out_base = out_ptr + batch * stride_out_batch + d * stride_out_d

   # loop over k dimension
   for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
       k_offset = k_tile * BLOCK_K
       offs_k = k_offset + tl.arange(0, BLOCK_K)
       mask_k = offs_k < N
       # load left tile [M, K]
       lhs_ptrs = left_base + (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
       lhs = tl.load(lhs_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)
       # load right tile [K, N] from transposed orientation
       rhs_ptrs = right_base + (offs_k[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_i)
       rhs = tl.load(rhs_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float16)
       # dot product accumulate
       acc += tl.dot(lhs, rhs)

   # store result
   out_ptrs = out_base + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
   tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left = F.linear(x, left_proj_weight)  # shape [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Mask handling
    if config.get("nomask", False) or mask_tensor is None:
        mask = torch.ones_like(input_tensor[...,0:1])  # shape [B,N,N,1]
    else:
        mask = mask_tensor.unsqueeze(-1)
        mask = mask.to(dtype=left.dtype)  # match dtype

    left = left * mask
    right = right * mask

    # Gating
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, hidden = left_h.shape

    # Output tensor
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # Prepare strides for kernel
    stride_left = left_h.stride()
    stride_right = right_h.stride()
    stride_out = out.stride()

    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64  # maybe 64 for better memory locality

    # Compute grid
    num_tiles_i = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_j = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = B * hidden * num_tiles_i * num_tiles_j
    grid = (total_tiles,)

    trimul_kernel[grid](
        left_h, right_h, out,
        B, N, hidden,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # After kernel, out is shape [B,N,N,hidden] (float32)
    # Apply to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # Apply out_gate and final linear
    out = out * out_gate
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)  # output dim = dim

    return out

"""
TriMul forward (outgoing) implementation with a custom Triton kernel.

The heavy part of the TriMul module is the three‑dimensional reduction:
    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]

We treat each hidden dimension `d` (and each batch) as an independent
matrix multiplication L (i×k) × Rᵀ (k×j).  The kernel follows the usual
blocked GEMM pattern (BLOCK_M × BLOCK_N tiles, summed over BLOCK_K).
All linear projections, layer‑norms and gates are performed in PyTorch
(using float32).  The tensors handed to the kernel are cast to float16
to reduce memory traffic, while accumulation is carried out in float32.
The kernel returns a float32 result which is then normalized,
gated and projected back to the original feature dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr,          # *half   left tensor   [B, N, N, D]
    right_ptr,         # *half   right tensor  [B, N, N, D]
    out_ptr,           # *float  output tensor [B, N, N, D]
    B, N, D,           # int32  batch, seq_len, hidden_dim
    # strides for left
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    # strides for right
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    # strides for out
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = Σₖ left[..., i, k, d] * right[..., j, k, d] using a blocked GEMM."""
    pid = tl.program_id(0)

    # Number of tiles in each dimension
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)
    tiles_per_batch_hidden = num_tiles_i * num_tiles_j

    # Map 1‑D pid → (batch, d, i_tile, j_tile)
    group = pid // tiles_per_batch_hidden
    tile_idx = pid % tiles_per_batch_hidden
    batch = group // D
    d = group % D

    i_tile = tile_idx // num_tiles_j
    j_tile = tile_idx % num_tiles_j

    # Offsets inside the tile
    i_offset = i_tile * BLOCK_M
    j_offset = j_tile * BLOCK_N
    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (batch, d)
    left_base = left_ptr + batch * stride_left_batch + d * stride_left_d
    right_base = right_ptr + batch * stride_right_batch + d * stride_right_d
    out_base = out_ptr + batch * stride_out_batch + d * stride_out_d

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_tile * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base \
            + (offs_i[:, None] * stride_left_i) \
            + (offs_k[None, :] * stride_left_k)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16)

        # Load right tile (transposed): shape (BLOCK_K, BLOCK_N)
        # right[b, j, k, d] -> we need (k, j)
        right_ptrs = right_base \
            + (offs_k[:, None] * stride_right_k) \
            + (offs_j[None, :] * stride_right_i)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0,
                        dtype=tl.float16)

        # GEMM accumulation
        acc += tl.dot(left, right)   # (M,N) accumulate in FP32

    # Store the result
    out_ptrs = out_base \
        + (offs_i[:, None] * stride_out_i) \
        + (offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    # Configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    use_nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2. Linear projections (left/right) and masking
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    if not use_nomask:
        # mask shape [B, N, N] → [B, N, N, 1]
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 3. Gating signals
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Prepare half‑precision tensors for the kernel
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    B, N, _, D = left_h.shape
    assert D == hidden_dim, "Hidden dimension mismatch"

    # Output buffer (float32)
    out = torch.empty((B, N, N, D), dtype=torch.float32, device=device)

    # Strides (convert to Python ints)
    stride_left = left_h.stride()
    stride_right = right_h.stride()
    stride_out = out.stride()

    # ------------------------------------------------------------------
    # 5. Triton kernel launch
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_tiles_i = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_j = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = B * D * num_tiles_i * num_tiles_j
    grid = (total_tiles,)

    trimul_kernel[grid](
        left_h,
        right_h,
        out,
        B,
        N,
        D,
        # left strides
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        # right strides
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        # out strides
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        # optional launch config (tuned for A100/B200 like GPUs)
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing: norm → gating → final linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    out = out * out_gate  # element‑wise gating (float32)

    to_out_weight = weights["to_out.weight"]   # shape [dim, hidden_dim]
    out = F.linear(out, to_out_weight)        # final projection → [B, N, N, dim]

    return out
