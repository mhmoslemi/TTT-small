"""
TriMul (outgoing) implementation using Triton.

The core of the operator is the reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is an O(N³·C) contraction over the sequence dimension.
We fuse this reduction into a Triton kernel that processes
tiles of (i, j, d) and iterates over k in small blocks.
All other linear layers and LayerNorms are performed with
torch.nn.functional for simplicity; at least the expensive
N³ reduction is GPU‑accelerated via Triton.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_reduce_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    d_offset,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    Compute a tile of:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    Parameters
    ----------
    left_ptr, right_ptr, out_ptr : pointers to the input/output tensors
    B, N, H : batch size, sequence length, hidden dimension
    stride_* : tensor strides (in elements, not bytes)
    d_offset : start index of the hidden‑dimension block
    BLOCK_* : compile‑time tile sizes
    """
    pid = tl.program_id(0)

    # number of tiles in i and j dimension
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    # decode program id -> (batch, i_tile, j_tile)
    b = pid // (num_i * num_j)
    rem = pid % (num_i * num_j)
    i0 = (rem // num_j) * BLOCK_M
    j0 = (rem % num_j) * BLOCK_N

    # tile offsets
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = (d_offset + offs_h) < H

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over k in blocks
    for k0 in range(0, N, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------- load left[b, i, k, d] -------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + (d_offset + offs_h)[None, None, :] * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        left = left.to(tl.float32)          # convert FP16 → FP32

        # ------- load right[b, j, k, d] -------
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_j[:, None, None] * stride_right_j
            + offs_k[None, :, None] * stride_right_k
            + (d_offset + offs_h)[None, None, :] * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        right = right.to(tl.float32)

        # ------- accumulate outer product over the K block -------
        for kk in range(0, BLOCK_K):
            # each slice is (M, D) and (N, D)
            l = left[:, kk, :]               # (M, D)
            r = right[:, kk, :]              # (N, D)
            # broadcast to (M, N, D) and accumulate
            acc += l[:, None, :] * r[None, :, :]

    # store the result (cast back to FP16)
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + (d_offset + offs_h)[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (binary 0/1)
        - weights      : dict of model weight tensors
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----- LayerNorm on the input -----
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # work in FP16 for the heavy tensors
    x_h = x.to(torch.float16)

    # ----- Linear projections (no bias) -----
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ----- Apply mask (if provided) -----
    if mask is not None:
        # mask shape [B,N,N] -> [B,N,N,1] and cast to FP16
        m = mask.to(left.dtype).unsqueeze(-1)
        left = left * m
        right = right * m

    # ----- Gating tensors -----
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ----- Apply gates -----
    left = left * left_gate
    right = right * right_gate

    # ----- Triton reduction over the sequence dimension -----
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=left.device)

    # tensor strides (in elements)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # tile sizes (tuned for the B200 GPU)
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 16
    BLOCK_D = 32  # hidden‑dimension tile (must be <= hidden_dim, mask handles the rest)

    num_i_tiles = (N + BLOCK_M - 1) // BLOCK_M
    num_j_tiles = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * num_i_tiles * num_j_tiles,)

    # launch the kernel for each hidden‑dim block
    for d_start in range(0, hidden_dim, BLOCK_D):
        trimul_reduce_kernel[grid](
            left,
            right,
            out,
            B,
            N,
            hidden_dim,
            stride_left[0],
            stride_left[1],
            stride_left[2],
            stride_left[3],
            stride_right[0],
            stride_right[1],
            stride_right[2],
            stride_right[3],
            stride_out[0],
            stride_out[1],
            stride_out[2],
            stride_out[3],
            d_start,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # ----- Convert back to FP32 for the remaining layers -----
    out = out.to(torch.float32)

    # ----- Final LayerNorm, gating and projection -----
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # out_gate is FP16 → auto‑upcast to FP32
    out = F.linear(out, weights["to_out.weight"])  # shape -> [B,N,N,dim]

    return out
