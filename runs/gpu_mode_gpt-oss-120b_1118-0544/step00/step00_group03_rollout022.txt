"""
TriMul outgoing kernel implementation.

The kernel computes the core Einstein-sum:
out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d].

The operation is expressed as a batched matrix multiplication over the
sequence dimension k, where each hidden dimension d is treated as an
independent batch. Triton is used to perform the N×N blockwise GEMM
for every (batch, hidden) pair.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)   # combined batch*hidden index
    pid_i = tl.program_id(1)       # block row (i)
    pid_j = tl.program_id(2)       # block col (j)

    # split combined index into batch and hidden-id
    b = pid_batch // H
    h = pid_batch - b * H

    # coordinate offsets
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds
    i_mask = i_off < N
    j_mask = j_off < N

    # base pointers for this (b, h)
    base_left = b * stride_left_batch + h * stride_left_h
    base_right = b * stride_right_batch + h * stride_right_h
    base_out = b * stride_out_batch + h * stride_out_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_off = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # left sub‑matrix (M × K)
        left_ptrs = left_ptr + base_left + i_off[:, None] * stride_left_i + k_off[None, :] * stride_left_k
        # right sub‑matrix (K × N) – note the transposed layout
        right_ptrs = right_ptr + base_right + k_off[:, None] * stride_right_k + j_off[None, :] * stride_right_j

        left = tl.load(left_ptrs,
                       mask=(i_mask[:, None] & k_mask[None, :]),
                       other=0.0,
                       dtype=tl.float16)
        right = tl.load(right_ptrs,
                        mask=(k_mask[:, None] & j_mask[None, :]),
                        other=0.0,
                        dtype=tl.float16)

        acc += tl.dot(left, right)   # fp16×fp16 → fp32 accumulation

    # write back
    out_ptrs = out_ptr + base_out + i_off[:, None] * stride_out_i + j_off[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=(i_mask[:, None] & j_mask[None, :]))

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update.
    data = (input_tensor, mask_tensor, weights, config)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask_tensor, weights, config = data

    # Configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # -------------------------------------------------
    # 1) Layer‑norm over the channel dimension (dim)
    # -------------------------------------------------
    # Using torch.nn.functional.layer_norm which accepts weight & bias
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # shape [B,N,N,dim]

    # -------------------------------------------------
    # 2) Linear projections + optional masking
    # -------------------------------------------------
    # left and right projections (dim -> hidden_dim)
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden_dim]

"""
TriMul outgoing kernel (AlphaFold3).

The core of the TriMul operation is the Einstein‑summation

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` have shape [B, N, N, H] (H = hidden_dim).
For each hidden dimension `d` this is exactly a matrix multiplication
(L_i,k) @ (R_j,k)^T.  The implementation treats every (batch, hidden)
pair as an independent GEMM and evaluates it with a Triton kernel that
uses block‑wise loading, fp16 arithmetic and fp32 accumulation.
All surrounding LayerNorms, linear projections and gating are
performed in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM kernel for the TriMul einsum.
    Each program instance computes a BLOCK_M × BLOCK_N tile of the
    output matrix for a specific (batch, hidden) pair.
    """
    pid_batch = tl.program_id(0)          # combined batch*hidden index
    pid_i = tl.program_id(1)              # tile row index (i)
    pid_j = tl.program_id(2)              # tile column index (j)

    # Recover batch and hidden indices
    b = pid_batch // H
    h = pid_batch - b * H

    # Offsets for the i / j dimensions
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_off < N
    j_mask = j_off < N

    # Base pointers for this (b, h) slice
    base_left = b * stride_left_batch + h * stride_left_h
    base_right = b * stride_right_batch + h * stride_right_h
    base_out = b * stride_out_batch + h * stride_out_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Reduction over the k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_off = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # left: (M × K) block
        left_ptrs = left_ptr + base_left \
                    + i_off[:, None] * stride_left_i \
                    + k_off[None, :] * stride_left_k
        # right: (K × N) block (note the transposed layout)
        right_ptrs = right_ptr + base_right \
                     + k_off[:, None] * stride_right_k \
                     + j_off[None, :] * stride_right_j

        left = tl.load(left_ptrs,
                       mask=(i_mask[:, None] & k_mask[None, :]),
                       other=0.0,
                       dtype=tl.float16)
        right = tl.load(right_ptrs,
                        mask=(k_mask[:, None] & j_mask[None, :]),
                        other=0.0,
                        dtype=tl.float16)

        acc += tl.dot(left, right)   # fp16 × fp16 → fp32 accumulation

    # Store the computed tile
    out_ptrs = out_ptr + base_out \
               + i_off[:, None] * stride_out_i \
               + j_off[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=(i_mask[:, None] & j_mask[None, :]))
