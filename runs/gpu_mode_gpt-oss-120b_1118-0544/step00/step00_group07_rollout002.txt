"""
Optimized Triton implementation of the “outgoing” Triangle Multiplicative Update
(TriMul) used in AlphaFold‑3. The heavy N³ part (einsum over the middle index)
is computed with a custom Triton GEMM‑style kernel that treats each hidden
channel as an independent batch dimension. All other linear layers,
LayerNorms and gating are performed with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,          # const *float  (B,N,N,H)  left projection
    right_ptr,         # const *float  (B,N,N,H)  right projection
    out_ptr,           # *float        (B,N,N,H)  output buffer
    B, N, H,           # int32 scalar batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    where left/right have shape (B,N,N,H). The kernel fuses the reduction
    over k using a blocked GEMM. Each program processes a tile of size
    BLOCK_M × BLOCK_N for a single (b,h) pair.
    """
    pid_batch = tl.program_id(0)   # combined batch index = b * H + h
    pid_i = tl.program_id(1)       # tile index along seq_len i
    pid_j = tl.program_id(2)       # tile index along seq_len j

    # Decode batch and hidden channel
    b = pid_batch // H
    h = pid_batch % H

    # Offsets for the i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_idx = k_start + tl.arange(0, BLOCK_K)

        # ------- load left tile (M × K) -------
        lhs_ptr = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + k_idx[None, :] * stride_left_k
        )
        lhs_mask = mask_i[:, None] & (k_idx[None, :] < N)
        lhs = tl.load(lhs_ptr, mask=lhs_mask, other=0.0)

        # ------- load right tile (N × K) -------
        rhs_ptr = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_j[:, None] * stride_right_j
            + k_idx[None, :] * stride_right_k
        )
        rhs_mask = mask_j[:, None] & (k_idx[None, :] < N)
        rhs = tl.load(rhs_ptr, mask=rhs_mask, other=0.0)

        # Accumulate dot product across K
        acc += tl.dot(lhs, tl.trans(rhs))

    # ------- store result -------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    store_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=store_mask)

    """
    Forward pass of TriMul (outgoing) using a fused Triton kernel for the
    N³ einsum. Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype float32).
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # config
    dim = config["dim"]                     # input/output feature dim
    hidden_dim = config["hidden_dim"]       # inner hidden dimension
    eps = 1e-5

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    right_gate_weight = weights["right_gate.weight"]

    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight).contiguous()
    right = torch.nn.functional.linear(x, right_proj_weight).contiguous()

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dimension)
    # -----------------------------------------------------------------
    if not config.get("nomask", True):
        mask_f = mask_tensor.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4) Gating
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core TriMul (einsum) via Triton
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape: [B, N, N, hidden_dim]
    out = torch.empty_like(left)

    # Strides (int64)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    total_batch = B * hidden_dim  # each (b, h) pair is a separate GEMM

    # Block sizes – tuned for typical A100/B200 memory hierarchy
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64

    grid = (
        total_batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # Ensure kernel completion before using the result
    torch.cuda.synchronize()

    # -----------------------------------------------------------------
    # 6) Output LayerNorm + gating
    # -----------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps
    )
    out_norm = out_norm * out_gate

    # -----------------------------------------------------------------
    # 7) Final linear projection back to original dimension
    # -----------------------------------------------------------------
    output = torch.nn.functional.linear(out_norm, to_out_weight)

    return output
