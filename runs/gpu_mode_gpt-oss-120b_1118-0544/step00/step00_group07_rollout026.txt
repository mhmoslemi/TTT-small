    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # Norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # shape [dim], bias same shape.
    # Use torch.nn.functional.layer_norm
    # Because we need to apply over dim axis (-1)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), norm_weight, norm_bias)

    # linear proj - use torch.nn.functional.linear (weight shape [out_features, in_features])
    left_proj_weight = weights["left_proj.weight"]   # shape [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"] # shape [hidden_dim, dim]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]    # shape [dim, hidden_dim]

    # Compute left,right
    # Input shape [B,N,N,C]
    # Use torch.nn.functional.linear applied on last dim
    left = torch.nn.functional.linear(x, left_proj_weight) # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask
    if mask is not None:
        # mask shape [B,N,N] (bool?) convert to float
        mask_f = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    out = torch.nn.functional.linear(out, to_out_weight)

def tri_mul_triton(left, right):
    # left,right shape [B,N,N,H] torch.float32, contiguous.
    B,N,N2,H = left.shape
    assert N2 == N
    # allocate output
    out = torch.empty_like(left)
    # get strides
    stride_b = left.stride(0)
    stride_i = left.stride(1)
    stride_k = left.stride(2)
    stride_h = left.stride(3)
    # same for right/out
    # Create grid
    BLOCK_I = 16
    BLOCK_J = 16
    BLOCK_K = 32
    BLOCK_H = 32 # hidden dim chunk
    num_i = (N + BLOCK_I - 1)//BLOCK_I
    num_j = (N + BLOCK_J - 1)//BLOCK_J
    # if H is not multiple of BLOCK_H, we need to handle remainder via mask.
    grid = (num_i, num_j, B)  # each program processes all hidden blocks internally
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_b, stride_i, stride_k, stride_h,
        # out strides
        stride_b, stride_i, stride_k, stride_h,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
    )
    return out

"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor x [B,N,N,C].
2. Project to hidden dimension H with two linear layers (left/right) and apply
   the optional mask.
3. Apply per‑position sigmoid gates (left, right, out) obtained from three
   additional linear layers.
4. Core operation:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched N×N matrix multiply for each hidden channel d.
   It is implemented as a tiled Triton kernel that iterates over blocks of
   i, j and k while accumulating over hidden channels in chunks.
5. Layer‑norm over the hidden dimension, element‑wise out‑gate, and final
   linear projection back to C.

The kernel processes tiles of size (BLOCK_I, BLOCK_J, BLOCK_H) and loops over
k‑blocks of size BLOCK_K.  All arithmetic is performed in float32 for
numerical stability; the result is stored as float32.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_batch, stride_i, stride_j, stride_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].

    left/right/out are stored as [B, N, N, H] (contiguous, last dim H).
    """
    pid_b = tl.program_id(2)          # batch dimension
    pid_i = tl.program_id(0)          # tile over axis i
    pid_j = tl.program_id(1)          # tile over axis j

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # generate index vectors (broadcastable shapes)
    i_idx = i_start + tl.arange(0, BLOCK_I)[:, None, None]   # (BLOCK_I,1,1)
    j_idx = j_start + tl.arange(0, BLOCK_J)[None, :, None]   # (1,BLOCK_J,1)

    i_valid = i_start + tl.arange(0, BLOCK_I) < N
    j_valid = j_start + tl.arange(0, BLOCK_J) < N

    num_h = tl.cdiv(H, BLOCK_H)
    num_k = tl.cdiv(N, BLOCK_K)

    for h_iter in range(num_h):
        h_offset = h_iter * BLOCK_H
        h_idx = h_offset + tl.arange(0, BLOCK_H)[None, None, :]   # (1,1,BLOCK_H)
        h_valid = h_idx < H

        # accumulator for this (i,j) tile and this hidden‑dim block
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        for k_iter in range(num_k):
            k_offset = k_iter * BLOCK_K
            k_idx = k_offset + tl.arange(0, BLOCK_K)[None, :, None]   # (1,BLOCK_K,1)

            # ----- left tile (i,k,d) -----
            left_ptrs = (
                left_ptr
                + pid_b * stride_batch
                + i_idx * stride_i
                + k_idx * stride_j
                + h_idx * stride_h
            )
            mask_l = (i_idx < N) & (k_idx < N) & h_valid
            left_tile = tl.load(left_ptrs, mask=mask_l, other=0.0)   # (BLOCK_I, BLOCK_K, BLOCK_H)

            # ----- right tile (j,k,d) -----
            right_ptrs = (
                right_ptr
                + pid_b * stride_batch
                + j_idx * stride_i
                + k_idx * stride_j
                + h_idx * stride_h
            )
            mask_r = (j_idx < N) & (k_idx < N) & h_valid
            right_tile = tl.load(right_ptrs, mask=mask_r, other=0.0)  # (BLOCK_J, BLOCK_K, BLOCK_H)

            # ----- accumulate over k -----
            for kk in range(BLOCK_K):
                # left_slice: (BLOCK_I, BLOCK_H)
                # right_slice: (BLOCK_J, BLOCK_H)
                left_slice = left_tile[:, kk, :]      # (BLOCK_I, BLOCK_H)
                right_slice = right_tile[:, kk, :]    # (BLOCK_J, BLOCK_H)

                # broadcast to (BLOCK_I, BLOCK_J, BLOCK_H) and accumulate
                acc += left_slice[:, None, :] * right_slice[None, :, :]

        # ----- write back -----
        out_ptrs = (
            out_ptr
            + pid_b * stride_batch
            + i_idx * stride_i
            + j_idx * stride_j
            + h_idx * stride_h
        )
        store_mask = i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :]
        tl.store(out_ptrs, acc, mask=store_mask)


def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton implementation of the core triple‑multiply.
    left / right: [B, N, N, H] (contiguous, float32)
    Returns out of the same shape.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, N2, H = left.shape
    assert N == N2, "pairwise dimensions must be equal"
    out = torch.empty_like(left)

    # strides for a contiguous tensor [B, N, N, H]
    stride_batch, stride_i, stride_j, stride_h = left.stride()

    # tile sizes – chosen to fit in shared memory & registers on A100/B200
    BLOCK_I = 16
    BLOCK_J = 16
    BLOCK_K = 32
    BLOCK_H = 32  # works for H up to 384 (12 blocks)

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
        B,
    )
    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_batch,
        stride_i,
        stride_j,
        stride_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"]) # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (shape [B,N,N] -> broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(left.dtype)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core triple‑multiply via Triton
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out = _tri_mul_triton(left, right)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6. Post‑processing (LayerNorm, out‑gate, final projection)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
