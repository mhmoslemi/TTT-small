"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

The kernel computes the core tensor contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
for each batch b and hidden dimension h.
All surrounding operations (LayerNorms, linear projections, gating, final
projection) are performed in PyTorch; only the cubic‐complexity reduction
is executed in Triton for speed.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def tri_mul_kernel(
    left_ptr,          # *float16
    right_ptr,         # *float16
    out_ptr,           # *float16
    B, H, N,           # runtime constants
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_i, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out = left @ right^T for each (b, h) pair."""
    pid_bn = tl.program_id(0)        # combined batch & head index
    pid_m = tl.program_id(1)         # block row (i)
    pid_n = tl.program_id(2)         # block col (j)

    # Decode batch and hidden index
    b = pid_bn // H
    h = pid_bn % H

    # Offsets of the tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (b, h)
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        cur_k = kk * BLOCK_K + offs_k
        mask_k = cur_k < N

        # Load a tile of left: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + cur_k[None, :] * stride_l_k
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load a tile of right: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + cur_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        b_ = tl.load(right_ptrs,
                     mask=mask_k[:, None] & mask_j[None, :],
                     other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b_)

    # Store the result
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Unpack config
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load weights (all are expected on the same device as input)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # Normalization of the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (float32)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Optional mask application
    # ------------------------------------------------------------------
    if (mask is not None) and (not nomask):
        # mask shape: [B, N, N] -> [B, N, N, 1]
        mask_ = mask.unsqueeze(-1).to(dtype=left.dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 and rearrange to [B, H, N, N] for the kernel
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).permute(0, 3, 1, 2).contiguous()
    right_fp16 = right.to(torch.float16).permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_fp16.shape  # after permute H == hidden_dim

    # Output buffer
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # Choose block sizes (tuned for typical N up to 1024)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (batch*head, rows, cols)
    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Launch kernel
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        H,
        N,
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        left_fp16.stride(3),
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Convert back to float32 and restore original layout [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm over hidden_dim, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
```
