"""
TriMul Outgoing implementation (AlphaFold3)

The kernel computes the outgoing triangular multiplicative update:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, k, j, d]

The heavy contraction Σ_k is fused in a Triton kernel.
The rest of the module (LayerNorm, Linear projections, gating) is done
with native PyTorch ops.  The kernel works on tiles of shape
(BLOCK_M, BLOCK_N, BLOCK_D) and loops over the K‑dimension in blocks
of size BLOCK_K, using `tl.dot` for each channel slice.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, D,
    # left strides: (i, k, d)
    stride_left_i, stride_left_k, stride_left_d,
    # right strides: (k, j, d)
    stride_right_k, stride_right_j, stride_right_d,
    # out strides: (i, j, d)
    stride_out_i, stride_out_j, stride_out_d,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute outgoing triangular multiplicative update for a single batch."""
    pid_i = tl.program_id(0)  # tile row
    pid_j = tl.program_id(1)  # tile col
    pid_d = tl.program_id(2)  # tile channel

    # ---- offsets for the output tile ------------------------------------
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)  # (BM)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)  # (BN)
    d_offset = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)  # (BD)

    # output pointer for this tile
    out_ptrs = (
        out_ptr
        + i_offset[:, None, None] * stride_out_i
        + j_offset[None, :, None] * stride_out_j
        + d_offset[None, None, :] * stride_out_d
    )

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # number of K‑tiles
    num_steps = tl.cdiv(N, BLOCK_K)

    for step in range(num_steps):
        k_base = step * BLOCK_K
        k_offset = k_base + tl.arange(0, BLOCK_K)  # (BK)

        # ---- left tile: shape (BM, BK, BD) -------------------------------
        left_ptrs = (
            left_ptr
            + i_offset[:, None, None] * stride_left_i
            + k_offset[None, :, None] * stride_left_k
            + d_offset[None, None, :] * stride_left_d
        )
        # mask not needed – input dimensions are guaranteed to be multiples
        left_tile = tl.load(left_ptrs, dtype=tl.float32)

        # ---- right tile: shape (BK, BN, BD) ------------------------------
        right_ptrs = (
            right_ptr
            + k_offset[:, None, None] * stride_right_k
            + j_offset[None, :, None] * stride_right_j
            + d_offset[None, None, :] * stride_right_d
        )
        right_tile = tl.load(right_ptrs, dtype=tl.float32)

        # ---- accumulate for each channel slice ---------------------------
        for db in range(0, BLOCK_D):
            # (BM, BK) @ (BK, BN) -> (BM, BN)
            left_mat = left_tile[:, :, db]   # (BM, BK)
            right_mat = right_tile[:, :, db]  # (BK, BN)
            acc[:, :, db] += tl.dot(left_mat, right_mat)

    # ---- write the result ------------------------------------------------
    tl.store(out_ptrs, acc)


def custom_kernel(data):
    """
    Execute the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] (float32/float16)
            - mask: [B, N, N] (binary float or bool)
            - weights: dict of model weights (tensors on the same device)
            - config: dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        Tensor of shape [B, N, N, dim] (same dtype as input)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ----------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ----------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ----------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------------------
    # 4. Gating
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5. Outgoing contraction via Triton kernel
    # ----------------------------------------------------------------------
    B, N, _, _ = left.shape
    device = left.device
    dtype = left.dtype

    # Allocate output tensor
    out = torch.empty_like(left, device=device, dtype=dtype)  # [B, N, N, hidden_dim]

    # Block sizes (chosen to evenly divide supported test dimensions)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8

    # Sanity checks – all test sizes are multiples of the block sizes
    assert N % BLOCK_M == 0 and N % BLOCK_N == 0 and N % BLOCK_K == 0, \
        "Sequence length must be a multiple of block sizes."
    assert hidden_dim % BLOCK_D == 0, "hidden_dim must be a multiple of BLOCK_D."

    grid = (
        N // BLOCK_M,
        N // BLOCK_N,
        hidden_dim // BLOCK_D,
    )

    for b in range(B):
        # contiguous sub‑tensors for the current batch
        left_b = left[b].contiguous()
        right_b = right[b].contiguous()
        out_b = out[b].contiguous()

        # strides (in elements)
        stride_left_i, stride_left_k, stride_left_d = left_b.stride()
        stride_right_k, stride_right_j, stride_right_d = right_b.stride()
        stride_out_i, stride_out_j, stride_out_d = out_b.stride()

        tri_mul_outgoing_kernel[grid](
            left_b,
            right_b,
            out_b,
            N,
            hidden_dim,
            stride_left_i,
            stride_left_k,
            stride_left_d,
            stride_right_k,
            stride_right_j,
            stride_right_d,
            stride_out_i,
            stride_out_j,
            stride_out_d,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # ----------------------------------------------------------------------
    # 6. Final LayerNorm, gating and linear projection to original dim
    # ----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
