    stride_right_b, stride_right_i, stride_right_k, stride_right_h,

    j_start = pid_j * BLOCK_N

        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    b = pid_bh // H
    h = pid_bh % H

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_blocks):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block [M, K]
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            h * stride_left_h + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right block transposed: we need shape [K, N] where rows are k, cols are j
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            h * stride_right_h + \
            offs_k[:, None] * stride_right_k + \
            offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # matrix multiply and accumulate
        acc += tl.dot(left, right)

    # write out
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        h * stride_out_h + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul module (outgoing version).
    Performs LayerNorm, linear projections, gating, a batched matrix multiplication
    (the N^3 contraction) using a Triton kernel, followed by a second LayerNorm,
    gating and a final linear projection.
    """
    input_tensor, mask, weights, config = data
    # unpack config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # optional: eps
    eps = 1e-5

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm over the last dimension (dim)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections (no bias)
    left = F.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Apply mask if provided
    if mask is not None:
        # mask shape [B,N,N] -> [B,N,N,1]
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gating values (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous tensors for Triton
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape
    # allocate output buffer
    out = torch.empty_like(left)  # same shape as left/right

    # Get strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Block sizes (tuned for B200, FP32 compute)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    # Launch Triton kernel
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Second LayerNorm over hidden dimension
    out_norm = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply output gate
    out_norm = out_norm * out_gate

    # Final linear projection back to original dim
    output = F.linear(out_norm, to_out_weight)

    return output

Triton‑accelerated forward pass of the “outgoing” TriMul operator.

The heavy N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication (left @ right.T) for each
hidden slice `d`.  All other operations (LayerNorms, linear projections,
gating, mask handling) are performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,               # const * left  [B,N,N,H]
    right_ptr,              # const * right [B,N,N,H]
    out_ptr,                # * out         [B,N,N,H]
    B, N, H,                # scalar sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out = left @ right.T per batch and hidden channel.
    left : (B, N, N, H)   -> rows i, reduction k
    right: (B, N, N, H)   -> rows j, reduction k (transposed on the fly)
    out  : (B, N, N, H)   -> (i,j) for each hidden h
    """
    pid_i = tl.program_id(0)          # block row in the output matrix
    pid_j = tl.program_id(1)          # block column
    pid_bh = tl.program_id(2)         # combines batch and hidden index

    b = pid_bh // H                   # batch index
    h = pid_bh % H                    # hidden index

    # starting coordinates of the block
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # absolute offsets inside each dimension
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (more precise than FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over the k dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_blocks):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k]  -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right[j,k] transposed on the fly => shape (BLOCK_K, BLOCK_N)
        # rows = k (use stride_right_k), cols = j (use stride_right_i)
        right_ptrs = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            offs_k[:, None] * stride_right_k + \
            offs_j[None, :] * stride_right_i
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # GEMM of shapes (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(left, right)

    # write the result back
    out_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the TriMul module (outgoing version).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (may be None)
        - weights      : dict of torch.Tensors containing all model weights
        - config       : dict with keys "dim", "hidden_dim", etc.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim], dtype torch.float32.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------- unpack weights -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------- first LayerNorm over the input dim -----------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B,N,N,dim]

    # ------- linear projections (no bias) -----------------------------------
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------- optional mask ---------------------------------------------------
    if mask is not None:
        # mask shape [B,N,N] -> [B,N,N,1] for broadcasting
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------- gating ---------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------- Triton kernel for the N³ contraction --------------------------
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, _ = left.shape
    out = torch.empty_like(left)  # allocate output buffer

    # strides in elements (not bytes)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # block sizes – tuned for the target GPU (B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        s_lb,
        s_li,
        s_lk,
        s_lh,
        s_rb,
        s_ri,
        s_rk,
        s_rh,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------- second LayerNorm over hidden dimension -------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------- output gating --------------------------------------------------
    out = out * out_gate

    # ------- final linear back to original dim ------------------------------
    output = F.linear(out, to_out_weight)  # [B,N,N,dim]

    return output
