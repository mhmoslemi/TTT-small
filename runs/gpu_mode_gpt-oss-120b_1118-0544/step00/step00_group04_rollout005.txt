def mask_gate_kernel(
        src_ptr,
        gate_ptr,
        mask_ptr,
        dst_ptr,
        B: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
        stride_src_b, stride_src_i, stride_src_j, stride_src_h,
        stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
        stride_mask_b, stride_mask_i, stride_mask_j,
        BLOCK_SIZE: tl.constexpr):
    idx = tl.program_id(0) * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask_idx = idx < total
    # compute coordinates
    # Use integer division and modulo
    h = idx % H
    seq = idx // H
    j = seq % N
    i = (seq // N) % N
    b = seq // (N * N)
    # offsets in elements:
    src_offset = b * stride_src_b + i * stride_src_i + j * stride_src_j + h * stride_src_h
    gate_offset = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # load values
    src = tl.load(src_ptr + src_offset, mask=mask_idx, other=0.0)
    gate = tl.load(gate_ptr + gate_offset, mask=mask_idx, other=0.0)
    mask = tl.load(mask_ptr + mask_offset, mask=mask_idx, other=0.0)
    # broadcast mask across H dimension, none needed because mask is scalar per position
    # Multiply
    dst = src * mask * gate
    tl.store(dst_ptr + src_offset, dst, mask=mask_idx)

"""
TriMul forward (outgoing) implementation with a Triton elementwise kernel.

The operation is:
1. LayerNorm over the last dimension.
2. Linear projections (left/right) and gating linear layers.
3. Apply mask, then multiply each projection by its gate (mask * gate).
   This elementwise step is fused in a Triton kernel.
4. For each hidden channel compute A @ Bᵀ where
   A = left_proj (B,N,N,hidden) and B = right_proj (B,N,N,hidden).
   This is performed with a batched GEMM (torch.bmm) after reshaping.
5. LayerNorm the result, apply the output gate, and a final linear projection.

Only the mask‑gate fusion is done in Triton; the heavy MatMul uses
torch's highly‑optimized batched GEMM.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def mask_gate_kernel(
    src_ptr,                # pointer to source tensor (left/right proj)
    gate_ptr,               # pointer to gate tensor (sigmoid)
    mask_ptr,               # pointer to mask tensor (broadcasted)
    dst_ptr,                # pointer to destination tensor
    B, N, H,                # dimensions: batch, seq_len, hidden_dim
    stride_src_b, stride_src_i, stride_src_j, stride_src_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_SIZE: tl.constexpr
):
    """Fuse: dst = src * mask * gate  (mask broadcasted over hidden dim)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # decompose linear index into (b,i,j,h)
    h = offs % H
    seq = offs // H
    j = seq % N
    i = (seq // N) % N
    b = seq // (N * N)

    src_offset  = b * stride_src_b  + i * stride_src_i  + j * stride_src_j  + h * stride_src_h
    gate_offset = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    src_val  = tl.load(src_ptr  + src_offset,  mask=mask, other=0.0)
    gate_val = tl.load(gate_ptr + gate_offset, mask=mask, other=0.0)
    m_val    = tl.load(mask_ptr + mask_offset, mask=mask, other=0.0)   # scalar per (b,i,j)

    out = src_val * m_val * gate_val
    tl.store(dst_ptr + src_offset, out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool/int/float)
            - weights_dict: dict with all linear / layer‑norm weights.
            - config_dict: dict containing "dim" and "hidden_dim".

    Returns:
        torch.Tensor of shape [B, N, N, dim].
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. LayerNorm over the last dimension
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 3. Gating (sigmoid) + mask fusion via Triton
    # -----------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # ensure tensors are contiguous for stride calculation
    left      = left.contiguous()
    right     = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    out_gate  = out_gate.contiguous()

    # mask handling
    if mask_tensor is None:
        mask = torch.ones((B, N, N), dtype=input_tensor.dtype, device=input_tensor.device)
    else:
        mask = mask_tensor.to(input_tensor.dtype)
    mask = mask.contiguous()

    # allocate fused results
    left_fused  = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    # Triton launch parameters
    BLOCK_SIZE = 1024
    total_elems = B * N * N * hidden_dim
    num_blocks = (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE

    # stride helpers (in element units)
    s_l_b, s_l_i, s_l_j, s_l_h = left.stride()
    s_g_l_b, s_g_l_i, s_g_l_j, s_g_l_h = left_gate.stride()

    s_r_b, s_r_i, s_r_j, s_r_h = right.stride()
    s_g_r_b, s_g_r_i, s_g_r_j, s_g_r_h = right_gate.stride()

    s_m_b, s_m_i, s_m_j = mask.stride()

    # fuse left
    mask_gate_kernel[(num_blocks,)](
        left.data_ptr(),
        left_gate.data_ptr(),
        mask.data_ptr(),
        left_fused.data_ptr(),
        B, N, hidden_dim,
        s_l_b, s_l_i, s_l_j, s_l_h,
        s_g_l_b, s_g_l_i, s_g_l_j, s_g_l_h,
        s_m_b, s_m_i, s_m_j,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # fuse right
    mask_gate_kernel[(num_blocks,)](
        right.data_ptr(),
        right_gate.data_ptr(),
        mask.data_ptr(),
        right_fused.data_ptr(),
        B, N, hidden_dim,
        s_r_b, s_r_i, s_r_j, s_r_h,
        s_g_r_b, s_g_r_i, s_g_r_j, s_g_r_h,
        s_m_b, s_m_i, s_m_j,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # -----------------------------------------------------------------
    # 4. Tri‑multiplicative reduction: out_ijk = Σ_k left_ik * right_jk
    #    This is a batched matmul: left @ rightᵀ for each hidden channel.
    # -----------------------------------------------------------------
    # reshape to (B*hidden, N, N) to use torch.bmm
    left_perm  = left_fused.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right_fused.permute(0, 3, 1, 2).contiguous()  # [B, hidden, N, N]

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)        # (B*H, N, N)
    right_mat = right_perm.view(BH, N, N)       # (B*H, N, N)

    # out_mat = left_mat @ right_matᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # restore [B, N, N, hidden]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5. Second LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    # apply output gate
    out_gated = out_norm * out_gate

    # final linear projection back to original dim
    to_out_weight = weights["to_out.weight"]
    out_final = torch.nn.functional.linear(out_gated, to_out_weight)  # [B,N,N,dim]

    return out_final
