def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype
    
    # LayerNorm - use weight and bias
    eps = 1e-5
    # Compute mean/var across last dim (dim)
    # layernorm weight shape [dim], bias shape [dim]
    # compute normalized: (x - mean) / sqrt(var+eps) * weight + bias
    
    # Input shape [B,N,N,dim]
    B,N,_ = input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2] # N same for second dim

    # Compute mean over last dimension dim
    x = input_tensor
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]
    
    # Linear projections without bias
    # left_proj(x_norm) = x_norm @ left_proj_weight.T (since weight shape hidden_dim x dim)
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])  # shape [B,N,N,hidden]
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    
    # Apply mask (if not nomask)
    if not config.get("nomask", False):
        # mask shape [B,N,N]; broadcast to hidden
        mask_unsq = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_unsq
        right = right * mask_unsq
    
    # Gating
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()
    
    left = left * left_gate
    right = right * right_gate
    
    # Core TriMul via Triton
    # Cast to fp16 for compute (if input dtype bigger)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out_fp32 = torch.empty_like(left, dtype=torch.float32)  # shape B,N,N,hidden
    
    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    # Cast strides to ints
    # left stride in elements
    # Need .stride()
    l_strides = left_fp16.stride()
    r_strides = right_fp16.stride()
    o_strides = out_fp32.stride()

import torch
import triton
import triton.language as tl

@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    using block-sparse matrix multiplication.
    `left_ptr`, `right_ptr`, `out_ptr` point to tensors of shape
    (B, N, N, H) with strides as supplied.
    """
    pid_batch = tl.program_id(0)  # combined batch * hidden index
    pid_i = tl.program_id(1)      # block row
    pid_j = tl.program_id(2)      # block column

    # recover batch and hidden indices
    b = pid_batch // H
    h = pid_batch % H

    # base offsets for each tensor
    offset_left = b * stride_left_b + h * stride_left_h
    offset_right = b * stride_right_b + h * stride_right_h
    offset_out = b * stride_out_b + h * stride_out_h

    # block coordinates
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # ranges inside the block
    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)

    # masks for out-of-bounds rows/cols
    mask_i = i < N
    mask_j = j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k blocks
    num_k = tl.cdiv(N, BLOCK_K)

    for k_iter in range(0, num_k):
        k_base = k_iter * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + offset_left + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)

        # right tile (transposed): shape (BLOCK_K, BLOCK_N)
        # we load right[j,k] but we need (k,j) -> swap strides
        b_ptrs = right_ptr + offset_right + k[:, None] * stride_right_k + j[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0, dtype=tl.float16)

        acc += tl.dot(a, b)

    # store result
    out_ptrs = out_ptr + offset_out + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    # Configuration

    hidden_dim = config["hidden_dim"]

    eps = 1e-5

    x = input_tensor

    x_norm = (x - mean) / torch.sqrt(var + eps)

    right = right * right_gate

    # Convert to fp16 for compute to reduce bandwidth, keep fp32 accumulator.
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # strides (in elements)
    s_left = left_fp16.stride()
    s_right = right_fp16.stride()
    s_out = out_fp32.stride()

    # Blocking parameters (tuned for B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

            triton.cdiv(N, BLOCK_M),

        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    eps_out = 1e-5
    out_mean = out.mean(dim=-1, keepdim=True)
    out_var = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - out_mean) / torch.sqrt(out_var + eps_out)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # --------- Gate and final linear ----------
    out_norm = out_norm * out_gate
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output

"""
Outgoing Triangle Multiplicative Update (TriMul) kernel.

The core operation is:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

where `left` and `right` have shape [B, N, N, H] (H = hidden_dim).
We implement this reduction with a Triton kernel that treats each (b, h)
pair as an independent program and performs a blocked matrix multiplication
over the N×N dimensions.  All surrounding LayerNorms, linear projections,
and gating are performed with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
    left, right, out have shape (B, N, N, H) in row‑major order.
    The kernel is launched with a 3‑D grid:
        axis 0: combined batch * hidden index (pid_batch)
        axis 1: block row of the output matrix (pid_i)
        axis 2: block column of the output matrix (pid_j)
    """
    pid_batch = tl.program_id(0)      # combined b * H
    pid_i = tl.program_id(1)          # block row
    pid_j = tl.program_id(2)          # block column

    # Recover batch and hidden indices
    b = pid_batch // H
    h = pid_batch % H

    # Base offsets for this (b, h) pair
    off_left = b * stride_left_b + h * stride_left_h
    off_right = b * stride_right_b + h * stride_right_h
    off_out = b * stride_out_b + h * stride_out_h

    # Block coordinates for i and j dimensions
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    # Out‑of‑bounds masks for rows / columns
    mask_i = i < N
    mask_j = j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_iter in range(0, num_k):
        k0 = k_iter * BLOCK_K
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Load A tile: left[b, i, k, h]  -> shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + off_left + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float32)

        # Load B tile transposed: we need right[b, j, k, h] but as (k, j)
        b_ptrs = right_ptr + off_right + k[:, None] * stride_right_k + j[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0,
                    dtype=tl.float32)

        # Perform the dot‑product for this k‑tile
        acc += tl.dot(a, b)

    # Store the result
    out_ptrs = out_ptr + off_out + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    B, N, _, _ = input_tensor.shape  # input shape [B, N, N, dim]

    eps = 1e-5

    # ---------- LayerNorm on the input (dim) ----------
    x = input_tensor
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # ---------- Linear projections (no bias) ----------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> broadcast to hidden dim
        mask_exp = mask.unsqueeze(-1).to(x_norm.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gating ----------
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Core TriMul using Triton ----------
    # Ensure contiguous tensors for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left, dtype=torch.float32).contiguous()

    # Strides (in elements, not bytes)
    s_left = left.stride()    # (B, N, N, H)
    s_right = right.stride()
    s_out = out.stride()

    # Blocking parameters – tuned for the B200 GPU
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (batch*hidden, N/BLOCK_M, N/BLOCK_N)
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    triton_trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # out now holds the result of the einsum (... i k h, ... j k h -> ... i j h)

    # ---------- LayerNorm on hidden dimension ----------
    eps_h = 1e-5
    out_mean = out.mean(dim=-1, keepdim=True)
    out_var = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - out_mean) / torch.sqrt(out_var + eps_h)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # ---------- Gate and final linear projection ----------
    out_norm = out_norm * out_gate
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output
