import torch

import torch.nn.functional as F

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm on input
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # Linear projections
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    if not nomask:
        # mask shape [B,N,N] -> [B,N,N,1]
        mask = mask_tensor.to(dtype).unsqueeze(-1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Rearrange for batched matmul over hidden dim
    # permute to [B, hidden, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    # flatten batch and hidden
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # transpose right for multiplication
    right_T = right_flat.transpose(1, 2).contiguous()

    # custom batched matmul kernel
    out_flat = batched_matmul(left_flat, right_T)  # returns shape (B*H, N, N)

    # reshape back
    out_hidden = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # to_out_norm
    out_norm = F.layer_norm(out_hidden,
                            normalized_shape=(hidden_dim,),
                            weight=weights["to_out_norm.weight"],
                            bias=weights["to_out_norm.bias"])

    # apply out gate
    out_gate = out_gate  # shape [B,N,N,hidden]
    out_norm = out_norm * out_gate

    # final linear projection to dim
    out = F.linear(out_norm, weights["to_out.weight"])

    return out

"""
custom_kernel: Triton‑accelerated forward pass of the TriMul “outgoing” module.

Algorithm
---------
1. Layer‑norm the input tensor (last dim = dim).
2. Compute three linear projections (left, right, gates) with weight tensors
   supplied in `weights`.  Gating tensors are passed through sigmoid.
3. Apply the optional mask (broadcast over the hidden dimension) and element‑wise
   gates.
4. For each hidden channel d compute
        out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
   which is equivalent to a batched matrix multiply
        out_d = left_d @ right_dᵀ .
   This is performed by a custom Triton kernel (`batched_matmul_kernel`).
5. Layer‑norm the result over the hidden dimension, multiply by the output gate
   and apply the final linear projection back to `dim`.

Only the heavy N³ reduction (step 4) is implemented in Triton; all other
operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched matrix multiplication C = A @ Bᵀ
    A : [batch, M, K]
    B : [batch, K, N]   (already transposed)
    C : [batch, M, N]
    """
    pid = tl.program_id(0)
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_pid_m * num_pid_n

    batch_id = pid // tiles_per_batch
    tile_id = pid % tiles_per_batch
    pid_m = tile_id // num_pid_n
    pid_n = tile_id % num_pid_n

    # Offsets inside the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = A_ptr + batch_id * stride_Ab
    b_batch = B_ptr + batch_id * stride_Bb
    c_batch = C_ptr + batch_id * stride_Cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = kk * BLOCK_K + offs_k
        mask_k = cur_k < K

        # Load A tile (M x K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_Am + cur_k[None, :] * stride_Ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (K x N) – note B is already transposed (K rows, N cols)
        b_ptrs = b_batch + (cur_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Write result
    c_ptrs = c_batch + (offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def batched_matmul(A: torch.Tensor, B_T: torch.Tensor, *, BLOCK_M=64, BLOCK_N=64, BLOCK_K=64):
    """
    Wrapper for batched_matmul_kernel.
    A : [batch, M, K]      (float32)
    B_T: [batch, K, N]     (float32)   – already transposed version of right matrix
    Returns C : [batch, M, N] (float32)
    """
    assert A.is_contiguous() and B_T.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B_T.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    grid = lambda meta: (batch * tl.cdiv(M, meta['BLOCK_M']) * tl.cdiv(N, meta['BLOCK_N']),)

    batched_matmul_kernel[grid](
        A_ptr=A,
        B_ptr=B_T,
        C_ptr=C,
        M=M,
        N=N,
        K=K,
        stride_Ab=A.stride(0), stride_Am=A.stride(1), stride_Ak=A.stride(2),
        stride_Bb=B_T.stride(0), stride_Bk=B_T.stride(1), stride_Bn=B_T.stride(2),
        stride_Cb=C.stride(0), stride_Cm=C.stride(1), stride_Cn=C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a custom Triton kernel
    for the N³ reduction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input layer‑norm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Linear projections + gates
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])        # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])      # [B,N,N,hidden]

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # optional mask
    if not nomask:
        mask = mask_tensor.to(dtype).unsqueeze(-1)       # [B,N,N,1]
        left = left * mask
        right = right * mask

    # apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 3. N³ reduction via batched matmul (left @ rightᵀ) for each hidden
    # -----------------------------------------------------------------
    # Rearrange to [B, hidden, N, N] and flatten batch & hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)            # [B*H, N, N]
    right_flat = right_perm.view(B * H, N, N)          # [B*H, N, N]
    right_T = right_flat.transpose(1, 2).contiguous() # [B*H, N, N] (K,N) view

    # Triton kernel computes C = left_flat @ right_T   (B*H, N, N)
    out_flat = batched_matmul(left_flat, right_T,
                              BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)

    # Restore original layout
    out_hidden = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 4. Output layer‑norm, gating, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```
