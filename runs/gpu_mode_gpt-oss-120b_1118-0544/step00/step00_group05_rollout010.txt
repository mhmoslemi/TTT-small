"""
TriMul “outgoing” implementation.

The kernel performs the heavy contraction:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where left/right are the projected and gated pairwise tensors.
The contraction is expressed as a batched GEMM with the hidden dimension
`h` fused into the third grid dimension (batch * hidden).  Each program
computes a TILE_M × TILE_N tile of the output using a K‑loop and tl.dot.
All other operations (LayerNorm, Linear, gating, final Linear) are done
with PyTorch for simplicity; the kernel provides the dominant O(N³) work
in float32 and returns a tensor ready for the remaining PyTorch post‑processing.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_out_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # sizes
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM for TriMul outward update.

    Computes, for a fixed batch b and hidden h:

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Grid: (ceil(N/BLOCK_M), ceil(N/BLOCK_N), B*H)
    """
    pid_i = tl.program_id(0)   # tile on the i axis
    pid_j = tl.program_id(1)   # tile on the j axis
    pid_bh = tl.program_id(2)  # combined batch * hidden index

    # decode batch and hidden from combined index
    b = pid_bh // H
    h = pid_bh % H

    # base pointers for this (b, h)
    base_l = left_ptr + b * stride_l_batch + h * stride_l_h
    base_r = right_ptr + b * stride_r_batch + h * stride_r_h
    base_o = out_ptr + b * stride_o_batch + h * stride_o_h

    # tile ranges
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # out‑of‑bounds masks for i/j
    mask_i = i < N
    mask_j = j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        ks = k_start + tl.arange(0, BLOCK_K)
        mask_k = ks < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = base_l + i[:, None] * stride_l_i + ks[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile transposed to (BLOCK_K, BLOCK_N)
        right_ptrs = base_r + ks[:, None] * stride_r_k + j[None, :] * stride_r_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # GEMM accumulation
        acc += tl.dot(left, right)

    # write result
    out_ptrs = base_o + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def tri_mul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched contraction left @ rightᵀ over the intermediate dimension.
    left / right: (B, N, N, H) float32
    Returns out of the same shape.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Strides in elements (not bytes)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    tri_mul_out_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_lb,
        s_li,
        s_lk,
        s_lh,
        s_rb,
        s_ri,
        s_rk,
        s_rh,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Arguments:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be ignored if config["nomask"]=True)
            - weights: dict of torch.Tensors containing all Linear/LayerNorm params
            - config: dict with at least keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Apply optional mask
    # -----------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction – Triton kernel
    # -----------------------------------------------------------------
    out = tri_mul_gemm(left, right)   # shape (B,N,N,hidden_dim)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
