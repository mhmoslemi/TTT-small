def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    b = pid_bh // H
    h = pid_bh % H

    # offsets for first element of block
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    # Create range indices
    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)

    # Mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Compute base pointers for left and right for batch and hidden
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # bounds for k
        mask_k = offs_k < N

        # load left tile [M,K]
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k

        # load right tile [K,N] (right transposed)
        # For right, row = k (since we want [K,N] shape), col = j
        right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # compute dot product
        acc += tl.dot(left, right)

        # increment offsets automatically via loop var K
        
    # store result
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Projections
    left_proj_weight = weights["left_proj.weight"]  # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]

    left = F.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # mask
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask
        right = right * mask

    # gating
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Compute TriMul einsum via Triton
    B, N, _, _ = left.shape  # shape (B,N,N,hidden_dim)
    # ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    # output buffer
    out = torch.empty_like(left)

    # define kernel
    @triton.jit
    def tri_mul_gemm_kernel(
        left_ptr, right_ptr, out_ptr,
        N, B, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        pid_i = tl.program_id(0)
        pid_j = tl.program_id(1)
        pid_bh = tl.program_id(2)

        b = pid_bh // H
        h = pid_bh % H

        i_start = pid_i * BLOCK_M
        j_start = pid_j * BLOCK_N

        offs_i = i_start + tl.arange(0, BLOCK_M)
        offs_j = j_start + tl.arange(0, BLOCK_N)

        mask_i = offs_i < N
        mask_j = offs_j < N

        left_base = left_ptr + b * stride_left_b + h * stride_left_h
        right_base = right_ptr + b * stride_right_b + h * stride_right_h
        out_base = out_ptr + b * stride_out_b + h * stride_out_h

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # iterate over K dimension
        for k in range(0, tl.cdiv(N, BLOCK_K)):
            k_start = k * BLOCK_K
            offs_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # load left tile [M,K]
            left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
            left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

            # load right tile [K,N] (right transposed)
            right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
            right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

            # accumulate dot product
            acc += tl.dot(left, right)

        out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
        tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    # compute strides
    left_stride_b, left_stride_i, left_stride_k, left_stride_h = left.stride()
    right_stride_b, right_stride_j, right_stride_k, right_stride_h = right.stride()
    out_stride_b, out_stride_i, out_stride_j, out_stride_h = out.stride()

    # block sizes (tunable)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N), B * hidden_dim)
    tri_mul_gemm_kernel[grid](
        left, right, out,
        N, B, hidden_dim,
        left_stride_b, left_stride_i, left_stride_k, left_stride_h,
        right_stride_b, right_stride_j, right_stride_k, right_stride_h,
        out_stride_b, out_stride_i, out_stride_j, out_stride_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # to_out_norm and gating
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    out = out_norm * out_gate

    to_out_weight = weights["to_out.weight"]
    output = F.linear(out, to_out_weight)

    return output

"""
TriMul (outgoing) forward implementation.

The operation consists of:
1. LayerNorm on the input pair representation.
2. Linear projections (left/right) and three gating linear layers.
3. Masking (if provided) and element‑wise gating.
4. A batched matrix multiplication implemented in Triton:
       out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
   This is equivalent to a GEMM per hidden channel (c) with the
   transposed right matrix. The kernel tiles the i/j dimensions
   (BLOCK_M × BLOCK_N) and reduces over k (BLOCK_K) while supporting
   an arbitrary batch size B and hidden dimension H.
5. LayerNorm on the hidden result, out‑gate gating, and a final
   linear projection back to the original channel dimension.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    N, B, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: out = left @ right.T  (per hidden channel)."""
    pid_i = tl.program_id(0)          # block row (i)
    pid_j = tl.program_id(1)          # block col (j)
    pid_bh = tl.program_id(2)         # combined batch & hidden index

    b = pid_bh // H                   # batch index
    h = pid_bh % H                    # hidden (channel) index

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over the shared sequence dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile (transposed): shape (BLOCK_K, BLOCK_N)
        # right[j, k] -> rows = k, cols = j
        right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # accumulate dot product over k
        acc += tl.dot(left, right)

    # write the result
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[:, None])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm on the input
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # (B, N, N, dim)

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3️⃣ Optional masking
    # -----------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4️⃣ Gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Triton kernel: out = Σ_k left[...,k,:] * right[...,k,:]
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # (B,N,N,hidden_dim)

    # strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tile sizes – chosen for good occupancy on an A100/B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    _tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        N,
        B,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Final LayerNorm, out‑gate, and linear projection
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
