import torch, triton, triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, N, hidden_dim,
    stride_b, stride_i, stride_j, stride_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)   # row block i
    pid_j = tl.program_id(1)   # col block j
    pid_bd = tl.program_id(2)  # combined batch*hidden index

    # Determine batch and hidden dim indices
    batch_idx = pid_bd // hidden_dim
    d = pid_bd % hidden_dim

    # Compute base pointer for this batch and hidden dim
    left_base = left_ptr + batch_idx * stride_b + d * stride_d
    right_base = right_ptr + batch_idx * stride_b + d * stride_d
    out_base = out_ptr + batch_idx * stride_b + d * stride_d

    # Coordinates of the block
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_iter * BLOCK_K

        # Offsets
        i_range = i_start + tl.arange(0, BLOCK_M)
        j_range = j_start + tl.arange(0, BLOCK_N)
        k_range = k_start + tl.arange(0, BLOCK_K)

        # Compute pointers
        # left[i,k]
        left_ptrs = left_base + i_range[:, None] * stride_i + k_range[None, :] * stride_j
        # right_T[k,j]  (right transposed)
        right_ptrs = right_base + k_range[:, None] * stride_j + j_range[None, :] * stride_i

        # Load with boundary mask
        mask_i = i_range[:, None] < N
        mask_j = j_range[None, :] < N
        mask_k = k_range[None, :] < N

        left = tl.load(left_ptrs, mask=mask_i & mask_k, other=0.0).to(tl.float32)
        right = tl.load(right_ptrs, mask=mask_k & mask_j, other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_base + i_range[:, None] * stride_i + j_range[None, :] * stride_j
    mask_store = (i_range[:, None] < N) & (j_range[None, :] < N)
    tl.store(out_ptrs, acc, mask=mask_store)

"""
TriMul (outgoing) forward pass.

The core of the operator is an O(N³) reduction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right are linear projections of the Layer‑Normed input and are
masked/gated element‑wise.  The reduction is implemented as a tiled
matrix‑multiply in Triton (L × Rᵀ) for every (batch, hidden) pair.
All other operations (LayerNorm, linear layers, sigmoid gates) are
performed with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,        # pointers
    batch, N, hidden_dim,                # sizes
    stride_b, stride_i, stride_j, stride_d,   # strides in elements
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out = left @ rightᵀ for each (batch, hidden) slice.

    Tensor layout (contiguous):
        (b, i, j, d)  with strides (stride_b, stride_i, stride_j, stride_d)
    left  : left[b, i, k, d]
    right : right[b, j, k, d]
    out   : out[b, i, j, d]
    """
    pid_i = tl.program_id(0)          # tile along i (rows)
    pid_j = tl.program_id(1)          # tile along j (cols)
    pid_bd = tl.program_id(2)         # fused batch * hidden dim index

    # De‑compose fused index
    b_idx = pid_bd // hidden_dim
    d_idx = pid_bd % hidden_dim

    # Base pointers for this (b,d) slice
    left_base  = left_ptr  + b_idx * stride_b + d_idx * stride_d
    right_base = right_ptr + b_idx * stride_b + d_idx * stride_d
    out_base   = out_ptr   + b_idx * stride_b + d_idx * stride_d

    # Tile origins
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Accumulator (float32 for numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k in blocks
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_iter * BLOCK_K

        # Index vectors
        i = i0 + tl.arange(0, BLOCK_M)
        j = j0 + tl.arange(0, BLOCK_N)
        k = k0 + tl.arange(0, BLOCK_K)

        # Pointers for left[i,k]  (shape M×K)
        left_ptrs = left_base + i[:, None] * stride_i + k[None, :] * stride_j
        # Pointers for rightᵀ[k,j] (shape K×N) – note transposed layout
        right_ptrs = right_base + k[:, None] * stride_j + j[None, :] * stride_i

        # Load with boundary checks
        mask_i = i[:, None] < N
        mask_j = j[None, :] < N
        mask_k = k[None, :] < N

        left_tile  = tl.load(left_ptrs,
                             mask=mask_i & mask_k,
                             other=0.0).to(tl.float32)
        right_tile = tl.load(right_ptrs,
                             mask=mask_k & mask_j,
                             other=0.0).to(tl.float32)

        # Accumulate dot product over k
        acc += tl.dot(left_tile, right_tile)

    # Store the result tile
    out_ptrs = out_base + i[:, None] * stride_i + j[None, :] * stride_j
    store_mask = (i[:, None] < N) & (j[None, :] < N)
    tl.store(out_ptrs, acc, mask=store_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Mask (if present)
    # -----------------------------------------------------------------
    if not nomask:
        m = mask.to(x.dtype).unsqueeze(-1)          # (B,N,N,1)
        left = left * m
        right = right * m

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid on linear maps of the normalized input)
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"]
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"]
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"]
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core O(N³) reduction via Triton kernel
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=left.device)

    # Strides in element counts (not bytes)
    stride_b, stride_i, stride_j, stride_d = left.stride()

    # Tile sizes – tuned for typical N (256‑1024) on an A200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    _tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        hidden_dim,
        stride_b,
        stride_i,
        stride_j,
        stride_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
