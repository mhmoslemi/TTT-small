def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # Compute offsets
    i_offsets = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offsets = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Determine batch and channel indices
    b = pid_bh // H
    h = pid_bh % H

    # Masks for out-of-bounds
    mask_i = i_offsets < N
    mask_j = j_offsets < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offsets = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < N

        # Load left tile [i,k]
        left_ptrs = left_ptr + b * stride_l_b + h * stride_l_h + i_offsets[:, None] * stride_l_i + k_offsets[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile [k,j]
        right_ptrs = right_ptr + b * stride_r_b + h * stride_r_h + k_offsets[:, None] * stride_r_k + j_offsets[None, :] * stride_r_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Accumulate dot product
        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_ptr + b * stride_o_b + h * stride_o_h + i_offsets[:, None] * stride_o_i + j_offsets[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul forward kernel (outgoing version) for AlphaFold3.

The core operation is:
    out[b, i, j, h] = sum_k left[b, i, k, h] * right[b, j, k, h]

where `left` and `right` are the gated, masked projections of the input.
The kernel implements a blocked matmul for each (batch, hidden) pair using
Triton, handling arbitrary sequence lengths with bounds checks.
All other operations (LayerNorm, gating, final projection) are performed
with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, H,                                     # runtime constants
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # Offsets in i and j dimensions
    i_offsets = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offsets = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Determine batch index and hidden‑channel index from combined axis
    b = pid_bh // H
    h = pid_bh % H

    # Bounds masks
    mask_i = i_offsets < N
    mask_j = j_offsets < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offsets = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < N

        # left[b, i, k, h]  -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + h * stride_l_h
            + i_offsets[:, None] * stride_l_i
            + k_offsets[None, :] * stride_l_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right[b, j, k, h] transposed to (k, j) -> shape (BLOCK_K, BLOCK_N)
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + h * stride_r_h
            + k_offsets[:, None] * stride_r_k
            + j_offsets[None, :] * stride_r_i
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # Accumulate matrix product over k
        acc += tl.dot(left, right)

    # Write the result to out[b, i, j, h]
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + h * stride_o_h
        + i_offsets[:, None] * stride_o_i
        + j_offsets[None, :] * stride_o_j
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def tri_mul_forward(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:

    left/right: [B, N, N, H] (float32)
    Returns out with same shape.

    BLOCK_N = 64
    BLOCK_K = 64

    )

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----- LayerNorm on the input -----
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ----- Linear projections (no bias) -----
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Mask (if provided) -----
    if not config.get("nomask", False):
        # mask shape: [bs, seq_len, seq_len]; broadcast on channel dim
        mask_unsq = mask.to(dtype).unsqueeze(-1)  # [bs, N, N, 1]
        left = left * mask_unsq
        right = right * mask_unsq

    # ----- Gating (sigmoid) -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----- Core triangular multiplicative update (Triton) -----
    out = tri_mul_forward(left, right)  # shape [bs, N, N, hidden_dim]

    # ----- Output LayerNorm -----
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ----- Apply output gate -----
    out = out * out_gate

    # ----- Final linear projection back to original dim -----
    output = F.linear(out, weights["to_out.weight"])  # shape [bs, N, N, dim]

    return output
```
