"""
TriMul (outgoing) implementation with a fused Triton kernel for the
pairwise contraction:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

The kernel treats each (batch, hidden) slice as an independent matrix
multiplication:
    out_{b,d} = left_{b,d} @ right_{b,d}^T,
where left_{b,d} and right_{b,d} are (N×N) matrices.  The kernel
tiles the M, N and K dimensions (i, j, k) to achieve high throughput
while the surrounding PyTorch code handles LayerNorm, linear
projections, gating, and the final output projection.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_lb, stride_lh, stride_li, stride_lk,
    stride_rb, stride_rh, stride_ri, stride_rk,
    stride_ob, stride_oh, stride_oi, stride_ok,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # 3‑D grid: (block_i, block_j, block_batch_head)
    pid_z = tl.program_id(axis=2)      # index over B*H
    pid_m = tl.program_id(axis=0)      # block row (i)
    pid_n = tl.program_id(axis=1)      # block col (j)

    # decode batch and head indices
    batch_id = pid_z // H
    head_id = pid_z % H

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for the selected (batch, head) slice
    left_base  = left_ptr  + batch_id * stride_lb + head_id * stride_lh
    right_base = right_ptr + batch_id * stride_rb + head_id * stride_rh
    out_base   = out_ptr   + batch_id * stride_ob + head_id * stride_oh

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: shape (BLOCK_M, BLOCK_K)  -> rows i, cols k
        a_ptrs = left_base + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # right (transposed for GEMM): shape (BLOCK_K, BLOCK_N)  -> rows k, cols j
        b_ptrs = right_base + offs_k[:, None] * stride_rk + offs_n[None, :] * stride_ri
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (M,K) * (K,N) -> (M,N)

    # write back the tile
    c_ptrs = out_base + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_ok
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32,_cuda)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (left/right) and gating values
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)          # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Mask handling (optional)
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gates
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel: pairwise contraction over k
    #    Shape after permute: [B, hidden, N, N]
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    out_perm = torch.empty_like(left_perm)   # [B, hidden, N, N]

    B, H, N, _ = left_perm.shape

    # strides (in elements)
    stride_lb, stride_lh, stride_li, stride_lk = left_perm.stride()
    stride_rb, stride_rh, stride_ri, stride_rk = right_perm.stride()
    stride_ob, stride_oh, stride_oi, stride_ok = out_perm.stride()

    # Tile sizes (tuned for N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * H )

    tri_mul_kernel[grid](
        left_perm, right_perm, out_perm,
        B, H, N,
        stride_lb, stride_lh, stride_li, stride_lk,
        stride_rb, stride_rh, stride_ri, stride_rk,
        stride_ob, stride_oh, stride_oi, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Back to original layout [B,N,N,hidden]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output layernorm, gating, and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate

    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # final shape [B,N,N,dim]

    return out
