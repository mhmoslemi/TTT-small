def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)  # batch * hidden index
    pid1 = tl.program_id(1)  # i tile
    pid2 = tl.program_id(2)  # j tile

    total_bh = B * H
    bh = pid0
    b = bh // H
    h = bh % H

    # Compute i and j offsets
    i_offset = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for boundary
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left_ptr base: left + b*stride_l_b + h*stride_l_h

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    eps = 1e-5

    # compute layer norm (norm)
    # mean and var across dim axis (last)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    # normative
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    # apply weight and bias
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    x_norm = x_norm * norm_weight + norm_bias

    # linear projections
    left_proj = torch.nn.functional.linear(x_norm, weights['left_proj.weight'])
    right_proj = torch.nn.functional.linear(x_norm, weights['right_proj.weight'])
    # gating projections
    left_gate = torch.nn.functional.linear(x_norm, weights['left_gate.weight']).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights['right_gate.weight']).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights['out_gate.weight']).sigmoid()   # shape B,N,N,H

    # apply mask if present (mask may be None or not; but in problem always given)
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(input_tensor.dtype)
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Launch kernel to compute out = einsum over k
    B, N, _, _ = input_tensor.shape  # after left shape B,N,N,hidden_dim

    # allocate output tensor

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)  # batch * hidden index
    pid1 = tl.program_id(1)  # i tile
    pid2 = tl.program_id(2)  # j tile

    # decode batch and hidden
    bh = pid0
    b = bh // H
    h = bh % H

    # compute tile offsets
    i_offset = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for i and j dims
    mask_i = i_offset < N
    mask_j = j_offset < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # offset for batch and hidden
    base_l = b * stride_l_b + h * stride_l_h
    base_r = b * stride_r_b + h * stride_r_h
    base_o = b * stride_o_b + h * stride_o_h

    # loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # compute pointer addresses for left tile (i, k)
        left_ptrs = left_ptr + base_l + i_offset[:, None] * stride_l_i + k_offset[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right tile (k, j) where we need k rows, j columns
        right_ptrs = right_ptr + base_r + k_offset[:, None] * stride_r_k + j_offset[None, :] * stride_r_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # accumulate

    # write back output
    out_ptrs = out_ptr + base_o + i_offset[:, None] * stride_o_i + j_offset[None, :] * stride_o_j

"""
TriMul "outgoing" forward pass implemented with a custom Triton kernel for the
core N³ reduction (einsum). The kernel computes:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where left/right are the projected and gated tensors.
All other operations (LayerNorm, linear projections, gates) are performed in
PyTorch. The kernel is tiled on the (i, j) dimensions and reduces over k in
blocks for high memory locality.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: batched GEMM over the k dimension for each hidden channel.
# -------------------------------------------------------------------------
@triton.jit
def trifold_mul_kernel(
    left_ptr, right_ptr, out_ptr,                     # pointers
    B, N, H,                                          # dimensions
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Program IDs
    pid_bh = tl.program_id(0)   # combined batch * hidden index
    pid_i  = tl.program_id(1)   # i‑tile
    pid_j  = tl.program_id(2)   # j‑tile

    # Decode batch and hidden (channel) index
    bh = pid_bh
    b  = bh // H
    h  = bh % H

    # Offsets within the tile
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary masks
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Base offsets for this (b,h) pair
    base_l = b * stride_l_b + h * stride_l_h
    base_r = b * stride_r_b + h * stride_r_h
    base_o = b * stride_o_b + h * stride_o_h

    # Accumulator for the tile (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over k in blocks
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # ----- load left[i,k] ------------------------------------------------
        # left shape: (B, N, N, H) -> left[b, i, k, h]
        left_ptrs = left_ptr + base_l \
                    + i_offset[:, None] * stride_l_i \
                    + k_offset[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16).to(tl.float32)   # cast to fp32

        # ----- load right[k,j] (transposed view) ---------------------------
        # right shape: (B, N, N, H) -> right[b, j, k, h]  -> we load as (k,j)
        right_ptrs = right_ptr + base_r \
                     + k_offset[:, None] * stride_r_k \
                     + j_offset[None, :] * stride_r_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0,
                        dtype=tl.float16).to(tl.float32)  # cast to fp32

        # Accumulate dot product over k
        acc += tl.dot(left, right)   # (M,K)·(K,N) -> (M,N)

    # ----- write the tile to output ---------------------------------------
    out_ptrs = out_ptr + base_o \
               + i_offset[:, None] * stride_o_i \
               + j_offset[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model weight tensors
            - config: dict containing "dim" and "hidden_dim" and optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # -----------------------------------------------------------------
    # LayerNorm on the last dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj  = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # Gates (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # Apply mask if present (mask is broadcast over hidden dimension)
    # -----------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # -----------------------------------------------------------------
    # Apply gates
    # -----------------------------------------------------------------
    left  = left_proj * left_gate
    right = right_proj * right_gate

    # -----------------------------------------------------------------
    # Cast to fp16 for the heavy N³ reduction
    # -----------------------------------------------------------------
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Output buffer (float32)
    out_fp32 = torch.empty((batch_size, seq_len, seq_len, hidden_dim),
                           dtype=torch.float32,
                           device=device)

    # -----------------------------------------------------------------
    # Prepare strides for the Triton kernel
    # -----------------------------------------------------------------
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_fp32.stride()

    # -----------------------------------------------------------------
    # Kernel launch parameters
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_size * hidden_dim,                         # pid0 : batch * hidden
        (seq_len + BLOCK_M - 1) // BLOCK_M,              # pid1 : i‑tiles
        (seq_len + BLOCK_N - 1) // BLOCK_N,              # pid2 : j‑tiles
    )

    # Launch the kernel
    trifold_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        batch_size,
        seq_len,
        hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # -----------------------------------------------------------------
    # Output LayerNorm (over hidden dimension)
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # Apply out_gate and final linear projection back to `dim`
    # -----------------------------------------------------------------
    out_norm = out_norm * out_gate
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
