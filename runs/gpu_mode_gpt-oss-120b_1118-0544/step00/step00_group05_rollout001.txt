"""
TriMul (outgoing) forward pass implemented with Triton.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B, N, N, C]).
2. Linear projections to a hidden dimension H:
   left = x @ W_left    , right = x @ W_right
3. Optional binary mask broadcast over H.
4. Compute sigmoid gates (left_gate, right_gate, out_gate) and apply them:
   left  *= sigmoid(x @ W_left_gate)
   right *= sigmoid(x @ W_right_gate)
5. Perform the core O(N³) contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batch of H·B independent matrix multiplies
   (N×N) @ (N×N)ᵀ.  Implemented as a fused Triton GEMM kernel.
6. Layer‑norm over the hidden dimension, multiply with out_gate and a final
   linear projection back to C.

Only the contraction (step 5) is executed on the GPU with Triton; all other
operations use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM kernel (A @ Bᵀ) for the TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                     # matrix dimensions (M rows, N cols, K inner)
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # Number of blocks in each dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # Decompose pid into (batch, block_m, block_n)
    batch = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for ko in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = ko * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers with batch offset
        a_ptrs = a_ptr + batch * stride_a_batch \
                 + offs_m[:, None] * stride_a_row \
                 + offs_k[None, :] * stride_a_col
        b_ptrs = b_ptr + batch * stride_b_batch \
                 + offs_k[:, None] * stride_b_row \
                 + offs_n[None, :] * stride_b_col

        # Load tiles (masked)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Write the result tile
    c_ptrs = c_ptr + batch * stride_c_batch \
             + offs_m[:, None] * stride_c_row \
             + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask_tensor:  torch.Tensor of shape [B, N, N] (may be None)
            - weights_dict: mapping of weight names to torch.Tensor
            - config_dict: must contain "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32, on the same device)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Apply mask if provided (mask broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton (batched GEMM)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Bring hidden dim to second axis for convenient tiling
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, K]
    # Transpose the last two axes of right before permuting
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, K, N]

    B, H, N, K = left_perm.shape  # here K == N
    batch_hidden = B * H

    # Flatten batch*hidden into a single dimension
    A = left_perm.view(batch_hidden, N, K)          # (M, N, K)
    B_mat = right_perm.view(batch_hidden, K, N)     # (M, K, N)

    # Prepare output buffer
    out_flat = torch.empty((batch_hidden, N, N), dtype=torch.float32, device=device)

    # Strides in element units (not bytes)
    stride_a_batch = A.stride(0)
    stride_a_row = A.stride(1)
    stride_a_col = A.stride(2)

    stride_b_batch = B_mat.stride(0)
    stride_b_row = B_mat.stride(1)   # K‑dimension stride
    stride_b_col = B_mat.stride(2)   # N‑dimension stride

    stride_c_batch = out_flat.stride(0)
    stride_c_row = out_flat.stride(1)
    stride_c_col = out_flat.stride(2)

    # Tile sizes – chosen to balance register usage & occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid size: one program instance per (batch_hidden, block_m, block_n)
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch_hidden * num_blocks_m * num_blocks_n,)

    # Launch kernel
    _batched_gemm_kernel[grid](
        A, B_mat, out_flat,
        N, N, N,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, to_out_weight)  # [B, N, N, C]

    return output
