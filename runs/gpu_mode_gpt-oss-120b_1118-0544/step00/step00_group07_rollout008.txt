def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_h, N,
    stride_l_batch, stride_l_row, stride_l_col,
    stride_r_batch, stride_r_row, stride_r_col,
    stride_o_batch, stride_o_row, stride_o_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # offsets for batch pointer
    batch_offset_l = pid_batch * stride_l_batch
    batch_offset_r = pid_batch * stride_r_batch
    batch_offset_o = pid_batch * stride_o_batch

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute number of K tiles
    num_tiles_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_tiles_k):
        # compute k offsets
        k_offs = k * BLOCK_K + offs_k
        # mask for this tile
        cur_mask_k = k_offs < N

        # load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + batch_offset_l + (offs_m[:, None] * stride_l_row) + (k_offs[None, :] * stride_l_col)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & cur_mask_k[None, :], other=0.0)

        # load B_T tile: shape (BLOCK_K, BLOCK_N)

    mask_store = mask_m[:, None] & mask_n[None, :]

"""
TriMul (outgoing) implementation for AlphaFold3.

The kernel computes the core contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
for all batches `b` and hidden channels `d`.  The input tensor
[ B, N, N, C ] is first LayerNormed, linearly projected to a hidden
dimension `H`, gated and masked, then the above contraction is performed
using a tiled Triton GEMM (left @ right^T) where the hidden dimension is
folded into the batch dimension (B*H).  The result is LayerNormed,
gated again and projected back to the original dimension `C`.

Only the heavy O(N³·H) contraction is realized in Triton; all other
operations use plain PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM where B = batch * hidden_dim,
#                A shape (B, M, K)  = left  [i,k],
#                B shape (B, K, N)  = right^T [k,j],
#                C shape (B, M, N)  = out   [i,j].
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # *float32
    right_ptr,         # *float32
    out_ptr,           # *float32
    batch_h: tl.int32, # B*H
    N: tl.int32,
    stride_l_batch: tl.int32,
    stride_l_row:   tl.int32,
    stride_l_col:   tl.int32,
    stride_r_batch: tl.int32,
    stride_r_row:   tl.int32,
    stride_r_col:   tl.int32,
    stride_o_batch: tl.int32,
    stride_o_row:   tl.int32,
    stride_o_col:   tl.int32,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)      # index over folded batch*hidden
    pid_m = tl.program_id(1)          # block row (i)
    pid_n = tl.program_id(2)          # block col (j)

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Global bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this batch entry
    batch_off_l = pid_batch * stride_l_batch
    batch_off_r = pid_batch * stride_r_batch
    batch_off_o = pid_batch * stride_o_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of tiles along K dimension
    n_tiles_k = tl.cdiv(N, BLOCK_K)

    for k in range(n_tiles_k):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < N

        # Load a tile of A: [BLOCK_M, BLOCK_K] (left)
        a_ptrs = (
            left_ptr
            + batch_off_l
            + (offs_m[:, None] * stride_l_row)
            + (cur_k[None, :] * stride_l_col)
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load a tile of B: [BLOCK_K, BLOCK_N] (right^T)
        b_ptrs = (
            right_ptr
            + batch_off_r
            + (cur_k[:, None] * stride_r_row)
            + (offs_n[None, :] * stride_r_col)
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        # GEMM on the tile
        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = (
        out_ptr
        + batch_off_o
        + (offs_m[:, None] * stride_o_row)
        + (offs_n[None, :] * stride_o_col)
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Compute the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) or None
            - weights: dict of model weight tensors
            - config: dict with keys "dim" and "hidden_dim" (int)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Extract config and weights
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm weights (for input)
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Linear projection weights (no bias)
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    # Gating weights
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    # Output projection & normalization
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3) Apply mask (if any) before gating
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B,N,N] -> [B,N,N,1] broadcast over hidden_dim
        mask_f = mask.to(dtype).unsqueeze(-1)
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B,N,N,hidden_dim]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction via Triton (batched GEMM)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Fold hidden dimension into the batch dimension for the kernel
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_flat  = left_perm.view(-1, N, N)   # [B*H, N, N]
    right_flat = right_perm.view(-1, N, N)  # [B*H, N, N]

    out_flat = torch.empty_like(left_flat)  # same dtype & device

    # Strides for the flattened tensors (they are contiguous)
    stride_l_batch = left_flat.stride(0)
    stride_l_row   = left_flat.stride(1)
    stride_l_col   = left_flat.stride(2)

    stride_r_batch = right_flat.stride(0)
    stride_r_row   = right_flat.stride(1)
    stride_r_col   = right_flat.stride(2)

    stride_o_batch = out_flat.stride(0)
    stride_o_row   = out_flat.stride(1)
    stride_o_col   = out_flat.stride(2)

    # Tile sizes – tuned for the B200 (64‑bit, ample shared memory)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        left_flat.shape[0],                         # batch*hidden
        (N + BLOCK_M - 1) // BLOCK_M,               # blocks over i
        (N + BLOCK_N - 1) // BLOCK_N,               # blocks over j
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        left_flat.shape[0],
        N,
        stride_l_batch,
        stride_l_row,
        stride_l_col,
        stride_r_batch,
        stride_r_row,
        stride_r_col,
        stride_o_batch,
        stride_o_row,
        stride_o_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate  # apply output gate

    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
```
