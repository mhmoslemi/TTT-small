def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # extract parameters
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # eps for layernorm:
    eps = 1e-5

    # extract weight tensors
    # we need bias for norm and to_out_norm
    norm_weight = weights["norm.weight"]   # shape [dim]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]   # shape [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]   # shape [dim, hidden_dim]

    # LayerNorm on input
    # use torch.nn.functional.layer_norm
    # no epsilon config, but we can use default eps
    # shape of normalized dim is last dim
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)  # shape [bs,N,N,dim]

    # Linear projections (no bias)
    # Torch linear = input @ weight.T + bias
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [bs,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    if not config.get('nomask', True):
        # mask shape [bs,N,N] -> broadcast
        mask_exp = mask.unsqueeze(-1).to(x_norm.dtype)
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # gating
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Optionally cast to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # allocate output
    out = torch.empty_like(left, dtype=torch.float16, device=device)  # shape [bs,N,N,hidden_dim]

    # Compute strides
    left_stride_b, left_stride_i, left_stride_k, left_stride_d = left.stride()
    right_stride_b, right_stride_i, right_stride_k, right_stride_d = right.stride()
    out_stride_b, out_stride_i, out_stride_j, out_stride_d = out.stride()

    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # compute grid sizes
    total_hd = bs * hidden_dim
    grid = (total_hd, (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)

    # launch kernel
    triton_kernel[grid](
        left, right, out,
        bs, N, hidden_dim,
        left_stride_b, left_stride_i, left_stride_k, left_stride_d,
        right_stride_b, right_stride_i, right_stride_k, right_stride_d,
        out_stride_b, out_stride_i, out_stride_j, out_stride_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # cast back to float32 for subsequent layernorm
    out = out.to(torch.float32)

    # to_out_norm: layernorm over hidden_dim (per-element)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # multiply by out_gate (shape [bs,N,N,hidden_dim])
    out = out * out_gate

    # final linear mapping hidden_dim -> dim
    out = torch.nn.functional.linear(out, to_out_weight)  # weight shape [dim, hidden_dim]; no bias.
    # shape [bs,N,N,dim]

    return out

import triton.language as tl

@triton.jit
def triton_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    Compute out[b,i,j,d] = sum_{k=0}^{N-1} left[b,i,k,d] * right[b,j,k,d]
    All inputs are in fp16, accumulator in fp32.
    """
    pid0 = tl.program_id(0)  # combined batch * channel
    pid1 = tl.program_id(1)  # block i
    pid2 = tl.program_id(2)  # block j

    # decode batch and dim
    total_hd = B * D
    b = pid0 // D
    d = pid0 % D

    # offsets
    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N

    # 1D range for dimensions
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_j = offs_j < N

"""
TriMul (outgoing) implementation with Triton.

The core operation computes, for every batch b, hidden channel d and pair (i, j):

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

All surrounding LayerNorms, linear projections and gating are performed with
PyTorch.  The reduction above is realised as a batched matrix multiplication
using a Triton kernel that tiles the i‑ and j‑dimensions (BLOCK_M × BLOCK_N
threads per program) and iterates over k in blocks (BLOCK_K).  The kernel works
in fp16 for bandwidth‑limited loads and accumulates in fp32 for numerical
stability.  The final output is cast back to fp32 for the subsequent
LayerNorm/reprojection steps.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, D,                               # batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    left, right, out are stored as fp16; accumulation is fp32.
    """
    pid0 = tl.program_id(0)   # combined batch * hidden channel
    pid1 = tl.program_id(1)   # block i
    pid2 = tl.program_id(2)   # block j

    # decode batch index and hidden channel index
    total_hd = B * D
    b = pid0 // D
    d = pid0 % D

    # start offsets for this tile
    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N

    # ranges within the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)               # (M,)
    offs_j = j0 + tl.arange(0, BLOCK_N)               # (N,)
    offs_k = tl.arange(0, BLOCK_K)                    # (K,)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k in blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k_start = kb * BLOCK_K
        cur_k = k_start + offs_k                     # (K,)
        mask_k = cur_k < N

        # offsets for left[b, i, k, d]
        left_offsets = (
            b * stride_left_b
            + d * stride_left_d
            + offs_i[:, None] * stride_left_i
            + cur_k[None, :] * stride_left_k
        )
        # offsets for right[b, j, k, d]
        right_offsets = (
            b * stride_right_b
            + d * stride_right_d
            + offs_j[:, None] * stride_right_i
            + cur_k[None, :] * stride_right_k
        )

        # load tiles, mask out‑of‑bounds entries
        left = tl.load(
            left_ptr + left_offsets,
            mask=mask_i[:, None] & mask_k[None, :],
            other=tl.float16(0.0),
        )
        right = tl.load(
            right_ptr + right_offsets,
            mask=mask_j[:, None] & mask_k[None, :],
            other=tl.float16(0.0),
        )
        # compute outer‑product accumulation over the K‑dimension
        for kk in range(0, BLOCK_K):
            l = left[:, kk].to(tl.float32)   # (M,)
            r = right[:, kk].to(tl.float32)  # (N,)
            acc += l[:, None] * r[None, :]   # (M,N)

    # store the result
    out_offsets = (
        b * stride_out_b
        + d * stride_out_d
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(
        out_ptr + out_offsets,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] float32
            - mask: [B, N, N] (bool/int) or None
            - weights: dict of model weight tensors
            - config: dict with keys "dim", "hidden_dim", "nomask", etc.

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    torch.cuda.set_device("cuda")
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32

    # -----------------------------------------------------------------
    # 2. Linear projections (dim -> hidden_dim)
    # -----------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Mask handling (if required)
    # -----------------------------------------------------------------
    if not config.get("nomask", True) and mask is not None:
        mask_exp = mask.to(x_norm.dtype).unsqueeze(-1)            # [B, N, N, 1]
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # -----------------------------------------------------------------
    # 4. Gating signals (sigmoid) and element‑wise gating
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # Cast to fp16 before the heavy reduction
    left = (left_proj * left_gate).to(torch.float16)
    right = (right_proj * right_gate).to(torch.float16)

    # -----------------------------------------------------------------
    # 5. Core reduction using Triton
    # -----------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float16)  # [B, N, N, hidden_dim]

    # Compute strides (in element units)
    s_lb, s_li, s_lk, s_ld = left.stride()
    s_rb, s_ri, s_rk, s_rd = right.stride()
    s_ob, s_oi, s_oj, s_od = out.stride()

    # Tile sizes – product must be <=1024 (max threads per program)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                     # batch * hidden channel
        (N + BLOCK_M - 1) // BLOCK_M,       # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,       # j‑tiles
    )

    triton_trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        s_lb,
        s_li,
        s_lk,
        s_ld,
        s_rb,
        s_ri,
        s_rk,
        s_rd,
        s_ob,
        s_oi,
        s_oj,
        s_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_stages=3,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 6. Post‑processing (LayerNorm, gating, final projection)
    # -----------------------------------------------------------------
    out_fp32 = out.to(torch.float32)  # convert back for LayerNorm

    # to_out_norm over the hidden dimension
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # apply out gate (still fp32)
    out_gated = out_norm * out_gate

    # final linear projection hidden_dim -> dim
    out_final = torch.nn.functional.linear(out_gated, weights["to_out.weight"])

    return out_final
