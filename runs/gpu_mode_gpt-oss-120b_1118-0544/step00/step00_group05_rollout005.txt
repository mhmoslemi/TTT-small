@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_bi, stride_bk, stride_bh,
    stride_rl, stride_rj, stride_rk, stride_rh,
    stride_ol, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bc = tl.program_id(2)

    b = pid_bc // H
    h = pid_bc % H

    # i and j indices
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block: shape (BLOCK_I, BLOCK_K)
        left_ptrs = left_ptr + b * stride_bl + offs_i[:, None] * stride_bi + offs_k[None, :] * stride_bk + h * stride_bh
        left_val = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right transposed block: shape (BLOCK_K, BLOCK_J)
        # right[b, j, k, h] -> we want (k, j)
        right_ptrs = right_ptr + b * stride_rl + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj + h * stride_rh
        right_val = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Accumulate dot product
        acc += tl.dot(left_val, right_val)

    # Store result
    out_ptrs = out_ptr + b * stride_ol + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj + h * stride_oh
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton-accelerated Triangle Multiplicative Update (outgoing) forward pass.
    Performs LayerNorm, linear projections, gating, and the core O(N^3) bilinear
    contraction using a custom Triton kernel that computes
    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h].
    The result is normalized, gated and projected back to the input dimension.
    """

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # epsilon for layer norm
    eps = 1e-5

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Mask handling
    if mask is not None:
        mask_f = mask.to(left.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f
    else:
        # no mask; optionally skip
        pass

    # Gating
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Triton kernel for contraction.
    B, N, _, _ = left.shape
    # Cast to half for faster kernel compute (keep out as float32)
    left_f = left.half()
    right_f = right.half()

    out_h = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # Launch kernel
    launch_trimul_kernel(left_f, right_f, out_h)

    # Post-processing
    # Norm on hidden dim
    out = torch.nn.functional.layer_norm(
        out_h,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
Triangle Multiplicative Update (outgoing) – Triton implementation
----------------------------------------------------------------
This file provides a `custom_kernel` function that implements the forward
pass of the TriMul module used in AlphaFold‑3 and related models.

The computation consists of:
1. LayerNorm on the input tensor.
2. Linear projections (left/right) and three gating projections.
3. Application of a mask (if provided) and element‑wise gates.
4. Core O(N³) contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is mathematically a batched matrix multiplication
   L_h @ R_hᵀ for each hidden channel h.  This is performed by a
   custom Triton kernel that tiles the i, j and k dimensions.
5. LayerNorm on the hidden dimension, gating with `out_gate`,
   and a final linear projection back to the original feature dimension.

Only the contraction (step 4) is executed on Triton; the surrounding
operations use PyTorch for simplicity.  The kernel loads `float16`
values, accumulates in `float32`, and writes the result as `float32`.

The kernel launch grid is (num_i_blocks, num_j_blocks, B*H) where the
third dimension encodes the batch index and hidden‑channel index.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,          # *half   [B,N,N,H] (left projection after gating)
    right_ptr,         # *half   [B,N,N,H] (right projection after gating)
    out_ptr,           # *float  [B,N,N,H] (output of the contraction)
    B, N, H,           # scalar integers
    stride_bl, stride_bi, stride_bk, stride_bh,   # strides for left
    stride_rl, stride_rj, stride_rk, stride_rh,   # strides for right
    stride_ol, stride_oi, stride_oj, stride_oh,   # strides for out
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]"""

    pid_i = tl.program_id(0)                # block index over i
    pid_j = tl.program_id(1)                # block index over j
    pid_bc = tl.program_id(2)               # combined batch & hidden index

    # Decode batch (b) and hidden channel (h) from the 3rd dimension
    b = pid_bc // H
    h = pid_bc % H

    # i and j coordinates for this block
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over k dimension in tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------- Load left block L[i,k] (shape BLOCK_I x BLOCK_K) -------
        left_ptrs = (
            left_ptr
            + b * stride_bl
            + offs_i[:, None] * stride_bi
            + offs_k[None, :] * stride_bk
            + h * stride_bh
        )
        L = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)  # cast half -> float32

        # ------- Load right block transposed Rᵀ[k,j] (shape BLOCK_K x BLOCK_J) -------
        # right layout is (b, j, k, h); we need (k, j) order.
        right_ptrs = (
            right_ptr
            + b * stride_rl
            + offs_k[:, None] * stride_rk
            + offs_j[None, :] * stride_rj
            + h * stride_rh
        )
        R_T = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # ---- Accumulate dot product L @ R_T ----
        acc += tl.dot(L, R_T)   # (BLOCK_I,BLOCK_K) x (BLOCK_K,BLOCK_J) -> (BLOCK_I,BLOCK_J)

    # ---- Store the result ----
    out_ptrs = (
        out_ptr
        + b * stride_ol
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
        + h * stride_oh
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def launch_tri_mul_kernel(left: torch.Tensor,
                          right: torch.Tensor,
                          out: torch.Tensor,
                          BLOCK_I: int = 64,
                          BLOCK_J: int = 64,
                          BLOCK_K: int = 32):
    """
    Launch the Triton kernel that computes the triple‑product contraction.
    left/right are expected to be half‑precision tensors of shape
    [B, N, N, H]; out is a float32 tensor of the same shape.
    """
    B, N, _, H = left.shape

    # Strides (in elements, not bytes)
    stride_bl, stride_bi, stride_bk, stride_bh = left.stride()
    stride_rl, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ol, stride_oi, stride_oj, stride_oh = out.stride()

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,   # i‑blocks
        (N + BLOCK_J - 1) // BLOCK_J,   # j‑blocks
        B * H,                          # combined batch‑hidden dimension
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_bl,
        stride_bi,
        stride_bk,
        stride_bh,
        stride_rl,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ol,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )
    # Kernel is asynchronous; synchronize for correctness in downstream ops
    torch.cuda.synchronize()


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update.
    Arguments
    ----------
    data : tuple(input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, C] (float32)
        mask         : torch.Tensor [B, N, N] (float32/bool) or None
        weights      : dict of model parameters (see implementation)
        config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None:
        # ensure mask is float and has a trailing singleton dimension
        mask_f = mask.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # 4. Gating
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core O(N³) contraction via Triton
    # -----------------------------------------------------------------
    # Cast to half for the kernel; the kernel accumulates in FP32.
    left_half = left.contiguous().half()
    right_half = right.contiguous().half()

    # Allocate output buffer for the hidden dimension (float32)
    B, N, _, _ = left.shape
    out_hidden = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    launch_tri_mul_kernel(left_half, right_half, out_hidden)

    # -----------------------------------------------------------------
    # 6. Post‑processing: normalization, out‑gate, final projection
    # -----------------------------------------------------------------
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_hidden = out_hidden * out_gate
    output = torch.nn.functional.linear(out_hidden, weights["to_out.weight"])

    return output
