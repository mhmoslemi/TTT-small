@triton.jit
def tri_mul_kernel_impl(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    # pid
    pid_i = tl.program_id(0)  # block index over i
    pid_j = tl.program_id(1)  # block index over j
    pid_z = tl.program_id(2)  # combined batch*head

    # Determine batch and head
    b = pid_z // D
    d = pid_z % D

    # Compute offsets
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # Offsets for each tile dimension
    i_offsets = i_start + tl.arange(0, BLOCK_I)
    j_offsets = j_start + tl.arange(0, BLOCK_J)

    # mask for out-of-bounds
    mask_i = i_offsets < N
    mask_j = j_offsets < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over k tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < N
        
        # left tile: left[b,i,k,d]
        left_ptrs = left_ptr + b * stride_left_b + d * stride_left_d \
                    + i_offsets[:, None] * stride_left_i \
                    + k_offsets[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile transposed: we need right[b,j,k,d] but want shape (k,j)
        right_ptrs = right_ptr + b * stride_right_b + d * stride_right_d \
                     + k_offsets[:, None] * stride_right_k \
                     + j_offsets[None, :] * stride_right_j
        # Note: stride_right_k is stride for k dimension, stride_right_j for j dimension.
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Cast to float32 for compute
        left_f = left.to(tl.float32)
        right_f = right.to(tl.float32)

        # Multiply-accumulate via dot (matrix multiplication)
        # tl.dot expects shapes (M,K) and (K,N) returning (M,N)
        acc += tl.dot(left_f, right_f)
    
    # Store result to out
    out_ptrs = out_ptr + b * stride_out_b + d * stride_out_d \
               + i_offsets[:, None] * stride_out_i \
               + j_offsets[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul "outgoing" implementation using Triton.

The heavy part of the TriMul operator is the tensor contraction

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` have shape [B, N, N, D] (D = hidden_dim).
This kernel treats each (batch, head) pair (i.e. a specific `b` and `d`)
as an independent task and computes a 2‑D tile of the output matrix
(i × j) by a blocked matrix‑multiply‑accumulate over the summed axis `k`.
All other operations (LayerNorm, Linear projections, gating, final
projection) are performed with PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def _tri_mul_kernel(
    left_ptr,          # *float32 left  [B,N,N,D]
    right_ptr,         # *float32 right [B,N,N,D]
    out_ptr,           # *float32 out   [B,N,N,D]
    B, N, D,
    # strides for left
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    # strides for right
    stride_r_b, stride_r_j, stride_r_k, stride_r_d,
    # strides for out
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    # compile‑time block sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)                # tile over i
    pid_j = tl.program_id(1)                # tile over j
    pid_z = tl.program_id(2)                # combined batch * head

    # --------------------------------------------------------------
    # decode batch and head (hidden) index from pid_z
    # --------------------------------------------------------------
    b = pid_z // D                           # batch index
    d = pid_z % D                            # hidden dimension index

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    i_offset = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # accumulator for the tile
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over the reduction axis k in blocks
    # --------------------------------------------------------------
    k = 0
    while k < N:
        cur_k = k + tl.arange(0, BLOCK_K)           # absolute k indices
        mask_k = cur_k < N

        # ----- left tile: shape (BLOCK_I, BLOCK_K) -----
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + d * stride_l_d
            + i_offset[:, None] * stride_l_i
            + cur_k[None, :] * stride_l_k
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # ----- right tile (transposed) : shape (BLOCK_K, BLOCK_J) -----
        # right is stored as (b, j, k, d).  We want (k, j) ordering.
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + d * stride_r_d
            + cur_k[:, None] * stride_r_k
            + j_offset[None, :] * stride_r_j
        )
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # ----- matrix multiply‑accumulate -----
        # cast to float32 for precision
        acc += tl.dot(left.to(tl.float32), right.to(tl.float32))

        k += BLOCK_K

    # --------------------------------------------------------------
    # write the computed tile back to global memory
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + d * stride_o_d
        + i_offset[:, None] * stride_o_i
        + j_offset[None, :] * stride_o_j
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_forward(left: torch.Tensor, right: torch.Tensor,
                     BLOCK_I=64, BLOCK_J=64, BLOCK_K=64) -> torch.Tensor:
    """
    Compute the contraction `out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]`
    using the Triton kernel above.
    """
    B, N, _, D = left.shape
    out = torch.empty_like(left)

    # strides (in elements) for each tensor
    s_l_b, s_l_i, s_l_k, s_l_d = left.stride()
    s_r_b, s_r_j, s_r_k, s_r_d = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_d = out.stride()

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,        # i‑tiles
        (N + BLOCK_J - 1) // BLOCK_J,        # j‑tiles
        B * D,                               # one program per (batch, head)
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        D,
        s_l_b,
        s_l_i,
        s_l_k,
        s_l_d,
        s_r_b,
        s_r_j,
        s_r_k,
        s_r_d,
        s_o_b,
        s_o_i,
        s_o_j,
        s_o_d,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # small launch config works well for the sizes we target
    )
    return out


def custom_kernel(data):
    """
    Entry point for the Triton‑accelerated Triangle Multiplicative Update (outgoing).

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim].
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1)             # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton
    # ------------------------------------------------------------------
    out = _tri_mul_forward(left, right, BLOCK_I=64, BLOCK_J=64, BLOCK_K=64)

    # ------------------------------------------------------------------
    # 6. LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7. Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
