def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # shapes
    B,N,N,dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Compute LayerNorm: using torch.nn.functional.layer_norm with weight and bias.
    # Here we have separate layernorm weight/bias for dim.
    # norm = (x - mean)/sqrt(var+eps) * weight + bias. Use epsilon 1e-5.
    eps = 1e-5
    weight = weights["norm.weight"]
    bias = weights["norm.bias"]
    # compute mean,var over last dim (dim)
    # x_norm = (x - mean) / sqrt(var+eps) * weight + bias
    x = input_tensor
    # flatten batch and seq dims for offset
    x_flat = x.view(-1, dim)
    mean = x_flat.mean(-1, keepdim=True)
    var = x_flat.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean.view(B,N,N,1)) / torch.sqrt(var.view(B,N,N,1) + eps)
    x_norm = x_norm * weight + bias

    # Linear projections:
    # left_proj: shape (dim, hidden_dim) weight shape [hidden_dim, dim]?? In nn.Linear weight shape (out_features, in_features).
    # In provided code left_proj = nn.Linear(dim, hidden_dim, bias=False). So weight shape [hidden_dim, dim].
    # Similarly for right_proj, left_gate, right_gate, out_gate.
    # We'll use torch.nn.functional.linear.
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Compute projections
    # left = torch.nn.functional.linear(x_norm, left_proj_weight) # shape B,N,N,hidden_dim
    # Use torch.einsum: but functional linear works on last dim.
    left = torch.nn.functional.linear(x_norm, left_proj_weight) # no bias
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask
    if mask is not None and mask.numel() > 0: # if mask provided
        mask_ = mask.unsqueeze(-1)  # B,N,N,1
        left = left * mask_
        right = right * mask_

    # Gates: linear + sigmoid
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute core: out = left @ right^T per channel
    # We'll call triton kernel with left/right as float16 maybe.
    # Use float16 for speed but we need float32 compute. We'll cast to float16 for memory and back to float32? input could be float32 we can cast to float16.
    # Ensure left and right contiguous.

    left_cont = left.contiguous()
    right_cont = right.contiguous()

    # output tensor allocate
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left_cont.device)

    # Determine block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Launch kernel using grid dimensions: (B*hidden_dim, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    total_bc = B * hidden_dim
    grid = (total_bc, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    # Compute strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left_cont.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_d = right_cont.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # Since our kernel expects (B,N,N,D) layout as described

        left_cont, right_cont, out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_d,
        stride_r_b, stride_r_i, stride_r_k, stride_r_d,
        stride_o_b, stride_o_i, stride_o_j, stride_o_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # After kernel, apply to_out_norm: LayerNorm over hidden_dim.
    # to_out_norm weight/bias length hidden_dim.
    # Compute mean/var over hidden_dim.

    # Apply out_gate and final linear
    out = out_norm * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)  # no bias

    return out

"""
TriMul "outgoing" forward pass implemented with Triton.

The heavy part of TriMul is the tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where left/right have shape (B, N, N, H) (H = hidden_dim).  For every
channel d this is exactly a matrix multiplication:
    out_d = left_d @ right_d.T
We implement this contraction with a Triton kernel that tiles the i‑ and
j‑axes (BLOCK_M × BLOCK_N) and performs a K‑reduction in blocks
(BLOCK_K).  The kernel runs on a combined batch‑channel dimension
(b * H) so it works for any batch size (1‑2) and hidden dimension
(≤384).  All other operations (LayerNorm, Linear projections, gates,
and final LayerNorm + Linear) are performed with PyTorch for simplicity.

The kernel computes in float32 while loading/storing in float16 to
reduce memory traffic.  The final output is float32 as required.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the core O(N^3) contraction
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    left/right: [B, N, N, H]   (layout: b, i, k, h)
    out       : [B, N, N, H]   (layout: b, i, j, h)
    """
    # ------------ program IDs ------------
    pid_bc = tl.program_id(0)   # combined batch*channel dimension
    pid_i  = tl.program_id(1)   # i‑tile
    pid_j  = tl.program_id(2)   # j‑tile

    total_bc = B * H
    b = pid_bc // H                     # batch index
    h = pid_bc %  H                     # hidden‑dim index

    # ------------ tile offsets ------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------ K‑reduction loop ------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left[b, i, k, h]  -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            b * stride_l_b + h * stride_l_h + \
            offs_i[:, None] * stride_l_i + \
            offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # Load right[b, j, k, h] transposed as (BLOCK_K, BLOCK_N)
        # (we need right_T[k, j] so that dot(left, right_T) sums over k)
        right_ptrs = right_ptr + \
            b * stride_r_b + h * stride_r_h + \
            offs_k[:, None] * stride_r_k + \
            offs_j[None, :] * stride_r_i
        right_T = tl.load(right_ptrs,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0).to(tl.float32)

        # Matrix multiply (M×K)·(K×N) -> (M×N)
        acc += tl.dot(left, right_T)

    # ------------ write back ------------
    out_ptrs = out_ptr + \
        b * stride_o_b + h * stride_o_h + \
        offs_i[:, None] * stride_o_i + \
        offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward (outgoing) implementation.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float, may be all‑ones)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim" (int)

    Returns:
        torch.Tensor: output [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Extract config
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # compute mean/var over last dim
    x = input_tensor
    # flatten batch*seq*seq for stable mean/var
    x_flat = x.view(-1, dim)
    mean = x_flat.mean(dim=1, keepdim=True)
    var = x_flat.var(dim=1, unbiased=False, keepdim=True)
    mean = mean.view(x.shape[:-1] + (1,))
    var = var.view(x.shape[:-1] + (1,))
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    # Project to hidden dimension (float16 for memory bandwidth)
    left = torch.nn.functional.linear(x_norm, left_proj_w).to(torch.float16)
    right = torch.nn.functional.linear(x_norm, right_proj_w).to(torch.float16)

    # ------------------------------------------------------------------
    # Apply mask if provided (mask shape [B,N,N])
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=torch.float32).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid().to(torch.float16)
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid().to(torch.float16)
    out_gate = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()          # keep fp32 for later

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core contraction via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape: [B, N, N, hidden_dim] (half)

    # Allocate output in fp32 (the kernel writes fp32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Block sizes (tuned for typical H100, can be adjusted)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Compute strides for the three tensors
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Grid: (B*H, ceil(N/B_M), ceil(N/B_N))
    total_bc = B * hidden_dim
    grid = (
        total_bc,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # to_out_norm (LayerNorm on hidden dimension)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps
    )

    # ------------------------------------------------------------------
    # Apply out_gate and final projection to original dim
    # ------------------------------------------------------------------
    out = out * out_gate  # out_gate is fp32, broadcasting over hidden dim
    to_out_w = weights["to_out.weight"]      # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)  # shape [B,N,N,dim]

    return out
