```
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # Get program IDs
    pid_m = tl.program_id(0)  # row block
    pid_n = tl.program_id(1)  # col block
    pid_b = tl.program_id(2)  # batch hidden index

    # Compute offsets
    # Compute tile start
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # K loop offsets range
    # Load a block of A (M x K) and B (K x N) for this batch index
    a_ptrs = a_ptr + pid_b * stride_batch + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_batch + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load blocks
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K) < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        # Compute dot product
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)
        # Increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store to C
    c_ptrs = c_ptr + pid_b * stride_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float32), mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])
```

"""
TriMul (outgoing) forward pass implemented with Triton for the heavy
pairwise multiply‑sum. The algorithm follows the reference PyTorch code:

1. LayerNorm over the channel dimension (dim).
2. Linear projections (left/right) and gating projections (left_gate,
   right_gate, out_gate) without bias.
3. Optional mask applied before gating.
4. Gate the left/right tensors.
5. Compute out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d] using a batched
   matrix multiplication implemented in Triton (left @ rightᵀ for each
   hidden channel).
6. LayerNorm on the hidden dimension, multiply by out_gate, and a final
   linear projection back to `dim`.

Only the N³‑like contraction (step 5) is executed in Triton; all other
operations use PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for left @ right.T over many hidden channels
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix sizes (M=N=seq_len, K=seq_len)
    stride_am, stride_ak,                    # strides for A (M dim, K dim)
    stride_bk, stride_bn,                    # strides for B (K dim, N dim)
    stride_cm, stride_cn,                    # strides for C (M dim, N dim)
    stride_batch,                            # stride to next batch*hidden entry
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """ Compute C = A @ Bᵀ for a batch of matrices.
        A, B, C have shape (batch_h, M, K), (batch_h, K, N), (batch_h, M, N)
        where batch_h = batch_size * hidden_dim.
    """
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch_hidden index

    # Offsets within the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in chunks of BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers to the current tiles of A and B
        a_ptrs = a_ptr + pid_b * stride_batch \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_b * stride_batch \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # Load tiles with out‑of‑bounds masking
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        # Accumulate the dot product
        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = c_ptr + pid_b * stride_batch \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor[B, N, N, dim]
        - mask: torch.Tensor[B, N, N] (may be None when config["nomask"] is True)
        - weights: dict with all linear and layer‑norm parameters
        - config: dict containing at least "dim", "hidden_dim", optionally "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm over channel dim
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left_proj = F.linear(x, left_proj_w)          # [B,N,N,hidden_dim]
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)    # [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4. Gating values (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply gates to the projected tensors
    left = left_proj * left_gate   # [B,N,N,hidden_dim]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Batched GEMM (left @ rightᵀ) via Triton
    # ------------------------------------------------------------------
    # Rearrange to bring hidden_dim into the batch dimension
    # left:  [B, N, N, H] -> [B, H, N, N]
    left_t = left.permute(0, 3, 1, 2).contiguous()
    # rightᵀ: [B, N, N, H] -> permute(0,3,2,1) -> [B, H, N, N]
    right_t = right.permute(0, 3, 2, 1).contiguous()

    # Flatten batch*hidden so each GEMM works on a (N,N) matrix pair
    left_flat = left_t.view(-1, N, N)    # shape (B*H, N, N)
    right_flat = right_t.view(-1, N, N)  # shape (B*H, N, N)
    out_flat = torch.empty_like(left_flat)   # (B*H, N, N)

    # Strides for Triton (all tensors are contiguous in row‑major order)
    stride_am = left_flat.stride(1)      # stride for M dimension
    stride_ak = left_flat.stride(2)      # stride for K dimension
    stride_bk = right_flat.stride(1)     # stride for K dimension in B
    stride_bn = right_flat.stride(2)     # stride for N dimension in B
    stride_cm = out_flat.stride(1)       # stride for M in C
    stride_cn = out_flat.stride(2)       # stride for N in C
    stride_batch = left_flat.stride(0)   # distance between successive batch‑hidden entries

    # Block sizes – chosen to give good occupancy for N up to 1024
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid: (rows, cols, batch_hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        left_flat.shape[0],
    )

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                     # M, N, K (all equal to seq_len)
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # apply out_gate (shape matches)

    # Final projection back to original dim
    to_out_w = weights["to_out.weight"]     # (dim, hidden_dim)
    out = F.linear(out, to_out_w)          # [B, N, N, dim]

    return out
