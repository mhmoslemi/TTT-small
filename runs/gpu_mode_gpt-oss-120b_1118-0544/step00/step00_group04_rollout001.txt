def layernorm_kernel(x_ptr, out_ptr,
                     weight_ptr, bias_ptr,
                     B, N, D,
                     stride_xb, stride_xi, stride_xj, stride_xd,
                     stride_ob, stride_oi, stride_oj, stride_od,
                     EPS: tl.constexpr):
    pid = tl.program_id(0)
    total = B*N*N
    b = pid // (N*N)
    idx = pid % (N*N)
    i = idx // N
    j = idx % N
    # compute base address for each row
    x_ptrs = x_ptr + b*stride_xb + i*stride_xi + j*stride_xj
    out_ptrs = out_ptr + b*stride_ob + i*stride_oi + j*stride_oj
    # compute sum and sum_sq
    sum_val = tl.zeros([BLOCK_SIZE], dtype=tl.float32) # not needed; single sum across dims.
    # Actually need scalar sum; so use scalar accumulator.
    sum_scalar = tl.float32(0.0)
    sum_sq_scalar = tl.float32(0.0)
    for d_offset in range(0, tl.cdiv(D, BLOCK_SIZE)):
        offs = d_offset * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(x_ptrs + offs * stride_xd, mask=mask, other=0.0)
        sum_scalar += tl.sum(x, mask=mask)
        sum_sq_scalar += tl.sum(x * x, mask=mask)
    mean = sum_scalar / D
    var = sum_sq_scalar / D - mean * mean
    rstd = tl.rsqrt(var + EPS)
    # second pass: write normalized values
    for d_offset in range(0, tl.cdiv(D, BLOCK_SIZE)):
        offs = d_offset * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
        mask = offs < D
        x = tl.load(x_ptrs + offs * stride_xd, mask=mask, other=0.0)
        weight = tl.load(weight_ptr + offs, mask=mask, other=0.0)
        bias = tl.load(bias_ptr + offs, mask=mask, other=0.0)
        x_norm = (x - mean) * rstd
        y = x_norm * weight + bias
        tl.store(out_ptrs + offs * stride_od, y, mask=mask)

def layernorm_kernel(
    x_ptr, out_ptr,
    weight_ptr, bias_ptr,
    B, N, D,
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_D: tl.constexpr,
    EPS: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    x_ptr_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    out_ptr_base = out_ptr + b * stride_ob + i * stride_oi + j * stride_oj

    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)
    num_iter = tl.cdiv(D, BLOCK_D)

    for it in range(num_iter):
        offs = it * BLOCK_D + tl.arange(0, BLOCK_D)
        mask = offs < D
        x = tl.load(x_ptr_base + offs * stride_xd, mask=mask, other=0.0)
        sum_val += tl.sum(x, mask=mask)
        sum_sq += tl.sum(x * x, mask=mask)

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + EPS)

    # second pass: write normalized values
    for it in range(num_iter):
        offs = it * BLOCK_D + tl.arange(0, BLOCK_D)
        mask = offs < D
        x = tl.load(x_ptr_base + offs * stride_xd, mask=mask, other=0.0)
        w = tl.load(weight_ptr + offs, mask=mask, other=0.0)
        b_ = tl.load(bias_ptr + offs, mask=mask, other=0.0)
        x_norm = (x - mean) * rstd
        y = x_norm * w + b_
        tl.store(out_ptr_base + offs * stride_od, y, mask=mask)

"""
TriMul (outgoing) kernel implementation.

The computation follows the PyTorch reference:

1. LayerNorm over the last dimension (dim) of the input tensor `x`.
2. Linear projections `left` and `right` (dim → hidden_dim) and gating
   linear layers (`left_gate`, `right_gate`, `out_gate`) followed by sigmoid.
3. Optional mask multiplication (broadcast over hidden_dim).
4. Element‑wise gating of `left` and `right`.
5. Core O(N³) reduction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is realized as a batched matrix multiplication per hidden channel:
       out = (left_perm @ right_permᵀ) where each channel is a separate batch.
6. LayerNorm over the hidden dimension, gating with `out_gate`,
   and final linear projection back to `dim`.

Two Triton kernels implement the LayerNorm steps (input‑norm and output‑norm).
All other operations are performed with PyTorch (highly‑optimized cuBLAS
for the reduction). The final output is `float32`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: LayerNorm over the last dimension.
# -------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # *const float* input
    out_ptr,             # *float* output
    weight_ptr,          # *const float* gamma (shape D)
    bias_ptr,            # *const float* beta  (shape D)
    B, N, D,             # int: batch, seq_len, hidden size
    stride_xb, stride_xi, stride_xj, stride_xd,  # strides for input
    stride_ob, stride_oi, stride_oj, stride_od,  # strides for output
    BLOCK_D: tl.constexpr,        # block size along hidden dimension
    EPS: tl.constexpr = 1e-5,    # epsilon for numerical stability
):
    pid = tl.program_id(0)
    total = B * N * N
    # Decode pid -> (b,i,j)
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # Base pointers for the row (b,i,j,:)
    x_ptr_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    out_ptr_base = out_ptr + b * stride_ob + i * stride_oi + j * stride_oj

    # ---------- First pass: compute mean and variance ----------
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)
    num_iter = tl.cdiv(D, BLOCK_D)

    for it in range(num_iter):
        offs = it * BLOCK_D + tl.arange(0, BLOCK_D)
        mask = offs < D
        x = tl.load(x_ptr_base + offs * stride_xd, mask=mask, other=0.0)
        sum_val += tl.sum(x, mask=mask)
        sum_sq += tl.sum(x * x, mask=mask)

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + EPS)

    # ---------- Second pass: write normalized values ----------
    for it in range(num_iter):
        offs = it * BLOCK_D + tl.arange(0, BLOCK_D)
        mask = offs < D
        x = tl.load(x_ptr_base + offs * stride_xd, mask=mask, other=0.0)

        w = tl.load(weight_ptr + offs, mask=mask, other=0.0)
        b_val = tl.load(bias_ptr + offs, mask=mask, other=0.0)

        x_norm = (x - mean) * rstd
        y = x_norm * w + b_val
        tl.store(out_ptr_base + offs * stride_od, y, mask=mask)


def layernorm_triton(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Apply LayerNorm over the last dimension of `x` using a Triton kernel.
    """
    assert x.is_contiguous()
    B, N, N2, D = x.shape
    assert N == N2, "Input must be square in the two middle dimensions."
    out = torch.empty_like(x)

    # Strides (in elements, not bytes)
    stride_xb, stride_xi, stride_xj, stride_xd = x.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    total = B * N * N
    grid = (total,)

    # Choose a block size that divides the hidden dimension well.
    BLOCK_D = 64 if D >= 64 else 32

    layernorm_kernel[grid](
        x,
        out,
        weight,
        bias,
        B, N, D,
        stride_xb, stride_xi, stride_xj, stride_xd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_D=BLOCK_D,
        EPS=eps,
    )
    return out


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Compute the outgoing Triangle Multiplicative Update (TriMul) without gradients.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              - input_tensor: torch.Tensor of shape [B, N, N, dim]
              - mask_tensor:  torch.Tensor of shape [B, N, N] or None
              - weights_dict: mapping of weight names to torch.Tensors
              - config_dict: {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Fetch weights (ensure they are on the right device and dtype)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=dtype)
    norm_bias = weights["norm.bias"].to(device=device, dtype=dtype)

    left_proj_weight = weights["left_proj.weight"].to(device=device, dtype=dtype)
    right_proj_weight = weights["right_proj.weight"].to(device=device, dtype=dtype)

    left_gate_weight = weights["left_gate.weight"].to(device=device, dtype=dtype)
    right_gate_weight = weights["right_gate.weight"].to(device=device, dtype=dtype)
    out_gate_weight = weights["out_gate.weight"].to(device=device, dtype=dtype)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device=device, dtype=dtype)

    to_out_weight = weights["to_out.weight"].to(device=device, dtype=dtype)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x_norm = layernorm_triton(input_tensor, norm_weight, norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2) Projections and Gates (all linear layers without bias)
    # ------------------------------------------------------------------
    # Linear projections (dim -> hidden_dim)
    left = F.linear(x_norm, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_weight)

    # Optional mask handling
    if mask is not None:
        # mask shape [B,N,N] -> broadcast to hidden_dim
        mask_exp = mask.to(dtype=dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Gating values (sigmoid)
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # Apply gates to projected tensors
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Core O(N^3) reduction: batched matmul per hidden channel
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Rearrange to (B*hidden_dim, N, N) to reuse cuBLAS GEMM
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B*H, N, N)
    right_perm = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Matrix multiplication with transposed second operand
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # (B*H, N, N)

    # Restore original layout [B, N, N, hidden_dim]
    out = out_perm.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4) Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    out_norm = layernorm_triton(out, to_out_norm_weight, to_out_norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 5) Apply out_gate and final projection back to dim
    # ------------------------------------------------------------------
    out_gated = out_norm * out_gate
    output = F.linear(out_gated, to_out_weight)   # shape [B,N,N,dim]

    return output
