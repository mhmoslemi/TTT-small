def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr
):
    pid_m = tl.program_id(0)  # i tile
    pid_n = tl.program_id(1)  # j tile
    pid_z = tl.program_id(2)  # batch * hidden

    # compute batch and hidden index
    b = pid_z // H
    h = pid_z % H

    # compute start offsets
    i0 = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j0 = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_i = i0 < N
    mask_j = j0 < N

    # accumulate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute offsets for k block
        k0 = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k0 < N

        # compute offset for left
        # broadcast i0 across k0
        left_ptrs = left_ptr + b*stride_left_b + i0[:, None] * stride_left_i \
                    + k0[None, :] * stride_left_k + h * stride_left_h

        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # compute offset for right
        right_ptrs = right_ptr + b*stride_right_b + j0[:, None] * stride_right_i \
                     + k0[None, :] * stride_right_k + h * stride_right_h

        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # dot product across k dimension
        acc += tl.dot(left, right)  # shape (BLOCK_M, BLOCK_N)
    # store output
    out_ptrs = out_ptr + b*stride_out_b + i0[:, None] * stride_out_i + j0[None, :] * stride_out_j + h * stride_out_h
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])

    # For convenience, we cast to floats
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # weights: norm.weight shape (dim,), norm.bias shape (dim,)
    # compute mean & var across last dim
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)

    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    var_out = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_out) / torch.sqrt(var_out + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # multiply by out_gate
    out_scaled = out_norm * out_gate

    # final linear projection (hidden_dim -> dim)
    output = torch.nn.functional.linear(out_scaled, weights["to_out.weight"])

    return output

"""
TrTriMul outgoing kernel (AlphaFold3 style).

Algorithm
---------
1. Layer‑normalize the input tensor.
2. Linear projections + sigmoid gates (left, right, out_gate).
3. Apply mask and gates to the projected tensors.
4. Compute the core “outgoing” multiplicative update:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched Gram‑matrix computation and is implemented in
   Triton using a blocked GEMM with `tl.dot` (the reduction over *k*).
5. Layer‑normalize the intermediate result, scale by the out‑gate and
   apply a final linear projection back to the original channel dimension.

Only the heavy N³ reduction (step 4) is performed with a custom Triton
kernel; all other operations are done with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,            # ptr to left tensor (fp16)
    right_ptr,           # ptr to right tensor (fp16)
    out_ptr,             # ptr to output tensor (fp32)
    B, N, H,             # batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    """Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_m = tl.program_id(0)      # i‑tile
    pid_n = tl.program_id(1)      # j‑tile
    pid_z = tl.program_id(2)      # fused batch * hidden index

    # ------------------------------------------------------------------
    # Derive batch and hidden indices from the fused dimension
    # ------------------------------------------------------------------
    b = pid_z // H
    h = pid_z % H

    # ------------------------------------------------------------------
    # Offsets for the i and j tiles
    # ------------------------------------------------------------------
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # ------------------------------------------------------------------
    # Accumulator in FP32
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k in blocks of size BLOCK_K
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ----- load left[b, i, k, h] -----
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + i[:, None] * stride_left_i
            + k[None, :] * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- load right[b, j, k, h] -----
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + j[:, None] * stride_right_i
            + k[None, :] * stride_right_k
            + h * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- dot‑product over k (tl.dot reduces over the last dim) -----
        acc += tl.dot(left, right)      # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Write the accumulated result
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i[:, None] * stride_out_i
        + j[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul update.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # -------------------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------------------
    # Input LayerNorm
    # -------------------------------------------------------------
    # Normalization across the channel dimension (last dim)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]          # [B,N,N,dim]

    # -------------------------------------------------------------
    # Linear projections (no bias)
    # -------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------------------
    # Optional masking
    # -------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.to(x.dtype).unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # -------------------------------------------------------------
    # Gates (sigmoid)
    # -------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------
    # Prepare for Triton kernel (convert to FP16 for memory bandwidth)
    # -------------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    B, N, _, _ = left_h.shape

    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # -------------------------------------------------------------
    # Strides (in elements, not bytes)
    # -------------------------------------------------------------
    s_lb, s_li, s_lk, s_lh = left_h.stride()
    s_rb, s_ri, s_rk, s_rh = right_h.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # -------------------------------------------------------------
    # Kernel launch configuration
    # -------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,           # i tiles
        (N + BLOCK_N - 1) // BLOCK_N,           # j tiles
        B * hidden_dim,                         # fused batch‑hidden dimension
    )

    tri_mul_outgoing_kernel[grid](
        left_h,
        right_h,
        out,
        B,
        N,
        hidden_dim,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    torch.cuda.synchronize()

    # -------------------------------------------------------------
    # Output LayerNorm (hidden_dim -> hidden_dim)
    # -------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # -------------------------------------------------------------
    # Apply out_gate and final linear projection
    # -------------------------------------------------------------
    out_scaled = out_norm * out_gate
    output = torch.nn.functional.linear(out_scaled, weights["to_out.weight"])

    return output
