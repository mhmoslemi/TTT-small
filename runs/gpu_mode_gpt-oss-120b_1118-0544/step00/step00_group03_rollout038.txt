@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_ai, stride_ak, stride_ad,
    stride_bb, stride_bk, stride_bj, stride_bd,
    stride_cb, stride_ci, stride_cj, stride_cd,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # row tile i
    pid_n = tl.program_id(1)  # col tile j
    pid_b = tl.program_id(2)  # batch hidden index

    # compute global offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # block for accumulated result
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # pointers to start of this batch in A and B
    a_batch_ptr = a_ptr + pid_b * stride_ab
    b_batch_ptr = b_ptr + pid_b * stride_bb

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        # offsets for A: rows i, cols k
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_ai + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (cur_k[None, :] < K), other=0.0)
        # offsets for B (transposed): rows k, cols j
        b_ptrs = b_batch_ptr + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=(cur_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)
        acc += tl.dot(a, b)
    # store results
    c_ptrs = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float32), mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

```
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,  # a: batch, row (i), col (k)
    stride_bm, stride_bk, stride_bj,  # b: batch, row (k), col (j)   (transposed)
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile row i
    pid_n = tl.program_id(1)   # tile col j
    pid_b = tl.program_id(2)   # batch index

    # Compute actual offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # Get base pointers for this batch
    a_batch_ptr = a_ptr + pid_b * stride_am
    b_batch_ptr = b_ptr + pid_b * stride_bm

    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        # Load A tile
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_ai + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (cur_k[None, :] < K), other=0.0)
        # Load B tile (transposed B: rows = k, cols = j)
        b_ptrs = b_batch_ptr + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=(cur_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)
        acc += tl.dot(a, b)
    # Store output
    c_ptrs = c_ptr + pid_b * stride_cm + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float32), mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))
```

"""
TriMul outgoing implementation (AlphaFold3).

Algorithm:
1. Layer‑norm the input pair representation.
2. Linear projection to hidden dimension for “left” and “right”.
3. Apply optional pair mask and sigmoid gates.
4. Core O(N³) update:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication of left[:, :, :, d] with
   right[:, :, :, d]ᵀ for each hidden channel d.
   Implemented with a Triton GEMM kernel (block‑wise, FP32 accumulation).
5. Layer‑norm over the hidden dimension, apply an output gate,
   and a final linear projection back to the original dimension.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batch GEMM for one hidden slice (A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                           # matrix sizes (all = seq_len)
    stride_am, stride_ai, stride_ak,    # A: batch, row(i), col(k)
    stride_bm, stride_bk, stride_bj,    # B (transposed): batch, row(k), col(j)
    stride_cm, stride_ci, stride_cj,    # C: batch, row(i), col(j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # tile row index i
    pid_n = tl.program_id(1)  # tile col index j
    pid_b = tl.program_id(2)  # batch index (original batch)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # base pointers for this batch
    a_batch = a_ptr + pid_b * stride_am
    b_batch = b_ptr + pid_b * stride_bm

    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_ai + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B tile (transposed view): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # write back C tile
    c_ptrs = c_ptr + pid_b * stride_cm + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Performs the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask_tensor, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim]
            mask_tensor : torch.Tensor [B, N, N] (broadcastable)
            weights     : dict of model weights
            config      : dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- 1. LayerNorm ----------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B,N,N,dim]

    # ---------- 2. Linear projections ----------
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B,N,N,hidden_dim]

    # ---------- 3. Optional mask ----------
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- 4. Gates ----------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left  = left  * left_gate
    right = right * right_gate

    # ---------- 5. Core TriMul (batched GEMM per hidden slice) ----------
    B, N, _, _ = left.shape
    out_hidden = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=left.device)

    # Strides (in elements, not bytes)
    s_bl, s_il, s_kl, s_dl = left.stride()
    s_br, s_ir, s_kr, s_dr = right.stride()
    s_outb, s_outi, s_outj, s_outd = out_hidden.stride()

    # Triton tile sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Launch a separate GEMM kernel for each hidden channel
    for d in range(hidden_dim):
        # Compute byte offsets for the d‑th hidden slice
        offset_bytes = d * s_dl * left.element_size()
        a_ptr = left.data_ptr() + offset_bytes
        b_ptr = right.data_ptr() + offset_bytes
        c_ptr = out_hidden.data_ptr() + d * s_outd * out_hidden.element_size()

        grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B)
        tri_mul_gemm_kernel[grid](
            a_ptr, b_ptr, c_ptr,
            N, N, N,
            s_bl, s_il, s_kl,           # A strides
            s_br, s_kr, s_ir,           # B strides (transposed)
            s_outb, s_outi, s_outj,     # C strides
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        )

    # ---------- 6. Output LayerNorm and linear ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    output = torch.nn.functional.linear(out, to_out_weight)  # [B,N,N,dim]

    return output
