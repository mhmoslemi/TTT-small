import torch
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a, b, output, size, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a_val = tl.load(a + offs, mask=mask).to(tl.float32)
    b_val = tl.load(b + offs, mask=mask).to(tl.float32)
    out = a_val * b_val
    tl.store(output + offs, out, mask=mask)

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul (outgoing) module.
    Steps:
    1. LayerNorm on the input.
    2. Linear projections for left/right and gates.
    3. Optional mask application.
    4. Triton kernel to fuse gate multiplication.
    5. Batched matrix multiplication (left @ right^T) using torch.bmm.
    6. Output LayerNorm, gating, and final linear projection.
    """
    import torch
    from torch.nn.functional import linear, layer_norm, sigmoid
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # 1. Input LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Note: layer_norm expects normalized_shape as last dimension.
    x = layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # 2. Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = linear(x, left_proj_weight)  # bias=None
    right = linear(x, right_proj_weight)

    # 2b. Gates (pre-sigmoid)
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = sigmoid(linear(x, left_gate_weight))
    right_gate = sigmoid(linear(x, right_gate_weight))
    out_gate = sigmoid(linear(x, out_gate_weight))

    # 3. Optional mask
    if mask is not None and not config.get("nomask", False):
        # mask is [B, N, N]; expand to broadcast over hidden_dim
        mask_exp = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # 4. Fuse gating via Triton
    # left = left * left_gate ; right = right * right_gate
    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    # Triton kernel launch parameters
    BLOCK_SIZE = 1024
    total_elements = left.numel()
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    elementwise_mul_kernel[grid](
        left.reshape(-1), left_gate.reshape(-1), left_gated.reshape(-1),
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    elementwise_mul_kernel[grid](
        right.reshape(-1), right_gate.reshape(-1), right_gated.reshape(-1),
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # 5. Batched matrix multiplication: left @ right^T per hidden channel
    # Permute to (B, hidden_dim, N, N) and (B, hidden_dim, N, N) with transpose on last two for right
    B, N, _, _ = left_gated.shape
    # left: (B, N, N, H) -> (B, H, N, N)
    left_perm = left_gated.permute(0, 3, 1, 2).contiguous()
    # right: (B, N, N, H) -> (B, H, N, N) where we transpose i/k
    right_perm = right_gated.permute(0, 3, 2, 1).contiguous()

    # Flatten batch and hidden_dim
    batch_hd = B * hidden_dim
    A = left_perm.view(batch_hd, N, N)
    B_mat = right_perm.view(batch_hd, N, N)

    # Use torch.bmm for batched matmul
    C = torch.bmm(A, B_mat)  # (batch_hd, N, N)

    # Reshape back to [B, N, N, hidden_dim]
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Output LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # 7. Multiply by out_gate
    out = out * out_gate

    # 8. Final linear projection to original dim
    to_out_weight = weights["to_out.weight"]
    out = linear(out, to_out_weight)

    return out
```

import torch
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK_SIZE: tl.constexpr):
    """Elementwise multiplication of two vectors a*b into out (float32)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask).to(tl.float32)
    b = tl.load(b_ptr + offs, mask=mask).to(tl.float32)
    out = a * b
    tl.store(out_ptr + offs, out, mask=mask)

def custom_kernel(data):
    """
    TriMul (outgoing) forward pass using Triton for gate fusion.
    Steps:
    1. LayerNorm on input.
    2. Linear projections for left/right and gates.
    3. Optional mask application.
    4. Triton kernel to fuse left/right gating.
    5. Batched matrix multiplication via torch.bmm (left @ right^T) per hidden channel.
    6. Output LayerNorm, gating, and final linear projection back to original dim.
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim].
    """
    import torch.nn.functional as F

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # 1. LayerNorm on input (dim)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, normalized_shape=(dim,),
                     weight=norm_weight, bias=norm_bias, eps=1e-5)

    # 2. Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # 2b. Gates (pre‑sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # 3. Apply mask if present and not disabled
    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # 4. Fuse gating via Triton (left = left * left_gate, right = right * right_gate)
    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    total_elements = left.numel()
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    elementwise_mul_kernel[grid](
        left.reshape(-1), left_gate.reshape(-1), left_gated.reshape(-1),
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    elementwise_mul_kernel[grid](
        right.reshape(-1), right_gate.reshape(-1), right_gated.reshape(-1),
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # 5. Batched matmul over the k dimension for each hidden channel.
    B, N, _, _ = left_gated.shape  # left_gated is [B,N,N,hidden_dim]
    # reshape to (B, hidden_dim, N, N) and transpose right appropriately
    left_perm = left_gated.permute(0, 3, 1, 2).contiguous()      # [B, H, N, N]
    right_perm = right_gated.permute(0, 3, 2, 1).contiguous()    # [B, H, N, N] (i ↔ k)

    batch_hd = B * hidden_dim
    A = left_perm.view(batch_hd, N, N)
    B_mat = right_perm.view(batch_hd, N, N)

    C = torch.bmm(A, B_mat)   # (batch_hd, N, N)

    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    # 6. Output LayerNorm over hidden_dim
    out = F.layer_norm(out, normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=1e-5)

    # 7. Apply out_gate
    out = out * out_gate

    # 8. Final linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"])

    return out

import torch
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK_SIZE: tl.constexpr):
    """Elementwise multiplication a * b → out (float32)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask).to(tl.float32)
    b = tl.load(b_ptr + offs, mask=mask).to(tl.float32)
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    TriMul (outgoing) forward pass with Triton‑fused gating.
    1️⃣ Layer‑norm the input.
    2️⃣ Linear projections for left/right channels.
    3️⃣ Compute sigmoid gates.
    4️⃣ (Optional) apply mask.
    5️⃣ Fuse left/right * gate via a small Triton kernel.
    6️⃣ Batched mat‑mul (left @ rightᵀ) per hidden channel using torch.bmm.
    7️⃣ Output LayerNorm over hidden_dim.
    8️⃣ Multiply by out‑gate and final linear projection to `dim`.
    Returns a tensor of shape [bs, seq_len, seq_len, dim].
    """
    import torch.nn.functional as F

    # unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # --------------------------------------------------------------------- #
    # 1. LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------- #
    # 3. Gates (sigmoid)
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------------- #
    # 4. Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------------- #
    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------------- #
    # 5. Fuse gating via Triton (left = left * left_gate, right = right * right_gate)
    # --------------------------------------------------------------------- #
    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    elementwise_mul_kernel[grid](
        left.reshape(-1),           # a_ptr
        left_gate.reshape(-1),      # b_ptr
        left_gated.reshape(-1),     # out_ptr
        total,
        BLOCK_SIZE=BLOCK,
    )
    elementwise_mul_kernel[grid](
        right.reshape(-1),
        right_gate.reshape(-1),
        right_gated.reshape(-1),
        total,
        BLOCK_SIZE=BLOCK,
    )

    # --------------------------------------------------------------------- #
    # 6. Batched mat‑mul: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as (B*hd) independent GEMMs with torch.bmm.
    # --------------------------------------------------------------------- #
    B, N, _, _ = left_gated.shape               # left_gated: [B, N, N, hidden_dim]

    # reshape to (B, hidden_dim, N, N) and prepare the transpose for right
    left_perm = left_gated.permute(0, 3, 1, 2).contiguous()      # [B, H, N, N] (i,k)
    right_perm = right_gated.permute(0, 3, 2, 1).contiguous()    # [B, H, N, N] (k,i)

    batch_hd = B * hidden_dim
    A = left_perm.view(batch_hd, N, N)      # (B*H, N, N)
    B_mat = right_perm.view(batch_hd, N, N) # (B*H, N, N)

    C = torch.bmm(A, B_mat)                 # (B*H, N, N)

    # reshape back to [B, N, N, hidden_dim]
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 7. Output LayerNorm over hidden_dim
    # --------------------------------------------------------------------- #
    out = F.layer_norm(

        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # 8. Apply out‑gate and final linear projection back to `dim`
    # --------------------------------------------------------------------- #
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
