def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    # extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if needed
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(left.dtype)  # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute paired matrix multiplication via Triton
    out = torch.empty((bs, N, N, hidden_dim), dtype=left.dtype, device=device)

    # Launch Triton kernel for each batch and hidden channel
    # Pre-calc grid sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    for b in range(bs):

            bmat = right[b, :, :, h]
            c = out[b, :, :, h]

                BLOCK_M=BLOCK_M,
                BLOCK_N=BLOCK_N,

"""
TriMul "outgoing" forward pass with a Triton‐accelerated 3‑tensor contraction.

The heavy part of the TriMul module is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batch of matrix multiplications: for each batch b and hidden
dimension d we compute  out_d = left_d @ right_dᵀ .
The implementation below:

* Performs LayerNorm, linear projections and sigmoid gates in pure PyTorch.
* Fuses the N³ contraction with a custom Triton kernel that tiles the
  (i, j) output plane and accumulates over k.
* Applies the final LayerNorm, out‑gate and output linear layer in PyTorch.

The kernel works for any batch size (≤2), sequence length N (≤1024) and
hidden dimension H (≤128).  It uses 32‑bit accumulation for numerical
stability.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ  where A and B are square N×N matrices.
# The kernel is launched for each (batch, hidden) pair.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix sizes (all equal to seq_len)
    stride_am, stride_ak,                    # strides of A (row, col)
    stride_bn, stride_bk,                    # strides of B (row, col) – note we load Bᵀ
    stride_cm, stride_cn,                    # strides of C (row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(axis=0)   # block index over rows (i)
    pid_n = tl.program_id(axis=1)   # block index over cols (j)

    # Compute the absolute offsets of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # A[i, k] block
        a_ptrs = a_ptr + (offs_m[:, None] * stride_am + k_off[None, :] * stride_ak)
        a = tl.load(
            a_ptrs,
            mask=(offs_m[:, None] < M) & (k_off[None, :] < K),
            other=0.0,
        )

        # Bᵀ[k, j] block  (i.e. B[j, k] in memory)
        b_ptrs = b_ptr + (k_off[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        b = tl.load(
            b_ptrs,
            mask=(k_off[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )

        # Accumulate the dot product
        acc += tl.dot(a, b)   # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)

    # Write the result back to C
    c_ptrs = c_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask)

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (may be ignored)
        - weights      : dict of torch.Tensors containing all model parameters
        - config       : dict with keys:
            * "dim" (int)
            * "hidden_dim" (int)
            * optional "nomask" (bool, default True)
            * optional "eps" (float, default 1e-5)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim]  (float32, on the same device as input)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Extract model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on input
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3. Optional masking
    # ------------------------------------------------------------------
    if not nomask:
        m = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton accelerated contraction: out = Σ_k left[...,k,:] * right[...,k,:]
    #    We treat each (batch, hidden) pair as an independent matrix‑multiply.
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=device)

    # Tiling parameters (tuned for typical B200 memory hierarchy)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N

    for b_idx in range(B):
        for h_idx in range(hidden_dim):
            a_mat = left[b_idx, :, :, h_idx]   # [N, N]
            b_mat = right[b_idx, :, :, h_idx]  # [N, N]
            c_mat = out[b_idx, :, :, h_idx]    # [N, N]

            stride_am, stride_ak = a_mat.stride()
            stride_bn, stride_bk = b_mat.stride()
            stride_cm, stride_cn = c_mat.stride()

            # Launch the Triton kernel for this (b, h) slice
            _tri_mul_kernel[
                (grid_m, grid_n)
            ](
                a_ptr=a_mat.data_ptr(),
                b_ptr=b_mat.data_ptr(),
                c_ptr=c_mat.data_ptr(),
                M=N,
                N=N,
                K=N,
                stride_am=stride_am,
                stride_ak=stride_ak,
                stride_bn=stride_bn,
                stride_bk=stride_bk,
                stride_cm=stride_cm,
                stride_cn=stride_cn,
                BLOCK_M=BLOCK_M,
                BLOCK_N=BLOCK_N,
                BLOCK_K=BLOCK_K,
                num_warps=4,
            );

    # ------------------------------------------------------------------
    # 6. Post‑processing
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )                                   # [B, N, N, hidden]

    out = out * out_gate                 # gating after norm

    out = torch.nn.functional.linear(out, to_out_weight)  # final projection → [B,N,N,dim]

    return out
