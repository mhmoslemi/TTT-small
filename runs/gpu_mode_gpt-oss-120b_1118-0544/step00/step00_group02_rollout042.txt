@triton.jit
def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr, eps: tl.constexpr):
    pid = tl.program_id(0)  # linear program id
    # Get block indices: we need to map program id to (b, i_block, j_block)
    # Let's combine dimensions: total blocks = B * ceil(N/BLOCK_M) * ceil(N/BLOCK_N)
    # Compute block indices:
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    total_blocks_per_batch = num_i * num_j
    b = pid // total_blocks_per_batch
    block_idx = pid % total_blocks_per_batch
    i_block = block_idx // num_j
    j_block = block_idx % num_j

    # Offsets for i and j
    i_offsets = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offsets = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    # Hidden offsets
    h_offsets = tl.arange(0, BLOCK_H)

    # Create masks for out-of-range i,j,h
    i_mask = i_offsets < N
    j_mask = j_offsets < N
    h_mask = h_offsets < H

    # Pointers
    # left loaded: shape [BLOCK_M, BLOCK_K, BLOCK_H] where k iterates.
    # right loaded: shape [BLOCK_N, BLOCK_K, BLOCK_H]
    # We need to accumulate for each h offset.

    # Initialize accumulator as zero float32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offsets
        k_offsets = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < N

        # load left tile
        # left_ptr + b*stride_bl + i_offsets[:,None,None]*stride_il + k_offsets[None,:,None]*stride_kl + h_offsets[None,None,:]*stride_hl
        left_ptrs = left_ptr + b * stride_bl + i_offsets[:, None, None] * stride_il + k_offsets[None, :, None] * stride_kl + h_offsets[None, None, :] * stride_hl
        left = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :], other=0.0).to(tl.float32)

        # load right tile: shape [BLOCK_N, BLOCK_K, BLOCK_H]
        right_ptrs = right_ptr + b * stride_br + j_offsets[:, None, None] * stride_jr + k_offsets[None, :, None] * stride_kr + h_offsets[None, None, :] * stride_hr
        right = tl.load(right_ptrs, mask=j_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :], other=0.0).to(tl.float32)

        # Multiply and accumulate: sum over k dimension.
        # We need elementwise product and sum across k: left[i,k,h] * right[j,k,h]; this is a multiplication across k dimension. The sum is across k, not across h.
        # For each h, we can compute dot product across k dimension: For each i and j and h:
        # acc[i,j,h] += sum_k left[i,k,h] * right[j,k,h]
        # This can be done by broadcasting the multiplication: left (M,K,H) * right (N,K,H) -> sum over K.
        # We can perform per-h dot product using tl.dot for each h slice. But we can compute using elementwise multiply and then sum across k dimension.
        # Because we need to sum across K dimension, we can use tl.dot for each h: tl.dot(left[:, :, h], right[:, :, h].T) would produce MxN.
        # However we can implement using a loop over h or vectorized via tl.broadcast? For simplicity, perform per-h based on BLOCK_H small: we will compute a dot for each h using tl.dot.

        # We'll accumulate using a for loop over h indexing.
        for h in range(0, BLOCK_H):
            # left slice shape (M,K), right slice shape (N,K)
            # Use tl.dot with transposed right.
            acc[:, :, h] += tl.dot(left[:, :, h], right[:, :, h].T)

    # Store result
    out_ptrs = out_ptr + b * stride_bo + i_offsets[:, None, None] * stride_io + j_offsets[None, :, None] * stride_jo + h_offsets[None, None, :] * stride_ho
    tl.store(out_ptrs, acc, mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :])

@triton.jit
def trmul_forward_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    # compute batch index, block i, block j
    b = pid // (num_i * num_j)
    rem = pid % (num_i * num_j)
    i_block = rem // num_j
    j_block = rem % num_j

    # offsets
    i_idx = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_idx = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    h_idx = tl.arange(0, BLOCK_H)

    # masks
    i_mask = i_idx < N
    j_mask = j_idx < N
    h_mask = h_idx < H

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # loop over K tiles
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_idx = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_idx < N

        # compute pointers to left and right
        # left tile shape [BLOCK_M, BLOCK_K, BLOCK_H]
        left_ptrs = left_ptr + b*stride_bl + i_idx[:, None, None]*stride_il + k_idx[None, :, None]*stride_kl + h_idx[None, None, :]*stride_hl
        left_tile = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :], other=0.0).to(tl.float32)

"""
TriMul Outgoing implementation (AlphaFold3 style).

Algorithm:
1. Layer‑norm the input tensor.
2. Linear projections to a hidden dimension for "left" and "right" tensors.
3. Optional mask application (broadcasted over the hidden dimension).
4. Gating: sigmoid(linear) applied element‑wise to the projected tensors.
5. Core TriMul reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication with a Triton kernel.
   The tensors are reshaped to (B*H, N, N) where H = hidden_dim, then a
   custom GEMM kernel computes the product for all batches in parallel.
6. Layer‑norm + output gating.
7. Final linear projection back to the original feature dimension.

The Triton kernel computes a standard GEMM for each (batch*hidden) slice,
processing tiles of size BLOCK_M×BLOCK_N with a K‑tile loop.
Only the reduction in step 5 is executed on Triton; all surrounding
operations use PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul reduction.
# ----------------------------------------------------------------------
@triton.jit
def _trmul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    BATCH, N,                           # dimensions
    stride_ah, stride_ai, stride_ak,    # strides for A
    stride_bh, stride_bi, stride_bk,    # strides for B
    stride_ch, stride_ci, stride_cj,    # strides for C (output)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k] .
    A,B,C are (batch, N, N). The kernel works on a 2‑D tile (i,j) for each
    batch entry.
    """
    pid = tl.program_id(0)

    # ----- compute batch, i‑tile, j‑tile indices -----
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    batch_id = pid // (num_i * num_j)
    block_id = pid % (num_i * num_j)
    i_block = block_id // num_j
    j_block = block_id % num_j

    # ----- offsets within the current tile -----
    i_off = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    k_off = tl.arange(0, BLOCK_K)

    i_mask = i_off < N
    j_mask = j_off < N

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- K‑loop -----
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = kk * BLOCK_K + k_off
        k_mask = cur_k < N

        # A: (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + batch_id * stride_ah \
                 + i_off[:, None] * stride_ai \
                 + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=i_mask[:, None] & k_mask[None, :],
                    other=0.0)

        # B: (BLOCK_N, BLOCK_K) – we need its transpose, so load as (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + batch_id * stride_bh \
                 + j_off[:, None] * stride_bi \
                 + cur_k[None, :] * stride_bk
        b = tl.load(b_ptrs,
                    mask=j_mask[:, None] & k_mask[None, :],
                    other=0.0)

        # b is (BLOCK_N, BLOCK_K); transpose to (BLOCK_K, BLOCK_N) for dot
        acc += tl.dot(a, b.T)

    # ----- write back -----
    c_ptrs = c_ptr + batch_id * stride_ch \
             + i_off[:, None] * stride_ci \
             + j_off[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

# ----------------------------------------------------------------------
# Front‑end function
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments:
        data: Tuple of (input_tensor, mask, weights, config)
              - input_tensor: [B, N, N, C]      (float32)
              - mask:        [B, N, N] (optional, broadcastable)
              - weights: dict with all linear / layer‑norm parameters
              - config: dict containing "dim", "hidden_dim", and "nomask"

    Returns:
        Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul reduction via a batched GEMM kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    device = left.device

    # reshape to (B*H, N, N) for the kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]
    left_flat = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)

    out_flat = torch.empty_like(left_flat)

    # strides for the flattened tensors (all int64)
    a_batch_stride, a_i_stride, a_k_stride = left_flat.stride()
    b_batch_stride, b_i_stride, b_k_stride = right_flat.stride()
    c_batch_stride, c_i_stride, c_j_stride = out_flat.stride()

    # kernel launch configuration
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B * hidden_dim * num_i * num_j
    grid = (total_blocks,)

    _trmul_gemm_kernel[grid](
        left_flat.data_ptr(),
        right_flat.data_ptr(),
        out_flat.data_ptr(),
        B * hidden_dim,
        N,
        a_batch_stride,
        a_i_stride,
        a_k_stride,
        b_batch_stride,
        b_i_stride,
        b_k_stride,
        c_batch_stride,
        c_i_stride,
        c_j_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6) Output layer‑norm and gating
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)  # [B, N, N, C]

    return out
