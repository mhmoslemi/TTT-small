def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B,N,N2,dim = input_tensor.shape
    assert N == N2

    # get dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layer_norm_epsilon", 1e-5)

    # load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if not None
    if mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(left)  # broadcast to hidden dim
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Compute contraction via Triton batched matmul kernel
    # Rearrange for batch matmul
    # permute dims to bring hidden_dim after batch for view
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose last two dims for right
    right_perm_t = right_perm.transpose(2, 3).contiguous()  # shape [B, hidden, N, N]

    # reshape to (batch_dim, M, K) and (batch_dim, K, N)
    B_hd = left_perm.shape[0] * left_perm.shape[1]  # B * hidden_dim
    left_mat = left_perm.view(B_hd, N, N)  # [B*hidden,N,N]
    right_mat = right_perm_t.view(B_hd, N, N)  # shape [B*hidden,N,N]; this is B*K,N? Actually relation: right_mat is B*hidden x K x N where K = N, N = N.

    # Compute output mat using Triton kernel
    # allocate output
    out_mat = torch.empty_like(left_mat)  # shape [B*hidden,N,N]

    # call kernel
    grid = (B_hd, (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)
    # but kernel expects program_id(0) batch axis, program_id(1) row block, program_id(2) col block.

    matmul_batch_kernel[grid](
        left_mat, right_mat, out_mat,
        N, N, N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()  # ensure kernel done

    # reshape out_mat back to [B, hidden, N, N]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # Apply to_out_norm layernorm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim]

    return out

@triton.jit
def matmul_batch_kernel(a_ptr, b_ptr, c_ptr,
                        M, N, K,
                        stride_aba, stride_a_row, stride_a_col,
                        stride_bba, stride_b_row, stride_b_col,
                        stride_cca, stride_c_row, stride_c_col,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to start of batch
    a_batch_ptr = a_ptr + pid_batch * stride_aba
    b_batch_ptr = b_ptr + pid_batch * stride_bba
    c_batch_ptr = c_ptr + pid_batch * stride_cca

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        # compute mask for K dimension
        mask_k = offs_k < K

        # load A tile
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_row) + (offs_k[None, :] * stride_a_col)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B tile
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_b_row) + (offs_n[None, :] * stride_b_col)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

    # write back
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_row) + (offs_n[None, :] * stride_c_col)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul "outgoing" kernel (AlphaFold3 style)

Algorithm
---------
1. Layer‑norm the input tensor `x` over its last dimension.
2. Linear projections for the left/right tensors and their sigmoid gates.
   Apply a (optional) pairwise mask before gating.
3. Fuse the left/right tensors into half‑precision to reduce memory pressure.
4. Compute the contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batched matrix multiplication for each hidden channel `d`.
   This is performed by a custom Triton kernel that treats the
   `(batch, hidden)` pair as the batch dimension of the GEMM.
5. Reshape the result back to `[B,N,N,hidden]`.
6. Layer‑norm over the hidden dimension, apply the output gate and a final
   linear projection back to `dim`.

The Triton kernel implements a strided‑batched GEMM:
C[b] = A[b] @ B[b]^T  for each batch `b`.
All other ops are executed with PyTorch for simplicity; the key O(N³)
reduction is accelerated on the GPU.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton batched GEMM kernel (C = A @ B^T)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # batch (including hidden) dimension
    pid_m = tl.program_id(1)              # block row
    pid_n = tl.program_id(2)              # block column

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # batch pointers
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # accumulator in FP32 (stable for many K‑steps)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for the current tile
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (offs_k[None, :] * stride_a_k)
        b_ptrs = b_batch + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)

        # load tiles (FP16 -> FP32)
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0,
                    dtype=tl.float16).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0,
                    dtype=tl.float16).to(tl.float32)

        # rank‑k update
        acc += tl.dot(a, b)

    # write out
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Main entry point ----------------------------------------------------
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the TriMul operator (outgoing version).

    Args:
        data: (input, mask, weights, config)
            - input: Tensor[B, N, N, dim]
            - mask : Tensor[B, N, N] (may be None)
            - weights: dict with the following keys
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight',
                'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            - config: dict with entries 'dim' and 'hidden_dim' (and optional
              'layer_norm_epsilon')

    Returns:
        Tensor[B, N, N, dim] – result of the TriMul forward pass.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layer_norm_epsilon", 1e-5)

    # ------------------------------------------------------------------
    # layer‑norm on the input
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # linear projections (no bias) and mask handling
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    if mask is not None:
        mask_ = mask.to(dtype).unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # cast to half‑precision before the heavy O(N³) reduction
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # bring hidden dimension next to batch for a strided‑batched GEMM
    # shape: [B, hidden, N, N] -> (B*hidden, N, N)
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()
    # right needs to be transposed on the last two axes for the inner‑product definition
    right_perm_t = right_perm.transpose(2, 3).contiguous()

    B, H, N, _ = left_perm.shape
    batch_hidden = B * H

    A = left_perm.view(batch_hidden, N, N)          # [B*H, M=N, K=N]
    B_mat = right_perm_t.view(batch_hidden, N, N)   # [B*H, K=N, N]

    # output tensor (FP16) – will be cast back to FP32 later
    C = torch.empty_like(A, device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # launch Triton GEMM
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_hidden,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        A,
        B_mat,
        C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),
        B_mat.stride(0), B_mat.stride(1), B_mat.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # make sure kernel has finished
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # reshape back to [B,N,N,hidden] in FP32
    # ------------------------------------------------------------------
    out = C.to(torch.float32).view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # post‑processing: LayerNorm over hidden dim, gating, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                                     # broadcast over dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # [B,N,N,dim]

    return out
