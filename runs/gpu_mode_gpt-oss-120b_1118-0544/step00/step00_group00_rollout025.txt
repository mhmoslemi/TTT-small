@triton.jit
def gated_layernorm_kernel(
    out_ptr, out_gate_ptr, norm_weight_ptr, norm_bias_ptr,
    B, N, hidden_dim,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    stride_g_b, stride_g_i, stride_g_j, stride_g_d,
    stride_nw, stride_nb,
    eps: tl.constexpr,
    BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    b = pid // (N * N)
    tmp = pid % (N * N)
    i = tmp // N
    j = tmp % N

    # compute base pointer for out and gate
    out_base = out_ptr + b * stride_o_b + i * stride_o_i + j * stride_o_j
    gate_base = out_gate_ptr + b * stride_g_b + i * stride_g_i + j * stride_g_j

    # ---- first pass: compute mean ----
    sum_val = tl.float32(0.0)
    for d in range(0, hidden_dim, BLOCK_D):
        cur_block = tl.arange(0, BLOCK_D)
        mask = cur_block < (hidden_dim - d)
        # load values
        out_vals = tl.load(out_base + d * stride_o_d + cur_block * stride_o_d,
                           mask=mask, other=0.0)
        gate_vals = tl.load(gate_base + d * stride_g_d + cur_block * stride_g_d,
                            mask=mask, other=0.0)
        gated = out_vals * gate_vals
        sum_val += tl.sum(gated, axis=0)

    mean = sum_val / hidden_dim

    # ---- second pass: compute variance ----
    sum_sq = tl.float32(0.0)
    for d in range(0, hidden_dim, BLOCK_D):
        cur_block = tl.arange(0, BLOCK_D)
        mask = cur_block < (hidden_dim - d)
        out_vals = tl.load(out_base + d * stride_o_d + cur_block * stride_o_d,
                           mask=mask, other=0.0)
        gate_vals = tl.load(gate_base + d * stride_g_d + cur_block * stride_g_d,
                            mask=mask, other=0.0)
        gated = out_vals * gate_vals
        diff = gated - mean
        sum_sq += tl.sum(diff * diff, axis=0)

    var = sum_sq / hidden_dim
    inv_std = 1.0 / tl.sqrt(var + eps)

    # norm weight and bias: 1D vectors of length hidden_dim
    # load in blocks, apply and store back
    for d in range(0, hidden_dim, BLOCK_D):
        cur_block = tl.arange(0, BLOCK_D)
        mask = cur_block < (hidden_dim - d)
        out_vals = tl.load(out_base + d * stride_o_d + cur_block * stride_o_d,
                           mask=mask, other=0.0)
        gate_vals = tl.load(gate_base + d * stride_g_d + cur_block * stride_g_d,
                            mask=mask, other=0.0)
        gated = out_vals * gate_vals
        normed = (gated - mean) * inv_std
        w = tl.load(norm_weight_ptr + d * stride_nw + cur_block * stride_nw,
                    mask=mask, other=0.0)
        b = tl.load(norm_bias_ptr + d * stride_nb + cur_block * stride_nb,
                    mask=mask, other=0.0)
        out_res = normed * w + b
        tl.store(out_base + d * stride_o_d + cur_block * stride_o_d,
                 out_res, mask=mask)

    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Cast maybe to float32 if not.
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    # Use torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps
    )
    # Projection weights
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    # Compute left and right projections
    # Using torch.nn.functional.linear for efficient compute
    # input shape [B,N,N,dim]; linear expects last dim weight shape (out_features, in_features)
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)
    # Apply mask if needed
    if mask is not None:
        mask_unsqueeze = mask.unsqueeze(-1)
        left_proj = left_proj * mask_unsqueeze
        right_proj = right_proj * mask_unsqueeze
    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate
    # Convert to float16 for speed (optional)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Reshape for batched matmul.
    # left: (B,N,N,hidden_dim) -> (B*hidden_dim,N,N)
    left_mat = left.permute(0,3,1,2).reshape(B*hidden_dim, N, N)

    right_mat = right.permute(0,3,2,1).reshape(B*hidden_dim, N, N)

    out = torch.bmm(left_mat, right_mat)  # shape (B*hidden_dim,N,N)
    # Reshape back
    out = out.reshape(B, hidden_dim, N, N).permute(0,2,3,1)  # [B,N,N,hidden_dim]

    to_out_norm_bias = weights["to_out_norm.bias"]
    # Launch Triton kernel

    total_elements = B * N * N
    # Choose block size for hidden_dim loops

    out_gate_strides = out_gate.stride()
    # Norm weight and bias strides (1D)
    norm_w_stride = to_out_norm_weight.stride(0)
    norm_b_stride = to_out_norm_bias.stride(0)

    gated_layernorm_kernel[grid](

    )

    # Final linear projection to output dim
    to_out_weight = weights["to_out.weight"]
    final = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim]

def custom_kernel(data):

    import triton
    import triton.language as tl
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # layernorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps
    )

    # linear projections and gates
    # Cast weights to appropriate dtype (float16) for speed
    dtype = torch.float16
    # Ensure input for linear is float16 after casting
    x_norm_h = x_norm.to(dtype)

    left_proj = torch.nn.functional.linear(x_norm_h, weights["left_proj.weight"].to(dtype))
    right_proj = torch.nn.functional.linear(x_norm_h, weights["right_proj.weight"].to(dtype))

    if mask is not None:
        mask_h = mask.unsqueeze(-1).to(dtype)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_h, weights["left_gate.weight"].to(dtype))).to(dtype)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_h, weights["right_gate.weight"].to(dtype))).to(dtype)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # reshape for batched matmul
    left_mat = left.permute(0,3,1,2).reshape(B*hidden_dim, N, N)
    right_mat = right.permute(0,3,2,1).reshape(B*hidden_dim, N, N)

    out = torch.bmm(left_mat, right_mat)  # half precision
    # reshape back
    out = out.reshape(B, hidden_dim, N, N).permute(0,2,3,1)  # [B,N,N,hidden_dim]

    # compute out_gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_h, weights["out_gate.weight"].to(dtype))).to(dtype)

    # final layernorm + gating using Triton kernel
    # kernel defined later
    BLOCK_D = 32
    total_elements = B * N * N
    grid = (total_elements,)
    out_strides = out.stride()
    out_gate_strides = out_gate.stride()
    # norm weight & bias: 1D
    norm_w = weights["to_out_norm.weight"]
    norm_b = weights["to_out_norm.bias"]

        norm_w, norm_b,

        out_gate_strides[0], out_gate_strides[1], out_gate_strides[2], out_gate_strides[3],

    )

    # final linear projection
    to_out_weight_h = weights["to_out.weight"].to(dtype)
    final = torch.nn.functional.linear(out, to_out_weight_h)  # [B,N,N,dim] half
    # convert back to float32
    return final.to(torch.float32)

    out_ptr, gate_ptr,
    norm_weight_ptr, norm_bias_ptr,
    B, N, hidden_dim,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    stride_g_b, stride_g_i, stride_g_j, stride_g_d,
    eps,
    BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # compute indices
    b = pid // (N * N)
    temp = pid % (N * N)
    i = temp // N
    j = temp % N

    # base pointers
    out_base = out_ptr + b * stride_o_b + i * stride_o_i + j * stride_o_j
    gate_base = gate_ptr + b * stride_g_b + i * stride_g_i + j * stride_g_j

    # ---- compute mean ----
    sum_val = tl.float32(0.0)
    for d in range(0, hidden_dim, BLOCK_D):
        cur = tl.arange(0, BLOCK_D)
        valid = cur < (hidden_dim - d)
        offs = d + cur

        gate_vals = tl.load(gate_base + offs * stride_g_d, mask=valid, other=0.0).to(tl.float32)

    # ---- compute variance ----

"""
TriMul (outgoing) forward implementation with Triton acceleration.

Algorithm
---------
1. Layer‑normalize the input tensor `x` over the last dimension (dim).
2. Linear projections:
   - left  = (x_norm @ left_proj_weight.T)  * left_gate
   - right = (x_norm @ right_proj_weight.T) * right_gate
   Optional mask is applied before gating.
   Computations are performed in fp16 for the heavy matmul.
3. Batched matrix multiplication:
   For each hidden channel `h`,
       out[..., i, j, h] = Σ_k left[..., i, k, h] * right[..., j, k, h]
   This is realised as a batched GEMM `left @ right.transpose(-1, -2)`.
4. Compute the output gate `out_gate = sigmoid(x_norm @ out_gate_weight.T)`.
5. Fuse element‑wise gating, LayerNorm over the hidden dimension,
   and the per‑channel affine transformation (`to_out_norm`) in a Triton
   kernel (`gated_layernorm_kernel`).
6. Final linear projection `to_out` maps the normalised hidden features
   back to the original `dim` dimension.

The Triton kernel processes one (batch, i, j) element per program and
iterates over the hidden dimension in blocks, performing a two‑pass
mean/variance computation followed by the normalisation and affine
transform. All heavy reductions are performed on‑GPU, while the outer
projections and batched GEMM leverage PyTorch's highly‑optimised kernels.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def gated_layernorm_kernel(
    out_ptr,            # fp16 hidden tensor [B,N,N,H]
    gate_ptr,           # fp16 gate tensor   [B,N,N,H]
    norm_w_ptr,         # fp32 weight vector [H]
    norm_b_ptr,         # fp32 bias   vector [H]
    B, N, H,            # runtime dims
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    stride_g_b, stride_g_i, stride_g_j, stride_g_h,
    eps,                # layer‑norm epsilon
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode (b, i, j) from linear pid
    b = pid // (N * N)
    tmp = pid % (N * N)
    i = tmp // N
    j = tmp % N

    # base pointers for this element
    out_base = out_ptr + b * stride_o_b + i * stride_o_i + j * stride_o_j
    gate_base = gate_ptr + b * stride_g_b + i * stride_g_i + j * stride_g_j

    # ---------- first pass : mean ----------
    sum_val = tl.float32(0.0)
    for off in range(0, H, BLOCK_H):
        cur = tl.arange(0, BLOCK_H)
        mask = cur < (H - off)
        idx = off + cur
        out_v = tl.load(out_base + idx * stride_o_h, mask=mask, other=0.0).to(tl.float32)
        gate_v = tl.load(gate_base + idx * stride_g_h, mask=mask, other=0.0).to(tl.float32)
        gated = out_v * gate_v
        sum_val += tl.sum(gated, axis=0)

    mean = sum_val / H

    # ---------- second pass : variance ----------
    sum_sq = tl.float32(0.0)
    for off in range(0, H, BLOCK_H):
        cur = tl.arange(0, BLOCK_H)
        mask = cur < (H - off)
        idx = off + cur
        out_v = tl.load(out_base + idx * stride_o_h, mask=mask, other=0.0).to(tl.float32)
        gate_v = tl.load(gate_base + idx * stride_g_h, mask=mask, other=0.0).to(tl.float32)
        gated = out_v * gate_v
        diff = gated - mean
        sum_sq += tl.sum(diff * diff, axis=0)

    var = sum_sq / H
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ---------- third pass : normalize, affine, store ----------
    for off in range(0, H, BLOCK_H):
        cur = tl.arange(0, BLOCK_H)
        mask = cur < (H - off)
        idx = off + cur
        out_v = tl.load(out_base + idx * stride_o_h, mask=mask, other=0.0).to(tl.float32)
        gate_v = tl.load(gate_base + idx * stride_g_h, mask=mask, other=0.0).to(tl.float32)
        gated = out_v * gate_v

        normed = (gated - mean) * inv_std

        w = tl.load(norm_w_ptr + idx, mask=mask, other=0.0)   # fp32
        b = tl.load(norm_b_ptr + idx, mask=mask, other=0.0)   # fp32

        out_fp32 = normed * w + b
        tl.store(out_base + idx * stride_o_h, out_fp32.to(tl.float16), mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated TriMul (outgoing) forward pass.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N]  (may be ignored if config["nomask"])
            - weights: dict of model weights (see problem statement)
            - config: dict with keys "dim", "hidden_dim", and optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float16

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm on input ----------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # fp32

    # Cast to half for the heavy linear operations
    x_h = x_norm.to(dtype)

    # ---------- Linear projections ----------
    left_proj = torch.nn.functional.linear(
        x_h, weights["left_proj.weight"].to(dtype)
    )  # [B,N,N,H]
    right_proj = torch.nn.functional.linear(
        x_h, weights["right_proj.weight"].to(dtype)
    )

    # Optional mask
    if not config.get("nomask", False):
        mask_h = mask.unsqueeze(-1).to(dtype)  # [B,N,N,1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ---------- Gates ----------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_h, weights["left_gate.weight"].to(dtype))
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_h, weights["right_gate.weight"].to(dtype))
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_h, weights["out_gate.weight"].to(dtype))
    )  # [B,N,N,H]

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ---------- Batched GEMM (TriMul reduction) ----------
    # reshape to (B*H, N, N) and (B*H, N, N) with right transposed on the last two axes
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)

    # Use fp16 batched matmul
    out = torch.bmm(left_mat, right_mat)  # (B*H, N, N) in fp16

    # Restore original layout [B,N,N,H]
    out = out.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- Fuse gate, layernorm and affine with Triton ----------
    BLOCK_H = 32  # block size for hidden dimension
    total_elements = B * N * N
    grid = (total_elements,)

    # strides for the tensors (number of elements, not bytes)
    out_strides = out.stride()
    gate_strides = out_gate.stride()
    norm_w = weights["to_out_norm.weight"]
    norm_b = weights["to_out_norm.bias"]

    gated_layernorm_kernel[grid](
        out, out_gate,
        norm_w, norm_b,
        B, N, hidden_dim,
        out_strides[0], out_strides[1], out_strides[2], out_strides[3],
        gate_strides[0], gate_strides[1], gate_strides[2], gate_strides[3],
        eps,
        BLOCK_H=BLOCK_H,
    )

    # ---------- Final linear projection ----------
    to_out_weight = weights["to_out.weight"].to(dtype)
    final = torch.nn.functional.linear(out, to_out_weight)  # [B,N,N,dim] fp16

    return final.to(torch.float32)
