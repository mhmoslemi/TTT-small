@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_l_batch, stride_l_row, stride_l_col,
    stride_r_batch, stride_r_row, stride_r_col,
    stride_o_batch, stride_o_row, stride_o_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loops over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        mask_m = offs_m < N
        mask_n = offs_n < N
        mask_k = offs_k < N

        # Load left block [M,K]
        a_ptrs = left_ptr + batch * stride_l_batch + offs_m[:, None] * stride_l_row + offs_k[None, :] * stride_l_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load right block transposed [K,N] -> we need (k,j)
        b_ptrs = right_ptr + batch * stride_r_batch + offs_k[:, None] * stride_r_col + offs_n[None, :] * stride_r_row
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Convert to float32 for accumulation
        acc += tl.dot(a, b).to(tl.float32)

    # Write back result
    c_ptrs = out_ptr + batch * stride_o_batch + offs_m[:, None] * stride_o_row + offs_n[None, :] * stride_o_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Mask handling
    if not config.get("nomask", False):
        # mask shape [B,N,N], may be bool or float
        if mask.dtype != x_norm.dtype:
            mask = mask.to(x_norm.dtype)
        mask = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Prepare for kernel: convert to half for speed
    left_h = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)  # shape (B, hidden_dim, N, N)
    right_h = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    # Flatten batch*hidden_dim dimension
    B, H, N, _ = left_h.shape
    left_flat = left_h.view(-1, N, N)  # shape (B*H, N, N)
    right_flat = right_h.view(-1, N, N)

    # Output placeholder
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Compute strides
    stride_l_batch, stride_l_row, stride_l_col = left_flat.stride()
    stride_r_batch, stride_r_row, stride_r_col = right_flat.stride()
    stride_o_batch, stride_o_row, stride_o_col = out_flat.stride()

    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    triton_trimul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_l_batch, stride_l_row, stride_l_col,
        stride_r_batch, stride_r_row, stride_r_col,
        stride_o_batch, stride_o_row, stride_o_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape result back to [B,N,N,hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,hidden_dim)
    out = out.to(torch.float32)  # back to fp32

    # Apply to_out_norm and out_gate
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_l_batch, stride_l_row, stride_l_col,
    stride_r_batch, stride_r_row, stride_r_col,
    stride_o_batch, stride_o_row, stride_o_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute batched product out = left @ right^T
    left, right, out are 3‑D tensors of shape (batch, N, N) stored
    with strides provided. The kernel processes one (batch) index per
    program_id(0) and tiles the output matrix with BLOCK_M×BLOCK_N.
    """
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        mask_m = offs_m < N
        mask_n = offs_n < N
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + batch * stride_l_batch + offs_m[:, None] * stride_l_row + offs_k[None, :] * stride_l_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right block transposed: we need (BLOCK_K, BLOCK_N) where first dim is k, second is j
        b_ptrs = right_ptr + batch * stride_r_batch + offs_k[:, None] * stride_r_col + offs_n[None, :] * stride_r_row
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product, cast result to FP32 for accumulation
        acc += tl.dot(a, b).to(tl.float32)

    # Store result
    c_ptrs = out_ptr + batch * stride_o_batch + offs_m[:, None] * stride_o_row + offs_n[None, :] * stride_o_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    Implements LayerNorm, linear projections, gating and the core
    `einsum('... i k d, ... j k d -> ... i j d')` using a custom Triton
    kernel that performs a batched matrix multiplication between `left`
    and the transpose of `right`.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. LayerNorm over channel dimension
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # 2. Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # 3. Optional masking
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # 4. Gating (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # 5. Prepare tensors for the Triton kernel
    # Convert to fp16 to reduce memory bandwidth
    left_h = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # [B, H, N, N]
    right_h = right.permute(0, 3, 1, 2).contiguous().to(torch.float16) # [B, H, N, N]

    B, H, N, _ = left_h.shape
    # Flatten batch * hidden_dim dimension
    left_flat = left_h.view(-1, N, N)
    right_flat = right_h.view(-1, N, N)

    # Allocate output tensor (fp16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Strides (in elements, not bytes)
    stride_l_batch, stride_l_row, stride_l_col = left_flat.stride()
    stride_r_batch, stride_r_row, stride_r_col = right_flat.stride()
    stride_o_batch, stride_o_row, stride_o_col = out_flat.stride()

    # 6. Launch Triton kernel (batched matmul left @ right^T)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    triton_trimul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_l_batch,
        stride_l_row,
        stride_l_col,
        stride_r_batch,
        stride_r_row,
        stride_r_col,
        stride_o_batch,
        stride_o_row,
        stride_o_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # 7. Reshape back to [B, N, N, hidden_dim] and cast to fp32
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # 8. Final LayerNorm on hidden dimension, gating and output projection
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass with a Triton kernel.

The operation is:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

We implement the costly Σ_k reduction as a batched matrix
multiplication left @ right.T for each hidden channel d,
using a custom Triton kernel. All remaining
operations (LayerNorm, linear projections, gating) are performed
with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def triton_trimul_kernel(
    left_ptr,          # const float*  [batch, N, N]
    right_ptr,         # const float*  [batch, N, N]
    out_ptr,           # float*        [batch, N, N]
    N,                 # int64_t matrix dimension
    stride_l_batch, stride_l_row, stride_l_col,
    stride_r_batch, stride_r_row, stride_r_col,
    stride_o_batch, stride_o_row, stride_o_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched matmul: out = left @ right.T
    left/right/out are 3‑D tensors of shape (batch, N, N) stored in
    row‑major order. One batch index per program_id(0); tiles are
    BLOCK_M×BLOCK_N.
    """
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k in range(0, num_k):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        mask_m = offs_m < N
        mask_n = offs_n < N
        mask_k = offs_k < N

        # left block: (M, K)
        a_ptrs = left_ptr + batch * stride_l_batch \
                 + offs_m[:, None] * stride_l_row \
                 + offs_k[None, :] * stride_l_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right block transposed -> (K, N)
        # We need right[k, j] where k varies fastest.
        b_ptrs = right_ptr + batch * stride_r_batch \
                 + offs_k[:, None] * stride_r_col \
                 + offs_n[None, :] * stride_r_row
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product; result is FP16 -> cast to FP32 before accumulation
        acc += tl.dot(a, b).to(tl.float32)

    # Write back
    c_ptrs = out_ptr + batch * stride_o_batch \
             + offs_m[:, None] * stride_o_row \
             + offs_n[None, :] * stride_o_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward of the TriMul “outgoing” module with Triton acceleration.

    Args:
        data: tuple (input, mask, weights, config)
            - input:   torch.Tensor [B, N, N, dim]
            - mask:    torch.Tensor [B, N, N] (may be None)
            - weights: dict of torch.Tensors with the model parameters
            - config:  dict containing at least “dim”, “hidden_dim” and
                       optional “nomask” flag

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted to hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the Triton kernel
    #    Convert to FP16 and bring the hidden dimension to the second axis.
    # ------------------------------------------------------------------
    left_h = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # [B, H, N, N]
    right_h = right.permute(0, 3, 1, 2).contiguous().to(torch.float16) # [B, H, N, N]

    B, H, N, _ = left_h.shape
    left_flat = left_h.view(-1, N, N)   # [B*H, N, N]
    right_flat = right_h.view(-1, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)  # result buffer

    # Strides in element units (not bytes)
    stride_l_batch, stride_l_row, stride_l_col = left_flat.stride()
    stride_r_batch, stride_r_row, stride_r_col = right_flat.stride()
    stride_o_batch, stride_o_row, stride_o_col = out_flat.stride()

    # ------------------------------------------------------------------
    # 6. Triton batched matmul: out = left @ right.T
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                 # batch * hidden_dim
        (N + BLOCK_M - 1) // BLOCK_M,          # rows
        (N + BLOCK_N - 1) // BLOCK_N,          # cols
    )

    triton_trimul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_l_batch,
        stride_l_row,
        stride_l_col,
        stride_r_batch,
        stride_r_row,
        stride_r_col,
        stride_o_batch,
        stride_o_row,
        stride_o_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden_dim] and cast to FP32
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, H, N, N)          # [B, H, N, N]
        .permute(0, 2, 3, 1)               # [B, N, N, H]
        .contiguous()
        .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # 8. Final LayerNorm, gating, and projection back to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
