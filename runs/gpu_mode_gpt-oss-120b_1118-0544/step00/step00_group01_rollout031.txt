    a_ptr, b_ptr, c_ptr,
    batch, N,

    stride_bh, stride_br, stride_bc,
    stride_ch, stride_ci, stride_cj,

    pid_m = tl.program_id(axis=1) # row tile index
    pid_n = tl.program_id(axis=2) # col tile index

    # Compute tile offsets for rows/cols
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

"""
TriMul outgoing implementation.

The kernel computes the core contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix multiplication
    out[b,d] = left[b,d] @ right[b,d].T
for each hidden channel d (depth = hidden_dim) and batch element.
The heavy O(N³·hidden_dim) work is performed by a custom Triton GEMM
kernel that processes the flattened batch dimension (B·hidden_dim) as
independent GEMM tasks.  Inputs are cast to FP16 for higher throughput,
while accumulation and the final output are FP32.  All remaining
operations (LayerNorm, linear projections, gating, final projection) are
executed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    batch, N,                           # batch*hidden, sequence length
    stride_a0, stride_ai, stride_ak,    # A strides: (batch, i, k)
    stride_b0, stride_br, stride_bc,    # B strides: (batch, j, k)
    stride_c0, stride_ci, stride_cj,    # C strides: (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)   # which (batch * hidden) slice
    pid_m = tl.program_id(axis=1) # i‑tile
    pid_n = tl.program_id(axis=2) # j‑tile

    # offsets within the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile:   shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + pid * stride_a0
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B tile (transposed on‑the‑fly):
        # we need B[j,k] but the GEMM expects (k,j), so we swap strides.
        b_ptrs = (
            b_ptr
            + pid * stride_b0
            + offs_k[:, None] * stride_bc   # k‑dimension (now rows)
            + offs_n[None, :] * stride_br   # j‑dimension (now columns)
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot: (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)   # a,b are FP16, acc accumulates in FP32

    # write back the result tile
    c_ptrs = (
        c_ptr
        + pid * stride_c0
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be None)
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----- LayerNorm on the last dimension -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim]

    # ----- Linear projections ----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ----- Mask (if present) -----------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ----- Gating ---------------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ----- Prepare tensors for the batched GEMM ---------------------------------------------
    # Move hidden dimension to second axis to obtain shape (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Cast to FP16 for the kernel (memory‑bandwidth friendly)
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    B, H, N = left_fp16.shape  # B=batch, H=hidden_dim, N=seq_len

    # Flatten batch and hidden into a single GEMM batch dimension
    left_flat = left_fp16.view(-1, N, N)   # [BH, N, N]
    right_flat = right_fp16.view(-1, N, N)

    # Output buffer (FP32 accumulator)
    out_flat = torch.empty(
        (left_flat.shape[0], N, N), dtype=torch.float32, device=device
    )

    # Strides in element units
    stride_a0, stride_ai, stride_ak = left_flat.stride()
    stride_b0, stride_br, stride_bc = right_flat.stride()
    stride_c0, stride_ci, stride_cj = out_flat.stride()

    # ----- Triton kernel launch ---------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        left_flat.shape[0],                     # batch*hidden
        (N + BLOCK_M - 1) // BLOCK_M,           # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,           # j‑tiles
    )
    _tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        left_flat.shape[0],
        N,
        stride_a0,
        stride_ai,
        stride_ak,
        stride_b0,
        stride_br,
        stride_bc,
        stride_c0,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----- Output LayerNorm and final projection -----------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return output
