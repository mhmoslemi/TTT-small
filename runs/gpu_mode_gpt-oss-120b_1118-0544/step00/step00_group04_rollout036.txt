"""
TriMul (outgoing) implementation with a Triton kernel for mask application.

Algorithm
---------
1. Layer‑normalize the input tensor.
2. Project to hidden dimension (left/right) and compute three gating tensors.
3. Apply gating and (optional) mask – mask multiplication is performed by a
   small Triton kernel that processes the 4‑D tensor in blocks.
4. Compute the core O(N³) reduction:
        out[i, j, d] = Σₖ left[i, k, d] * right[j, k, d]
   This is a batched matrix‑multiply for each hidden channel:
        left  → (B, H, i, k)
        right → (B, H, j, k) -> transpose to (B, H, k, j)
        out   = torch.matmul(left, right_T) → (B, H, i, j)
        out   → (B, i, j, H)
5. Multiply by the output gate, apply a second LayerNorm, and project back
   to the original dimension.
All heavy linear ops stay on the GPU’s high‑performance BLAS; the mask
application demonstrates the use of a custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: multiply a [B, N, N, H] tensor by a [B, N, N] mask (in‑place)
# --------------------------------------------------------------------------- #
@triton.jit
def mask_tensor_kernel(
    t_ptr,                # pointer to the tensor to be masked
    mask_ptr,             # pointer to the mask
    B, N, H,              # dimensions
    stride_t_b, stride_t_i, stride_t_j, stride_t_h,   # strides of t
    stride_m_b, stride_m_i, stride_m_j,               # strides of mask
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # i‑tile index
    pid_j = tl.program_id(2)          # j‑tile index

    # absolute coordinates of the tile
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # ---------- load a scalar mask for each (i,j) ----------
    mask_off = pid_b * stride_m_b + i[:, None] * stride_m_i + j[None, :] * stride_m_j
    mask_val = tl.load(mask_ptr + mask_off,
                       mask=(i[:, None] < N) & (j[None, :] < N),
                       other=0.0)                     # (BLOCK_I, BLOCK_J)
    mask_val = mask_val[:, :, None]                     # broadcast to H dim

    # ---------- loop over the hidden dimension ----------
    cur_h = 0
    while cur_h < H:
        h = cur_h + tl.arange(0, BLOCK_H)
        valid_h = h < H

        # offsets for the tensor
        t_off = (pid_b * stride_t_b
                 + i[:, None, None] * stride_t_i
                 + j[None, :, None] * stride_t_j
                 + h[None, None, :] * stride_t_h)

        # mask for out‑of‑bounds (i,j,h)
        m = (i[:, None, None] < N) & (j[None, :, None] < N) & valid_h[None, None, :]

        # load, apply mask, store back
        val = tl.load(t_ptr + t_off, mask=m, other=0.0)
        val = val * mask_val
        tl.store(t_ptr + t_off, val, mask=m)

        cur_h += BLOCK_H


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.
    `data` is a tuple: (input_tensor, mask, weights, config).
    Returns the processed tensor of shape [B, N, N, dim].
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------- #
    # Unpack configuration
    # ------------------------------------------------------------------- #
    dim = config["dim"]               # input / output channel dimension
    hidden = config["hidden_dim"]     # hidden dimension (H)
    B, N, _, _ = input_tensor.shape   # [B, N, N, dim]

    # ------------------------------------------------------------------- #
    # Convenience pointers to the weight tensors (convert to fp16 for speed)
    # ------------------------------------------------------------------- #
    def to_fp16(t):
        return t.to(torch.float16)

    norm_weight = to_fp16(weights["norm.weight"])
    norm_bias   = to_fp16(weights["norm.bias"])
    left_proj_weight  = to_fp16(weights["left_proj.weight"])
    right_proj_weight = to_fp16(weights["right_proj.weight"])
    left_gate_weight  = to_fp16(weights["left_gate.weight"])
    right_gate_weight = to_fp16(weights["right_gate.weight"])
    out_gate_weight   = to_fp16(weights["out_gate.weight"])
    to_out_norm_weight = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_bias   = to_fp16(weights["to_out_norm.bias"])
    to_out_weight      = to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------- #
    # 1) Input LayerNorm (stay in fp32 for numerical stability, then cast)
    # ------------------------------------------------------------------- #
    x_norm = F.layer_norm(input_tensor,
                          normalized_shape=(dim,),
                          weight=norm_weight,
                          bias=norm_bias,
                          eps=1e-5)
    x = x_norm.to(torch.float16)  # fp16 compute from now on

    # ------------------------------------------------------------------- #
    # 2) Linear projections + gating
    # ------------------------------------------------------------------- #
    left_proj = F.linear(x, left_proj_weight)           # [B, N, N, H]
    right_proj = F.linear(x, right_proj_weight)

    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------- #
    # 3) Optional mask – apply via Triton kernel (in‑place)
    # ------------------------------------------------------------------- #
    if mask is not None and mask.numel() > 0:
        # broadcast mask to hidden dim (float16)
        mask_fp = mask.to(torch.float16)

        # choose block sizes (tuned for typical B200 resources)
        BLOCK_I = 32
        BLOCK_J = 32
        BLOCK_H = 32

        grid = (B,
                (N + BLOCK_I - 1) // BLOCK_I,
                (N + BLOCK_J - 1) // BLOCK_J)

        # left
        mask_tensor_kernel[grid](
            left, mask_fp,
            B, N, hidden,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            mask_fp.stride(0), mask_fp.stride(1), mask_fp.stride(2),
            BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_H=BLOCK_H)

        # right
        mask_tensor_kernel[grid](
            right, mask_fp,
            B, N, hidden,
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            mask_fp.stride(0), mask_fp.stride(1), mask_fp.stride(2),
            BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_H=BLOCK_H)

    # ------------------------------------------------------------------- #
    # 4) Core reduction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched matrix multiply per hidden channel.
    # ------------------------------------------------------------------- #
    # Rearrange tensors to (B, H, i, k) and (B, H, j, k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, H, i, k]
    right_swapped = right.permute(0, 3, 2, 1).contiguous()    # [B, H, j, k]
    # Transpose the last two dimensions of right to get (k, j)
    right_T = right_swapped.permute(0, 1, 3, 2).contiguous()  # [B, H, k, j]

    # Batched matmul over the hidden channel dimension
    out = torch.matmul(left_perm, right_T)                     # [B, H, i, j]

    # Restore original layout [B, i, j, H]
    out = out.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 5) Apply output gate, second LayerNorm and final projection
    # ------------------------------------------------------------------- #
    out = out * out_gate

    out = F.layer_norm(out,
                       normalized_shape=(hidden,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = F.linear(out, to_out_weight)   # shape [B, N, N, dim]

    # Cast back to FP32 as required by the interface
    return out.to(torch.float32)
```
