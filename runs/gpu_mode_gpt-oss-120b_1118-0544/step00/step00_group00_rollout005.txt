@triton.jit
def triton_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k, stride_h,
    stride_b_r, stride_i_r, stride_k_r, stride_h_r,
    stride_b_o, stride_i_o, stride_j_o, stride_h_o,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)  # batch * hidden
    pid_i = tl.program_id(1)   # block i
    pid_j = tl.program_id(2)   # block j

    total_bh = B * H
    b = pid_bh // H
    h = pid_bh % H

    # Compute starting indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # mask for i/j bound
    i_mask = i0 + tl.arange(0, BLOCK_M) < N
    j_mask = j0 + tl.arange(0, BLOCK_N) < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k0 in range(0, N, BLOCK_K):
        # compute k range
        k_max = tl.minimum(k0 + BLOCK_K, N)
        cur_k = k_max - k0  # actual size

        # Offsets for left: (i, k)
        i_idx = i0 + tl.arange(0, BLOCK_M)
        k_idx = k0 + tl.arange(0, BLOCK_K)

        left_ptrs = left_ptr + b*stride_b + i_idx[:, None] * stride_i + k_idx[None, :] * stride_k + h * stride_h
        # mask for left load
        k_mask = k_idx < N
        left_mask = i_mask[:, None] & k_mask[None, :]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # Offsets for right transposed: (k, j)
        # right[j, k] -> we need shape (K, N) after transpose
        j_idx = j0 + tl.arange(0, BLOCK_N)
        right_ptrs = right_ptr + b*stride_b_r + j_idx[None, :] * stride_i_r + k_idx[:, None] * stride_k_r + h * stride_h_r
        # mask for right load
        right_mask = k_mask[:, None] & j_mask[None, :]
        right = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(left, right)   # left (M,K), right (K,N)
    # store results
    out_ptrs = out_ptr + b*stride_b_o + i0 + tl.arange(0, BLOCK_M)[:, None] * stride_i_o + (j0 + tl.arange(0, BLOCK_N))[None, :] * stride_j_o + h * stride_h_o
    out_mask = i_mask[:, None] & j_mask[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # dtype
    dtype = input_tensor.dtype
    device = input_tensor.device

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear projections (weights are [hidden_dim, dim])
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask if needed
    if not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gating
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for einsum
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Compute output via Triton kernel
    out_fp16 = triton_einsum(left_fp16, right_fp16)  # shape [B,N,N,hidden_dim], dtype float16

    # Convert back to float32 for subsequent layernorm and linear
    out = out_fp16.to(torch.float32)

    # Output layernorm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # Apply out_gate
    out = out * out_gate

    # Final linear projection
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

def tri_mul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k, stride_h,
    stride_b_r, stride_i_r, stride_k_r, stride_h_r,
    stride_b_o, stride_i_o, stride_j_o, stride_h_o,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    total_bh = B * H
    b = pid // H
    h = pid % H

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    i_range = tl.arange(0, BLOCK_M)
    j_range = tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    i_mask = i0 + i_range < N
    j_mask = j0 + j_range < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k0 in range(0, N, BLOCK_K):
        cur_k = tl.minimum(k0 + BLOCK_K, N)
        k_valid = k_range < (cur_k - k0)  # but easier: k_idx = k0 + k_range; k_mask = k_idx < N

        k_mask = k_idx < N

        left_ptrs = left_ptr + b*stride_b + i_idx[:, None] * stride_i + k_idx[None, :] * stride_k + h * stride_h

"""
TriMul outgoing update (forward only).

The kernel performs the core 4‑D Einstein summation:
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]

We fuse the reduction using a tiled GEMM‑style Triton kernel.
All other inexpensive ops (LayerNorm, linear projections, sigmoids,
masking and final projection) are executed with PyTorch.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the einsum "left[i,k] * right[j,k] -> out[i,j]"
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # left strides
    stride_bl, stride_il, stride_kl, stride_hl,
    # right strides
    stride_br, stride_ir, stride_kr, stride_hr,
    # out strides
    stride_bo, stride_io, stride_jo, stride_ho,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)      # combined batch * hidden
    pid_i   = tl.program_id(1)      # tile row  (i)
    pid_j   = tl.program_id(2)      # tile col  (j)

    total_bh = B * H
    b = pid_bh // H                # batch index
    h = pid_bh %  H                # hidden index

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # range vectors
    i_range = tl.arange(0, BLOCK_M)
    j_range = tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    i_valid = i0 + i_range < N
    j_valid = j0 + j_range < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduce over k
    # ------------------------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        cur_k = tl.minimum(k0 + BLOCK_K, N)
        k_len = cur_k - k0                     # actual k‑tile length
        k_idx = k0 + k_range
        k_valid = k_idx < N

        # ------- left tile (i,k) -------
        i_idx = i0 + i_range
        left_ptrs = (
            left_ptr
            + b * stride_bl
            + i_idx[:, None] * stride_il
            + k_idx[None, :] * stride_kl
            + h * stride_hl
        )
        left_mask = i_valid[:, None] & k_valid[None, :]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # ------- right tile transposed (k,j) -------
        j_idx = j0 + j_range
        right_ptrs = (
            right_ptr
            + b * stride_br
            + j_idx[None, :] * stride_ir   # j dimension → stride of first N
            + k_idx[:, None] * stride_kr   # k dimension → stride of second N
            + h * stride_hr
        )
        right_mask = k_valid[:, None] & j_valid[None, :]
        right = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # dot: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    # write back
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_bo
        + (i0 + i_range)[:, None] * stride_io
        + (j0 + j_range)[None, :] * stride_jo
        + h * stride_ho
    )
    out_mask = i_valid[:, None] & j_valid[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


def tri_mul_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Execute the Triton kernel.  Input tensors must be on the same CUDA device,
    have shape [B, N, N, H] and be either fp16 or fp32 (they will be cast to fp16).
    Returns a tensor of the same shape and dtype as the inputs.
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape
    # cast to fp16 for the kernel (GPU Tensor Cores)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out = torch.empty_like(left_fp16)

    # stride values in element units (not bytes)
    s_bl, s_il, s_kl, s_hl = left_fp16.stride()
    s_br, s_ir, s_kr, s_hr = right_fp16.stride()
    s_bo, s_io, s_jo, s_ho = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                                 # batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
    )

    tri_mul_einsum_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        H,
        s_bl,
        s_il,
        s_kl,
        s_hl,
        s_br,
        s_ir,
        s_kr,
        s_hr,
        s_bo,
        s_io,
        s_jo,
        s_ho,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,   # enough warps for the tile size
    )
    return out


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    data = (input_tensor, mask, weights, config)
    """
    torch.cuda.empty_cache()  # ensure a clean launch environment
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core einsum via Triton (out shape [B,N,N,hidden_dim])
    # ------------------------------------------------------------------
    out = tri_mul_einsum(left, right)          # still fp32 (cast back later)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
