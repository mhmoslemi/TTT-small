def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # LayerNorm input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # mask
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp
    # gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
    left = left * left_gate
    right = right * right_gate
    # Cast to half for kernel
    left = left.half()
    right = right.half()
    # allocate output buffer
    out = torch.empty_like(left, dtype=torch.float32, device=device) # shape (B,N,N,hidden_dim)
    # call triton kernel
    # define block sizes
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_D = 32
    # get strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_k, stride_o_h = out.stride()
    B, N, N2, D = left.shape
    assert N == N2
    grid = ( ( (N + BLOCK_I - 1)//BLOCK_I ), ( (N + BLOCK_J -1)//BLOCK_J ), B )
    tri_mul_outgoing_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_k, stride_o_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_D=BLOCK_D,
    )
    # Now out is shape (B,N,N,hidden_dim) float32
    # Apply to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    # Apply out_gate gating
    out = out * out_gate
    # Final linear projection to dim
    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out, to_out_weight)

    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_k, stride_o_h,

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # compute offsets
    i_offsets = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offsets = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    i_mask = i_offsets < N
    j_mask = j_offsets < N

    d_start = 0

        d_mask = d_offsets < H

"""
TriMul outgoing update (AlphaFold3) forward pass with Triton acceleration.

The kernel computes, for each batch b, hidden slice h and pair (i, j):
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

All LayerNorms, linear projections and gates are performed with PyTorch.
The heavy O(N³·H) reduction over the sequence dimension k is fused in a
Triton kernel that tiles the i‑ and j‑axes and processes a small hidden
tile (BLOCK_D) at a time. The kernel works for arbitrary batch size, sequence
length and hidden dimension (within the limits of the block sizes) and
produces a float32 output compatible with the reference implementation.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,                # const float16 *
    right_ptr,               # const float16 *
    out_ptr,                 # float32 *
    B, N, H,                 # int32
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # int32 stride of left
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,   # int32 stride of right
    stride_o_b, stride_o_i, stride_o_k, stride_o_h,   # int32 stride of out
    d_start,                 # int32 start index of hidden tile
    BLOCK_I: tl.constexpr,   # tile size along i dimension
    BLOCK_J: tl.constexpr,   # tile size along j dimension
    BLOCK_D: tl.constexpr,   # tile size along hidden dimension
):
    """Triton kernel for the outgoing triangular multiplicative update."""
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # i/j offsets for this tile
    i_off = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_mask = i_off < N
    j_mask = j_off < N

    # hidden offsets for the current tile
    d_off = d_start + tl.arange(0, BLOCK_D)
    d_mask = d_off < H

    # accumulator (float32) for a (BLOCK_I, BLOCK_J, BLOCK_D) tile
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # loop over the sequence dimension k
    k = tl.zeros([1], dtype=tl.int32)  # scalar counter
    while k[0] < N:
        k_idx = k[0]

        # pointer to left[b, i_off, k_idx, d_off]
        left_ptrs = (
            left_ptr
            + pid_b * stride_l_b
            + i_off[:, None] * stride_l_i
            + k_idx * stride_l_k
            + d_off[None, :] * stride_l_h
        )
        # pointer to right[b, j_off, k_idx, d_off]
        right_ptrs = (
            right_ptr
            + pid_b * stride_r_b
            + j_off[:, None] * stride_r_i
            + k_idx * stride_r_k
            + d_off[None, :] * stride_r_h
        )

        # load vectors (float16 -> cast to float32)
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None] & d_mask[None, :],
            other=0.0,
            dtype=tl.float16,
        ).to(tl.float32)
        right = tl.load(
            right_ptrs,
            mask=j_mask[:, None] & d_mask[None, :],
            other=0.0,
            dtype=tl.float16,
        ).to(tl.float32)

        # outer product across i and j for each hidden channel
        acc += left[:, None, :] * right[None, :, :]

        k += 1

    # write the result tile
    out_ptrs = (
        out_ptr
        + pid_b * stride_o_b
        + i_off[:, None, None] * stride_o_i
        + j_off[None, :, None] * stride_o_k
        + d_off[None, None, :] * stride_o_h
    )
    store_mask = i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :]
    tl.store(out_ptrs, acc, mask=store_mask)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N]  (may be all‑ones)
        - weights      : dict of parameter tensors
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional masking (mask shape [B,N,N])
    # -----------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).type_as(left)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # 4) Gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Triton kernel for the O(N³·H) reduction
    # -----------------------------------------------------------------
    # Cast to half for the kernel (saves memory bandwidth)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape
    H = hidden_dim

    # Allocate output buffer for the hidden representation
    out_hidden = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Strides (in elements, not bytes)
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_i, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_k, s_o_h = out_hidden.stride()

    # Tile sizes (tuned for typical B200 GPU)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_D = 8   # hidden‑channel tile; 8*32*32 ≈ 8 192 float registers

    grid_i = (N + BLOCK_I - 1) // BLOCK_I
    grid_j = (N + BLOCK_J - 1) // BLOCK_J
    grid = (grid_i, grid_j, B)   # (i‑tile, j‑tile, batch)

    # Loop over hidden tiles on the host side
    for d_start in range(0, H, BLOCK_D):
        tri_mul_outgoing_kernel[
            grid
        ](
            left.data_ptr(),
            right.data_ptr(),
            out_hidden.data_ptr(),
            B,
            N,
            H,
            s_l_b,
            s_l_i,
            s_l_k,
            s_l_h,
            s_r_b,
            s_r_i,
            s_r_k,
            s_r_h,
            s_o_b,
            s_o_i,
            s_o_k,
            s_o_h,
            d_start,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_D=BLOCK_D,
            num_warps=4,  # modest warp count; Triton chooses a default if omitted
        )

    # -----------------------------------------------------------------
    # 6) Output LayerNorm, gating, and final projection
    # -----------------------------------------------------------------
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate
    output = torch.nn.functional.linear(out_hidden, weights["to_out.weight"])

    return output
