"""
TriMul (outgoing) forward implementation with a Triton kernel for the
core O(N³) contraction:

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

The contraction is realized as a batched matrix multiplication
for each hidden channel d (hidden_dim).  The Triton kernel computes
many (i, j) tiles in parallel and accumulates over k.  All other
operations (LayerNorm, linear projections, gating) are performed in
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_gemm_kernel(
    L_ptr, R_ptr, O_ptr,          # pointers to left, right_T, output
    B, H, N,                      # batch, hidden, seq_len
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_k, stride_r_j,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: for each (b, h) compute L[b,h] @ R_T[b,h] (N×N)."""
    pid = tl.program_id(axis=0)

    # -------------------------------------------------------------
    # Decode the 1‑D program id into (b, h, block_m, block_n)
    # -------------------------------------------------------------
    num_m_blocks = tl.cdiv(N, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)
    tiles_per_bh = num_m_blocks * num_n_blocks
    bh_idx = pid // tiles_per_bh                # which (b, h) pair
    tile_idx = pid % tiles_per_bh

    block_m = tile_idx // num_n_blocks          # i‑tile
    block_n = tile_idx % num_n_blocks           # j‑tile

    b = bh_idx // H
    h = bh_idx % H

    # -------------------------------------------------------------
    # Compute the global i / j coordinates for the tile
    # -------------------------------------------------------------
    i = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # -----------------------------------------------------------------
    # Pointers to the start of the (i) rows of L and (j) columns of R_T
    # -----------------------------------------------------------------
    L_base = L_ptr + b * stride_l_b + h * stride_l_h + i[:, None] * stride_l_i
    R_base = R_ptr + b * stride_r_b + h * stride_r_h + j[None, :] * stride_r_j

    # -----------------------------------------------------------------
    # Accumulator for the tile (BLOCK_M x BLOCK_N)
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Loop over the reduction dimension k
    # -------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_blocks):
        cur_k = kk * BLOCK_K
        k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Load a (BLOCK_M x BLOCK_K) sub‑matrix from L
        L = tl.load(
            L_base + k[None, :] * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a (BLOCK_K x BLOCK_N) sub‑matrix from R_T
        R = tl.load(
            R_base + k[:, None] * stride_r_k,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # Rank‑K update
        acc += tl.dot(L, R)

    # -------------------------------------------------------------
    # Store the result
    # -------------------------------------------------------------
    O_ptrs = O_ptr + b * stride_o_b + h * stride_o_h + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(O_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def _launch_tri_gemm(left: torch.Tensor, right_t: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, N, N)  – rows indexed by i, columns by k
    right_t: (B, H, N, N) – rows indexed by k, columns by j (already transposed)
    Returns out with shape (B, H, N, N)
    """
    assert left.dtype == right_t.dtype == torch.float32
    B, H, N, _ = left.shape

    # Output tensor
    out = torch.empty_like(left)

    # Strides (in elements)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k = left.stride()
    stride_r_b, stride_r_h, stride_r_k, stride_r_j = right_t.stride()
    stride_o_b, stride_o_h, stride_o_i, stride_o_j = out.stride()

    # Choose tile sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Number of program instances
    num_m_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_n_blocks = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = B * H * num_m_blocks * num_n_blocks
    grid = (total_tiles,)

    # Launch kernel
    tri_gemm_kernel[grid](
        left,
        right_t,
        out,
        B,
        H,
        N,
        stride_l_b,
        stride_l_h,
        stride_l_i,
        stride_l_k,
        stride_r_b,
        stride_r_h,
        stride_r_k,
        stride_r_j,
        stride_o_b,
        stride_o_h,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of AlphaFold‑3's outgoing TriMul.
    Arguments:
        data = (input_tensor, mask, weights, config)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on the same device).
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # (B, N, N, hidden)
    right = F.linear(x, right_proj_weight)  # (B, N, N, hidden)

    # -----------------------------------------------------------------
    # 3. Optional mask application
    # -----------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid of linear transforms of the normalized input)
    # -----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # (B,N,N,hidden)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # (B,N,N,hidden)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # (B,N,N,hidden)

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Prepare tensors for the batched GEMM: (B, H, N, N)
    # -----------------------------------------------------------------
    # left: (B, N, N, H) -> (B, H, N, N) with rows=i, cols=k
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: (B, N, N, H) -> transpose second/third -> (B, H, N, N) rows=k, cols=j
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Core contraction via Triton GEMM kernel
    # -----------------------------------------------------------------
    out_hidden = _launch_tri_gemm(left_perm, right_perm)   # (B, H, N, N)

    # Restore original layout (B, N, N, H)
    out_hidden = out_hidden.permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7. Post‑processing: LayerNorm, gating and final projection
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]   # (dim, hidden)

    out_hidden_norm = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # Apply the output gate
    out_gated = out_hidden_norm * out_gate

    # Final linear projection back to dim
    out = F.linear(out_gated, to_out_weight)  # (B, N, N, dim)

    return out
