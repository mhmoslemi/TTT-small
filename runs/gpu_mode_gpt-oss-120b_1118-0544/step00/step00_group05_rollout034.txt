@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_bi, stride_bk, stride_bh,
    stride_rl, stride_ri, stride_rk, stride_rh,
    stride_ol, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    pid_h = tl.program_id(1)

    # compute block indices in i, j
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_i * num_j

    batch = pid // blocks_per_batch
    block_id = pid % blocks_per_batch
    i_block = block_id // num_j
    j_block = block_id % num_j

    i_offset = i_block * BLOCK_M
    j_offset = j_block * BLOCK_N
    h_offset = pid_h * BLOCK_H

    # ranges
    i_range = i_offset + tl.arange(0, BLOCK_M)
    j_range = j_offset + tl.arange(0, BLOCK_N)
    h_range = h_offset + tl.arange(0, BLOCK_H)

    i_mask = i_range < N
    j_mask = j_range < N
    h_mask = h_range < H

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # Loop over reduction dimension K
    for K_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = K_iter * BLOCK_K
        k_range = k_offset + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # Load left block: shape (M, K, H)
        left_ptrs = left_ptr + \
            batch * stride_bl + \
            i_range[:, None, None] * stride_bi + \
            k_range[None, :, None] * stride_bk + \
            h_range[None, None, :] * stride_bh
        left = tl.load(left_ptrs,
            mask=i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :],
            other=0.0).to(tl.float32)

        # Load right block transposed: shape (K, N, H)
        right_ptrs = right_ptr + \
            batch * stride_rl + \
            k_range[:, None, None] * stride_rk + \
            j_range[None, :, None] * stride_ri + \
            h_range[None, None, :] * stride_rh
        right = tl.load(right_ptrs,
            mask=k_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
            other=0.0).to(tl.float32)

        # Compute dot per hidden channel
        for h in range(0, BLOCK_H):
            # left_slice: (M, K)
            left_slice = left[:, :, h]
            # right_slice: (K, N) (since right shape (K,N,H))
            right_slice = right[:, :, h]
            # Dot: (M,K) @ (K,N) -> (M,N)
            acc[:, :, h] += tl.dot(left_slice, right_slice)

    # Store results
    out_ptrs = out_ptr + \
        batch * stride_ol + \
        i_range[:, None, None] * stride_oi + \
        j_range[None, :, None] * stride_oj + \
        h_range[None, None, :] * stride_oh

    tl.store(out_ptrs,
        acc.to(tl.float16),
        mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :])

"""TriMul outgoing kernel implementation.

This module implements the forward pass of the AlphaFold3 "outgoing" TriMul
operator. The heavy O(N³) contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is performed with a custom Triton kernel that tiles the three dimensions
(i, j and the reduction k) and also tiles the hidden dimension `d`.  The
kernel accumulates in float32 and stores the result in float16; the rest of
the pipeline (layer‑norms, linear projections, gating) is done with PyTorch
operations for simplicity.

The kernel is launched with a 2‑D grid:
  axis‑0 : (batch, i‑tile, j‑tile) combined,
  axis‑1 : hidden‑dim tile.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_bi, stride_bk, stride_bh,
    stride_rl, stride_ri, stride_rk, stride_rh,
    stride_ol, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    """Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].

    Tiling strategy:
      - Tile i dimension (BLOCK_M) and j dimension (BLOCK_N).
      - Tile the reduction k dimension (BLOCK_K).
      - Tile the hidden channel h dimension (BLOCK_H) across the second
        program‑id axis.
    Accumulation is performed in float32 to preserve precision.
    """
    pid = tl.program_id(0)
    pid_h = tl.program_id(1)

    # number of tiles
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_i * num_j

    batch = pid // tiles_per_batch
    tile_id = pid % tiles_per_batch
    i_tile = tile_id // num_j
    j_tile = tile_id % num_j

    i0 = i_tile * BLOCK_M
    j0 = j_tile * BLOCK_N
    h0 = pid_h * BLOCK_H

    # ranges
    i_range = i0 + tl.arange(0, BLOCK_M)
    j_range = j0 + tl.arange(0, BLOCK_N)
    h_range = h0 + tl.arange(0, BLOCK_H)

    i_mask = i_range < N
    j_mask = j_range < N
    h_mask = h_range < H

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # iterate over k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_iter * BLOCK_K
        k_range = k0 + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # ----- load left tile (M, K, H) -----
        left_ptrs = left_ptr + \
            batch * stride_bl + \
            i_range[:, None, None] * stride_bi + \
            k_range[None, :, None] * stride_bk + \
            h_range[None, None, :] * stride_bh
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :],
            other=0.0
        ).to(tl.float32)   # (M, K, H)

        # ----- load right tile transposed (K, N, H) -----
        # we load with k as leading dimension so that we can use tl.dot
        right_ptrs = right_ptr + \
            batch * stride_rl + \
            k_range[:, None, None] * stride_rk + \
            j_range[None, :, None] * stride_ri + \
            h_range[None, None, :] * stride_rh
        right = tl.load(
            right_ptrs,
            mask=k_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
            other=0.0
        ).to(tl.float32)   # (K, N, H)

        # ----- compute matmul for each hidden slice -----
        for h in range(0, BLOCK_H):
            # left slice: (M, K), right slice: (K, N)
            lhs = left[:, :, h]        # (M, K)
            rhs = right[:, :, h]       # (K, N)
            acc[:, :, h] += tl.dot(lhs, rhs)   # (M, N)

    # ----- store result -----
    out_ptrs = out_ptr + \
        batch * stride_ol + \
        i_range[:, None, None] * stride_oi + \
        j_range[None, :, None] * stride_oj + \
        h_range[None, None, :] * stride_oh

    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :]
    )

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (Tuple):
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with all required weight tensors
            - config: dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # ---------------- LayerNorm over channel dim ----------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------------- Linear projections ----------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------------- mask handling ----------------
    if mask is None or config.get("nomask", False):
        mask_tensor = torch.ones_like(input_tensor[..., 0], dtype=x.dtype, device=device)
    else:
        mask_tensor = mask.to(dtype=x.dtype, device=device)
    mask_tensor = mask_tensor.unsqueeze(-1)   # shape [B,N,N,1]

    left = left * mask_tensor
    right = right * mask_tensor

    # ---------------- Gating ----------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------------- Triton contraction ----------------
    # Cast to half for the kernel (accumulation in fp32 inside the kernel)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)
    out_h = torch.empty_like(left_h)  # shape [B,N,N,hidden_dim], dtype half

    B, N, _, _ = left_h.shape

    # strides (in elements)
    stride_bl = left_h.stride(0)
    stride_bi = left_h.stride(1)
    stride_bk = left_h.stride(2)
    stride_bh = left_h.stride(3)

    stride_rl = right_h.stride(0)
    stride_ri = right_h.stride(1)
    stride_rk = right_h.stride(2)
    stride_rh = right_h.stride(3)

    stride_ol = out_h.stride(0)
    stride_oi = out_h.stride(1)
    stride_oj = out_h.stride(2)
    stride_oh = out_h.stride(3)

    # Tunable tile sizes
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    # Hidden dimension is tiled; choose a block that divides nicely up to 128
    BLOCK_H = 32 if hidden_dim >= 32 else hidden_dim

    grid = (
        B * ((N + BLOCK_M - 1) // BLOCK_M) * ((N + BLOCK_N - 1) // BLOCK_N),
        (hidden_dim + BLOCK_H - 1) // BLOCK_H,
    )

    # launch kernel
    trimul_outgoing_kernel[grid](
        left_h,
        right_h,
        out_h,
        B, N, hidden_dim,
        stride_bl, stride_bi, stride_bk, stride_bh,
        stride_rl, stride_ri, stride_rk, stride_rh,
        stride_ol, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # back to fp32 for the rest of the pipeline
    out = out_h.to(torch.float32)

    # ---------------- Output LayerNorm, gating and final projection ----------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate  # shape [B,N,N,hidden_dim]
    out = F.linear(out, weights["to_out.weight"])

    return out
