def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm for input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Project
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask if present
    if not nomask and mask is not None:
        mask_f = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Optionally cast to float16 for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to bring hidden_dim to second dim for batch matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    right_t = right_perm.transpose(-2, -1).contiguous()

    B, H, N, _ = left_perm.shape

    M = B * H
    # reshape for kernel
    left_mat = left_perm.view(M, N, N)
    right_mat = right_t.view(M, N, N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float32)  # accumulate in fp32

    # strides
    stride_am = left_mat.stride(0)
    stride_ai = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_bm = right_mat.stride(0)
    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)

    stride_cm = out_mat.stride(0)
    stride_ci = out_mat.stride(1)
    stride_cj = out_mat.stride(2)

    # kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (M, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    tri_mul_kernel[grid](
        left_mat, right_mat, out_mat,
        M, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bn,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # apply out_gate
    out = out * out_gate

    # final linear projection
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
Triton implementation of the outgoing TriMul operation.
Given input tensor x [B,N,N,D] it applies:
- LayerNorm over D
- Linear projections left/right without bias
- Optional mask
- Gated linear transformations
- Core O(N^3) triple multiplication: out_{i,j,d}=sum_k left_{i,k,d} * right_{j,k,d}
  Implemented as a batched matrix multiplication in a Triton kernel
- LayerNorm over hidden dimension
- Output gating and final linear projection back to original dimension.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch = tl.program_id(axis=0)
    pid_i = tl.program_id(axis=1)
    pid_j = tl.program_id(axis=2)

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # predicates for i/j inside bounds
    i_in_bounds = offs_i < N
    j_in_bounds = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        k_in_bounds = cur_k < K

        # pointers
        a_ptrs = a_ptr + batch * stride_am + offs_i[:, None] * stride_ai + cur_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch * stride_bm + cur_k[:, None] * stride_bk + offs_j[None, :] * stride_bn

        # load with masking
        a = tl.load(a_ptrs,
                    mask = i_in_bounds[:, None] & k_in_bounds[None, :],
                    other = 0.0)
        b = tl.load(b_ptrs,
                    mask = k_in_bounds[:, None] & j_in_bounds[None, :],
                    other = 0.0)

        acc += tl.dot(a, b)

    c_ptrs = c_ptr + batch * stride_cm + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    store_mask = i_in_bounds[:, None] & j_in_bounds[None, :]

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[B, N, N, D] (float32)
            - mask: torch.Tensor[B, N, N] (bool/float) or None
            - weights: dict of raw nn.Linear / LayerNorm parameters
            - config: dict containing at least "dim" and "hidden_dim" and optional "nomask"

    Returns:
        torch.Tensor[B, N, N, D] – result of the TriMul forward pass (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------- LayerNorm over the channel dimension --------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # -------- Linear projections (no bias) --------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -------- Optional masking --------
    if (not nomask) and mask is not None:
        mask_f = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_f
        right = right * mask_f

    # -------- Gating --------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------
    # Core O(N^3) multiplication: out_{i,j,d}=sum_k left_{i,k,d} * right_{j,k,d}
    # Implemented as a batched GEMM in Triton.
    # ----------------------------------------------------
    # bring hidden dimension to a separate batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # transpose k and j for the right operand so that matmul matches the einsum
    right_t = right_perm.transpose(-2, -1).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    M = B * H

    # flatten the (B,H) batch dimensions
    a = left_perm.view(M, N, N)
    b = right_t.view(M, N, N)
    c = torch.empty_like(a, dtype=torch.float32)   # result in fp32

    # strides in element units (not bytes)
    stride_am = a.stride(0)
    stride_ai = a.stride(1)
    stride_ak = a.stride(2)

    stride_bm = b.stride(0)
    stride_bk = b.stride(1)
    stride_bn = b.stride(2)

    stride_cm = c.stride(0)
    stride_ci = c.stride(1)
    stride_cj = c.stride(2)

    # Block sizes – chosen to fit typical GPU shared memory limits
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (M,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tri_mul_kernel[grid](
        a, b, c,
        M, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bn,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back to (B, N, N, H)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1)  # (B, N, N, H)

    # -------- Output LayerNorm and gating --------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    out = out * out_gate

    # -------- Final linear projection back to original dim --------
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass with a Triton kernel for the O(N³·H) core.

The implementation follows the reference PyTorch module:
1. LayerNorm over the input channel dimension.
2. Linear projections (left/right) and three gating projections.
3. Optional per‑pair mask applied before gating.
4. Core tensor contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM in Triton (left * rightᵀ for every
   hidden channel).  The (B·H) hidden‑channel batches are flattened to a
   single batch dimension for the kernel.
5. LayerNorm over the hidden dimension, gating, and final linear projection
   back to the original channel dimension.

Only the heavy triple‑multiplication is performed in Triton; the surrounding
operations use regular PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication:
       C[b,i,j] = Σ_k A[b,i,k] * B[b,k,j]
       where A and B are square (N×N) and the batch dimension M = B·H.
    """
    batch = tl.program_id(axis=0)          # flattened batch (B·H)
    pid_i = tl.program_id(axis=1)          # tile row
    pid_j = tl.program_id(axis=2)          # tile column

    # Tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_in_bounds = offs_i < N
    j_in_bounds = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis k
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        k_in_bounds = cur_k < K

        # Pointers with proper strides
        a_ptrs = a_ptr \
            + batch * stride_am \
            + offs_i[:, None] * stride_ai \
            + cur_k[None, :] * stride_ak

        b_ptrs = b_ptr \
            + batch * stride_bm \
            + cur_k[:, None] * stride_bk \
            + offs_j[None, :] * stride_bn

        # Load with masking (out‑of‑bounds entries are zero)
        a = tl.load(a_ptrs,
                    mask = i_in_bounds[:, None] & k_in_bounds[None, :],
                    other = 0.0)
        b = tl.load(b_ptrs,
                    mask = k_in_bounds[:, None] & j_in_bounds[None, :],
                    other = 0.0)

        acc += tl.dot(a, b)      # (BLOCK_M, BLOCK_N)

    # Store the result
    c_ptrs = c_ptr \
        + batch * stride_cm \
        + offs_i[:, None] * stride_ci \
        + offs_j[None, :] * stride_cj

    store_mask = i_in_bounds[:, None] & j_in_bounds[None, :]
    tl.store(c_ptrs, acc, mask=store_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[B, N, N, D] (float32, on GPU)
            - mask: torch.Tensor[B, N, N] (bool/float) or None
            - weights: dict of raw parameters (no nn.Module wrappers)
            - config: dict with keys
                * "dim"         – input channel dimension D
                * "hidden_dim"  – projection dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor[B, N, N, D] – the module output (float32, on GPU)
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------------- LayerNorm over the channel dimension ----------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ---------------- Linear projections (no bias) -----------------------
    left = F.linear(x, weights["left_proj.weight"])      # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])    # (B,N,N,H)

    # ---------------- Optional masking -----------------------------------
    if (not nomask) and mask is not None:
        m = mask.unsqueeze(-1).to(left.dtype)            # (B,N,N,1)
        left = left * m
        right = right * m

    # ---------------- Gating (sigmoid) ----------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------- Core O(N³·H) multiplication --------------------
    # Bring the hidden dimension to a separate batch axis.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)  -> (i,k)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)  -> (j,k)

    # Transpose the last two dimensions of right so that the inner
    # dimension aligns with left's reduction axis.
    right_t = right_perm.transpose(-2, -1).contiguous()  # (B, H, N, N) -> (k,j)

    B, H, N, _ = left_perm.shape
    M = B * H                                         # flattened batch

    # Flatten the (B,H) axes to a single batch dimension
    a = left_perm.view(M, N, N)                       # (M, i, k)
    b = right_t.view(M, N, N)                         # (M, k, j)
    c = torch.empty_like(a, dtype=torch.float32)      # (M, i, j)

    # Strides in element units (not bytes)
    stride_am = a.stride(0)
    stride_ai = a.stride(1)
    stride_ak = a.stride(2)

    stride_bm = b.stride(0)
    stride_bk = b.stride(1)
    stride_bn = b.stride(2)

    stride_cm = c.stride(0)
    stride_ci = c.stride(1)
    stride_cj = c.stride(2)

    # Block sizes – tuned for typical A100/H100 shared‑memory limits
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        M,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_kernel[grid](
        a, b, c,
        M, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bn,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1)      # (B, N, N, H)

    # ---------------- Output LayerNorm, gating and projection ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate                               # gating

    # Final linear projection back to the original channel dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
