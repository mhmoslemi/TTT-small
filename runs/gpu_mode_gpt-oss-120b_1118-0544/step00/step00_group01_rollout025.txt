import triton.language as tl

@triton.jit
def batched_matmul_triton(a_ptr, b_ptr, c_ptr,
                          M, N, K,
                          stride_aba, stride_ama, stride_ak,
                          stride_bba, stride_bbn, stride_bk,
                          stride_cca, stride_ccb,
                          BLOCK_M: tl.constexpr,
                          BLOCK_N: tl.constexpr,
                          BLOCK_K: tl.constexpr):
    pid_task = tl.program_id(0)  # 0..total_tasks-1
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block column

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers + offsets
    a_ptrs = a_ptr + pid_task * stride_aba + offs_m[:, None] * stride_ama + offs_k[None, :] * stride_ak

   pid_i = tl.program_id(1)
   pid_j = tl.program_id(2)

    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)

        right = right * mask_h

"""
TriMul (outgoing) implementation with a Triton kernel for the heavy
contraction.

The forward pass consists of:
1. LayerNorm on the input tensor.
2. Linear projections (left, right) and three gate projections.
3. Optional masking and gating (elementwise).
4. Core contraction:
      out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   which is equivalent to a batched matrix multiplication
   left_h @ right_h^T for each hidden dimension `h`.
   This O(N³) step is performed by a custom Triton kernel that
   processes `batch * hidden_dim` independent GEMMs of size N×N.
5. LayerNorm on the hidden dimension, gating by `out_gate`,
   and a final linear projection back to `dim`.

Only the contraction is implemented on the GPU with Triton; the rest
uses standard PyTorch operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for left @ right^T  (float32 inputs)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_fp32(
    a_ptr, b_ptr, c_ptr,            # pointers
    M, N, K,                        # matrix dimensions (all == seq_len)
    stride_aba, stride_ama, stride_ak,   # a strides: batch, row, col
    stride_bba, stride_bbn, stride_bk,   # b strides: batch, row, col
    stride_cca, stride_ccb,              # c strides: batch, row (col stride = 1)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_task = tl.program_id(0)      # task = batch * hidden_dim
    pid_m = tl.program_id(1)         # block row
    pid_n = tl.program_id(2)         # block col

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k_iter in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k_iter * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for the current K‑block
        a_ptrs = (
            a_ptr
            + pid_task * stride_aba
            + offs_m[:, None] * stride_ama
            + offs_k[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_task * stride_bba
            + offs_n[:, None] * stride_bbn
            + offs_k[None, :] * stride_bk
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        acc += tl.dot(a, b)   # sum over K‑block

    # Store the result
    c_ptrs = (
        c_ptr
        + pid_task * stride_cca
        + offs_m[:, None] * stride_ccb
        + offs_n[None, :]
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Extract configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(device))          # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"].to(device))        # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # Gates (sigmoid of linear projection)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(device)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(device)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(device)).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core contraction – batched GEMM over the sequence dimension
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, N, N, hidden_dim
    # Move hidden_dim to the second axis and make it contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    total_tasks = B * hidden_dim                     # flattened batch * hidden
    a = left_perm.view(total_tasks, N, N)            # [tasks, N, N]
    b = right_perm.view(total_tasks, N, N)           # [tasks, N, N]
    c = torch.empty((total_tasks, N, N), dtype=torch.float32, device=device)

    # Strides (in elements)
    stride_a_batch, stride_a_row, stride_a_col = a.stride()
    stride_b_batch, stride_b_row, stride_b_col = b.stride()
    stride_c_batch, stride_c_row, stride_c_col = c.stride()  # col stride == 1

    # Block sizes (tuned for the B200 architecture)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        total_tasks,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_fp32[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        c_ptr=c.data_ptr(),
        M=N,
        N=N,
        K=N,
        stride_aba=stride_a_batch,
        stride_ama=stride_a_row,
        stride_ak=stride_a_col,
        stride_bba=stride_b_batch,
        stride_bbn=stride_b_row,
        stride_bk=stride_b_col,
        stride_cca=stride_c_batch,
        stride_ccb=stride_c_row,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # Reshape back to [B, N, N, hidden]
    out_hidden = c.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # Final LayerNorm, out‑gate, and linear projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device),
    )
    out_norm = out_norm * out_gate.to(device)    # gate in float32

    out = F.linear(out_norm, weights["to_out.weight"].to(device))   # [B, N, N, dim]

    return out
