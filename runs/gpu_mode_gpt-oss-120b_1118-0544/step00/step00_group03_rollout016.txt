import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def triangular_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    B, N, H,
    stride_Ab, stride_Ai, stride_Ak, stride_Ah,
    stride_Bb, stride_Bk, stride_Bj, stride_Bh,
    stride_Cb, stride_Ci, stride_Cj, stride_Ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # compute b,h
    bh = pid_bh
    b = bh // H
    h = bh % H

    # offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers per batch and hidden dim
    a_base = A_ptr + b * stride_Ab + h * stride_Ah
    b_base = B_ptr + b * stride_Bb + h * stride_Bh
    c_base = C_ptr + b * stride_Cb + h * stride_Ch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A block (i,k)
        a_ptrs = a_base + (offs_i[:, None] * stride_Ai) + (offs_k[None, :] * stride_Ak)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load B block (k,j) - note B has dimensions (k, j)
        b_ptrs = b_base + (offs_k[:, None] * stride_Bk) + (offs_j[None, :] * stride_Bj)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_base + (offs_i[:, None] * stride_Ci) + (offs_j[None, :] * stride_Cj)

```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Possibly config may contain "nomask", default false
    nomask = config.get("nomask", False)
    # ensure contiguous and float32
    x = input_tensor.contiguous()
    # Layer norm
    x = F.layer_norm(x,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=1e-5)
    # projections
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])
    # gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # mask handling
    if not nomask:
        mask_exp = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # gating
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ensure contiguous for kernel
    left = left.contiguous()
    right = right.contiguous()

    # allocate output of matmul
    out = torch.empty((bs, N, N, hidden_dim),
                      dtype=left.dtype,
                      device=left.device)

    # Precompute strides
    stride_Ab = left.stride(0)
    stride_Ai = left.stride(1)
    stride_Ak = left.stride(2)
    stride_Ah = left.stride(3)

    stride_Bb = right.stride(0)
    stride_Bi = right.stride(1)  # this corresponds to k dimension after transpose
    stride_Bj = right.stride(2)
    stride_Bh = right.stride(3)

    stride_Cb = out.stride(0)
    stride_Ci = out.stride(1)
    stride_Cj = out.stride(2)
    stride_Ch = out.stride(3)

    # Block sizes: choose based on N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            bs * hidden_dim)

    triangular_mul_kernel[grid](
        left,
        right,
        out,
        bs,
        N,
        hidden_dim,
        stride_Ab, stride_Ai, stride_Ak, stride_Ah,
        stride_Bb, stride_Bi, stride_Bj, stride_Bh,
        stride_Cb, stride_Ci, stride_Cj, stride_Ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # Post processing: out_norm (LayerNorm over hidden_dim)
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=1e-5)

    out = out * out_gate

    # final linear projection
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul outgoing update (AlphaFold3) – GPU Triton implementation.

This kernel computes the core O(N³·C_hidden) operation:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

for all batches b and hidden channels h.  The computation is a batched
matrix multiplication left @ rightᵀ for each hidden channel.  The surrounding
operations (LayerNorm, linear projections, gates, final projection) are
performed with regular PyTorch calls.  Only the matrix multiplication is
off‑loaded to a custom Triton kernel for speed.

The kernel tiles the output (i, j) plane with BLOCK_M × BLOCK_N tiles and
reduces over k in BLOCK_K sized chunks, using tl.dot for high‑throughput
FMA.  The batch × hidden dimension is fused into a single program‑id axis
to avoid extra kernel launches.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: compute out = left @ rightᵀ per hidden channel
# ----------------------------------------------------------------------
@triton.jit
def triangular_mul_kernel(
    A_ptr,                      # left  [B, N, N, H] (i, k)
    B_ptr,                      # right [B, N, N, H] (j, k)  – interpreted as (k, j)
    C_ptr,                      # out   [B, N, N, H] (i, j)
    B, N, H,                    # runtime sizes
    stride_Ab, stride_Ai, stride_Ak, stride_Ah,
    stride_Bb, stride_Bk, stride_Bj, stride_Bh,
    stride_Cb, stride_Ci, stride_Cj, stride_Ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_i = tl.program_id(0)        # tile row (i)
    pid_j = tl.program_id(1)        # tile col (j)
    pid_bh = tl.program_id(2)       # fused batch * hidden index

    # ------------ decode batch and hidden index ------------
    batch = pid_bh // H
    hidden = pid_bh % H

    # offsets within N
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (batch, hidden) slice
    a_base = A_ptr + batch * stride_Ab + hidden * stride_Ah
    b_base = B_ptr + batch * stride_Bb + hidden * stride_Bh
    c_base = C_ptr + batch * stride_Cb + hidden * stride_Ch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------ k‑loop (reduction) ------------
    for _k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = _k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A block: left[b, i, k, h]
        a_ptrs = a_base + (offs_i[:, None] * stride_Ai) + (offs_k[None, :] * stride_Ak)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load B block: right[b, k, j, h] (notice k dimension first)
        b_ptrs = b_base + (offs_k[:, None] * stride_Bk) + (offs_j[None, :] * stride_Bj)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # inner product
        acc += tl.dot(a, b)

    # ------------ store result ------------
    c_ptrs = c_base + (offs_i[:, None] * stride_Ci) + (offs_j[None, :] * stride_Cj)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] on the same device as the inputs.
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input layer‑norm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2) Linear projections + gates
    # --------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])       # [B,N,N,H]
    right_proj = F.linear(x, weights["right_proj.weight"])     # [B,N,N,H]

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # optional mask (broadcast over hidden dim)
    if not nomask:
        mask_exp = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # apply elementwise gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # make tensors contiguous for the kernel
    left = left.contiguous()
    right = right.contiguous()

    # --------------------------------------------------------------
    # 3) Core TriMul: out = left @ rightᵀ per hidden channel (Triton)
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim),
                      dtype=left.dtype,
                      device=left.device)

    # strides (in elements, not bytes)
    stride_Ab, stride_Ai, stride_Ak, stride_Ah = left.stride()
    stride_Bb, stride_Bi, stride_Bj, stride_Bh = right.stride()
    stride_Cb, stride_Ci, stride_Cj, stride_Ch = out.stride()

    # Configuration of the tiling. These values work well for N up‑to 1024.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,            # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,            # j‑tiles
        B * hidden_dim,                          # fused batch × hidden tiles
    )

    triangular_mul_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        stride_Ab, stride_Ai, stride_Ak, stride_Ah,
        stride_Bb, stride_Bi, stride_Bj, stride_Bh,
        stride_Cb, stride_Ci, stride_Cj, stride_Ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )

    # --------------------------------------------------------------
    # 4) Output LayerNorm, gating, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
