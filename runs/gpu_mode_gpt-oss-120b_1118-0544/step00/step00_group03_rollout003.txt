def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # shape check
    B, N, _, C = input_tensor.shape
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    # Step 1: layernorm input: using weights "norm.weight" and bias.  # given we have elementwise weight and bias; can use torch.nn.functional.layer_norm.
    # Use torch.nn.functional.layer_norm(x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias) but our x is shape [..., dim] where dim is last
    # We'll compute x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # However we will need to ensure shape: for each element in last dimension.

    norm_weight, norm_bias = weights["norm.weight"], weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projection: left = x @ left_proj_weight^T (weight shape: hidden_dim, dim) because Linear with bias=False. In PyTorch, torch.nn.functional.linear(input, weight, bias=None) expects weight shape [out_features, in_features].
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    # left shape: [B,N,N,hidden_dim] -> using torch.einsum? Actually we can use torch.nn.functional.linear over last dimension.
    left = torch.nn.functional.linear(x, left_proj_weight) # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask: mask shape [B,N,N]; unsqueeze(-1) -> broadcast across hidden_dim.
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)
        left = left * mask
        right = right * mask
    else:
        mask = torch.ones((B,N,N,1), dtype=left.dtype, device=left.device)

    # Gates: linear then sigmoid
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Prepare for batched matmul: rearrange dims
    left_perm = left.permute(0,3,1,2).contiguous()  # [B, hidden_dim, N, N]
    right_perm = right.permute(0,3,1,2).contiguous() # same

    # flatten batch*hidden_dim:
    batch_hd = B * hidden_dim
    left_flat = left_perm.view(batch_hd, N, N)
    right_flat = right_perm.view(batch_hd, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # call Triton kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32  # or 64 maybe.
    # grid dims:
    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch_hd, num_tiles_m * num_tiles_n)

    # compute strides:
    stride_ah = left_flat.stride(0)
    stride_ak = left_flat.stride(1)
    stride_ad = left_flat.stride(2)
    stride_bh = right_flat.stride(0)
    stride_bn = right_flat.stride(1)
    stride_bk = right_flat.stride(2)
    stride_ch = out_flat.stride(0)
    stride_ck = out_flat.stride(1)
    stride_cd = out_flat.stride(2)

    # launch kernel
    batched_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        stride_ah, stride_ak, stride_ad,
        stride_bh, stride_bn, stride_bk,
        stride_ch, stride_ck, stride_cd,
        BLOCK_M, BLOCK_N, BLOCK_K,
        num_tiles_m, # maybe group?
        # no additional args
    )
    # reshape out back
    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous() # shape [B,N,N,hidden_dim]
    # to_out_norm (LayerNorm over hidden_dim)

    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # gating with out_gate (broadcast)
    out = out * out_gate

    # final linear to dim
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul (outgoing) implementation with Triton acceleration.

The core operation is the reduction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix multiplication
    left_d @ right_d^T  for each hidden dimension d.
We perform the projection, gating and layer‑norms in PyTorch and
off‑load the O(N²·hidden_dim) matrix multiplication to a Triton kernel.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes (all = seq_len)
    stride_ah, stride_ak, stride_ad,   # strides for A: batch, M, K
    stride_bh, stride_bn, stride_bk,   # strides for B: batch, N, K
    stride_ch, stride_ck, stride_cd,   # strides for C: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C[b, i, j] = Σ_k A[b, i, k] * B[b, j, k]   (B is transposed on the fly)
    A, B, C are 3‑D tensors of shape [batch, M/N, K].
    """
    pid_batch = tl.program_id(0)          # batch * hidden_dim dimension
    pid_tile = tl.program_id(1)           # tile index over M and N

    # Number of tiles along each axis
    num_tiles_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid_tile // num_tiles_n       # tile row
    pid_n = pid_tile % num_tiles_n        # tile column

    # Offsets within the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointer for this batch element
    batch_offset_A = pid_batch * stride_ah
    batch_offset_B = pid_batch * stride_bh
    batch_offset_C = pid_batch * stride_ch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = tl.arange(0, BLOCK_K) + k_offset
        mask_k = cur_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptr = A_ptr + batch_offset_A + offs_m[:, None] * stride_ak + cur_k[None, :] * stride_ad
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (transposed on the fly): shape (BLOCK_K, BLOCK_N)
        b_ptr = B_ptr + batch_offset_B + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)   # accumulation in FP32

    # Write results back to C
    c_ptr = C_ptr + batch_offset_C + offs_m[:, None] * stride_ck + offs_n[None, :] * stride_cd
    tl.store(c_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.
    Arguments:
        data = (input_tensor, mask_tensor, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # Unpack arguments
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional masking (mask shape [B, N, N])
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating and sigmoid
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for batched matmul: reshape to [B*hidden_dim, N, N]
    # ------------------------------------------------------------------
    # Shape: [B, N, N, hidden_dim] -> permute to bring hidden_dim next to batch
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, hidden_dim, N, N]

    B, N = input_tensor.shape[0], input_tensor.shape[1]

    batch_hd = B * hidden_dim
    left_flat = left_perm.view(batch_hd, N, N)   # [B*hidden_dim, N, N]
    right_flat = right_perm.view(batch_hd, N, N) # same layout

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # 6. Triton batched matrix multiplication (A @ B^T)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch_hd, num_tiles_m * num_tiles_n)

    # Strides in element units
    stride_ah = left_flat.stride(0)
    stride_ak = left_flat.stride(1)
    stride_ad = left_flat.stride(2)

    stride_bh = right_flat.stride(0)
    stride_bn = right_flat.stride(1)
    stride_bk = right_flat.stride(2)

    stride_ch = out_flat.stride(0)
    stride_ck = out_flat.stride(1)
    stride_cd = out_flat.stride(2)

    batched_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        stride_ah, stride_ak, stride_ad,
        stride_bh, stride_bn, stride_bk,
        stride_ch, stride_ck, stride_cd,
        BLOCK_M, BLOCK_N, BLOCK_K,
        # no additional constexprs
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Second LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )

    # ------------------------------------------------------------------
    # 9. Output gating and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```
