@triton.jit
def triu_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_o0, stride_o1, stride_o2, stride_o3,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    EPS: tl.constexpr,
):
    pid_bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    b = pid_bh // H
    h = pid_bh % H

    # compute starting offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out-of-range
    mask_m = offs_m < N
    mask_n = offs_n < N

    # compute pointer offsets
    # left: L[b,h,i,k]
    a_ptrs = left_ptr + b*stride_l0 + h*stride_l1 + offs_m[:, None]*stride_l2 + offs_k[None,:]*stride_l3

    # right: B[k,j] where B is right transposed: right[b,h,j,k]
    # Offsets: b*stride_r0 + h*stride_r1 + offs_k[:, None]*stride_r3 + offs_n[None,:]*stride_r2
    b_ptrs = right_ptr + b*stride_r0 + h*stride_r1 + offs_k[:, None]*stride_r3 + offs_n[None,:]*stride_r2

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None,:] < N), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < N) & mask_n[None,:], other=0.0)
        # compute dot product: a (M,K) dot b (K,N)
        acc += tl.dot(a, b)

        # increment pointers to next k block
        a_ptrs += BLOCK_K * stride_l3
        b_ptrs += BLOCK_K * stride_r3

    # apply mask for out-of-range rows/cols
    # store result
    out_ptrs = out_ptr + b*stride_o0 + h*stride_o1 + offs_m[:, None]*stride_o2 + offs_n[None,:]*stride_o3
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None,:])

```python
import torch
import triton
import triton.language as tl

@triton.jit
def triu_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_o0, stride_o1, stride_o2, stride_o3,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,h,i,j] = sum_k left[b,h,i,k] * right[b,h,j,k]
    using tiled matrix multiplication.
    left: shape (B, H, N, N) with layout [B, H, i, k]
    right: shape (B, H, N, N) with layout [B, H, j, k]
    out:   shape (B, H, N, N) with layout [B, H, i, j]
    """
    pid_bh = tl.program_id(0)   # batch * hidden index
    pid_m  = tl.program_id(1)   # tile row i
    pid_n  = tl.program_id(2)   # tile col j

    # decode batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # Pointers to the start of the tiles (no K offset yet)
    a_ptrs = left_ptr + b * stride_l0 + h * stride_l1 + offs_m[:, None] * stride_l2 + offs_k[None, :] * stride_l3
    # right is accessed as right[b, h, k, j] i.e. we swap the last two dims
    b_ptrs = right_ptr + b * stride_r0 + h * stride_r1 + offs_k[:, None] * stride_r3 + offs_n[None, :] * stride_r2

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)

        # advance K dimension
        a_ptrs += BLOCK_K * stride_l3
        b_ptrs += BLOCK_K * stride_r3

    # Store result
    c_ptrs = out_ptr + b * stride_o0 + h * stride_o1 + offs_m[:, None] * stride_o2 + offs_n[None, :] * stride_o3
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    Computes out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
    where
        left  : (B, H, N, N) laid out as [B, H, i, k]
        right : (B, H, N, N) laid out as [B, H, j, k] (accessed transposed)
        out   : (B, H, N, N) laid out as [B, H, i, j]
    The computation is tiled with block sizes BLOCK_M × BLOCK_N and
    summed over K in steps of BLOCK_K.
    """

    pid_m  = tl.program_id(1)   # tile row (i)
    pid_n  = tl.program_id(2)   # tile column (j)

    b = pid_bh // H
    h = pid_bh % H

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_n = offs_n < N
    mask_k = offs_k < N

              + h * stride_l1
              + offs_m[:, None] * stride_l2
              + offs_k[None, :] * stride_l3)

              + offs_n[None, :] * stride_r2)

    if not config.get("nomask", True):
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ----- Gating ----- 
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----- Triton matmul (core TriMul) -----
    # Bring hidden dimension to the second axis for convenient tiling.
    left_view = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_view.shape  # H == hidden_dim

    out_view = torch.empty_like(left_view)  # (B, H, N, N)

    # Tunable block sizes (work well on A100/B200 GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                  # batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,           # row tiles
        (N + BLOCK_N - 1) // BLOCK_N,           # col tiles
    )

    triu_mul_matmul_kernel[grid](
        left_view,
        right_view,
        out_view,
        B,
        H,
        N,
        left_view.stride(0),
        left_view.stride(1),
        left_view.stride(2),
        left_view.stride(3),
        right_view.stride(0),
        right_view.stride(1),
        right_view.stride(2),
        right_view.stride(3),
        out_view.stride(0),
        out_view.stride(1),
        out_view.stride(2),
        out_view.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout [B, N, N, hidden_dim]
    out = out_view.permute(0, 2, 3, 1).contiguous()

    # ----- Output LayerNorm, gating and final projection -----
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    out_norm = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps
    )
    out = out_norm * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # back to dim

    return out
```
