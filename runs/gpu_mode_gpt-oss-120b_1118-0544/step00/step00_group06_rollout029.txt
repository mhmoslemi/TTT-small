def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # left_proj, right_proj
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # Apply mask if present
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1)  # shape [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left_proj = left_proj * left_gate
    right_proj = right_proj * right_gate

    # core contraction: compute out_hidden
    B, N, _, _ = input_tensor.shape
    # Prepare output tensor
    out_hidden = torch.empty((B, N, N, hidden_dim), dtype=input_tensor.dtype, device=device)

    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides
    stride_left = left_proj.stride()
    stride_right = right_proj.stride()
    stride_out = out_hidden.stride()

    # we need 4 strides each
    stride_lb, stride_li, stride_lk, stride_ld = stride_left
    stride_rb, stride_rj, stride_rk, stride_rd = stride_right
    stride_ob, stride_oi, stride_oj, stride_od = stride_out

    # compute grid
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )
    tri_mul_kernel[grid](
        left_proj,
        right_proj,
        out_hidden,
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # normalize result
    out_norm = torch.nn.functional.layer_norm(out_hidden, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out_norm = out_norm * out_gate
    # final linear projection
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    return output

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The kernel computes the core cubic contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
for each batch `b` and hidden dimension `d`.  This is an
N×N matrix multiplication for every hidden channel, i.e.
out_d = left_d @ right_dᵀ.  The surrounding LayerNorms,
Linear projections and sigmoid gates are performed in
PyTorch; only the O(N³·hidden_dim) contraction is off‑loaded
to Triton.

The Triton kernel is a tiled GEMM where each program id
covers a tile of the output matrix (i‑tile, j‑tile) for a
specific (batch, hidden‑dim) pair.  The loops over the
reduction dimension `k` are blocked (BLOCK_K) and the inner
product is accumulated in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,          # const float* left  [B, N, N, D]
    right_ptr,         # const float* right [B, N, N, D]
    out_ptr,           # float* out         [B, N, N, D]
    B,
    N,
    D,
    stride_lb, stride_li, stride_lk, stride_ld,   # strides for left
    stride_rb, stride_rj, stride_rk, stride_rd,   # strides for right
    stride_ob, stride_oi, stride_oj, stride_od,   # strides for out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]"""

    pid_i = tl.program_id(0)    # tile along i
    pid_j = tl.program_id(1)    # tile along j
    pid_bd = tl.program_id(2)   # combined batch * hidden_dim

    # Decode batch and hidden dimension from pid_bd
    b = pid_bd // D
    d = pid_bd % D

    # Offsets for the i and j tiles
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers to the start of the (b,d) slice
    left_base = left_ptr + b * stride_lb + d * stride_ld
    right_base = right_ptr + b * stride_rb + d * stride_rd
    out_base = out_ptr + b * stride_ob + d * stride_od

    # Accumulator for the output tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_offset = kb * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + (offs_i[:, None] * stride_li) + (offs_k[None, :] * stride_lk)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile transposed: shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + (offs_k[:, None] * stride_rk) + (offs_j[None, :] * stride_rj)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Accumulate dot product
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # Store the result
    out_ptrs = out_base + (offs_i[:, None] * stride_oi) + (offs_j[None, :] * stride_oj)
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        output tensor of shape [bs, seq_len, seq_len, dim]
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                # input / output feature dimension
    hidden_dim = config["hidden_dim"]  # inner hidden dimension

    # ------------------------------------------------------------------
    # layer‑norm on the input (last dimension)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid) and gated projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Core cubic contraction using Triton
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape

    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out_hidden = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # Block sizes (tuned for typical B200 GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides (in elements) for each tensor
    s_lb, s_li, s_lk, s_ld = left.stride()
    s_rb, s_rj, s_rk, s_rd = right.stride()
    s_ob, s_oi, s_oj, s_od = out_hidden.stride()

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,                # batch * hidden_dim as third axis
    )

    tri_mul_kernel[grid](
        left,
        right,
        out_hidden,
        B,
        N,
        hidden_dim,
        s_lb, s_li, s_lk, s_ld,
        s_rb, s_rj, s_rk, s_rd,
        s_ob, s_oi, s_oj, s_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm, gate, final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(out_hidden,
                            normalized_shape=(hidden_dim,),
                            weight=weights["to_out_norm.weight"],
                            bias=weights["to_out_norm.bias"])
    out_norm = out_norm * out_gate

    output = F.linear(out_norm, weights["to_out.weight"])

    return output
