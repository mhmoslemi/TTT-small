@triton.jit
def trmul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_lb, stride_li, stride_lk,
    stride_rb, stride_rj, stride_rk,
    stride_ob, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # compute block indices
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    total_per_batch = num_i * num_j

    batch = pid // total_per_batch
    pid = pid % total_per_batch

    i_block = pid // num_j
    j_block = pid % num_j

    i0 = i_block * BLOCK_M
    j0 = j_block * BLOCK_N

    i_idx = i0 + tl.arange(0, BLOCK_M)
    j_idx = j0 + tl.arange(0, BLOCK_N)

    mask_i = i_idx < N
    mask_j = j_idx < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    k = 0
    while k < N:
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        left_ptrs = left_ptr + batch * stride_lb + i_idx[:, None] * stride_li + cur_k[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0, dtype=tl.float32)

        right_ptrs = right_ptr + batch * stride_rb + j_idx[:, None] * stride_rj + cur_k[None, :] * stride_rk
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0, dtype=tl.float32)

        # outer product across k
        for kk in range(BLOCK_K):
            # left column vector shape (BLOCK_M,)
            left_vec = left[:, kk]   # (BLOCK_M,)
            right_vec = right[:, kk] # (BLOCK_N,)
            acc += left_vec[:, None] * right_vec[None, :]

        k += BLOCK_K

    # store result
    out_ptrs = out_ptr + batch * stride_ob + i_idx[:, None] * stride_oi + j_idx[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul "outgoing" implementation using Triton.

The forward pass consists of:
1. LayerNorm on the input.
2. Linear projections (left/right) and gating (left/right/out).
3. Optional mask applied to the projected tensors.
4. Contraction over the intermediate sequence dimension:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is implemented as a block‑wise GEMM per hidden channel using a Triton
   kernel that computes a tile of the (i,j) matrix for a single channel.
5. LayerNorm on the hidden dimension, gating, and final linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trmul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_lb, stride_li, stride_lk,
    stride_rb, stride_rj, stride_rk,
    stride_ob, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of tiles in each dimension
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    total_per_batch = num_i * num_j

    batch = pid // total_per_batch
    pid = pid % total_per_batch

    i_block = pid // num_j
    j_block = pid % num_j

    i0 = i_block * BLOCK_M
    j0 = j_block * BLOCK_N

    i_idx = i0 + tl.arange(0, BLOCK_M)
    j_idx = j0 + tl.arange(0, BLOCK_N)

    mask_i = i_idx < N
    mask_j = j_idx < N

    # accumulator for the (i,j) tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    k = 0
    while k < N:
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # load left tile  [M, K]
        left_ptrs = left_ptr + batch * stride_lb + i_idx[:, None] * stride_li + cur_k[None, :] * stride_lk
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float32,
        )

        # load right tile [N, K]
        right_ptrs = right_ptr + batch * stride_rb + j_idx[:, None] * stride_rj + cur_k[None, :] * stride_rk
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float32,
        )

        # outer‑product accumulation over the K tile
        for kk in range(BLOCK_K):
            # each column is a vector of length M / N
            left_vec = left[:, kk]          # (BLOCK_M,)
            right_vec = right[:, kk]        # (BLOCK_N,)
            acc += left_vec[:, None] * right_vec[None, :]

        k += BLOCK_K

    # write the tile to output
    out_ptrs = out_ptr + batch * stride_ob + i_idx[:, None] * stride_oi + j_idx[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """TriMul forward – outgoing version.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: [B, N, N, dim]
            mask:         [B, N, N]  (may be ignored if config["nomask"] is True)
            weights: dict with the model parameters
            config:  dict with "dim", "hidden_dim", "nomask" (bool)

    Returns:
        Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----- LayerNorm on the input -----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ----- Linear projections -----
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # ----- Optional mask -----
    if not nomask:
        mask = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ----- Gating -----
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----- Contraction: out = Σ_k left[...,k,:] * right[...,k,:] -----
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Triton block sizes (chosen to keep register usage modest)
    BLOCK_M = 8
    BLOCK_N = 8
    BLOCK_K = 32

    n_i = (N + BLOCK_M - 1) // BLOCK_M
    n_j = (N + BLOCK_N - 1) // BLOCK_N
    total_progs = B * n_i * n_j

    # Loop over hidden channels – launch one kernel per channel
    for h in range(hidden_dim):
        left_h = left[..., h]   # shape [B, N, N]
        right_h = right[..., h] # shape [B, N, N]
        out_h = out[..., h]     # shape [B, N, N]

        # Strides (in elements)
        stride_lb = left_h.stride(0)
        stride_li = left_h.stride(1)
        stride_lk = left_h.stride(2)

        stride_rb = right_h.stride(0)
        stride_rj = right_h.stride(1)
        stride_rk = right_h.stride(2)

        stride_ob = out_h.stride(0)
        stride_oi = out_h.stride(1)
        stride_oj = out_h.stride(2)

        trmul_contract_kernel[
            (total_progs,)
        ](
            left_h,
            right_h,
            out_h,
            B,
            N,
            stride_lb,
            stride_li,
            stride_lk,
            stride_rb,
            stride_rj,
            stride_rk,
            stride_ob,
            stride_oi,
            stride_oj,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

    # Ensure kernel completion before proceeding
    torch.cuda.synchronize()

    # ----- Post‑processing -----
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # [B, N, N, hidden_dim]

    out = out * out_gate  # gating

    to_out_weight = weights["to_out.weight"]  # shape (dim, hidden_dim)
    out = F.linear(out, to_out_weight)       # final projection to dim

    return out
