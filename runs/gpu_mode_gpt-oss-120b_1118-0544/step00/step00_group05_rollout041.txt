def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_bi, stride_left_bk, stride_left_bd,
    stride_right_bj, stride_right_bk, stride_right_bd, 
    stride_out_bi, stride_out_bj, stride_out_bd,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    # Determine block indices
    # Compute number of blocks in i and j.
    total_i = tl.cdiv(N, BLOCK_I)
    total_j = tl.cdiv(N, BLOCK_J)
    total_b = B
    pid_b = pid // (total_i * total_j)
    pid_rem = pid % (total_i * total_j)
    pid_i = pid_rem // total_j
    pid_j = pid_rem % total_j

    i_offset = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    j_mask = j_offset < N

import torch
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    pid_d = tl.program_id(1)

    # compute block ids
    total_i = tl.cdiv(N, BLOCK_I)
    total_j = tl.cdiv(N, BLOCK_J)
    total_b = B
    # flatten mapping: b, i0, j0
    b_idx = pid // (total_i * total_j)
    rem = pid % (total_i * total_j)
    i_idx = rem // total_j
    j_idx = rem % total_j

    # offsets
    i_offset = i_idx * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = j_idx * BLOCK_J + tl.arange(0, BLOCK_J)
    d_offset = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    i_mask = i_offset < N
    j_mask = j_offset < N
    d_mask = d_offset < D

    # base batch offset in bytes? Actually pointer arithmetic uses element offsets not bytes; we can just compute offsets as integers multiplied by strides.
    batch_offset = b_idx * stride_left_b  # left and right share same batch stride

    # init accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # loop over k blocks
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # left pointer
        left_ptrs = left_ptr + batch_offset \
                    + i_offset[:, None, None] * stride_left_i \
                    + k_offset[None, :, None] * stride_left_k \
                    + d_offset[None, None, :] * stride_left_d
        left_tile = tl.load(left_ptrs,
                            mask=i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],
                            other=0.0)

        # right pointer

                    + j_offset[:, None, None] * stride_right_i \
                    + k_offset[None, :, None] * stride_right_k \
                    + d_offset[None, None, :] * stride_right_d

                             mask=j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],
                             other=0.0)

            # if k is out-of-bounds, the values are zero due to mask
            l = left_tile[:, kk, :]  # (BLOCK_I, BLOCK_D)
            r = right_tile[:, kk, :]  # (BLOCK_J, BLOCK_D)
            # broadcast multiplication
            acc += l[:, None, :] * r[None, :, :]

               + i_offset[:, None, None] * stride_out_i \
               + j_offset[None, :, None] * stride_out_j \
               + d_offset[None, None, :] * stride_out_d

"""
TriMul (outgoing) implementation using Triton.

The operator works on a 4‑D tensor `x` of shape [B, N, N, C] and performs:
  1. LayerNorm over the last dimension.
  2. Linear projections to a hidden dimension H (no bias).
  3. Element‑wise gates (sigmoid of a linear projection).
  4. Masking (if a mask is supplied).
  5. Contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
     which is a batched Gram‑matrix per hidden channel.
  6. LayerNorm over the hidden dimension, channel‑wise gating,
     and a final linear projection back to C.

Step 5 is the heavy N³·H work; it is implemented in a Triton kernel.
Each program (kernel instance) computes a tile of the output
(i‑block × j‑block × d‑block) and iterates over the k‑dimension in
tiles, accumulating the product in registers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------- #
# Triton kernel for the 4‑D contraction (the O(N³·H) part of TriMul)
# -------------------------------------------------------------------------- #
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,    # [B, N, N, H] – left projected tensor
    right_ptr,   # [B, N, N, H] – right projected tensor
    out_ptr,     # [B, N, N, H] – result tensor
    B, N, H,
    # strides for left (identical to right, but kept separate for clarity)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # strides for out
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    # 2‑D program grid:
    #   axis 0 -> (b, i_tile, j_tile)   (covers B·⌈N/BI⌉·⌈N/BJ⌉ blocks)
    #   axis 1 -> h_tile                (covers ⌈H/BH⌉ blocks)
    pid = tl.program_id(0)
    pid_h = tl.program_id(1)

    # ------------------------------------------------------------------ #
    # Decode grid indices
    # ------------------------------------------------------------------ #
    total_i = tl.cdiv(N, BLOCK_I)
    total_j = tl.cdiv(N, BLOCK_J)

    b_idx = pid // (total_i * total_j)
    rem = pid % (total_i * total_j)
    i_tile = rem // total_j
    j_tile = rem % total_j

    # Offsets for the current tile
    i_off = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)
    h_off = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    i_mask = i_off < N
    j_mask = j_off < N
    h_mask = h_off < H

    # Batch offset (same for left, right and out because they are
    # contiguous with the same leading stride)
    batch_off = b_idx * stride_left_b

    # ------------------------------------------------------------------ #
    # Accumulator for the tile
    # ------------------------------------------------------------------ #
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------ #
    # Loop over k in tiles of size BLOCK_K
    # ------------------------------------------------------------------ #
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # ----- load a block of left (i, k, h) ---------------------------------
        left_ptrs = left_ptr + batch_off \
                    + i_off[:, None, None] * stride_left_i \
                    + k_off[None, :, None] * stride_left_k \
                    + h_off[None, None, :] * stride_left_h
        left_tile = tl.load(
            left_ptrs,
            mask=i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :],
            other=0.0,
        )  # (BI, BK, BH)

        # ----- load a block of right (j, k, h) --------------------------------
        right_ptrs = right_ptr + batch_off \
                     + j_off[:, None, None] * stride_right_i \
                     + k_off[None, :, None] * stride_right_k \
                     + h_off[None, None, :] * stride_right_h
        right_tile = tl.load(
            right_ptrs,
            mask=j_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :],
            other=0.0,
        )  # (BJ, BK, BH)

        # ----- accumulate over the k dimension ---------------------------------
        # Unroll over the inner k dimension (BK is a compile‑time constant)
        for kk in range(0, BLOCK_K):
            # left[i, kk, h]  -> (BI, BH)
            # right[j, kk, h] -> (BJ, BH)
            l = left_tile[:, kk, :]        # (BI, BH)
            r = right_tile[:, kk, :]       # (BJ, BH)
            # broadcast to (BI, BJ, BH) and add to accumulator
            acc += l[:, None, :] * r[None, :, :]

    # ------------------------------------------------------------------ #
    # Store the computed tile
    # ------------------------------------------------------------------ #
    out_ptrs = out_ptr + batch_off \
               + i_off[:, None, None] * stride_out_i \
               + j_off[None, :, None] * stride_out_j \
               + h_off[None, None, :] * stride_out_h
    tl.store(
        out_ptrs,
        acc,
        mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
    )


# -------------------------------------------------------------------------- #
# Helper that launches the Triton kernel
# -------------------------------------------------------------------------- #
def _contract_left_right(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Perform the einsum `... i k d, ... j k d -> ... i j d` using Triton."""
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # tensor strides (element strides, not byte strides)
    sL = left.stride()   # (b, i, k, h)
    sR = right.stride()  # (b, j, k, h)
    sO = out.stride()    # (b, i, j, h)

    # Tile sizes – chosen to fit comfortably in registers and shared memory
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 8   # hidden‑dim sub‑tile

    grid_i = (B * ((N + BLOCK_I - 1) // BLOCK_I) * ((N + BLOCK_J - 1) // BLOCK_J),
              (H + BLOCK_H - 1) // BLOCK_H)

    _trimul_outgoing_kernel[grid_i](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=sL[0],
        stride_left_i=sL[1],
        stride_left_k=sL[2],
        stride_left_h=sL[3],
        stride_right_b=sR[0],
        stride_right_i=sR[1],
        stride_right_k=sR[2],
        stride_right_h=sR[3],
        stride_out_b=sO[0],
        stride_out_i=sO[1],
        stride_out_j=sO[2],
        stride_out_h=sO[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )
    return out


# -------------------------------------------------------------------------- #
# Entry point – replicates the forward pass of the reference PyTorch module
# -------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, C]
        - mask_tensor  : torch.Tensor, shape [B, N, N] (may be all‑ones)
        - weights_dict : mapping of parameter names to torch.Tensors
        - config_dict  : contains "dim" and "hidden_dim" entries

    Returns
    -------
    torch.Tensor : shape [B, N, N, C] – the updated pair representation
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------ #
    # 1. Initial LayerNorm over the channel dimension C
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-6,
    )

    # ------------------------------------------------------------------ #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------ #
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------ #
    # 3. Masking (if a mask is provided)
    # ------------------------------------------------------------------ #
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------ #
    # 4. Gating
    # ------------------------------------------------------------------ #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Contraction – the O(N³·H) work performed by Triton
    # ------------------------------------------------------------------ #
    contracted = _contract_left_right(left, right)   # shape [B, N, N, H]

    # ------------------------------------------------------------------ #
    # 6. Output LayerNorm over hidden dimension H
    # ------------------------------------------------------------------ #
    contracted = F.layer_norm(
        contracted,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-6,
    )

    # ------------------------------------------------------------------ #
    # 7. Apply final channel‑wise gate
    # ------------------------------------------------------------------ #
    contracted = contracted * out_gate

    # ------------------------------------------------------------------ #
    # 8. Final linear projection back to C
    # ------------------------------------------------------------------ #
    out = F.linear(contracted, weights["to_out.weight"])

    return out
