"""
TriMul forward (outgoing) implementation.

The heavy part (`einsum('... i k d, ... j k d -> ... i j d')`) is executed on
GPU by a Triton kernel.  The kernel computes, for each batch and hidden‑channel
block, a blocked matrix multiplication
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d] .
`left` and `right` already contain the linear projections, mask and elementwise
gates.  The kernel operates on tiles (BLOCK_M × BLOCK_N) of the output and a
small tile (BLOCK_D) of the hidden dimension, accumulating the result in
float32.  All other operations (LayerNorm, linear projections, sigmoid gates,
final LayerNorm / linear) are performed with PyTorch for simplicity.

The function `custom_kernel` expects a tuple:
    (input_tensor, mask_tensor, weight_dict, config_dict)
and returns the final Tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: blocked matrix multiplication with hidden dimension loop
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    L_ptr, R_ptr, O_ptr,               # pointers
    B, N, H,                            # batch, seq_len, hidden
    stride_BL, stride_iL, stride_kL, stride_hL,
    stride_BR, stride_iR, stride_kR, stride_hR,
    stride_BO, stride_iO, stride_jO, stride_hO,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid_m = tl.program_id(0)               # row tile
    pid_n = tl.program_id(1)               # col tile
    pid_hb = tl.program_id(2)              # batch * hidden‑block index

    # ----- decode batch and hidden‑block -----
    num_hblocks = tl.cdiv(H, BLOCK_D)
    b = pid_hb // num_hblocks
    hb = pid_hb % num_hblocks
    h_start = hb * BLOCK_D

    # ----- offsets within the tensors -----
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = h_start + tl.arange(0, BLOCK_D)

    # clamp indices to stay inside the allocated memory (out‑of‑range entries will be masked)
    i_off = tl.minimum(offs_i, N - 1)
    j_off = tl.minimum(offs_j, N - 1)
    d_off = tl.minimum(offs_d, H - 1)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ----- loop over the reduction dimension k -----
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        k_off = tl.minimum(offs_k, N - 1)
        mask_k = offs_k < N

        # masks for loads
        mask_L = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
        mask_R = mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]

        # pointers to the current tile
        L_ptrs = (L_ptr
                  + b * stride_BL
                  + i_off[:, None, None] * stride_iL
                  + k_off[None, :, None] * stride_kL
                  + d_off[None, None, :] * stride_hL)

        R_ptrs = (R_ptr
                  + b * stride_BR
                  + j_off[:, None, None] * stride_iR
                  + k_off[None, :, None] * stride_kR
                  + d_off[None, None, :] * stride_hR)

        # load tiles (fp16 -> fp32)
        L = tl.load(L_ptrs, mask=mask_L, other=0.0).to(tl.float32)   # (M, K, D)
        R = tl.load(R_ptrs, mask=mask_R, other=0.0).to(tl.float32)   # (N, K, D)

        # accumulate per hidden slice
        for d in range(0, BLOCK_D):
            # L[d] shape (M, K), R[d] shape (N, K)
            Ld = L[:, :, d]
            Rd = R[:, :, d]
            # Rd is transposed inside tl.dot (trans_b=True) => (K, N)
            acc[:, :, d] = acc[:, :, d] + tl.dot(Ld, Rd, trans_b=True)

    # ----- write the result -----
    O_ptrs = (O_ptr
              + b * stride_BO
              + i_off[:, None, None] * stride_iO
              + j_off[None, :, None] * stride_jO
              + d_off[None, None, :] * stride_hO)

    mask_O = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    tl.store(O_ptrs, acc, mask=mask_O)


# ----------------------------------------------------------------------
# Helper to launch the Triton kernel
# ----------------------------------------------------------------------
def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: [B, N, N, H] tensors (any floating dtype, preferably fp16)
    returns out: [B, N, N, H] tensor (float32)
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    # Ensure contiguous layout for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()

    out = torch.empty_like(left, dtype=torch.float32)

    # strides (int64)
    stride_BL, stride_iL, stride_kL, stride_hL = left.stride()
    stride_BR, stride_iR, stride_kR, stride_hR = right.stride()
    stride_BO, stride_iO, stride_jO, stride_hO = out.stride()

    # tile sizes (tuned for typical B200 memory/SM count)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64
    BLOCK_D = 16

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * ((H + BLOCK_D - 1) // BLOCK_D),
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_BL,
        stride_iL,
        stride_kL,
        stride_hL,
        stride_BR,
        stride_iR,
        stride_kR,
        stride_hR,
        stride_BO,
        stride_iO,
        stride_jO,
        stride_hO,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Arguments:
        data = (input_tensor, mask_tensor, weight_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (same dtype as the model parameters, float32)
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Load model weights (all are 1‑D or 2‑D tensors already on the right device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (dim -> hidden_dim)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, left_proj_weight)   # [B,N,N,hidden]
    right = F.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # broadcast over hidden dim
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid(sigmoid(linear(x_norm))))
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul (einsum) via Triton kernel
    #    Convert to fp16 for reduced memory traffic, kernel accumulates in fp32.
    # ------------------------------------------------------------------
    left_k = left.to(torch.float16)
    right_k = right.to(torch.float16)

    intermediate = _tri_mul_triton(left_k, right_k)   # shape [B,N,N,hidden] float32

    # ------------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, out‑gate, final linear to dim
    # ------------------------------------------------------------------
    # to_out_norm is a LayerNorm over the hidden dimension
    out = F.layer_norm(
        intermediate,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    # apply outgoing gate (broadcast over hidden dim)
    out = out * out_gate
    # final linear back to original dimension
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
