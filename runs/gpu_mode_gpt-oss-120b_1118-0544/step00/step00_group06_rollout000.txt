@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_z = tl.program_id(0)  # batch*channel
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch, channel
    b = pid_z // H
    h = pid_z % H

    # Compute offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M,K)
        left_ptrs = left_ptr + b*stride_left_b + offs_i[:, None]*stride_left_i + offs_k[None, :] * stride_left_k + h*stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile transposed (K,N) i.e., (K,N) where N is j
        right_ptrs = right_ptr + b*stride_right_b + offs_k[:, None]*stride_right_k + offs_j[None, :] * stride_right_i + h*stride_right_h
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Multiply-Accumulate
        acc += tl.dot(left, right)  # shape (M,N)

    # Store result
    out_ptrs = out_ptr + b*stride_out_b + offs_i[:, None]*stride_out_i + offs_j[None, :] * stride_out_j + h*stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul forward kernel (outgoing version) with Triton acceleration.

The core of the module is the tensor contraction:
    out[i, j, d] = Σ_k left[i, k, d] * right[j, k, d]
which is equivalent to a batched matrix multiplication
    out[..., :, :, d] = left[..., :, :, d] @ right[..., :, :, d].T

We implement this contraction with a Triton kernel that
  * treats each (batch, hidden) pair as an independent GEMM,
  * tiles the computation along the i, j, and k dimensions,
  * accumulates in float32 (even when the inputs are float16),
  * writes the result back to a contiguous output tensor.

All other operations (LayerNorm, Linear projections, sigmoid gates,
masking, final normalization and projection) are performed with
standard PyTorch for simplicity.  The kernel therefore provides the
most expensive O(N³) part of the forward pass.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,      # const *float32
    right_ptr,     # const *float32
    out_ptr,       # *float32
    B, N, H,       # int32: batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h].
    left/right/out are assumed to be in contiguous (NCHW-like) layout:
        (B, N, N, H) with strides given explicitly.
    """
    pid_z = tl.program_id(0)      # combined batch*hidden index
    pid_m = tl.program_id(1)      # tile over i (row)
    pid_n = tl.program_id(2)      # tile over j (col)

    # derive batch and hidden-index from pid_z
    b = pid_z // H
    h = pid_z % H

    # Offsets for the i and j dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of tiles along the reduction dimension k
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----------- load left tile (M x K) -----------------
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k + \
            h * stride_left_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ----------- load right tile (K x N) -----------------
        # we load it already transposed: rows = k, cols = j
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            offs_k[:, None] * stride_right_k + \
            offs_j[None, :] * stride_right_i + \
            h * stride_right_h
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # ----------- dot product & accumulate ----------------
        acc += tl.dot(left, right)   # (M, K) @ (K, N) -> (M, N)

    # ----------- write back --------------------------------
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j + \
        h * stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def triton_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper for tri_mul_kernel.  Left/right shape: (B, N, N, H) float32.
    Returns output of same shape.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # tune tile sizes (experimentally good for N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8
    )
    return out


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel
    for the expensive contraction.
    Arguments:
        data = (input, mask, weights, config)
            input   : torch.Tensor [B, N, N, dim] (float32)
            mask    : torch.Tensor [B, N, N]   (bool or float) or None
            weights : dict of learned parameters (see docstring)
            config  : dict with keys 'dim' and 'hidden_dim' (and possibly 'nomask')
    Returns:
        output tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) First LayerNorm (over dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections
    # ------------------------------------------------------------------
    # (B,N,N,dim) @ (hidden_dim, dim) -> (B,N,N,hidden_dim)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Masking (if present)
    # ------------------------------------------------------------------
    if (mask is None) or nomask:
        # no masking required
        masked_left = left
        masked_right = right
    else:
        # mask shape (B,N,N) -> (B,N,N,1) for broadcasting
        mask_f = mask.unsqueeze(-1).to(x.dtype)
        masked_left = left * mask_f
        masked_right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = masked_left * left_gate
    right = masked_right * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction via Triton kernel
    # ------------------------------------------------------------------
    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = triton_tri_mul(left, right)   # shape (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
