def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # layernorm
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-12)

    # linear projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # mask
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)  # shape B,N,N,1
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    left = left_proj * left_gate
    right = right_proj * right_gate

    # allocate output
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # Stides
    stride_l_b, stride_l_i, stride_l_j, stride_l_d = left.stride()
    stride_r_b, stride_r_i, stride_r_j, stride_r_d = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # Perform contraction via Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_D = 8  # tile over hidden dimension

    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    num_d = (hidden_dim + BLOCK_D - 1) // BLOCK_D
    grid = (B * num_i * num_j, num_d)

    tri_mul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_j, stride_l_d,
        stride_r_b, stride_r_i, stride_r_j, stride_r_d,
        stride_o_b, stride_o_i, stride_o_j, stride_o_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D
    )

    # Apply to_out_norm and gating and final linear
    # to_out_norm
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-12)

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
Triton implementation of the “outgoing” Triangle Multiplicative Update (TriMul).

The kernel computes, for every batch b, every pair of positions i,j and every
hidden channel d,

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are already projected, gated and (optionally) masked.
The computation is a batched matrix multiplication of `left[...,d]` with the
transpose of `right[...,d]`.  The kernel tiles the i‑, j‑, k‑ and d‑dimensions
using a 4‑dimensional block (M,N,K,D) and accumulates the partial results with
`tl.dot`.  The outer Python code handles LayerNorms, linear projections, gates
and the final output transformation, while the heavy O(N³·C) contraction is
fused in the Triton kernel.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel: tiled batched matmul over the k dimension for a block of
# hidden channels (D).  The kernel is launched with a 2‑D grid:
#   grid[0] : (batch, i‑tile, j‑tile)
#   grid[1] : d‑tile
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # const float*   [B,N,N,D]
    right_ptr,              # const float*   [B,N,N,D]
    out_ptr,                # float*         [B,N,N,D]

    B, N, D,                # int32 scalars
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,   # int32 strides for left
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,   # int32 strides for right (transposed)
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,   # int32 strides for out

    BLOCK_M: tl.constexpr,  # tile size for i dimension
    BLOCK_N: tl.constexpr,  # tile size for j dimension
    BLOCK_K: tl.constexpr,  # tile size for k reduction
    BLOCK_D: tl.constexpr,  # tile size for hidden dimension
):
    pid = tl.program_id(0)
    pid_d = tl.program_id(1)               # which D‑tile we are processing

    # --------------------------------------------------------------
    # Compute 3‑D indices (batch, i‑tile, j‑tile)
    # --------------------------------------------------------------
    num_i_tiles = tl.cdiv(N, BLOCK_M)
    num_j_tiles = tl.cdiv(N, BLOCK_N)

    b = pid // (num_i_tiles * num_j_tiles)
    tmp = pid % (num_i_tiles * num_j_tiles)
    i_tile = tmp // num_j_tiles
    j_tile = tmp % num_j_tiles

    # Offsets for this tile
    offs_i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    # Masks for out‑of‑bounds (i,j,d)
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # --------------------------------------------------------------
    # Accumulator for the output tile (i, j, d)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over k‑tiles
    # --------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)          # k‑dim offsets
        mask_k = offs_k < N

        # ----------------------------------------------------------
        # Load left: shape (M, K, D)
        # ----------------------------------------------------------
        left_ptrs = left_ptr + \
            b * stride_l_b + \
            offs_i[:, None, None] * stride_l_i + \
            offs_k[None, :, None] * stride_l_k + \
            offs_d[None, None, :] * stride_l_d
        left_mask = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # ----------------------------------------------------------
        # Load right transposed: shape (K, N, D)
        #   we need right[j, k, d] -> (k, j, d) for dot product
        # ----------------------------------------------------------
        right_ptrs = right_ptr + \
            b * stride_r_b + \
            offs_k[:, None, None] * stride_r_k + \
            offs_j[None, :, None] * stride_r_i + \
            offs_d[None, None, :] * stride_r_d
        right_mask = mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        right = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # ----------------------------------------------------------
        # Compute dot products for every d in the tile
        # ----------------------------------------------------------
        for d_idx in range(0, BLOCK_D):
            # Skip d‑values that are out of range (all‑zero tile anyway)
            # The mask already zeroed those entries.
            a = left[:, :, d_idx]   # (M, K)
            b_mat = right[:, :, d_idx]   # (K, N)
            acc[:, :, d_idx] += tl.dot(a, b_mat)   # (M, N)

    # --------------------------------------------------------------
    # Store the result
    # --------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_o_b + \
        offs_i[:, None, None] * stride_o_i + \
        offs_j[None, :, None] * stride_o_j + \
        offs_d[None, None, :] * stride_o_d
    out_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict of torch.Tensors containing all linear/LN weights
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1. LayerNorm on the input
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-12,
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden_dim]
    right_proj = F.linear(x, weights["right_proj.weight"])    # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 3. Gating values
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).type_as(x)   # [B,N,N,1]
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # --------------------------------------------------------------
    # 5. Apply gates
    # --------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Ensure all tensors are contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()

    # --------------------------------------------------------------
    # 6. Contract across the triangle dimension using Triton
    # --------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (in element units, not bytes)
    stride_l_b, stride_l_i, stride_l_j, stride_l_d = left.stride()
    stride_r_b, stride_r_i, stride_r_j, stride_r_d = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # Tile sizes (chosen to fit typical GPU registers)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8        # hidden‑dim tile

    num_i_tiles = (N + BLOCK_M - 1) // BLOCK_M
    num_j_tiles = (N + BLOCK_N - 1) // BLOCK_N
    num_d_tiles = (hidden_dim + BLOCK_D - 1) // BLOCK_D

    # Grid: (batch * i_tiles * j_tiles, d_tiles)
    grid = (B * num_i_tiles * num_j_tiles, num_d_tiles)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_j, stride_l_d,
        stride_r_b, stride_r_i, stride_r_j, stride_r_d,
        stride_o_b, stride_o_i, stride_o_j, stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 7. Output LayerNorm, gating and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-12,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
