@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, H,
                   stride_left_b, stride_left_i, stride_left_k, stride_left_h,
                   stride_right_b, stride_right_i, stride_right_k, stride_right_h,
                   stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

    pid_bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    b = pid_bh // H
    h = pid_bh % H

    # compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid i, j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for left (A) block (M x K)
        a_ptrs = left_ptr + \
                 b * stride_left_b + \
                 h * stride_left_h + \
                 offs_i[:, None] * stride_left_i + \
                 offs_k[None, :] * stride_left_k

        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # pointers for right (B) block (N x K) transposed => (K x N)
        # right shape: (B, j, k, H) i.e., (B, N, N, H). We need B_T: (K x N)
        b_ptrs = right_ptr + \
                 b * stride_right_b + \
                 h * stride_right_h + \
                 offs_k[:, None] * stride_right_k + \
                 offs_j[None, :] * stride_right_i

        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product accumulate
        acc += tl.dot(a, b)

    # store result
    out_ptrs = out_ptr + \
               b * stride_out_b + \
               h * stride_out_h + \
               offs_i[:, None] * stride_out_i + \
               offs_j[None, :] * stride_out_j

    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul outgoing kernel (AlphaFold3) – Triton implementation.

The heavy O(N³) contraction:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

is performed with a custom Triton kernel that does a batched matrix
multiplication per hidden dimension `h`.  All other operations (LayerNorm,
linear projections, gating, final LayerNorm and output Linear) are done
with PyTorch for simplicity.

The kernel tiles the i‑ and j‑dimensions (BLOCK_M × BLOCK_N) and reduces
over k in blocks of size BLOCK_K, handling arbitrary sequence lengths
(N) via masks.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel for the core contraction (per hidden dimension)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_bh = tl.program_id(0)   # combined batch * hidden dimension
    pid_i = tl.program_id(1)    # i‑tile
    pid_j = tl.program_id(2)    # j‑tile

    # decode batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # range of i / j indices handled by this block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator for the partial sum
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kb * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------------------- load left block (M x K) --------------------
        a_ptrs = left_ptr \
                 + b * stride_left_b \
                 + h * stride_left_h \
                 + offs_i[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # -------------------- load right block (K x N) -------------------
        # right layout: (B, j, k, H) → we need (K, N) where K = k, N = j
        b_ptrs = right_ptr \
                 + b * stride_right_b \
                 + h * stride_right_h \
                 + offs_k[:, None] * stride_right_k \
                 + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # -------------------- dot product accumulation --------------------
        acc += tl.dot(a, b)

    # -------------------- store result --------------------
    out_ptrs = out_ptr \
               + b * stride_out_b \
               + h * stride_out_h \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Helper to launch the kernel
# ----------------------------------------------------------------------
def tri_mul_forward(left: torch.Tensor, right: torch.Tensor,
                    BLOCK_M=64, BLOCK_N=64, BLOCK_K=64) -> torch.Tensor:
    """
    left, right: [B, N, N, H] (float32)
    returns out of same shape using the triton kernel.
    """
    assert left.shape == right.shape
    B, N, N2, H = left.shape
    assert N == N2

    out = torch.empty_like(left)

    # strides (int64) for pointer arithmetic
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    grid = (
        B * H,                         # batch * hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,  # i tiles
        (N + BLOCK_N - 1) // BLOCK_N   # j tiles
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # a reasonable default; Triton will pick if omitted
    )
    return out

# ----------------------------------------------------------------------
# Main entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    data: Tuple (input_tensor, mask, weights, config)
    Returns: output tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # Extract config
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on the input (normalize over the channel dimension)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-12)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]        # (H, dim)
    right_proj_weight = weights["right_proj.weight"]      # (H, dim)

    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Apply mask if provided (mask shape: [B, N, N])
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating weights
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core contraction via Triton
    # ------------------------------------------------------------------
    out = tri_mul_forward(left, right,
                          BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # Post‑processing: hidden‑dim LayerNorm, gating, and final linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-12)

    out = out * out_gate                       # element‑wise gating

    to_out_weight = weights["to_out.weight"]   # (dim, H)
    out = F.linear(out, to_out_weight)        # final shape [B, N, N, dim]

    return out
