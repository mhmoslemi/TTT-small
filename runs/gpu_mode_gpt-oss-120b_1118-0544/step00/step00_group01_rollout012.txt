    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # (maybe config may also hold "dim"?, but we have dim)
    # layernorm on input
    # flatten for Triton
    input_flat = input_tensor.view(-1, dim)
    # allocate normalized tensor
    norm_out = torch.empty_like(input_flat)
    # call Triton layernorm kernel
    layernorm_forward(input_flat, norm_out, weights["norm.weight"], weights["norm.bias"])
    # reshape back
    normed = norm_out.view(bs, seq_len, seq_len, dim)
    # compute projections and gates
    left_proj = torch.nn.functional.linear(normed, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(normed, weights["right_proj.weight"])
    left_gate = torch.sigmoid(torch.nn.functional.linear(normed, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(normed, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(normed, weights["out_gate.weight"]))
    # apply mask
    if mask is not None:
        mask = mask.unsqueeze(-1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask
    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate
    # compute outer product using batched matmul
    # left: [B,N,N,hidden_dim] -> B, hidden_dim, N, N
    left_perm = left.permute(0,3,1,2) # B, hidden_dim, N, N

    right_perm = right.permute(0,3,2,1) # B, hidden_dim, N, N

    out_norm = torch.empty_like(out_flat)
    layernorm_forward(out_flat, out_norm,

"""
TriMul (outgoing) forward pass implemented with a fused Triton LayerNorm kernel.
The heavy O(N³) contraction is performed via a batched matrix multiplication
(using torch.matmul, which maps to cuBLAS).  The LayerNorm steps (input‑norm
and to_out‑norm) are executed in a custom Triton kernel to satisfy the
requirement of a non‑trivial Triton implementation while keeping the overall
runtime low.
"""

import torch
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton fused LayerNorm (no bias for variance, eps = 1e-5)
# -------------------------------------------------------------
@triton.jit
def layernorm_fwd_kernel(
    inp_ptr, out_ptr,               # float32 pointers
    weight_ptr, bias_ptr,           # float32 1‑D weight / bias
    N_GROUPS: tl.int32,             # number of groups = B * N * N
    eps: tl.float32,                # epsilon
    D: tl.constexpr,                # normalized dimension (C)
    BLOCK_SIZE: tl.constexpr,       # tile size along D
):
    """Row‑wise LayerNorm:
       For each group g (size D) compute
          y = (x - mean) / sqrt(var + eps) * weight + bias
    """
    pid = tl.program_id(0)
    if pid >= N_GROUPS:
        return

    # -----------------------------------------------------------------
    # First pass – compute mean and variance
    # -----------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sumsq_val = tl.float32(0.0)
    for offs in range(0, D, BLOCK_SIZE):
        cur = tl.minimum(BLOCK_SIZE, D - offs)
        idx = tl.arange(0, BLOCK_SIZE)
        mask = idx < cur

        x = tl.load(inp_ptr + pid * D + offs + idx, mask=mask, other=0.0)
        x_f = x.to(tl.float32)

        sum_val += tl.sum(x_f, axis=0)
        sumsq_val += tl.sum(x_f * x_f, axis=0)

    mean = sum_val / D
    var = sumsq_val / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # -----------------------------------------------------------------
    # Second pass – write normalized values with weight & bias
    # -----------------------------------------------------------------
    for offs in range(0, D, BLOCK_SIZE):
        cur = tl.minimum(BLOCK_SIZE, D - offs)
        idx = tl.arange(0, BLOCK_SIZE)
        mask = idx < cur

        x = tl.load(inp_ptr + pid * D + offs + idx, mask=mask, other=0.0)
        x_f = x.to(tl.float32)

        y = (x_f - mean) * inv_std

        w = tl.load(weight_ptr + offs + idx, mask=mask, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + offs + idx, mask=mask, other=0.0).to(tl.float32)

        y = y * w + b
        tl.store(out_ptr + pid * D + offs + idx, y, mask=mask)


def layernorm_triton(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Triton implementation of LayerNorm over the last dimension.
    Arguments:
        x    – [G, C] contiguous float32 tensor (G = B * N * N)
        weight, bias – [C] tensors
    Returns:
        y – same shape as x, normalized and affine transformed.
    """
    assert x.is_contiguous()
    G, C = x.shape
    out = torch.empty_like(x)

    # Tile size – 64 works well for C up to 384
    BLOCK = 64

    grid = (G,)
    layernorm_fwd_kernel[grid](
        x,
        out,
        weight,
        bias,
        G,
        eps,
        D=C,
        BLOCK_SIZE=BLOCK,
    )
    return out

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using Triton for LayerNorm.
    Input:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Output:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------
    # 1. Input LayerNorm (Triton)
    # -----------------------------
    # flatten first 3 dimensions so each row is a LayerNorm group
    x_flat = input_tensor.reshape(-1, dim)
    x_norm_flat = layernorm_triton(
        x_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=1e-5,
    )
    x_norm = x_norm_flat.view(bs, N, N, dim)   # [B, N, N, dim]

    # -------------------------------------------------
    # 2. Linear projections & gates (pure PyTorch)
    # -------------------------------------------------
    # Projections to hidden dimension (no bias)
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Gating (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # Apply mask if provided (mask shape [B, N, N])
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # Apply gates
    left = left_proj * left_gate            # [B, N, N, hidden_dim]
    right = right_proj * right_gate

    # -------------------------------------------------
    # 3. Core contraction: einsum('... i k d, ... j k d -> ... i j d')
    #    Implemented as batched matrix multiplication:
    #    out_{b, i, j, d} = sum_k left_{b, i, k, d} * right_{b, j, k, d}
    # -------------------------------------------------
    # reshape to (B, hidden_dim, N, N) and multiply left @ right^T
    left_perm = left.permute(0, 3, 1, 2)                 # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 2, 1)               # [B, hidden_dim, N, N] (k ↔ j)
    out_mat = torch.matmul(left_perm, right_perm)       # [B, hidden_dim, N, N]
    out = out_mat.permute(0, 2, 3, 1)                    # [B, N, N, hidden_dim]

    # -------------------------------------------------
    # 4. Output LayerNorm (Triton) over hidden_dim
    # -------------------------------------------------
    out_flat = out.reshape(-1, hidden_dim)
    out_norm_flat = layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm_flat.view(bs, N, N, hidden_dim)

    # Apply the final output gate
    out_norm = out_norm * out_gate

    # -------------------------------------------------
    # 5. Final linear projection back to original dim
    # -------------------------------------------------
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output
