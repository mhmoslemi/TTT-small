@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Get program IDs
    pid_m = tl.program_id(0)  # row tile index
    pid_n = tl.program_id(1)  # col tile index
    pid_z = tl.program_id(2)  # batch*head index
    # Determine batch and head
    batch_id = pid_z // H
    head_id = pid_z % H

    # offs for i,j block
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension (reduction)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Compute offsets for the k block
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            batch_id * stride_left_b + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k + \
            head_id * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right block: shape (BLOCK_N, BLOCK_K)
        # Right is indexed as [batch, j, k, head], we need j and k
        right_ptrs = right_ptr + \
            batch_id * stride_right_b + \
            offs_j[:, None] * stride_right_j + \
            offs_k[None, :] * stride_right_k + \
            head_id * stride_right_h
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # Compute dot product over k: left (M,K) * right (N,K)?? 
        # Need right transposed: right[j,k] => we need right[j,k] dot left[i,k] across k.
        # Since both have shape (M,K) and (N,K), we can do tl.dot(left, right, trans_b=True)
        # But tl.dot only does dot between (M,K) and (K,N). So we need to supply right as (K,N). 
        # So we need to load right transposed: shape (BLOCK_K, BLOCK_N)
        right_T = tl.trans(right) # shape (BLOCK_K, BLOCK_N)
        # Accumulate
        acc += tl.dot(left, right_T)
    
    # Store result
    out_ptrs = out_ptr + \
        batch_id * stride_out_b + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j + \
        head_id * stride_out_h
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # We can assume mask is None or a tensor. If mask is None, set ones.
    if mask is None:
        mask = torch.ones_like(input_tensor[...,0], dtype=input_tensor.dtype, device=input_tensor.device)
    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    
    # Apply mask
    mask_unsqueeze = mask.unsqueeze(-1).type_as(left_proj)
    left = left_proj * mask_unsqueeze
    right = right_proj * mask_unsqueeze
    
    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    
    # Apply gates to left and right
    left = left * left_gate
    right = right * right_gate
    
    # Convert to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    
    # Compute contraction
    out = tri_mul_triton(left, right)  # returned as float16
    # Convert back to float32
    out = out.to(torch.float32)
    
    # to_out_norm (LayerNorm over hidden_dim)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    
    # Multiply out_gate (still float32)
    out_gate = out_gate.to(out.dtype) # ensure same dtype
    out = out * out_gate
    
    # Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    
    return out
```

"""
Custom TriMul operator: Combines layernorm, linear projections, gating, and an
efficient Triton kernel for the triple‑product contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
The kernel treats each head (d) as an independent matrix multiplication
(left @ right^T) and tiles the N×N output using BLOCK_M×BLOCK_N
tiles with a K‑reduction of size BLOCK_K.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched per‑head GEMM: out = left @ right^T  (left/right shape B×N×N×H)."""
    pid_m = tl.program_id(0)   # tile row index
    pid_n = tl.program_id(1)   # tile col index
    pid_z = tl.program_id(2)   # combined batch‑head index

    # decode batch and head from pid_z
    batch_id = pid_z // H
    head_id  = pid_z % H

    # offsets within the N dimension
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base offsets for this batch/head
    batch_off = batch_id * stride_left_b
    head_off   = head_id * stride_left_h

    # accumulator (float32 for numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a BLOCK_M×BLOCK_K panel of left (i, k)
        left_ptrs = left_ptr + \
            batch_off + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k + \
            head_off
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # Load a BLOCK_N×BLOCK_K panel of right (j, k)
        right_ptrs = right_ptr + \
            batch_off + \
            offs_j[:, None] * stride_right_j + \
            offs_k[None, :] * stride_right_k + \
            head_off
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0)

        # transpose right so it becomes K×BLOCK_N
        right_T = tl.trans(right)        # shape (BLOCK_K, BLOCK_N)

        # matrix‑multiply the two panels
        acc += tl.dot(left, right_T)    # (BLOCK_M, BLOCK_N)

    # write back the result (cast back to the input dtype, here fp16)
    out_ptrs = out_ptr + \
        batch_id * stride_out_b + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j + \
        head_off
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Execute the Triton kernel for left/right tensors of shape
    [B, N, N, H] (dtype torch.float16). Returns a tensor of the same shape.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    # ensure contiguous for predictable strides
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # strides in elements
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

            triton.cdiv(N, BLOCK_N),
            B * H)

        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

"""
Outgoing TriMul implementation.

This kernel performs the heavy O(N³·H) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
by treating each head `d` as an independent matrix multiply
    left_slice @ right_sliceᵀ   (N×N @ N×N → N×N)
and tiling the N×N output with BLOCK_M×BLOCK_N tiles.
All other operations (LayerNorm, Linear projections, gates) are handled in PyTorch.
The Triton kernel uses fp16 inputs with fp32 accumulation for speed and
numerical stability.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched per‑head GEMM: out = left @ rightᵀ  (shape B×N×N×H)."""
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile column
    pid_z = tl.program_id(2)   # combined batch‑head index

    # Decode batch and head from linear pid_z
    batch_id = pid_z // H
    head_id = pid_z % H

    # Offsets in the N dimension
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base offsets for the selected batch / head
    batch_off = batch_id * stride_left_b
    head_off   = head_id * stride_left_h

    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis k
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a BLOCK_M×BLOCK_K tile from left (i, k)
        left_ptrs = left_ptr + \
            batch_off + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k + \
            head_off
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # Load a BLOCK_N×BLOCK_K tile from right (j, k)
        right_ptrs = right_ptr + \
            batch_off + \
            offs_j[:, None] * stride_right_j + \
            offs_k[None, :] * stride_right_k + \
            head_off
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0)

        # Transpose right to K×BLOCK_N and perform dot
        right_T = tl.trans(right)          # (BLOCK_K, BLOCK_N)
        acc += tl.dot(left, right_T)       # (BLOCK_M, BLOCK_N)

    # Store the fp16 result back to global memory
    out_ptrs = out_ptr + \
        batch_id * stride_out_b + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j + \
        head_off
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Run the Triton kernel on half‑precision tensors of shape [B, N, N, H].
    Returns a tensor of the same shape (fp16).
    """
    assert left.shape == right.shape, "left and right must have identical shapes"
    B, N, _, H = left.shape
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # Strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – tuned for typical B200/A100 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M),    # rows
            triton.cdiv(N, BLOCK_N),    # cols
            B * H)                      # batch × head

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out
