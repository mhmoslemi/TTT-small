    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid_n = tl.program_id(1) # col block
    pid_bd = tl.program_id(2) # combined batch and d
    b = pid_bd // D
    d = pid_bd % D

"""
TriMul (outgoing) implementation using Triton.

The module computes, for each batch b and hidden channel d,
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right are obtained from a layer‑normed input tensor by
linear projections, an optional mask and sigmoid gates.
The heavy N×N reduction over the summation index k is performed by a
custom Triton kernel that computes a batched GEMM for each (b, d)
pair:
    left[b,…,d] @ right[b,…,d].T   (shape N×N).

All other operations (LayerNorm, linear projections, gating,
final LayerNorm and output projection) are done with PyTorch for
simplicity and numerical stability. The kernel works in fp16
(half) for bandwidth while accumulating in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,          # *fp16  [B, N, N, D]
    right_ptr,         # *fp16  [B, N, N, D]
    out_ptr,           # *fp32  [B, N, N, D]
    B, N, D,           # runtime sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Program IDs
    pid_i = tl.program_id(0)               # block row (i)
    pid_j = tl.program_id(1)               # block col (j)
    pid_bd = tl.program_id(2)              # combined batch & hidden dim

    # Decode batch and hidden dimension
    b = pid_bd // D
    d = pid_bd % D

    # Offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------ load left tile (M x K) ------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + d * stride_left_d
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # ------------ load right tile (K x N) ------------
        # right is indexed as (b, j, k, d); we need (K, N) layout for dot
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
            + d * stride_right_d
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # Dot product accumulates in fp32
        acc += tl.dot(left, right)

    # ------------ store result (M x N) ------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + d * stride_out_d
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (float or bool)
            - weights: dict of weight tensors
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------
    # 1. Input layer‑norm
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (float32)
    # -----------------------------------------------------------------
    left_proj = F.linear(x_norm, weights["left_proj.weight"], bias=None)   # [B,N,N,hidden_dim]
    right_proj = F.linear(x_norm, weights["right_proj.weight"], bias=None)

    # -----------------------------------------------------------------
    # 3. Apply mask (if provided)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.to(left_proj.dtype).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_
        right_proj = right_proj * mask_
    # if mask is None, skip (equivalent to all‑ones)

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"], bias=None))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"], bias=None))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"], bias=None))

    left = left_proj * left_gate    # [B,N,N,hidden_dim]
    right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5. Triton kernel: out = Σ_k left[...,k] * right[...,k]^T
    # -----------------------------------------------------------------
    # Cast to half for the kernel (memory and bandwidth)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (in element units)
    slb, sli, slk, sld = left_fp16.stride()
    srb, srj, srk, srd = right_fp16.stride()
    sob, soi, soj, sod = out.stride()

    # Blocking parameters (tuned for A100/B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # blocks over i
        (N + BLOCK_N - 1) // BLOCK_N,          # blocks over j
        B * hidden_dim,                        # one program per (batch, hidden)
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden_dim,
        slb,
        sli,
        slk,
        sld,
        srb,
        srj,
        srk,
        srd,
        sob,
        soi,
        soj,
        sod,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 6. Output layer‑norm, gating, final projection
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_scaled = out_norm * out_gate
    output = F.linear(out_scaled, weights["to_out.weight"], bias=None)   # [B,N,N,dim]

    return output
