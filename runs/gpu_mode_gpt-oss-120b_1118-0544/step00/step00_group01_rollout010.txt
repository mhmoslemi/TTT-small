    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm (input)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if present
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)  # broadcast to hidden dim
        left_proj = left_proj * mask_
        right_proj = right_proj * mask_

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left_proj = left_proj * left_gate
    right_proj = right_proj * right_gate

    # cast to fp16 for kernel
    left_fp16 = left_proj.to(torch.float16).contiguous()
    right_fp16 = right_proj.to(torch.float16).contiguous()
    B, N, _, _ = left_fp16.shape

    # allocate output fp32
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

    # Get strides
    stride_a_batch, stride_a_i, stride_a_k, stride_a_d = left_fp16.stride()
    stride_b_batch, stride_b_j, stride_b_k, stride_b_d = right_fp16.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_d = out.stride()

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_m_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_n_blocks = (N + BLOCK_N - 1) // BLOCK_N
    tiles_per_mat = num_m_blocks * num_n_blocks
    total_mats = B * hidden_dim
    grid = (total_mats * tiles_per_mat,)

    tri_mul_kernel[grid](

        B, N, hidden_dim,
        stride_a_batch, stride_a_i, stride_a_k, stride_a_d,
        stride_b_batch, stride_b_j, stride_b_k, stride_b_d,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The kernel computes, for each batch `b` and hidden dimension `d`,
the matrix product:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
which is equivalent to (left[..., d] @ right[..., d].T).
All other operations (LayerNorms, linear projections, gating) are done
with PyTorch; only the O(N³) reduction is fused in Triton for speed.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr,                # * left  (B,N,N,HD)  fp16
    b_ptr,                # * right (B,N,N,HD)  fp16
    c_ptr,                # * out   (B,N,N,HD)  fp32
    B, N, HD,                     # runtime sizes
    stride_a_batch, stride_a_i, stride_a_k, stride_a_d,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_d,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of tiles per matrix
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    tiles_per_mat = num_m * num_n

    # Which (batch, hidden) matrix and which tile inside it?
    mat_id = pid // tiles_per_mat          # 0 .. B*HD-1
    tile_id = pid % tiles_per_mat

    d = mat_id % HD                       # hidden index
    b = mat_id // HD                      # batch index

    m_block = tile_id // num_n
    n_block = tile_id % num_n

    i0 = m_block * BLOCK_M
    j0 = n_block * BLOCK_N

    i_idx = i0 + tl.arange(0, BLOCK_M)
    j_idx = j0 + tl.arange(0, BLOCK_N)

    i_mask = i_idx < N
    j_mask = j_idx < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K
        k_idx = k0 + tl.arange(0, BLOCK_K)
        k_mask = k_idx < N

        # A block: left[b, i, k, d]   shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + b * stride_a_batch
            + d * stride_a_d
            + i_idx[:, None] * stride_a_i
            + k_idx[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=tl.float16(0.0))

        # B block transposed: right[b, j, k, d] -> we need (k, j) shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            b_ptr
            + b * stride_b_batch
            + d * stride_b_d
            + k_idx[:, None] * stride_b_k
            + j_idx[None, :] * stride_b_j
        )
        b = tl.load(b_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=tl.float16(0.0))

        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Store result
    c_ptrs = (
        c_ptr
        + b * stride_c_batch
        + d * stride_c_d
        + i_idx[:, None] * stride_c_i
        + j_idx[None, :] * stride_c_j
    )
    tl.store(c_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


def custom_kernel(data):
    """
    Executes the TriMul forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (or None)
        - weights      : dict of weight tensors
        - config       : dict containing at least 'dim', 'hidden_dim', 'nomask'

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (dtype float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- LayerNorm on input ----------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- Linear projections ----------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------- Mask (if any) ----------
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)  # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ---------- Gating ----------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare for Triton kernel ----------
    # Cast to half-precision for the heavy O(N³) work
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape

    # Output buffer (float32, as required by subsequent LayerNorm)
    out = torch.empty(
        (B, N, N, hidden_dim), dtype=torch.float32, device=device
    )

    # Strides (in elements, not bytes)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_d = left_fp16.stride()
    stride_b_batch, stride_b_j, stride_b_k, stride_b_d = right_fp16.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_d = out.stride()

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_m_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_n_blocks = (N + BLOCK_N - 1) // BLOCK_N
    tiles_per_mat = num_m_blocks * num_n_blocks
    total_mats = B * hidden_dim
    grid = (total_mats * tiles_per_mat,)

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden_dim,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_a_d,
        stride_b_batch,
        stride_b_j,
        stride_b_k,
        stride_b_d,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        stride_c_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Post‑processing ----------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_norm = out_norm * out_gate

    # Final linear projection back to original dimension
    output = torch.nn.functional.linear(
        out_norm, weights["to_out.weight"]
    )  # shape [B,N,N,dim]

    return output
