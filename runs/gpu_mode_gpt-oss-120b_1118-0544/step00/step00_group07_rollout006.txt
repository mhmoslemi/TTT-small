def layer_norm(x, weight, bias, eps=1e-5):
    # compute mean and variance across last dim
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_hat = (x - mean) * torch.rsqrt(var + eps)
    if weight is not None:
        x_hat = x_hat * weight
    if bias is not None:
        x_hat = x_hat + bias
    return x_hat

"""
Custom Triton kernel for the "outgoing" Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last dimension.
2. Linear projections (no bias) to a hidden dimension `H`:
      left  = x @ left_proj_weight.T
      right = x @ right_proj_weight.T
3. Apply optional binary mask (broadcast on the hidden dim).
4. Compute gating values (sigmoid of linear maps) and multiply:
      left  = left  * sigmoid(x @ left_gate_weight.T)
      right = right * sigmoid(x @ right_gate_weight.T)
5. Core O(N³) operation:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM using a Triton kernel:
      out = left @ rightᵀ   (per hidden channel)
6. Layer‑norm over the hidden dimension, apply an output gate,
   and a final linear map back to the original dimension.
The kernel only handles step 5; all other steps use PyTorch for
clarity. The kernel works for any batch size `B∈{1,2}`,
sequence length `N`, and hidden dimension `H`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def triton_tri_mul_kernel(
    left_ptr,                     # ptr to left [B,N,N,H] (float16)
    right_ptr,                    # ptr to right [B,N,N,H] (float16)
    out_ptr,                      # ptr to output [B,N,N,H] (float32)
    B, H,                         # runtime batch size, hidden dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    SEQ_LEN: tl.constexpr,        # N (compile‑time for loop unrolling)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row  (i)
    pid_n = tl.program_id(1)          # tile col  (j)
    pid_z = tl.program_id(2)          # combined batch & hidden index

    # decode batch and hidden indices
    b = pid_z // H
    h = pid_z % H

    # offsets for the output tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < SEQ_LEN
    mask_j = offs_j < SEQ_LEN

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, SEQ_LEN, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < SEQ_LEN

        # left block:   [i, k]
        a_ptrs = (left_ptr
                  + b * stride_left_b
                  + offs_i[:, None] * stride_left_i
                  + offs_k[None, :] * stride_left_k
                  + h * stride_left_h)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # right block transposed: we need right[k, j] (original right[j,k])
        b_ptrs = (right_ptr
                  + b * stride_right_b
                  + offs_k[:, None] * stride_right_k
                  + offs_j[None, :] * stride_right_j
                  + h * stride_right_h)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # matrix multiplication (accumulates in float32)
        acc += tl.dot(a, b)

    # store the result
    out_ptrs = (out_ptr
                + b * stride_out_b
                + offs_i[:, None] * stride_out_i
                + offs_j[None, :] * stride_out_j
                + h * stride_out_h)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def _layer_norm(x, weight, bias, eps=1e-5):
    """Layer‑norm over the last dimension (supports weight/bias)."""
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    if weight is not None:
        x_norm = x_norm * weight
    if bias is not None:
        x_norm = x_norm + bias
    return x_norm


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    cubic reduction step.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------------- LayerNorm ----------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = _layer_norm(input_tensor, norm_weight, norm_bias)  # (B,N,N,dim)

    # ---------------- Linear projections ----------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    # ---------------- Mask (if present) ----------------
    if not nomask and mask_tensor is not None:
        # mask shape [B,N,N] -> [B,N,N,1]
        mask = mask_tensor.to(dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ---------------- Gates ----------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ---------------- Triton cubic reduction ----------------
    # Cast to half for the kernel (good memory bandwidth)
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    B, N, _, H = left_h.shape

    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # strides (in elements) for contiguous tensors
    s_left_b, s_left_i, s_left_k, s_left_h = left_h.stride()
    s_right_b, s_right_j, s_right_k, s_right_h = right_h.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()

    # Tile sizes (tuned for typical B200 memory hierarchy)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    triton_tri_mul_kernel[grid](
        left_h,
        right_h,
        out,
        B,
        H,
        s_left_b, s_left_i, s_left_k, s_left_h,
        s_right_b, s_right_j, s_right_k, s_right_h,
        s_out_b, s_out_i, s_out_j, s_out_h,
        SEQ_LEN=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------------- Output LayerNorm and gating ----------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = _layer_norm(out, to_out_norm_w, to_out_norm_b)

    out = out * out_gate  # element‑wise gate

    # ---------------- Final linear projection ----------------
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)  # (B,N,N,dim)

    return out
```
