    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)
    # Compute batch idx and hidden idx
    b = pid_hb // H
    h = pid_hb % H

    # Offsets
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # Create range vectors
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Mask for out-of-range
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # Compute pointers
    # left pointer: left[b, i, k, h]
    left_ptrs = left_ptr + b * stride_left_b + (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k) + h * stride_left_h
    # right pointer: right[b, j, k, h] but we want rows k, cols j for dot
    # (k, j) ordering: (offs_k[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_i) + h * stride_right_h
    right_ptrs = right_ptr + b * stride_right_b + (offs_k[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_i) + h * stride_right_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k in blocks
    for start_k in range(0, N, BLOCK_K):
        # compute block k offsets
        cur_k = start_k + offs_k
        cur_mask_k = cur_k < N

        # recompute pointers offset for each block: we can increment pointers each iteration rather than recomputing
        # We'll use pointer increments: left_tile_ptrs = left_ptr + start_k * stride_left_k maybe

def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)

    b = pid_hb // H
    h = pid_hb % H

    # offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    # mask for bounds
    mask_i = i < N
    mask_j = j < N

    # calculate base pointers without k
    left_base = left_ptr + b * stride_left_b + (i[:, None] * stride_left_i) + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + (j[None, :] * stride_right_i) + h * stride_right_h

    num_k = tl.cdiv(N, BLOCK_K)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for _ in range(num_k):

        left_tile = tl.load(left_base + (k[None, :] * stride_left_k), mask=mask_i[:, None] & (k[None, :] < N), other=0.0).to(tl.float32)

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)

    b = pid_hb // H
    h = pid_hb % H

    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)

    for kk in range(num_k):
        k_start = kk * BLOCK_K
        cur_k = k_start + k
        mask_k = cur_k < N

        left_offset = b * stride_l_b + (i[:, None] * stride_l_i) + (cur_k[None, :] * stride_l_k) + h * stride_l_h
        right_offset = b * stride_r_b + (cur_k[:, None] * stride_r_k) + (j[None, :] * stride_r_i) + h * stride_r_h

        left_tile = tl.load(left_ptr + left_offset,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)
        right_tile = tl.load(right_ptr + right_offset,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # Cast to fp32 if needed
        left_tile = left_tile.to(tl.float32)
        right_tile = right_tile.to(tl.float32)

        acc += tl.dot(left_tile, right_tile)

    out_offset = b * stride_o_b + (i[:, None] * stride_o_i) + (j[None, :] * stride_o_j) + h * stride_o_h

"""
TriMul (outgoing) implementation with Triton.

The heavy part of TriMul is the tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication for each hidden channel d.
We fuse this contraction into a custom Triton kernel that tiles the
(i, j) output dimensions and iterates over k in blocks. The rest
(layer‑norms, linear projections, sigmoid gates, final projection)
is performed with PyTorch which is already highly optimized.

The kernel processes one hidden channel (d) at a time; the third
program‑axis enumerates `batch * hidden_dim`.  Each program computes a
tile of size BLOCK_M × BLOCK_N of the output slice and accumulates the
dot‑product over K‑tiles.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute the outgoing TriMul contraction.

    Arguments
    ----------
    left_ptr, right_ptr, out_ptr : pointers to [B,N,N,H] tensors (float32)
    B, N, H                     : batch size, sequence length, hidden dim
    stride_*                    : strides of the tensors in element units
    BLOCK_M, BLOCK_N, BLOCK_K   : compile‑time tile sizes
    """
    # Program IDs for the (i, j, batch*hidden) grid
    pid_i = tl.program_id(0)          # tile row (i)
    pid_j = tl.program_id(1)          # tile col (j)
    pid_hb = tl.program_id(2)         # combined batch & hidden index

    # De‑compose batch and hidden channel
    b = pid_hb // H
    h = pid_hb % H

    # Tile offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    # Bounds masks for i and j
    mask_i = i < N
    mask_j = j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    # Main K‑loop
    for kk in range(num_k):
        k_start = kk * BLOCK_K
        cur_k = k_start + k                     # absolute k indices
        mask_k = cur_k < N

        # Offsets for left[b,i,cur_k,h]   shape (BLOCK_M, BLOCK_K)
        off_left = (
            b * stride_l_b
            + (i[:, None] * stride_l_i)
            + (cur_k[None, :] * stride_l_k)
            + h * stride_l_h
        )
        # Offsets for right[b,j,cur_k,h]  shape (BLOCK_K, BLOCK_N)
        off_right = (
            b * stride_r_b
            + (cur_k[:, None] * stride_r_k)
            + (j[None, :] * stride_r_i)
            + h * stride_r_h
        )

        left_tile = tl.load(left_ptr + off_left,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)
        right_tile = tl.load(right_ptr + off_right,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # Ensure fp32 accumulation
        left_tile = left_tile.to(tl.float32)
        right_tile = right_tile.to(tl.float32)

        acc += tl.dot(left_tile, right_tile)

    # Store the result tile to out[b,i,j,h]
    off_out = (
        b * stride_o_b
        + (i[:, None] * stride_o_i)
        + (j[None, :] * stride_o_j)
        + h * stride_o_h
    )
    tl.store(out_ptr + off_out, acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (float or bool)
        - weights      : dict with all linear / layer‑norm parameters
        - config       : dict containing "dim", "hidden_dim", "nomask" ...

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -------------------- 1. LayerNorm on input --------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # -------------------- 2. Linear projections --------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # -------------------- 3. Optional masking --------------------
    # mask: [B,N,N] -> [B,N,N,1] for broadcasting
    if config.get("nomask", False):
        mask_tensor = torch.ones_like(mask, dtype=dtype, device=device)
    else:
        mask_tensor = mask.to(dtype)
    mask_tensor = mask_tensor.unsqueeze(-1)      # [B,N,N,1]

    left = left * mask_tensor
    right = right * mask_tensor

    # -------------------- 4. Gating --------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -------------------- 5. Triton contraction --------------------
    B, N, _, _ = left.shape
    out = torch.empty_like(left)   # allocate [B,N,N,hidden_dim]

    # Strides in elements (PyTorch gives strides in element counts)
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_i, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    # Tile sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        s_l_b, s_l_i, s_l_k, s_l_h,
        s_r_b, s_r_i, s_r_k, s_r_h,
        s_o_b, s_o_i, s_o_j, s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------- 6. Post‑processing ----------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm over hidden dimension
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # Apply output gate and final linear projection
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # shape [B,N,N,dim]

    return out
