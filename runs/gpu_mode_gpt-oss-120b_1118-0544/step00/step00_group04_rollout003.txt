@triton.jit
def batched_gemm_kernel(A, B, C,
                        stride_ab_batch, stride_a_m, stride_a_k,
                        stride_bb_batch, stride_b_k, stride_b_n,
                        stride_cb_batch, stride_c_m, stride_c_n,
                        M, N, K,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # coordinate ranges
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers offset for batch
    a_ptr = A + pid_batch * stride_ab_batch
    b_ptr = B + pid_batch * stride_bb_batch
    c_ptr = C + pid_batch * stride_cb_batch

    # Masks for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # compute offsets for K dimension
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # mask for K
        mask_k = offs_k < K

        # load a block
        a = tl.load(a_ptr + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k),

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Unpack weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm
    # Using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # (B,N,N,hidden_dim)

"""
TriMul forward kernel (outgoing version).

The heavy part of the TriMul operation is the einsum:
    out[..., i, j, d] = sum_k left[..., i, k, d] * right[..., j, k, d]

For a fixed hidden dimension `d` this is a matrix multiplication:
    out_d = left_d @ right_d.T   (shape N×N)

We implement this batched matrix multiplication with a custom Triton kernel.
All other operations (LayerNorm, Linear projections, gating) are performed
with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def batched_gemm_kernel(
    A, B, C,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # 3‑D grid: (batch, block_m, block_n)
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the current batch
    a_batch = A + pid_batch * stride_a_batch
    b_batch = B + pid_batch * stride_b_batch
    c_batch = C + pid_batch * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block [M, K]
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load B block transposed -> shape [K, N]
        b = tl.load(
            b_batch + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # Accumulate
        acc += tl.dot(a, b)

    # Write the result (convert back to FP16)
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n),
        c,
        mask=mask_m[:, None] & mask_n[None, :]
    )

def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask: torch.Tensor, shape [B, N, N] (may be ignored if nomask)
            - weights: dict of torch.Tensors containing all Linear/LayerNorm params
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----- unpack weights -----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ----- Stage 0: first LayerNorm -----
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias
    )

    # ----- Stage 1: projections -----
    left = F.linear(x_norm, left_proj_weight)          # (B,N,N,hidden_dim)
    right = F.linear(x_norm, right_proj_weight)

    # ----- Stage 2: gates -----
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Apply mask if needed (mask is 0/1)
    if not nomask:
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ----- Convert to FP16 for the heavy GEMM -----
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # Reshape to (B*hidden_dim, N, N) in a contiguous layout
    B, N, _, _ = left_f16.shape
    left_trans = left_f16.permute(0, 3, 1, 2).contiguous()
    right_trans = right_f16.permute(0, 3, 1, 2).contiguous()
    total_batches = B * hidden_dim
    left_mat = left_trans.view(total_batches, N, N)
    right_mat = right_trans.view(total_batches, N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # Strides (converted to Python ints for Triton)
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_n, stride_b_k = right_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    # Block sizes (tuned for A100/B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid dimensions
    grid = (
        total_batches,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch the Triton kernel: compute left @ right^T for each hidden batch
    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        N, N, N,                     # M, N, K (all equal to sequence length)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out_fp16 = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out_fp16.to(torch.float32)

    # ----- Stage 3: second LayerNorm -----
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias
    )

    # Apply output gate
    out = out * out_gate

    # ----- Final linear projection -----
    out = F.linear(out, to_out_weight)   # bias is omitted (bias=False)

    return out
