@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)  # b * H
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch and hidden indices
    b = pid0 // H
    h = pid0 % H

    # offsets within block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid ranges
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: [M, K]
        left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k

        # right: [K, N]

        acc += tl.dot(left, right)  # result shape (BLOCK_M, BLOCK_N)

"""
TriMul (outgoing) implementation using Triton.

The core computation is:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, k, j, h]

where `left` and `right` are linear projections of the normalized input,
scaled by learned gates and an optional pairwise mask.
The heavy O(N³·H) reduction is performed by a custom Triton kernel that
tiles the output (i, j) dimensions and accumulates over the k dimension.
All other operations (LayerNorms, linear projections, gating) are done in
PyTorch.  The kernel works with FP16 inputs and accumulates in FP32
for numerical stability, then writes the result back as FP16.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ------------------------------------------------------------
# Triton kernel for the outgoing triangular multiplicative update
# ------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,          # const half*   [B, N, N, H]   (i, k)
    right_ptr,         # const half*   [B, N, N, H]   (k, j)
    out_ptr,           #       half*   [B, N, N, H]   (i, j)
    B, N, H,           # problem dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)          # flattened batch * hidden index
    pid_m = tl.program_id(1)         # i tile
    pid_n = tl.program_id(2)         # j tile

    # Decompose pid0 into batch index and hidden index
    b = pid0 // H
    h = pid0 % H

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in BLOCK_K sized chunks
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------- load left[b, i, k, h] -------------------
        # shape (BLOCK_M, BLOCK_K)
        left_off = (
            b * stride_left_b + h * stride_left_h +
            offs_m[:, None] * stride_left_i +
            offs_k[None, :] * stride_left_k
        )
        left = tl.load(left_ptr + left_off,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ------------------- load right[b, k, j, h] ------------------
        # shape (BLOCK_K, BLOCK_N)
        right_off = (
            b * stride_right_b + h * stride_right_h +
            offs_k[:, None] * stride_right_i +
            offs_n[None, :] * stride_right_k
        )
        right = tl.load(right_ptr + right_off,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # -------------------- dot product accumulate -----------------
        acc += tl.dot(left, right)        # (BLOCK_M, BLOCK_N)

    # Write result back as FP16
    out_off = (
        b * stride_out_b + h * stride_out_h +
        offs_m[:, None] * stride_out_i +
        offs_n[None, :] * stride_out_k
    )
    tl.store(out_ptr + out_off,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ------------------------------------------------------------
# Entrypoint required by the evaluator
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors containing all learnable params
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    B, N, _, _ = input_tensor.shape  # shape: [B, N, N, dim]

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # same shape

    # ------------------------------------------------------------------
    # Apply optional mask (broadcast over the channel dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1]
        mask = mask.to(dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating mechanisms
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy O(N^3) kernel
    # ------------------------------------------------------------------
    left_half = left.to(torch.float16).contiguous()
    right_half = right.to(torch.float16).contiguous()
    out_half = torch.empty_like(left_half)  # placeholder for results

    # ------------------------------------------------------------------
    # Prepare strides (in elements) for the Triton kernel
    # ------------------------------------------------------------------
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_half.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_half.stride()
    stride_out_b, stride_out_i, stride_out_k, stride_out_h = out_half.stride()

    # ------------------------------------------------------------------
    # Tile sizes (tuned for typical B200 GPUs)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch the Triton kernel
    tri_mul_outgoing_kernel[grid](
        left_half,
        right_half,
        out_half,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_k,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Remaining post‑processing in float32
    # ------------------------------------------------------------------
    out_fp32 = out_half.to(torch.float32)          # [B, N, N, H] fp32

    # LayerNorm on the hidden dimension
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )                                            # [B, N, N, H] fp32

    # Apply the output gate
    out_norm = out_norm * out_gate                 # broadcast over H

    # Final linear projection back to dim
    to_out_weight = weights["to_out.weight"]       # shape [dim, hidden_dim]
    output = F.linear(out_norm, to_out_weight)    # [B, N, N, dim] fp32

    return output
