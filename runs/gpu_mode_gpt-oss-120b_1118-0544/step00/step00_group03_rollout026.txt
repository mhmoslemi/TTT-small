def custom_kernel(data):
    input, mask, weights, config = data
    # ensure tensors on same device (should be given)
    B,N,N,_ = input.shape
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    # optionally config may contain 'nomask' boolean; if True ignore mask.
    nomask = config.get('nomask', False)
    if nomask:
        mask_tensor = torch.ones((B,N,N), dtype=input.dtype, device=input.device)
    else:
        mask_tensor = mask
    # 1. layernorm
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # x = (x - mean)/sqrt(var+eps) * weight + bias
    mean = input.mean(dim=-1, keepdim=True)
    var = ((input - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (input - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # 2. Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # 3. Apply mask
    mask_f = mask_tensor.unsqueeze(-1).type_as(x_norm)   # (B,N,N,1)
    left = left * mask_f
    right = right * mask_f

    # 4. Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # 5. Compute einsum using Triton kernel
    out = torch.empty((B,N,N,hidden_dim), dtype=input.dtype, device=input.device)
    # we need to call kernel with all tensors contiguous
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()
    # compute strides
    stride_a_b = left.stride(0)
    stride_a_i = left.stride(1)
    stride_a_k = left.stride(2)
    stride_a_d = left.stride(3)
    stride_b_b = right.stride(0)
    stride_b_i = right.stride(1)   # ~same as left.stride(1)
    stride_b_k = right.stride(2)
    stride_b_d = right.stride(3)
    stride_c_b = out.stride(0)
    stride_c_i = out.stride(1)
    stride_c_j = out.stride(2)
    stride_c_d = out.stride(3)
    # For B_T (right transposed), we need to supply stride for "row" (k) as stride_b_k and for "col" (j) as stride_b_i.
    # We'll pass them as stride_bk (notes)
    # We'll also have stride for hidden dimension d (stride_ad/d).
    # but we already included d offset in base pointer.

    # Determine BLOCK sizes: use typical 128x128 tiling maybe, and BLOCK_K 32.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # K tiling
    # Compute grid sizes
    grid = ( (N + BLOCK_M - 1)//BLOCK_M,
             (N + BLOCK_N - 1)//BLOCK_N,
             B * hidden_dim )
    # Launch kernel
    tri_mul_kernel[grid](
        left, right, out,
        N,
        stride_a_b, stride_a_i, stride_a_k, stride_a_d,
        stride_b_b, stride_b_i, stride_b_k, stride_b_d,
        stride_c_b, stride_c_i, stride_c_j, stride_c_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # 6. Apply to_out_norm: LayerNorm on hidden_dim per location (i,j per batch)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # compute mean & var across last dim hidden_dim for out
    out_mean = out.mean(dim=-1, keepdim=True)
    out_var = ((out - out_mean) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - out_mean) / torch.sqrt(out_var + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # Apply out_gate (elementwise multiplication)
    out_norm = out_norm * out_gate

    # Final linear to dim
    final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    # final shape (B,N,N,dim)
    return final

"""
TriMul “outgoing” implementation using Triton for the cubic einsum.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension (dim).
2. Apply two linear projections (left/right) and three gating linear layers.
3. Mask the projected tensors if a mask is supplied.
4. Compute the core einsum
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is equivalent to a batched matrix multiplication
        out_slice = left_slice @ right_slice.T
   for each (batch, hidden) pair.  The multiplication is performed by a
   custom Triton kernel that tiles the N×N matrices (BLOCK_M×BLOCK_N) and
   accumulates over K‑tiles.
5. Layer‑norm the result in the hidden dimension, apply the output gate,
   and a final linear projection back to the original dimension.

The kernel processes `batch * hidden_dim` independent GEMMs in parallel,
using program ID axis‑2 for the combined batch‑hidden index.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel for the cubic einsum (the bottleneck)
# ----------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to left, right, output
    N,                                 # sequence length
    BATCH, HIDDEN,                     # batch size and hidden dimension
    stride_a_batch, stride_ai, stride_ak, stride_ad,
    stride_b_batch, stride_bk, stride_bi, stride_bd,
    stride_c_batch, stride_ci, stride_cj, stride_cd,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # block row (i)
    pid_n = tl.program_id(1)  # block col (j)
    pid_z = tl.program_id(2)  # combined (batch, hidden)

    # Determine which batch and hidden slice this program works on
    batch_idx = pid_z // HIDDEN
    hidden_idx = pid_z % HIDDEN

    # Base pointers for this (batch, hidden) slice
    a_base = a_ptr + batch_idx * stride_a_batch + hidden_idx * stride_ad
    b_base = b_ptr + batch_idx * stride_b_batch + hidden_idx * stride_bd
    c_base = c_ptr + batch_idx * stride_c_batch + hidden_idx * stride_cd

    # Offsets for the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Pointers for the current K‑tile
        a_ptrs = a_base + offs_m[:, None] * stride_ai + k_offset[None, :] * stride_ak
        b_ptrs = b_base + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bi

        # Load tiles (zero‑pad out‑of‑bounds)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float32)

        # Accumulate partial product
        acc += tl.dot(a, b)

    # Write result
    c_ptrs = c_base + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------------- #
# Public entry point – the forward pass of the “outgoing” TriMul operator
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    # --------------------------------------------------------------------- #
    # Config and weight extraction
    # --------------------------------------------------------------------- #
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape
    nomask = config.get("nomask", False)

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1. Input layer‑norm
    # --------------------------------------------------------------------- #
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # --------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left = F.linear(x, left_proj_weight)    # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------------- #
    # 3. Apply mask (if any)
    # --------------------------------------------------------------------- #
    if nomask or mask_tensor is None:
        mask = torch.ones((batch_size, seq_len, seq_len),
                          dtype=dtype,
                          device=device)
    else:
        mask = mask_tensor
    mask = mask.unsqueeze(-1).type_as(x)  # [B,N,N,1] -> broadcast over hidden

    left = left * mask
    right = right * mask

    # --------------------------------------------------------------------- #
    # 4. Gating (sigmoid of linear projections)
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Cubic einsum via Triton – computes out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------------- #
    out = torch.empty((batch_size, seq_len, seq_len, hidden_dim),
                      dtype=dtype,
                      device=device)

    # Ensure contiguous layout for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # Strides (in elements)
    stride_a_batch, stride_ai, stride_ak, stride_ad = left.stride()
    stride_b_batch, stride_bi, stride_bk, stride_bd = right.stride()
    stride_c_batch, stride_ci, stride_cj, stride_cd = out.stride()

    # Triton launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = ( (seq_len + BLOCK_M - 1) // BLOCK_M,
             (seq_len + BLOCK_N - 1) // BLOCK_N,
             batch_size * hidden_dim )

    tri_mul_kernel[grid](
        left, right, out,
        seq_len,
        batch_size, hidden_dim,
        stride_a_batch, stride_ai, stride_ak, stride_ad,
        stride_b_batch, stride_bk, stride_bi, stride_bd,
        stride_c_batch, stride_ci, stride_cj, stride_cd,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # Synchronize to make sure the kernel has finished before next ops
    torch.cuda.synchronize()

    # --------------------------------------------------------------------- #
    # 6. Output layer‑norm, gating and final linear
    # --------------------------------------------------------------------- #
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate  # broadcast over hidden dimension

    output = F.linear(out, to_out_weight)   # back to [B,N,N,dim]

    return output
