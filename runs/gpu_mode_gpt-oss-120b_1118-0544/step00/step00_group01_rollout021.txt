def triton_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bi, stride_bk,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for rows/cols
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out-of-bound
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to beginning of the batch slice
    a_batch_ptr = a_ptr + pid_batch * stride_ah
    b_batch_ptr = b_ptr + pid_batch * stride_bh
    c_batch_ptr = c_ptr + pid_batch * stride_ch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block [BLOCK_M, BLOCK_K]
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B block [BLOCK_K, BLOCK_N]

        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

    # Write back to C

    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

TriMul forward (outgoing) implemented with a fused Triton GEMM.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,D]).
2. Project `x` with two linear layers (left/right) and apply element‑wise
   gating (sigmoid of three more linear layers).  A mask (if present) is

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to a batched matrix multiplication
        left_mat @ right_matᵀ
   where each batch corresponds to (b,h).  We launch a custom Triton kernel
   that computes this GEMM for all (B·hidden) matrices.
4. Apply a second layer‑norm, an output gate, and a final linear projection
   back to the original feature dimension.
All heavy arithmetic (the N³ contraction) is performed in the Triton kernel.
"""

import torch
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B   (A: (M,K), B: (K,N))
# -------------------------------------------------------------
@triton.jit
def _triton_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bi, stride_bk,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # index over (B * hidden)
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block col

    # Coordinates of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the beginning of the current batch slice
    a_batch_ptr = a_ptr + pid_batch * stride_ah
    b_batch_ptr = b_ptr + pid_batch * stride_bh
    c_batch_ptr = c_ptr + pid_batch * stride_ch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile [BLOCK_M, BLOCK_K]
        a = tl.load(
            a_batch_ptr + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile [BLOCK_K, BLOCK_N]
        b = tl.load(
            b_batch_ptr + offs_k[:, None] * stride_bi + offs_n[None, :] * stride_bk,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate the matrix product for this tile
        acc += tl.dot(a, b)

    # Store the result tile into C
    c = c_batch_ptr + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])


def _launch_triton_gemm(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor,
                        BLOCK_M=64, BLOCK_N=64, BLOCK_K=64):
    """
    Launch the Triton GEMM kernel for tensors:
        a: [B*H, M, K]
        b: [B*H, K, N]   (already transposed)
        c: [B*H, M, N]
    """
    assert a.is_contiguous() and b.is_contiguous() and c.is_contiguous()
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "Incompatible inner dimensions"
    grid = (batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _triton_gemm_kernel[grid](
        a, b, c,
        M, N, K,
        a.stride(0), a.stride(1), a.stride(2),
        b.stride(0), b.stride(1), b.stride(2),
        c.stride(0), c.stride(1), c.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )


# -------------------------------------------------------------
# Custom kernel entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, D]
        - mask         : torch.Tensor or None, shape [B, N, N]
        - weights      : dict of torch.Tensors with the model parameters
        - config       : dict containing at least "dim" and "hidden_dim"
                         (and optionally "nomask")

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (dtype float32).
    """
    # Unpack arguments
    x, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) Input layer‑norm
    # -----------------------------------------------------------------
    # mean / var over the last dimension (feature dim)
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional mask
    # -----------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        m = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left_proj = left_proj * m
        right_proj = right_proj * m

    # -----------------------------------------------------------------
    # 4) Gates (sigmoid of linear layers)
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left_proj * left_gate          # [B,N,N,hidden]
    right = right_proj * right_gate       # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 5) Core TriMul contraction via a fused Triton GEMM
    # -----------------------------------------------------------------
    # Move hidden dim next to batch and make tensors contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, hidden, N, N]

    B, H, N, _ = left_perm.shape
    # Flatten batch * hidden into a single GEMM batch dimension
    left_mat = left_perm.view(B * H, N, N)                # [B*H, M=N, K=N]
    right_mat = right_perm.view(B * H, N, N)              # [B*H, J=N, K=N]

    # Transpose the second matrix so that we compute A @ Bᵀ
    right_T = right_mat.transpose(1, 2).contiguous()     # [B*H, K=N, N=N]

    out_mat = torch.empty_like(left_mat)                  # [B*H, N, N]

    # Launch the Triton kernel
    _launch_triton_gemm(left_mat, right_T, out_mat,
                        BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)

    # Reshape back to [B, N, N, hidden]
    out_hidden = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6) Output layer‑norm, gating and final linear projection
    # -----------------------------------------------------------------
    mean_o = out_hidden.mean(dim=-1, keepdim=True)
    var_o = out_hidden.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_hidden - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    out_norm = out_norm * out_gate           # element‑wise gate

    # Final linear back to original dimension
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output.to(torch.float32)
