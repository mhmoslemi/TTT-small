def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # Compute normalized x
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # Compute left/right projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    # compute gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()
    # Apply mask
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask
    # Apply gates
    left = left * left_gate
    right = right * right_gate
    # Cast left/right to half for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    B, N, _, _ = left.shape
    # reshape to [B*hidden_dim, N, N] for kernel
    left_perm = left_fp16.permute(0,3,1,2).contiguous()  # [B, hidden_dim, N, N]
    right_perm = right_fp16.permute(0,3,1,2).contiguous()
    # flatten batch*hidden_dim
    total_batch = B * hidden_dim
    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_perm.view(total_batch, N, N)
    # allocate output
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)  # will hold hidden_dim dim result
    # compute matmul left @ right^T via kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    # compute grid
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m, num_blocks_n, total_batch)
    # compute strides
    a_ptr = left_flat
    b_ptr = right_flat
    c_ptr = out_flat
    # compute stride in elements (not bytes) using .stride()
    stride_aa = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)
    stride_bb = right_flat.stride(0)
    stride_bk = right_flat.stride(2)  # col stride (original)
    stride_bn = right_flat.stride(1)  # row stride (original)
    stride_cc = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)
    # launch kernel
    matmul_kernel[grid](
        a_ptr, b_ptr, c_ptr,
        N, N, N,
        stride_aa, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cc, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1)  # shape [B,N,N,hidden_dim]
    # Apply output norm and gate
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    # final projection to dim
    out = torch.nn.functional.linear(out, to_out_weight)
    return out

"""
TriMul forward kernel (outgoing version) implemented with Triton.

The operation:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where
    left  = (x_norm @ W_left)  * mask * σ(x_norm @ W_left_gate)
    right = (x_norm @ W_right) * mask * σ(x_norm @ W_right_gate)

The heavy O(N³) contraction is performed with a fused Triton batched matrix‑multiply
kernel (left @ rightᵀ) over the sequence dimension.  All other steps (LayerNorm,
Linear projections, sigmoid gates, output normalization, output gate and final
projection) are carried out with PyTorch for simplicity.

The kernel processes a batch×hidden_dim “matrix” of size N×N for each channel.
Each kernel launch works on a single (batch*hidden) slice and computes the
output tile‑wise with BLOCK_M×BLOCK_N tiles and a reduction over K in BLOCK_K
steps.  Inputs are provided in float16 (to reduce memory traffic) while the
accumulator is float32 for numeric stability.  The final result is returned as
float32.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix sizes (all = seq_len)
    stride_aa, stride_am, stride_ak,   # strides for a: batch, row (M), col (K)
    stride_bb, stride_bk, stride_bn,   # strides for b: batch, K (col), N (row) – b is treated as (K,N)
    stride_cc, stride_cm, stride_cn,   # strides for c: batch, row (M), col (N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)      # tile row
    pid_n = tl.program_id(1)      # tile col
    batch = tl.program_id(2)      # which (batch*hidden) slice

    # Compute offsets within the matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer to the start of the current batch slice
    a_batch_ptr = a_ptr + batch * stride_aa
    b_batch_ptr = b_ptr + batch * stride_bb
    c_batch_ptr = c_ptr + batch * stride_cc

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        k_range = k_offset + tl.arange(0, BLOCK_K)
        mask_k = k_range < K

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am
                                + k_range[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B block (transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch_ptr + (k_range[:, None] * stride_bk
                                + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # Write result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm
                            + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Load weights (all on the same device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (to hidden_dim)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid) – computed from the normalized input
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()  # later used after to_out_norm

    # ------------------------------------------------------------------
    # 4) Optional mask (mask may be None or all‑ones)
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(x_norm.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Cast to half‑precision for the heavy contraction
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape  # N is the sequence length

    # Rearrange to (batch*hidden, N, N) and make contiguous
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()
    total_batch = B * hidden_dim

    left_flat = left_perm.view(total_batch, N, N)   # [B*hidden, N, N]
    right_flat = right_perm.view(total_batch, N, N)

    # ------------------------------------------------------------------
    # 7) Allocate output buffer for the contracted tensor (float32)
    # ------------------------------------------------------------------
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 8) Triton kernel launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (num_blocks_m, num_blocks_n, total_batch)

    # Strides (in elements, not bytes)
    stride_aa = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bb = right_flat.stride(0)
    # For b we treat it as (K, N) where K = original column stride
    stride_bk = right_flat.stride(2)   # stride over K (original column)
    stride_bn = right_flat.stride(1)   # stride over N (original row)

    stride_cc = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # Launch the kernel
    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_aa, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cc, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9) Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 10) Output LayerNorm, gate and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
