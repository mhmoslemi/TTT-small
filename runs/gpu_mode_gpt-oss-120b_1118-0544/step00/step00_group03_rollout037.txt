```
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr
):
    pid_m = tl.program_id(0) # i tile
    pid_n = tl.program_id(1) # j tile
    pid_b = tl.program_id(2) # batch index
    pid_d = tl.program_id(3) # d block index

    # compute start offsets
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N
    d_start = pid_d * BLOCK_D

    # compute pointer base for this batch
    offs_b = pid_b * stride_left_b   # left and right same stride for batch

    # pointers for left and right base
    left_base = left_ptr + offs_b
    right_base = right_ptr + offs_b
    out_base = out_ptr + offs_b

    # create range for i, j, k, d offsets
    i = i_start + tl.arange(0, BLOCK_M)
    j = j_start + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)
    d = d_start + tl.arange(0, BLOCK_D)

    # mask out-of-bounds
    mask_i = i < N
    mask_j = j < N
    mask_d = d < D

    # initialize accumulator shape (BLOCK_M, BLOCK_N, BLOCK_D)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over K dimension
    for k_iter in range(0, N, BLOCK_K):
        cur_K = min(BLOCK_K, N - k_iter)
        # k indexes
        k = k_iter + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # Load left block for each d: shape (BLOCK_M, cur_K, cur_D) but we will load per d inside nested loop.
        # We'll compute for each d slice:
        for d_off in range(0, BLOCK_D, 1): # We'll load each d separately bottom maybe ineff.
            # get actual d index
            d_idx = d_start + d_off
            if d_idx >= D: continue   # bound check but we can rely on mask

            # compute pointer offsets
            left_ptrs = left_base + i[:,None] * stride_left_i + (k_iter + k[:cur_K]) * stride_left_k + d_idx * stride_left_h
            # left_ptrs shape (BLOCK_M, cur_K)
            left = tl.load(left_ptrs, mask=mask_i[:,None] & k_mask[None,:], other=0.0).to(tl.float32)

            # right block:
            right_ptrs = right_base + j[:,None] * stride_right_j + (k_iter + k[:cur_K]) * stride_right_k + d_idx * stride_right_h
            # Actually right needed to index (j, k) order: right[j,k,d] shape (BLOCK_N, cur_K).
            # but we intend to access right[j,k,d]; we need to sum over k. It's fine.
            right = tl.load(right_ptrs, mask=mask_j[:,None] & k_mask[None,:], other=0.0).to(tl.float32)

            # Multiply left and right (broadcast) -> we need outer product shape (BLOCK_M, BLOCK_N)

"""
TriMul outgoing kernel (AlphaFold3 style) implemented with Triton.

The heavy part of the TriMul forward pass is the Einstein summation:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

For each hidden dimension `d` this corresponds to a matrix multiplication
    left[b, :, :, d] @ right[b, :, :, d].T
with size (N, N) × (N, N).  This kernel performs that batched GEMM in
FP16 (accumulating in FP32) for all batches and hidden channels in a single
launch.  The surrounding layer‑norm, linear projections and gating are
performed in PyTorch for simplicity.

The kernel receives pointers to the left and right tensors sliced at a
specific batch and hidden channel (computed inside the kernel from a
combined program id) and writes the result to the output tensor.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the TriMul einsum (outgoing version)
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,                     # pointers
    M, N, K,                                          # matrix sizes (all = seq_len)
    stride_lb, stride_li, stride_lk, stride_lh,       # left strides: batch, i, k, hidden
    stride_rb, stride_rj, stride_rk, stride_rh,       # right strides: batch, j, k, hidden
    stride_ob, stride_oi, stride_oj, stride_oh,       # out strides: batch, i, j, hidden
    hidden_dim,                                       # hidden dimension (runtime)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)      # tile row (i)
    pid_n = tl.program_id(1)      # tile column (j)
    pid_z = tl.program_id(2)      # combined batch * hidden index

    # ------------------------------------------------------------------- #
    # Decode batch and hidden index from pid_z
    # ------------------------------------------------------------------- #
    b = pid_z // hidden_dim
    d = pid_z % hidden_dim

    # Base pointers for the selected batch and hidden channel
    left_base = left_ptr + b * stride_lb + d * stride_lh
    right_base = right_ptr + b * stride_rb + d * stride_rh
    out_base = out_ptr + b * stride_ob + d * stride_oh

    # ------------------------------------------------------------------- #
    # Tile offsets
    # ------------------------------------------------------------------- #
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Loop over K dimension (reduction)
    # ------------------------------------------------------------------- #
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_start = kb * BLOCK_K
        cur_k = k_start + tl.arange(0, BLOCK_K)          # absolute k indices
        mask_k = cur_k < K

        # Load a tile of left: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base \
                    + offs_i[:, None] * stride_li \
                    + cur_k[None, :] * stride_lk
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load a tile of right (transposed): shape (BLOCK_K, BLOCK_N)
        # right is indexed as (j, k) in memory, we need (k, j)
        right_ptrs = right_base \
                     + cur_k[:, None] * stride_rk \
                     + offs_j[None, :] * stride_rj
        b = tl.load(right_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # GEMM accumulate
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------- #
    # Write back result (convert to FP16)
    # ------------------------------------------------------------------- #
    out_ptrs = out_base \
               + offs_i[:, None] * stride_oi \
               + offs_j[None, :] * stride_oj
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------------------- #
# Entry point used by the evaluator
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul update.

    Args:
        data: tuple (input_tensor, mask_tensor, weights, config)
            input_tensor: [B, N, N, dim] (torch.float32)
            mask_tensor:  [B, N, N]  (torch.float32 or bool) – may be all‑ones
            weights: dict of model parameters (torch.float32)
            config: dict with keys "dim", "hidden_dim" (ints)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------- #
    # LayerNorm on the input (last dimension)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # Linear projections (no bias)
    # ------------------------------------------------------------------- #
    # left_proj : (B,N,N,hidden_dim)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # Mask handling
    # ------------------------------------------------------------------- #
    if mask_tensor is not None:
        # Ensure mask is float (0/1) and broadcast to last dim
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask
    # ------------------------------------------------------------------- #
    # Gating
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # Cast to FP16 before the heavy GEMM
    # ------------------------------------------------------------------- #
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim
    # Allocate output of the einsum (still FP16)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------- #
    # Gather strides for the kernel
    # ------------------------------------------------------------------- #
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # ------------------------------------------------------------------- #
    # Kernel launch configuration
    # ------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    # Launch the Triton kernel
    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        N,               # M
        N,               # N
        N,               # K
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------- #
    # Remaining post‑processing (still in PyTorch)
    # ------------------------------------------------------------------- #
    # Convert back to FP32 for the final LayerNorm and linear projection
    out = out.to(torch.float32)

    # to_out_norm (LayerNorm over hidden_dim)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply the output gate (broadcasted)
    out_gate = out_gate.to(out.dtype)          # ensure same dtype
    out = out * out_gate

    # Final linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"])

    return out
