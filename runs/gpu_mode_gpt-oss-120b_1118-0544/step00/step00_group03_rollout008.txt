    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_flat, left_proj_weight).view(B, N, N, H)
    right = torch.nn.functional.linear(x_flat, right_proj_weight).view(B, N, N, H)

    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_weight)).view(B, N, N, H)

"""
TriMul (outgoing) forward pass implementation.

The function `custom_kernel` takes a tuple
(data = (input_tensor, mask, weights, config)) and returns the processed tensor.
Key steps:
1) Layer‑norm over the last channel dimension.
2) Linear projections (left/right) and three sigmoidal gates.
3) Apply mask (if any) and element‑wise gating.
4) Compute the core O(N³) contraction
       out[b,i,j,d] = Σₖ left[b,i,k,d]·right[b,j,k,d]
   using a fused batched GEMM Triton kernel.
5) Layer‑norm over the hidden dimension, apply the output gate,
   and a final linear projection back to the original channel size.

The heavy reduction (step 4) is performed by a custom Triton kernel that
processes a batch of matrix multiplications (batch = B·hidden_dim) with
block‑wise tiling (BLOCK_M×BLOCK_N tiles, inner K‑tile BLOCK_K).  Inputs
are cast to fp16 for bandwidth, accumulation is fp32 for numerical
stability, and the result is converted back to fp32 for the remaining
PyTorch ops.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    B, M, N, K,                               # dimensions
    stride_a_batch, stride_a_m, stride_a_k,   # strides for A
    stride_b_batch, stride_b_k, stride_b_n,   # strides for B (already transposed)
    stride_c_batch, stride_c_m, stride_c_n,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # ---- block indices -------------------------------------------------
    num_m_blocks = tl.cdiv(M, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_m_blocks * num_n_blocks)
    pid = pid % (num_m_blocks * num_n_blocks)
    pid_m = pid // num_n_blocks
    pid_n = pid % num_n_blocks

    # ---- offsets within the block --------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ---- accumulator (fp32) -------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K dimension -----------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # pointers for this tile
        a = tl.load(
            a_ptr
            + batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k,
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_ptr
            + batch * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n,
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )

        # up‑cast to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # ---- write result ---------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = (
        c_ptr
        + batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, c, mask=mask)

    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask          : torch.Tensor or None, shape [B, N, N] (bool/float)
        - weights       : dict of torch.Tensor containing all model weights
        - config        : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    B, N, _, C = input_tensor.shape
    H = hidden_dim  # hidden dimension (often 128)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) LayerNorm over channel dimension C
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (C,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Linear projections & gates (all in float32)
    # ------------------------------------------------------------------
    x_flat = x.reshape(-1, C)                         # (B*N*N, C)

    left_proj_w  = weights["left_proj.weight"]        # (H, C)
    right_proj_w = weights["right_proj.weight"]       # (H, C)
    left_gate_w  = weights["left_gate.weight"]        # (H, C)
    right_gate_w = weights["right_gate.weight"]       # (H, C)
    out_gate_w   = weights["out_gate.weight"]         # (H, C)

    left   = F.linear(x_flat, left_proj_w).view(B, N, N, H)
    right  = F.linear(x_flat, right_proj_w).view(B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x_flat, left_gate_w)).view(B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_w)).view(B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(B, N, N, H)

    # ------------------------------------------------------------------
    # 3) Apply mask (if present) and gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask = mask.to(dtype).unsqueeze(-1)            # float32 mask
        left = left * mask
        right = right * mask

    # element‑wise gating
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Core O(N³) contraction using a fused Triton batched GEMM
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B*H, N, N) for batched GEMM
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # we need right transposed on the last two dims (K axis)
    right_h_T = right_h.transpose(1, 2).contiguous()

    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    batch = B * H
    num_m_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_n_blocks = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch * num_m_blocks * num_n_blocks

    grid = (total_blocks,)

    # strides
    a_batch_stride = left_h.stride(0)
    a_m_stride     = left_h.stride(1)
    a_k_stride     = left_h.stride(2)

    b_batch_stride = right_h_T.stride(0)
    b_k_stride     = right_h_T.stride(1)
    b_n_stride     = right_h_T.stride(2)

    c_batch_stride = out_h.stride(0)
    c_m_stride     = out_h.stride(1)
    c_n_stride     = out_h.stride(2)

    # launch Triton kernel
    batched_matmul_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h_T,
        c_ptr=out_h,
        B=batch,
        M=N,
        N=N,
        K=N,
        stride_a_batch=a_batch_stride,
        stride_a_m=a_m_stride,
        stride_a_k=a_k_stride,
        stride_b_batch=b_batch_stride,
        stride_b_k=b_k_stride,
        stride_b_n=b_n_stride,
        stride_c_batch=c_batch_stride,
        stride_c_m=c_m_stride,
        stride_c_n=c_n_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion before moving on
    torch.cuda.synchronize()

    # Convert back to fp32 and restore original layout
    out = out_h.to(torch.float32).view(B, H, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Final LayerNorm, output gating, and projection back to dim
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate          # apply output gate (broadcasted over dim)

    to_out_w = weights["to_out.weight"]   # (dim, H)
    out = F.linear(out, to_out_w)         # final projection → [B, N, N, dim]

    return out
