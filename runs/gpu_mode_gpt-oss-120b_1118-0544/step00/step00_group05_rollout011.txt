@triton.jit
def layernorm_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr,
                     B, N, C,
                     stride_xb, stride_xi, stride_xj, stride_xc,
                     stride_yb, stride_yi, stride_yj, stride_yc,
                     stride_w, stride_b,
                     BLOCK_C: tl.constexpr, EPS: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base ptr for this (b,i,j)
    x_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    y_base = y_ptr + b * stride_yb + i * stride_yi + j * stride_yj

    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)
    # First pass (mean/var)
    for off in range(0, C, BLOCK_C):
        cur = off + tl.arange(0, BLOCK_C)
        mask = cur < C
        # load vector
        x = tl.load(x_base + cur * stride_xc, mask=mask, other=0.0).to(tl.float32)  # shape=BLOCK_C
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

    mean = sum_val / C
    var = sum_sq / C - mean * mean
    rstd = 1.0 / tl.sqrt(var + EPS)

    # Second pass: write normalized
    for off in range(0, C, BLOCK_C):
        cur = off + tl.arange(0, BLOCK_C)
        mask = cur < C
        x = tl.load(x_base + cur * stride_xc, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + cur * stride_w, mask=mask, other=0.0).to(tl.float32)
        b_ = tl.load(bias_ptr + cur * stride_b, mask=mask, other=0.0).to(tl.float32)
        y = (x - mean) * rstd
        y = y * w + b_
        # cast back to original dtype (float32)
        tl.store(y_base + cur * stride_yc, y, mask=mask)

def layernorm_triton(x, weight, bias, eps=1e-5, block_c=32):
    B, N, N2, C = x.shape # note N2 same as N
    assert N == N2
    y = torch.empty_like(x)
    # compute strides (int)
    stride_xb, stride_xi, stride_xj, stride_xc = x.stride()
    stride_yb, stride_yi, stride_yj, stride_yc = y.stride()
    stride_w = weight.stride(0) if weight.ndim >0 else 1
    stride_b = bias.stride(0) if bias.ndim>0 else 1
    grid = lambda META: (B * N * N + META['BLOCK_M'] - 1) // META['BLOCK_M']
    # Actually BLOCK_M is not defined; we use each pid for each element, not blocks.
    # So we set kernel to have 1D grid of size total.

    # In triton, grid size is number of program IDs. Provide integer total.
    grid = (B * N * N, )
    # Need to call kernel
    layernorm_kernel[grid](
        x, y, weight, bias,
        B, N, C,
        stride_xb, stride_xi, stride_xj, stride_xc,
        stride_yb, stride_yi, stride_yj, stride_yc,
        stride_w, stride_b,
        BLOCK_C=block_c,
        EPS=eps
    )
    return y

"""
TriMul (outgoing) forward implementation with Triton accelerated LayerNorm.

The heavy O(N³) contraction is performed via a batched GEMM
(left @ rightᵀ) using PyTorch's highly‑optimized cuBLAS kernels,
while the two LayerNorms (input norm and output norm) are fused
into custom Triton kernels that compute mean/variance per
(N, N) position without extra Python loops.
"""

import torch
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: LayerNorm over the last dimension of a 4‑D tensor
# -------------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,          # input and output pointers
    weight_ptr, bias_ptr,  # LayerNorm parameters (1‑D)
    B, N, C,               # tensor dimensions
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_yb, stride_yi, stride_yj, stride_yc,
    stride_w, stride_b,
    BLOCK_C: tl.constexpr, # compile‑time tile size for the channel dim
    EPS: tl.constexpr,     # epsilon for numerical stability
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode (b, i, j) from linear program id
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base pointers for this (b,i,j) element
    x_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    y_base = y_ptr + b * stride_yb + i * stride_yi + j * stride_yj

    # -----------------------------------------------------------------
    # First pass: compute mean and variance over C channels
    # -----------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    num_blocks = tl.cdiv(C, BLOCK_C)
    for block_idx in range(num_blocks):
        off = block_idx * BLOCK_C
        cur = off + tl.arange(0, BLOCK_C)
        mask = cur < C
        x = tl.load(x_base + cur * stride_xc, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

    C_f = tl.float32(C)
    mean = sum_val / C_f
    var = sum_sq / C_f - mean * mean
    rstd = 1.0 / tl.sqrt(var + EPS)

    # -----------------------------------------------------------------
    # Second pass: write normalized result, applying weight & bias
    # -----------------------------------------------------------------
    for block_idx in range(num_blocks):
        off = block_idx * BLOCK_C
        cur = off + tl.arange(0, BLOCK_C)
        mask = cur < C
        x = tl.load(x_base + cur * stride_xc, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + cur * stride_w, mask=mask, other=0.0).to(tl.float32)
        b_ = tl.load(bias_ptr + cur * stride_b, mask=mask, other=0.0).to(tl.float32)
        y = (x - mean) * rstd
        y = y * w + b_
        tl.store(y_base + cur * stride_yc, y, mask=mask)


def layernorm_triton(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5,
                     block_c: int = 32) -> torch.Tensor:
    """
    Apply LayerNorm (per‑position) using Triton.
    x: [B, N, N, C] (float32, contiguous)
    weight, bias: [C]
    Returns y with same shape as x.
    """
    assert x.is_contiguous()
    B, N, N2, C = x.shape
    assert N == N2, "input must be square on the pairwise axes"
    y = torch.empty_like(x)

    # strides (in elements, not bytes)
    sxb, sxi, sxj, sxc = x.stride()
    syb, syi, syj, syc = y.stride()
    sw = weight.stride(0) if weight.ndim > 0 else 1
    sb = bias.stride(0) if bias.ndim > 0 else 1

    grid = (B * N * N,)

    layernorm_kernel[grid](
        x, y,
        weight, bias,
        B, N, C,
        sxb, sxi, sxj, sxc,
        syb, syi, syj, syc,
        sw, sb,
        BLOCK_C=block_c,
        EPS=eps,
    )
    return y

    if mask is None:
        mask = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=input_tensor.dtype,
            device=input_tensor.device,
        )
    mask = mask.to(input_tensor.dtype).unsqueeze(-1)  # [B, N, N, 1]

    # -----------------------------------------------------------------
    # Input LayerNorm (Triton)
    # -----------------------------------------------------------------
    with torch.no_grad():
        x_norm = layernorm_triton(input_tensor, norm_w, norm_b, eps=eps, block_c=32)

    # -----------------------------------------------------------------
    # Linear projections + gating
    # -----------------------------------------------------------------
    # Linear layers operate on the last dimension
    left = torch.nn.functional.linear(x_norm, left_proj_w)        # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_w)      # [B,N,N,hidden_dim]

    # Apply mask to the raw projections
    left = left * mask
    right = right * mask

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_w))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Batched contraction: left @ rightᵀ for each hidden channel
    # -----------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) so that we can use torch.matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hd, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, hd, N, N]

    # Matrix multiplication across the N dimension
    out = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # [B, hd, N, N]

    # Restore original layout
    out = out.permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # Output LayerNorm (Triton)
    # -----------------------------------------------------------------
    out_norm = layernorm_triton(out, to_out_norm_w, to_out_norm_b, eps=eps, block_c=32)

    # Apply final gating
    out_norm = out_norm * out_gate

    # -----------------------------------------------------------------
    # Final linear projection back to the original dimensionality
    # -----------------------------------------------------------------
    output = torch.nn.functional.linear(out_norm, to_out_w)   # [B, N, N, dim]

    return output
```
