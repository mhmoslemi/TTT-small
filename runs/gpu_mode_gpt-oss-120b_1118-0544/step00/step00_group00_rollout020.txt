def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(input_tensor,
                                            (dim,),
                                            weight=weights["norm.weight"],
                                            bias=weights["norm.bias"],
                                            eps=1e-5)
    # Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    if not nomask:
        # mask shape [B,N,N], convert to same dtype
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    B, N, _, _ = left_fp16.shape

    # Allocate output
    out_fp16 = torch.empty_like(left_fp16)  # same shape and dtype

    # Strides
    stride_a_batch = left_fp16.stride(0)
    stride_ai = left_fp16.stride(1)
    stride_ak = left_fp16.stride(2)

    stride_b_batch = right_fp16.stride(0)
    stride_bk = right_fp16.stride(2)   # row (k) stride
    stride_bj = right_fp16.stride(1)   # column (j) stride

    stride_c_batch = out_fp16.stride(0)
    stride_ci = out_fp16.stride(1)
    stride_cj = out_fp16.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * hidden_dim)
    tri_mul_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        B, N, hidden_dim,
        stride_a_batch, stride_ai, stride_ak,
        stride_b_batch, stride_bk, stride_bj,
        stride_c_batch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert out to fp32 for following ops
    out = out_fp16.to(torch.float32)

    # Apply to_out_norm
    out = torch.nn.functional.layer_norm(

        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

"""
TriMul (outgoing) forward implementation using Triton.

The heavy part of TriMul is the tensor contraction
    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
which is equivalent to a batched matrix multiplication Lₕ @ Rₕᵀ for each
hidden dimension h.  This kernel tiles the (i, j) output plane with
BLOCK_M × BLOCK_N tiles and reduces over k in blocks of size BLOCK_K.
Masking and gating are applied before the reduction with regular
PyTorch kernels; the reduction itself is performed in mixed‑precision
(fp16 inputs, fp32 accumulator) for speed.

The kernel receives the flattened batch×hidden dimension as the third
program ID, computes the correct offsets using the original strides of
the (B, N, N, H) layout and stores the result back in fp16.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to left, right, out
    B, N, H,                           # batch, sequence length, hidden dim
    stride_ab, stride_ai, stride_ak,   # A: batch, i (row), k (col)
    stride_bb, stride_bk, stride_bj,   # B: batch, k (row), j (col)  (B is transposed)
    stride_cb, stride_ci, stride_cj,   # C: batch, i (row), j (col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # i‑block
    pid_n = tl.program_id(1)   # j‑block
    pid_z = tl.program_id(2)   # combined batch*hidden index

    # recover batch and hidden index from pid_z
    b_idx = pid_z // H
    h_idx = pid_z % H

    # base pointers for the chosen (b, h) slice
    a_base = a_ptr + b_idx * stride_ab + h_idx
    b_base = b_ptr + b_idx * stride_bb + h_idx
    c_base = c_ptr + b_idx * stride_cb + h_idx

    offs_i = pid_m * BLOCK_M
    offs_j = pid_n * BLOCK_N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for kp in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kp * BLOCK_K

        # indices for A (i,k)
        i_idx = offs_i + tl.arange(0, BLOCK_M)[:, None]
        k_idx = k_start + tl.arange(0, BLOCK_K)[None, :]
        a_ptrs = a_base + i_idx * stride_ai + k_idx * stride_ak
        a_mask = (i_idx < N) & (k_idx < N)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # indices for Bᵀ (k,j) i.e. original right[j,k]
        j_idx = offs_j + tl.arange(0, BLOCK_N)[None, :]
        b_ptrs = b_base + k_idx * stride_bk + j_idx * stride_bj
        b_mask = (k_idx < N) & (j_idx < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # fused multiply‑add
        acc += tl.dot(a, b)

    # write back C (i,j)
    i_idx = offs_i + tl.arange(0, BLOCK_M)[:, None]
    j_idx = offs_j + tl.arange(0, BLOCK_N)[None, :]
    c_ptrs = c_base + i_idx * stride_ci + j_idx * stride_cj
    c_mask = (i_idx < N) & (j_idx < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        output tensor of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input tensor (over the last dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Apply optional mask and gating
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_exp = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to fp16 and launch the Triton kernel for the einsum
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape  # B, seq_len, seq_len, hidden_dim
    out_fp16 = torch.empty_like(left_fp16)  # same shape, fp16

    # Strides (in number of elements, not bytes)
    stride_ab = left_fp16.stride(0)   # batch stride = N*N*H
    stride_ai = left_fp16.stride(1)   # row (i) stride    = N*H
    stride_ak = left_fp16.stride(2)   # col (k) stride    = H

    stride_bb = right_fp16.stride(0)  # same as A
    stride_bk = right_fp16.stride(2)  # row (k) stride in transposed view = H
    stride_bj = right_fp16.stride(1)  # col (j) stride                      = N*H

    stride_cb = out_fp16.stride(0)    # batch stride for C = N*N*H
    stride_ci = out_fp16.stride(1)    # row (i) stride    = N*H
    stride_cj = out_fp16.stride(2)    # col (j) stride    = H

    # Tile sizes – chosen to fit typical A100 registers
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # blocks over i dimension
        triton.cdiv(N, BLOCK_N),          # blocks over j dimension
        B * hidden_dim,                   # combined batch * hidden dimension
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        stride_ab,
        stride_ai,
        stride_ak,
        stride_bb,
        stride_bk,
        stride_bj,
        stride_cb,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, output gating and final linear
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)   # to_out_norm works in fp32

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    output = F.linear(out, weights["to_out.weight"])
    return output
