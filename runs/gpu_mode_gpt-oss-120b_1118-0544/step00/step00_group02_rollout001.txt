@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH_H, N,
    stride_left_bh, stride_left_i, stride_left_k,
    stride_right_bh, stride_right_j, stride_right_k,  # note ordering for B_T
    stride_out_bh, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Program id identifies a block in (block_m, block_n, batch_h).
    batch_h = tl.program_id(2)  # block along batch*channel dimension
    block_i = tl.program_id(0)
    block_j = tl.program_id(1)
    # Compute offsets.
    i_base = block_i * BLOCK_M
    j_base = block_j * BLOCK_N
    # Pointers for A (left): start point for this batch_h
    left_batch_ptr = left_ptr + batch_h * stride_left_bh
    # left_i pointers: block rows
    left_ptrs = left_batch_ptr + (i_base + tl.arange(0, BLOCK_M)[:, None]) * stride_left_i + tl.arange(0, BLOCK_K)[None, :] * stride_left_k

    # right pointer: treat as B_T, where we need rows along k and columns along j.
    # For B_T, "row" index is k, "col" index is j.
    right_batch_ptr = right_ptr + batch_h * stride_right_bh
    right_ptrs = right_batch_ptr + (j_base + tl.arange(0, BLOCK_N)[None, :]) * stride_right_j + tl.arange(0, BLOCK_K)[:, None] * stride_right_k

    # Use accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k in steps of BLOCK_K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load A block
        a = tl.load(left_ptrs, mask=(i_base + tl.arange(0, BLOCK_M)[:, None] < N) & (k*BLOCK_K + tl.arange(0, BLOCK_K)[None, :] < N), other=0.0)
        # load B_T block
        b = tl.load(right_ptrs, mask=(j_base + tl.arange(0, BLOCK_N)[None, :] < N) & (k*BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < N), other=0.0)
        # compute dot
        acc += tl.dot(a, b)
        # increment pointers
        left_ptrs += BLOCK_K * stride_left_k
        right_ptrs += BLOCK_K * stride_right_k

    # Store result
    out_batch_ptr = out_ptr + batch_h * stride_out_bh
    out_i_ptrs = out_batch_ptr + (i_base + tl.arange(0, BLOCK_M)[:, None]) * stride_out_i + (j_base + tl.arange(0, BLOCK_N)[None, :]) * stride_out_j
    tl.store(out_i_ptrs, acc.to(tl.float16) if acc.dtype==tl.float32 else acc, mask=(i_base + tl.arange(0, BLOCK_M)[:, None] < N) & (j_base + tl.arange(0, BLOCK_N)[None, :] < N))

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_bh, stride_left_i, stride_left_k,
    stride_right_bh, stride_right_j, stride_right_k,
    stride_out_bh, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    # offsets
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N
    # compute mask for i,j range
    i_mask = i_offset + tl.arange(0, BLOCK_M) < N
    j_mask = j_offset + tl.arange(0, BLOCK_N) < N
    # pointers to left block
    left_base = left_ptr + pid_bh * stride_left_bh
    # each row i, each k.
    # left indices: [i_offset + i, k]
    k_range = tl.arange(0, BLOCK_K)
    i_range = tl.arange(0, BLOCK_M)
    a_ptrs = left_base + (i_offset + i_range[:, None]) * stride_left_i + (k_range[None, :] ) * stride_left_k
    # pointers to right (transposed)
    right_base = right_ptr + pid_bh * stride_right_bh
    # right indices: [j_offset + j, k] -> we want B_T row index k (fast), col index j
    b_ptrs = right_base + (j_offset + tl.arange(0, BLOCK_N)[None, :]) * stride_right_j + (k_range[:, None]) * stride_right_k
    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # loop over k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=i_mask[:, None] & (kk*BLOCK_K + k_range[None,:] < N), other=0.0)
        b = tl.load(b_ptrs, mask=j_mask[None,:] & (kk*BLOCK_K + k_range[:, None] < N), other=0.0)
        acc += tl.dot(a, b)
        # advance pointers
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k
    # store result
    out_base = out_ptr + pid_bh * stride_out_bh
    out_ptrs = out_base + (i_offset + i_range[:, None]) * stride_out_i + (j_offset + tl.arange(0, BLOCK_N)[None,:]) * stride_out_j
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None,:])

"""
TriMul outgoing update (AlphaFold3) forward pass.

Implements the core reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where left/right are linear projections of the input with gating and optional masking.
The reduction is performed by a custom Triton kernel that computes a batched
matrix multiplication L @ Rᵀ for each hidden channel, exploiting FP16 compute
and FP32 accumulation. Remaining operations (LayerNorm, linear projections,
gating, final projection) are carried out with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication with right transposed.
# Input tensors have shape (B*H, N, N) and dtype float16.
# Output tensor has shape (B*H, N, N) and dtype float32.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                # *fp16
    right_ptr,               # *fp16   (original right, not transposed)
    out_ptr,                 # *fp32
    N: tl.int32,
    stride_left_bh: tl.int32,
    stride_left_i: tl.int32,
    stride_left_k: tl.int32,
    stride_right_bh: tl.int32,
    stride_right_j: tl.int32,   # original j dimension
    stride_right_k: tl.int32,   # original k dimension
    stride_out_bh: tl.int32,
    stride_out_i: tl.int32,
    stride_out_j: tl.int32,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(2)           # batch*hidden channel index
    pid_i = tl.program_id(0)            # block row (i)
    pid_j = tl.program_id(1)            # block col (j)

    # base offsets for this block
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # range indices
    i_range = tl.arange(0, BLOCK_M)
    j_range = tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    # masks for i/j within N
    i_valid = i0 + i_range < N
    j_valid = j0 + j_range < N

    # Pointers to the start of the selected batch*hidden slice
    left_base = left_ptr + pid_bh * stride_left_bh
    right_base = right_ptr + pid_bh * stride_right_bh
    out_base = out_ptr + pid_bh * stride_out_bh

    # Pointers to the start of the block (k=0)
    a_ptrs = left_base + (i0 + i_range[:, None]) * stride_left_i + (k_range[None, :] ) * stride_left_k
    # For right, we need Bᵀ so rows are k and columns are j.
    b_ptrs = right_base + (j0 + j_range[None, :]) * stride_right_j + (k_range[:, None]) * stride_right_k

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = kk * BLOCK_K
        k_valid = cur_k + k_range < N

        # load A block (left)
        a = tl.load(a_ptrs,
                    mask=i_valid[:, None] & k_valid[None, :],
                    other=0.0)

        # load B block (right transposed)
        b = tl.load(b_ptrs,
                    mask=j_valid[None, :] & k_valid[:, None],
                    other=0.0)

        # dot product; tl.dot accumulates in FP32
        acc += tl.dot(a, b)

        # advance pointers along k dimension
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # Store result (FP32)
    out_ptrs = out_base + (i0 + i_range[:, None]) * stride_out_i + (j0 + j_range[None, :]) * stride_out_j
    mask_store = i_valid[:, None] & j_valid[None, :]
    tl.store(out_ptrs, acc, mask=mask_store)


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel for the batched matmul.
# ----------------------------------------------------------------------
def triton_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: (B*H, N, N) float16
    returns out: (B*H, N, N) float32
    """
    assert left.is_contiguous() and right.is_contiguous()
    BH, N, _ = left.shape
    out = torch.empty((BH, N, N), device=left.device, dtype=torch.float32)

    # strides (in elements)
    s_left_bh, s_left_i, s_left_k = left.stride()
    s_right_bh, s_right_i, s_right_k = right.stride()
    s_out_bh, s_out_i, s_out_j = out.stride()

    # Choose block sizes; values tuned for typical RTX/A100 GPUs.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            BH)

    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        N=N,
        stride_left_bh=s_left_bh,
        stride_left_i=s_left_i,
        stride_left_k=s_left_k,
        stride_right_bh=s_right_bh,
        stride_right_j=s_right_i,   # original j stride
        stride_right_k=s_right_k,   # original k stride
        stride_out_bh=s_out_bh,
        stride_out_i=s_out_i,
        stride_out_j=s_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul update.
    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float32/bool) or None
            - weights: dict of model parameters (all float32)
            - config: dict containing at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    nomask = config.get("nomask", False)
    if mask is not None and not nomask:
        # mask shape: [B, N, N]; expand to broadcast over hidden dim
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare for Triton reduction ----------
    # Move hidden dimension to second axis for contiguous (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Flatten batch and hidden dim: (B*H, N, N)
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N).contiguous()
    right_flat = right_perm.view(BH, N, N).contiguous()

    # Cast to FP16 for the kernel (good speed, fp32 accumulation inside)
    left_f16 = left_flat.to(torch.float16)
    right_f16 = right_flat.to(torch.float16)

    # ---------- Triton batched matmul (L @ Rᵀ) ----------
    out_flat = triton_tri_mul(left_f16, right_f16)  # (BH, N, N) float32

    # ---------- Reshape back ----------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden_dim)

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final projection ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
