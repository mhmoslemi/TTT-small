"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The heavy part of the TriMul module is the contraction:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
where left/right are the projected and gated pairwise representations.
We implement this contraction as a blocked GEMM over the (i, j, d) tile
using Triton while the surrounding linear layers, layer‑norms and gates are
performed with PyTorch.  The kernel blocks the i and j dimensions (M, N)
and the hidden dimension d (D) while iterating over k in blocks (K), thus
reducing memory traffic and enabling high‑throughput accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # problem size
    B, N, H,
    # strides (elements) for left
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides for right (same layout as left)
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    # strides for output
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
    NUM_D_TILES: tl.constexpr,
):
    # -----------------------------------------------------------------
    # program ids
    pid_i = tl.program_id(0)   # i‑tile
    pid_j = tl.program_id(1)   # j‑tile
    pid_z = tl.program_id(2)   # combined batch and d‑tile

    batch = pid_z // NUM_D_TILES
    d_tile = pid_z % NUM_D_TILES

    # -----------------------------------------------------------------
    # tile offsets
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N
    d_offset = d_tile * BLOCK_D

    # -----------------------------------------------------------------
    # index vectors
    i_idx = i_offset + tl.arange(0, BLOCK_M)
    j_idx = j_offset + tl.arange(0, BLOCK_N)
    d_idx = d_offset + tl.arange(0, BLOCK_D)

    # bounds masks for output tile (used when storing)
    mask_i = i_idx < N
    mask_j = j_idx < N
    mask_d = d_idx < H
    mask_out = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]

    # -----------------------------------------------------------------
    # accumulators
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # -----------------------------------------------------------------
    # loop over k dimension in blocks
    k = 0
    while k < N:
        # actual k indices for this block
        k_idx = k + tl.arange(0, BLOCK_K)

        # masks for k dimension
        mask_k = k_idx < N

        # -----------------------------------------------------------------
        # offsets for left (i, k, d)
        offs_i = (i_idx[:, None, None] * stride_l_i)
        offs_k = (k_idx[None, :, None] * stride_l_k)
        offs_d = (d_idx[None, None, :] * stride_l_h)
        ptrs_left = left_ptr \
            + batch * stride_l_b \
            + offs_i + offs_k + offs_d
        mask_left = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
        left = tl.load(ptrs_left, mask=mask_left, other=0.0)   # (M, K, D)

        # -----------------------------------------------------------------
        # offsets for right (j, k, d)
        offs_j = (j_idx[:, None, None] * stride_r_i)
        # k stride is identical
        ptrs_right = right_ptr \
            + batch * stride_r_b \
            + offs_j + offs_k + offs_d
        mask_right = mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
        right = tl.load(ptrs_right, mask=mask_right, other=0.0)   # (N, K, D)

        # -----------------------------------------------------------------
        # accumulate over k (inner loop over the K block)
        for kk in range(0, BLOCK_K):
            # l_vec: (M, D), r_vec: (N, D)
            l_vec = left[:, kk, :]      # (M, D)
            r_vec = right[:, kk, :]     # (N, D)
            # outer product over i and j for this hidden slice
            acc += l_vec[:, None, :] * r_vec[None, :, :]   # (M, N, D)

        k += BLOCK_K

    # -----------------------------------------------------------------
    # store the result tile
    offs_i_out = (i_idx[:, None, None] * stride_o_i)
    offs_j_out = (j_idx[None, :, None] * stride_o_j)
    offs_d_out = (d_idx[None, None, :] * stride_o_h)
    ptrs_out = out_ptr \
        + batch * stride_o_b \
        + offs_i_out + offs_j_out + offs_d_out
    tl.store(ptrs_out, acc, mask=mask_out)


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module with Triton acceleration.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim] torch.Tensor
            - mask_tensor:  [B, N, N]   torch.Tensor  (may be None)
            - weights_dict: dict of torch.Tensor with the required weights
            - config_dict:  dict containing "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # LayerNorm on the input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # -----------------------------------------------------------------
    # Linear projections (no bias)
    left_proj_w = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]    # (hidden_dim, dim)
    left_gate_w = weights["left_gate.weight"]      # (hidden_dim, dim)
    right_gate_w = weights["right_gate.weight"]    # (hidden_dim, dim)
    out_gate_w = weights["out_gate.weight"]        # (hidden_dim, dim)

    # reshape for efficient linear layers
    x_flat = x.reshape(-1, dim)                     # (B*N*N, dim)

    left = F.linear(x_flat, left_proj_w).reshape(B, N, N, hidden_dim)
    right = F.linear(x_flat, right_proj_w).reshape(B, N, N, hidden_dim)

    left_gate = torch.sigmoid(F.linear(x_flat, left_gate_w)).reshape(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_w)).reshape(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w)).reshape(B, N, N, hidden_dim)

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # apply mask if present
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # Triton kernel for the contraction
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # strides (in elements) for tensors
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Blocking parameters (tuned for the B200)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 16   # must divide hidden_dim or be handled by mask
    num_i_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_j_blocks = (N + BLOCK_N - 1) // BLOCK_N
    num_d_tiles = (hidden_dim + BLOCK_D - 1) // BLOCK_D

    grid = (num_i_blocks, num_j_blocks, B * num_d_tiles)

    tri_mul_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_ptr=out.data_ptr(),
        B=B,
        N=N,
        H=hidden_dim,
        stride_l_b=stride_l_b,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_l_h=stride_l_h,
        stride_r_b=stride_r_b,
        stride_r_i=stride_r_i,
        stride_r_k=stride_r_k,
        stride_r_h=stride_r_h,
        stride_o_b=stride_o_b,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        stride_o_h=stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        NUM_D_TILES=num_d_tiles,
    )
    # -----------------------------------------------------------------
    # post‑processing: LayerNorm over hidden_dim, out‑gate, final linear
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]   # (dim, hidden_dim)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    # final projection back to model dimension
    out = torch.matmul(out, to_out_w.t())

    return out
