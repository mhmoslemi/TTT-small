"""
TriMul (outgoing) forward pass with a Triton‐accelerated sigmoid.
- LayerNorm on the input is done with torch.nn.functional.layer_norm.
- Linear projections and the output LayerNorm use PyTorch (highly optimized).
- The three gate activations are computed with a custom Triton kernel that
  applies the element‑wise sigmoid (1/(1+exp(-x))).
- The outgoing triangular multiplicative update is performed as a batched
  matrix multiplication for each hidden channel:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,k,j,d]
  which is realized by reshaping the tensors to (B*H, N, N) and using
  torch.bmm.
- Finally a second LayerNorm, out‑gate scaling and a linear projection back
  to the original dimensionality are applied.
"""

import math
import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise sigmoid (used for the three gating tensors)
# ----------------------------------------------------------------------
@triton.jit
def sigmoid_kernel(x_ptr, y_ptr, numel, TILE: tl.constexpr):
    """Compute y = sigmoid(x) for a contiguous 1‑D tensor.

    Arguments
    ---------
    x_ptr, y_ptr : pointers to the input and output tensors (float32)
    numel        : total number of elements in the tensor
    TILE         : compile‑time tile size (e.g. 128)
    """
    pid = tl.program_id(0)
    offset = pid * TILE
    idx = tl.arange(0, TILE)
    mask = offset + idx < numel

    x = tl.load(x_ptr + offset + idx, mask=mask, other=0.0)
    y = 1.0 / (1.0 + tl.exp(-x))
    tl.store(y_ptr + offset + idx, y, mask=mask)


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of torch tensors containing all learned parameters
        - config       : dict with at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------ 1. Input LayerNorm (torch, very fast) ------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------ 2. Linear projections (left & right) ------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # ------------ 3. Apply mask if present ------------
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------ 4. Gate pre‑activations (linear) ------------
    left_gate_pre = torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    right_gate_pre = torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    out_gate_pre = torch.nn.functional.linear(x_norm, weights["out_gate.weight"])

    # Ensure contiguous tensors for Triton
    left_gate_pre = left_gate_pre.contiguous()
    right_gate_pre = right_gate_pre.contiguous()
    out_gate_pre = out_gate_pre.contiguous()

    # Prepare output tensors for the sigmoid
    left_gate = torch.empty_like(left_gate_pre)
    right_gate = torch.empty_like(right_gate_pre)
    out_gate = torch.empty_like(out_gate_pre)

    # ------------ 5. Triton sigmoid for the three gates ------------
    TILE = 128  # tile size (must be compile‑time constant)
    # left gate
    total = left_gate_pre.numel()
    grid = (math.ceil(total / TILE),)
    sigmoid_kernel[grid](left_gate_pre, left_gate, total, TILE=TILE)
    # right gate
    total = right_gate_pre.numel()
    grid = (math.ceil(total / TILE),)
    sigmoid_kernel[grid](right_gate_pre, right_gate, total, TILE=TILE)
    # out gate
    total = out_gate_pre.numel()
    grid = (math.ceil(total / TILE),)
    sigmoid_kernel[grid](out_gate_pre, out_gate, total, TILE=TILE)

    # ------------ 6. Apply gating to the projected tensors ------------
    left = left * left_gate
    right = right * right_gate

    # ------------ 7. Outgoing triangular multiplicative update ------------
    # Reshape to (B*H, N, N) so each hidden channel is a separate matrix
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Batched matrix multiplication: (i,k) @ (k,j) -> (i,j)
    out_flat = torch.bmm(left_flat, right_flat)  # [B*H, N, N]

    # Restore original layout [B, N, N, hidden_dim]
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------ 8. Output LayerNorm ------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------ 9. Scale by out_gate ------------
    out = out * out_gate

    # ------------ 10. Final linear projection back to dim ------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
