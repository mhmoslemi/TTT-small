@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_aba, stride_ams, stride_akm,  # Actually a_ptr: (batch, m, k) strides: stride_ab, stride_am, stride_ak
    stride_bba, stride_bbk, stride_bkn,
    stride_cca, stride_ccm, stride_ccn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    total_m = tl.cdiv(M, BLOCK_M)
    total_n = tl.cdiv(N, BLOCK_N)

    batch_id = pid // (total_m * total_n)
    block_index = pid % (total_m * total_n)
    block_m = block_index // total_n
    block_n = block_index % total_n

    # offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Mask because M,N may not be multiple of block size
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers
    a_ptrs = a_ptr + batch_id * stride_aba + offs_m[:, None] * stride_ams + tl.arange(0, BLOCK_K)[None, :] * stride_akm
    b_ptrs = b_ptr + batch_id * stride_bba + tl.arange(0, BLOCK_K)[:, None] * stride_bbk + offs_n[None, :] * stride_bkn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load A and B tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k*BLOCK_K+tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k*BLOCK_K+tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
        # compute
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_akm
        b_ptrs += BLOCK_K * stride_bbk

    # write out
    c_ptrs = c_ptr + batch_id * stride_cca + offs_m[:, None] * stride_ccm + offs_n[None, :] * stride_ccn
    tl.store(c_ptrs, acc.to(tl.float16) if acc.dtype==tl.float16 else acc, mask=mask_m[:, None] & mask_n[None, :])

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑normalize the 4‑D input tensor (last dimension = dim).
2. Apply three linear projections (left, right, and three gates) using the
   provided weights.  Gating values are passed through sigmoid and multiplied
   with the projected tensors.
3. The core operation is an O(N³) sum:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is equivalent to a batched matrix multiplication where each hidden
   channel is treated as an independent batch:
        left  → (B·H, N, N)  (rows i, cols k)
        right → (B·H, N, N).T (rows k, cols j)
   The kernel `batched_matmul_kernel` computes this GEMM for all batches,
   tiling the M (i) and N (j) dimensions and streaming over K (k).
4. The result is reshaped back to [B, N, N, hidden_dim],
   layer‑normalized, multiplied by the output gate and finally projected
   back to the original dimension.
The heavy O(N³) part is performed in Triton; all other tensor
manipulations use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM (A @ Bᵀ) where each hidden channel is a batch.
# A: (batch, M, K)   – left tensor
# B: (batch, K, N)   – right tensor already transposed
# C: (batch, M, N)   – output
# ----------------------------------------------------------------------------- #
@triton.jit
def batched_matmul_kernel(
    a_ptr,                # *float32
    b_ptr,                # *float32
    c_ptr,                # *float32
    total_batch: tl.int32,   # total number of batches (B*H)
    M: tl.int32,              # seq_len (rows of A)
    N: tl.int32,              # seq_len (cols of Bᵀ)
    K: tl.int32,              # seq_len (inner dimension)
    stride_a_batch, stride_am, stride_ak,   # strides for A
    stride_b_batch, stride_bk, stride_bn,   # strides for Bᵀ
    stride_c_batch, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # Number of tiles in M/N dimensions
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    batch_id = pid // (num_m * num_n)
    pid_in_batch = pid % (num_m * num_n)
    block_m = pid_in_batch // num_n
    block_n = pid_in_batch % num_n

    # Offsets for the current M/N tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current batch
    a_batch_ptr = a_ptr + batch_id * stride_a_batch
    b_batch_ptr = b_ptr + batch_id * stride_b_batch
    c_batch_ptr = c_ptr + batch_id * stride_c_batch

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointer arithmetic for the current K‑tile
        a_ptrs = a_batch_ptr + \
                 offs_m[:, None] * stride_am + \
                 offs_k[None, :] * stride_ak
        b_ptrs = b_batch_ptr + \
                 offs_k[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn

        # Load tiles with masking
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Write back the result
    c_ptrs = c_batch_ptr + \
             offs_m[:, None] * stride_cm + \
             offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------------- #
# Entry point used by the evaluator
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (may be None)
        - weights      : dict of weight tensors
        - config       : dict containing at least 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim]
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1. LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # --------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])    # [B, N, N, H]

    # --------------------------------------------------------------------- #
    # 3. Optional mask (broadcast over the hidden dimension)
    # --------------------------------------------------------------------- #
    if mask is not None and mask.numel() > 0:
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------- #
    # 4. Gating (sigmoid of linear projections)
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Core TriMul: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM using Triton.
    # --------------------------------------------------------------------- #
    B, N, _, _ = left.shape

    # Reshape so that each hidden channel becomes an independent batch:
    #   (B, N, N, H) -> (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # Transpose the inner dimension of right (k ↔ j) so that we can perform
    # a regular matrix multiplication A @ Bᵀ.
    right_flat_t = right_flat.transpose(1, 2).contiguous()

    # Allocate output tensor
    out_flat = torch.empty_like(left_flat, device=device)

    # Tiling parameters (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    total_batch = B * hidden_dim
    grid = (total_batch * num_m * num_n,)

    # Strides for the three tensors (in elements, not bytes)
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat_t.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # Launch the Triton kernel
    batched_matmul_kernel[grid](
        left_flat,
        right_flat_t,
        out_flat,
        total_batch,
        N,               # M
        N,               # N
        N,               # K
        stride_a_batch, stride_am, stride_ak,
        stride_b_batch, stride_bk, stride_bn,
        stride_c_batch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 6. Output LayerNorm, gating and final linear projection
    # --------------------------------------------------------------------- #
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
