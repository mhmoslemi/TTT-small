"""
TriMul (outgoing) forward pass implemented with a fused Triton GEMM.

Algorithm:
1. Layer‑norm the input over the last dimension.
2. Linear projections to hidden_dim (left/right) and corresponding sigmoid gates.
3. Apply optional mask to left/right projections, then multiply with their gates.
4. For each hidden channel compute L @ Rᵀ where L,R ∈ ℝ^{N×N}.
   This is realized as a batched GEMM (B·hidden_dim batches) using a custom
   Triton kernel that multiplies A (M×K) with Bᵀ (K×N) → C (M×N).
5. Layer‑norm over hidden_dim, apply the output gate, and finally linearly project
   back to the original dimension.
All heavy computation (step 4) is performed in Triton; the rest uses
torch.nn.functional for simplicity and correctness.
"""

import torch
import triton
import triton.language as tl
import math

# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for out = L @ Rᵀ   (L,R: [batch, N, N])
# ----------------------------------------------------------------------------- #
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    N,                                 # matrix size (square)
    stride_ah, stride_ai, stride_ak,   # strides for A: batch, i, k
    stride_bh, stride_bi, stride_bk,   # strides for B (treated as transposed): batch, j, k
    stride_ch, stride_ci, stride_cj,   # strides for C: batch, i, j
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """ Compute C[b,i,j] = sum_k A[b,i,k] * B[b,j,k] (i.e. A @ Bᵀ). """
    pid_m = tl.program_id(axis=0)   # block row
    pid_n = tl.program_id(axis=1)   # block col
    pid_b = tl.program_id(axis=2)   # batch (B * hidden_dim)

    # ------------------------ offsets & masks -----------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this block (without the k dimension)
    a_base = a_ptr + pid_b * stride_ah + offs_m[:, None] * stride_ai
    b_base = b_ptr + pid_b * stride_bh + offs_n[None, :] * stride_bi

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_offset = k * BLOCK_K
        k_range = k_offset + tl.arange(0, BLOCK_K)

        mask_k = k_range < N

        # Load A block: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_base + k_range[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B block as (BLOCK_K, BLOCK_N) – note the swapped strides.
        b = tl.load(
            b_base + k_range[:, None] * stride_bk,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # GEMM on this sub‑tile
        acc += tl.dot(a, b)

    # --------------------------- store result ------------------------------
    c_ptrs = c_ptr + pid_b * stride_ch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------------- #
# Custom kernel entry point
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operation.

    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] torch.Tensor (float32)
            - mask:        [B, N, N] torch.Tensor (bool/int) – may be ignored
            - weights:     dict with all required weight tensors
            - config:      dict with keys ["dim", "hidden_dim", "nomask"]

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------------- #
    # Load weights
    # --------------------------------------------------------------------- #
    # LayerNorms
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # Linear projections (bias=False)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    # Final projection
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------------- #
    # 1) Input layer‑norm over dim
    # --------------------------------------------------------------------- #
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------------- #
    # 2) Linear projections to hidden_dim and sigmoid gates
    # --------------------------------------------------------------------- #
    # Projections
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B,N,N,H]
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight) # [B,N,N,H]

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # --------------------------------------------------------------------- #
    # 3) Optional mask (only on left/right before gating)
    # --------------------------------------------------------------------- #
    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1] broadcast
        mask_ = mask.to(dtype).unsqueeze(-1)
        left_proj = left_proj * mask_
        right_proj = right_proj * mask_

    # Apply gates
    left = left_proj * left_gate   # [B,N,N,H]
    right = right_proj * right_gate

    # Free unused tensors
    del left_proj, right_proj, left_gate, right_gate, x_norm
    torch.cuda.empty_cache()

    # --------------------------------------------------------------------- #
    # 4) Batched GEMM: out = left @ rightᵀ  (per hidden channel)
    # --------------------------------------------------------------------- #
    B, N, _, _ = left.shape  # batch size, seq_len
    # Rearrange to (B, H, N, N) -> flatten batch*H
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_h = B * hidden_dim
    left_mat = left_perm.view(batch_h, N, N)
    right_mat = right_perm.view(batch_h, N, N)

    # Allocate output buffer (float32)
    out_mat = torch.empty_like(left_mat, dtype=torch.float32)

    # Strides (int64)
    stride_ah, stride_ai, stride_ak = left_mat.stride()
    stride_bh, stride_bi, stride_bk = right_mat.stride()
    stride_ch, stride_ci, stride_cj = out_mat.stride()

    # Block sizes (tuned for typical N up to 1024)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # 32 works well for float32 on A100/B200

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_h,
    )
    tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        stride_ah,
        stride_ai,
        stride_ak,
        stride_bh,
        stride_bi,
        stride_bk,
        stride_ch,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # Clean temp buffers
    del left, right, left_perm, right_perm, left_mat, right_mat, out_mat
    torch.cuda.empty_cache()

    # --------------------------------------------------------------------- #
    # 5) Output LayerNorm over hidden_dim, multiply by out_gate, final linear
    # --------------------------------------------------------------------- #
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # [B,N,N,H]

    out_gated = out_norm * out_gate  # broadcasting over hidden_dim

    # Final linear projection back to dim (no bias)
    output = torch.nn.functional.linear(out_gated, to_out_weight)  # [B,N,N,dim]

    return output
