@triton.jit
def left_right_gate_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, H,
    stride_lproj_b, stride_lproj_i, stride_lproj_j, stride_lproj_h,
    stride_rproj_b, stride_rproj_i, stride_rproj_j, stride_rproj_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lout_b, stride_lout_i, stride_lout_j, stride_lout_h,
    stride_rout_b, stride_rout_i, stride_rout_j, stride_rout_h,
    BLOCK_H: tl.constexpr,
):
    pid0 = tl.program_id(0)  # (b,i,j)
    pid1 = tl.program_id(1)  # hidden block
    total = B * N * N
    if pid0 >= total:
        return

    # decode b,i,j
    b = pid0 // (N * N)
    rem = pid0 % (N * N)
    i = rem // N
    j = rem % N

    # hidden offset
    h_offset = pid1 * BLOCK_H
    offs_h = tl.arange(0, BLOCK_H)
    h = h_offset + offs_h

    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    mask_val = tl.load(mask_ptr + mask_offset)  # scalar

    # load left_proj and left_gate vector
    base_lp = b * stride_lproj_b + i * stride_lproj_i + j * stride_lproj_j
    base_lg = b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j
    base_rp = b * stride_rproj_b + i * stride_rproj_i + j * stride_rproj_j
    base_rg = b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j

    # mask for when h < H
    mask = h < H
    left_proj = tl.load(left_proj_ptr + base_lp + offs_h * stride_lproj_h, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + base_lg + offs_h * stride_lgate_h, mask=mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + base_rp + offs_h * stride_rproj_h, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + base_rg + offs_h * stride_rgate_h, mask=mask, other=0.0)

    left_out = left_proj * left_gate * mask_val
    right_out = right_proj * right_gate * mask_val

    # store output
    base_lo = b * stride_lout_b + i * stride_lout_i + j * stride_lout_j
    base_ro = b * stride_rout_b + i * stride_rout_i + j * stride_rout_j

"""
Fast TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the input `x` (shape [B,N,N,C]).
2. Linear projections to hidden dimension H:
   * left_proj = x @ W_left        (no bias)
   * right_proj = x @ W_right
   * left_gate  = sigmoid(x @ W_left_gate)
   * right_gate = sigmoid(x @ W_right_gate)
   * out_gate   = sigmoid(x @ W_out_gate)   # shape [B,N,N,H]
3. Fuse mask, left/right gates and projections in a Triton kernel:
        left  = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask
   (mask broadcasted over H).
4. Perform the “triangle” reduction using a batched GEMM:
        left_perm  = left .permute(0,3,1,2)   # [B,H,N,N]
        right_perm = right.permute(0,3,2,1)   # [B,H,N,N]
        out_perm   = torch.matmul(left_perm, right_perm)   # sum_k left[i,k] * right[j,k]
        out        = out_perm.permute(0,2,3,1)   # [B,N,N,H]
5. Apply output LayerNorm, out‑gate and final linear projection back to C.

Only the mask‑gate fusion is implemented in Triton; all dense linear layers
and the heavy N³ reduction use highly‑optimised PyTorch batched‐matmul.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _apply_mask_and_gates_kernel(
    # input tensors
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    # output tensors
    left_out_ptr, right_out_ptr,
    # shapes / strides
    B, N, H,
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_h,
    stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_h,
    BLOCK_H: tl.constexpr,
):
    """
    For each (b,i,j) compute a slice of the hidden dimension:
        left  = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask
    mask is a scalar broadcasted over H.
    """
    pid0 = tl.program_id(0)   # encodes (b,i,j)
    pid1 = tl.program_id(1)   # encodes block of hidden dim

    total = B * N * N
    if pid0 >= total:
        return

    # decode batch, row, col from flat index
    b = pid0 // (N * N)
    rem = pid0 % (N * N)
    i = rem // N
    j = rem % N

    # hidden indices for this block
    h_start = pid1 * BLOCK_H
    h_off = tl.arange(0, BLOCK_H)
    h = h_start + h_off
    h_mask = h < H

    # base offset for (b,i,j) without hidden
    base_lp = b * stride_lp_b + i * stride_lp_i + j * stride_lp_j
    base_rp = b * stride_rp_b + i * stride_rp_i + j * stride_rp_j
    base_lg = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j
    base_rg = b * stride_rg_b + i * stride_rg_i + j * stride_rg_j
    base_lo = b * stride_lo_b + i * stride_lo_i + j * stride_lo_j
    base_ro = b * stride_ro_b + i * stride_ro_i + j * stride_ro_j

    # scalar mask (float)
    mask_val = tl.load(mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j)

    # load vectors
    left_proj = tl.load(left_proj_ptr + base_lp + h * stride_lp_h,
                       mask=h_mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + base_rp + h * stride_rp_h,
                        mask=h_mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + base_lg + h * stride_lg_h,
                       mask=h_mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + base_rg + h * stride_rg_h,
                        mask=h_mask, other=0.0)

    # fuse mask and gates
    left_out = left_proj * left_gate * mask_val
    right_out = right_proj * right_gate * mask_val

    # store results
    tl.store(left_out_ptr + base_lo + h * stride_lo_h,
             left_out, mask=h_mask)
    tl.store(right_out_ptr + base_ro + h * stride_ro_h,
             right_out, mask=h_mask)


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. LayerNorm on input
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # -----------------------------------------------------------------
    # 2. Linear projections + gates (float32)
    # -----------------------------------------------------------------
    left_proj = F.linear(x_norm, weights["left_proj.weight"])
    right_proj = F.linear(x_norm, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 3. Prepare mask (float32) – treat as all‑ones if no mask is required
    # -----------------------------------------------------------------
    if nomask or mask is None:
        mask_f = torch.ones_like(mask, dtype=dtype, device=device) if mask is not None else torch.ones(
            (x.shape[0], x.shape[1], x.shape[2]), dtype=dtype, device=device
        )
    else:
        mask_f = mask.to(dtype=dtype)

    # -----------------------------------------------------------------
    # 4. Fuse mask and gates via Triton kernel
    # -----------------------------------------------------------------
    B, N, _, _ = x.shape
    # allocate outputs
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # constants
    BLOCK_H = 64  # hidden‑dim tile size (covers up to 384)

    # compute grid dimensions
    total_ij = B * N * N
    grid = (total_ij, (hidden_dim + BLOCK_H - 1) // BLOCK_H)

    # launch kernel
    _apply_mask_and_gates_kernel[
        grid
    ](
        left_proj, right_proj,
        left_gate, right_gate,
        mask_f,
        left, right,
        # shape / strides
        B, N, hidden_dim,
        *left_proj.stride(),
        *right_proj.stride(),
        *left_gate.stride(),
        *right_gate.stride(),
        *mask_f.stride(),
        *left.stride(),
        *right.stride(),
        BLOCK_H=BLOCK_H,
    )

    # -----------------------------------------------------------------
    # 5. Batched triangular multiplication using torch.matmul
    # -----------------------------------------------------------------
    # left: [B,N,N,H]  -> permute to [B,H,N,N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: [B,N,N,H] -> permute to [B,H,N,N] with axes (k,j) for matmul
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    # batch matmul over the last two dims
    out_perm = torch.matmul(left_perm, right_perm)          # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()         # (B, N, N, H)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate and final projection
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
