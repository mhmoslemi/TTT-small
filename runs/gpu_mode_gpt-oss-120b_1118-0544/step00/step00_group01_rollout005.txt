import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Triton kernel for the core TriMul contraction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    left/right shapes: [B,N,N,hidden_dim]
    Output shape: [B,N,N,hidden_dim]
    """
    pid_bn = tl.program_id(0)   # combined batch and hidden idx
    pid_m = tl.program_id(1)    # block row (i)
    pid_n = tl.program_id(2)    # block col (j)

    # decode batch and hidden dimension
    b = pid_bn // hidden_dim
    d = pid_bn % hidden_dim

    # compute row and column indices
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (i,k)
        a_ptrs = left_ptr + b * stride_left_batch + d * stride_left_d \
                 + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile transposed: (k,j)
        b_ptrs = right_ptr + b * stride_right_batch + d * stride_right_d \
                 + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = out_ptr + b * stride_out_batch + d * stride_out_d \
             + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Fast forward pass of the TriMul (outgoing) operator.
    The heavy contraction `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`
    is performed by a custom Triton kernel; all other linear /
    LayerNorm / gating steps are done with PyTorch for simplicity.
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. LayerNorm over the last channel dimension
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # -----------------------------------------------------------------
    # 2. Linear projections + gating
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_proj = F.linear(x, left_proj_weight)               # [B,N,N,hidden_dim]
    right_proj = F.linear(x, right_proj_weight)

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left_proj * left_gate
    right = right_proj * right_gate

    if mask is not None:
        # mask shape [B,N,N] -> broadcast to hidden_dim
        mask_exp = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # 3. Core triangular multiplication (einsum) via Triton
    # -----------------------------------------------------------------
    batch_size, seq_len, _, _ = left.shape

    out = torch.empty_like(left)

    # strides in number of elements (not bytes)
    lsb, lsi, lsk, lsd = left.stride()
    rsb, rsi, rsk, rsd = right.stride()
    osb, osi, osj, osd = out.stride()

    # Choose block sizes (tuned for typical B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (batch_size * hidden_dim,
            (seq_len + BLOCK_M - 1) // BLOCK_M,
            (seq_len + BLOCK_N - 1) // BLOCK_N)

    tri_mul_matmul_kernel[grid](
        left, right, out,
        batch_size, seq_len, hidden_dim,
        lsb, lsi, lsk, lsd,
        rsb, rsi, rsk, rsd,
        osb, osi, osj, osd,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # -----------------------------------------------------------------
    # 4. Post‑processing: LayerNorm, out‑gate and final linear
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    out_norm = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                            bias=to_out_norm_bias, eps=eps)
    out_norm = out_norm * out_gate
    output = F.linear(out_norm, to_out_weight)

    return output

"""
TriMul (outgoing) forward pass.

The most expensive part is the three‑tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right are the gated projections of the input tensor.
We implement this contraction with a custom Triton kernel that performs a
batched matrix multiplication for every (batch, head) pair:
    out[d] = left[d] @ right[d].T   (shape (N,N) × (N,N) → (N,N))

All other operations (LayerNorm, linear projections, sigmoids, gating
and the final output projection) are carried out with PyTorch for
simplicity. The kernel works for any batch size ≤2, sequence length up
to 1024 and hidden dimension 128.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Kernel for: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Arguments:
        left_ptr, right_ptr, out_ptr : pointers to the tensors.
        B, N, H                     : batch size, sequence length, hidden dim.
        stride_*                    : strides (in elements) for each tensor.
        BLOCK_M, BLOCK_N, BLOCK_K   : tile sizes.
    """
    pid_bn = tl.program_id(0)          # combined batch * hidden index
    pid_m = tl.program_id(1)           # block row (i)
    pid_n = tl.program_id(2)           # block col (j)

    # decode batch and head index
    b = pid_bn // H
    h = pid_bn % H

    # row / column offsets inside the current block
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K) -> left[b,i,k,h]
        a_ptrs = (
            left_ptr
            + b * stride_left_batch
            + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile transposed: shape (BLOCK_K, BLOCK_N) -> right[b,j,k,h]ᵀ
        b_ptrs = (
            right_ptr
            + b * stride_right_batch
            + h * stride_right_h
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_N)

    # write result
    c_ptrs = (
        out_ptr
        + b * stride_out_batch
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor or None, shape [B, N, N]
        - weights      : dict of torch.Tensor containing all linear and norm params
        - config       : dict with keys ``dim`` and ``hidden_dim`` (and optionally ``nomask``)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] after the TriMul update.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (dim -> dim)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # -----------------------------------------------------------------
    # 2) Linear projections + gating
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    # projections
    left_proj = F.linear(x, left_proj_weight)      # [B,N,N,hidden_dim]
    right_proj = F.linear(x, right_proj_weight)

    # gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # optional mask (broadcast over hidden_dim)
    if mask is not None:
        mask_exp = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # 3) Core TriMul contraction via Triton
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty_like(left)

    # strides (in elements)
    lsb, lsi, lsk, lsh = left.stride()
    rsb, rsi, rsk, rsh = right.stride()
    osb, osi, osj, osh = out.stride()

    # tile sizes – tuned for typical B200 hardware
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                      # program_id(0) – batch * head
        (N + BLOCK_M - 1) // BLOCK_M,       # program_id(1) – i‑blocks
        (N + BLOCK_N - 1) // BLOCK_N,       # program_id(2) – j‑blocks
    )

    tri_mul_matmul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        lsb, lsi, lsk, lsh,
        rsb, rsi, rsk, rsh,
        osb, osi, osj, osh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 4) Post‑processing: LayerNorm over hidden_dim, out‑gate, final linear
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    out_norm = F.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight,
        bias=to_out_norm_bias, eps=eps
    )
    out_norm = out_norm * out_gate
    output = F.linear(out_norm, to_out_weight)

    return output
