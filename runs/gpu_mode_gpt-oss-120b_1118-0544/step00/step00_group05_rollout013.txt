"""
TriMul (outgoing) forward implementation with Triton.

The heavy part of the operation is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication for each hidden dimension `d`.
We fuse this O(N³·C) compute in a custom Triton kernel that
processes tiles of the output (i,j) while looping over the reduction axis k.
All other operations (LayerNorm, linear projections, gating) are performed
with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM per hidden dimension (left @ right^T)
# ----------------------------------------------------------------------
@triton.jit
def _trimal_outgoing_kernel(
    out_ptr, left_ptr, right_ptr,
    B, N, H,                          # batch size, seq length, hidden dim
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    The kernel iterates over tiles (i,j) of size BLOCK_M×BLOCK_N
    and reduces over k in steps of BLOCK_K.
    """
    pid_batch = tl.program_id(0)      # combined (b, h) index
    pid_i = tl.program_id(1)          # tile row
    pid_j = tl.program_id(2)          # tile col

    # Decompose combined batch index into (batch, hidden)
    b = pid_batch // H
    h = pid_batch % H

    # Base pointers for this (b,h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # Tile start positions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction axis k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_iter * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load left tile   (M x K)
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left_mask = (offs_i[:, None] < N) & (offs_k[None, :] < N)
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # Load right tile transposed (K x N) => we need right[j,k] but
        # accessed as (k, j) for the dot product
        right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right_mask = (offs_k[:, None] < N) & (offs_j[None, :] < N)
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # GEMM accumulate: (M×K) @ (K×N) -> (M×N)
        acc += tl.dot(left_tile, right_tile)

    # Write back result
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    out_mask = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(out_ptrs, acc, mask=out_mask)

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # Shape after projection: [B, N, N, hidden_dim]
    left = F.linear(x, left_proj_weight)   # -> (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 4. Apply mask if present
    # ------------------------------------------------------------------
    if mask is not None:
        # mask shape [B, N, N] -> broadcast over hidden_dim
        mask_exp = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 5. Apply element‑wise gating
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous layout for Triton
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 6. Triton kernel: out = left @ rightᵀ per hidden dimension
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # Strides (in number of elements)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()

    # Tile sizes – chosen to fit typical shared‑memory limits on A200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                     # combined batch & hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,       # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N        # j‑tiles
    )

    _trimal_outgoing_kernel[grid](
        out,
        left,
        right,
        B, N, hidden_dim,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Output LayerNorm (hidden_dim)
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # ------------------------------------------------------------------
    # 8. Final gating and linear projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # shape [B, N, N, dim]

    return out
```
