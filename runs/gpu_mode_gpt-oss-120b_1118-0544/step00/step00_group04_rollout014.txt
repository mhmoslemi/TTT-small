    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Flatten
    x_flat = x_norm.view(-1, dim)

    # Projections
    left_proj_weight = weights["left_proj.weight"] # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_flat, left_proj_weight) # (B*N*N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, right_proj_weight)

    # Reshape back
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

import triton
import triton.language as tl

@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # decode batch and hidden dim
    batch = pid_bh // hidden_dim
    d = pid_bh % hidden_dim

    # compute start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out-of-bound i,j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base pointers
    left_base = left_ptr + batch * stride_left_b + d * stride_left_d
    right_base = right_ptr + batch * stride_right_b + d * stride_right_d
    out_base = out_ptr + batch * stride_out_b + d * stride_out_d

    # pointers for i and j
    left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
    right_ptrs = right_base + offs_j[:, None] * stride_right_i + offs_k[None, :] * stride_right_k

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_iter = (N + BLOCK_K - 1) // BLOCK_K
    for it in range(num_iter):
        k_offset = it * BLOCK_K
        cur_k = k_offset + offs_k
        kmask = cur_k < N

        left = tl.load(left_ptrs + k_offset * stride_left_k, mask=mask_i[:, None] & kmask[None, :], other=0.0)

"""
custom_kernel for the “outgoing” TriMul operator.

Algorithm
---------
1. Layer‑normalize the input tensor `x` (shape [B, N, N, C]).
2. Linear projections (no bias) generate `left` and `right` tensors of shape
   [B, N, N, H] (`H = hidden_dim`).
3. (optional) Apply the pairwise mask.
4. Compute three gates (sigmoid of a linear projection of `x`) and multiply
   them element‑wise with `left`, `right` and later with the reduction result.
5. **Triton kernel** – compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   for every batch `b` and hidden dimension `h` using a blocked GEMM
   (block‑wise dot product).  The kernel handles out‑of‑bounds tiles and works
   in float32 accumulator while loading the inputs as float32.
6. Layer‑norm the reduction result, multiply by the output gate, and apply the
   final linear projection back to `dim`.
7. Return a float32 tensor of shape [B, N, N, dim].

The heavy O(N³) reduction is performed in the Triton kernel, all other
operations are done with PyTorch primitives.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def _tri_mul_reduce_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Grid: (num_i_blocks, num_j_blocks, B*H)
    pid_i = tl.program_id(0)      # block index along the i‑axis
    pid_j = tl.program_id(1)      # block index along the j‑axis
    pid_bh = tl.program_id(2)     # combined batch & hidden index

    batch = pid_bh // H
    hidden = pid_bh % H

    # Starting coordinates of the tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # 1‑D offsets within the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Boundary masks for i and j dimensions
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers that include batch and hidden offsets
    left_base  = left_ptr  + batch * stride_left_b  + hidden * stride_left_h
    right_base = right_ptr + batch * stride_right_b + hidden * stride_right_h
    out_base   = out_ptr   + batch * stride_out_b   + hidden * stride_out_h

    # Accumulator for the current (BLOCK_M x BLOCK_N) tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k in blocks of size BLOCK_K
    for k0 in range(0, N, BLOCK_K):
        cur_k = k0 + offs_k                # absolute k indices
        kmask = cur_k < N                  # valid k mask (1‑D)

        # Load left[i, k]   shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_base + offs_i[:, None] * stride_left_i
                     + cur_k[None, :]   * stride_left_k,
            mask=mask_i[:, None] & kmask[None, :],
            other=0.0
        )
        # Load right[j, k]  shape (BLOCK_N, BLOCK_K)
        right = tl.load(
            right_base + offs_j[:, None] * stride_right_i
                      + cur_k[None, :]   * stride_right_k,
            mask=mask_j[:, None] & kmask[None, :],
            other=0.0
        )
        # Accumulate dot(left, right) → (BLOCK_M, BLOCK_N)
        acc += tl.dot(left, right)

    # Store the tile back to global memory (apply i/j boundary mask)
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be dummy when nomask=True)
            - weights: dict with all model parameters (see code)
            - config: dict containing at least ``dim`` and ``hidden_dim`` and
                      optionally ``nomask`` (bool).

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # 1. Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2. LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    # Ensure contiguous layout for the upcoming view
    x = x.contiguous()

    # ------------------------------------------------------------------
    # 3. Linear projections (left/right) and gates
    # ------------------------------------------------------------------
    # Flatten the last dimension for a batched linear
    x_flat = x.view(-1, dim)                       # (B*N*N, dim)

    # Projections
    left_proj_w = weights["left_proj.weight"]      # (H, dim)
    right_proj_w = weights["right_proj.weight"]

    left = torch.nn.functional.linear(x_flat, left_proj_w)   # (B*N*N, H)
    right = torch.nn.functional.linear(x_flat, right_proj_w)

    left = left.view(B, N, N, hidden_dim).contiguous()
    right = right.view(B, N, N, hidden_dim).contiguous()

    # Optional mask (broadcast over hidden_dim)
    if not nomask:
        mask_exp = mask.unsqueeze(-1).type_as(left)   # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x_flat, left_gate_w).view(B, N, N, hidden_dim)
    right_gate = torch.nn.functional.linear(x_flat, right_gate_w).view(B, N, N, hidden_dim)
    out_gate = torch.nn.functional.linear(x_flat, out_gate_w).view(B, N, N, hidden_dim)

    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Triton kernel for the O(N³) reduction
    # ------------------------------------------------------------------
    # Output tensor before the final output projection
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Block sizes (tuned for typical B200 memory/clusters)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid dimensions
    num_i_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_j_blocks = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_i_blocks, num_j_blocks, B * hidden_dim)

    # Strides (in elements, not bytes)
    s_left = left.stride()   # (B_stride, i_stride, k_stride, h_stride)
    s_right = right.stride()
    s_out = out.stride()

    _tri_mul_reduce_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------
    # 5. Post‑reduction LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Apply the output gate (broadcast over the hidden dimension)
    out_norm = out_norm * out_gate

    # Final linear projection back to `dim`
    to_out_w = weights["to_out.weight"]            # (dim, hidden_dim)
    out_flat = out_norm.view(-1, hidden_dim)
    final = torch.nn.functional.linear(out_flat, to_out_w) \
                .view(B, N, N, dim)

    return final
