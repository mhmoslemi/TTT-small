```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_j, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    h = pid_h

    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_blocks):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers
        left_ptrs = left_ptr + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k + h * stride_left_h
        right_ptrs = right_ptr + offs_j[:, None] * stride_right_j + offs_k[None, :] * stride_right_k + h * stride_right_h

        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # compute partial product and sum over k
        # left: (BLOCK_I, BLOCK_K)
        # right: (BLOCK_J, BLOCK_K)
        # broadcast to (BLOCK_I, BLOCK_J, BLOCK_K)
        # sum over k axis=2
        prod = left[:, None, :] * right[None, :, :]  # shape (B_I, B_J, B_K)
        acc += tl.sum(prod, axis=2)

    # store result
    out_ptrs = out_ptr + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j + h * stride_out_h
    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=mask_out)

def tri_mul_forward(x, mask, weights, config):
    B,N,Ndim = x.shape ; dim = config["dim"]; hidden = config["hidden_dim"]
    # ensure shape matches
    # LayerNorm
    x_norm = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Projections
    left = torch.einsum('b i j d, h d -> b i j h', x_norm, weights["left_proj.weight"])
    right = torch.einsum('b i j d, h d -> b i j h', x_norm, weights["right_proj.weight"])

    # apply mask
    mask_exp = mask.unsqueeze(-1).to(x_norm.dtype)
    left = left * mask_exp
    right = right * mask_exp

    # Gates
    left_gate = torch.sigmoid(torch.einsum('b i j d, h d -> b i j h', x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.einsum('b i j d, h d -> b i j h', x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.einsum('b i j d, h d -> b i j h', x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Now compute einsum reduction using Triton kernel
    B, N, _, H = left.shape
    out = torch.empty((B, N, N, H), dtype=x.dtype, device=x.device)

    # constants
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    # For each batch:
    for b in range(B):
        left_b = left[b]   # shape (N,N,H)
        right_b = right[b]
        out_b = out[b]

        # Launch kernel
        grid = (
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J,
            H,
        )

            left_b,
            right_b,
            out_b,
            N, H,
            left_b.stride(0), left_b.stride(1), left_b.stride(2),

            out_b.stride(0), out_b.stride(1), out_b.stride(2),
            BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K,
        )
    # Now apply to_out_norm
    out = F.layer_norm(out, (hidden,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out * out_gate
    # final linear
    out = torch.einsum('b i j h, d h -> b i j d', out, weights["to_out.weight"])
    return out

"""
TriMul (outgoing) forward pass implemented with Triton.

The heavy part of the TriMul operator is the Einstein‑summation
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication over the K dimension for every
hidden channel d.  This kernel tiles the I and J dimensions (32×32) and
reduces over K in blocks of 32, accumulating in FP32.  The rest of the
module (LayerNorms, linear projections, and sigmoids) is performed with
PyTorch.  The kernel is launched once per batch and once per hidden
channel, handling arbitrary sequence lengths up to the limits used in the
tests.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: computes the pairwise reduction over the K dimension.
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,           # const float*   [i,k,d]
    right_ptr,          # const float*   [j,k,d]
    out_ptr,            # float*         [i,j,d]
    N,                  # int: sequence length
    H,                  # int: hidden dimension
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_j, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    h = pid_h

    # offsets for the current tile
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_blocks):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers for the current k‑block, channel h
        left_ptrs = left_ptr \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k \
            + h * stride_left_h

        right_ptrs = right_ptr \
            + offs_j[:, None] * stride_right_j \
            + offs_k[None, :] * stride_right_k \
            + h * stride_right_h

        # Load tiles (masked)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # (BLOCK_I, BLOCK_K)
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)          # (BLOCK_J, BLOCK_K)

        # Broadcast and accumulate: Σ_k left[i,k] * right[j,k]
        prod = left[:, None, :] * right[None, :, :]               # (BLOCK_I, BLOCK_J, BLOCK_K)
        acc += tl.sum(prod, axis=2)                               # (BLOCK_I, BLOCK_J)

    # Write the result
    out_ptrs = out_ptr \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j \
        + h * stride_out_h

    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=mask_out)


# ----------------------------------------------------------------------
# Entry point expected by the evaluator.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, C]
        - mask        : torch.Tensor of shape [B, N, N] (float or bool)
        - weights     : dict of model parameters
        - config      : dict with keys "dim", "hidden_dim", and optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. LayerNorm on the input (over the last dimension = dim)
    # --------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) for left/right and gates
    # --------------------------------------------------------------
    # shape helper: einsum "b i j d, h d -> b i j h"
    left = torch.einsum("b i j d, h d -> b i j h", x, weights["left_proj.weight"])
    right = torch.einsum("b i j d, h d -> b i j h", x, weights["right_proj.weight"])

    # apply mask if supplied (mask is [B,N,N])
    mask_exp = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # gates (sigmoid after linear)
    left_gate = torch.sigmoid(
        torch.einsum("b i j d, h d -> b i j h", x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.einsum("b i j d, h d -> b i j h", x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.einsum("b i j d, h d -> b i j h", x, weights["out_gate.weight"])
    )

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 3. Triton kernel for the pairwise reduction over K
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Blocking parameters (tuned for B200 GPUs)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    for b in range(B):
        # Views for a single batch – shape (N, N, H)
        left_b = left[b].contiguous()
        right_b = right[b].contiguous()
        out_b = out[b]

        grid = (
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J,
            H,
        )
        trimul_einsum_kernel[grid](
            left_b,
            right_b,
            out_b,
            N,
            H,
            left_b.stride(0),
            left_b.stride(1),
            left_b.stride(2),
            right_b.stride(0),
            right_b.stride(1),
            right_b.stride(2),
            out_b.stride(0),
            out_b.stride(1),
            out_b.stride(2),
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
        )
    # --------------------------------------------------------------
    # 4. Post‑processing: LayerNorm, out‑gate, final linear
    # --------------------------------------------------------------
    # Normalize over hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Apply the output gate
    out = out * out_gate

    # Final linear projection back to original dim
    out = torch.einsum("b i j h, d h -> b i j d", out, weights["to_out.weight"])
    return out
