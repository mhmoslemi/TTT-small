"""
TriMul (outgoing) implementation with a custom Triton kernel.

The kernel computes the contraction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
for all batches `b` and hidden channels `h`.  This is equivalent to a batched
matrix multiplication between `left[..., :, :, h]` and `right[..., :, :, h].T`
for each hidden dimension.  The rest of the module (LayerNorms, linear
projections, gating) is performed with PyTorch; only the O(N³) contraction is
off‑loaded to Triton.

The forward pass follows the reference PyTorch implementation:
    1. LayerNorm on the input.
    2. Linear projections (`left_proj`, `right_proj`).
    3. Optional mask application.
    4. Gating (`left_gate`, `right_gate`, `out_gate`).
    5. Triton kernel for the triple‑sum contraction.
    6. LayerNorm on the hidden dimension, gating, final linear projection.
"""

import torch
import triton
import triton.language as tl
import math
from typing import Tuple, Dict

# --------------------------------------------------------------
# Triton kernel: batched matmul for the TriMul contraction
# --------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    B, N, H,                            # dimensions
    stride_ab, stride_ai, stride_ak, stride_ah,   # strides for a (left)
    stride_bb, stride_bj, stride_bk, stride_bh,   # strides for b (right)
    stride_cb, stride_ci, stride_cj, stride_ch,   # strides for c (output)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_bh = tl.program_id(2)      # b * H + h
    pid_m = tl.program_id(0)       # block row (i)
    pid_n = tl.program_id(1)       # block col (j)

    # decode batch and hidden channel
    b = pid_bh // H
    h = pid_bh % H

    # compute offsets for this tile
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers with proper stride arithmetic
        a_ptrs = (
            a_ptr
            + b * stride_ab
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
            + h * stride_ah
        )
        b_ptrs = (
            b_ptr
            + b * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
            + h * stride_bh
        )

        # load tiles (mask out‑of‑bounds with zeros)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # accumulate dot product
        acc += tl.dot(a, b)

    # store result
    c_ptrs = (
        c_ptr
        + b * stride_cb
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
        + h * stride_ch
    )
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def _trimul_forward(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the TriMul contraction using the Triton kernel.

    Args:
        left:  Tensor of shape [B, N, N, H] (float32/float16).
        right: Tensor of same shape.

    Returns:
        out: Tensor of shape [B, N, N, H] (same dtype as inputs).
    """
    assert left.shape == right.shape
    B, N, N2, H = left.shape
    assert N == N2, "Pairwise dimensions must be equal."

    # Ensure contiguous layout for deterministic stride handling
    left = left.contiguous()
    right = right.contiguous()

    out = torch.empty_like(left)

    # Extract strides (in elements, not bytes)
    stride_ab, stride_ai, stride_ak, stride_ah = left.stride()
    stride_bb, stride_bj, stride_bk, stride_bh = right.stride()
    stride_cb, stride_ci, stride_cj, stride_ch = out.stride()

    # Tunable block sizes - chosen to fit typical B200 sweet spot
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_ab,
        stride_ai,
        stride_ak,
        stride_ah,
        stride_bb,
        stride_bj,
        stride_bk,
        stride_bh,
        stride_cb,
        stride_ci,
        stride_cj,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (bool or float)
        - weights      : dict of model weight tensors
        - config       : dict containing 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim dimension)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask application (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # Ensure mask has compatible dtype
        mask = mask.to(dtype)
        mask = mask.unsqueeze(-1)                # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear transforms of the normalized input)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. TriMul contraction via Triton
    # ------------------------------------------------------------------
    # left/right shape: [B, N, N, hidden_dim]
    out_hidden = _trimul_forward(left, right)  # still [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_hidden_norm = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    # Apply output gate (broadcasted over hidden dim)
    out_hidden_norm = out_hidden_norm * out_gate

    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out_hidden_norm, to_out_weight)  # [B,N,N,dim]

    return out
