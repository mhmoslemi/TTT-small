@triton.jit
def elemwise_gating_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK: tl.constexpr
):
    # compute linear offsets for each element
    offset = tl.program_id(0) * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offset < total
    if not tl.any(mask):
        return

    # Compute indices for masked entries only.
    # We'll compute b,i,j,h.
    idx = offset

"""
TriMul (outgoing) forward pass implemented with a small Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B,N,N,C]).
2. Linear projections to hidden_dim:
      left  = x_norm @ left_proj_weight.T
      right = x_norm @ right_proj_weight.T
   Gating values (sigmoid of linear layers) are also computed.
3. Triton kernel `mask_gate_kernel` fuses
      left  = left  * mask * left_gate
      right = right * mask * right_gate
   in‑place for all (b,i,j,hd) elements.
4. Compute the core “triangle” multiplication:
      out[b,i,j,hd] = Σ_k left[b,i,k,hd] * right[b,j,k,hd]
   This is realized by reshaping left/right so that a batched GEMM
   (`torch.bmm`) performs the sum over `k` for every hidden dimension.
5. Apply a second layer‑norm on the result, multiply by the output gate,
   and finish with a linear projection back to the original channel size.

Only the elementwise mask‑gate fusion is performed by Triton; everything else
uses highly‑optimized PyTorch/CuBLAS kernels.
"""

import torch
import triton
import triton.language as tl

# -----------------------------------------------------------------------------#
# Triton kernel: fuse mask, left/right gates, and in‑place write back.
# -----------------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    left_ptr, right_ptr,      # [B,N,N,HD] float32, in‑place output
    left_gate_ptr, right_gate_ptr,  # [B,N,N,HD] float32
    mask_ptr,                # [B,N,N] float32 (broadcasted)
    B, N, HD,                # compile‑time constants
    stride_left_b, stride_left_i, stride_left_j, stride_left_hd,
    stride_right_b, stride_right_i, stride_right_j, stride_right_hd,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_hd,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_hd,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK: tl.constexpr,
):
    # Linear index for this thread
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * HD
    mask = offsets < total
    if not tl.any(mask):
        return

    # Decode multi‑dimensional indices
    idx = offsets
    hd = idx % HD
    idx = idx // HD
    j = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # Convert to int64 for pointer arithmetic
    b = tl.cast(b, tl.int64)
    i = tl.cast(i, tl.int64)
    j = tl.cast(j, tl.int64)
    hd = tl.cast(hd, tl.int64)

    # Pointers to the elements we need
    left_off = (b * stride_left_b + i * stride_left_i +
                j * stride_left_j + hd * stride_left_hd)
    right_off = (b * stride_right_b + i * stride_right_i +
                 j * stride_right_j + hd * stride_right_hd)
    lgate_off = (b * stride_lgate_b + i * stride_lgate_i +
                 j * stride_lgate_j + hd * stride_lgate_hd)
    rgate_off = (b * stride_rgate_b + i * stride_rgate_i +
                 j * stride_rgate_j + hd * stride_rgate_hd)
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # Load values (masked load prevents OOB)
    left_val = tl.load(left_ptr + left_off, mask=mask)
    right_val = tl.load(right_ptr + right_off, mask=mask)
    mask_val = tl.load(mask_ptr + mask_off, mask=mask)

    lgate_val = tl.load(left_gate_ptr + lgate_off, mask=mask)
    rgate_val = tl.load(right_gate_ptr + rgate_off, mask=mask)

    # Apply mask then gating
    left_val = left_val * mask_val * lgate_val
    right_val = right_val * mask_val * rgate_val

    # Store results back in‑place
    tl.store(left_ptr + left_off, left_val, mask=mask)
    tl.store(right_ptr + right_off, right_val, mask=mask)

    B, N, N2, HD = left.shape

def layer_norm(x, weight, bias, eps=1e-5):
    """
    Simple LayerNorm implementation using torch ops.
    """
    # x: [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    if weight is not None:
        x_norm = x_norm * weight
    if bias is not None:
        x_norm = x_norm + bias
    return x_norm


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    # Extract config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # Ensure tensors are on CUDA and have correct dtype
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Prepare mask (broadcasted later)
    if nomask or mask_tensor is None:
        mask = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=dtype,
            device=device,
        )
    else:
        mask = mask_tensor.to(dtype=dtype)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------ #
    x_norm = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
    )  # shape [B,N,N,dim]

    # ------------------------------------------------------------------ #
    # 2. Linear projections and gates (bias‑free)
    # ------------------------------------------------------------------ #
    # Project to hidden_dim
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Gates (sigmoid of linear)
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    )  # shape [B,N,N,hidden_dim]

    # ------------------------------------------------------------------ #
    # 3. Apply mask and gates via Triton (in‑place on left/right)
    # ------------------------------------------------------------------ #
    apply_mask_and_gate(left, right, left_gate, right_gate, mask)

    # ------------------------------------------------------------------ #
    # 4. Core triangular multiplication using batched GEMM
    #    out[b,i,j,hd] = Σ_k left[b,i,k,hd] * right[b,j,k,hd]
    # ------------------------------------------------------------------ #
    B, N, _, HD = left.shape
    # Permute to bring hidden_dim to batch dimension for torch.bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B,HD,N,N] -> (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B,HD,N,N] -> (k,j)

    # Reshape to (B*HD, N, N) matrices
    left_mat = left_perm.view(B * HD, N, N)
    right_mat = right_perm.view(B * HD, N, N)

    # Batched matrix multiplication (sum over k)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*HD, N, N)

    # Reshape back to [B,N,N,HD]
    out = out_mat.view(B, HD, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 5. Output layer‑norm, gating, final linear projection
    # ------------------------------------------------------------------ #
    out_norm = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply output gate
    out_gated = out_norm * out_gate

    # Final linear projection back to original dim
    output = torch.nn.functional.linear(out_gated, weights["to_out.weight"])

    return output
