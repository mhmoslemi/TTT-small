@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K, H,
    stride_left_b, stride_left_h, stride_left_m, stride_left_k,
    stride_right_b, stride_right_h, stride_right_k, stride_right_n,
    stride_out_b, stride_out_h, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)   # combined batch*head index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute batch and head indices
    b = pid // H
    h = pid % H

    # Compute base pointers
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Offsets for M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Mask out-of-bounds for M,N
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block (M x K)
        a_ptrs = left_base + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B_T block (K x N) where B_T is right transposed

        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Multiply-accumulate
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = out_base + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n

"""
custom Triton implementation of the outgoing TriMul operator (AlphaFold3).

The heavy part – the N³ reduction 
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication in a Triton kernel.
All other operations (LayerNorm, linear projections, gating, final projection)
are performed in PyTorch for clarity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K, H,
    stride_left_b, stride_left_h, stride_left_m, stride_left_k,
    stride_right_b, stride_right_h, stride_right_k, stride_right_n,
    stride_out_b, stride_out_h, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid = tl.program_id(0)      # combined batch*head index
    pid_m = tl.program_id(1)    # block row
    pid_n = tl.program_id(2)    # block column

    # decode batch and head
    b = pid // H
    h = pid % H

    # base pointers for this (b,h) slice
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B_T block (right transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot‑product accumulation
        acc += tl.dot(a, b)

    # store the result tile
    c_ptrs = out_base + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def _layer_norm(x, weight, bias, eps=1e-5):
    """LayerNorm over the last dimension (no bias handling inside torch)."""
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    if weight is not None:
        x_norm = x_norm * weight
    if bias is not None:
        x_norm = x_norm + bias
    return x_norm


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, same device as inputs)
    """
    # ----------------------------------------------------------------------
    # unpack inputs
    # ----------------------------------------------------------------------
    x, mask, weights, cfg = data  # x: [B,N,N,D]

    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    device = x.device
    dtype = x.dtype

    # ----------------------------------------------------------------------
    # LayerNorm on the input
    # ----------------------------------------------------------------------
    x = _layer_norm(
        x,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ----------------------------------------------------------------------
    # Linear projections (no bias)
    # ----------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------------------
    # Apply mask if present (mask shape [B,N,N] -> broadcast over hidden_dim)
    # ----------------------------------------------------------------------
    if not nomask:
        mask = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------------------
    # Compute gates (sigmoid) and apply
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # Rearrange for the kernel: (B, hidden_dim, N, N)
    # ----------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape  # H == hidden_dim, _ == N again

    # output tensor before the final LayerNorm
    out_perm = torch.empty_like(left)  # (B, H, N, N), float32

    # ----------------------------------------------------------------------
    # Triton kernel launch
    # ----------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_ptr=out_perm.data_ptr(),
        M=N,
        N=N,
        K=N,
        H=H,
        stride_left_b=left.stride(0),
        stride_left_h=left.stride(1),
        stride_left_m=left.stride(2),
        stride_left_k=left.stride(3),
        stride_right_b=right.stride(0),
        stride_right_h=right.stride(1),
        stride_right_k=right.stride(3),  # k dimension in right (col)
        stride_right_n=right.stride(2),  # j dimension in right (row)
        stride_out_b=out_perm.stride(0),
        stride_out_h=out_perm.stride(1),
        stride_out_m=out_perm.stride(2),
        stride_out_n=out_perm.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ----------------------------------------------------------------------
    # Restore original layout: (B, N, N, H)
    # ----------------------------------------------------------------------
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,hidden_dim)

    # ----------------------------------------------------------------------
    # Post‑reduction LayerNorm, gating, and final projection
    # ----------------------------------------------------------------------
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate  # element‑wise gating (same shape as out)

    # final linear back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
