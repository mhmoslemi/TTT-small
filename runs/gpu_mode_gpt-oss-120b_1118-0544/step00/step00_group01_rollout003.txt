@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # each program processes a (batch_idx, block_m, block_n)
    # this kernel uses 1D grid where each pid maps to a block of output for a specific batch
    batch_idx = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_in_batch = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = pid_in_batch // tl.cdiv(N, BLOCK_N)
    pid_n = pid_in_batch % tl.cdiv(N, BLOCK_N)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for a and b
    a_ptrs = a_ptr + batch_idx * stride_a_batch + offs_m[:, None] * stride_a_m + (tl.arange(0, BLOCK_K))[None, :] * stride_a_k
    b_ptrs = b_ptr + batch_idx * stride_b_batch + (tl.arange(0, BLOCK_K))[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k*BLOCK_K+tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k*BLOCK_K+tl.arange(0, BLOCK_K) < K)[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # store
    c_ptrs = c_ptr + batch_idx * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B, N, N, C]).
2. Linear projections → `left` and `right` of shape [B, N, N, H].
3. Optional pairwise mask (broadcast on the feature dim).
4. Gating (sigmoid) on the projected tensors.
5. Core TriMul:
   out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   For each hidden channel this is a matrix multiplication
   L (N×N) @ Rᵀ (N×N).  We pack the batch and hidden dimensions,
   then run a batched GEMM kernel written in Triton.
6. Layer‑norm the result, apply the output gate and a final linear
   projection back to the original channel dimension.

Only the heavy N³ matrix multiplication is performed in Triton;
all other operations use efficient PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ B for a batch of matrices.
    A: (batch, M, K)   B: (batch, K, N)   C: (batch, M, N)
    """
    pid = tl.program_id(0)

    # Number of blocks per batch
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_pid_m * num_pid_n

    batch_idx = pid // blocks_per_batch
    pid_in_batch = pid % blocks_per_batch
    pid_m = pid_in_batch // num_pid_n
    pid_n = pid_in_batch % num_pid_n

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the first block of the tile
    a_tile_ptr = a_ptr + batch_idx * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_tile_ptr = b_ptr + batch_idx * stride_b_batch \
                 + tl.arange(0, BLOCK_K)[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        k_idx = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(a_tile_ptr,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0)
        b = tl.load(b_tile_ptr,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0)

        acc += tl.dot(a, b)

        a_tile_ptr += BLOCK_K * stride_a_k
        b_tile_ptr += BLOCK_K * stride_b_k

    # Store the result
    c_tile_ptr = c_ptr + batch_idx * stride_c_batch \
                 + offs_m[:, None] * stride_c_m \
                 + offs_n[None, :] * stride_c_n
    tl.store(c_tile_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using Triton.
    A: (B, M, K)   B: (B, K, N)   ->  C: (B, M, N)
    All tensors must be contiguous and on CUDA.
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = tl.cdiv(M, BLOCK_M)
    grid_n = tl.cdiv(N, BLOCK_N)
    total_blocks = batch * grid_m * grid_n

    _batched_matmul_kernel[total_blocks](
        A, B, C,
        batch, M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, C]
        - mask: torch.Tensor of shape [B, N, N] (may be None)
        - weights: dict of model parameters
        - config: dict containing 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor of shape [B, N, N, C]
    """
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Optional pairwise mask
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask = mask.to(dtype)  # broadcast to feature dim
        mask = mask.unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul (batched GEMM per hidden channel)
    # ------------------------------------------------------------------
    # Rearrange for GEMM: (B, H, N, N)   L @ Rᵀ
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (transpose k/j)

    B, H, N, _ = left_perm.shape
    A_flat = left_perm.view(B * H, N, N)      # (B*H, N, N)
    B_flat = right_perm.view(B * H, N, N)     # (B*H, N, N)

    # Batched GEMM using Triton
    out_flat = _batched_matmul_triton(A_flat, B_flat)  # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
