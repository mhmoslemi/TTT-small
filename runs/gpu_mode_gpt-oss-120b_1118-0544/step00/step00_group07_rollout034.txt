@triton.jit
def apply_gate_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_gb, stride_gi, stride_gk, stride_gh,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k,
    stride_outl_b, stride_outl_i, stride_outl_k, stride_outl_h,
    stride_outr_b, stride_outr_i, stride_outr_k, stride_outr_h,
    BLOCK: tl.constexpr):
    pid = tl.program_id(axis=0)
    total = B * N * N
    # mapping thread id to indices
    b = pid // (N * N)
    idx = pid % (N * N)
    i = idx // N
    k = idx % N
    # Process along hidden dim in blocks
    for h_offset in range(0, H, BLOCK):
        h = h_offset + tl.arange(0, BLOCK)
        mask = tl.load(mask_ptr + b * stride_m_b + i * stride_m_i + k * stride_m_k, mask=True, other=0.0)

        left_gate = tl.load(left_gate_ptr + b*stride_gb + i*stride_gi + k*stride_gk + h*stride_gh, mask=h < H, other=0.0)
        right = tl.load(right_ptr + b*stride_r_b + i*stride_r_i + k*stride_r_k + h*stride_r_h, mask=h < H, other=0.0)
        right_gate = tl.load(right_gate_ptr + b*stride_gb + i*stride_gi + k*stride_gk + h*stride_gh, mask=h < H, other=0.0)

    return out

"""
Triangle Multiplicative Update (TriMul) – “outgoing” version.

This implementation follows the reference PyTorch module but fuses the mask
application and the three gating multiplications into a single Triton kernel.
All linear projections, LayerNorms and the final linear projection are
performed with PyTorch ops, while the element‑wise gating kernel is executed
on the GPU to reduce memory traffic.
The core bilinear reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is computed with a single torch.einsum, which maps to a highly‑optimized
batched GEMM on the underlying CUDA library.
"""
import torch
import triton
import triton.language as tl


@triton.jit
def _apply_gate_kernel(
    # inputs
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    # outputs
    out_left_ptr, out_right_ptr,
    # dimensions
    B, N, H,
    # strides for left (B,N,N,H)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides for gates (same layout as left/right)
    stride_g_b, stride_g_i, stride_g_k, stride_g_h,
    # strides for right (B,N,N,H)
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    # strides for mask (B,N,N)
    stride_m_b, stride_m_i, stride_m_k,
    # strides for output tensors
    stride_outl_b, stride_outl_i, stride_outl_k, stride_outl_h,
    stride_outr_b, stride_outr_i, stride_outr_k, stride_outr_h,
    # compile‑time constant: size of the hidden dimension block
    BLOCK_H: tl.constexpr,
):
    """Fuse mask, left/right gates and write gated tensors."""
    pid = tl.program_id(0)                # one program per (b,i,k) entry
    total = B * N * N
    # decode (b,i,k) from linear pid
    b = pid // (N * N)
    idx = pid % (N * N)
    i = idx // N
    k = idx % N

    # hidden‑dimension offsets
    h = tl.arange(0, BLOCK_H)             # [BLOCK_H]

    # mask (scalar per (b,i,k))
    mask_off = b * stride_m_b + i * stride_m_i + k * stride_m_k
    mask_val = tl.load(mask_ptr + mask_off)   # scalar broadcasted later

    # compute base offsets (without hidden dim)
    off_l   = b * stride_l_b + i * stride_l_i + k * stride_l_k
    off_g   = b * stride_g_b + i * stride_g_i + k * stride_g_k
    off_r   = b * stride_r_b + i * stride_r_i + k * stride_r_k
    off_outl = b * stride_outl_b + i * stride_outl_i + k * stride_outl_k
    off_outr = b * stride_outr_b + i * stride_outr_i + k * stride_outr_k

    # bounds check for hidden dimension tail
    valid = h < H

    # load vectors of length BLOCK_H
    left      = tl.load(left_ptr     + off_l   + h * stride_l_h,
                       mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr+ off_g   + h * stride_g_h,
                       mask=valid, other=0.0)
    right     = tl.load(right_ptr    + off_r   + h * stride_r_h,
                       mask=valid, other=0.0)
    right_gate= tl.load(right_gate_ptr+ off_g   + h * stride_g_h,
                       mask=valid, other=0.0)

    # fused gated values
    left_f  = left  * mask_val * left_gate
    right_f = right * mask_val * right_gate

    # store results
    tl.store(out_left_ptr + off_outl + h * stride_outl_h, left_f,  mask=valid)
    tl.store(out_right_ptr+ off_outr + h * stride_outr_h, right_f, mask=valid)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict of torch.Tensors with all model parameters
            - config: dict containing "dim", "hidden_dim", "nomask" etc.

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm on the input ----------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ---------- Linear projections ----------
    left  = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # ---------- Gate values ----------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # ensure contiguity for Triton kernels
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # ---------- Mask handling ----------
    # If there is no mask (config["nomask"] or mask is None), use an all‑ones mask.
    if mask is None or config.get("nomask", False):
        mask_tensor = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask_tensor = mask.to(dtype).contiguous()

    # ---------- Allocate tensors for gated outputs ----------
    gated_left = torch.empty_like(left)
    gated_right = torch.empty_like(right)

    # ---------- Launch Triton kernel to apply mask & gates ----------
    total_programs = B * N * N
    grid = (total_programs,)

    _apply_gate_kernel[grid](
        left, left_gate,
        right, right_gate,
        mask_tensor,
        gated_left, gated_right,
        B, N, hidden_dim,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        mask_tensor.stride(0), mask_tensor.stride(1), mask_tensor.stride(2),
        gated_left.stride(0), gated_left.stride(1), gated_left.stride(2), gated_left.stride(3),
        gated_right.stride(0), gated_right.stride(1), gated_right.stride(2), gated_right.stride(3),
        BLOCK_H=hidden_dim,
        num_warps=4  # modest amount of warps for this kernel
    )

    # ---------- Core triangle multiplicative update ----------
    # out[b,i,j,h] = Σ_k gated_left[b,i,k,h] * gated_right[b,j,k,h]
    out = torch.einsum('b i k h, b j k h -> b i j h', gated_left, gated_right)

    # ---------- Post‑processing ----------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
