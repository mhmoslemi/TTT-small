    #    left_proj.weight: [hidden_dim, dim]

    #    right_gate.weight: [hidden_dim, dim]

```
@triton.jit
def fused_proj_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    HAS_MASK: tl.constexpr,
    BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return
    b = pid // (N * N)
    idx = pid % (N * N)
    i = idx // N
    j = idx % N

    offs = tl.arange(0, BLOCK_H)
    proj_ptrs = proj_ptr + b*stride_proj_b + i*stride_proj_i + j*stride_proj_j + offs*stride_proj_h
    gate_ptrs = gate_ptr + b*stride_gate_b + i*stride_gate_i + j*stride_gate_j + offs*stride_gate_h
    out_ptrs = out_ptr + b*stride_out_b + i*stride_out_i + j*stride_out_j + offs*stride_out_h

    proj = tl.load(proj_ptrs)
    gate_pre = tl.load(gate_ptrs)
    gate = 1.0 / (1.0 + tl.exp(-gate_pre))

    out = proj * gate
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + b*stride_mask_b + i*stride_mask_i + j*stride_mask_j)
        out = out * mask_val

    tl.store(out_ptrs, out)
```

"""
TriMul forward (outgoing) implementation.

Algorithm
---------
1. Input LayerNorm over the last dimension (dim).
2. Linear projections: left_proj, right_proj, left_gate, right_gate, out_gate.
3. Fuse three element‑wise ops in a Triton kernel:
      left  = left_proj  * sigmoid(left_gate)  * mask
      right = right_proj * sigmoid(right_gate) * mask
   (mask is ignored when config["nomask"] is True.)
4. Reduction over the pairwise dimension using a batched GEMM:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is performed as a bmm on tensors shaped (B*hidden, N, N).
5. LayerNorm on the reduced tensor (hidden_dim), multiply by out_gate.
6. Final linear projection back to the original dimension (dim).

Only step 3 is performed in a custom Triton kernel; all other
operations use highly‑optimized PyTorch routines.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------- #
# Triton kernel: fuse projection, gating (sigmoid) and optional mask #
# -------------------------------------------------------------------- #
@triton.jit
def fused_proj_gate_mask_kernel(
    proj_ptr,       # left/right projection          [B,N,N,H]
    gate_ptr,       # left/right gate pre‑sigmoid    [B,N,N,H]
    mask_ptr,       # mask tensor (B,N,N) or dummy  []
    out_ptr,        # output tensor (B,N,N,H)

    B, N, H,        # dimensions
    # strides (in elements)
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b,  stride_out_i,  stride_out_j,  stride_out_h,

    HAS_MASK: tl.constexpr,   # compile‑time flag
    BLOCK_H: tl.constexpr,    # hidden dimension (must equal H)
):
    pid = tl.program_id(0)                     # one program per (b,i,j)
    total = B * N * N
    if pid >= total:
        return

    b = pid // (N * N)
    idx = pid % (N * N)
    i = idx // N
    j = idx % N

    # vector of hidden indices processed by this program
    offs = tl.arange(0, BLOCK_H)

    # pointers to the slices we need
    proj = proj_ptr + b * stride_proj_b + i * stride_proj_i + j * stride_proj_j + offs * stride_proj_h
    gate = gate_ptr + b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + offs * stride_gate_h
    out  = out_ptr  + b * stride_out_b  + i * stride_out_i  + j * stride_out_j  + offs * stride_out_h

    # load and apply sigmoid
    proj_val = tl.load(proj)
    gate_pre = tl.load(gate)
    gate_sig = 1.0 / (1.0 + tl.exp(-gate_pre))

    result = proj_val * gate_sig

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j)
        result = result * mask_val

    tl.store(out, result)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ensure tensors are on the same device and correct dtype
    device = input_tensor.device
    dtype = input_tensor.dtype

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj   = F.linear(x, weights["left_proj.weight"])
    right_proj  = F.linear(x, weights["right_proj.weight"])
    left_gate_p = F.linear(x, weights["left_gate.weight"])
    right_gate_p= F.linear(x, weights["right_gate.weight"])
    out_gate_p  = F.linear(x, weights["out_gate.weight"])

    # -----------------------------------------------------------------
    # 3) Fuse projection, gating (sigmoid) and optional mask via Triton
    # -----------------------------------------------------------------
    # mask handling
    nomask = config.get("nomask", True)
    has_mask = (not nomask) and mask is not None
    if has_mask:
        mask = mask.to(dtype)        # ensure same dtype as tensors
    else:
        # dummy mask (not used when HAS_MASK=False)
        mask = torch.empty(0, device=device, dtype=dtype)

    B, N, _, _ = input_tensor.shape

    # allocate output tensors for the fused results
    left  = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # strides (in elements) for all tensors
    lproj_s = left_proj.stride()
    lgate_s = left_gate_p.stride()
    rproj_s = right_proj.stride()
    rgate_s = right_gate_p.stride()
    mask_s  = mask.stride() if has_mask else (0, 0, 0)
    out_s   = left.stride()   # left and right share same layout

    # compile‑time block size = hidden_dim (must be const for each launch)
    BLOCK_H = hidden_dim

    total_vectors = B * N * N
    grid = (total_vectors,)

    # left side
    fused_proj_gate_mask_kernel[grid](
        left_proj, left_gate_p, mask, left,
        B, N, hidden_dim,
        lproj_s[0], lproj_s[1], lproj_s[2], lproj_s[3],
        lgate_s[0], lgate_s[1], lgate_s[2], lgate_s[3],
        mask_s[0], mask_s[1], mask_s[2],
        out_s[0],  out_s[1],  out_s[2],  out_s[3],
        HAS_MASK=has_mask,
        BLOCK_H=BLOCK_H,
        num_warps=4,               # 128 threads → 4 warps
    )

    # right side
    fused_proj_gate_mask_kernel[grid](
        right_proj, right_gate_p, mask, right,
        B, N, hidden_dim,
        rproj_s[0], rproj_s[1], rproj_s[2], rproj_s[3],
        rgate_s[0], rgate_s[1], rgate_s[2], rgate_s[3],
        mask_s[0], mask_s[1], mask_s[2],
        out_s[0],  out_s[1],  out_s[2],  out_s[3],
        HAS_MASK=has_mask,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # out_gate sigmoid (no mask)
    out_gate = torch.sigmoid(out_gate_p)   # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # 4) Reduction over the pairwise dimension (batched GEMM)
    # -----------------------------------------------------------------
    # reshape to (B*hidden, N, N) for batched matrix multiply
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # left @ right^T  (sum over k)
    out_mat = torch.bmm(left_, right_.transpose(1, 2))  # (B*hidden, N, N)

    # reshape back to [B,N,N,hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5) Final LayerNorm, gating, and output projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
