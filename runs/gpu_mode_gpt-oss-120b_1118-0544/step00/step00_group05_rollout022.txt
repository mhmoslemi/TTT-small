    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    device = input_tensor.device
    dtype = input_tensor.dtype

    # Cast to fp16

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

import torch
import triton
import triton.language as tl

@triton.jit
def fused_mul3_kernel(a_ptr, b_ptr, c_ptr, out_ptr, n_elems,
                      BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elems
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    c = tl.load(c_ptr + offs, mask=mask)
    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul (outgoing) operator.
    The kernel fuses element‑wise masking and gating, then uses
    batched GEMM (torch.bmm) to compute the contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    followed by layer‑norm, gating and a final linear projection.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # Ensure tensors are on GPU
    assert device.type == "cuda", "Tensor must be on CUDA device"
    # Use float32 for correctness
    x = input_tensor.to(torch.float32)

    # ---- LayerNorm on the last dimension ----
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # ---- Linear projections ----
    left_proj_weight = weights["left_proj.weight"].to(torch.float32)
    right_proj_weight = weights["right_proj.weight"].to(torch.float32)
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ---- Gating (sigmoid) ----
    left_gate_weight = weights["left_gate.weight"].to(torch.float32)
    right_gate_weight = weights["right_gate.weight"].to(torch.float32)
    out_gate_weight = weights["out_gate.weight"].to(torch.float32)

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # ---- Mask handling ----
    if mask is None:
        mask = torch.ones(x.shape[:3], device=device, dtype=torch.float32)
    else:
        mask = mask.to(torch.float32)
    mask = mask.unsqueeze(-1)  # [B,N,N,1]

    # ---- Fuse elementwise multiplication (left = left * mask * left_gate,
    #                                   right = right * mask * right_gate) ----
    total_el = left.numel()
    # allocate output buffers
    left_fused = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    BLOCK_SIZE = 1024
    grid = lambda meta: (triton.cdiv(total_el, meta['BLOCK_SIZE']),)
    # launch for left
    fused_mul3_kernel[grid](
        left.view(-1), mask.view(-1), left_gate.view(-1),
        left_fused.view(-1), total_el,
        BLOCK_SIZE=BLOCK_SIZE
    )
    # launch for right
    fused_mul3_kernel[grid](
        right.view(-1), mask.view(-1), right_gate.view(-1),
        right_fused.view(-1), total_el,
        BLOCK_SIZE=BLOCK_SIZE
    )

    # ---- Batched matrix multiplication over the contracted dimension k ----
    # reshape to [B, hidden_dim, N, N]
    left_perm = left_fused.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fused.permute(0, 3, 1, 2).contiguous()
    # transpose the second operand on the last two axes
    right_t = right_perm.transpose(-1, -2).contiguous()

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_t.view(B * H, N, N)

    # Using torch.bmm for the batched GEMM
    out_mat = torch.bmm(left_mat, right_mat)  # [B*H, N, N]

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden_dim]

    # ---- Output layernorm (hidden_dim) ----
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,),
                                         weight=to_out_norm_weight,
                                         bias=to_out_norm_bias)

    # ---- Apply output gate ----
    out = out * out_gate

    # ---- Final linear projection back to dim ----
    to_out_weight = weights["to_out.weight"].to(torch.float32)
    out = torch.nn.functional.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
```

"""
TriMul (outgoing) forward pass with Triton acceleration.

The kernel fuses the element‑wise multiplication of the projected tensor with
its gate (after applying the optional mask).  The costly contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is performed as a batched GEMM using torch.bmm (leveraging cuBLAS).  The
remaining operations (layer‑norms, gates, final projection) are expressed with
standard PyTorch ops.  The final output is float32.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def mul2_kernel(a_ptr, b_ptr, out_ptr, n_elems,
                BLOCK_SIZE: tl.constexpr):
    """Element‑wise multiplication a * b → out (1‑D flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elems
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul2(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton mul2 kernel on two equally‑shaped float32 tensors."""
    assert a.shape == b.shape, "Shapes must match for elementwise mul2."
    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)  # 1‑D launch grid
    mul2_kernel[grid](
        a.view(-1), b.view(-1), out.view(-1), total,
        BLOCK_SIZE=BLOCK,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask:        torch.Tensor of shape [B, N, N] (bool or float)
            - weights:     dict of model parameters (see code)
            - config:      dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32).
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    assert device.type == "cuda", "custom_kernel requires CUDA tensors."

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input channel dimension (dim)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)
    x = torch.nn.functional.layer_norm(
        input_tensor.to(torch.float32),
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float32)
    right_proj_w = weights["right_proj.weight"].to(torch.float32)

    left = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B, N, N, hidden]

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dimension)
    # -----------------------------------------------------------------
    if mask is None:
        mask = torch.ones(x.shape[:3], device=device, dtype=torch.float32)
    else:
        mask = mask.to(torch.float32)
    mask = mask.unsqueeze(-1)               # [B, N, N, 1]
    left = (left * mask).contiguous()
    right = (right * mask).contiguous()

    # -----------------------------------------------------------------
    # 4. Gating values (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float32)
    right_gate_w = weights["right_gate.weight"].to(torch.float32)
    out_gate_w = weights["out_gate.weight"].to(torch.float32)

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # -----------------------------------------------------------------
    # 5. Fuse (left * left_gate) and (right * right_gate) with Triton
    # -----------------------------------------------------------------
    left = _elemwise_mul2(left, left_gate)   # still [B, N, N, hidden]
    right = _elemwise_mul2(right, right_gate)

    # -----------------------------------------------------------------
    # 6. Contract over the third dimension (k) via batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -----------------------------------------------------------------
    # reshape to (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # transpose the second operand on its last two axes (k ↔ j)
    right_t = right_perm.transpose(-1, -2).contiguous()

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)      # (B*H, N, N)
    right_mat = right_t.view(B * H, N, N)      # (B*H, N, N)

    # Batched matrix multiplication using cuBLAS (torch.bmm)
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N)

    # reshape back to [B, N, N, hidden]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7. Output LayerNorm (over hidden dimension)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float32)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 8. Apply output gate
    # -----------------------------------------------------------------
    out = out * out_gate

    # -----------------------------------------------------------------
    # 9. Final linear projection back to `dim`
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float32)
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim]

    # Ensure float32 output
    return out.to(torch.float32)
