import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_bh, stride_i, stride_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # offsets for output tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer for this (b, h) slice
    base = pid_bh * stride_bh

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k]
        left_ptrs = left_ptr + base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right[k,j] (notice the transpose)
        right_ptrs = right_ptr + base + offs_k[:, None] * stride_i + offs_j[None, :] * stride_k
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left, right)

    out_ptrs = out_ptr + base + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=mask_out)

    """
    Triton-accelerated outgoing Triangle Multiplicative Update (TriMul).
    Performs the heavy pairwise reduction   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    using a custom Triton kernel, while the surrounding linear layers,
    LayerNorms and gated activations are handled by PyTorch.
    """
    input_tensor, mask_tensor, weights, config = data
    # Ensure torch no_grad for speed
    torch.set_grad_enabled(False)

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm on the input (last dim)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Optional mask (broadcast)
    if not nomask:
        mask = mask_tensor.to(input_tensor.dtype).unsqueeze(-1)  # shape [B, N, N, 1]
        left = left * mask
        right = right * mask

    # Gated activations
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Prepare tensors for Triton: move hidden_dim to second dimension
    # shape after permute: [B, hidden_dim, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, N2 = left.shape
    assert N == N2, "Sequence length dimensions must match"

    # Flatten (B, H) into a single dimension to simplify kernel indexing
    left_flat = left.reshape(-1, N, N).contiguous()
    right_flat = right.reshape(-1, N, N).contiguous()
    out_flat = torch.empty_like(left_flat)

    # Strides for the flattened tensors (all three have the same layout)
    stride_bh = left_flat.stride(0)   # stride between different (b,h) slices (= N*N)
    stride_i = left_flat.stride(1)    # stride for row index i (= N)
    stride_k = left_flat.stride(2)    # stride for column index k (= 1)

    total_bh = left_flat.shape[0]

    # Choose block sizes. 64 works well for typical N up to 1024.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_bh,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_bh,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to original 4‑D shape [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm over hidden_dim
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # Gating
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,         # *float32   left tensor  (flattened B*H, N, N)
    right_ptr,        # *float32   right tensor (flattened B*H, N, N)
    out_ptr,          # *float32   output tensor (flattened B*H, N, N)
    N,                # int32      sequence length
    stride_bh,        # int32      stride between (b,h) slices (= N*N)
    stride_i,         # int32      stride for row index i (= N)
    stride_k,         # int32      stride for column index k (= 1)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication with a transposed RHS:
    out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
    where the right operand is logically transposed (k,j) in memory.
    """
    pid_bh = tl.program_id(0)                 # flattened (batch * head) dimension
    pid_i = tl.program_id(1)                  # block index along the i axis
    pid_j = tl.program_id(2)                  # block index along the j axis

    # Offsets for the i and j dimensions of the output tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointer for this (b,h) slice
    base = pid_bh * stride_bh

    # Accumulator in FP32 (even if inputs are FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for kk in range(num_k):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k] tile
        left_ptrs = (
            left_ptr
            + base
            + offs_i[:, None] * stride_i
            + offs_k[None, :] * stride_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right[k,j] tile – note the transposition (k as row, j as column)
        right_ptrs = (
            right_ptr
            + base
            + offs_k[:, None] * stride_i   # row index = k uses stride_i
            + offs_j[None, :] * stride_k   # column index = j uses stride_k
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # Store the result
    out_ptrs = (
        out_ptr
        + base
        + offs_i[:, None] * stride_i
        + offs_j[None, :] * stride_k
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).
    Implements the heavy reduction    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    via a custom Triton kernel. All surrounding LayerNorms, linear projections,
    and gated activations are performed with PyTorch.
    """
    input_tensor, mask_tensor, weights, config = data

    torch.set_grad_enabled(False)  # forward only

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- LayerNorm on the input ----------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- Linear projections (no bias) ----------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if not nomask:
        mask = mask_tensor.to(input_tensor.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gating ----------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare for Triton (move hidden_dim forward) ----------
    # Shapes: [B, N, N, hidden_dim] -> [B, hidden_dim, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, N2 = left.shape
    assert N == N2, "Inconsistent sequence dimensions"

    # Flatten (B, H) into one dimension for simplified kernel launching
    left_flat = left.reshape(-1, N, N).contiguous()
    right_flat = right.reshape(-1, N, N).contiguous()
    out_flat = torch.empty_like(left_flat)

    # Strides for the flattened tensors (identical for all three)
    stride_bh = left_flat.stride(0)   # = N*N
    stride_i = left_flat.stride(1)    # = N
    stride_k = left_flat.stride(2)    # = 1

    total_bh = left_flat.shape[0]     # = B * hidden_dim

    # Block sizes – 64 works well for a wide range of N (up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_bh,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # ---------- Launch Triton kernel ----------
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_bh,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- Output LayerNorm, gating and final projection ----------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
