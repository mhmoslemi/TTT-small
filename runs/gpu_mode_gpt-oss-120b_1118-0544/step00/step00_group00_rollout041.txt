def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # retrieve weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    device = input_tensor.device

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Mask broadcast if mask is not None and if mask dtype?
    mask_broadcast = mask.unsqueeze(-1).to(x_norm.dtype)  # shape [B,N,N,1]
    left = left * mask_broadcast
    right = right * mask_broadcast

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Allocate output buffer (float32)
    out_fp32 = torch.empty((left_fp16.shape[0], left_fp16.shape[1], left_fp16.shape[2], hidden_dim), dtype=torch.float32, device=device)

    # Launch kernel
    B, N, _, _ = left_fp16.shape  # B,N,N,hidden
    # grid
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_D = 32

    num_blocks_M = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_N = (N + BLOCK_N - 1) // BLOCK_N
    num_blocks_D = (hidden_dim + BLOCK_D - 1) // BLOCK_D
    total_blocks = B * num_blocks_M * num_blocks_N * num_blocks_D
    grid = (total_blocks,)

    # Get strides
    stride_lb = left_fp16.stride(0)
    stride_li = left_fp16.stride(1)
    stride_lk = left_fp16.stride(2)
    stride_ld = left_fp16.stride(3)

    stride_rb = right_fp16.stride(0)
    stride_rj = right_fp16.stride(1)
    stride_rk = right_fp16.stride(2)
    stride_rd = right_fp16.stride(3)

    stride_ob = out_fp32.stride(0)
    stride_oi = out_fp32.stride(1)
    stride_oj = out_fp32.stride(2)
    stride_od = out_fp32.stride(3)

    tri_mul_kernel[grid](
        left_fp16, right_fp16, out_fp32,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
    )

    # Continue with post-processing
    out = out_fp32

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # gating
    out = out * out_gate

    # final linear projection
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    # compute block indices
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    num_blocks_d = tl.cdiv(D, BLOCK_D)
    total_blocks_per_batch = num_blocks_m * num_blocks_n * num_blocks_d
    b = pid // total_blocks_per_batch
    tmp = pid % total_blocks_per_batch
    i_block = tmp // (num_blocks_n * num_blocks_d)

    j_block = tmp2 // num_blocks_d
    d_block = tmp2 % num_blocks_d

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    """
    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
    Blocked implementation over i, j, k, d dimensions using TensorCore dot products.
    """
    pid = tl.program_id(0)

    # number of blocks along each dimension
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    num_blocks_d = (D + BLOCK_D - 1) // BLOCK_D
    blocks_per_batch = num_blocks_m * num_blocks_n * num_blocks_d

    b = pid // blocks_per_batch
    tmp = pid % blocks_per_batch
    i_block = tmp // (num_blocks_n * num_blocks_d)
    tmp2 = tmp % (num_blocks_n * num_blocks_d)
    j_block = tmp2 // num_blocks_d
    d_block = tmp2 % num_blocks_d

    # offsets for i, j, d
    offs_i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    base_d = d_block * BLOCK_D
    offs_d = base_d + tl.arange(0, BLOCK_D)

    # masks for out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # number of k‑tiles
    num_k_tiles = (N + BLOCK_K - 1) // BLOCK_K

    for k_tile in range(num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # masks for the two loads (i‑k and j‑k slices)
        mask_ik = mask_i[:, None] & mask_k[None, :]
        mask_jk = mask_j[:, None] & mask_k[None, :]

        # iterate over the d‑dimension within the block
        for d_local in range(0, BLOCK_D):
            d = base_d + d_local
            d_valid = d < D

            # left: shape (M, K)
            left_ptrs = left_ptr + b * stride_lb \
                        + offs_i[:, None] * stride_li \
                        + offs_k[None, :] * stride_lk \
                        + d * stride_ld
            left = tl.load(left_ptrs,
                           mask=mask_ik & d_valid,
                           other=0.0).to(tl.float32)

            # right transposed: shape (K, N)
            right_ptrs = right_ptr + b * stride_rb \
                         + offs_k[:, None] * stride_rk \
                         + offs_j[None, :] * stride_rj \
                         + d * stride_rd
            right = tl.load(right_ptrs,
                            mask=mask_jk & d_valid,
                            other=0.0).to(tl.float32)

            # matrix‑multiply‑accumulate
            acc[:, :, d_local] += tl.dot(left, right)   # (M, N)

    # store results, respecting i/j/d bounds
    mask_ij = mask_i[:, None] & mask_j[None, :]
    for d_local in range(0, BLOCK_D):
        d = base_d + d_local
        d_valid = d < D
        out_ptrs = out_ptr + b * stride_ob \
                   + offs_i[:, None] * stride_oi \
                   + offs_j[None, :] * stride_oj \
                   + d * stride_od
        tl.store(out_ptrs,
                 acc[:, :, d_local],
                 mask=mask_ij & d_valid)

def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” update.
    Implements the expensive einsum reduction with a custom Triton kernel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # -------------------- unpack weights --------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -------------------- layer‑norm --------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # -------------------- linear projections --------------------
    left = F.linear(x, left_proj_weight)          # [B, N, N, H]
    right = F.linear(x, right_proj_weight)

    # -------------------- mask (broadcast) --------------------
    # mask shape: [B, N, N]; expand to last dim
    mask_ = mask.unsqueeze(-1).to(x.dtype)
    left = left * mask_
    right = right * mask_

    # -------------------- gating --------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -------------------- Triton reduction (i‑j‑k) --------------------
    # cast to fp16 for better bandwidth; kernel accumulates in fp32
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape
    out_fp32 = torch.empty((B, N, N, hidden_dim),
                           dtype=torch.float32,
                           device=device)

    # -------------------- launch kernel --------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_D = 32

    num_blocks_M = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_N = (N + BLOCK_N - 1) // BLOCK_N
    num_blocks_D = (hidden_dim + BLOCK_D - 1) // BLOCK_D
    total_blocks = B * num_blocks_M * num_blocks_N * num_blocks_D
    grid = (total_blocks,)

    stride_lb = left_fp16.stride(0)
    stride_li = left_fp16.stride(1)
    stride_lk = left_fp16.stride(2)
    stride_ld = left_fp16.stride(3)

    stride_rb = right_fp16.stride(0)
    stride_rj = right_fp16.stride(1)
    stride_rk = right_fp16.stride(2)
    stride_rd = right_fp16.stride(3)

    stride_ob = out_fp32.stride(0)
    stride_oi = out_fp32.stride(1)
    stride_oj = out_fp32.stride(2)
    stride_od = out_fp32.stride(3)

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # -------------------- post‑processing --------------------
    out = out_fp32

    # Layer‑norm on the reduced tensor
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to original dimension
    out = F.linear(out, to_out_weight)

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,               # const float16 *
    right_ptr,              # const float16 *
    out_ptr,                # float32 *
    B, N, D,                # tensor dimensions
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    Compute the core TriMul reduction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    The kernel blocks over the i, j and hidden‑dim (d) axes and
    reduces over k using Tensor‑Core dot products.  Accumulation
    happens in float32, while the inputs are stored in float16
    (the usual mixed‑precision trade‑off for speed).
    """
    pid = tl.program_id(0)

    # -------- grid decomposition ---------------------------------
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    num_blocks_d = (D + BLOCK_D - 1) // BLOCK_D
    blocks_per_batch = num_blocks_m * num_blocks_n * num_blocks_d

    b = pid // blocks_per_batch
    tmp = pid % blocks_per_batch
    i_block = tmp // (num_blocks_n * num_blocks_d)
    tmp2 = tmp % (num_blocks_n * num_blocks_d)
    j_block = tmp2 // num_blocks_d
    d_block = tmp2 % num_blocks_d

    # -------- offsets for this program ---------------------------
    offs_i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    base_d = d_block * BLOCK_D
    offs_d = base_d + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # -------- accumulator (float32) ------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # -------- k‑tiling loop ------------------------------------
    num_k_tiles = (N + BLOCK_K - 1) // BLOCK_K
    for k_tile in range(num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        mask_ik = mask_i[:, None] & mask_k[None, :]   # (M, K)
        mask_jk = mask_j[:, None] & mask_k[None, :]   # (N, K)

        # -------- loop over hidden dimension inside the block ----
        for d_local in range(0, BLOCK_D):
            d = base_d + d_local
            d_valid = d < D

            # left[b, i, k, d]  -> (M, K)
            left_ptrs = (
                left_ptr
                + b * stride_lb
                + offs_i[:, None] * stride_li
                + offs_k[None, :] * stride_lk
                + d * stride_ld
            )
            left = tl.load(left_ptrs, mask=mask_ik & d_valid, other=0.0).to(tl.float32)

            # right[b, j, k, d]   (need k as first dim for dot)
            # we load the transposed view: (K, N)
            right_ptrs = (
                right_ptr
                + b * stride_rb
                + offs_k[:, None] * stride_rk
                + offs_j[None, :] * stride_rj
                + d * stride_rd
            )
            right = tl.load(right_ptrs, mask=mask_jk & d_valid, other=0.0).to(tl.float32)

            # matrix‑multiply‑accumulate
            acc[:, :, d_local] += tl.dot(left, right)   # (M, N)

    # -------- write back -------------------------------------------------
    mask_ij = mask_i[:, None] & mask_j[None, :]
    for d_local in range(0, BLOCK_D):
        d = base_d + d_local
        d_valid = d < D
        out_ptrs = (
            out_ptr
            + b * stride_ob
            + offs_i[:, None] * stride_oi
            + offs_j[None, :] * stride_oj
            + d * stride_od
        )
        tl.store(out_ptrs, acc[:, :, d_local], mask=mask_ij & d_valid)

    stride_lk = left_fp16.stride(2)
    stride_ld = left_fp16.stride(3)

    stride_rj = right_fp16.stride(1)
    stride_rk = right_fp16.stride(2)
    stride_rd = right_fp16.stride(3)

    stride_oi = out_fp32.stride(1)
    stride_oj = out_fp32.stride(2)
    stride_od = out_fp32.stride(3)

        right_fp16,

        hidden_dim,

        stride_lk,
        stride_ld,
