def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # LayerNorm: x_norm = (x - mean)/sqrt(var+eps) * weight + bias
    # We can use torch.nn.functional.layer_norm:
    x_norm = torch.nn.functional.layer_norm(
        x, normalized_shape=(dim,),
        weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps
    )
    # Project left/right
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])  # shape [B,N,N,hidden]
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # Apply mask if provided (if mask is not None)
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)  # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # Apply gates & mask multiplication already applied
    left = left * left_gate
    right = right * right_gate

    # Permute for GEMM: (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    import triton.language as tl

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    B_H: tl.constexpr,
    N: tl.constexpr,
    stride_a_bh, stride_a_m, stride_a_k,
    stride_b_bh, stride_b_n, stride_b_k,
    stride_c_bh, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Load A block [M,K]
        a_ptrs = a_ptr + pid_bh * stride_a_bh + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k

import torch
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    BxH, N,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load block of A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + pid_bh * stride_a_batch + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
        a_mask = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)

        # Load block of B transposed: we want B_T with shape (BLOCK_K, BLOCK_N)
        # B in memory is [batch, N, N]; we load B at (k, n)
        b_ptrs = b_ptr + pid_bh * stride_b_batch + offs_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col
        b_mask = (offs_k[:, None] < N) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        # Compute dot product

    # Store result

    c_mask = (offs_m[:, None] < N) & (offs_n[None, :] < N)

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    # define kernel inside function
    @triton.jit
    def batched_gemm_kernel( ... ): ...

    # unpack data
    input_tensor, mask, weights, config = data
    # dims
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)   # if True, skip mask

    # LayerNorm on last dim
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps
    )

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if needed
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x_norm.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # prepare for reduction
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch and head dimensions
    B = bs
    H = hidden_dim
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)
    out_flat = torch.empty_like(left_flat)  # shape [B*H, N, N]

    # compute strides for kernel
    stride_a_batch = left_flat.stride(0)
    stride_a_row = left_flat.stride(1)
    stride_a_col = left_flat.stride(2)
    stride_b_batch = right_flat.stride(0)
    stride_b_row = right_flat.stride(1)
    stride_b_col = right_flat.stride(2)
    stride_c_batch = out_flat.stride(0)
    stride_c_row = out_flat.stride(1)
    stride_c_col = out_flat.stride(2)

    # decide block sizes based on N
    BLOCK = 64
    BLOCK_M = BLOCK_N = BLOCK_K = BLOCK

    # launch kernel
    grid = (B * H, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

        left_flat,
        right_flat,
        out_flat,

    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

import triton.language as tl

def custom_kernel(data):
    """
    Triton-accelerated forward pass for the TriMul 'outgoing' operator.
    The kernel performs the core N‑wise reduction `out[i,j] = Σ_k left[i,k] * right[j,k]`
    using a batched GEMM (A @ Bᵀ) over the combined batch‑head dimension.
    All other operations (LayerNorm, linear projections, gating, final projection)
    are performed with PyTorch for simplicity.
    """
    # unpack input
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # -------------------------------------------------
    # 1. LayerNorm on last dimension
    # -------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3. Optional mask
    # -------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # -------------------------------------------------
    # 4. Gating
    # -------------------------------------------------
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5. Batched GEMM: left @ rightᵀ  (N³ reduction)
    # -------------------------------------------------
    # reshape to (B, H, N, N) and then merge B and H
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B = bs
    H = hidden_dim
    left_flat = left_perm.view(B * H, N, N)   # (BH, N, N)
    right_flat = right_perm.view(B * H, N, N) # (BH, N, N)
    out_flat = torch.empty_like(left_flat)    # (BH, N, N)

    # strides (int)
    stride_a_batch = left_flat.stride(0)
    stride_a_row   = left_flat.stride(1)
    stride_a_col   = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_row   = right_flat.stride(1)   # original row stride (j)
    stride_b_col   = right_flat.stride(2)   # original col stride (k)

    stride_c_batch = out_flat.stride(0)
    stride_c_row   = out_flat.stride(1)
    stride_c_col   = out_flat.stride(2)

    # ------------------------------------------------------------------
    # Triton kernel: batched GEMM A @ Bᵀ
    # ------------------------------------------------------------------
    @triton.jit
    def batched_gemm_kernel(
        a_ptr, b_ptr, c_ptr,
        batch, N,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
    ):
        pid_bh = tl.program_id(0)          # index for merged batch‑head
        pid_m  = tl.program_id(1)          # tile row
        pid_n  = tl.program_id(2)          # tile col

        # coordinate offsets for this tile
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        # accumulator in FP32
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # iterate over K dimension
        for k in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

            # ---- A block (M x K) ----
            a_ptrs = a_ptr + pid_bh * stride_a_batch + \
                     offs_m[:, None] * stride_a_row + \
                     offs_k[None, :] * stride_a_col
            a_mask = (offs_m[:, None] < N) & (offs_k[None, :] < N)
            a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)

            # ---- Bᵀ block (K x N) ----
            # NOTE: we load B with swapped strides to obtain the transpose on‑the‑fly
            b_ptrs = b_ptr + pid_bh * stride_b_batch + \
                     offs_k[:, None] * stride_b_col + \
                     offs_n[None, :] * stride_b_row
            b_mask = (offs_k[:, None] < N) & (offs_n[None, :] < N)
            b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

            acc += tl.dot(a, b)

        # write back result
        c_ptrs = c_ptr + pid_bh * stride_c_batch + \
                 offs_m[:, None] * stride_c_row + \
                 offs_n[None, :] * stride_c_col
        c_mask = (offs_m[:, None] < N) & (offs_n[None, :] < N)
        tl.store(c_ptrs, acc, mask=c_mask)

    # Choose block sizes (tuned for typical N up to 1024)
    BLOCK = 64
    BLOCK_M = BLOCK_N = BLOCK_K = BLOCK

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden_dim)
    out_hidden = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # -------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output
