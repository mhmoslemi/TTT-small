class TriMulTriton(nn.Module):
    def __init__(self, input_dim, hidden_dim, config):
        super().__init__()
        self.norm = nn.LayerNorm(input_dim)
        self.left_proj = nn.Linear(input_dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(input_dim, hidden_dim, bias=False)
        self.left_gate = nn.Linear(input_dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(input_dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(input_dim, hidden_dim, bias=False)
        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, input_dim, bias=False)

    def forward(self, x, mask=None):
        z = self.norm(x)
        left = self.left_proj(z)
        right = self.right_proj(z)
        if mask is not None:
            left = left * mask[:, :, :]
            right = right * mask[:, :, :]
        left_gate = (self.left_gate(z.t())).sigmoid().t()
        right_gate = (self.right_gate(z.t())).sigmoid().t()
        out_gate = (self.out_gate(z.t())).sigmoid().t()
        out = torch.einsum('...ikd,...jkd->...ijd', 
                           left * left_gate, 
                           right * right_gate)
        z_out = self.to_out_norm(out)
        z_out = z_out * out_gate[:, :, :, None]
        return self.to_out(z_out)
