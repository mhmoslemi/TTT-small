@triton.jit
def tri_mul_out_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_x = tl.program_id(0) # i block
    pid_y = tl.program_id(1) # j block
    pid_z = tl.program_id(2) # combined batch*head

    b = pid_z // H
    h = pid_z % H

    # Compute offsets
    i_offset = pid_x * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_y * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out-of-bounds on i and j
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, N, BLOCK_K):
        k_offset = k + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left tile
        left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + i_offset[:,None] * stride_left_i + k_offset[None,:] * stride_left_k
        left = tl.load(left_ptrs, mask = mask_i[:,None] & mask_k[None,:], other=0.0).to(tl.float32)

        # Load right tile (transposed: k x j)
        right_ptrs = right_ptr + b*stride_right_b + h*stride_right_h + k_offset[:,None] * stride_right_k + j_offset[None,:] * stride_right_j
        right = tl.load(right_ptrs, mask = mask_k[:,None] & mask_j[None,:], other=0.0).to(tl.float32)

        # Perform dot product
        acc += tl.dot(left, right)

    # Write back results
    out_ptrs = out_ptr + b*stride_out_b + h*stride_out_h + i_offset[:,None] * stride_out_i + j_offset[None,:] * stride_out_j
    tl.store(out_ptrs, acc, mask = mask_i[:,None] & mask_j[None,:])

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# kernel definition

@triton.jit
def tri_mul_out_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_x = tl.program_id(0)  # i block
    pid_y = tl.program_id(1)  # j block
    pid_z = tl.program_id(2)  # batch * head

    # decode batch and head
    b = pid_z // H
    h = pid_z % H

    # Compute base offsets for i and j
    i_base = pid_x * BLOCK_M
    j_base = pid_y * BLOCK_N
    i_offset = i_base + tl.arange(0, BLOCK_M)
    j_offset = j_base + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_base = k_iter * BLOCK_K
        k_offset = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left (i,k)
        left_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + i_offset[:, None] * stride_left_i + k_offset[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)
        # Load right (k,j) - note transposed view
        right_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + k_offset[:, None] * stride_right_k + j_offset[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # Write out
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

```python
def custom_kernel(data):
    """
    Triton-accelerated TriMul "outgoing" operator, matching the forward
    pass of the reference PyTorch implementation.  The heavy
    O(N^3) einsum is computed with a custom Triton kernel; the
    surrounding linear layers, layer‑norms and sigmoids are performed
    with PyTorch for simplicity.
    """
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm on last dimension (dim)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections: [B,N,N,dim] -> [B,N,N,hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])   # shape B,N,N,hidden_dim
    right = F.linear(x, weights["right_proj.weight"])

    if mask is not None:
        # mask shape [B,N,N], broadcast to hidden_dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Allocate output for accumulated result
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

    # Get strides (in elements, not bytes)
    s_lb = left_fp16.stride(0)  # batch stride
    s_li = left_fp16.stride(1)  # i stride
    s_lk = left_fp16.stride(2)  # k stride
    s_lh = left_fp16.stride(3)  # hidden stride

    s_rb = right_fp16.stride(0)
    s_rj = right_fp16.stride(1)   # j stride (same as i)
    s_rk = right_fp16.stride(2)   # k stride
    s_rh = right_fp16.stride(3)

    s_ob = out.stride(0)
    s_oi = out.stride(1)
    s_oj = out.stride(2)
    s_oh = out.stride(3)

    # Choose block sizes based on N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid dimensions
    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             bs * hidden_dim )

    tri_mul_out_kernel[grid](
        left_fp16, right_fp16, out,
        bs, N, hidden_dim,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rj, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4
    )

    # Post processing: layer norm on hidden_dim
    out_norm = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # Multiply by out_gate
    out_norm = out_norm * out_gate

    # Final linear projection to dim
    out_final = F.linear(out_norm, weights["to_out.weight"])

    return out_final
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_out_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Triton kernel for the core TriMul contraction:
    out[b,i,j,h] = sum_{k} left[b,i,k,h] * right[b,j,k,h]
    """
    pid_x = tl.program_id(0)   # i block
    pid_y = tl.program_id(1)   # j block
    pid_z = tl.program_id(2)   # combined batch * head

    # decode batch index and head index
    b = pid_z // H
    h = pid_z % H

    # offsets for i and j dimensions
    i_base = pid_x * BLOCK_M
    j_base = pid_y * BLOCK_N
    i_off = i_base + tl.arange(0, BLOCK_M)
    j_off = j_base + tl.arange(0, BLOCK_N)

    mask_i = i_off < N
    mask_j = j_off < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_base = k_iter * BLOCK_K
        k_off = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # left: shape (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + i_off[:, None] * stride_left_i
            + k_off[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right transposed: shape (BLOCK_K, BLOCK_N)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + k_off[:, None] * stride_right_k
            + j_off[None, :] * stride_right_i
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(left, right)

    # write back
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + i_off[:, None] * stride_out_i
        + j_off[None, :] * stride_out_j
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated implementation of the “outgoing” Triangle
    Multiplicative Update (TriMul) forward pass.  The O(N³) contraction
    is performed in the `tri_mul_out_kernel`; all linear layers,
    LayerNorms and sigmoids are handled with PyTorch for simplicity.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----- LayerNorm on the input -----
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ----- Linear projections (dim -> hidden_dim) -----
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Optional masking -----
    if mask is not None and mask.numel() > 0:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ----- Gates (sigmoid of linear layers) -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----- Triton kernel for the einsum contraction -----
    # Convert to half to cut bandwidth, keep accumulation in FP32
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (in elements)
    s_lb, s_li, s_lk, s_lh = left_fp16.stride()
    s_rb, s_ri, s_rk, s_rh = right_fp16.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Blocking parameters – tuned for typical sequence lengths up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs * hidden_dim,
    )

    tri_mul_out_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        bs,
        N,
        hidden_dim,
        s_lb,
        s_li,
        s_lk,
        s_lh,
        s_rb,
        s_ri,
        s_rk,
        s_rh,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ----- Post‑processing: norm, gate, final linear -----
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out

    # --------------------------------------------------------------

        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------

    right = F.linear(x, weights["right_proj.weight"])
