        x_norm = torch.nn.functional.layer_norm(input_tensor,
                                                normalized_shape=(dim,),

import triton

import math

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr

    pid = tl.program_id(0)
    pid_n = tl.program_id(1)

    num_pid_m = tl.cdiv(N, BLOCK_M)

    pid_bh = pid // num_pid_m
    pid_m = pid % num_pid_m

    b = pid_bh // H
    h = pid_bh % H

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul module (outgoing version).
    Performs layernorm, linear projections, optional masking, gating,
    a batched matrix multiplication (implemented in Triton), output
    layernorm, a final gate and linear projection.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # retrieve weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1) input layer norm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # 2) projections
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)   # (B,N,N,hidden)
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    # 3) optional mask
    if not config.get("nomask", True):
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # 4) gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # 5) core triangular multiplication via Triton
    out = torch.empty_like(left)  # shape (B,N,N,hidden)
    tri_mul_contract(left, right, out)

    # 6) output layer norm and gate
    out_normed = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out_normed = out_normed * out_gate

    # 7) final linear projection back to dim
    result = torch.nn.functional.linear(out_normed, to_out_weight)

    return result

"""
TriMul (outgoing version) forward pass with a Triton kernel for the core
contraction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

The kernel treats each (batch, hidden) pair as an independent matrix
multiplication (left @ rightᵀ) and tiles the multiplication over the
sequence dimension N. All other operations (LayerNorm, linear projections,
gates, masking) are performed with PyTorch.
"""

import math
import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM where the right operand is transposed.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid = tl.program_id(0)          # encodes (batch, hidden) and block on M
    pid_n = tl.program_id(1)        # block index on N

    # number of blocks covering the M dimension (i‑index)
    num_pid_m = tl.cdiv(N, BLOCK_M)

    # decode batch*hidden and block‑M
    pid_bh = pid // num_pid_m
    pid_m = pid % num_pid_m

    # batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑indices
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑indices
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for the selected batch and hidden channel
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K dimension (the summed‑over axis)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (M, K) -> left[i, k, h]
        a_ptrs = left_base + (offs_m[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # right tile transposed: we need (K, N) where element (k, j) = right[b, j, k, h]
        b_ptrs = right_base + (offs_k[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_i)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # accumulate dot product: (M, K) · (K, N) -> (M, N)
        acc += tl.dot(a, b)

    # write the result tile
    c_ptrs = out_base + (offs_m[:, None] * stride_o_i) + (offs_n[None, :] * stride_o_j)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

                     right: torch.Tensor,
                     out: torch.Tensor,
                     BLOCK_M: int = 64,
                     BLOCK_N: int = 64,
                     BLOCK_K: int = 32) -> torch.Tensor:
    """
    Compute the TriMul contraction using the Triton GEMM kernel.
    Arguments:
        left, right: tensors of shape (B, N, N, H) (contiguous, float32)
        out: pre‑allocated tensor of the same shape as left/right
    """

    # ensure contiguous memory layout (required for stride correctness)
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # strides for the 4‑D layout (batch, i, k, hidden)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    total_bh = B * H
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (total_bh * num_pid_m, num_pid_n)

    tri_mul_gemm_kernel[grid](

        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out

    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4) Gating pathways
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Core triangular multiplication (performed by Triton)
    # ------------------------------------------------------------------
    out = torch.empty_like(left)          # shape (B,N,N,H)
    tri_mul_contract(left, right, out)   # fills `out` in‑place

    # ------------------------------------------------------------------
    # 6) Output layer‑norm, gating and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out_norm = out_norm * out_gate

    result = torch.nn.functional.linear(out_norm, to_out_weight)   # (B,N,N,dim)

    return result
```
