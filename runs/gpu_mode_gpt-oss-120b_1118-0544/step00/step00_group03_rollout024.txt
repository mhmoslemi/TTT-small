    hidden_dim = config["hidden_dim"]

    norm_bias = weights["norm.bias"]

        offs_i = i_start + tl.arange(0, BLOCK_M)
        offs_j = j_start + tl.arange(0, BLOCK_N)
        offs_d = d_start + tl.arange(0, BLOCK_D)

        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_d = offs_d < D

        # accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # iteration over k
        for k_offset in range(0, N, BLOCK_K):
            offs_k = k_offset + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # compute pointers for left: shape (M,K,D)
            a_ptrs = (left_ptr
                      + batch * stride_lb
                      + (offs_i[:, None, None] * stride_li)
                      + (offs_k[None, :, None] * stride_lk)
                      + (offs_d[None, None, :] * stride_ld))
            a = tl.load(a_ptrs,
                        mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                        other=0.0)

            # compute pointers for right transposed: shape (K,N,D)
            b_ptrs = (right_ptr
                      + batch * stride_rb
                      + (offs_k[:, None, None] * stride_rk)  # row = k
                      + (offs_j[None, :, None] * stride_rj)  # col = j
                      + (offs_d[None, None, :] * stride_rd))
            b = tl.load(b_ptrs,
                        mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
                        other=0.0)

            # compute dot product per d
            for d in range(BLOCK_D):
                # Slicing
                a_d = a[:, :, d]   # (M, K)
                b_d = b[:, :, d]   # (K, N)
                acc[:, :, d] += tl.dot(a_d, b_d)

        # store results
        out_ptrs = (out_ptr
                    + batch * stride_ob
                    + (offs_i[:, None, None] * stride_oi)
                    + (offs_j[None, :, None] * stride_oj)
                    + (offs_d[None, None, :] * stride_od))
        tl.store(out_ptrs,
                 acc,
                 mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])

    # launch kernel
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * ((D + BLOCK_D - 1) // BLOCK_D)
    )
    trimul_einsum_kernel[grid](
        left_fp16,
        right_fp16,
        out_core,
        B, N, D,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D
    )
    # Now out_core shape [B,N,N,hidden_dim] (float32)
    # apply output layernorm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_core = torch.nn.functional.layer_norm(out_core, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    # gating
    out_core = out_core * out_gate
    # final linear
    to_out_weight = weights["to_out.weight"] # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out_core, to_out_weight)  # bias=False
    return out

"""
Custom Triton kernel for the "outgoing" Triangle Multiplicative Update (TriMul).

The forward pass consists of:
1. LayerNorm over the channel dimension.
2. Linear projections to a hidden dimension (left/right) with optional mask.
3. Sigmoid gates applied to the projected tensors.
4. Core contraction:   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   (implemented as a batched GEMM where the right operand is accessed
   transposed in the second sequence dimension).
5. LayerNorm + gate on the hidden output.
6. Final linear projection back to the original channel dimension.

Step 4 is performed by a Triton kernel that tiles the i‑, j‑, k‑ and
hidden‑dim (d) axes.  The kernel works with fp16 inputs for speed,
accumulates in fp32, and writes a fp32 output.
"""
import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def _trimul_core_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].

    The right tensor is accessed with its first two sequence axes swapped,
    i.e. we treat it as (k, j) to achieve the required transpose.
    """
    pid_m = tl.program_id(0)                     # i‑tile
    pid_n = tl.program_id(1)                     # j‑tile
    pid_z = tl.program_id(2)                     # combined batch and d‑tile

    # --------------------------------------------------------------
    # Decode batch index and hidden‑dim tile index
    # --------------------------------------------------------------
    num_d_blocks = tl.cdiv(D, BLOCK_D)
    batch = pid_z // num_d_blocks
    d_block = pid_z % num_d_blocks

    # --------------------------------------------------------------
    # Offsets for this tile
    # --------------------------------------------------------------
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N
    d_start = d_block * BLOCK_D

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_d = d_start + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # --------------------------------------------------------------
    # Accumulator (fp32) for the tile
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    for k_offset in range(0, N, BLOCK_K):
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----------- Load left tile (i, k, d) -------------
        a_ptrs = (left_ptr
                  + batch * stride_lb
                  + (offs_i[:, None, None] * stride_li)
                  + (offs_k[None, :, None] * stride_lk)
                  + (offs_d[None, None, :] * stride_ld))
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )   # shape (M, K, D)

        # ----------- Load right tile transposed (k, j, d) -------------
        b_ptrs = (right_ptr
                  + batch * stride_rb
                  + (offs_k[:, None, None] * stride_rk)      # row = k
                  + (offs_j[None, :, None] * stride_rj)      # col = j
                  + (offs_d[None, None, :] * stride_rd))
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )   # shape (K, N, D)

        # ----------- Multiply‑accumulate for each hidden channel ----------
        for d in range(BLOCK_D):
            # Slice the current channel
            a_d = a[:, :, d]          # (M, K)
            b_d = b[:, :, d]          # (K, N)
            # tl.dot performs a matrix multiplication over the K dimension
            acc[:, :, d] += tl.dot(a_d, b_d)

    # --------------------------------------------------------------
    # Store the result
    # --------------------------------------------------------------
    out_ptrs = (out_ptr
                + batch * stride_ob
                + (offs_i[:, None, None] * stride_oi)
                + (offs_j[None, :, None] * stride_oj)
                + (offs_d[None, None, :] * stride_od))
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


def custom_kernel(data):
    """
    Entry point used by the evaluator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper aliases for weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden_dim (bias=False)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_expanded = mask.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask_expanded
        right = right * mask_expanded

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction using Triton
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel (speed, memory)
    left_fp16 = left.to(torch.half)
    right_fp16 = right.to(torch.half)

    B, N, _, D = left_fp16.shape  # D == hidden_dim

    # Allocate fp32 output for the core tensor
    out_core = torch.empty(left_fp16.shape, dtype=torch.float32, device=left_fp16.device)

    # Strides in elements (not bytes)
    stride_l = left_fp16.stride()    # (batch, i, k, d)
    stride_r = right_fp16.stride()
    stride_o = out_core.stride()

    # Kernel block sizes (tuned for RTX/GA100, works on B200 as well)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * ((D + BLOCK_D - 1) // BLOCK_D),
    )

    _trimul_core_kernel[grid](
        left_fp16,
        right_fp16,
        out_core,
        B,
        N,
        D,
        stride_l[0],
        stride_l[1],
        stride_l[2],
        stride_l[3],
        stride_r[0],
        stride_r[1],
        stride_r[2],
        stride_r[3],
        stride_o[0],
        stride_o[1],
        stride_o[2],
        stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm + gate
    # ------------------------------------------------------------------
    out_core = F.layer_norm(
        out_core,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out_core = out_core * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out_core, to_out_weight)   # bias=False as per definition

    return out
