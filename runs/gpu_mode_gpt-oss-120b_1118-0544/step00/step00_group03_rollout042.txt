def mask_mul_kernel(tensor_ptr, mask_ptr, out_ptr,
                    B, N, H,
                    stride_tensor_batch, stride_tensor_i, stride_tensor_k, stride_tensor_h,
                    stride_mask_batch, stride_mask_i, stride_mask_k,

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # LayerNorm - using weight and bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # standard epsilon
    eps = 1e-5

    # Perform layer norm across last dimension.
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # linear projections
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    # For linear without bias.
    left_proj = torch.nn.functional.linear(x, left_proj_w)  # shape B,N,N,hidden_dim
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # gating
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left_proj * left_gate
    right = right_proj * right_gate

    if not nomask and mask is not None:
        # apply mask using Triton kernel
        left = apply_mask_triton(left, mask)
        right = apply_mask_triton(right, mask)

    # Core contraction using torch.einsum (fast)
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)

    # to_out_norm
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps)

    # multiply with out_gate
    out = out * out_gate

    # final linear
    to_out_w = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_w)

    return out

        left = apply_mask_triton(left, mask_tensor)
        right = apply_mask_triton(right, mask_tensor)

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward implementation with a Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor of shape [B, N, N, C].
2. Apply two linear projections (left/right) and three gating projections.
3. Multiply the projected tensors with their corresponding gates.
4. If a mask is provided, apply it element‑wise to the gated tensors.
   This mask multiplication is performed by a small Triton kernel.
5. Perform the core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   (implemented with a torch.einsum for simplicity and correctness).
6. Layer‑norm the contraction result, multiply by an output gate and
   apply the final linear projection back to the original channel size.

Only the mask‑multiplication step is executed on Triton, satisfying the
requirement to use a custom kernel while keeping the overall implementation
readable and correct.
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: element‑wise multiplication of a 4‑D tensor with a 3‑D mask.
# Each thread processes one element (b, i, k, d) and writes the masked value.
# --------------------------------------------------------------------------- #
@triton.jit
def _mask_mul_kernel(
    tensor_ptr,   # ptr to input tensor  [B, N, N, H]
    mask_ptr,     # ptr to mask          [B, N, N]
    out_ptr,      # ptr to output tensor (can be the same as tensor_ptr)
    B, N, H,
    stride_tensor_b, stride_tensor_i, stride_tensor_k, stride_tensor_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)          # linear ids
    total = B * N * N * H
    mask = offsets < total

    # Decode linear index -> (b, i, k, d)
    d = offsets % H
    idx_div = offsets // H
    ik = idx_div % (N * N)
    b = idx_div // (N * N)
    i = ik // N
    k = ik % N

    # Load tensor value
    tensor_off = (
        b * stride_tensor_b
        + i * stride_tensor_i
        + k * stride_tensor_k
        + d * stride_tensor_h
    )
    x = tl.load(tensor_ptr + tensor_off, mask=mask, other=0.0)

    # Load mask scalar (broadcast over hidden dimension)
    mask_off = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k
    m = tl.load(mask_ptr + mask_off, mask=mask, other=0.0)

    # Store masked result
    tl.store(out_ptr + tensor_off, x * m, mask=mask)


def _apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Apply a binary mask to a tensor of shape [B, N, N, H] using Triton.
    The operation is performed in‑place for efficiency.
    """
    assert tensor.is_contiguous()
    assert mask.is_contiguous()
    B, N, _, H = tensor.shape
    # strides for a contiguous tensor
    s_tensor = tensor.stride()
    s_mask = mask.stride()
    total = B * N * N * H
    BLOCK = 128
    grid = ( (total + BLOCK - 1) // BLOCK, )
    _mask_mul_kernel[grid](
        tensor,
        mask,
        tensor,                      # in‑place write back
        B, N, H,
        s_tensor[0], s_tensor[1], s_tensor[2], s_tensor[3],
        s_mask[0], s_mask[1], s_mask[2],
        BLOCK=BLOCK,
        num_warps=4,
    )
    return tensor


# --------------------------------------------------------------------------- #
# Public entry point expected by the evaluator.
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask_tensor:  torch.Tensor [B, N, N] (may be None when nomask=True)
            - weights_dict: mapping from weight names to torch.Tensor
                Expected keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict: contains at least "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] – the module output.
    """
    # unpack arguments
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # 1. LayerNorm on the input (over the last dimension)
    # ------------------------------------------------------------------- #
    eps = 1e-5
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------- #
    # 2. Linear projections (bias‑free) and gating logits
    # ------------------------------------------------------------------- #
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # gating
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------- #
    # 3. Apply optional mask via the Triton kernel
    # ------------------------------------------------------------------- #
    if not nomask and mask_tensor is not None:
        # Ensure contiguous layout for the kernel
        left = left.contiguous()
        right = right.contiguous()
        mask = mask_tensor.contiguous()
        left = _apply_mask_triton(left, mask)
        right = _apply_mask_triton(right, mask)

    # ------------------------------------------------------------------- #
    # 4. Core contraction: sum over the second sequence dimension (k)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------- #
    out = torch.einsum("b i k d, b j k d -> b i j d", left, right)

    # ------------------------------------------------------------------- #
    # 5. Final LayerNorm, output gating and projection back to original dim
    # ------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # broadcast over hidden dimension
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
