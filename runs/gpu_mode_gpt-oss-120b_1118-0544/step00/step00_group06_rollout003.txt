def elementwise_mul_kernel(

"""
TriMul (outgoing) forward pass with fused elementwise multiplication in Triton.

The kernel `fused_mul_three` computes the element‑wise product of three
contiguous tensors (a * b * c). This is used to fuse the mask, projection,
and gate multiplications for the left and right tensors.

The remaining operations (LayerNorm, linear projections, gating, and the
batched contraction) are performed with PyTorch primitives, leveraging
cuBLAS for the heavy N³ computation.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: elementwise multiplication of three tensors
# ----------------------------------------------------------------------
@triton.jit
def fused_mul_three_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,
    N_ELEMS,                # total number of elements
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < N_ELEMS

    a = tl.load(a_ptr + offsets, mask=mask, other=0.0)
    b = tl.load(b_ptr + offsets, mask=mask, other=0.0)
    c = tl.load(c_ptr + offsets, mask=mask, other=0.0)

    out = a * b * c
    tl.store(out_ptr + offsets, out, mask=mask)


def fused_mul_three(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """
    Compute element‑wise a * b * c using Triton.
    All inputs must have the same shape, dtype and be contiguous.
    """
    assert a.shape == b.shape == c.shape, "Shapes must match"
    assert a.is_contiguous() and b.is_contiguous() and c.is_contiguous()
    total = a.numel()
    out = torch.empty_like(a)

    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul_three_kernel[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        c_ptr=c.data_ptr(),
        out_ptr=out.data_ptr(),
        N_ELEMS=total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,          # B200 has many cores; 4 warps is a safe default
    )
    return out


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: Tuple containing (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, same device as input)
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Config parameters
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)   # (B,N,N,hidden_dim)
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # ------------------------------------------------------------------
    # 4. Mask handling (broadcast to hidden_dim)
    # ------------------------------------------------------------------
    B, N, _, _ = left_proj.shape
    if nomask:
        # All‑ones mask, already the correct shape after broadcasting
        mask_expanded = torch.ones_like(left_proj, dtype=dtype, device=device)
    else:
        # mask_tensor: [B, N, N]; expand to [B, N, N, hidden_dim]
        mask_expanded = (
            mask_tensor.unsqueeze(-1)
            .to(dtype)
            .expand(-1, -1, -1, hidden_dim)
            .contiguous()
        )

    # Ensure contiguity for Triton
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_expanded = mask_expanded.contiguous()

    # ------------------------------------------------------------------
    # 5. Fuse mask, projection, and gate via Triton
    # ------------------------------------------------------------------
    left = fused_mul_three(left_proj, left_gate, mask_expanded)   # (B,N,N,hidden_dim)
    right = fused_mul_three(right_proj, right_gate, mask_expanded)

    # ------------------------------------------------------------------
    # 6. Batched contraction: out[i,j] = Σ_k left[i,k] * right[j,k] (per hidden dim)
    #    Implemented with torch.bmm after reshaping.
    # ------------------------------------------------------------------
    # Permute to bring hidden_dim to second axis for easy reshape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Merge batch and hidden dimensions -> (B*H, N, N)
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # Matrix multiplication with transposed right operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 8. Apply out_gate (element‑wise) and final linear projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate  # (B,N,N,hidden_dim)

    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    final_out = torch.nn.functional.linear(out_norm, to_out_weight)  # (B,N,N,dim)

    return final_out
