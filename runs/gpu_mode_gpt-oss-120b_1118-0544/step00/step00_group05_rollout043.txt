@triton.jit
def tri_mul_kernel(left, right, out, B, N, H,
                   stride_left_b, stride_left_i, stride_left_k, stride_left_h,
                   stride_right_b, stride_right_j, stride_right_k, stride_right_h,
                   stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                   BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, D_VEC: tl.constexpr):
    # Get program IDs
    pid_i = tl.program_id(0)  # tile index for i dimension
    pid_j = tl.program_id(1)  # tile index for j dimension
    pid_b = tl.program_id(2)  # batch index

    # Offsets
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)  # k within tile
    offs_d = tl.arange(0, D_VEC)

    # mask for i,j within N
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Compute base pointers for this batch
    left_ptr = left + pid_b * stride_left_b
    right_ptr = right + pid_b * stride_right_b
    out_ptr = out + pid_b * stride_out_b

    # Accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, D_VEC), dtype=tl.float32)

    # Loop over k dimension
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + offs_k
        mask_k = cur_k < N

        # Load left: shape (BLOCK_I, BLOCK_K, D_VEC)
        # offsets: i*stride_i + k*stride_k + d*stride_h
        left_offsets = (offs_i[:, None, None] * stride_left_i
                        + cur_k[None, :, None] * stride_left_k
                        + offs_d[None, None, :] * stride_left_h)
        left_vals = tl.load(left_ptr + left_offsets,
                            mask = mask_i[:, None, None] & mask_k[None, :, None],
                            other=0.0)

        # Load right: shape (BLOCK_J, BLOCK_K, D_VEC)
        right_offsets = (offs_j[:, None, None] * stride_right_j
                         + cur_k[None, :, None] * stride_right_k
                         + offs_d[None, None, :] * stride_right_h)
        right_vals = tl.load(right_ptr + right_offsets,
                             mask = mask_j[:, None, None] & mask_k[None, :, None],
                             other=0.0)

        # Multiply and accumulate over k dimension:
        # Expand left to (BLOCK_I, 1, BLOCK_K, D_VEC)
        # Expand right to (1, BLOCK_J, BLOCK_K, D_VEC)
        # Compute elementwise product then sum over k.
        # We can do using tl.reshape or broadcast.

        # Use broadcasting by adding dimensions:
        # left_vals (i,k,d) -> (i,1,k,d)
        # right_vals (j,k,d) -> (1,j,k,d)
        prod = left_vals[:, None, :, :] * right_vals[None, :, :, :]  # shape (i,j,k,d)
        # Sum over k
        acc += tl.sum(prod, axis=2)

    # Apply mask for i/j and d if needed for partial block at boundaries
    mask_ij_d = mask_i[:, None] & mask_j[None, :] & (offs_d[None, None] < H)
    # Store result
    out_offsets = (offs_i[:, None, None] * stride_out_i
                   + offs_j[None, :, None] * stride_out_j
                   + offs_d[None, None, :] * stride_out_h)
    tl.store(out_ptr + out_offsets,
             acc,

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(left, right, out, B, N, H,
                   stride_l_b, stride_l_i, stride_l_k, stride_l_h,
                   stride_r_b, stride_r_j, stride_r_k, stride_r_h,
                   stride_o_b, stride_o_i, stride_o_j, stride_o_h,
                   BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, D_VEC: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_b = tl.program_id(2)

    # Offsets within block
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    k = tl.arange(0, BLOCK_K)
    d = tl.arange(0, D_VEC)

    mask_i = i < N
    mask_j = j < N

    # base pointers for this block
    left_base = left + pid_b * stride_l_b
    right_base = right + pid_b * stride_r_b
    out_base = out + pid_b * stride_o_b

    acc = tl.zeros((BLOCK_I, BLOCK_J, D_VEC), dtype=tl.float32)

    for block_start in range(0, N, BLOCK_K):
        cur_k = block_start + k
        mask_k = cur_k < N

        # offsets for left (i,k,d)
        left_offsets = (i[:, None, None] * stride_l_i
                        + cur_k[None, :, None] * stride_l_k
                        + d[None, None, :] * stride_l_h)

        left_val = tl.load(left_base + left_offsets,
                           mask = mask_i[:, None, None] & mask_k[None, :, None],
                           other=0.0)

        # offsets for right (j,k,d)
        right_offsets = (j[:, None, None] * stride_r_j
                         + cur_k[None, :, None] * stride_r_k
                         + d[None, None, :] * stride_r_h)

        right_val = tl.load(right_base + right_offsets,
                            mask = mask_j[:, None, None] & mask_k[None, :, None],
                            other=0.0)

        # product and sum over k
        # left: (i,k,d), right: (j,k,d) -> broadcast to (i,j,k,d)
        prod = left_val[:, None, :, :] * right_val[None, :, :, :]  # (i,j,k,d)
        acc += tl.sum(prod, axis=2)  # sum over k dimension

    # Store
    # masks for i, j within bounds and d within hidden dim (d < H)
    mask_ij = mask_i[:, None] & mask_j[None, :]
    mask_ij_d = mask_ij[:, :, None] & (d[None, None, :] < H)
    out_offsets = (i[:, None, None] * stride_o_i
                   + j[None, :, None] * stride_o_j
                   + d[None, None, :] * stride_o_h)

    tl.store(out_base + out_offsets, acc, mask=mask_ij_d)

def custom_kernel(data):
    """
    Triton-accelerated outgoing Triangle Multiplicative Update (TriMul) forward pass.
    The kernel fuses the O(N^3) contraction out = sum_k left[...,k,:] * right[...,k,:]
    across the sequence dimension using tile-based accumulation.
    """
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape  # shape [B,N,N,dim]

    # LayerNorm (x = norm(x))
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Mask handling
    if mask is not None:
        # Ensure mask dtype matches input
        mask = mask.to(dtype=dtype)
        mask = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates (sigmoid)
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Ensure tensors are contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()

    # Allocate output tensor for contraction (B,N,N,hidden_dim)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Determine strides (int64)
    stride_l = left.stride()   # (B,N,N,hidden_dim) strides order
    stride_r = right.stride()
    stride_o = out.stride()

    # Kernel launch config
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    D_VEC = 8  # must divide hidden_dim (128/384)

    # Number of hidden slices
    for h_start in range(0, hidden_dim, D_VEC):
        h_end = h_start + D_VEC
        # Slice views (they share memory, no copy)
        left_slice = left[..., h_start:h_end]
        right_slice = right[..., h_start:h_end]
        out_slice = out[..., h_start:h_end]

        # Compute strides for sliced tensors (the stride for the hidden dim stays the same)
        stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_slice.stride()
        stride_r_b, stride_r_j, stride_r_k, stride_r_h = right_slice.stride()
        stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_slice.stride()

        # Grid dimensions (i, j, batch)
        grid = (
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J,
            B,
        )

        # Launch kernel
        tri_mul_kernel[grid](
            left_slice,
            right_slice,
            out_slice,
            B,
            N,
            hidden_dim,   # pass full hidden_dim for bounds checking
            stride_l_b,
            stride_l_i,
            stride_l_k,
            stride_l_h,
            stride_r_b,
            stride_r_j,
            stride_r_k,
            stride_r_h,
            stride_o_b,
            stride_o_i,
            stride_o_j,
            stride_o_h,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            D_VEC=D_VEC,
        )

    # out now holds summed contraction results

    # Apply to_out_norm (LayerNorm over hidden_dim)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # Apply out gate
    out = out * out_gate

    # Final linear projection back to dim
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, D_VEC: tl.constexpr):
    """Kernel computing out[b,i,j,d] = sum_{k=0}^{N-1} left[b,i,k,d] * right[b,j,k,d]
    for a slice of the hidden dimension of size D_VEC.
    The kernel tiles over the i and j dimensions and iterates over k in blocks.
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_b = tl.program_id(2)

    # Coordinates
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    k = tl.arange(0, BLOCK_K)
    d = tl.arange(0, D_VEC)

    mask_i = i < N
    mask_j = j < N

    # Base pointers for this batch
    left_base = left_ptr + pid_b * stride_l_b
    right_base = right_ptr + pid_b * stride_r_b
    out_base = out_ptr + pid_b * stride_o_b

    # Accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, D_VEC), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + k
        mask_k = cur_k < N

        # Load tile from left: shape (BLOCK_I, BLOCK_K, D_VEC)
        left_offset = (
            i[:, None, None] * stride_l_i
            + cur_k[None, :, None] * stride_l_k
            + d[None, None, :] * stride_l_h
        )
        left_tile = tl.load(
            left_base + left_offset,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0
        )
        # Load tile from right: shape (BLOCK_J, BLOCK_K, D_VEC)
        right_offset = (
            j[:, None, None] * stride_r_j
            + cur_k[None, :, None] * stride_r_k
            + d[None, None, :] * stride_r_h
        )
        right_tile = tl.load(
            right_base + right_offset,
            mask=mask_j[:, None, None] & mask_k[None, :, None],
            other=0.0
        )
        # Accumulate across K dimension using an inner static loop
        for kk in range(0, BLOCK_K):
            # Extract column kk (shape (BLOCK_I, D_VEC) and (BLOCK_J, D_VEC))
            left_col = left_tile[:, kk, :]   # (BLOCK_I, D_VEC)
            right_col = right_tile[:, kk, :] # (BLOCK_J, D_VEC)
            # Outer product for each hidden dimension
            # broadcasting to (BLOCK_I, BLOCK_J, D_VEC)
            acc += left_col[:, None, :] * right_col[None, :, :]   # (i,j,d)
    # Store result
    out_offset = (
        i[:, None, None] * stride_o_i
        + j[None, :, None] * stride_o_j
        + d[None, None, :] * stride_o_h
    )
    # mask for valid i/j (d always within slice)
    mask_ij = mask_i[:, None] & mask_j[None, :]
    tl.store(out_base + out_offset, acc, mask=mask_ij[:, :, None])
    
def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.
    The heavy O(N^3) contraction is performed by the Triton kernel `tri_mul_kernel`.
    """
    import torch

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape   # shape [B, N, N, dim]

    # LayerNorm over the channel dimension
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Optional mask
    if mask is not None:
        mask = mask.to(dtype=dtype)
        mask = mask.unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # Gating
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous layout for Triton
    left = left.contiguous()
    right = right.contiguous()

    # Allocate output tensor for the contraction
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Kernel launch parameters
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    D_VEC = 8   # hidden dimension tile. Must divide hidden_dim (128, 384)

    # Grid (i_tiles, j_tiles, batch)
    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
        B,
    )

    # Slice and launch kernel over hidden dimension
    for h_start in range(0, hidden_dim, D_VEC):
        left_slice = left[..., h_start:h_start + D_VEC]
        right_slice = right[..., h_start:h_start + D_VEC]
        out_slice = out[..., h_start:h_start + D_VEC]

        # Strides (they are ints)
        stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_slice.stride()
        stride_r_b, stride_r_j, stride_r_k, stride_r_h = right_slice.stride()
        stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_slice.stride()

        # Launch kernel
        tri_mul_kernel[grid](
            left_slice,
            right_slice,
            out_slice,
            B,
            N,
            hidden_dim,
            stride_l_b,
            stride_l_i,
            stride_l_k,
            stride_l_h,
            stride_r_b,
            stride_r_j,
            stride_r_k,
            stride_r_h,
            stride_o_b,
            stride_o_i,
            stride_o_j,
            stride_o_h,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            D_VEC=D_VEC,
        )

    # Post-processing: layer norm, out gate, final linear
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output

"""
TriMul (outgoing) forward pass.

The heavy O(N³·C) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed by a Triton kernel that tiles the i and j axes and iterates over
k in blocks.  The hidden dimension (C) is processed in small fixed‑size
chunks (D_VEC) to keep register pressure low.  All other operations
(layer‑norm, linear projections, gating) are carried out in PyTorch.
"""
import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    D_VEC: tl.constexpr,
):
    """Compute a slice of the tri‑multiplicative update:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] for d in a D_VEC‑wide slice.
    """
    pid_i = tl.program_id(0)   # tile index over the i‑axis
    pid_j = tl.program_id(1)   # tile index over the j‑axis
    pid_b = tl.program_id(2)   # batch index

    # Block coordinates
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    k = tl.arange(0, BLOCK_K)
    d = tl.arange(0, D_VEC)

    mask_i = i < N
    mask_j = j < N

    # Base pointers for this batch
    left_base = left_ptr + pid_b * stride_l_b
    right_base = right_ptr + pid_b * stride_r_b
    out_base = out_ptr + pid_b * stride_o_b

    # Accumulator for the result slice
    acc = tl.zeros((BLOCK_I, BLOCK_J, D_VEC), dtype=tl.float32)

    # Loop over the reduction dimension K in blocks
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + k
        mask_k = cur_k < N

        # -------- load a tile from left (i, k, d) ----------
        left_off = (
            i[:, None, None] * stride_l_i
            + cur_k[None, :, None] * stride_l_k
            + d[None, None, :] * stride_l_h
        )
        left_tile = tl.load(
            left_base + left_off,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )  # (BLOCK_I, BLOCK_K, D_VEC)

        # -------- load a tile from right (j, k, d) ----------
        right_off = (
            j[:, None, None] * stride_r_j
            + cur_k[None, :, None] * stride_r_k
            + d[None, None, :] * stride_r_h
        )
        right_tile = tl.load(
            right_base + right_off,
            mask=mask_j[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )  # (BLOCK_J, BLOCK_K, D_VEC)

        # -------- accumulate over the K block ----------
        # Unroll the inner reduction over k (compile‑time loop)
        for kk in range(0, BLOCK_K):
            left_col = left_tile[:, kk, :]   # (BLOCK_I, D_VEC)
            right_col = right_tile[:, kk, :]  # (BLOCK_J, D_VEC)
            # Outer product for each hidden dimension
            acc += left_col[:, None, :] * right_col[None, :, :]  # (i, j, d)

    # -------- store the result ----------
    out_off = (
        i[:, None, None] * stride_o_i
        + j[None, :, None] * stride_o_j
        + d[None, None, :] * stride_o_h
    )
    # Mask only i and j (the hidden slice is always inside bounds)
    mask_ij = mask_i[:, None] & mask_j[None, :]
    tl.store(out_base + out_off, acc, mask=mask_ij[:, :, None])


def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.
    The contraction over the sequence dimension is executed by `tri_mul_kernel`.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape  # input shape: [B, N, N, dim]

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over the hidden dimension)
    if mask is not None:
        mask = mask.to(dtype=dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton contraction (left × rightᵀ summed over k)
    left = left.contiguous()
    right = right.contiguous()

    # Output tensor for the O(N³) part
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Kernel launch configuration
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 16   # smaller block to keep register usage modest
    D_VEC = 8      # hidden‑dimension tile (must divide hidden_dim)

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
        B,
    )

    # Process the hidden dimension in chunks of D_VEC
    for h_start in range(0, hidden_dim, D_VEC):
        left_slice = left[..., h_start : h_start + D_VEC]
        right_slice = right[..., h_start : h_start + D_VEC]
        out_slice = out[..., h_start : h_start + D_VEC]

        # Strides for the sliced tensors (int64)
        stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_slice.stride()
        stride_r_b, stride_r_j, stride_r_k, stride_r_h = right_slice.stride()
        stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_slice.stride()

        # Launch the kernel for this hidden slice
        tri_mul_kernel[grid](
            left_slice,
            right_slice,
            out_slice,
            B,
            N,
            hidden_dim,
            stride_l_b,
            stride_l_i,
            stride_l_k,
            stride_l_h,
            stride_r_b,
            stride_r_j,
            stride_r_k,
            stride_r_h,
            stride_o_b,
            stride_o_i,
            stride_o_j,
            stride_o_h,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            D_VEC=D_VEC,
        )

    # ------------------------------------------------------------------
    # 6. Post‑processing: layer‑norm, output gate, final linear projection
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
