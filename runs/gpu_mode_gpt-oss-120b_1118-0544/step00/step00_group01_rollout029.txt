def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    epsilon = 1e-5

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=epsilon)

    # Projections
    left_proj_weight = weights["left_proj.weight"]   # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [bs, N,N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if present.
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1)  # shape [bs,N,N,1]
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute product using Triton
    # allocate output tensor
    out = torch.empty_like(left)  # shape [bs, N,N, hidden_dim] float32

    # compute strides for left, right, out
    left_strides = left.stride()   # tuple (stride_b, stride_i, stride_k, stride_h)
    right_strides = right.stride()
    out_strides = out.stride()

    # total_batches = bs * hidden_dim
    total_batches = bs * hidden_dim

    # Ensure left, right, out are contiguous
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # kernel launch config
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = ( (seq_len + BLOCK_M - 1) // BLOCK_M,
             (seq_len + BLOCK_N - 1) // BLOCK_N,
             total_batches )

    triton_trimul_kernel[grid](
        left_ptr = left.data_ptr(),
        right_ptr = right.data_ptr(),
        out_ptr = out.data_ptr(),
        B = bs,
        N = seq_len,
        H = hidden_dim,
        stride_left_b = left_strides[0],
        stride_left_i = left_strides[1],
        stride_left_k = left_strides[2],
        stride_left_h = left_strides[3],
        stride_right_b = right_strides[0],
        stride_right_i = right_strides[1],
        stride_right_k = right_strides[2],
        stride_right_h = right_strides[3],
        stride_out_b = out_strides[0],
        stride_out_i = out_strides[1],
        stride_out_j = out_strides[2],
        stride_out_h = out_strides[3],
        BLOCK_M = BLOCK_M,
        BLOCK_N = BLOCK_N,
        BLOCK_K = BLOCK_K,
        num_stages = 3,
        num_warps = 4,
    )
    # at this point out contains sum over k product

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_normed = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=epsilon)

    # multiply out_gate
    out_normed = out_normed * out_gate

    # final linear projection back to dim
    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out_normed, to_out_weight)   # shape [bs,N,N,dim]

    return output

import triton.language as tl

@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(2)  # batch * hidden
    pid_m = tl.program_id(0)  # i block
    pid_n = tl.program_id(1)  # j block

    # compute batch and hidden indices
    total_h = H
    b = pid_b // total_h
    h = pid_b % total_h

    # start offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base pointers for this batch+hidden
    # compute per element base offset for left and right
    # left base: b*stride_left_b + h*stride_left_h

    pid_b = tl.program_id(2)      # flattened batch * hidden dimension
    pid_m = tl.program_id(0)      # block row (i)
    pid_n = tl.program_id(1)      # block col (j)

    # Recover original batch and hidden channel indices
    b = pid_b // H
    h = pid_b % H

    # Compute start offsets for this tile
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for the current (b, h)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis k
    for kb in range(tl.cdiv(N, BLOCK_K)):
        k_start = kb * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile A[i,k]
        a_ptrs = left_base + (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile B_T[k,j]  (original right[j,k])
        b_ptrs = right_base + (offs_k[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_i)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Matrix multiplication of the tile
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Write the result
    out_ptrs = out_base + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton-accelerated forward pass for the outgoing Triangle Multiplicative Update (TriMul).
    The heavy O(N^3) pairwise product is performed by a fused block‑matrix multiplication
    kernel (left @ rightᵀ) for each hidden channel, while the remaining linear and
    LayerNorm operations are carried out with PyTorch.
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    # Config parameters
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----- LayerNorm on the input (over the channel dimension) -----
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # shape [B,N,N,dim]

    # ----- Linear projections (no bias) -----
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ----- Optional masking -----
    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ----- Gating linear layers + sigmoid -----
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.nn.functional.linear(x, left_gate_weight).sigmoid()   # [B,N,N,hidden_dim]
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ----- Triton kernel for the einsum (left @ rightᵀ) -----
    B, N, _, _ = left.shape
    out = torch.empty_like(left)   # will hold [B,N,N,hidden_dim]

    # Ensure contiguous layout for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # Strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tiling parameters (tuned for A100/B200, can be adjusted)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid definition: (num_blocks_m, num_blocks_n, total_batches = B*hidden_dim)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    triton_trimul_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_ptr=out.data_ptr(),
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )

    # ----- Output LayerNorm, gating and final projection -----
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate

    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    output = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,dim]

    return output

"""TriMul (outgoing) forward pass with Triton acceleration.

The expensive part of the TriMul module is the three‑dimensional
summation 
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
For every hidden channel *d* this is a matrix multiplication
`left @ rightᵀ` of two N×N matrices.  The kernel below fuses the
batched matmul across the hidden dimension using a tiled
implementation (BLOCK_M×BLOCK_N×BLOCK_K) and accumulates in FP32.
All other operations (LayerNorm, linear projections, sigmoids) are
performed with PyTorch.

The kernel receives flattened batch‑hidden indices, loads tiles of the
left and transposed‑right matrices, performs `tl.dot`, and writes the
result back to the output tensor.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul per hidden channel: out = left @ rightᵀ."""
    pid_b = tl.program_id(2)      # batch * hidden index
    pid_m = tl.program_id(0)      # block row (i)
    pid_n = tl.program_id(1)      # block column (j)

    # Recover original batch and hidden indices
    b = pid_b // H
    h = pid_b % H

    # Starting indices of the current tile
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for the selected (b, h)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction axis k
    for kb in range(tl.cdiv(N, BLOCK_K)):
        k_start = kb * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile A[i, k]
        a_ptrs = left_base + (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile Bᵀ[k, j]  (original right[j, k])
        b_ptrs = right_base + (offs_k[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_i)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # Write the tile to output
    out_ptrs = out_base + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
