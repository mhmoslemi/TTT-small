def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # Expect shape [bs, N, N, dim]
    bs, N, _, dim = input.shape
    hidden_dim = config["hidden_dim"]; eps = 1e-5

    # layernorm: x_norm = (x - mean) * rsqrt(var+eps) * norm_weight + norm_bias
    # compute mean and var across last dim.
    x = input_tensor
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    rsig = (var + eps).rsqrt()
    x_norm = (x - mean) * rsig
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # projection
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1)  # shape [bs, N, N, 1]
    else:
        mask_ = torch.ones_like(left_proj)  # but unsqueeze would be [bs,N,N,1]
        # We'll just set mask_ = 1.
    left = left_proj * mask_
    right = right_proj * mask_

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Compute core multiplication via triton kernel
    # allocate output tensor (float32) shape [bs, N, N, hidden_dim]
    out = torch.empty(bs, N, N, hidden_dim, dtype=torch.float32, device=x.device)

    # Use triton kernel for each batch
    import triton
    import triton.language as tl

    # define kernel inside function? We'll define outside earlier.
    # but we need pointer offsets for left,right,out.

    # These tensors are contiguous? left shape [bs,N,N,hidden_dim] with C contiguous row major.
    # Stride order: stride_b = left.stride(0); stride_i = left.stride(1); stride_k = left.stride(2); stride_d = left.stride(3)

        num_j = (N + BLOCK_J - 1) // BLOCK_J

"""
TriMul (outgoing) implementation with a Triton kernel for the O(N³) reduction.

The forward pass consists of:
1. Layer‑norm on the input tensor.
2. Linear projections (left/right) and 3 gating linear layers.
3. Optional masking of the projected tensors.
4. Core operation:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   computed with a block‑wise Triton kernel (i‑j‑d blocks,
   reduction over k).
5. A second Layer‑norm, output‑gate multiplication and final linear
   projection back to the original dimension.
All tensors are kept on the CUDA device; the heavy N³ reduction is
performed in the Triton kernel, while the rest uses PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr,               # *float32
    right_ptr,              # *float32
    out_ptr,                # *float32
    # strides (in elements, not bytes)
    stride_left_i, stride_left_k, stride_left_d,
    stride_right_i, stride_right_k, stride_right_d,
    stride_out_i, stride_out_j, stride_out_d,
    N, D,                   # sequence length, hidden dimension
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of blocks in each dimension
    num_i = (N + BLOCK_I - 1) // BLOCK_I
    num_j = (N + BLOCK_J - 1) // BLOCK_J
    num_d = (D + BLOCK_D - 1) // BLOCK_D

    i_block = (pid // (num_j * num_d)) % num_i
    j_block = (pid // num_d) % num_j
    d_block = pid % num_d

    i_off = i_block * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = j_block * BLOCK_J + tl.arange(0, BLOCK_J)
    d_off = d_block * BLOCK_D + tl.arange(0, BLOCK_D)

    i_mask = i_off < N
    j_mask = j_off < N
    d_mask = d_off < D

    # accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    num_k_blocks = (N + BLOCK_K - 1) // BLOCK_K
    for kb in range(num_k_blocks):
        k_off = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # left[i,k,d]   shape (I, K, D)
        left_ptrs = (
            left_ptr
            + i_off[:, None, None] * stride_left_i
            + k_off[None, :, None] * stride_left_k
            + d_off[None, None, :] * stride_left_d
        )
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # right[k,j,d]   shape (K, J, D)  <-- note k is first dimension for dot
        right_ptrs = (
            right_ptr
            + k_off[:, None, None] * stride_right_k
            + j_off[None, :, None] * stride_right_i
            + d_off[None, None, :] * stride_right_d
        )
        right = tl.load(
            right_ptrs,
            mask=k_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # accumulate over k for each d in the block
        for dd in range(0, BLOCK_D):
            left_slice = left[:, :, dd]   # (I, K)
            right_slice = right[:, :, dd] # (K, J)
            acc[:, :, dd] = acc[:, :, dd] + tl.dot(left_slice, right_slice)

    # store result
    out_ptrs = (
        out_ptr
        + i_off[:, None, None] * stride_out_i
        + j_off[None, :, None] * stride_out_j
        + d_off[None, None, :] * stride_out_d
    )
    tl.store(
        out_ptrs,
        acc,
        mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :],
    )


def custom_kernel(data):
    """
    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : [B, N, N, dim]  (float32)
            mask_tensor  : [B, N, N]      (bool or float)
            weights_dict : {
                "norm.weight": [dim],
                "norm.bias"  : [dim],
                "left_proj.weight": [hidden_dim, dim],
                "right_proj.weight": [hidden_dim, dim],
                "left_gate.weight": [hidden_dim, dim],
                "right_gate.weight": [hidden_dim, dim],
                "out_gate.weight":  [hidden_dim, dim],
                "to_out_norm.weight": [hidden_dim],
                "to_out_norm.bias":   [hidden_dim],
                "to_out.weight":      [dim, hidden_dim],
            }
            config_dict : {"dim": int, "hidden_dim": int}
    Returns:
        output tensor [B, N, N, dim] (float32)
    """
    eps = 1e-5

    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    (inp, mask, weights, config) = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm on input
    # ------------------------------------------------------------------
    inp_norm = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias) and gating linear layers
    # ------------------------------------------------------------------
    left_proj = F.linear(inp_norm, weights["left_proj.weight"])
    right_proj = F.linear(inp_norm, weights["right_proj.weight"])

    # handle mask (may be None or a tensor)
    if mask is None:
        mask_ = torch.ones((B, N, N), dtype=inp.dtype, device=inp.device)
    else:
        mask_ = mask.to(inp.dtype)
    mask_ = mask_.unsqueeze(-1)  # [B, N, N, 1]

    # apply mask before gating (as in reference code)
    left = left_proj * mask_
    right = right_proj * mask_

    # gates
    left_gate = torch.sigmoid(F.linear(inp_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(inp_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(inp_norm, weights["out_gate.weight"]))

    # combine with gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³) reduction via Triton
    # ------------------------------------------------------------------
    out = torch.empty(
        (B, N, N, hidden_dim), dtype=torch.float32, device=inp.device
    )

    # block sizes (tuned for typical N up to 1024, hidden_dim ≤ 384)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_D = 8
    BLOCK_K = 32

    # strides (in element counts, not bytes)
    s_li = left.stride(1)   # over i
    s_lk = left.stride(2)   # over k
    s_ld = left.stride(3)   # over d

    s_ri = right.stride(1)  # over i (used as j in kernel)
    s_rk = right.stride(2)  # over k
    s_rd = right.stride(3)  # over d

    s_oi = out.stride(1)
    s_oj = out.stride(2)
    s_od = out.stride(3)

    for b in range(B):
        # base pointers for the current batch
        left_ptr = left.data_ptr() + b * left.stride(0)
        right_ptr = right.data_ptr() + b * right.stride(0)
        out_ptr = out.data_ptr() + b * out.stride(0)

        num_i = (N + BLOCK_I - 1) // BLOCK_I
        num_j = (N + BLOCK_J - 1) // BLOCK_J
        num_d = (hidden_dim + BLOCK_D - 1) // BLOCK_D
        total_blocks = num_i * num_j * num_d

        _tri_mul_kernel[(total_blocks,)](
            left_ptr,
            right_ptr,
            out_ptr,
            s_li,
            s_lk,
            s_ld,
            s_ri,
            s_rk,
            s_rd,
            s_oi,
            s_oj,
            s_od,
            N,
            hidden_dim,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_D=BLOCK_D,
            BLOCK_K=BLOCK_K,
        )

    # ------------------------------------------------------------------
    # Second LayerNorm, output gate and final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
```
