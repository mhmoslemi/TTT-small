def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, D,
    stride_left_i, stride_left_k, stride_left_d,
    stride_right_i, stride_right_k, stride_right_d,
    stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_m = tl.program_id(0)  # i block
    pid_n = tl.program_id(1)  # j block
    pid_d = tl.program_id(2)  # d block

    # block offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    # masks for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # Loop over k dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)  # N is seq_len
    for pk in range(num_k_blocks):
        k_base = pk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)  # shape (BLOCK_K,)
        mask_k = offs_k < N

        # load left block: shape (BLOCK_M, BLOCK_K, BLOCK_D)
        # compute offsets:
        left_ptrs = left_ptr + (offs_i[:, None, None] * stride_left_i
                                + offs_k[None, :, None] * stride_left_k
                                + offs_d[None, None, :] * stride_left_d)
        # load right block: shape (BLOCK_N, BLOCK_K, BLOCK_D)
        right_ptrs = right_ptr + (offs_j[:, None, None] * stride_right_i
                                 + offs_k[None, :, None] * stride_right_k
                                 + offs_d[None, None, :] * stride_right_d)

        left_block = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)
        right_block = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)

        # For each d, compute left_block[:,:,d] @ right_block[:,:,d].T? Actually left: (i,k,d); right: (j,k,d)

        # For each d, compute elementwise multiplication and sum over k:
        # out[i,j,d] += sum_k left[i,k,d] * right[j,k,d].
        # We can compute via broadcasting: left_block: (i,k,d); right_block: (j,k,d)
        # Perform `tl.dot`? Not available for broadcasting across d. So do elementwise multiplication and sum across k:
        # We can reshape left and right to (i,k,d) and (j,k,d); compute `tmp = left_block * right_block` using broadcast w.r.t i and j axes.
        # To broadcast, we need to expand dimensions: left_block: (i,k,1,d), right_block: (1,j,k,d) -> shape (i,j,k,d). Then sum over k.
        # But Triton cannot easily broadcast along new axes; we can manually loop over k within block:
        # For kk in range(BLOCK_K):

def custom_kernel(data):
    """Implement TriMul 'outgoing' update via Triton kernel.
    The core einsum `out_{i,j,d}=sum_k left_{i,k,d} * right_{j,k,d}` is executed in Triton.
    """
    import torch
    import triton
    import triton.language as tl
    from torch.nn import functional as F

    input_tensor, mask_tensor, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # For safety cast to float32
    eps = 1e-5

    # LayerNorm over dim
    # Using F.layer_norm with provided weight and bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # shape (bs,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # mask
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1)  # shape (bs,N,N,1)
        left = left * mask
        right = right * mask
    # gating
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    # Output buffer for hidden dim result
    out_hidden = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

    # Triton kernel definition
    @triton.jit
    def trimul_kernel(
        left_ptr, right_ptr, out_ptr,
        N, D,
        stride_li, stride_lk, stride_ld,
        stride_ri, stride_rk, stride_rd,
        stride_oi, stride_oj, stride_od,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
        BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
    ):

    # Use heuristics

    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8

    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid_d = (hidden_dim + BLOCK_D - 1) // BLOCK_D

        right_ptr = right_h[b].data_ptr()
        out_ptr = out_hidden[b].data_ptr()

        stride_right_i, stride_right_k, stride_right_d = right_h.stride()[0], right_h.stride()[1], right_h.stride()[2]
        stride_out_i, stride_out_j, stride_out_d = out_hidden.stride()[0], out_hidden.stride()[1], out_hidden.stride()[2]

        trimul_kernel[ (grid_i, grid_j, grid_d) ](
            left_ptr,
            right_ptr,
            out_ptr,
            N,
            hidden_dim,
            stride_left_i, stride_left_k, stride_left_d,
            stride_right_i, stride_right_k, stride_right_d,
            stride_out_i, stride_out_j, stride_out_d,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D

    # Now apply to_out_norm, out_gate, to_out linear
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # layer norm over hidden dim on out_hidden (shape [bs,N,N,hidden_dim])
    out_normed = F.layer_norm(out_hidden, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out_gate

    # Final linear
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out_normed, to_out_weight)  # shape [bs,N,N,dim]

    return out

"""
TriMul 'outgoing' update (AlphaFold3).

The heavy computation:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is performed by a custom Triton kernel that computes a batched
matrix‑multiply for each hidden channel d (i.e. left_{i,k,d} @
right_{j,k,d}.T).  The surrounding operations (LayerNorms, linear
projections, gating, mask handling and the final output projection)
are carried out with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm on input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [bs, N, N, dim]

    # ------------------------------------------------------------------
    # Projections
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [bs, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Optional mask (mask shape: [bs, N, N])
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1)  # [bs, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate   # [bs, N, N, hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 for the Triton kernel
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    del left, right, x  # free memory

    # ------------------------------------------------------------------
    # Triton kernel: compute sum_k left[i,k,d] * right[j,k,d]
    # ------------------------------------------------------------------
    @triton.jit
    def trimul_kernel(
        left_ptr,          # fp16   [N, N, D]
        right_ptr,         # fp16   [N, N, D]
        out_ptr,           # fp32   [N, N, D]
        N, D,
        stride_li, stride_lk, stride_ld,
        stride_ri, stride_rk, stride_rd,
        stride_oi, stride_oj, stride_od,
        BLOCK_M: tl.constexpr,
        BLOCK_N: tl.constexpr,
        BLOCK_K: tl.constexpr,
        BLOCK_D: tl.constexpr,
    ):
        pid_i = tl.program_id(0)   # block index in i dimension
        pid_j = tl.program_id(1)   # block index in j dimension
        pid_d = tl.program_id(2)   # block index in hidden dimension

        # Offsets for the i, j, d dimensions
        offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
        offs_d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_d = offs_d < D

        # Number of blocks along the reduction dimension k
        num_k_blocks = tl.cdiv(N, BLOCK_K)

        # Loop over hidden channels inside this block
        for d_idx in range(BLOCK_D):
            d = pid_d * BLOCK_D + d_idx
            if d >= D:
                # out‑of‑range channel – skip
                continue

            # accumulator for a single channel
            acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

            # Reduction over k
            for pk in range(num_k_blocks):
                k_base = pk * BLOCK_K
                offs_k = k_base + tl.arange(0, BLOCK_K)
                mask_k = offs_k < N

                # Pointers to the (i,k) slice of left and the (j,k) slice of right
                left_ptrs = left_ptr + (
                    offs_i[:, None] * stride_li
                    + offs_k[None, :] * stride_lk
                    + d * stride_ld
                )
                right_ptrs = right_ptr + (
                    offs_j[:, None] * stride_ri
                    + offs_k[None, :] * stride_rk
                    + d * stride_rd
                )

                left = tl.load(
                    left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16,
                )
                right = tl.load(
                    right_ptrs,
                    mask=mask_j[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16,
                )
                # left: (M, K)   right: (N, K)  -> need right.T => (K, N)
                acc += tl.dot(left, tl.trans(right))   # (M, N)

            # Store the result for this hidden channel
            out_ptrs = out_ptr + (
                offs_i[:, None] * stride_oi
                + offs_j[None, :] * stride_oj
                + d * stride_od
            )
            tl.store(
                out_ptrs,
                acc,
                mask=mask_i[:, None] & mask_j[None, :],
            )

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8

    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid_d = (hidden_dim + BLOCK_D - 1) // BLOCK_D

    out_hidden = torch.empty(
        (bs, N, N, hidden_dim), dtype=torch.float32, device=device
    )

    # Launch one kernel per batch (bs ≤ 2)
    for b in range(bs):
        left_ptr = left_h[b]
        right_ptr = right_h[b]
        out_ptr = out_hidden[b]

        stride_li, stride_lk, stride_ld = left_h.stride()
        stride_ri, stride_rk, stride_rd = right_h.stride()
        stride_oi, stride_oj, stride_od = out_hidden.stride()

        trimul_kernel[
            (grid_i, grid_j, grid_d)
        ](
            left_ptr,
            right_ptr,
            out_ptr,
            N,
            hidden_dim,
            stride_li,
            stride_lk,
            stride_ld,
            stride_ri,
            stride_rk,
            stride_rd,
            stride_oi,
            stride_oj,
            stride_od,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm over hidden dim, gating and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_normed = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # [bs, N, N, hidden_dim]

    out_normed = out_normed * out_gate  # apply final gate

    to_out_weight = weights["to_out.weight"]
    out = F.linear(out_normed, to_out_weight)  # [bs, N, N, dim]

    return out
```
