def custom_kernel(data):

    # Extract config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    x = input_tensor

    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,

    pid_n = tl.program_id(1)
    pid_z = tl.program_id(2)

    # decode batch and hidden channel
    batch_channel = pid_z

    d = batch_channel % D

    # offsets within N dimension
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)

    mask_i = i < N
    mask_j = j < N

    # loop over K dimension

        mask_k = k < N

        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

def tri_mul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM that implements out‑going triangle multiplicative update:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    The operation is performed in FP16 with FP32 accumulation.
    """
    assert left.shape == right.shape
    B, N, _, D = left.shape
    # ensure contiguous layout
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_d = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    BLOCK = 64
    grid = ( (N + BLOCK - 1) // BLOCK,
             (N + BLOCK - 1) // BLOCK,
             B * D )

    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        D,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_d,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_d,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_d,
        BLOCK_M=BLOCK,
        BLOCK_N=BLOCK,
        BLOCK_K=BLOCK,
        NUM_WARPS=4,
    )
    return out

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriangleMultiplicativeUpdate.
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # LayerNorm over last dim
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Optional mask (multiply by 0/1)
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to half for the heavy O(N^3) reduction
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Core TriMul via Triton
    out = tri_mul_gemm(left, right)  # shape [B,N,N,hidden_dim] dtype half

    # Back to FP32 for the remaining layers
    out = out.to(torch.float32)

    # Output LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Output gate
    out = out * out_gate

    # Final linear projection to original dim
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) accelerated with Triton.

Algorithm
---------
Given X ∈ ℝ^{B×N×N×C}:
1. Layer‑norm over the last dimension C.
2. Linear projections (no bias) to hidden dimension H:
   L = X·W_left ,  R = X·W_right
3. Optional pairwise mask M ∈ {0,1}^{B×N×N}:
   L ← L·M , R ← R·M  (broadcast on H)
4. Compute element‑wise gates:
   G_L = sigmoid(X·W_left_gate) , G_R = sigmoid(X·W_right_gate)
   G_out = sigmoid(X·W_out_gate)
   L ← L·G_L , R ← R·G_R
5. Core operation (dense O(N³) part):
   out[b,i,j,h] = Σ_k L[b,i,k,h] * R[b,j,k,h]
   This is a batched matrix multiplication per hidden channel:
   out_{b,h} = L_{b,h} @ (R_{b,h})^T
   Implemented with a custom Triton kernel (FP16 compute, FP32 accumulation).
6. Layer‑norm over hidden dimension, gated by G_out and final linear
   projection back to C.

The heavy reduction in step 5 is performed on GPU with tiling (BLOCK=64)
and 64‑bit indexing; the remaining lightweight operations stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the O(N³) reduction (outgoing version)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where left/right have shape (B,N,N,H). The kernel works on a tile
    (BLOCK_M × BLOCK_N) of the output and accumulates over K in
    BLOCK_K‑sized chunks. Accumulation is in FP32, storage in FP16.
    """
    pid_m = tl.program_id(0)          # tile along i
    pid_n = tl.program_id(1)          # tile along j
    pid_z = tl.program_id(2)          # combined batch * hidden index

    # decode batch & hidden channel
    b = pid_z // H
    h = pid_z % H

    # calculate absolute i/j indices for this tile
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # pointers to left[b,i,k,h]
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + h * stride_l_h
            + i[:, None] * stride_l_i
            + k[None, :] * stride_l_k
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        left = left.to(tl.float32)

        # pointers to right[b,j,k,h] (will be transposed for dot)
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + h * stride_r_h
            + j[:, None] * stride_r_i   # j uses the same stride as i in the original tensor
            + k[None, :] * stride_r_k
        )
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        right = right.to(tl.float32)

        # (BLOCK_M × BLOCK_K) @ (BLOCK_K × BLOCK_N) -> (BLOCK_M × BLOCK_N)
        acc += tl.dot(left, tl.trans(right))

    # store result as FP16
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + h * stride_o_h
        + i[:, None] * stride_o_i
        + j[None, :] * stride_o_j
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def tri_mul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Helper that launches :func:`tri_mul_gemm_kernel`. Both inputs must have
    shape (B, N, N, H) and dtype torch.float16. Returns a tensor of the same
    shape and dtype containing the outgoing triangular update.
    """
    assert left.shape == right.shape, "left and right must share shape"
    B, N, _, H = left.shape

    # Ensure contiguous layout for simple stride calculations
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # same dtype (fp16)

    # strides are expressed in element counts
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_i, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    BLOCK = 64  # tile size; works well for N up to 1024 on A100‑class GPUs
    grid = (
        (N + BLOCK - 1) // BLOCK,   # blocks along i
        (N + BLOCK - 1) // BLOCK,   # blocks along j
        B * H,                       # combined batch & hidden dimension
    )

    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_l_b,
        s_l_i,
        s_l_k,
        s_l_h,
        s_r_b,
        s_r_i,
        s_r_k,
        s_r_h,
        s_o_b,
        s_o_i,
        s_o_j,
        s_o_h,
        BLOCK_M=BLOCK,
        BLOCK_N=BLOCK,
        BLOCK_K=BLOCK,
        num_warps=4,          # 4 warps (~128 threads) per block
    )
    return out


# ----------------------------------------------------------------------
# Public entry point matching the required signature
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).
    Arguments:
        data = (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float)
        - weights      : dict of model tensors
        - config       : dict containing "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, C]
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply optional pairwise mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy O(N³) reduction via Triton
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    out = tri_mul_gemm(left, right)               # shape [B,N,N,hidden_dim], fp16
    out = out.to(torch.float32)                    # back to fp32 for the rest

    # ------------------------------------------------------------------
    # 6. Output normalisation, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
