    a_ptr, b_ptr, c_ptr,
    stride_ab, stride_ai, stride_ak, stride_ah,
    stride_bb, stride_bi, stride_bk, stride_bh,
    stride_cb, stride_ci, stride_cj, stride_ch,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):

    num_h_blocks = tl.cdiv(H, BLOCK_H)
    batch = pid // num_h_blocks
    hb = pid % num_h_blocks
    # offsets base
    i0 = tl.program_id(1) * BLOCK_M
    j0 = tl.program_id(2) * BLOCK_N

    # Global indices
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    # Range for hidden channels:
    h_base = hb * BLOCK_H
    h = h_base + tl.arange(0, BLOCK_H)

    # Boundary masks
    i_mask = i < N
    j_mask = j < N
    h_mask = h < H

    # Initialize accumulator for each hidden channel
    # Create a 3D accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # Loop over K dimension
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k0 = k_tile * BLOCK_K
        k = k0 + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # Load left tile for each hidden channel in loop (inner)
        # We'll loop over hidden channels within block
        # Could vectorize across hidden dims using broadcast but we handle loop:
        for h_idx in range(0, BLOCK_H):
            # Compute mask for this hidden channel
            valid_h = (h_idx < H - h_base)  # can use h_mask[h_idx] boolean

            # Calculate pointers for left (A) [M,K] slice
            a_ptrs = a_ptr + batch * stride_ab \
                     + i[:, None] * stride_ai \
                     + k[None, :] * stride_ak \
                     + (h_base + h_idx) * stride_ah

            left = tl.load(a_ptrs, mask=(i_mask[:, None] & k_mask[None, :] & valid_h), other=0.0).to(tl.float32)  # shape (BLOCK_M, BLOCK_K)

            # Load right tile transposed for B => shape (K,N)
            # We need B_T such that index is (k, j)
            b_ptrs = b_ptr + batch * stride_bb \
                     + k[:, None] * stride_bi \
                     + j[None, :] * stride_bk \
                     + (h_base + h_idx) * stride_bh

            right_T = tl.load(b_ptrs, mask=(k_mask[:, None] & j_mask[None, :] & valid_h), other=0.0).to(tl.float32)  # shape (BLOCK_K, BLOCK_N)

            # Compute dot, shape (M,N)
            prod = tl.dot(left, right_T)  # returns (BLOCK_M, BLOCK_N)

            # Accumulate into acc for this h_idx
            acc[:, :, h_idx] = acc[:, :, h_idx] + prod

    # Store results to C
    # compute output pointers for each hidden slice
    for h_idx in range(0, BLOCK_H):
        c_ptrs = c_ptr + batch * stride_cb \
                 + i[:, None] * stride_ci \
                 + j[None, :] * stride_cj \
                 + (h_base + h_idx) * stride_ch

        tl.store(c_ptrs, acc[:, :, h_idx].to(c_ptr.dtype), mask=(i_mask[:, None] & j_mask[None, :] & (h_idx < H - h_base)))

    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply mask if any
    if mask is not None and mask.numel() > 0:
        mask_f = mask.unsqueeze(-1).to(x.dtype)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f
    # else no mask

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Prepare output tensor
    out_h = torch.empty_like(left_h) # shape [B,N,N,hidden_dim]

    # Determine strides
    stride_b_l, stride_i_l, stride_k_l, stride_h_l = left_h.stride()
    stride_b_r, stride_i_r, stride_k_r, stride_h_r = right_h.stride()
    stride_b_o, stride_i_o, stride_j_o, stride_h_o = out_h.stride()

    # set BLOCK sizes (tuned)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 8

    # compute grid
    num_h_blocks = (hidden_dim + BLOCK_H - 1) // BLOCK_H

"""
TriMul "outgoing" implementation.
The kernel computes, for each batch b, hidden block hb (size BLOCK_H),
and tile (i0, j0) of the output, the reduction:
out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
where left/right are the projected and gated pairwise tensors.
The reduction is performed as a batched GEMM using Triton's tiled
matrix multiplication. The rest of the module (LayerNorms,
gate sigmoid, linear projections) are done in PyTorch.
"""

import torch
import triton
import triton.language as tl

# Triton kernel
@triton.jit
def trimul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_ab, stride_ai, stride_ak, stride_ah,
    stride_bb, stride_bi, stride_bk, stride_bh,
    stride_cb, stride_ci, stride_cj, stride_ch,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    # -------------------------------
    # program ids
    pid = tl.program_id(0)               # fused batch * hidden‑block
    pid_m = tl.program_id(1)             # block row (i)
    pid_n = tl.program_id(2)             # block col (j)

    num_h_blocks = tl.cdiv(H, BLOCK_H)
    batch = pid // num_h_blocks
    h_block = pid % num_h_blocks
    h_base = h_block * BLOCK_H          # first hidden index handled by this block

    # tile start indices
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # range vectors
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)           # will be offset inside the loop
    h = h_base + tl.arange(0, BLOCK_H)

    # bounds
    i_mask = i < N
    j_mask = j < N
    k_mask = k < N          # will be recomputed each tile
    h_mask = h < H

    # accumulator: (M, N, Hblock)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # number of k‑tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k_tiles):
        k0 = k_tile * BLOCK_K
        cur_k = k0 + k
        k_mask = cur_k < N

        # loop over hidden channels inside the block
        for h_idx in range(0, BLOCK_H):
            # check hidden validity
            valid_h = (h_base + h_idx) < H

            # ---------- load A (left) ----------
            a_ptrs = a_ptr + batch * stride_ab \
                     + i[:, None] * stride_ai \
                     + cur_k[None, :] * stride_ak \
                     + (h_base + h_idx) * stride_ah
            a = tl.load(a_ptrs,
                        mask=(i_mask[:, None] & k_mask[None, :] & valid_h),
                        other=0.0).to(tl.float32)          # (M, K)

            # ---------- load B^T (right transposed) ----------
            # we need B_T[k, j] = right[j, k]
            b_ptrs = b_ptr + batch * stride_bb \
                     + cur_k[:, None] * stride_bi \
                     + j[None, :] * stride_bk \
                     + (h_base + h_idx) * stride_bh
            b = tl.load(b_ptrs,
                        mask=(k_mask[:, None] & j_mask[None, :] & valid_h),
                        other=0.0).to(tl.float32)          # (K, N)

            # dot product over k → (M, N)
            prod = tl.dot(a, b)                      # float32

            # accumulate
            acc[:, :, h_idx] = acc[:, :, h_idx] + prod

    # ---------- store result ----------
    for h_idx in range(0, BLOCK_H):
        valid_h = (h_base + h_idx) < H
        c_ptrs = c_ptr + batch * stride_cb \
                 + i[:, None] * stride_ci \
                 + j[None, :] * stride_cj \
                 + (h_base + h_idx) * stride_ch
        tl.store(c_ptrs,
                 acc[:, :, h_idx].to(c_ptr.dtype),
                 mask=(i_mask[:, None] & j_mask[None, :] & valid_h))

def custom_kernel(data):
    """
    Executes the TriMul “outgoing” update using a Triton kernel for the
    N³ reduction and PyTorch for the surrounding linear / LayerNorm work.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] (may be None)
        - weights      : dict with all Linear and LayerNorm parameters
        - config       : dict containing at least "dim", "hidden_dim",
                        and the boolean flag "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-6

    # ------------------- LayerNorm on the input -------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B,N,N,dim]

    # ------------------- Linear projections -------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------- Gates (sigmoid) -------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # ------------------- Mask handling -------------------
    if mask is not None and mask.numel() > 0 and not config.get("nomask", False):
        mask_f = mask.unsqueeze(-1).to(x.dtype)    # [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------- Apply gates -------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Cast to half for the kernel (reduces memory traffic)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Output buffer (float32 because the kernel accumulates in fp32)
    out_hidden = torch.empty(
        (batch_size, seq_len, seq_len, hidden_dim),
        dtype=torch.float32,
        device=device,
    )

"""
TriMul "outgoing" implementation.

The heavy N³ reduction (outgoing triangle multiplicative update) is
performed by a custom Triton kernel.  Given tensors

    left[b, i, k, h]  = proj_left(x)[b, i, k, h] * left_gate[x][b, i, k, h]
    right[b, j, k, h] = proj_right(x)[b, j, k, h] * right_gate[x][b, j, k, h]

the kernel computes for every batch b, tile (i, j) and hidden block h:

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

which is mathematically a batched matrix multiplication
    out_h = left_h @ right_hᵀ
for each hidden slice h.  The surrounding LayerNorms, gates and final
linear projection are done with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: tiled batched GEMM (A @ Bᵀ) over the hidden dimension.
# -------------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    a_ptr, b_ptr, c_ptr,                         # pointers
    stride_ab, stride_ai, stride_ak, stride_ah,  # strides for A (left)
    stride_bb, stride_bi, stride_bk, stride_bh,  # strides for B (right)
    stride_cb, stride_ci, stride_cj, stride_ch,  # strides for C (output)
    B, N, H,                                      # runtime sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    # -----------------------------------------------------------------
    # 1) Program ids
    # -----------------------------------------------------------------
    pid = tl.program_id(0)          # fused batch × hidden‑block axis
    pid_m = tl.program_id(1)        # i‑tile
    pid_n = tl.program_id(2)        # j‑tile

    # number of hidden blocks and current indices
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    batch = pid // num_h_blocks
    h_block = pid % num_h_blocks
    h_base = h_block * BLOCK_H                     # first hidden index of this block

    # tile start positions
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # index vectors
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)                     # will be offset inside the loop
    h = h_base + tl.arange(0, BLOCK_H)

    # bounds
    i_mask = i < N
    j_mask = j < N
    h_mask = h < H

    # accumulator for the (M, N, Hblock) tile
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # -----------------------------------------------------------------
    # 2) Loop over the reduction dimension k (runtime while‑loop)
    # -----------------------------------------------------------------
    k_offset = 0
    while k_offset < N:
        cur_k = k_offset + k                     # actual k indices for this tile
        k_mask = cur_k < N

        # ----- inner loop over hidden channels inside the block -----
        for h_idx in range(0, BLOCK_H):
            # is this hidden channel within the valid range?
            valid_h = (h_base + h_idx) < H

            # -------- load A slice: shape (M, K) ----------
            a_ptrs = a_ptr + batch * stride_ab \
                     + i[:, None] * stride_ai \
                     + cur_k[None, :] * stride_ak \
                     + (h_base + h_idx) * stride_ah
            a = tl.load(
                a_ptrs,
                mask=(i_mask[:, None] & k_mask[None, :] & valid_h),
                other=0.0).to(tl.float32)      # (M, K)

            # -------- load Bᵀ slice: shape (K, N) ----------
            # Bᵀ[k, j] = right[b, j, k]; we therefore swap the strides.
            b_ptrs = b_ptr + batch * stride_bb \
                     + cur_k[:, None] * stride_bk \
                     + j[None, :] * stride_bi \
                     + (h_base + h_idx) * stride_bh
            b = tl.load(
                b_ptrs,
                mask=(k_mask[:, None] & j_mask[None, :] & valid_h),
                other=0.0).to(tl.float32)      # (K, N)

            # -------- dot product over k ----------
            prod = tl.dot(a, b)                  # (M, N)

            # accumulate
            acc[:, :, h_idx] = acc[:, :, h_idx] + prod

        k_offset += BLOCK_K

    # -----------------------------------------------------------------
    # 3) Store the result
    # -----------------------------------------------------------------
    for h_idx in range(0, BLOCK_H):
        valid_h = (h_base + h_idx) < H
        c_ptrs = c_ptr + batch * stride_cb \
                 + i[:, None] * stride_ci \
                 + j[None, :] * stride_cj \
                 + (h_base + h_idx) * stride_ch
        tl.store(
            c_ptrs,
            acc[:, :, h_idx],
            mask=(i_mask[:, None] & j_mask[None, :] & valid_h)
        )


# -------------------------------------------------------------------------
# Host entry point: TriMul forward pass (no gradients)
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C]
        - mask         : torch.Tensor [B, N, N] (may be None)
        - weights      : dict of Torch tensors containing all Linear and
                         LayerNorm parameters
        - config       : dict with keys 'dim', 'hidden_dim', 'nomask'

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-6

    # -----------------------------------------------------------------
    # LayerNorm on the input
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, C]

    # -----------------------------------------------------------------
    # Linear projections
    # -----------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # Gates (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # Optional mask (applied before gating, as in the reference)
    # -----------------------------------------------------------------
    if mask is not None and mask.numel() > 0 and not config.get("nomask", False):
        mask_f = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # -----------------------------------------------------------------
    # Apply gates
    # -----------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Cast to half for the kernel (reduces bandwidth)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # -----------------------------------------------------------------
    # Output buffer for the tri‑multiplicative reduction
    # -----------------------------------------------------------------
    out_hidden = torch.empty(
        (B, N, N, hidden_dim), dtype=torch.float32, device=device
    )

    # -----------------------------------------------------------------
    # Strides (in elements)
    # -----------------------------------------------------------------
    stride_ab, stride_ai, stride_ak, stride_ah = left_h.stride()
    stride_bb, stride_bi, stride_bk, stride_bh = right_h.stride()
    stride_cb, stride_ci, stride_cj, stride_ch = out_hidden.stride()

    # -----------------------------------------------------------------
    # Triton launch configuration
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 8

    num_h_blocks = (hidden_dim + BLOCK_H - 1) // BLOCK_H
    grid = (
        B * num_h_blocks,                       # fused batch × hidden‑block axis
        (N + BLOCK_M - 1) // BLOCK_M,           # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,           # j‑tiles
    )

    trimul_kernel[grid](
        left_h,                     # a_ptr
        right_h,                    # b_ptr
        out_hidden,                 # c_ptr
        stride_ab, stride_ai, stride_ak, stride_ah,
        stride_bb, stride_bi, stride_bk, stride_bh,
        stride_cb, stride_ci, stride_cj, stride_ch,
        B, N, hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # to_out_norm
    # -----------------------------------------------------------------
    out_normed = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # Apply output gate and final linear projection
    # -----------------------------------------------------------------
    out_normed = out_normed * out_gate
    result = torch.nn.functional.linear(
        out_normed, weights["to_out.weight"]
    )   # [B, N, N, C]

    return result
