@triton.jit
def matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_aba, stride_a_row, stride_a_col,
    stride_bba, stride_b_row, stride_b_col,
    stride_cca, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # get batch id and tile indices
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute pointer base
    a_ptrs = A_ptr + batch_id * stride_aba + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
    b_ptrs = B_ptr + batch_id * stride_bba + offs_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k iterations
    num_k = tl.cdiv(K, BLOCK_K)
    for k_iter in range(0, num_k):
        # load
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)
        # Cast to float32
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        # dot product
        acc += tl.dot(a, b)

        # increment pointers
        a_ptrs += BLOCK_K * stride_a_col
        b_ptrs += BLOCK_K * stride_b_row

    # Write back results
    c_ptrs = C_ptr + batch_id * stride_cca + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    # Apply mask for valid C entries
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections → left, right (shape [B,N,N,H]).
3. Optional mask multiplication.
4. Compute gating sigmoids and apply them to left/right.
5. Compute the pairwise reduction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply of size (N×N) for each (b,d).
   The reduction is performed by a Triton kernel that treats (b,d) as a
   single batch dimension and carries out a standard GEMM in FP16 (accumulated
   in FP32).
6. Reshape the result back to [B,N,N,H] and apply a second LayerNorm.
7. Multiply by the output gate and a final linear projection back to the
   original feature dimension.

Only the O(N³·H) reduction step is inside Triton; all other operations use
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix‑multiply (FP16 → FP32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dimensions (runtime)
    stride_aba, stride_a_row, stride_a_col,   # a: (batch, M, K) strides
    stride_bba, stride_b_row, stride_b_col,   # b: (batch, K, N) strides
    stride_cca, stride_c_row, stride_c_col,   # c: (batch, M, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, K, BLOCK_K):
        k_range = k + tl.arange(0, BLOCK_K)

        # ---- load A tile (M x K) ----
        a_ptrs = (
            a_ptr
            + batch_id * stride_aba
            + offs_m[:, None] * stride_a_row
            + k_range[None, :] * stride_a_col
        )
        a_mask = (offs_m[:, None] < M) & (k_range[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)

        # ---- load B tile (K x N) ----
        b_ptrs = (
            b_ptr
            + batch_id * stride_bba
            + k_range[:, None] * stride_b_row
            + offs_n[None, :] * stride_b_col
        )
        b_mask = (k_range[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        # ---- accumulate ----
        acc += tl.dot(a, b)

    # ---- store result (convert back to FP16) ----
    c_ptrs = (
        c_ptr
        + batch_id * stride_cca
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (unused when config['nomask']==True)
        - weights      : dict of torch.Tensor with all linear and layer‑norm parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask", ...

    Returns
    -------
    torch.Tensor  [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])        # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask is [B,N,N]; broadcast to hidden dim
        m = mask.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) and application
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton reduction: out = left @ right^T  (batched GEMM)
    # ------------------------------------------------------------------
    # Cast to FP16 for the kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # (B, N, N, H) -> (B, H, N, N)   make contiguous for easy view
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()       # [B,H,N,N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()     # [B,H,N,N]
    # transpose the last two axes of right for the matmul
    right_perm_T = right_perm.permute(0, 1, 3, 2).contiguous()  # [B,H,N,N] (K↔N swapped)

    # flatten batch and hidden dim → single batch dimension for the kernel
    B_h = B * hidden_dim
    left_flat = left_perm.view(B_h, N, N)          # (batch, M, K)
    right_flat = right_perm_T.view(B_h, N, N)      # (batch, K, N)

    # output placeholder
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # strides (in elements)
    stride_a_batch = left_flat.stride(0)
    stride_a_row   = left_flat.stride(1)  # stride for M dimension
    stride_a_col   = left_flat.stride(2)  # stride for K dimension

    stride_b_batch = right_flat.stride(0)
    stride_b_row   = right_flat.stride(1)  # stride for K dimension (row)
    stride_b_col   = right_flat.stride(2)  # stride for N dimension (col)

    stride_c_batch = out_flat.stride(0)
    stride_c_row   = out_flat.stride(1)
    stride_c_col   = out_flat.stride(2)

    # kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        B_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,               # M
        N,               # N (output cols)
        N,               # K (shared dim)
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden_dim)
    out_perm = out_flat.view(B, hidden_dim, N, N)          # (B,H,N,N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()        # (B,N,N,H)
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Second LayerNorm, gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
