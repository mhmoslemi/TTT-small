@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_a_batch,
    stride_bk, stride_bn, stride_b_batch,
    stride_cm, stride_cn, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    FP16: tl.constexpr,
):
    pid = tl.program_id(0)
    batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_band = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = pid_band // tl.cdiv(N, BLOCK_N)
    pid_n = pid_band % tl.cdiv(N, BLOCK_N)
    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Load a (M, K)
        a_ptrs = a_ptr + batch * stride_a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=offs_m[:, None] < M and offs_k[None, :] < K, other=0.0)
        # Load b (N, K) but we need B^T: we want b[k, n] i.e., from original B (N,K) we need k row index. So we load b with reversed strides:
        b_ptrs = b_ptr + batch * stride_b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=offs_k[:, None] < K and offs_n[None, :] < N, other=0.0)
        # dot
        acc += tl.dot(a, b)
    # Store C
    c_ptrs = c_ptr + batch * stride_c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16) if FP16 else acc)
```

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_abatch,
    stride_bk, stride_bn, stride_bbatch,
    stride_cm, stride_cn, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_blocks_m * num_blocks_n)
    pid_tile = pid % (num_blocks_m * num_blocks_n)
    pid_m = pid_tile // num_blocks_n
    pid_n = pid_tile % num_blocks_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    a_ptrs = a_ptr + batch * stride_abatch + offs_m[:, None] * stride_am + 0 * stride_ak

    b_ptrs_base = b_ptr + batch * stride_bbatch
    # we will compute per k loop later

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Load a tile
        a_tile_ptrs = a_ptr + batch * stride_abatch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    # device
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape  # dim we get later

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Linear projections (float32)
    x_flat = x.view(-1, dim)  # (B*N*N, dim)
    left = torch.nn.functional.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    # Mask handling
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, weights["left_gate.weight"])).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, weights["right_gate.weight"])).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, weights["out_gate.weight"])).view(B, N, N, hidden_dim)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Rearrange for Triton kernel: (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden_dim
    # Flatten batch and hidden dimensions
    left_flat = left.view(BH, N, N)
    right_flat = right.view(BH, N, N)

    # Cast to fp16 for kernel
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    # Triton kernel launch
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Strides
    stride_am = left_fp16.stride(1)
    stride_ak = left_fp16.stride(2)
    stride_abatch = left_fp16.stride(0)
    stride_bk = right_fp16.stride(2)   # original k stride
    stride_bn = right_fp16.stride(1)   # original n stride
    stride_bbatch = right_fp16.stride(0)
    stride_cm = out_fp16.stride(1)
    stride_cn = out_fp16.stride(2)
    stride_cbatch = out_fp16.stride(0)

    grid = (BH * ((N + BLOCK_M - 1) // BLOCK_M) * ((N + BLOCK_N - 1) // BLOCK_N),)

    @triton.jit
    def tri_mul_kernel(a_ptr, b_ptr, c_ptr,
                       M, N, K,
                       stride_am, stride_ak, stride_abatch,
                       stride_bk, stride_bn, stride_bbatch,
                       stride_cm, stride_cn, stride_cbatch,
                       BLOCK_M: tl.constexpr,
                       BLOCK_N: tl.constexpr,
                       BLOCK_K: tl.constexpr):
        pid = tl.program_id(0)
        num_blocks_m = tl.cdiv(M, BLOCK_M)
        num_blocks_n = tl.cdiv(N, BLOCK_N)
        batch = pid // (num_blocks_m * num_blocks_n)
        pid_tile = pid % (num_blocks_m * num_blocks_n)
        pid_m = pid_tile // num_blocks_n
        pid_n = pid_tile % num_blocks_n

        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        # mask for out-of-bound
        mask_m = offs_m < M
        mask_n = offs_n < N

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        for k in range(0, tl.cdiv(K, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < K

            # Load A tile (M x K)
            a_ptrs = a_ptr + batch * stride_abatch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
            a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)

            # Load B tile transposed (K x N)
            b_ptrs = b_ptr + batch * stride_bbatch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
            b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0, dtype=tl.float16)

            acc += tl.dot(a, b)

        # Store C tile
        c_ptrs = c_ptr + batch * stride_cbatch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
        tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    # launch
    tri_mul_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        N, N, N,
        stride_am, stride_ak, stride_abatch,
        stride_bk, stride_bn, stride_bbatch,
        stride_cm, stride_cn, stride_cbatch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    torch.cuda.synchronize()

    # Cast back to float32
    out = out_fp16.to(torch.float32)

    # Reshape back to (B, N, N, hidden_dim)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output processing
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor.
2. Four linear projections (left, right, left_gate, right_gate, out_gate) are
   performed on the normalized tensor (no biases).  Gates are passed through a
   sigmoid.
3. Mask (if present) is applied to the left and right tensors, then
   element‑wise gated (left *= left_gate, right *= right_gate).
4. The core O(N³) operation:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is computed as a batched matrix multiplication where each hidden dimension
   `d` is treated as an independent batch.  The multiplication is
   performed in a custom Triton kernel using tiling over M, N and K (all = N).
   Computation uses FP16 inputs with FP32 accumulation for speed.
5. The result (shape [B,N,N,hidden_dim]) is layer‑normed, multiplied
   element‑wise with `out_gate`, and finally projected back to the original
   dimension with `to_out`.
6. The function returns the tensor of shape [B, N, N, dim].

The Triton kernel implements a standard blocked GEMM with the right operand
transposed on‑the‑fly (i.e. it computes A @ Bᵀ).  This matches the
`einsum('... i k d, ... j k d -> ... i j d')` used in the reference
implementation.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_abatch,
    stride_bk, stride_bn, stride_bbatch,
    stride_cm, stride_cn, stride_cbatch,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM with B transposed on‑the‑fly."""
    pid = tl.program_id(0)

    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch = pid // (num_blocks_m * num_blocks_n)
    tile = pid % (num_blocks_m * num_blocks_n)
    pid_m = tile // num_blocks_n
    pid_n = tile % num_blocks_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = a_ptr + batch * stride_abatch + \
                 offs_m[:, None] * stride_am + \
                 offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0, dtype=tl.float16)

        # Load B tile transposed (K x N) -> original B is (N, K)
        b_ptrs = b_ptr + batch * stride_bbatch + \
                 offs_k[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0, dtype=tl.float16)

        acc += tl.dot(a, b)

    # Store C tile
    c_ptrs = c_ptr + batch * stride_cbatch + \
             offs_m[:, None] * stride_cm + \
             offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass for the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float)
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    import torch

    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = x.shape

    # ---------- LayerNorm ----------
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ---------- Linear projections ----------
    # Flatten the first three dims to use torch.nn.functional.linear
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)

    left = torch.nn.functional.linear(
        x_flat, weights["left_proj.weight"]
    ).view(B, N, N, hidden_dim)

    right = torch.nn.functional.linear(
        x_flat, weights["right_proj.weight"]
    ).view(B, N, N, hidden_dim)

    # ---------- Optional mask ----------
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask
        right = right * mask

    # ---------- Gates ----------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, weights["left_gate.weight"])
    ).view(B, N, N, hidden_dim)

    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, weights["right_gate.weight"])
    ).view(B, N, N, hidden_dim)

    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, weights["out_gate.weight"])
    ).view(B, N, N, hidden_dim)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare for Triton GEMM ----------
    # Shape to (B, hidden_dim, N, N) then fuse B and hidden_dim
    left = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)
    right_flat = right.view(BH, N, N)

    # Cast to FP16 for the kernel (memory‑friendly & fast)
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    # ---------- Triton launch parameters ----------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    stride_am = left_fp16.stride(1)
    stride_ak = left_fp16.stride(2)
    stride_abatch = left_fp16.stride(0)

    stride_bk = right_fp16.stride(2)   # original k‑stride (inner)
    stride_bn = right_fp16.stride(1)   # original n‑stride (outer)
    stride_bbatch = right_fp16.stride(0)

    stride_cm = out_fp16.stride(1)
    stride_cn = out_fp16.stride(2)
    stride_cbatch = out_fp16.stride(0)

    grid = (BH *
            ((N + BLOCK_M - 1) // BLOCK_M) *
            ((N + BLOCK_N - 1) // BLOCK_N),)

    # ---------- Run Triton kernel ----------
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N, N, N,
        stride_am, stride_ak, stride_abatch,
        stride_bk, stride_bn, stride_bbatch,
        stride_cm, stride_cn, stride_cbatch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # Cast back to FP32 and reshape to original layout
    out = out_fp16.to(torch.float32)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # ---------- Post‑processing ----------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"]
    )   # (B,N,N,dim)

    return out
