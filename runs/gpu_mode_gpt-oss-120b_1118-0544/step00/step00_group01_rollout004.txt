@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    BH, N,
    stride_left_bh, stride_left_i, stride_left_k,
    stride_right_bh, stride_right_j, stride_right_k,
    stride_out_bh, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    total_blocks = BH * num_blocks_m * num_blocks_n
    # compute bh idx
    bh_idx = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    block_m = block_idx // num_blocks_n
    block_n = block_idx % num_blocks_n

    # compute start offsets
    i0 = block_m * BLOCK_M
    j0 = block_n * BLOCK_N

    # create offset ranges
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out-of-bound
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers for left tile (BH, N, N)
    left_ptrs = left_ptr + bh_idx * stride_left_bh + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
    # pointers for right tile (BH, N, N) but we need (k, j) shape
    right_ptrs = right_ptr + bh_idx * stride_right_bh + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_offset in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_offset * BLOCK_K
        # compute actual k offsets
        cur_k = k + offs_k
        # bound check for k (optional)
        k_mask = cur_k < N

        # load left & right
        left = tl.load(left_ptrs, mask=mask_i[:, None] & k_mask[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=k_mask[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left, right)

        # increment pointers
        left_ptrs += BLOCK_K * stride_left_k
        right_ptrs += BLOCK_K * stride_right_k

    # store result
    out_ptrs = out_ptr + bh_idx * stride_out_bh + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Extract weights as tensors (they're already torch tensors on device)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on dim
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B,N,N,dim]

    # Linear projections
    # Using torch.nn.functional.linear: expects input shape [..., dim], weight shape [out_features, dim]
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # Apply mask if present
    if mask is not None:
        # mask shape [B,N,N] possibly bool/float
        mask_expand = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left_proj * mask_expand * left_gate
        right = right_proj * mask_expand * right_gate
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Prepare for matmul
    B, N, _, _ = left.shape

    # reshape to (B*hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N).contiguous()
    right_flat = right_perm.view(BH, N, N).contiguous()

    # Cast to half for kernel
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # Output buffer
    out_fp16 = torch.empty_like(left_fp16)

    # Launch kernel
    grid = lambda meta: ( (BH * ((N + meta['BLOCK_M'] - 1)//meta['BLOCK_M']) * ((N + meta['BLOCK_N'] - 1)//meta['BLOCK_N'])), )
    tri_mul_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        BH, N,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2),
        out_fp16.stride(0), out_fp16.stride(1), out_fp16.stride(2),
        BLOCK_M=128, BLOCK_N=128, BLOCK_K=32,
    )

    # reshape back
    out = out_fp16.to(torch.float32).view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    # Apply out gate
    out_gated = out_norm * out_gate

    # Linear to original dim
    out_final = torch.nn.functional.linear(out_gated, to_out_weight)

    return out_final
```

"""
TriMul forward pass (outgoing) with a fused Triton kernel for the
pairwise sum over the sequence dimension (the expensive O(N³) part).

The kernel computes, for each batch b and hidden dimension d,
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication of shape (B*hidden, N, N).
The rest of the operations (LayerNorm, linear projections, gating) are
performed with PyTorch for simplicity.
"""

from __future__ import annotations
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched matmul (left * right^T) over the k dimension.
# Input shapes:
#   left  : (BH, N, N)  where BH = B * hidden_dim, layout (bh, i, k)
#   right : (BH, N, N)  layout (bh, k, j)  (i.e. transposed on the fly)
# Output:
#   out   : (BH, N, N)  layout (bh, i, j)
# --------------------------------------------------------------------------- #
@triton.jit
def _tri_mul_matmul_kernel(
    left_ptr,               # * half
    right_ptr,              # * half
    out_ptr,                # * half
    BH, N,                  # scalar parameters
    stride_left_bh, stride_left_i, stride_left_k,
    stride_right_bh, stride_right_k, stride_right_j,
    stride_out_bh, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # Number of blocks per dimension
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    # Decode the program id into batch*hidden, block_m and block_n
    bh_idx = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    block_m = block_idx // num_blocks_n
    block_n = block_idx % num_blocks_n

    # Offsets within the current block
    i_offset = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds rows / cols
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_tiles):
        cur_k = kk * BLOCK_K + k_offset
        mask_k = cur_k < N

        # left tile: (BLOCK_M, BLOCK_K)  -> (i, k)
        left_ptrs = (
            left_ptr
            + bh_idx * stride_left_bh
            + i_offset[:, None] * stride_left_i
            + cur_k[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=tl.float16(0.0),
        )

        # right tile: (BLOCK_K, BLOCK_N)  -> (k, j)
        right_ptrs = (
            right_ptr
            + bh_idx * stride_right_bh
            + cur_k[:, None] * stride_right_k
            + j_offset[None, :] * stride_right_j
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=tl.float16(0.0),
        )

        # accumulate
        acc += tl.dot(left, right)   # dot: (M,K) @ (K,N) -> (M,N)

    # Store the result (cast back to fp16)
    out_ptrs = (
        out_ptr
        + bh_idx * stride_out_bh
        + i_offset[:, None] * stride_out_i
        + j_offset[None, :] * stride_out_j
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# --------------------------------------------------------------------------- #
# Public API
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module using a Triton kernel.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be all‑ones if nomask)
            - weights: dict with the following keys (tensors on the same device):
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict containing at least:
                * "dim" (int), "hidden_dim" (int)
                * optional "nomask" (bool)
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------- #
    # Unpack arguments
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # Load weight tensors
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    eps = 1e-5

    # ------------------------------------------------------------------- #
    # LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------- #
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------- #
    # Linear projections
    # ------------------------------------------------------------------- #
    left_proj = F.linear(x_norm, left_proj_weight)   # [B, N, N, hidden_dim]
    right_proj = F.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------- #
    # Gating (sigmoid)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # ------------------------------------------------------------------- #
    # Apply mask (if present) and gates
    # ------------------------------------------------------------------- #
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # mask: [B, N, N] -> broadcast over hidden_dim
        mask_exp = mask.unsqueeze(-1).to(dtype)  # [B, N, N, 1]
        left = left_proj * mask_exp * left_gate
        right = right_proj * mask_exp * right_gate

    # ------------------------------------------------------------------- #
    # Prepare tensors for the Triton batched matmul
    # ------------------------------------------------------------------- #
    B, N, _, _ = left.shape
    # reshape to (B, hidden_dim, N, N) and then merge batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    # view as (BH, N, N)
    left_flat = left_perm.view(BH, N, N).contiguous()
    right_flat = right_perm.view(BH, N, N).contiguous()

    # Cast to half for the kernel (input may be float32)
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # Allocate output buffer (half)
    out_fp16 = torch.empty_like(left_fp16)

    # ------------------------------------------------------------------- #
    # Launch Triton kernel
    # ------------------------------------------------------------------- #
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = lambda meta: (
        BH
        * ((N + meta["BLOCK_M"] - 1) // meta["BLOCK_M"])
        * ((N + meta["BLOCK_N"] - 1) // meta["BLOCK_N"]),
    )

    _tri_mul_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        BH,
        N,
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        right_fp16.stride(0),
        right_fp16.stride(1),  # k dimension stride
        right_fp16.stride(2),  # j dimension stride
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to float32 and reshape to original layout
    out = (
        out_fp16.to(torch.float32)
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------- #
    # Post‑matmul LayerNorm, gating, and final linear projection
    # ------------------------------------------------------------------- #
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # [B, N, N, hidden_dim]

    out_gated = out_norm * out_gate  # element‑wise gate

    # Final linear back to original dim
    out_final = F.linear(out_gated, to_out_weight)  # [B, N, N, dim]

    return out_final
